#RequireContext CManiaApp

declare Text G_Title;
declare CUILayer G_Layer;

Text Render() {
	return """<?xml version="1.0" encoding="utf-8"?>
<manialink version="3" name="PageScript">
  <frame pos="0 0">
    <label id="LabelHello" pos="0 0" size="100 10" text="{{{G_Title}}}" halign="center" />
  </frame>
  <script><![CDATA[
#RequireContext CMlScriptIngame

declare CMlLabel G_LabelHello;

G_LabelHello = (Page.GetFirstChild("LabelHello") as CMlLabel);

G_LabelHello.Value = "Ready";
while (True) {
	yield;
}
]]></script>
</manialink>""";
}

main() {
	G_Title = "Razor Manialink";

	G_Layer = UILayerCreate();
	G_Layer.ManialinkPage = Render();
	while (True) {
		yield;
	}
}
