#RequireContext CTmMode

#Include "TextLib" as TL
#Include "MyLib.Script.txt" as MyLib

#Const C_MaxPlayers 16

#Setting S_RoundTime 180 as _("Round time (s)")
#Setting S_AllowRespawn True as _("Allow respawn")

declare Integer G_Score;
declare Text G_State;
declare Integer OldScore;

Integer Private_ComputeBonus(Integer _BasePoints) {
	if ((_BasePoints > 0 && _BasePoints < 50)) {
		return _BasePoints * 2;
	}
	if (_BasePoints >= 50) {
		return _BasePoints + 10;
	}
	return 0;
}

main() {
	G_State = "Idle";

	MyLib::Nice();
	log("Mode starting...");
	G_State = "Running";
	for (I, 0, C_MaxPlayers - 1) {
		log("Slot " ^ I ^ " reserved");
		log("""Mega string
{{{I}}} reserved""");
	}
	while (True) {
		yield;
		if (G_Score >= 100) {
			G_State = "Finished";
			log("Round over");
		}
		if (G_Score != OldScore) {
			log("Score changed from " ^ OldScore ^ " to " ^ G_Score);
			declare Integer Bonus = Private_ComputeBonus(G_Score);
			log("Bonus points: " ^ Bonus);
			OldScore = G_Score;
		}
	}
}
