/** `yield` the menu script during initialization
 * We don't want to execute the `CMGame_Yield()` function
 * before the complete initialization of the menu
 */
Void Private_InitializationYield() {
	+++CMGame_BeforeInitializationYield+++
	if (CMGame_G_State.CurrentNow > 0 && Now > CMGame_G_State.CurrentNow) {
		log("WARNING: [MenuBase] A frame was skipped with `yield`, `sleep()` or `wait()` ("^CMGame_G_State.CurrentNow^" > "^Now^"). This is considered bad practice. Always use `Private_InitializationYield()` to advance to the next frame.");
	}
	yield;
	CMGame_G_State.CurrentNow = Now;
	+++CMGame_AfterInitializationYield+++
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** `yield` the menu script
 * It should be the only place that directly calls the `yield` instruction
 * Everywhere else this `CMGame_Yield()` function must be called instead
 */
Void CMGame_Yield() {
	+++CMGame_BeforeYield+++
	if (CMGame_G_State.CurrentNow > 0 && Now > CMGame_G_State.CurrentNow) {
		log("WARNING: [MenuBase] A frame was skipped with `yield`, `sleep()` or `wait()` ("^CMGame_G_State.CurrentNow^" > "^Now^"). This is considered bad practice. Always use `CMGame_Yield()` to advance to the next frame.");
	}
	yield;
	CMGame_G_State.CurrentNow = Now;
	+++CMGame_AfterYield+++
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Sleep for a given duration
Void CMGame_Sleep(Integer _Duration) {
	declare Integer SleepEndTime = Now + _Duration;
	while (Now < SleepEndTime) CMGame_Yield();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Main
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
main() {
	CMGame_Synchro::SetTitleMenuInitializationStatus(System, CMGame_Synchro::C_Status_Initializing);

	CMGame_G_State = C_CMGame_State_Null;
	CMGame_G_State.CurrentNow = Now;

	LayersDefaultManialinkVersion = 3;

	// First wait for the main user
	while (UserMgr == Null || UserMgr.MainUser == Null) Private_InitializationYield();
	// Then wait for the persistent variables to be ready to read and write
	while (!UserMgr.MainUser.PersistentIsReady) Private_InitializationYield();

	---CMGame_ErrorInit---
	---CMGame_ConfigInit---
	---CMGame_BuildInfoInit---
	---CMGame_AudioInit---
	---CMGame_StylesheetInit---

	+++CMGame_LibrariesLoad+++

	+++CMGame_MenuStart+++

	CMGame_Synchro::SetTitleMenuInitializationStatus(System, CMGame_Synchro::C_Status_Initialized);

	while(True) {
		CMGame_Yield();
		+++CMGame_MenuLoop+++
	}

	+++CMGame_MenuEnd+++

	+++CMGame_LibrariesUnload+++

	// yield one last time to allow any remaining log to be sent
	CMGame_Yield();
}
