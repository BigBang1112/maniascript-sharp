				Scores::UpdatePlayerBestLapIfBetter(Event.Player);

***Match_EndRound***
***
if (!StateMgr::HasForcedState(StateMgr::C_State_Playing)) {
	//Wait for the end of the record transmission
	while (GhostUpload::IsUploading()) MB_Yield();

	// Update variables
	Map_Map = UpdateMapNetShare(Map_Campaign.Id);

	// Retrieve the latest ghost
	declare CGhost Ghost = Ghost_RetrieveFromPlayer(Players[0]);
	if (Ghost != Null) {
		G_Ghost_LastPlatformSession = Platform::ReleaseGhost(G_Ghost_LastPlatformSession);
		G_Ghost_LastPlatformSession = Platform::K_GhostInfo { Ghost = Ghost };
	}

	// Start replay outro
	declare Integer LastRaceReplayStartTime = Now;
	declare Integer GhostRestartTime = -1;
	declare Ident GhostToFollowInstanceId = NullId;
	declare CGhost GhostToFollow = Ghost_RetrieveFromPlayer(Players[0]);
	Ghosts_SetStartTime(LastRaceReplayStartTime);
	if (GhostToFollow != Null) {
		GhostRestartTime = Now + GhostToFollow.Result.Time;
		GhostToFollowInstanceId = GhostMgr.Ghost_Add(GhostToFollow, False);
	}
	if (GhostToFollowInstanceId != NullId) {
		UIManager.UIAll.Spectator_SetForcedTarget_Ghost(GhostToFollowInstanceId);
	}
	declare CUIConfig::EUISequence PrevUISequence = UIManager.UIAll.UISequence;
	UIManager.UIAll.UISequence = CUIConfig::EUISequence::EndRound;
	UIManager.UIAll.ForceSpectator = True;
	UIManager.UIAll.SpectatorForceCameraType = 0;

	// Process end race menu events
	StateMgr::ForcePlayersStates([StateMgr::C_State_EndRaceMenu]);
	UIModules_EndRaceMenu::SetEarlyMedalDisplay(CurrentMedal < Map_Map.Medal);
	UIModules_EndRaceMenu::SetVisibilityOfNewMedal(CurrentMedal < Map_Map.Medal);
	declare Boolean IsOutroComplete = IsOutroComplete();
	while (MB_MapIsRunning() && StateMgr::HasForcedState(StateMgr::C_State_EndRaceMenu)) {
		ProcessEndRaceMenuEvent();
		MB_Yield();

		// Loop ghost for replay
		if (
			(IsOutroComplete && GhostRestartTime >= 0 && Now >= GhostRestartTime) ||
			(!IsOutroComplete && IsOutroComplete())
		) {
			IsOutroComplete = True;
			if (GhostToFollow != Null) {
				LastRaceReplayStartTime = Now;
				Ghosts_SetStartTime(LastRaceReplayStartTime);
				GhostRestartTime = LastRaceReplayStartTime + GhostToFollow.Result.Time;
			} else {
				GhostRestartTime = -1;
			}
		}
	}

	// Stop replay outro
	Ghosts_SetStartTime(-1);
	UIManager.UIAll.UISequence = PrevUISequence;
	UIManager.UIAll.ForceSpectator = False;
	UIManager.UIAll.SpectatorForceCameraType = -1;
	UIManager.UIAll.Spectator_SetForcedTarget_Clear();
	if (GhostToFollowInstanceId != NullId) {
		GhostMgr.Ghost_Remove(GhostToFollowInstanceId);
		GhostToFollowInstanceId = NullId;
	}
	if (GhostToFollow != Null && DataFileMgr.Ghosts.existskey(GhostToFollow.Id)) {
		DataFileMgr.Ghost_Release(GhostToFollow.Id);
	}
	GhostToFollow = Null;
}

/// Set the map in the NetShare component
CampaignStruct::LibCampaignStruct_K_Map UpdateMapNetShare(Integer _CampaignId) {
	if (Map == Null || Map.MapInfo == Null) return CampaignStruct::LibCampaignStruct_K_Map {};
	declare CampaignStruct::LibCampaignStruct_K_Map NetShareMap;
	declare Integer RespawnCount = -1; //< ScoreMgr will update this variable
	declare Integer Points = -1; //< ScoreMgr will update this variable
	declare Integer Score = -1; //< ScoreMgr will update this variable
	declare Integer Medal;

	// Update medal and full score from ScoreMgr
	Medal = ScoreMgr.Map_GetMedal(
		MainUser::GetMainUserId(),
		Map.MapInfo.MapUid,
		MenuConst::C_ScopeType_PersonalBest,
		"",
		MenuConst::C_GameMode_Platform,
		""
	);
	ScoreMgr.Map_GetRecordFullScore(
		MainUser::GetMainUserId(),
		Map.MapInfo.MapUid,
		MenuConst::C_ScopeType_PersonalBest,
		"",
		MenuConst::C_GameMode_Platform,
		"",
		Score,
		Points,
		RespawnCount
	);

	// Update the NetShare map
	if (_CampaignId != CampaignStruct::C_Campaign_NullId) {
		NetShareMap = CampaignStore::SetMapMedalAndScore(_CampaignId, Map.MapInfo.MapUid, Medal, Score, Points, RespawnCount);
	} else { //< When playing a Platform track that is not in a campaign (e.g. using the 'Play a track' button in the 'Local' menu)
		NetShareMap = CampaignStruct::GetMapFromMapInfo(Map.MapInfo);
		NetShareMap.RespawnCount = RespawnCount;
		NetShareMap.Points = Points;
		NetShareMap.Score = Score;
		NetShareMap.Medal = Medal;
	}
	NetShare::SetMap(NetShareMap);

	return NetShareMap;
}

// Get the colored "Personnal Best" text
Text GetPBName() {
	//L16N [Record] Name displayed above the ghost of the player's best score
	return "$"^C_GhostLabel_PB^TL::GetTranslatedText(_("Personal best"));
}

/// Fetch the best ghost of the race for the player
K_Private_Ghost UpdatePBGhost_Yield(K_Private_Ghost _PreviousGhost) {
	declare K_Private_Ghost Ghost = _PreviousGhost;

	if (Ghost.GhostInstanceId != NullId) {
		GhostMgr.Ghost_Remove(_PreviousGhost.GhostInstanceId);
	}
	if (Ghost.GhostId != NullId && DataFileMgr.Ghosts.existskey(Ghost.GhostId)) {
		DataFileMgr.Ghost_Release(Ghost.GhostId);
	}
	Ghost.GhostInstanceId = NullId;
	Ghost.GhostId = NullId;

	declare Task::K_Task Task_GetRecordGhost = Task::Create(ScoreMgr, ScoreMgr.Map_GetRecordGhost_v2(
		MainUser::GetMainUserId(),
		Map.MapInfo.MapUid,
		MenuConst::C_ScopeType_PersonalBest,
		"",
		MenuConst::C_GameMode_Platform,
		""
	));

	while (Task::IsInitialized(Task_GetRecordGhost)) {
		Task_GetRecordGhost = Task::Update(Task_GetRecordGhost);
		if (!Task::IsRunning(Task_GetRecordGhost)) {
			declare CTaskResult_Ghost SourceTask = Task::GetSourceTask_Ghost(Task_GetRecordGhost);
			if (Task::IsSuccess(Task_GetRecordGhost) && SourceTask != Null && SourceTask.Ghost != Null) {
				//L16N [Record] Best score done by the player. PB stands for Personnal Best. The translation must be 3 or less letters because it's displayed on the back of the car (as the trigram).
				SourceTask.Ghost.Trigram = _("|Personal best|PB");
				SourceTask.Ghost.Nickname = GetPBName();
				Ghost.GhostInstanceId = GhostMgr.Ghost_Add(SourceTask.Ghost, True);
				Ghost.GhostId = SourceTask.Ghost.Id;
			}
			Task_GetRecordGhost = Task::Destroy(Task_GetRecordGhost);
		}
		MB_Yield();
	}

	return Ghost;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if all players outro are finished
Boolean IsOutroComplete() {
	foreach (Player in AllPlayers) {
		if (Race::IsWatchingOutro(Player)) {
			return False;
		}
	}
	return True;
}

/// Check if an event from the StartRaceMenu UI has been received and apply it if needed
Void ProcessStartRaceMenuEvent() {
	switch (UIModules_StartRaceMenu::GetLastEventType()) {
		case Const::C_RaceMenuEvent_MapRecordsUpdated: {
			declare CampaignStruct::LibCampaignStruct_K_Map Map_Map for This;
			declare CampaignStruct::LibCampaignStruct_K_Campaign Map_Campaign for This;
			Map_Map = UpdateMapNetShare(Map_Campaign.Id);
		}
		case Const::C_RaceMenuEvent_Exit: MB_StopServer();
		case Const::C_StartRaceMenuEvent_StartRace: StateMgr::ForcePlayersStates([StateMgr::C_State_Playing]);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if an event from the PauseMenu UI has been received and apply it if needed
Void ProcessPauseMenuEvent() {
	switch (UIModules_PauseMenu::GetLastEventType()) {
		case Const::C_RaceMenuEvent_NextMap: MB_StopMap();
		case Const::C_RaceMenuEvent_ChangeOpponents: {
			StateMgr::ForcePlayersStates([StateMgr::C_State_Playing]);
			MB_StopRound();
		}
	}
}

/// Check if an event from the EndRaceMenu UI has been received and apply it if needed
Void ProcessEndRaceMenuEvent() {
	switch (UIModules_EndRaceMenu::GetLastEventType()) {
		case Const::C_EndRaceMenuEvent_Improve: StateMgr::ForcePlayersStates([StateMgr::C_State_Playing]);
		case Const::C_RaceMenuEvent_NextMap: MB_StopMap();
		case Const::C_RaceMenuEvent_Exit: MB_StopServer();
		case Const::C_RaceMenuEvent_MapRecordsUpdated: {
			declare CampaignStruct::LibCampaignStruct_K_Map Map_Map for This;
			declare CampaignStruct::LibCampaignStruct_K_Campaign Map_Campaign for This;
			Map_Map = UpdateMapNetShare(Map_Campaign.Id);
		}
		case Const::C_EndRaceMenuEvent_SaveReplay: {
			G_Task_SaveReplay = Platform::SaveReplay(G_Ghost_LastPlatformSession.Ghost.Id, G_Task_SaveReplay);
			while (Task::IsInitialized(G_Task_SaveReplay)) {
				G_Task_SaveReplay = Task::Update(G_Task_SaveReplay);
				if (!Task::IsRunning(G_Task_SaveReplay)) {
					if (Task::IsSuccess(G_Task_SaveReplay)) UIModules_EndRaceMenu::SetReplaySaved();
					G_Task_SaveReplay = Task::Destroy(G_Task_SaveReplay);
				}
				MB_Yield();
			}
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if an event from the Records UI has been received and apply it if needed
Void ProcessRecordsEvent() {
	foreach (Event in UIManager.PendingEvents) {
		if (Event.Type == CUIConfigEvent::EType::OnLayerCustomEvent) {
			switch (Event.CustomEventType) {
				case UIModules_Record::C_Event_UpdatePBGhostVisibility: {
					if (UIModules_Record::PBGhostIsVisible(Event)) {
						G_BestGhost = UpdatePBGhost_Yield(G_BestGhost);
					} else if (G_BestGhost.GhostInstanceId != NullId) {
						GhostMgr.Ghost_Remove(G_BestGhost.GhostInstanceId);
						G_BestGhost.GhostInstanceId = NullId;
					}
				}
			}
		}
	}
}
