	foreach (Start in Map::GetStarts()) Starts[Start.Order] = Start;

***Match_StartRound***
***
// Reset the EndRaceMenu UIModule

		Race::Start(Player, Map::GetStart(), 0, -1);

}

// Process start race menu events

	StateMgr::ForcePlayersStates([StateMgr::C_State_StartRaceMenu]);
	while (MB_MapIsRunning() && StateMgr::HasForcedState(StateMgr::C_State_StartRaceMenu)) {
		ProcessStartRaceMenuEvent();
		MB_Yield();
	}
}
ModeUtils::PopAndApplyUISequence(UIManager.UIAll);

// Ensure that the volume balance enters Playing state
UIModules_StartRaceMenu::SetVolumeBalanceState_Playing();

// Update play mode
MB_EnablePlayMode(StateMgr::HasForcedState(StateMgr::C_State_Playing));

// Spawn players for the race
StartTime = Now + Race::C_SpawnDuration;
RoyalTimeAttack::ResetSegmentTimes(Scores[0]);
RoyalTimeAttack::Start(Map_Starts, Players[0], StartTime, False);
***

***Match_PlayLoop***
***
// Process pause menu events
ProcessPauseMenuEvent();

// Manage race events
foreach (Event in Race::GetPendingEvents()) {
	if (Event.Type != Events::C_Type_SkipOutro) Race::ValidEvent(Event);
	switch (Event.Type) {
		case Events::C_Type_SkipOutro: Race::InvalidEvent(Event); //< Prevent the players from pressing respawn/give up to cut the finish outro and respawn faster
		case Events::C_Type_GiveUp: {
			RoyalTimeAttack::ResetSegmentTimes(Scores[0]);
			StartTime = Now + Race::C_SpawnDuration;
		}
		case Events::C_Type_Waypoint: {
			if (Event.IsEndRace && Event.Player != Null) {
				Race::Stop(Event.Player, C_DelayBeforeStartingNextSegment, -1);
				RoyalTimeAttack::CompleteSegment(Event.Player, Event.RaceTime, StartTime + RoyalTimeAttack::GetSegmentTimes(Event.Player.Score).get(-1, 0), StartTime, Map_Starts.count);
				if (RoyalTimeAttack::GetCompletedSegmentsNb(Event.Player.Score) >= Map_Starts.count) {
					declare PlayerData::K_Times Times = PlayerData::K_Times {
						SegmentRaceTimes = RoyalTimeAttack::GetSegmentRaceTimes(Event.Player.Score),
						Total = RoyalTimeAttack::GetSegmentTimes(Event.Player.Score)[-1]
					};
					StateMgr::ForcePlayersStates([StateMgr::C_State_Waiting]);
					PlayerData::SendPlayerTimesEvent(Event.Player, Times);
					if (Map_SessionBestRaceTime > Times.Total || UIModules_ScoresTable::GetCustomTimes().count <= 0) {
						UIModules_ScoresTable::SetCustomTimes([Event.Player.User.WebServicesUserId => Times.Total]);
						Map_SessionBestRaceTime = Times.Total;
					}
					MB_StopRound();
				}
			}
		}
	}
}

// Manage C++ events
foreach (Event in PendingEvents) {
	if (Event.HasBeenPassed || Event.HasBeenDiscarded) continue;
	Events::Invalid(Event);
}

// Spawn players
if (PlayersNbDead > 0 && MB_RoundIsRunning()) {
	foreach (Player in Players) {
		if (Player.SpawnStatus == CSmPlayer::ESpawnStatus::NotSpawned && Race::IsReadyToStart(Player)) {
			RoyalTimeAttack::Start(Map_Starts, Player, StartTime + RoyalTimeAttack::GetSegmentTimes(Player.Score).get(-1, 0), StartTime, False);
		}
	}
}
***

***Match_EndRound***
***
MB_EnablePlayMode(False);
MB_Sleep(C_DelayBeforeStartingNextSegment);

// Special case for Royal achievements now that the Royal matchmaking mode is removed
// see: https://gitlab.nadeo.org/trackmania-next/tmnext/-/issues/6556
for (Score in Scores) {
	Tracking::SendPlayerMatchResult(UIManager, Score.User, 1, False);
}

// Process end race menu events
if (!StateMgr::HasForcedState(StateMgr::C_State_Playing)) {
	StateMgr::ForcePlayersStates([StateMgr::C_State_EndRaceMenu]);
	while (MB_MapIsRunning() && StateMgr::HasForcedState(StateMgr::C_State_EndRaceMenu)) {
		ProcessEndRaceMenuEvent();
		MB_Yield();
	}
}
***

***Match_EndMap***

// Ensure that we stop the match (after a vote for the next map, ...)
StateMgr::ForcePlayersStates([StateMgr::C_State_Waiting]);
Race::StopSkipOutroAll();
MB_StopMatch();
***

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
///	Check if an event from the StartRaceMenu UI has been received and apply it if needed
Void ProcessStartRaceMenuEvent() {
	switch (UIModules_StartRaceMenu::GetLastEventType()) {
		case Const::C_StartRaceMenuEvent_StartRace: StateMgr::ForcePlayersStates([StateMgr::C_State_Playing]);
		case Const::C_RaceMenuEvent_Exit: MB_StopServer();
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
///	Check if an event from the PauseMenu UI has been received and apply it if needed
Void ProcessPauseMenuEvent() {
	switch (UIModules_PauseMenu::GetLastEventType()) {
		case Const::C_RaceMenuEvent_NextMap: MB_StopMap();
		case Const::C_RaceMenuEvent_ChangeOpponents: {
			StateMgr::ForcePlayersStates([StateMgr::C_State_Playing]);
			MB_StopRound();
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if an event from the EndRaceMenu UI has been received and apply it if needed
Void ProcessEndRaceMenuEvent() {
	switch (UIModules_EndRaceMenu::GetLastEventType()) {
		case Const::C_EndRaceMenuEvent_Improve: StateMgr::ForcePlayersStates([StateMgr::C_State_Playing]);
		case Const::C_RaceMenuEvent_NextMap: MB_StopMap();
		case Const::C_RaceMenuEvent_Exit: MB_StopServer();
	}
}
