***Match_Yield***
***
declare MatchMgr::K_Match Server_Match for This = MatchMgr::C_Match_Null;
declare Boolean MatchPlayerListUpdated = False;

// Manage XmlRpc events
for (Event in XmlRpc.PendingEvents) {
	if (Event.Type == CXmlRpcEvent::EType::CallbackArray) {
		switch (Event.ParamArray1) {
			case Matchmaking::C_Method_MatchStarted: {
				Server_Match = MatchMgr::StartMatch(Server_Match);
			}
			case Matchmaking::C_Method_MatchCompleted: {
				Server_Match = MatchMgr::CompleteMatch(Server_Match);
			}
			case Matchmaking::C_Method_AddPlayer: {
				declare Matchmaking::K_Method_AddPlayer EventInfo = Matchmaking::GetEventInfo_Method_AddPlayer(Event);
				if (EventInfo.AccountId != "" && EventInfo.Clan > 0) {
					MatchPlayerListUpdated = True;
					Server_Match = MatchMgr::AddPlayer(Server_Match, EventInfo.AccountId, EventInfo.Clan);

					// Unspawn player to spawn them in the right team
					declare CSmPlayer Player = ModeUtils::GetPlayerFromAccountId(EventInfo.AccountId);
					if (Player != Null) {
						if (!MatchMgr::IsInCorrectClan(Server_Match, Player)) {
							if (Race::IsRacing(Player)) Race::StopSkipOutro(Player);
							SetPlayerClan(Player, MatchMgr::GetClan(Server_Match, Player));
						}
						if (Player.Score != Null) {
							WarmUp::CanPlay(Player.Score, MatchMgr::CanPlay(Server_Match, Player));
						}
					}
				}
			}
			case Matchmaking::C_Method_RemovePlayer: {
				declare Text AccountId = Matchmaking::GetEventInfo_Method_RemovePlayer(Event);
				if (AccountId != "") {
					MatchPlayerListUpdated = True;
					Server_Match = MatchMgr::RemovePlayer(Server_Match, AccountId);

					// Unspawn player if they are not in a team anymore
					declare CSmPlayer Player = ModeUtils::GetPlayerFromAccountId(AccountId);
					if (Player != Null && Race::IsRacing(Player)) {
						Race::StopSkipOutro(Player);
					}
				}
			}
		}
	}
}

for (Event in PendingEvents) {
	switch (Event.Type) {
		// Initialize players when they join the server
		case CSmModeEvent::EType::OnPlayerAdded: {
			StateMgr::InitializePlayer(Event.Player);
			CarRank::InitializePlayer(Event.Player);
			UIModules_LiveRanking::ForceUpdate();

			if (Event.Player != Null) {
				SetPlayerClan(Event.Player, MatchMgr::GetClan(Server_Match, Event.Player));
				MatchMgr::ApplySpectatorModeRestriction(ServerAdmin, Server_Match, Event.Player.User);

				if (Event.Player.User != Null) {
					Records::AddToQueueIfNeeded(Event.Player.User.WebServicesUserId);
				}

				if (Event.Player.Score != Null) {
					WarmUp::CanPlay(Event.Player.Score, MatchMgr::CanPlay(Server_Match, Event.Player));
				}
			}
		}
		case CSmModeEvent::EType::OnPlayerRemoved: {
			UIModules_LiveRanking::ForceUpdate();
		}
	}
}

declare Task::K_Task[] Server_UploadGhostsTasks for This = [];
if (Server_UploadGhostsTasks.count > 0) {
	declare Integer TaskKey = 0;
	while (TaskKey < Server_UploadGhostsTasks.count) {
		declare Task::K_Task UpdatedTask = Task::Update(Server_UploadGhostsTasks[TaskKey]);
		if (Task::IsRunning(UpdatedTask)) {
			Server_UploadGhostsTasks[TaskKey] = UpdatedTask;
			TaskKey += 1;
		} else {
			if (Log::CanLog()) {
				declare Task::K_SourceTaskError SourceTaskError = Task::GetSourceTaskError(UpdatedTask);
				Log::Log("""[Ranked][Bots] Ghosts upload task result > IsInitialized : {{{UpdatedTask.IsInitialized}}} | IsProcessing : {{{UpdatedTask.IsProcessing}}} | IsSuccess : {{{UpdatedTask.IsSuccess}}} | Error type : {{{SourceTaskError.Type}}} | Error code : {{{SourceTaskError.Code}}} | Error description : {{{SourceTaskError.Description}}}""");
			}
			UpdatedTask = Task::Destroy(UpdatedTask);
			Server_UploadGhostsTasks.removekey(TaskKey);
		}
	}
}

if (MatchPlayerListUpdated) {
	for (Player in AllPlayers) {
		MatchMgr::ApplySpectatorModeRestriction(ServerAdmin, Server_Match, Player.User);
	}
}

ServerAuth::Yield();
StateMgr::Yield();
TMWTMatchmakingMarkers::Yield();
Records::Yield();
Server_Match = MatchMgr::UpdateMatch(Server_Match);

UpdateHeader(C_MatchPointsLimit, S_MapPointsLimit, S_MatchInfo);

// Create bots after the match has been initialized to ensure that the matchmaking system has set the bot settings properly
Map_BotToGhostIds = [];
if (S_Bots_EnablePlaying && (S_Bots_Clan == 1 || S_Bots_Clan == 2)) {
	declare Ident[] WaitForBotsCreation;
	for (I, 1, C_BotsPerTeam) {
		declare CUser FakeUser <=> Users_CreateFake(Bot::GetUniqueName(), S_Bots_Clan);
		if (FakeUser != Null) {
			Map_BotToGhostIds[FakeUser.Id] = [];
			if (!WaitForBotsCreation.exists(FakeUser.Id)) {
				WaitForBotsCreation.add(FakeUser.Id);
			}
			Server_Match = MatchMgr::AddBot(Server_Match, FakeUser.WebServicesUserId, S_Bots_Clan);
			Log::Log("""[Ranked][Bots] Create bot {{{FakeUser.Id}}} ({{{FakeUser.Name}}}) [{{{FakeUser.WebServicesUserId}}}]""");
		}
	}

	// We have to wait for the bots to be properly added to the game
	while (MB_MapIsRunning() && WaitForBotsCreation.count > 0) {
		MB_Yield();
		for (Player in AllPlayers) {
			if (Player.User != Null && WaitForBotsCreation.exists(Player.User.Id)) {
				WaitForBotsCreation.remove(Player.User.Id);
			}
		}
	}
}

// Start task to get players PB now so we can process it during warm-up
if (S_Bots_EnablePlaying && Map != Null && Map.MapInfo != Null) {
	declare Text[] AccountIds = MatchMgr::GetPlayerAccountIds(Server_Match);
	Map_GetPlayerPBListTask = Task::Create(
		This,
		This.MapRecord_GetListByMapAndPlayerList(
			NullId,
			AccountIds,
			Map.MapInfo.MapUid,
			MenuConst::C_ScopeType_PersonalBest,
			"",
			MenuConst::C_GameMode_TimeAttack,
			""
		)
	);
	Log::Log("[Ranked][Bots] Request players PB > AccountIds: "^AccountIds);
}

// Warm up

for (Score in Scores) {
	WarmUp::CanPlay(Score, True);
}
for (Player in AllPlayers) {
	WarmUp::CanPlay(Player.Score, MatchMgr::CanPlay(Server_Match, Player));
}
UIModules_Countdown::SetVisibleFor(UIModules_Countdown::C_Target_Nobody);
MB_WarmUp(S_WarmUpNb, S_WarmUpDuration * 1000, S_WarmUpTimeout * 1000);
UIModules_Countdown::SetVisibleFor(UIModules_Countdown::C_Target_Everyone);
UIModules_ScoresTable::SetFooterInfo("");

UpdateHeader(C_MatchPointsLimit, S_MapPointsLimit, S_MatchInfo);

// Wait players PB
declare Integer PlayersPBTimeout = Now + 10000;
Map_GetPlayerPBListTask = Task::Update(Map_GetPlayerPBListTask);
Log::Log("""[Ranked][Bots] Waiting players PB : {{{Task::IsInitialized(Map_GetPlayerPBListTask) && Task::IsRunning(Map_GetPlayerPBListTask)}}}""");
while (
	MB_MapIsRunning() &&
	Task::IsInitialized(Map_GetPlayerPBListTask) &&
	Task::IsRunning(Map_GetPlayerPBListTask) &&
	PlayersPBTimeout > Now &&
	Map_BotToGhostIds.count > 0
) {
	MB_Yield();
	Map_GetPlayerPBListTask = Task::Update(Map_GetPlayerPBListTask);
}

// Read players PB
declare Integer[Text] PlayersPB;
if (Task::IsInitialized(Map_GetPlayerPBListTask) && !Task::IsRunning(Map_GetPlayerPBListTask)) {
	declare CTaskResult_MapRecordList SourceTask <=> Task::GetSourceTask_MapRecordList(Map_GetPlayerPBListTask);
	if (Task::IsSuccess(Map_GetPlayerPBListTask) && SourceTask != Null) {
		for (MapRecord in SourceTask.MapRecordList) {
			PlayersPB[MapRecord.WebServicesUserId] = MapRecord.Time;
		}
		for (AccountId in MatchMgr::GetPlayerAccountIds(Server_Match)) {
			if (!PlayersPB.existskey(AccountId)) {
				PlayersPB[AccountId] = -1;
			}
		}
		Log::Log("[Ranked][Bots] Successfully retrieved players PB : "^PlayersPB);
	} else {
		Log::Log("[Ranked][Bots] Failed to get players PB > SourceTask : "^SourceTask);
	}
}
Map_GetPlayerPBListTask = Task::Destroy(Map_GetPlayerPBListTask);

// Start downloading ghosts for the bots
declare Int2[Ident] BotTeamLevels;
declare Integer GhostsPerBot = ML::Max(0, S_Bots_GhostsPerBot);
if (Map_BotToGhostIds.count > 0 && GhostsPerBot > 0) {
	declare Int2 RaceLevelMinMax = GetRaceLevelMinMax(Map_RaceLevels);
	declare Integer[] PBTimes;
	for (AccountId => Time in PlayersPB) {
		PBTimes.add(Time);
	}

				GetRaceLevel(Map_RaceLevels, ML::NearestInteger(PBTime * S_Bots_PBMultiplier)) + S_Bots_LevelShift,
				RaceLevelMinMax.X,
				RaceLevelMinMax.Y
			);
		}

		_Player != Null &&
		_Player.User != Null &&
		!_Player.User.IsFakeUser &&
		MatchMgr::CanPlay(_Match, _Player)
	);
}

/// Compute the player round points for the latest race
Void ComputePlayersRoundPoints() {
	declare Integer I = 0;
	declare Integer[] PointsRepartition = PointsRepartition::GetPointsRepartition();
	for (Score in Race::GetPrevRaceRanking()) {
		if (Scores::GetPlayerPrevRaceTime(Score) > 0) {
			declare Integer Points = 0;
			if (PointsRepartition.count > 0) {
				if (PointsRepartition.existskey(I)) {
					Points = PointsRepartition[I];
				} else {
					Points = PointsRepartition[PointsRepartition.count - 1];
				}
			}
			Scores::SetPlayerRoundPoints(Score, Points);
			I += 1;
		} else {
			Scores::SetPlayerRoundPoints(Score, 0);
		}
	}
}

/** Get the ranking of the players in the match
 * Sort player by match points then best race time on the last track then by name
 */
CSmScore[] GetPlayersMatchRanking() {
	declare CSmScore[] SortedScores;

	declare Integer[Ident] ScoreIdsToSortByMatchPoints;
	for (Score in Scores) {
		ScoreIdsToSortByMatchPoints[Score.Id] = Scores::GetPlayerMatchPoints(Score);
	}

	declare Ident[][] ScoreIdsSortedByMatchPoints = Scores::SortIdsByPoints(ScoreIdsToSortByMatchPoints, Scores::C_Order_Descending, 0);
	for (MatchPointsScoreIds in ScoreIdsSortedByMatchPoints) {
		declare Ident[][Integer] ScoreIdsSortedByWaypointTimes;
		if (MatchPointsScoreIds.count <= 1) {
			ScoreIdsSortedByWaypointTimes = [0 => MatchPointsScoreIds];
		} else {
			declare Integer[][Ident] ScoreIdsToSortByWaypointTimes;
			for (ScoreId in MatchPointsScoreIds) {
				ScoreIdsToSortByWaypointTimes[ScoreId] = CommonUtils::ToScriptArray(Scores[ScoreId].BestRaceTimes);
			}
			ScoreIdsSortedByWaypointTimes = Scores::SortIdsByWaypointTimes(ScoreIdsToSortByWaypointTimes);
		}
		for (WaypointTimesScoreIds in ScoreIdsSortedByWaypointTimes) {
			declare Ident[][] ScoreIdsSortedByName;
			if (WaypointTimesScoreIds.count <= 1) {
				ScoreIdsSortedByName = [WaypointTimesScoreIds];
			} else {
				declare Text[Ident] ScoreIdsToSortByName;
				for (ScoreId in WaypointTimesScoreIds) {
					if (Scores[ScoreId].User == Null) {
						ScoreIdsToSortByName[ScoreId] = "";
					} else {
						ScoreIdsToSortByName[ScoreId] = Scores[ScoreId].User.Name;
					}
				}
				ScoreIdsSortedByName = Scores::SortIdsByName(ScoreIdsToSortByName, Scores::C_Order_Ascending, "");
			}
			for (NameScoreIds in ScoreIdsSortedByName) {
				for (ScoreId in NameScoreIds) {
					SortedScores.add(Scores[ScoreId]);
				}
			}
		}
	}

	return SortedScores;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Get the ranking of clans in the race
 *	The key is the rank and the value the clan
 *	So [1 => 1, 2 => 2, 3 => 2, 4 => 1] means clan 1 in first and fourth place,
 *	clan 2 in second and third place.
 */
Integer[Integer] GetClansRaceRanking() {
	declare Integer[][Ident] PlayerTimes;
	for (Score in Scores) {
		if (Scores::GetPlayerPrevRaceTime(Score) > 0) {
			PlayerTimes[Score.Id] = CommonUtils::ToScriptArray(Score.PrevRaceTimes);
		}
	}
	declare Ident[][Integer] SortedScoreIds = Scores::SortIdsByWaypointTimes(PlayerTimes);
	declare Integer[Integer] ClansRanking;
	for (ScoreIds in SortedScoreIds) {
		for (ScoreId in ScoreIds) {
			ClansRanking[ClansRanking.count + 1] = Scores[ScoreId].TeamNum;
		}
	}
	return ClansRanking;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Give points to the clans based on their rank in the race
Void GiveClansRacePoints(Integer[Integer] _ClansRaceRanking) {
	declare Integer FirstPlaceClan = _ClansRaceRanking.get(1, C_DNF);
	declare Integer SecondPlaceClan = _ClansRaceRanking.get(2, C_DNF);
	declare Integer ThirdPlaceClan = _ClansRaceRanking.get(3, C_DNF);

	if (FirstPlaceClan != C_DNF) {
		// Ace -> [A, A, B, B] - [A, A, B, DNF] - [A, A, DNF, DNF] - [A, DNF, DNF, DNF]
		if (FirstPlaceClan == SecondPlaceClan || SecondPlaceClan == C_DNF) {
			Scores::AddClanMapPoints(FirstPlaceClan, C_Points_Ace);
			UIModules_EventMessage::TriggerEventForClan(UIModules_EventMessage::C_Type_Ace, FirstPlaceClan);
		}
		// Victory [A, B, A, B] - [A, B, A, DNF] - [A, B, DNF, DNF]
		else if (FirstPlaceClan == ThirdPlaceClan || ThirdPlaceClan == C_DNF) {
			Scores::AddClanMapPoints(FirstPlaceClan, C_Points_VictoryWin);
			Scores::AddClanMapPoints(SecondPlaceClan, C_Points_VictoryLose);
			UIModules_EventMessage::TriggerEventForClan(UIModules_EventMessage::C_Type_Victory, FirstPlaceClan);
		}
		// Draw [A, B, B, A] - [A, B, B, DNF]
		else {
			Scores::AddClanMapPoints(FirstPlaceClan, C_Points_Draw);
			Scores::AddClanMapPoints(SecondPlaceClan, C_Points_Draw);
			UIModules_EventMessage::TriggerEvent(UIModules_EventMessage::C_Type_Draw);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Check if a points limit is reached without a tie between the two clans
 *
 *	@param	_PointsLimit	The number of points to win a track
 *	@param	_Clan1Points	The number of points of clan 1
 *	@param	_Clan2Points	The number of points of clan 2
 *
 *	@return		True if it is the case, false otherwise
 */
Boolean PointsLimitReached(Integer _PointsLimit, Integer _Clan1Points, Integer _Clan2Points) {
	return (_Clan1Points != _Clan2Points && (_Clan1Points >= _PointsLimit || _Clan2Points >= _PointsLimit));
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Check if we should go to the next track
 *
 *	@param	_MapPointsLimit		The number of points to win a track
 *
 *	@return		True if it is the case, false otherwise
 */
Boolean MapIsOver(Integer _MapPointsLimit) {
	return PointsLimitReached(_MapPointsLimit, Scores::GetClanMapPoints(1), Scores::GetClanMapPoints(2));
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Check if we should go to the next match
 *
 *	@param	_MatchPointsLimit		The number of points to win a match
 *
 *	@return		True if it is the case, false otherwise
 */
Boolean MatchIsOver(Integer _MatchPointsLimit) {
	return PointsLimitReached(_MatchPointsLimit, Scores::GetClanMatchPoints(1), Scores::GetClanMatchPoints(2));
}

/// Reset the ghost recording of all players
Void ResetGhostsRecording() {
	for (Player in AllPlayers) {
		if (Player.User != Null && !Player.User.IsFakeUser) {
			GhostRecorder_SetEnabled(Player, False);
			GhostRecorder_Clear(Player);
			Log::Log("""[Ranked][Bots] Reset ghost recording for {{{Player.User.WebServicesUserId}}} ({{{Player.User.Login}}})[{{{Player.Id}}}]""");
			// The ghosts from the GhostRecorder are independent
			// from the ghosts of the mode (`Ghosts` array from `CMode`).
			// The C++ takes care of releasing the GhostRecorder ghosts
			// without having to call `Ghost_Release()` manually.
			// eg: calling `GhostRecorder_Clear(Player)` will release all the ghost of the player.
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Release the ghosts generated by the mode
Void ReleaseGhosts(Ident[] _DownloadedGhostIds) {
	// Release downloaded ghosts
	for (Player in AllPlayers) {
		if (Player.User != Null && Player.User.IsFakeUser) {
			GhostDriver_Playlist_Clear(Player);
		}
	}
	for (GhostId in _DownloadedGhostIds) {
		Ghost_Release(GhostId);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the ghosts that the player will play
Void SetGhostDriverPlaylist(CSmPlayer _Player, Ident[] _GhostIds) {
	if (_Player == Null || _Player.User == Null || !_Player.User.IsFakeUser) return;

	declare Ident[] TMWTMatchmaking_GhostIds for _Player = [];
	if (TMWTMatchmaking_GhostIds.count <= 0) {
		TMWTMatchmaking_GhostIds = _GhostIds;
	}

	declare Boolean GhostFound = False;
	GhostDriver_Playlist_Clear(_Player);
	while (TMWTMatchmaking_GhostIds.count > 0 && !GhostFound) {
		declare Ident GhostId = TMWTMatchmaking_GhostIds[ML::Rand(0, TMWTMatchmaking_GhostIds.count - 1)];
		TMWTMatchmaking_GhostIds.remove(GhostId);
		if (Ghosts.existskey(GhostId)) {
			GhostFound = True;
			GhostDriver_Playlist_Add(_Player, Ghosts[GhostId]);
			Log::Log("""[Ranked][Bots] Add ghost {{{GhostId}}} to bot {{{_Player.User.Id}}} playlist""");
		}
	}
	if (!GhostFound) {
		Log::Log("""[Ranked][Bots] No ghost found for bot {{{_Player.User.Id}}} playlist""");
	}
}

/// Get the race levels for the times on the given map
Int2[Int2] GetMapRaceLevels(CMap _Map) {
	if (_Map == Null) return [];

	declare Integer UnderAuthorTime = ML::FloorInteger(ML::Max(0., _Map.TMObjective_AuthorTime * 0.9));
	declare Integer OverBronzeTime = ML::CeilingInteger(ML::Max(0., _Map.TMObjective_BronzeTime * 1.1));

	return [
		<C_NoTime, UnderAuthorTime> => <0, 0>,
		<UnderAuthorTime, _Map.TMObjective_AuthorTime> => <1, 10>,
		<_Map.TMObjective_AuthorTime, _Map.TMObjective_GoldTime> => <11, 30>,
		<_Map.TMObjective_GoldTime, _Map.TMObjective_SilverTime> => <31, 50>,
		<_Map.TMObjective_SilverTime, _Map.TMObjective_BronzeTime> => <51, 70>,
		<_Map.TMObjective_BronzeTime, OverBronzeTime> => <71, 100>,
		<OverBronzeTime, C_NoTime> => <101, 101>
	];
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the minimum and maximum race level
Int2 GetRaceLevelMinMax(Int2[Int2] _MapRaceLevels) {
	declare Boolean Init = True;
	declare Int2 LevelMinMax = <0, 0>;
	for (LevelRange in _MapRaceLevels) {
		if (Init) {
			Init = False;
			LevelMinMax = LevelRange;
		} else {
			if (LevelRange.X < LevelMinMax.X) LevelMinMax.X = LevelRange.X;
			if (LevelRange.Y > LevelMinMax.Y) LevelMinMax.Y = LevelRange.Y;
		}
	}
	return LevelMinMax;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the level of a player's last race
Integer GetRaceLevel(Int2[Int2] _MapRaceLevels, Integer _RaceTime) {
	if (_RaceTime <= 0) return C_InvalidRaceLevel;

	for (TimeRange => LevelRange in _MapRaceLevels) {
		if (
			TimeRange.X == C_NoTime &&
			TimeRange.Y != C_NoTime &&
			_RaceTime <= TimeRange.Y
		) {
			return LevelRange.X;
		} else if (
			TimeRange.X != C_NoTime &&
			TimeRange.Y != C_NoTime &&
			_RaceTime > TimeRange.X &&
			_RaceTime <= TimeRange.Y
		) {
			if (TimeRange.X == TimeRange.Y) {
				return LevelRange.X;
			} else {
				return LevelRange.Y - ML::NearestInteger(((TimeRange.Y * 1.) - _RaceTime) / (TimeRange.Y - TimeRange.X) * (LevelRange.Y - LevelRange.X));
			}
		} else if (
			TimeRange.X != C_NoTime &&
			TimeRange.Y == C_NoTime &&
			_RaceTime > TimeRange.X
		) {
			return LevelRange.X;
		}
	}

	return C_InvalidRaceLevel;
}
