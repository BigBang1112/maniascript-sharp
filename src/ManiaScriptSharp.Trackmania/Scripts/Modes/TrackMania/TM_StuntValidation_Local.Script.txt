***
StateMgr::ForcePlayersStates([StateMgr::C_State_Playing]);

// Initialize stunt session
StartTime = Now + Race::C_SpawnDuration;

Stunt::Start(Players[0], StartTime);
***

***Match_PlayLoop***
***
// Process UI events
ProcessPauseMenuEvent();

// C++ events
foreach (Event in PendingEvents) {
	if (Event.HasBeenPassed || Event.HasBeenDiscarded || !MB_RoundIsRunning()) continue;
	switch (Event.Type) {
		case CSmModeEvent::EType::OnStuntFigure: {
			Stunt::OnFigureEvent(Event.Player, Event.StuntFigure);
			CarRank::Update(CarRank::C_SortCriteria_TotalPoints);
		}
		case CSmModeEvent::EType::OnPlayerTriggersWaypoint: {
			Stunt::OnPlayerTriggersWaypoint(Event.Player, Event.Landmark);
			if (Event.Landmark.Waypoint.IsFinish) {
				EndPlayerStuntSession(Event.Player, Event.WaypointTime);
			}
			Events::Valid(Event); //< Validates that the checkpoint has been passed. This enables the checkpoints launched with the replay to work
		}
		case CSmModeEvent::EType::OnPlayerRequestRespawn: {
			Stunt::OnPlayerRequestRespawn(Event.Player, Event.GiveUp);
		}
		default: Events::Invalid(Event);
	}
}

// End the round when time limit is reached
if (EndTime > 0 && Now >= EndTime) {
	EndPlayerStuntSession(Players[0], EndTime - StartTime);
}
***

***Match_EndRound***
***
// Update variables
EndTime = -1; //< Stop the timer
Net_RaceValidation_NewScore = Players[0].Score.Points;

// Retrieve the latest ghost
declare CGhost Ghost = Ghost_RetrieveFromPlayer(Players[0]);
if (Ghost != Null) {
	G_Ghost_LastStuntSession = Stunt::ReleaseGhost(G_Ghost_LastStuntSession);
	G_Ghost_LastStuntSession = Stunt::K_GhostInfo { Ghost = Ghost };
}

// Process end race menu events
Net_StuntEndRaceMenu_IsVisible = True;
StateMgr::ForcePlayersStates([StateMgr::C_State_EndRaceMenu]);
while (MB_MapIsRunning() && StateMgr::HasForcedState(StateMgr::C_State_EndRaceMenu)) {
	ProcessEndRaceMenuEvent();
	MB_Yield();
}

// Hide end race menu
Net_StuntValidation_ReplaySaved = False;
Net_StuntEndRaceMenu_IsVisible = False;

// Save new score if better
if (Net_RaceValidation_NewScore >= 0 && (Net_RaceValidation_NewScore > MapGet::GetSharedAuthorScore() || MapGet::GetSharedAuthorScore() < 0)) {
	MapGet::ShareAuthorScore(Net_RaceValidation_NewScore);
	Net_RaceValidation_AuthorScore = MapGet::GetSharedAuthorScore();
}
***

G_Task_SaveReplay = Task::Destroy(G_Task_SaveReplay);

/// Stop and validate the current stunt session
Void EndPlayerStuntSession(CSmPlayer _Player, Integer _RaceTime) {
	Player_SetFinishTime(_Player, _RaceTime, Race::GetPlayerDefaultStart(_Player) as CSmMapLandmark); // to register the ghost
	Scores::UpdatePlayerPrevRace(_Player);
	Scores::UpdatePlayerBestRaceIfBetter(_Player);
	StateMgr::ForcePlayersStates([StateMgr::C_State_Waiting]);
	MB_StopRound();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if an event from the EndRaceMenu UI has been received and apply it if needed
Void ProcessEndRaceMenuEvent() {
	foreach (Event in UIManager.PendingEvents) {
		switch (Event.Type) {
			case CUIConfigEvent::EType::OnLayerCustomEvent: {
				switch (Event.CustomEventType) {
					case StuntConst::C_EventType_Quit: MB_StopServer();
					case StuntConst::C_EventType_Retry: StateMgr::ForcePlayersStates([StateMgr::C_State_Playing]);
					case StuntConst::C_EventType_SaveReplay: {
						G_Task_SaveReplay = Stunt::SaveReplay(G_Ghost_LastStuntSession.Ghost.Id, G_Task_SaveReplay);
						while (Task::IsInitialized(G_Task_SaveReplay)) {
							G_Task_SaveReplay = Task::Update(G_Task_SaveReplay);
							if (!Task::IsRunning(G_Task_SaveReplay)) {
								declare netwrite Boolean Net_StuntValidation_ReplaySaved for Teams[0] = False;
								Net_StuntValidation_ReplaySaved = Task::IsSuccess(G_Task_SaveReplay);
								G_Task_SaveReplay = Task::Destroy(G_Task_SaveReplay);
							}
							MB_Yield();
						}
					}
				}
			}
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if an event from the PauseMenu UI has been received and apply it if needed
Void ProcessPauseMenuEvent() {
	foreach (Event in UIManager.PendingEvents) {
		switch (Event.Type) {
			case CUIConfigEvent::EType::OnLayerCustomEvent: {
				switch (Event.CustomEventType) {
					case StuntConst::C_EventType_Retry: {
						StateMgr::ForcePlayersStates([StateMgr::C_State_Playing]);
						MB_StopRound();
					}
				}
			}
		}
	}
