/**
 * Manage the routes of the settings plugin
 */
#Const Version "1.0.0"
#Const ScriptName "Plugins/Nadeo/Trackmania/Settings/Routes.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Menus/Router@2.Script.txt" as Router
#Include "Libs/Nadeo/CMGame/Utils/ManiaView@3.Script.txt" as MV
#Include "Plugins/Nadeo/CMPlugins/Common/Widgets/Inputs.Script.txt" as Widget_Inputs
#Include "Plugins/Nadeo/Trackmania/ZIndex.Script.txt" as ZIndex
#Include "Plugins/Nadeo/Trackmania/Settings/Pages/Settings.Script.txt" as Page_Settings
#Include "Plugins/Nadeo/Trackmania/Settings/RoutePaths.Script.txt" as RoutePaths

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_FragmentGroup_Base "trackmania-settings-fragment-group-base"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Create all routes in the router
Void Initialize(CManiaApp _This) {
	// Create page fragments
	Router::CreateFragment(_This, RoutePaths::C_Fragment_PageSettings, Page_Settings::GetManialink());

	// Create widget fragments
	Router::CreateFragment(_This, RoutePaths::C_Fragment_WidgetInputs, MV::ManialinkPage(Widget_Inputs::Component(), ZIndex::C_ZIndex_SettingsWidgetInputs));

	// Create fragment groups
	declare Text[] FragmentGroupBase = [];
	Router::CreateFragmentGroup(_This, C_FragmentGroup_Base, FragmentGroupBase);

	// Create page routes
	Router::CreateRoute(_This, RoutePaths::C_Route_PageEmpty, []);
	Router::CreateRoute(_This, RoutePaths::C_Route_PageSettings, [C_FragmentGroup_Base, RoutePaths::C_Fragment_PageSettings]);
}
