/**
 * Request a visibility change of the settings
 */
#Const Version "1.0.0"
#Const ScriptName "Plugins/Nadeo/Trackmania/Settings/Visibility.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Request a visibility change of the settings
Void Request(CSystemPlatform _System, Boolean _IsVisible) {
	declare Boolean Trackmania_PluginsSettings_SetVisibilityRequest_Pending for _System = False;
	declare Boolean Trackmania_PluginsSettings_SetVisibilityRequest_IsVisible for _System = False;
	Trackmania_PluginsSettings_SetVisibilityRequest_Pending = True;
	Trackmania_PluginsSettings_SetVisibilityRequest_IsVisible = _IsVisible;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Check if someone is requesting a visibility change
Boolean Protected_IsRequesting(CSystemPlatform _System) {
	declare Boolean Trackmania_PluginsSettings_SetVisibilityRequest_Pending for _System = False;
	return Trackmania_PluginsSettings_SetVisibilityRequest_Pending;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Get the visibility request
Boolean Protected_GetRequest(CSystemPlatform _System) {
	declare Boolean Trackmania_PluginsSettings_SetVisibilityRequest_Pending for _System = False;
	declare Boolean Trackmania_PluginsSettings_SetVisibilityRequest_IsVisible for _System = False;
	Trackmania_PluginsSettings_SetVisibilityRequest_Pending = False;
	return Trackmania_PluginsSettings_SetVisibilityRequest_IsVisible;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Set the active visibility of the settings
Void Protected_SetIsVisible(CSystemPlatform _System, Boolean _IsVisible) {
	declare Boolean Trackmania_PluginsSettings_IsVisible for _System = False;
	Trackmania_PluginsSettings_IsVisible = _IsVisible;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Get the active visibility of the settings
Boolean GetIsVisible(CSystemPlatform _System) {
	declare Boolean Trackmania_PluginsSettings_IsVisible for _System = False;
	return Trackmania_PluginsSettings_IsVisible;
}
