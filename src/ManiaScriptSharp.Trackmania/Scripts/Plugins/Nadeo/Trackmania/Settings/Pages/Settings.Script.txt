/**
 * Settings page
 */
#Const Version "1.0.0"
#Const ScriptName "Plugins/Nadeo/Trackmania/Settings/Pages/Settings.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Utils/ManiaView2.Script.txt" as MV
#Include "Libs/Nadeo/Trackmania/Components/Settings.Script.txt" as Settings
#Include "Plugins/Nadeo/Trackmania/ZIndex.Script.txt" as ZIndex

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Private_Name "Plugins_Page_Settings"
#Const C_Private_IsPlugin True

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the page manialink
Text GetManialink() {
	return MV::Create(
C_Private_Name, 3,
"""
<frame z-index="{{{ZIndex::C_ZIndex_SettingsPageSettings}}}">
	<frameinstance modelid="{{{Settings::C_Name}}}" z-index="2" />
</frame>
""",
"""
#Include "Libs/Nadeo/CMGame/Menus/NavigationInputs.Script.txt" as NavigationInputs
#Include "Libs/Nadeo/CMGame/Menus/Router@2.Script.txt" as Router
#Include "Plugins/Nadeo/Trackmania/Settings/Visibility.Script.txt" as Visibility
#Include "Plugins/Nadeo/Trackmania/InputPriority.Script.txt" as InputPriority

#Const C_SettingsNavigationRequestId "plugins-trackmania-settings-pages-settings"
""",
"""
***MainInit***
***
declare Boolean ResetSettingCategory = True;
***

***MainStart***
***
{{{Settings::P}}}EnableCredits(False);
{{{Settings::P}}}EnableProfile(False);
{{{Settings::P}}}EnableLatencyTest(False);
{{{Settings::P}}}EnableControllerInfo(False);
{{{Settings::P}}}SetBackgroundVisibility(True);
{{{Settings::P}}}UseLocalEvents(True);
***

***MainLoop***
***
for (Event in Router::GetPendingEvents(This)) {
	switch (Event.Type) {
		case Router::C_EventType_EnterRoute: {
			Private_EnableSettings(This, True, ResetSettingCategory);
			ResetSettingCategory = True;
		}
		case Router::C_EventType_LeaveRoute: {
			Private_EnableSettings(This, False);
		}
	}
}

for (Event in {{{Settings::P}}}PendingEvents) {
	switch (Event.Type) {
		case {{{Settings::P}}}C_EventType_Back: {
			Visibility::Request(System, False);
		}
	}
}
***

Void Private_EnableSettings(CMlScript _Context, Boolean _IsEnabled, Boolean _ResetSettingCategory) {
	NavigationInputs::EnableNavigationInputs(_Context, C_SettingsNavigationRequestId, _IsEnabled, InputPriority::C_InputPriority_SettingsPageSettings);
	{{{Settings::P}}}Enable(_IsEnabled, _ResetSettingCategory);
}
Void Private_EnableSettings(CMlScript _Context, Boolean _IsEnabled) {
	NavigationInputs::EnableNavigationInputs(_Context, C_SettingsNavigationRequestId, _IsEnabled, InputPriority::C_InputPriority_SettingsPageSettings);
	{{{Settings::P}}}Enable(_IsEnabled);
}
""",
[
	Settings::Component(C_Private_IsPlugin)
],
[]
	);
}
