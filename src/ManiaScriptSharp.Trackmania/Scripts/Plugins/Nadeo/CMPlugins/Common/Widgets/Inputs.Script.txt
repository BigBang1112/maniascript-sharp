/// Get the component template
Text Private_Template() {
	return
"""
<frame pos="160 {{{-90 + (C_Private_WidgetHeight * 0.5)}}}">
	<CMGameExpendableAsset2
		id="{{{C_Private_Id_AssetBackground}}}"
		z-index="0"
		$halign="right"
		$valign="center"
		$color="{{{Stylesheet::GetColorHex6(Stylesheet::C_Color_BG_Black)}}}"
		$styles="{{{C_Private_Style_LeftEdgeSoloRounded}}}"
	/>
	<CMGameButtonsList3
		id="{{{C_Private_Id_ButtonsListInputs}}}"
		z-index="1"
		$halign="right"
		$valign="center"
	/>
</frame>
"""
	;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the buttons displayed in the list
Void Private_SetButtons(CMlScript _This, K_Private_Instance _Instance, Text[Text] _Buttons) {
	Component_ButtonsList::SetButtons(_Instance.Controls.ButtonsList_Inputs, _Buttons);
	if (_Buttons.count > 0) {
		Component_ExpendableAsset::SetSize(
			_Instance.Controls.Asset_Background,
			<Component_ButtonsList::GetSize(_This, _Instance.Controls.ButtonsList_Inputs).X + 10., C_Private_WidgetHeight>
		);
		_Instance.Controls.Asset_Background.Visible = True;
	} else {
		_Instance.Controls.Asset_Background.Visible = False;
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize the component
Void Private_Hook_ComponentInit(CMlScript _This, CMlFrame _Frame_Inputs) {
	declare K_Private_Instance CMPlugins_CommonWidgetsInputs_Instance for _Frame_Inputs = C_Private_Instance_Null;
	CMPlugins_CommonWidgetsInputs_Instance = C_Private_Instance_Null;

	CMPlugins_CommonWidgetsInputs_Instance.Controls.Asset_Background = (_Frame_Inputs.GetFirstChild(C_Private_Id_AssetBackground) as CMlFrame);
	CMPlugins_CommonWidgetsInputs_Instance.Controls.ButtonsList_Inputs = (_Frame_Inputs.GetFirstChild(C_Private_Id_ButtonsListInputs) as CMlFrame);

	declare CManiaApp LocalParentApp <=> Context::GetParentApp(_This);
	if (LocalParentApp != Null) {
		declare K_Private_Global CMPlugins_CommonWidgetsInputs_Global for LocalParentApp = C_Private_Global_Null;
		Private_SetButtons(_This, CMPlugins_CommonWidgetsInputs_Instance, CMPlugins_CommonWidgetsInputs_Global.Buttons);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the component
Void Private_Hook_ComponentAfterYield(CMlScript _This, CMlFrame _Frame_Inputs) {
	declare CManiaApp LocalParentApp <=> Context::GetParentApp(_This);
	if (LocalParentApp != Null) {
		declare K_Private_Global CMPlugins_CommonWidgetsInputs_Global for LocalParentApp = C_Private_Global_Null;
		declare K_Private_Instance CMPlugins_CommonWidgetsInputs_Instance for _Frame_Inputs = C_Private_Instance_Null;
		if (CMPlugins_CommonWidgetsInputs_Instance.ButtonsUpdate != CMPlugins_CommonWidgetsInputs_Global.ButtonsUpdate) {
			CMPlugins_CommonWidgetsInputs_Instance.ButtonsUpdate = CMPlugins_CommonWidgetsInputs_Global.ButtonsUpdate;
			Private_SetButtons(_This, CMPlugins_CommonWidgetsInputs_Instance, CMPlugins_CommonWidgetsInputs_Global.Buttons);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize the page
Void Private_Hook_PageMainInit(CMlScript _This) {
	Component_ExpendableAsset::CreateStyle(
		_This,
		C_Private_Style_LeftEdgeSoloRounded,
		Component_ExpendableAsset::K_Properties {
			Enabled = [Component_ExpendableAsset::C_Property_Images],
			Images = [
				Component_ExpendableAsset::C_Position_Left => Component_ExpendableAsset::K_Image {
					Url = ButtonStyles::C_ImageUrl_Common_ButtonRounded_Start,
					Size = <128., 256.>,
					ColorMode = Component_ExpendableAsset::C_ColorMode_Colorize
				},
				Component_ExpendableAsset::C_Position_Middle => Component_ExpendableAsset::K_Image {
					ColorMode = Component_ExpendableAsset::C_ColorMode_Background
				},
				Component_ExpendableAsset::C_Position_Right => Component_ExpendableAsset::K_Image {
					ColorMode = Component_ExpendableAsset::C_ColorMode_Background
				}
			]
		},
		C_Private_ComputeMergedProperties
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the buttons to display in the component
Void SetButtons(CManiaApp _ManiaApp, Text[Text] _Buttons) {
	declare K_Private_Global CMPlugins_CommonWidgetsInputs_Global for _ManiaApp = C_Private_Global_Null;
	if (CMPlugins_CommonWidgetsInputs_Global.Buttons != _Buttons) {
		CMPlugins_CommonWidgetsInputs_Global.Buttons = _Buttons;
		CMPlugins_CommonWidgetsInputs_Global.ButtonsUpdate += 1;
	}
}
Void SetButtons(CMlScript _This, Text[Text] _Buttons) {
	declare CManiaApp LocalParentApp <=> Context::GetParentApp(_This);
	if (LocalParentApp != Null) SetButtons(LocalParentApp, _Buttons);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Reset the buttons displayed in the component
Void ResetButtons(CManiaApp _ManiaApp) {
	SetButtons(_ManiaApp, []);
}
Void ResetButtons(CMlScript _This) {
	declare CManiaApp LocalParentApp <=> Context::GetParentApp(_This);
	if (LocalParentApp != Null) ResetButtons(LocalParentApp);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the inputs widget component
MV::K_Component Component() {
	return MV::K_Component {
		Name = C_ComponentName,
		Path = ScriptName,
		Class = C_ComponentClass,
		Template = Private_Template(),
		Hooks = [
			MV::C_Hook_PageMainInit => """{{{C_ComponentName}}}::Private_Hook_PageMainInit""",
			MV::C_Hook_ComponentInit => """{{{C_ComponentName}}}::Private_Hook_ComponentInit""",
			MV::C_Hook_ComponentAfterYield => """{{{C_ComponentName}}}::Private_Hook_ComponentAfterYield"""
		],
		Components = [
			Component_ExpendableAsset::Component(),
			Component_ButtonsList::Component()
		]
	};
}
