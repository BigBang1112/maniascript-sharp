/** `yield` the plugin script
 * It should be the only place that directly calls the `yield` instruction
 * Everywhere else this `CMPlugins_Yield()` function must be called instead
 */
Void CMPlugins_Yield() {
	+++CMPlugins_BeforeYield+++
	if (CMPlugins_G_State.CurrentNow > 0 && Now > CMPlugins_G_State.CurrentNow) {
		log("WARNING: [PluginBase] A frame was skipped with `yield`, `sleep()` or `wait()` ("^CMPlugins_G_State.CurrentNow^" > "^Now^"). This is considered bad practice. Always use `CMPlugins_Yield()` to advance to the next frame.");
	}
	yield;
	CMPlugins_G_State.CurrentNow = Now;
	+++CMPlugins_AfterYield+++
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Sleep for a given duration
Void CMPlugins_Sleep(Integer _Duration) {
	declare Integer SleepEndTime = Now + _Duration;
	while (Now < SleepEndTime) CMPlugins_Yield();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Main
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
main() {
	CMPlugins_G_State = C_CMPlugins_State_Null;
	CMPlugins_G_State.CurrentNow = Now;

	LayersDefaultManialinkVersion = 3;

	// First wait for the main user
	while (UserMgr == Null || UserMgr.MainUser == Null) CMPlugins_Yield();
	// Then wait for the persistent variables to be ready to read and write
	while (!UserMgr.MainUser.PersistentIsReady) CMPlugins_Yield();
	+++CMPlugins_Synchronization+++

	---CMPlugins_AudioInit---

	+++CMPlugins_LibrariesLoad+++

	+++CMPlugins_PluginStart+++

	while(True) {
		CMPlugins_Yield();
		+++CMPlugins_PluginLoop+++
	}

	+++CMPlugins_PluginEnd+++

	+++CMPlugins_LibrariesUnload+++

	// yield one last time to allow any remaining log to be sent
	CMPlugins_Yield();
}
