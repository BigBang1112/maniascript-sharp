
#Include "Libs/Nadeo/System/InternalEvents/EventsStructs.Script.txt" as Structs

declare Integer[] G_LibEventsConsumer_EventsToDestroy;

Structs::SInternalEvent[Integer] PendingEvents() {
	declare Structs::SInternalEvent[Integer] LibEventsProducer_UnprocessedEvents for ManiaPlanet;
	return LibEventsProducer_UnprocessedEvents;
}

Integer GetLatestChangeCount() {
	declare Integer LibEventsProducer_UnprocessedEvents_LatestChangeCount for ManiaPlanet;
	return LibEventsProducer_UnprocessedEvents_LatestChangeCount;
}

Void PassOn(Structs::SInternalEvent _Event) {
	declare Structs::SInternalEvent[Integer] LibEventsProducer_UnprocessedEvents for ManiaPlanet;
	if (!LibEventsProducer_UnprocessedEvents.existskey(_Event.EventId)) return;
	G_LibEventsConsumer_EventsToDestroy.add(_Event.EventId);
}

Void DestroyPassedEvents() {
	declare Structs::SInternalEvent[Integer] LibEventsProducer_UnprocessedEvents for ManiaPlanet;

	while (G_LibEventsConsumer_EventsToDestroy.existskey(0)) {
		declare Boolean Removed;

		Removed = LibEventsProducer_UnprocessedEvents.removekey(G_LibEventsConsumer_EventsToDestroy[0]);
		Removed = G_LibEventsConsumer_EventsToDestroy.removekey(0);
	}
}
