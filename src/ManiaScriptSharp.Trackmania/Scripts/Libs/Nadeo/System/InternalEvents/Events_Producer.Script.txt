#Include "Libs/Nadeo/System/InternalEvents/EventsStructs.Script.txt" as Structs

Void Private_SendInternalEvent(Text _Action, Text _Param, Text[] _Params) {
	if (_Action == "") return;

	declare Structs::SInternalEvent[Integer] LibEventsProducer_UnprocessedEvents for ManiaPlanet;
	declare Integer LibEventsProducer_LastInternalEventId for ManiaPlanet;

	declare Structs::SInternalEvent InternalEvent;

	InternalEvent.EventId	= LibEventsProducer_LastInternalEventId;
	InternalEvent.Action	= _Action;
	InternalEvent.Param		= _Param;
	InternalEvent.Params	= _Params;

	LibEventsProducer_UnprocessedEvents[InternalEvent.EventId] = InternalEvent;
	LibEventsProducer_LastInternalEventId += 1;
    
	declare Integer LibEventsProducer_UnprocessedEvents_LatestChangeCount for ManiaPlanet;
    LibEventsProducer_UnprocessedEvents_LatestChangeCount += 1;
}

Void SendInternalEvent(Text _Action) {
	Private_SendInternalEvent(_Action, "", []);
}

Void SendInternalEvent(Text _Action, Text _Param) {
	Private_SendInternalEvent(_Action, _Param, []);
}

Void SendInternalEvent(Text _Action, Text[] _Params) {
	Private_SendInternalEvent(_Action, "", _Params);
}
