		Quad_PlayerAvatarImage.ImageUrl = LocalUser.CountryFlagUrl;
	else
		Quad_PlayerAvatarImage.ImageUrl = LocalUser.AvatarUrl;
}

	if (Event.Type == CMlScriptEvent::Type::MouseOver) {
		if (Event.ControlId == "Quad_AvatarChangeActionSpace") {
			Frame_AvatarEditHighlighted.Show();
		}
	}
	else if (Event.Type == CMlScriptEvent::Type::MouseOut) {
		if (Event.ControlId == "Quad_AvatarChangeActionSpace") {
			Frame_AvatarEditHighlighted.Hide();
		}
	}

	Frame_EditingDisabledWarning.Visible = LayerProfile_CantEditProfileReason != "";

	Frame_AvatarHoverImage.Visible = !Frame_EditingDisabledWarning.Visible;
}

if (PrevLocalUserName != LocalUser.Name) {
	PrevLocalUserName = LocalUser.Name;

	Label_PlayerNicknameDisplay.Value = LocalUser.Name;
}

if (PrevLocalUserManiaStars != LocalUser.FameStars) {
	PrevLocalUserManiaStars = LocalUser.FameStars;

	declare ManiaStarsCount = ML::Clamp(LocalUser.FameStars, 0, 5);
	Frame_PlayerManiaStars.Visible = ManiaStarsCount > 0;

		foreach (I => Quad in Quad_PlayerManiaStars) {
			Quad.Visible = I + 1 <= ManiaStarsCount;
			Quad.RelativePosition_V3.X = I * 4.5 - ((ManiaStarsCount - 1) / 2.) * 4.5;
		}
	}
}

if (PrevLocalUserColor != LocalUser.Color) {
	PrevLocalUserColor = LocalUser.Color;

	Quad_PlayerColorAvatarFrame	.Colorize = LocalUser.Color;
	Quad_PlayerColorAvatarGlow	.Colorize = LocalUser.Color;
	Quad_PlayerColorLine0		.BgColor = LocalUser.Color;
	Quad_PlayerColorLine1		.BgColor = LocalUser.Color;
	Label_LocalUserColor0		.TextColor = LocalUser.Color;
}

if (PrevLocalUserDescription != LocalUser.Description) {
	PrevLocalUserDescription = LocalUser.Description;

	TextEdit_PlayerDescription.Value = LocalUser.Description;
	PrevPlayerDescriptionEdit = LocalUser.Description;
}

}
if (UpdatePlayerTags) {
	UpdatePlayerTags = False;

		Frame.Visible = LocalUser.Tags_Favored_Indices.existskey(I);
		if (!Frame.Visible) continue;

		declare Order = LocalUser.Tags_Favored_Indices[I];
		declare TagId = LocalUser.Tags_Id[Order];

		declare Quad_PlayerTagType	<=> (Frame.GetFirstChild("Quad_PlayerTagType")	as CMlQuad);
		declare Quad_PlayerTagImage	<=> (Frame.GetFirstChild("Quad_PlayerTagImage")	as CMlQuad);

		Quad_PlayerTagImage.ImageUrl = "file://Tags/" ^ LocalUser.Login ^ "/" ^ TagId;

		switch (LocalUser.Tags_Type[Order]) {
			case CUser::ETagType::Nadeo		: Quad_PlayerTagType.Substyle = "TagTypeNadeo";
			case CUser::ETagType::Gold		: Quad_PlayerTagType.Substyle = "TagTypeGold";
			case CUser::ETagType::Silver	: Quad_PlayerTagType.Substyle = "TagTypeSilver";
			case CUser::ETagType::Bronze	: Quad_PlayerTagType.Substyle = "TagTypeBronze";
		}
	}
}

if (PrevColorPrimary != G_CommonBase_PageColorPrimary) {
	PrevColorPrimary = G_CommonBase_PageColorPrimary;
	PrevSettingsUpdate = -1;
}

foreach (Event in Input.PendingEvents) {
	if (!ListenToGamepadInputs()) break;

	if (Event.Type == CInputEvent::EType::PadButtonPress) {
		if (Event.Button == CInputEvent::EButton::Up || Event.Button == CInputEvent::EButton::LeftStick_Up) {
			PrevSetting(LayerProfile_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerProfile_LastSettingsUpdate = Now;
		}

		else if (Event.Button == CInputEvent::EButton::Down || Event.Button == CInputEvent::EButton::LeftStick_Down) {
			NextSetting(LayerProfile_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerProfile_LastSettingsUpdate = Now;
		}
	}
}

if (PrevSettingsUpdate != LayerProfile_LastSettingsUpdate) {
	PrevSettingsUpdate = LayerProfile_LastSettingsUpdate;

	declare SplitZone = TL::Split("|", LocalUser.ZonePath);
	declare Separator = "$<$" ^ TL::ColorToText(G_CommonBase_PageColorPrimary) ^ " ⏵$>";

	declare Boolean IsTrackmaniaStandalone = (ManiaPlanet.LoadedTitle != Null && ManiaPlanet.LoadedTitle.TitleId == "Trackmania");
	if (IsTrackmaniaStandalone)
		Quad_PlayerAvatarImage.ImageUrl = LocalUser.CountryFlagUrl;
	else
		Quad_PlayerAvatarImage.ImageUrl = LocalUser.AvatarUrl;

	Quad_PlayerAvatarImage.RefreshImages();
	Quad_PlayerCurZoneIcon.ImageUrl = LocalUser.LadderZoneFlagUrl;
	Label_PlayerCurZoneText.Value = TL::Join(Separator, SplitZone);
	FormatSettingCards(LayerProfile_SettingsList);

	declare TotalHeight = 130.;

	Frame_OptionsListSteam.Visible = ManiaPlanet.System.Platform == CSystemPlatform::ESystemPlatform::Steam;

	if (Frame_OptionsListSteam.Visible) {
		TotalHeight += 24.;
	}

	Frame_OptionsListPrivacy.RelativePosition_V3.Y = -TotalHeight;
	TotalHeight += 26.;

	if (Frame_OptionsListTitle.Visible) {
		Frame_OptionsListTitle.RelativePosition_V3.Y = -TotalHeight;
		TotalHeight += 39.;
	}

	SetScrollPageHeight(TotalHeight);
}

	""", "");
}

Void Private_CustomEvent(Text _Name, Text[] _Params) {
	LayerCustomEvent(G_ManialinkLayerProfile, _Name, _Params);
}

Void Build() {
	G_ManialinkLayerProfile = UILayerCreate();
	G_ManialinkLayerProfile.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerProfile.ManialinkPage = Private_CreateManialink();
}

CUILayer GetLayer() {
	return G_ManialinkLayerProfile;
}

Void SetVisibility(Boolean _Visible) {
	G_ManialinkLayerProfile.IsVisible = _Visible;
	Private_CustomEvent("SetPageVisibility", [TL::ToText(_Visible)]);
}

Void ShowPage(Integer _Direction) {
	Private_CustomEvent("ShowPage", [TL::ToText(_Direction)]);
}

Void HidePage(Integer _Direction) {
	Private_CustomEvent("HidePage", [TL::ToText(_Direction)]);
}

Void SetColor(Vec3 _Color) {
	Private_CustomEvent("SetColor", [_Color.X^" "^_Color.Y^" "^_Color.Z]);
}

Void UpdateSettings() {
	declare Integer LayerProfile_LastSettingsUpdate for G_ManialinkLayerProfile.LocalPage;
	LayerProfile_LastSettingsUpdate = Now;
}

Void SetSettingsList(Structs::SSettingModel[] _SettingsList) {
	declare Structs::SSettingModel[] LayerProfile_SettingsList for G_ManialinkLayerProfile.LocalPage;
	declare Integer LayerProfile_LastSettingsUpdate for G_ManialinkLayerProfile.LocalPage;

	LayerProfile_SettingsList = _SettingsList;
	LayerProfile_LastSettingsUpdate = Now;
}

Void SetCantEditProfileReason(Text _WarningMessage) {
	declare Text LayerProfile_CantEditProfileReason for G_ManialinkLayerProfile.LocalPage;
	LayerProfile_CantEditProfileReason = _WarningMessage;
}
