
#Include "MathLib" as ML
#Include "TextLib" as TL
#Include "ColorLib" as CL
#Include "Libs/Nadeo/System/Settings/Profile/ProfileAPI.Script.txt"		as ProfileAPI
#Include "Libs/Nadeo/System/Settings/Layer_SettingsTabs.Script.txt"		as LayerSettingsTabs
#Include "Libs/Nadeo/System/Settings/Profile/Layer_Profile.Script.txt"	as LayerProfile
#Include "Libs/Nadeo/System/Settings/SettingsStructs.Script.txt"		as SettingsStructs

#Const C_LibProfileController_SettingsTabId		"Profile"

declare Boolean G_LibProfileController_PendingChanges;
declare Text 	G_LibProfileController_TitleUsedForTheTab;

Void Private_UpdateProfileSettings() {
	if (ManiaPlanet.UserMgr.MainUserProfile == Null) return;

	LayerProfile::SetSettingsList(ProfileAPI::GetSettingsList());

	declare Boolean IsTrackmaniaStandalone = G_LibProfileController_TitleUsedForTheTab == "Trackmania";

	switch (ProfileAPI::CustomizationLevel()) {
		case -1 : LayerProfile::SetCantEditProfileReason(_("You cannot customize your profile when offline."));
		case 0 : LayerProfile::SetCantEditProfileReason(_("You cannot customize your profile while playing."));
		case 1 : {
			if (IsTrackmaniaStandalone)
				LayerProfile::SetCantEditProfileReason("");
			else
				LayerProfile::SetCantEditProfileReason(_("Demo accounts have limited profile customization options."));
		}
		default	: LayerProfile::SetCantEditProfileReason("");
	}
	LayerProfile::UpdateSettings();
}

Text GetTabId() {
	return C_LibProfileController_SettingsTabId;
}

Void SetVisibility(Boolean _Visible) {
	LayerProfile::SetVisibility(_Visible);
}

Void Show(Integer _Direction) {
	LayerProfile::ShowPage(_Direction);
}

Void Hide(Integer _Direction) {
	LayerProfile::HidePage(_Direction);
}

Void SetLayerColor(Vec3 _Color) {
	LayerProfile::SetColor(_Color);
}

CUILayer GetLayer() {
	return LayerProfile::GetLayer();
}

Boolean HasUnsavedChanges() {
	declare HasChanges = G_LibProfileController_PendingChanges;
	G_LibProfileController_PendingChanges = False;
	return HasChanges;
}

Void Load() {
	LayerProfile::Build();

	LayerSettingsTabs::CreateTab(C_LibProfileController_SettingsTabId, _("|Settings|Profile"), "tab-profile.dds");

	if (ManiaPlanet.LoadedTitle == Null)
		G_LibProfileController_TitleUsedForTheTab = "";
	else
		G_LibProfileController_TitleUsedForTheTab = ManiaPlanet.LoadedTitle.TitleId;
		
	Private_UpdateProfileSettings();
}

Void Loop() {
	ProfileAPI::Loop();

	declare Text CurrentlyLoadedTitleId = "";
	if (ManiaPlanet.LoadedTitle != Null)
		CurrentlyLoadedTitleId = ManiaPlanet.LoadedTitle.TitleId;
	if (G_LibProfileController_TitleUsedForTheTab != CurrentlyLoadedTitleId) {
		G_LibProfileController_TitleUsedForTheTab = CurrentlyLoadedTitleId;
		Private_UpdateProfileSettings();
	}

	foreach (Event in PendingEvents) {
		declare UserSettings <=> ManiaPlanet.UserMgr.MainUserProfile;

		if (Event.Type == CManiaAppEvent::EType::LayerCustomEvent) {
			if (Event.CustomEventLayer != LayerProfile::GetLayer()) continue;

			switch (Event.CustomEventType) {
				case "OpenPlayerPage" : {
					OpenLink("https://players.trackmania.com/", CManiaApp::ELinkType::ExternalBrowser);
				}
				case "CopyAccountId" : {
					ClipboardSet(LocalUser.WebServicesUserId);
				}
				case "ToggleBoolean" : {
					if (!Event.CustomEventData.existskey(0)) continue;

					declare Setting = ProfileAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;

					declare NewValue = !Setting.CurrentValueBoolean;

					switch (Setting.Id) {
						case "PrivacyAdsEnabled"		: UserSettings.Adverts_Enabled						= NewValue;
						case "PrivacyAdsShareData"		: UserSettings.Adverts_UsePersonnalData				= NewValue;
					}

					G_LibProfileController_PendingChanges = True;
					Private_UpdateProfileSettings();
				}
			}

			if (ProfileAPI::CustomizationLevel() <= 0) continue;

			switch (Event.CustomEventType) {
				case "EditProfileAvatar" : {
					ProfileAPI::EditAvatar();
				}
				case "EditProfileHorn" : {
					ProfileAPI::EditHorn();
				}
				case "EditProfilePassword" : {
					ProfileAPI::EditPassword();
				}
				case "EditProfileZone" : {
					declare SettingsStructs::K_PluginEvents PluginEvents for System;
					PluginEvents.Events[PluginEvents.Count] = SettingsStructs::K_PluginEvent {
						Type = SettingsStructs::C_PluginEventType_PushRoute,
						Data = "/welcomechangezone",
						SetCurrentPathAsParent = True
					};
					PluginEvents.Count += 1;
				}
				case "SetProfileDescription" : {
					if (!Event.CustomEventData.existskey(0)) continue;
					UserSettings.User_Description = Event.CustomEventData[0];
					G_LibProfileController_PendingChanges = True;
				}

				case "SetSettingValue" : {
					if (!Event.CustomEventData.existskey(1)) continue;

					declare SettingId = Event.CustomEventData[0];
					declare NewValue = Event.CustomEventData[1];

					switch (SettingId) {
						case "EditProfileNickname" : {
							if (NewValue != "") {
								UserSettings.Account_Name = TL::SubString(NewValue, 0, 75);
								G_LibProfileController_PendingChanges = True;
							}
						}
						case "EditProfileTrigram" : {
							if (NewValue != "") {
								UserSettings.User_Trigram = TL::SubString(NewValue, 0, 3);
								G_LibProfileController_PendingChanges = True;
							}
						}
						case "EditProfileColor" : {
							UserSettings.User_LightTrailHue = ML::Clamp(TL::ToReal(NewValue), 0., 1.);
							G_LibProfileController_PendingChanges = True;
						}
						case "EditProfileClublink" : {
							UserSettings.User_ClubLinkUrl = NewValue;
							G_LibProfileController_PendingChanges = True;
						}
						case "EditProfileEmail" : {
							//if (UserSettings.Account_EMail != NewValue && TL::RegexMatch("^[^@]+@[^@.].+\\.[^@.].+$", NewValue, "i").count <= 0) {
								UserSettings.Account_EMail = NewValue;
								G_LibProfileController_PendingChanges = True;
							//}
						}
						case "ParentalLockPassword" : {
							ProfileAPI::StoreParentalLockPassword(NewValue);
						}
					}

					Private_UpdateProfileSettings();
				}

				case "StopSliderDragging" : {
					Private_UpdateProfileSettings();
				}

				case "SettingEnum" : {
					if (!Event.CustomEventData.existskey(1)) continue;

					declare Setting = ProfileAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;

					declare Direction = TL::ToInteger(Event.CustomEventData[1]);
					declare Key = Setting.AvailableOptions.keyof(Setting.CurrentValueText) + Direction;

					if (!Setting.AvailableOptions.existskey(Key)) continue;
					declare NewValue = Setting.AvailableOptions[Key];

					switch (Setting.Id) {
						case "PrivacyEnableSkins" : {
							switch (NewValue) {
								case "On"	: UserSettings.Custom_PlayerModels = CUserV2Profile::ECustomPlayerModels::All;
								case "2D"	: UserSettings.Custom_PlayerModels = CUserV2Profile::ECustomPlayerModels::OnlyTextures;
								case "Off"	: UserSettings.Custom_PlayerModels = CUserV2Profile::ECustomPlayerModels::None;
							}
						}
					}

					G_LibProfileController_PendingChanges = True;
					Private_UpdateProfileSettings();
				}

				case "ToggleBoolean" : {
					if (!Event.CustomEventData.existskey(0)) continue;

					declare Setting = ProfileAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;

					declare NewValue = !Setting.CurrentValueBoolean;

					switch (Setting.Id) {
						case "ProfileAutoConnect"		: UserSettings.Account_EnableAutoConnect			= NewValue;
						case "EditProfileReceiveMails"	: UserSettings.Account_AcceptNews					= NewValue;
						case "ParentalLockTitle"		: UserSettings.Title_IsForbiddenWithParentalLock	= NewValue;
						case "ParentalLockOnce"			: ProfileAPI::ParentalLockUnlockOnce(NewValue);
						case "SteamLinksOverlay"		: UserSettings.Steam_OpenLinksInSteam				= NewValue;
						case "SteamSyncWorkshop"		: UserSettings.Steam_SynchonizeWorkshopFiles		= NewValue;
						case "PrivacyEnableChat"		: UserSettings.Custom_EnableChat					= NewValue;
						case "PrivacyEnableAvatars"		: UserSettings.Custom_EnableAvatars					= NewValue;
						case "PrivacyEnableHorns"		: UserSettings.Custom_EnableUnlimitedHorns			= NewValue;
						case "TitleSaveAllReplays"		: UserSettings.Online_AutoSaveReplay				= NewValue;
						case "TitleSeparateRounds"		: UserSettings.Online_SaveRoundReplaysSeparately	= NewValue;

						case "TitleAdvancedEditor" : {
							if (NewValue) {
								UserSettings.Editors_MapEditorMode = CUserV2Profile::EMapEditorMode::Ask;
							} else {
								UserSettings.Editors_MapEditorMode = CUserV2Profile::EMapEditorMode::Advanced;
							}
						}
					}

					G_LibProfileController_PendingChanges = True;
					Private_UpdateProfileSettings();
				}

				case "ParentalLockToggle" : {
					ProfileAPI::ToggleParentalLock();
				}

				case "SliderIncrement" : {
					if (!Event.CustomEventData.existskey(1)) continue;

					declare Setting = ProfileAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;

					declare Direction = TL::ToInteger(Event.CustomEventData[1]);
					
					switch (Setting.Id) {
						case "EditProfileColor" : {
							declare NewValue = Setting.CurrentValueReal + Setting.ValueStep * Direction;
							UserSettings.User_LightTrailHue = ML::Clamp(NewValue, Setting.ValueRangeMin, Setting.ValueRangeMax);
							G_LibProfileController_PendingChanges = True;
						}
					}

					Private_UpdateProfileSettings();
				}
			}
		}
	}

	if (ProfileAPI::PendingUpdate()) {
		Private_UpdateProfileSettings();
	}
}
