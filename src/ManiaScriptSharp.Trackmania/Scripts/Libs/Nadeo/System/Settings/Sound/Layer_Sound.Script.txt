
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Settings/SettingsStructs.Script.txt"	as Structs
#Include "Libs/Nadeo/System/Settings/Layer_CommonBase.Script.txt"	as LayerBase


#Const C_ImgBase "file://Media/Manialinks/Common/Settings/"

declare CUILayer G_ManialinkLayerSound;

Text Private_CreateManialink() {
	LayerBase::InitSettingsCards();

	return LayerBase::CreateLayerMaialink("Settings:Sound", """

<frame id="Frame_PageSoundSettings" pos="0 0">
	<frame pos="0 -7">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="General" translate="1" />
		<quad class="Separator" pos="0 2" /> 
		{{{LayerBase::AddSettingsCards(5)}}}
	</frame>
</frame>

""", "", """
SetScrollPageHeight(37.);

declare SSettingModel[] LayerSound_SettingsList for Page;
declare Integer LayerSound_LastSettingsUpdate for Page;

declare PrevSettingsUpdate = -1;

G_CommonBase_DisablePreciseValues = True;
""", """

foreach (Event in Input.PendingEvents) {
	if (!ListenToGamepadInputs()) break;

	if (Event.Type == CInputEvent::EType::PadButtonPress) {
		if (Event.Button == CInputEvent::EButton::Up || Event.Button == CInputEvent::EButton::LeftStick_Up) {
			PrevSetting(LayerSound_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerSound_LastSettingsUpdate = Now;
		}

		else if (Event.Button == CInputEvent::EButton::Down || Event.Button == CInputEvent::EButton::LeftStick_Down) {
			NextSetting(LayerSound_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerSound_LastSettingsUpdate = Now;
		}
	}
}

if (PrevSettingsUpdate != LayerSound_LastSettingsUpdate) {
	PrevSettingsUpdate = LayerSound_LastSettingsUpdate;

	foreach (I => Setting in LayerSound_SettingsList) {
		LayerSound_SettingsList[I].CurrentValueReal = ML::Clamp(Setting.CurrentValueReal, -40., 0.);
	}
	FormatSettingCards(LayerSound_SettingsList);
}

	""", "");
}

Void Private_CustomEvent(Text _Name, Text[] _Params) {
	LayerCustomEvent(G_ManialinkLayerSound, _Name, _Params);
}

Void Build() {
	G_ManialinkLayerSound = UILayerCreate();
	G_ManialinkLayerSound.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerSound.ManialinkPage = Private_CreateManialink();
}

CUILayer GetLayer() {
	return G_ManialinkLayerSound;
}

Void SetVisibility(Boolean _Visible) {
    G_ManialinkLayerSound.IsVisible = _Visible;
	Private_CustomEvent("SetPageVisibility", [TL::ToText(_Visible)]);
}

Void ShowPage(Integer _Direction) {
	Private_CustomEvent("ShowPage", [TL::ToText(_Direction)]);
}

Void HidePage(Integer _Direction) {
	Private_CustomEvent("HidePage", [TL::ToText(_Direction)]);
}

Void SetColor(Vec3 _Color) {
	Private_CustomEvent("SetColor", [_Color.X^" "^_Color.Y^" "^_Color.Z]);
}

Void SetSettingsList(Structs::SSettingModel[] _SettingsList) {

	declare Structs::SSettingModel[] LayerSound_SettingsList for G_ManialinkLayerSound.LocalPage;
	declare Integer LayerSound_LastSettingsUpdate for G_ManialinkLayerSound.LocalPage;

	LayerSound_SettingsList = _SettingsList;
	LayerSound_LastSettingsUpdate = Now;
}
