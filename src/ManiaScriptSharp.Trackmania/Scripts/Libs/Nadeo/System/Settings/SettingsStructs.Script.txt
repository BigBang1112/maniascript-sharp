/// Get the configuration object
K_Config GetConfig() {
	declare Integer CMGame_Config_ConfigDirtyCounter for System = -2;

	declare K_Config PluginSystemSettings_Config for System = K_Config {};
	declare Integer PluginSystemSettings_ConfigDirtyCounter for System = -1;

	if (PluginSystemSettings_ConfigDirtyCounter != CMGame_Config_ConfigDirtyCounter) {
		declare Text CMGame_Config_ConfigJson for System = "";
		declare Text CMGame_Config_ConfigStructVersion for System = "";

		if (
			CMGame_Config_ConfigStructVersion != "" &&
			CMGame_Config_ConfigStructVersion != C_ConfigStructVersion
		) {
			log("WARNING [Plugins][SettingsStruct] The `K_Config` structure from the plugin is not at the same version than the one from the title");
		}

		PluginSystemSettings_ConfigDirtyCounter = CMGame_Config_ConfigDirtyCounter;
		PluginSystemSettings_Config.fromjson(CMGame_Config_ConfigJson);
	}

	return PluginSystemSettings_Config;
}
