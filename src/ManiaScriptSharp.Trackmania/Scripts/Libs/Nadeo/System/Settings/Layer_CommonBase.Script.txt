Void Private_UpdateScrollBar() {
	declare Frame_PageContentsContainer <=> (Page.GetFirstChild("Frame_PageContentsContainer") as CMlFrame);

	declare ScrollBarPageHeight		= Frame_PageContentsContainer.ScrollMax.Y + C_CommonBase_ScrollBarVisibleHeight;
	declare ScrollBarVisibleHeight	= Frame_PageContentsContainer.Size.Y;
	G_CommonBase_ScrollBarEnabled	= ScrollBarPageHeight > ScrollBarVisibleHeight;

	declare Frame_PageScrollingBar <=> (Page.GetFirstChild("Frame_PageScrollingBar") as CMlFrame);
	Frame_PageScrollingBar.Visible = G_CommonBase_ScrollBarEnabled;

	if (!G_CommonBase_ScrollBarEnabled) return;

	declare ScrollBarCurOffset = ML::Clamp(Frame_PageContentsContainer.ScrollOffset.Y, 0., ScrollBarPageHeight - ScrollBarVisibleHeight);

	declare HandleSizeY = ScrollBarVisibleHeight * (ScrollBarVisibleHeight / ScrollBarPageHeight);
	declare HandlePositionY = (ScrollBarCurOffset / ScrollBarPageHeight) * ScrollBarVisibleHeight;

	declare Quad_ScrollBarHandle		<=> (Page.GetFirstChild("Quad_ScrollBarHandle")			as CMlQuad);
	declare Quad_ScrollBarDragHandle	<=> (Page.GetFirstChild("Quad_ScrollBarDragHandle")		as CMlQuad);

	Quad_ScrollBarHandle.Size.Y = HandleSizeY;
	Quad_ScrollBarDragHandle.Size.Y = HandleSizeY;
	Quad_ScrollBarHandle.RelativePosition_V3.Y = -HandlePositionY;
	Quad_ScrollBarDragHandle.RelativePosition_V3.Y = -HandlePositionY;
}

Void SetScrollPageHeight(Real _PageHeight) {
	declare Frame_PageContentsContainer <=> (Page.GetFirstChild("Frame_PageContentsContainer") as CMlFrame);
	Frame_PageContentsContainer.ScrollMax.Y = ML::Max(_PageHeight - C_CommonBase_ScrollBarVisibleHeight, 0.);

	Private_UpdateScrollBar();
}

Void AnimScrollPage(Real _ScrollDirection) {
	if (!G_CommonBase_ScrollBarEnabled) return;

	declare Frame_PageContentsContainer <=> (Page.GetFirstChild("Frame_PageContentsContainer") as CMlFrame);
	Frame_PageContentsContainer.ScrollOffset.Y = ML::Clamp(Frame_PageContentsContainer.ScrollOffset.Y + _ScrollDirection, Frame_PageContentsContainer.ScrollMin.Y, Frame_PageContentsContainer.ScrollMax.Y);
}

Void ScrollPageTo(Real _TargetOffset) {
	if (!G_CommonBase_ScrollBarEnabled) return;

	declare Frame_PageContentsContainer <=> (Page.GetFirstChild("Frame_PageContentsContainer") as CMlFrame);
	Frame_PageContentsContainer.ScrollOffset.Y = _TargetOffset;
}

Void SetPageVisibility(Boolean _Visible) {
	declare Frame_GeneralPageContainer <=> (Page.GetFirstChild("Frame_GeneralPageContainer") as CMlFrame);
	Frame_GeneralPageContainer.Visible = _Visible;
}

Void ShowPage(Integer _Direction) {
	declare Frame_PageAnimSwipe <=> (Page.GetFirstChild("Frame_PageAnimSwipe") as CMlFrame);
	AnimMgr.Flush(Frame_PageAnimSwipe);
	Frame_PageAnimSwipe.Show();

	G_CommonBase_PageIsActive = True;

	if (_Direction == 0) {
		Frame_PageAnimSwipe.RelativePosition_V3.X = 0.;
		return;
	}

	Frame_PageAnimSwipe.RelativePosition_V3.X = 320.;
	if (_Direction < 0) Frame_PageAnimSwipe.RelativePosition_V3.X *= -1.;

	AnimMgr.Add(Frame_PageAnimSwipe, "<frame pos='0 0' />", 250, CAnimManager::EAnimManagerEasing::CircOut);
}

Void HidePage(Integer _Direction) {
	declare Frame_PageAnimSwipe <=> (Page.GetFirstChild("Frame_PageAnimSwipe") as CMlFrame);
	AnimMgr.Flush(Frame_PageAnimSwipe);

	G_CommonBase_PageIsActive = False;

	if (_Direction == 0) {
		Frame_PageAnimSwipe.RelativePosition_V3.X = 320.;
		return;
	}

	Frame_PageAnimSwipe.RelativePosition_V3.X = 0.;

	declare TargetX = 320.;
	if (_Direction > 0) TargetX *= -1.;

	AnimMgr.Add(Frame_PageAnimSwipe, "<frame pos='"^TargetX^" 0' />", 250, CAnimManager::EAnimManagerEasing::CircOut);
}

Boolean ListenToGamepadInputs() {
	if (!ManiaPlanet.Dialog_IsFinished || !G_CommonBase_PageIsActive) return False;

	declare Frame_GeneralPageContainer <=> (Page.GetFirstChild("Frame_GeneralPageContainer") as CMlFrame);
	return Frame_GeneralPageContainer.Visible;
}

Text HexDigit(Integer _Digit) {
	switch (_Digit) {
		case 10 : return "A";
		case 11 : return "B";
		case 12 : return "C";
		case 13 : return "D";
		case 14 : return "E";
		case 15 : return "F";
	}
	return TL::ToText(_Digit);
}

Text DecToHex(Integer _Dec) {
	return HexDigit((_Dec / 16) % 16) ^ HexDigit(_Dec % 16);
}

Text RgbToHex(Vec3 _RGB) {
	declare Text String;
	for (I, 0, 2) String ^= DecToHex(ML::FloorInteger(_RGB[I] * 255));
	return String;
}

Void Private_PageControllers() {
	foreach (Event in PendingEvents) {
		if (Event.Type == CMlScriptEvent::Type::MouseClick) {
			if (Event.ControlId == "Quad_SettingCardClickable") {
				declare Text SettingType for Event.Control;
				declare Text SettingId for Event.Control;

				switch (SettingType) {
					case "Bool" : {
						SendCustomEvent("ToggleBoolean", [SettingId]);
					}
					case "Action" : {
						SendCustomEvent(SettingId, []);
					}
					case "Input" : {
						SendCustomEvent("EditAction", [SettingId]);
					}
				}
			}

			else if (Event.ControlId == "Quad_SettingEnumArrowPrev") {
				declare Text SettingId for Event.Control;
				SendCustomEvent("SettingEnum", [SettingId, "-1"]);
			}

			else if (Event.ControlId == "Quad_SettingEnumArrowNext") {
				declare Text SettingId for Event.Control;
				SendCustomEvent("SettingEnum", [SettingId, "1"]);
			}

			G_CommonBase_UseSettingHover = False;
		}

		else if (Event.Type == CMlScriptEvent::Type::MouseOut) {
			if (Event.ControlId == "Quad_ScrollBarDragHandle") {
				G_CommonBase_IsScrollHovered = False;
			}

			else if (Event.ControlId == "Quad_SliderActionSpace") {
				G_CommonBase_SettingSliderHoverId = "";
			} else if (Event.ControlId == "Label_SettingInfoToggle") {
				(Page.GetFirstChild("Frame_SettingInfo") as CMlFrame).Hide();
			}
		}

		else if (Event.Type == CMlScriptEvent::Type::MouseOver) {
			if (Event.ControlId == "Quad_ScrollBarDragHandle") {
				G_CommonBase_IsScrollHovered = True;
			}

			else if (Event.ControlId == "Quad_SliderActionSpace") {
				if (G_CommonBase_SettingSliderHoverId != "" || G_CommonBase_SettingSliderDraggingId != "") continue;

				declare Text SettingId for Event.Control;
				declare Text SettingType for Event.Control;
				declare Vec2 SettingLimits for Event.Control;
				declare Vec2 SettingStepScale for Event.Control;
				declare CMlFrame SliderHandle for Event.Control;
				declare Boolean SettingAlternativeDisplay for Event.Control;

				G_CommonBase_SettingSliderHoverId		= SettingId;
				G_CommonBase_SettingSliderPositionStart	= Event.Control.AbsolutePosition_V3;
				G_CommonBase_SettingSliderIsColor		= SettingType == "Color";
				G_CommonBase_SettingSliderHandle		<=> SliderHandle;
				G_CommonBase_SettingSliderLimits		= SettingLimits;
				G_CommonBase_SettingSliderStepScale		= SettingStepScale;
			} else if (Event.ControlId == "Label_SettingInfoToggle") {
				declare Frame_SettingInfo <=> (Page.GetFirstChild("Frame_SettingInfo") as CMlFrame);
				Frame_SettingInfo.RelativePosition_V3 = Event.Control.AbsolutePosition_V3;
				declare Text SettingInfo for Event.Control;
				declare CMlLabel Label_SettingInfo = (Frame_SettingInfo.GetFirstChild("Label_SettingInfo") as CMlLabel);
				Label_SettingInfo.Value = TL::GetTranslatedText(SettingInfo);
				(Frame_SettingInfo.GetFirstChild("Quad_SettingInfo") as CMlQuad).Size.Y = Label_SettingInfo.ComputeHeight(Label_SettingInfo.Value);
				
				Frame_SettingInfo.Show();
			}

			G_CommonBase_UseSettingHover = False;
		}

		else if (Event.Type == CMlScriptEvent::Type::EntrySubmit) {
			declare Text SettingId for Event.Control;
			SendCustomEvent("SetSettingValue", [SettingId, (Event.Control as CMlEntry).Value]);
		}

		else if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
			switch (Event.CustomEventType) {
				case "SetPageVisibility" : {
					if (!Event.CustomEventData.existskey(0)) continue;
					SetPageVisibility(Event.CustomEventData[0] == "True");
				}
				case "ShowPage" : {
					if (!Event.CustomEventData.existskey(0)) continue;
					ShowPage(TL::ToInteger(Event.CustomEventData[0]));
				}
				case "HidePage" : {
					if (!Event.CustomEventData.existskey(0)) continue;
					HidePage(TL::ToInteger(Event.CustomEventData[0]));
				}
				case "SetColor" : {
					if (!Event.CustomEventData.existskey(0)) continue;
					declare SplitColor = TL::Split(" ", Event.CustomEventData[0]);
					G_CommonBase_PageColorPrimary = <1., 1., 1.>;

					if (SplitColor.count != 3) continue;
					for (I, 0, 2) G_CommonBase_PageColorPrimary[I] = TL::ToReal(SplitColor[I]);

					Page.GetClassChildren("Separator", Page.MainFrame, True);

					foreach (Control in Page.GetClassChildren_Result) {
						if (Control is CMlQuad) {
							(Control as CMlQuad).BgColor = G_CommonBase_PageColorPrimary;
						}
					}
				}
			}
		}
	}

	declare Frame_PageContentsContainer <=> (Page.GetFirstChild("Frame_PageContentsContainer") as CMlFrame);
	declare PrevScrollOffsetY for Page = -1.;

	if (G_CommonBase_PrevMouseLeftButton != Input.MouseLeftButton) {
		G_CommonBase_PrevMouseLeftButton = Input.MouseLeftButton;

		if (Input.MouseLeftButton) {
			if (G_CommonBase_IsScrollHovered) {
				G_CommonBase_IsScrollDragging = True;
				G_CommonBase_ScrollDragStartPositionY = Input.MousePos.Y;
				G_CommonBase_ScrollDragStartOffset = Frame_PageContentsContainer.ScrollOffset.Y;
			}

			if (G_CommonBase_SettingSliderHoverId != "") {
				G_CommonBase_SettingSliderDraggingId = G_CommonBase_SettingSliderHoverId;

				if (G_CommonBase_SettingSliderIsColor) {
					declare Frame_ColorPreviewTooltip <=> (Page.GetFirstChild("Frame_ColorPreviewTooltip") as CMlFrame);

					AnimMgr.Flush(Frame_ColorPreviewTooltip);
					AnimMgr.Add(Frame_ColorPreviewTooltip, "<frame scale='1' />", 125, CAnimManager::EAnimManagerEasing::CircOut);
				}
			}
		}

		else {
			if (G_CommonBase_IsScrollDragging) {
				G_CommonBase_IsScrollDragging = False;
			}

			if (G_CommonBase_SettingSliderDraggingId != "") {
				declare PositionDiff = Input.MousePos.X - G_CommonBase_SettingSliderPositionStart.X - 1;
				declare Ratio = ML::Clamp(PositionDiff, 0., 60.) / 60.;
				declare Real NewValue = G_CommonBase_SettingSliderLimits.X + Ratio * (G_CommonBase_SettingSliderLimits.Y - G_CommonBase_SettingSliderLimits.X);
				NewValue = ML::ToReal(ML::NearestInteger(NewValue / G_CommonBase_SettingSliderStepScale.X)) * G_CommonBase_SettingSliderStepScale.X;
				
				PositionDiff = 60. * (NewValue - G_CommonBase_SettingSliderLimits.X) / (G_CommonBase_SettingSliderLimits.Y - G_CommonBase_SettingSliderLimits.X);
				if (G_CommonBase_SettingSliderStepScale.Y > 0)	NewValue = ML::Exp(NewValue);
				
				SendCustomEvent("SetSettingValue", [G_CommonBase_SettingSliderDraggingId, TL::ToText(NewValue)]);

				G_CommonBase_SettingSliderDraggingId = "";

				if (G_CommonBase_SettingSliderIsColor) {
					declare Frame_ColorPreviewTooltip <=> (Page.GetFirstChild("Frame_ColorPreviewTooltip") as CMlFrame);

					AnimMgr.Flush(Frame_ColorPreviewTooltip);
					AnimMgr.Add(Frame_ColorPreviewTooltip, "<frame scale='0' />", 125, CAnimManager::EAnimManagerEasing::CircIn);
				}
			}
		}
	}

	if (Input.MouseLeftButton) {
		if (G_CommonBase_IsScrollDragging) {
			declare ScrollBarPageHeight = Frame_PageContentsContainer.ScrollMax.Y + C_CommonBase_ScrollBarVisibleHeight;
			declare PositionDiff = G_CommonBase_ScrollDragStartPositionY - Input.MousePos.Y;
			declare OffsetDiff = PositionDiff * (ScrollBarPageHeight / C_CommonBase_ScrollBarVisibleHeight);

			Frame_PageContentsContainer.ScrollOffset.Y = ML::Clamp(G_CommonBase_ScrollDragStartOffset + OffsetDiff, 0., ScrollBarPageHeight - C_CommonBase_ScrollBarVisibleHeight);
			Private_UpdateScrollBar();
		}

		if (G_CommonBase_SettingSliderDraggingId != "") {
			declare PositionDiff = Input.MousePos.X - G_CommonBase_SettingSliderPositionStart.X - 1;
			declare Ratio = ML::Clamp(PositionDiff, 0., 60.) / 60.;
			declare Real NewValue = G_CommonBase_SettingSliderLimits.X + Ratio * (G_CommonBase_SettingSliderLimits.Y - G_CommonBase_SettingSliderLimits.X);
			NewValue = ML::ToReal(ML::NearestInteger(NewValue / G_CommonBase_SettingSliderStepScale.X)) * G_CommonBase_SettingSliderStepScale.X;
			
			PositionDiff = 60. * (NewValue - G_CommonBase_SettingSliderLimits.X) / (G_CommonBase_SettingSliderLimits.Y - G_CommonBase_SettingSliderLimits.X);
			
			if (G_CommonBase_SettingSliderStepScale.Y > 0)	NewValue = ML::Exp(NewValue);

			SendCustomEvent("SliderTempValue", [G_CommonBase_SettingSliderDraggingId, TL::ToText(NewValue)]);

			G_CommonBase_SettingSliderHandle.RelativePosition_V3 = <ML::Clamp(PositionDiff, 0., 60.), 0.>;

			declare Entry_PreciseSliderInput <=> (G_CommonBase_SettingSliderHandle.Parent.GetFirstChild("Entry_PreciseSliderInput") as CMlEntry);
			Entry_PreciseSliderInput.Value = TL::FormatReal(NewValue, 3, False, False);

			if (G_CommonBase_SettingSliderIsColor) {
				declare Quad_SettingSliderColor		<=> (G_CommonBase_SettingSliderHandle.GetFirstChild("Quad_SettingSliderColor") as CMlQuad);
				declare Frame_ColorPreviewTooltip	<=> (Page.GetFirstChild("Frame_ColorPreviewTooltip")	as CMlFrame);
				declare Label_ColorPreviewValue		<=> (Page.GetFirstChild("Label_ColorPreviewValue")		as CMlLabel);
				declare Label_ColorPreviewArrow		<=> (Page.GetFirstChild("Label_ColorPreviewArrow")		as CMlLabel);
				declare Quad_ColorPreviewCard		<=> (Page.GetFirstChild("Quad_ColorPreviewCard")		as CMlQuad);

				Frame_ColorPreviewTooltip.RelativePosition_V3 = G_CommonBase_SettingSliderPositionStart + <ML::Clamp(PositionDiff, 0., 60.) + 1., -4.>;

				declare RGB = CL::HsvToRgb(<Ratio, 1., 1.>);
				Label_ColorPreviewValue.Value = "#" ^ RgbToHex(RGB);
				Label_ColorPreviewArrow.TextColor = RGB;
				Quad_ColorPreviewCard.BgColor = RGB;
				Quad_SettingSliderColor.BgColor = RGB;
			}
		}
	}

	if (PrevScrollOffsetY != Frame_PageContentsContainer.ScrollOffset.Y) {
		PrevScrollOffsetY = Frame_PageContentsContainer.ScrollOffset.Y;
		Private_UpdateScrollBar();
	}

	foreach (Event in Input.PendingEvents) {
		if (!ListenToGamepadInputs()) break;

		if (Event.Type == CInputEvent::EType::PadButtonPress) {
			if (Event.Button == CInputEvent::EButton::A) {
				declare Done = False;
				if (G_CommonBase_SelectedSettingFrame != Null) {
					declare Entry_SettingValueEditable		<=> (G_CommonBase_SelectedSettingFrame.GetFirstChild("Entry_SettingValueEditable")		as CMlEntry);
					if (Entry_SettingValueEditable != Null && Entry_SettingValueEditable.Visible) {
						Entry_SettingValueEditable.Focus();
						Done = True;
					}
				}
				if (!Done) {
					SendCustomEvent("ToggleBoolean", [G_CommonBase_SelectedSettingId]);
					SendCustomEvent("EditAction", [G_CommonBase_SelectedSettingId]);
					SendCustomEvent(G_CommonBase_SelectedSettingId, []);
				}
				G_CommonBase_UseSettingHover = True;
			}

			else if (Event.Button == CInputEvent::EButton::Left || Event.Button == CInputEvent::EButton::LeftStick_Left) {
				SendCustomEvent("SettingEnum", [G_CommonBase_SelectedSettingId, "-1"]);
				SendCustomEvent("SliderIncrement", [G_CommonBase_SelectedSettingId, "-1"]);
				G_CommonBase_UseSettingHover = True;
			}

			else if (Event.Button == CInputEvent::EButton::Right || Event.Button == CInputEvent::EButton::LeftStick_Right) {
				SendCustomEvent("SettingEnum", [G_CommonBase_SelectedSettingId, "1"]);	
				SendCustomEvent("SliderIncrement", [G_CommonBase_SelectedSettingId, "1"]);	// TODO: compute the new value here and send the SetSettingValue event.
				G_CommonBase_UseSettingHover = True;
			}
		}
	}
}

Void FormatSettingCards(SSettingModel[] _Settings) {

	G_CommonBase_SelectedSettingFrame = Null;

	foreach (I => Setting in _Settings) {
		declare Frame <=> (Page.GetFirstChild("Frame_SettingCard#"^I) as CMlFrame);
		if (Frame == Null) continue;

		declare Frame_SettingModelNormal	<=> (Frame.GetFirstChild("Frame_SettingModelNormal")	as CMlFrame);

		declare Label_SettingName				<=> (Frame.GetFirstChild("Label_SettingName")				as CMlLabel);
		declare Label_SettingValueDisplayOnly	<=> (Frame.GetFirstChild("Label_SettingValueDisplayOnly")	as CMlLabel);
		declare Entry_SettingValueEditable		<=> (Frame.GetFirstChild("Entry_SettingValueEditable")		as CMlEntry);
		declare Label_SettingValueBoolean		<=> (Frame.GetFirstChild("Label_SettingValueBoolean")		as CMlLabel);
		declare Label_SettingInfoToggle			<=> (Frame.GetFirstChild("Label_SettingInfoToggle")			as CMlLabel);

		declare Frame_SettingSliderReal		<=> (Frame.GetFirstChild("Frame_SettingSliderReal")		as CMlFrame);
		declare Entry_PreciseSliderInput	<=> (Frame.GetFirstChild("Entry_PreciseSliderInput")	as CMlEntry);
		declare Label_PreciseSliderInputRO	<=> (Frame.GetFirstChild("Label_PreciseSliderInputRO")	as CMlLabel);
		declare Quad_SliderTypeNormal		<=> (Frame.GetFirstChild("Quad_SliderTypeNormal")		as CMlQuad);
		declare Quad_SliderTypeColor		<=> (Frame.GetFirstChild("Quad_SliderTypeColor")		as CMlQuad);
		declare Frame_SettingSliderHandle	<=> (Frame.GetFirstChild("Frame_SettingSliderHandle")	as CMlFrame);
		declare Quad_SettingSliderHandle	<=> (Frame.GetFirstChild("Quad_SettingSliderHandle")	as CMlQuad);
		declare Quad_SliderActionSpace		<=> (Frame.GetFirstChild("Quad_SliderActionSpace")		as CMlQuad);
		declare Frame_SettingSliderColor	<=> (Frame.GetFirstChild("Frame_SettingSliderColor")	as CMlFrame);
		declare Quad_SettingSliderColor		<=> (Frame.GetFirstChild("Quad_SettingSliderColor")		as CMlQuad);

		declare Frame_SettingEnumArrows		<=> (Frame.GetFirstChild("Frame_SettingEnumArrows")		as CMlFrame);
		declare Label_SettingEnumArrowPrev	<=> (Frame.GetFirstChild("Label_SettingEnumArrowPrev")	as CMlLabel);
		declare Label_SettingEnumArrowNext	<=> (Frame.GetFirstChild("Label_SettingEnumArrowNext")	as CMlLabel);
		declare Quad_SettingEnumArrowPrev	<=> (Frame.GetFirstChild("Quad_SettingEnumArrowPrev")	as CMlQuad);
		declare Quad_SettingEnumArrowNext	<=> (Frame.GetFirstChild("Quad_SettingEnumArrowNext")	as CMlQuad);

		declare Quad_SettingCardClickable	<=> (Frame.GetFirstChild("Quad_SettingCardClickable")	as CMlQuad);
		declare Quad_SettingCardEnabled		<=> (Frame.GetFirstChild("Quad_SettingCardEnabled")		as CMlQuad);
		declare Quad_SettingCardDisabled	<=> (Frame.GetFirstChild("Quad_SettingCardDisabled")	as CMlQuad);
		declare Quad_SettingCardSelected	<=> (Frame.GetFirstChild("Quad_SettingCardSelected")	as CMlQuad);
		declare Label_SettingNeedsResart	<=> (Frame.GetFirstChild("Label_SettingNeedsResart")	as CMlLabel);

		Frame_SettingModelNormal.Visible = !Setting.IsDevice;

		declare IsTextDisplay = Setting.Type == "Enum" || (Setting.Type == "Int") || Setting.Type == "Action" || Setting.Type == "Text" || Setting.Type == "Input";
		declare IsSliderSetting = Setting.Type == "Real" || Setting.Type == "Color" || Setting.Type == "IntSlider";
		declare IsTextReadOnly = IsTextDisplay && (Setting.Type == "Enum" || Setting.Type == "Input" || Setting.Type == "Action" || Setting.IsDisabled);
		declare IsTextEditable = IsTextDisplay && (Setting.Type == "Text" || Setting.Type == "Int") && !Setting.IsDisabled;

		Label_SettingValueBoolean.Visible = Setting.Type == "Bool";
		Label_SettingValueDisplayOnly.Visible = IsTextDisplay && IsTextReadOnly;
		Label_SettingInfoToggle.Visible = Setting.Info != "";
		Entry_SettingValueEditable.Visible = IsTextDisplay && IsTextEditable;

		Frame_SettingSliderReal		.Visible = IsSliderSetting;
		Quad_SliderTypeNormal		.Visible = Setting.Type == "Real" || Setting.Type == "IntSlider";
		Quad_SliderTypeColor		.Visible = Setting.Type == "Color";
		Entry_PreciseSliderInput	.Visible = !G_CommonBase_DisablePreciseValues && Setting.Type == "Real" && !Setting.IsDisabled;
		Label_PreciseSliderInputRO	.Visible = !G_CommonBase_DisablePreciseValues && Setting.Type == "Real" && Setting.IsDisabled;
		Frame_SettingSliderColor	.Visible = Quad_SliderTypeColor.Visible;
		Quad_SettingSliderHandle	.Visible = !Quad_SliderTypeColor.Visible;

		Label_SettingEnumArrowPrev.Opacity = 1.;
		Label_SettingEnumArrowNext.Opacity = 1.;

		Frame_SettingEnumArrows.Visible = Setting.Type == "Enum";

		declare IsClickable = Setting.Type == "Bool" || Setting.Type == "Action" || Setting.Type == "Input";

		Quad_SettingCardClickable.Visible	= !Setting.IsDisabled && IsClickable;
		Quad_SettingCardEnabled.Visible		= !Setting.IsDisabled && !IsClickable;
		Quad_SettingCardDisabled.Visible	= Setting.IsDisabled;
		Quad_SliderActionSpace.Visible		= !Setting.IsDisabled;
		Label_SettingNeedsResart.Visible	= Setting.NeedsRestart;

		if (IsSliderSetting) {
			declare ClampedValue = 0.;
			if (Setting.Type == "IntSlider")
				ClampedValue  = ML::Clamp(ML::ToReal(Setting.CurrentValueInteger), Setting.ValueRangeMin, Setting.ValueRangeMax);
			else
				ClampedValue = ML::Clamp(Setting.CurrentValueReal, Setting.ValueRangeMin, Setting.ValueRangeMax);
				
			declare Ratio = 0.;
			if (Setting.AlternativeDisplay) {	// log scale.
				declare Width = ML::Ln(Setting.ValueRangeMax) - ML::Ln(Setting.ValueRangeMin);
				declare Distance = ML::Ln(ClampedValue) - ML::Ln(Setting.ValueRangeMin);
				if (Width > 0) Ratio = Distance / Width;
			} else {
				declare Width = Setting.ValueRangeMax - Setting.ValueRangeMin;
				declare Distance = ClampedValue - Setting.ValueRangeMin;
				if (Width > 0) Ratio = Distance / Width;
			}

			Frame_SettingSliderHandle.RelativePosition_V3.X = Quad_SliderTypeNormal.Size.X * Ratio;
			Entry_PreciseSliderInput.Value = TL::FormatReal(Setting.CurrentValueReal, 3, False, False);
			Label_PreciseSliderInputRO.Value = Entry_PreciseSliderInput.Value;

			if (Quad_SliderTypeColor.Visible) {
				Quad_SettingSliderColor.BgColor = CL::HsvToRgb(<Ratio, 1., 1.>);
			}
		}

		else if (Setting.Type == "Enum") {
			declare DisplayValue = Setting.UpcomingValueText;

			if (Setting.UpcomingValueText == "") {
				DisplayValue = Setting.CurrentValueText;
			}

			declare Key = Setting.AvailableOptions.keyof(DisplayValue);

			if (Setting.OptionsLabels.existskey(DisplayValue)) {
				DisplayValue = Setting.OptionsLabels[DisplayValue];
			}

			declare HasPrev = Key > 0;
			declare HasNext = Key < Setting.AvailableOptions.count - 1;

			Label_SettingValueDisplayOnly.Value = DisplayValue;
			Quad_SettingEnumArrowPrev.Visible = !Setting.IsDisabled && HasPrev;
			Quad_SettingEnumArrowNext.Visible = !Setting.IsDisabled && HasNext;

			if (!HasPrev) Label_SettingEnumArrowPrev.Opacity /= 4.;
			if (!HasNext) Label_SettingEnumArrowNext.Opacity /= 4.;

			foreach (I => Quad in [Quad_SettingEnumArrowPrev, Quad_SettingEnumArrowNext]) {
				declare Text SettingId for Quad;
				SettingId = Setting.Id;
			}
		}
		
		else if (Setting.Type == "Int") {
			Label_SettingValueDisplayOnly.Value = TL::ToText(Setting.CurrentValueInteger);
			Entry_SettingValueEditable.Value = TL::ToText(Setting.CurrentValueInteger);
		}
		
		else if (Setting.Type == "Input") {
			Label_SettingValueDisplayOnly.Value = TL::Join(", ", Setting.CurrentBindings);
		}
		
		else {
			Label_SettingValueDisplayOnly.Value = Setting.CurrentValueText;
			Entry_SettingValueEditable.Value = Setting.CurrentValueText;
		}

		if (Label_SettingValueBoolean.Visible) {
			if (Setting.CurrentValueBoolean) {
				Label_SettingValueBoolean.Value = "";
			} else {
				Label_SettingValueBoolean.Value = "";
			}
		}

		Label_SettingName.Value = TL::GetTranslatedText(Setting.Name);

		foreach (Label in [Label_SettingName, Label_SettingValueDisplayOnly, Label_SettingValueBoolean, Label_PreciseSliderInputRO]) {
			Label.Opacity = 1.;

			if (Setting.IsDisabled) {
				Label.Opacity /= 2.;
			}
		}

		if (!Setting.IsDisabled) {
			Quad_SliderTypeColor.Opacity = 1.;
		} else {
			Quad_SliderTypeColor.Opacity = .5;
			Label_SettingEnumArrowPrev.Opacity /= 2.;
			Label_SettingEnumArrowNext.Opacity /= 2.;
		}

		declare CMlControl[] Controls;
		Controls.add(Quad_SettingCardClickable);
		Controls.add(Quad_SliderActionSpace);
		Controls.add(Entry_SettingValueEditable);
		Controls.add(Entry_PreciseSliderInput);
		Controls.add(Label_SettingInfoToggle);

		foreach (Control in Controls) {
			declare Text SettingId for Control;
			declare Text SettingType for Control;
			declare Vec2 SettingStepScale	for Control;
			declare Vec2 SettingLimits 		for Control;
			declare Text SettingInfo 		for Control;

			SettingId = Setting.Id;
			SettingType = Setting.Type;
			SettingInfo = Setting.Info;
			if (Setting.Type == "Real" || Setting.Type == "Color" || Setting.Type == "IntSlider") { // IsSliderSetting
				if (Setting.AlternativeDisplay)  {
					SettingLimits 		= <ML::Ln(Setting.ValueRangeMin), ML::Ln(Setting.ValueRangeMax)>;
					SettingStepScale 	= <Setting.ValueStep,  1.>; 
				} else {
					SettingLimits 		= <Setting.ValueRangeMin, Setting.ValueRangeMax>;
					SettingStepScale	= <Setting.ValueStep, -1.>;
				}
			}
		}

		declare CMlFrame SliderHandle for Quad_SliderActionSpace;
		SliderHandle <=> Frame_SettingSliderHandle;

		declare Boolean PrevIsSelected for Frame;
		declare IsSelected = G_CommonBase_UseSettingHover && G_CommonBase_SelectedSettingId == Setting.Id;
		if (IsSelected)
			G_CommonBase_SelectedSettingFrame = Frame;

		if (PrevIsSelected != IsSelected) {
			PrevIsSelected = IsSelected;

			AnimMgr.Flush(Quad_SettingCardSelected);

			declare Duration = 100;
			declare Easing = CAnimManager::EAnimManagerEasing::Linear;
			declare Labels = [Label_SettingName, Label_SettingValueDisplayOnly, Label_SettingValueBoolean, Label_SettingEnumArrowPrev, Label_SettingEnumArrowNext, Label_PreciseSliderInputRO];
			declare Quads = [Quad_SliderTypeNormal, Quad_SettingSliderHandle];

			if (IsSelected) {
				AnimMgr.Add(Quad_SettingCardSelected, "<quad opacity='1' />", Duration, Easing);
				AnimMgr.Add(Entry_SettingValueEditable, "<entry textcolor='000' />", Duration, Easing);
				AnimMgr.Add(Entry_PreciseSliderInput, "<entry textcolor='000' />", Duration, Easing);

				foreach (Label in Labels) {
					AnimMgr.Add(Label, "<label textcolor='000' />", Duration, Easing);
				}

				foreach (Quad in Quads) {
					AnimMgr.Add(Quad, "<quad bgcolor='000' />", Duration, Easing);
				}

				Page.ScrollToControl(Quad_SettingCardSelected);
			}

			else {
				AnimMgr.Add(Quad_SettingCardSelected, "<quad opacity='0' />", Duration, Easing);
				AnimMgr.Add(Entry_SettingValueEditable, "<entry textcolor='FFF' />", Duration, Easing);
				AnimMgr.Add(Entry_PreciseSliderInput, "<entry textcolor='FFF' />", Duration, Easing);

				foreach (Label in Labels) {
					AnimMgr.Add(Label, "<label textcolor='FFF' />", Duration, Easing);
				}

				foreach (Quad in Quads) {
					AnimMgr.Add(Quad, "<quad bgcolor='FFF' />", Duration, Easing);
				}
			}
		}
	}
}

Void PrevSetting(SSettingModel[] _SettingsList) {
	declare PrevSettingIsTarget = G_CommonBase_SelectedSettingId == "";

	for (I, 0, _SettingsList.count - 1) {
		declare Setting = _SettingsList[_SettingsList.count - I - 1];

		if (!PrevSettingIsTarget && Setting.Id == G_CommonBase_SelectedSettingId) {
			PrevSettingIsTarget = True;
			continue;
		}

		if (!PrevSettingIsTarget || Setting.IsDisabled) continue;

		G_CommonBase_SelectedSettingId = Setting.Id;
		break;
	}
}

Void NextSetting(SSettingModel[] _SettingsList) {
	declare NextSettingIsTarget = G_CommonBase_SelectedSettingId == "";

	foreach (Setting in _SettingsList) {
		if (!NextSettingIsTarget && Setting.Id == G_CommonBase_SelectedSettingId) {
			NextSettingIsTarget = True;
			continue;
		}

		if (!NextSettingIsTarget || Setting.IsDisabled) continue;

		G_CommonBase_SelectedSettingId = Setting.Id;
		break;
	}
}

Void InitSettingsCards() {
	G_LibLayerCommonBase_LastSettingOrder = 0;
}

Text AddSettingsCards(Integer _Quantity) {
	if (_Quantity <= 0) return "";

	declare Text Result;

	for (I, 0, _Quantity - 1) {
		declare Y = I * -7.5;
		Result ^= "<frameinstance modelid='FrameModel_SettingCard' id='Frame_SettingCard#"^G_LibLayerCommonBase_LastSettingOrder^"' pos='0 "^Y^"' />";
		G_LibLayerCommonBase_LastSettingOrder += 1;
	}

	return Result;
}
