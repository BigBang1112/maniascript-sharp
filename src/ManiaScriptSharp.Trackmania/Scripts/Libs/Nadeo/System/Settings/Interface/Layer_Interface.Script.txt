
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Settings/SettingsStructs.Script.txt"	as Structs
#Include "Libs/Nadeo/System/Settings/Layer_CommonBase.Script.txt"	as LayerBase

#Const C_ImgBase "file://Media/Manialinks/Common/Settings/Interface/"

declare CUILayer G_ManialinkLayerInterface;

Text Private_CreateManialink() {
	LayerBase::InitSettingsCards();

	return LayerBase::CreateLayerMaialink("Settings:Interface", """

<frame id="Frame_PageInterfacePreview" pos="0 0">
	<frame pos="80 -20">
		<quad id="Quad_PreviewCrosshair" size="14 14" halign="center" valign="center" autoscale="0" image="{{{C_ImgBase}}}CrosshairRocket.png" />
		<quad class="PreviewColors" keepratio="Clip" image="{{{C_ImgBase}}}PreviewColors.dds" />
		<quad id="Quad_PreviewTeammates" class="PreviewColors" keepratio="Clip" image="{{{C_ImgBase}}}MaskTeammates.dds" />
		<quad id="Quad_PreviewOpponents" class="PreviewColors" keepratio="Clip" image="{{{C_ImgBase}}}MaskOpponents.dds" />
	</frame>
</frame>

<frame id="Frame_PageInterfaceSettings" pos="0 -42">
	<frame pos="0 -7" id="Frame_PageInterfaceSettings_Crosshair">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Crosshair" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(6)}}}
	</frame>

	<frame pos="0 -61.5" id="Frame_PageInterfaceSettings_PlayersAppearance">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Players appearance" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(4)}}}
	</frame>
	<frame pos="0 -101." id="Frame_PageInterfaceSettings_PingMarkers" >
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Ping markers" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(5)}}}
	</frame>
	<frame pos="0 -149." id="Frame_PageInterfaceSettings_HudMarkers" >
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Hud markers" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(2)}}}
	</frame>
	<frame pos="0 -174." id="Frame_PageInterfaceSettings_Miscs" >
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Miscellaneous" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(8)}}}
	</frame>
	<frame pos="0 -174." id="Frame_PageInterfaceSettings_MapEditorSettings" >
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Map editor quick start" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(4)}}}
	</frame>
</frame>

""", "", """
SetScrollPageHeight(267.5);

declare Frame_PageInterfacePreview 						<=> (Page.GetFirstChild("Frame_PageInterfacePreview") 						as CMlFrame);
declare Frame_PageInterfaceSettings 					<=> (Page.GetFirstChild("Frame_PageInterfaceSettings") 						as CMlFrame);
declare Frame_PageInterfaceSettings_Crosshair 			<=> (Page.GetFirstChild("Frame_PageInterfaceSettings_Crosshair") 			as CMlFrame);
declare Frame_PageInterfaceSettings_PlayersAppearance 	<=> (Page.GetFirstChild("Frame_PageInterfaceSettings_PlayersAppearance") 	as CMlFrame);
declare Frame_PageInterfaceSettings_PingMarkers 		<=> (Page.GetFirstChild("Frame_PageInterfaceSettings_PingMarkers") 			as CMlFrame);
declare Frame_PageInterfaceSettings_HudMarkers 			<=> (Page.GetFirstChild("Frame_PageInterfaceSettings_HudMarkers") 			as CMlFrame);
declare Frame_PageInterfaceSettings_Miscs	 			<=> (Page.GetFirstChild("Frame_PageInterfaceSettings_Miscs") 				as CMlFrame);
declare Frame_PageInterfaceSettings_MapEditorSettings	<=> (Page.GetFirstChild("Frame_PageInterfaceSettings_MapEditorSettings") 	as CMlFrame);

declare Quad_PreviewCrosshair <=> (Page.GetFirstChild("Quad_PreviewCrosshair") as CMlQuad);
declare Quad_PreviewTeammates <=> (Page.GetFirstChild("Quad_PreviewTeammates") as CMlQuad);
declare Quad_PreviewOpponents <=> (Page.GetFirstChild("Quad_PreviewOpponents") as CMlQuad);

declare SSettingModel[] LayerInterface_SettingsList for Page;
declare Integer LayerInterface_LastSettingsUpdate for Page;

declare UpdateColorsPreview = True;
declare CrosshairColor = G_CommonBase_PageColorPrimary;
declare TeammatesColor = G_CommonBase_PageColorPrimary;
declare OpponentsColor = <1., 0., 0.>;

declare PrevSettingsUpdate = -1;
declare PrevPageColor = G_CommonBase_PageColorPrimary;

""", """

foreach (Event in Input.PendingEvents) {
	if (!ListenToGamepadInputs()) break;

	if (Event.Type == CInputEvent::EType::PadButtonPress) {
		if (Event.Button == CInputEvent::EButton::Up || Event.Button == CInputEvent::EButton::LeftStick_Up) {
			PrevSetting(LayerInterface_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerInterface_LastSettingsUpdate = Now;
		}

		else if (Event.Button == CInputEvent::EButton::Down || Event.Button == CInputEvent::EButton::LeftStick_Down) {
			NextSetting(LayerInterface_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerInterface_LastSettingsUpdate = Now;
		}
	}
}

if (PrevSettingsUpdate != LayerInterface_LastSettingsUpdate) {
	PrevSettingsUpdate = LayerInterface_LastSettingsUpdate;
	FormatSettingCards(LayerInterface_SettingsList);
	
	declare Boolean IsTrackmaniaStandalone = (ManiaPlanet.LoadedTitle != Null && ManiaPlanet.LoadedTitle.TitleId == "Trackmania");
	if (IsTrackmaniaStandalone) {		
		Frame_PageInterfacePreview						.Visible 				= False;
		Frame_PageInterfaceSettings						.RelativePosition_V3.Y 	= -7.;
		Frame_PageInterfaceSettings_Crosshair			.Visible				= False;
		Frame_PageInterfaceSettings_PlayersAppearance	.Visible				= False;
		Frame_PageInterfaceSettings_PingMarkers			.Visible				= False;
		Frame_PageInterfaceSettings_HudMarkers			.RelativePosition_V3.Y	= 0.;
		Frame_PageInterfaceSettings_Miscs				.RelativePosition_V3.Y	= -25.;
		Frame_PageInterfaceSettings_MapEditorSettings	.Visible 				= True;
		Frame_PageInterfaceSettings_MapEditorSettings	.RelativePosition_V3.Y	= -94.5;
		SetScrollPageHeight(112.5);
	} else {
		Frame_PageInterfacePreview						.Visible 				= True;
		Frame_PageInterfaceSettings						.RelativePosition_V3.Y 	= -42.;
		Frame_PageInterfaceSettings_Crosshair			.Visible				= True;
		Frame_PageInterfaceSettings_PlayersAppearance	.Visible				= True;
		Frame_PageInterfaceSettings_PingMarkers			.Visible				= True;
		Frame_PageInterfaceSettings_HudMarkers			.RelativePosition_V3.Y	= -149.;
		Frame_PageInterfaceSettings_Miscs				.RelativePosition_V3.Y	= -196.5;
		Frame_PageInterfaceSettings_MapEditorSettings	.Visible 				= False;
		SetScrollPageHeight(252.5);
		UpdateColorsPreview = True;
	}
}

if (PrevPageColor != G_CommonBase_PageColorPrimary) {
	PrevPageColor = G_CommonBase_PageColorPrimary;
	UpdateColorsPreview = True;
}

if (UpdateColorsPreview) {
	UpdateColorsPreview = False;

	declare Hue = CL::RgbToHsv(G_CommonBase_PageColorPrimary).X;
	TeammatesColor = CL::HsvToRgb(<Hue, 1., 1.>);

	Hue += .5;
	while (Hue >= 1.) Hue -= 1.;

	CrosshairColor = TeammatesColor;
	OpponentsColor = CL::HsvToRgb(<Hue, 1., 1.>);

	declare UserSettings <=> ManiaPlanet.UserMgr.MainUserProfile;

	if (UserSettings.Interface_CrosshairEnableCustomColor) {
		CrosshairColor = CL::HsvToRgb(<UserSettings.Interface_CrosshairLinearHue, UserSettings.Interface_CrosshairSaturation, 1.>);
	}

	if (UserSettings.Interface_AllyEnableCustomColor) {
		TeammatesColor = CL::HsvToRgb(<UserSettings.Interface_AllyLinearHue, 1., 1.>);
	}

	if (UserSettings.Interface_OppoEnableCustomColor) {
		OpponentsColor = CL::HsvToRgb(<UserSettings.Interface_OppoLinearHue, 1., 1.>);
	}

	Quad_PreviewCrosshair.Colorize		= CrosshairColor;
	Quad_PreviewTeammates.Colorize		= TeammatesColor;
	Quad_PreviewOpponents.Colorize		= OpponentsColor;
	Quad_PreviewCrosshair.RelativeScale	= UserSettings.Interface_CrosshairSize;
	Quad_PreviewCrosshair.Opacity		= UserSettings.Interface_CrosshairOpacity;
}

	""", "");
}

Void Private_CustomEvent(Text _Name, Text[] _Params) {
	LayerCustomEvent(G_ManialinkLayerInterface, _Name, _Params);
}

Void Build() {
	G_ManialinkLayerInterface = UILayerCreate();
	G_ManialinkLayerInterface.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerInterface.ManialinkPage = Private_CreateManialink();
}

CUILayer GetLayer() {
	return G_ManialinkLayerInterface;
}

Void SetVisibility(Boolean _Visible) {
    G_ManialinkLayerInterface.IsVisible = _Visible;
	Private_CustomEvent("SetPageVisibility", [TL::ToText(_Visible)]);
}

Void ShowPage(Integer _Direction) {
	Private_CustomEvent("ShowPage", [TL::ToText(_Direction)]);
}

Void HidePage(Integer _Direction) {
	Private_CustomEvent("HidePage", [TL::ToText(_Direction)]);
}

Void SetColor(Vec3 _Color) {
	Private_CustomEvent("SetColor", [_Color.X^" "^_Color.Y^" "^_Color.Z]);
}

Void SetSettingsList(Structs::SSettingModel[] _SettingsList) {
	declare Structs::SSettingModel[] LayerInterface_SettingsList for G_ManialinkLayerInterface.LocalPage;
	declare Integer LayerInterface_LastSettingsUpdate for G_ManialinkLayerInterface.LocalPage;

	LayerInterface_SettingsList = _SettingsList;
	LayerInterface_LastSettingsUpdate = Now;
}
