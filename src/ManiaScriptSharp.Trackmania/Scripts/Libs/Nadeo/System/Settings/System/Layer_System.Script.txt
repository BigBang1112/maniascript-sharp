
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Settings/SettingsStructs.Script.txt"	as Structs
#Include "Libs/Nadeo/System/Settings/Layer_CommonBase.Script.txt"	as LayerBase


#Const C_ImgBase "file://Media/Manialinks/Common/Settings/"

declare CUILayer G_ManialinkLayerSystem;

Text Private_CreateManialink() {
	LayerBase::InitSettingsCards();

	return LayerBase::CreateLayerMaialink("Settings:System", """

<frame id="Frame_PageSystemSettings" pos="0 0">
	<frame pos="0 -7">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Credits" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(1)}}}
	</frame>

	<frame pos="0 -25">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Language" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(1)}}}
	</frame>
	
	<frame pos="0 -43">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Network" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(4)}}}
	</frame>
	
	<frame pos="0 -84">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Chat" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(1)}}}
	</frame>
	
	<frame pos="0 -102">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Downloads" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(4)}}}
	</frame>
</frame>

""", "", """
SetScrollPageHeight(130.);

declare SSettingModel[] LayerSystem_SettingsList for Page;
declare Integer LayerSystem_LastSettingsUpdate for Page;

declare PrevSettingsUpdate = -1;
""", """
foreach (Event in Input.PendingEvents) {
	if (!ListenToGamepadInputs()) break;

	if (Event.Type == CInputEvent::EType::PadButtonPress) {
		if (Event.Button == CInputEvent::EButton::Up || Event.Button == CInputEvent::EButton::LeftStick_Up) {
			PrevSetting(LayerSystem_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerSystem_LastSettingsUpdate = Now;
		}

		else if (Event.Button == CInputEvent::EButton::Down || Event.Button == CInputEvent::EButton::LeftStick_Down) {
			NextSetting(LayerSystem_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerSystem_LastSettingsUpdate = Now;
		}
	}
}

if (PrevSettingsUpdate != LayerSystem_LastSettingsUpdate) {
	PrevSettingsUpdate = LayerSystem_LastSettingsUpdate;
	FormatSettingCards(LayerSystem_SettingsList);
}

	""", "");
}

Void Private_CustomEvent(Text _Name, Text[] _Params) {
	LayerCustomEvent(G_ManialinkLayerSystem, _Name, _Params);
}

Void Build() {
	G_ManialinkLayerSystem = UILayerCreate();
	G_ManialinkLayerSystem.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerSystem.ManialinkPage = Private_CreateManialink();
}

CUILayer GetLayer() {
	return G_ManialinkLayerSystem;
}

Void SetVisibility(Boolean _Visible) {
    G_ManialinkLayerSystem.IsVisible = _Visible;
	Private_CustomEvent("SetPageVisibility", [TL::ToText(_Visible)]);
}

Void ShowPage(Integer _Direction) {
	Private_CustomEvent("ShowPage", [TL::ToText(_Direction)]);
}

Void HidePage(Integer _Direction) {
	Private_CustomEvent("HidePage", [TL::ToText(_Direction)]);
}

Void SetColor(Vec3 _Color) {
	Private_CustomEvent("SetColor", [_Color.X^" "^_Color.Y^" "^_Color.Z]);
}

Void SetSettingsList(Structs::SSettingModel[] _SettingsList) {
	declare Structs::SSettingModel[] LayerSystem_SettingsList for G_ManialinkLayerSystem.LocalPage;
	declare Integer LayerSystem_LastSettingsUpdate for G_ManialinkLayerSystem.LocalPage;

	LayerSystem_SettingsList = _SettingsList;
	LayerSystem_LastSettingsUpdate = Now;
}
