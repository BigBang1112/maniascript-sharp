
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Chat/ChatSettings.Script.txt"				as ChatSettings
#Include "Libs/Nadeo/System/Settings/System/SystemAPI.Script.txt"		as SystemAPI
#Include "Libs/Nadeo/System/Settings/Layer_SettingsTabs.Script.txt"		as LayerSettingsTabs
#Include "Libs/Nadeo/System/Settings/System/Layer_System.Script.txt"	as LayerSystem
#Include "Libs/Nadeo/System/Settings/SettingsStructs.Script.txt"		as SettingsStructs

#Const C_LibSystemController_SettingsTabId		"System"

declare Boolean G_LibSystemController_PendingChanges;

Void Private_UpdateSystemSettings() {
	declare SettingsList = SystemAPI::GetSettingsList();

	LayerSystem::SetSettingsList(SettingsList);
}

Text GetTabId() {
	return C_LibSystemController_SettingsTabId;
}

Void SetVisibility(Boolean _Visible) {
	LayerSystem::SetVisibility(_Visible);
}

Void Show(Integer _Direction) {
	LayerSystem::ShowPage(_Direction);
}

Void Hide(Integer _Direction) {
	LayerSystem::HidePage(_Direction);
}

Void SetLayerColor(Vec3 _Color) {
	LayerSystem::SetColor(_Color);
}

CUILayer GetLayer() {
	return LayerSystem::GetLayer();
}

Boolean HasPendingChanges() {
	return G_LibSystemController_PendingChanges;
}

Void MarkNoPendingChanges() {
	G_LibSystemController_PendingChanges = False;	
}

Void Load() {
	LayerSystem::Build();

	LayerSettingsTabs::CreateTab(C_LibSystemController_SettingsTabId, _("|Settings|System"), "tab-system.dds");

	Private_UpdateSystemSettings();
}

Void Loop() {
	SystemAPI::Loop();

	foreach (Event in PendingEvents) {
		if (Event.Type == CManiaAppEvent::EType::LayerCustomEvent) {
			if (Event.CustomEventLayer != LayerSystem::GetLayer()) continue;

			switch (Event.CustomEventType) {
				case "Credits" : {
					declare SettingsStructs::K_PluginEvents PluginEvents for System;
					PluginEvents.Events[PluginEvents.Count] = SettingsStructs::K_PluginEvent {
						Type = SettingsStructs::C_PluginEventType_PushRoute,
						Data = "/credits",
						SetCurrentPathAsParent = True
					};
					PluginEvents.Count += 1;
				}
				case "CleanCache" : {
					SystemAPI::CleanCache();
				}
				case "SetSettingValue" : {
					if (!Event.CustomEventData.existskey(1)) continue;

					declare SettingId = Event.CustomEventData[0];
					declare NewValue = Event.CustomEventData[1];

					switch (SettingId) {
						case "ChatStatusText" : { }		// unused
						case "JoinLink" : {
							if (TL::StartsWith("#", NewValue) || TL::StartsWith("maniaplanet://", NewValue, True, False))
								OpenLink(NewValue, CManiaApp::ELinkType::ManialinkBrowser);
							else
								OpenLink("#join="^NewValue, CManiaApp::ELinkType::ManialinkBrowser);
						}
					}

					Private_UpdateSystemSettings();
				}

				case "SettingEnum" : {
					if (!Event.CustomEventData.existskey(1)) continue;

					declare Setting = SystemAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;

					declare Direction = TL::ToInteger(Event.CustomEventData[1]);
					declare Key = Setting.AvailableOptions.keyof(Setting.CurrentValueText) + Direction;

					if (!Setting.AvailableOptions.existskey(Key)) continue;
					declare NewValue = Setting.AvailableOptions[Key];

					switch (Setting.Id) {
						case "Language"				: {
							ManiaPlanet.Language_NextStart = NewValue;
							G_LibSystemController_PendingChanges = True;
						}
						
						case "NetworkSpeed" : {
							switch (NewValue) {
								case "Custom"		: ManiaPlanet.NetworkSpeed_NextStart = CManiaPlanet::ENetworkSpeed::Custom;
								case "100Kbps"		: ManiaPlanet.NetworkSpeed_NextStart = CManiaPlanet::ENetworkSpeed::_100Kbps;
								case "1Mbps"		: ManiaPlanet.NetworkSpeed_NextStart = CManiaPlanet::ENetworkSpeed::_1Mbps;
								case "10Mbps"		: ManiaPlanet.NetworkSpeed_NextStart = CManiaPlanet::ENetworkSpeed::_10Mbps;
								case "100Mbps"		: ManiaPlanet.NetworkSpeed_NextStart = CManiaPlanet::ENetworkSpeed::_100Mbps;
							}
						}
					}

					Private_UpdateSystemSettings();
				}

				case "ToggleBoolean" : {
					if (!Event.CustomEventData.existskey(0)) continue;

					declare Setting = SystemAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;

					declare NewValue = !Setting.CurrentValueBoolean;
					
					declare UserSettings <=> ManiaPlanet.UserMgr.MainUserProfile;

					switch (Setting.Id) {
						case "ChatAllowHide" 						: if(UserSettings != Null) { UserSettings.Interface_AllowChatHiding = NewValue; }
						case "EnableMapModLocators"			:	ManiaPlanet.Network_EnableMapModLocators		= NewValue;
						case "EnableMapSkinLocators"		:	ManiaPlanet.Network_EnableMapSkinLocators		= NewValue;
						case "EnableVehicleSkinLocators"	:	ManiaPlanet.Network_EnableVehicleSkinLocators	= NewValue;
					}

					Private_UpdateSystemSettings();
				}
			}
		}
	}

	if (SystemAPI::PendingUpdate()) {
		Private_UpdateSystemSettings();
	}
}
