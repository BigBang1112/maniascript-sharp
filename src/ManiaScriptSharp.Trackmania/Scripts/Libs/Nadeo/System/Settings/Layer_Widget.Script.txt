
#Include "Libs/Nadeo/System/Styles/CommonStyles.Script.txt" as Styles

#Const C_ImgBase "file://Media/Manialinks/Common/"

declare CUILayer G_ManialinkLayerWidget;

Text Private_CreateManialink() {
	return """
<manialink version="3" name="Settings:Widget">
<stylesheet>
	{{{Styles::Get()}}}
</stylesheet>

<frame id="Frame_SettingsWidget" pos="116 89" z-index="2">
	<quad id="Quad_ToggleSettingsMenu" ScriptEvents="1"
				halign="center" valign="center" pos="0 -3" size="6 6"
				style="UICommon64_1" substyle="Parameters_light" />

	<quad id="Quad_ToggleUplayOverlay" pos="-32 -3" halign="center" valign="center" image="{{{C_ImgBase^"Ubisoft_Logo.dds"}}}" size="4.2 4.15" ScriptEvents="1" z-index="-1" hidden="1" />
</frame>

<script><!--

#Include "AnimLib" as AL
#Include "MathLib" as ML
#Include "TextLib" as TL

main() {
	declare Quad_ToggleSettingsMenu <=> (Page.GetFirstChild("Quad_ToggleSettingsMenu") as CMlQuad);
	
	if (System.Platform == CSystemPlatformScript::ESystemPlatform::UPlay) {
		declare Quad_ToggleUplayOverlay <=> (Page.GetFirstChild("Quad_ToggleUplayOverlay") as CMlQuad);
		Quad_ToggleUplayOverlay.Visible = True;
	}
	
	declare PrimaryColor = <1., 1., 1.>;
	declare DisableForceShowSystemBar = -1;
	
	while (True) {
		yield;
		if(	PendingEvents.count == 0 && 
				DisableForceShowSystemBar == -1) 
					continue;
					
		foreach (Event in PendingEvents) {
			if (Event.Type == CMlScriptEvent::Type::MouseClick) {
				if (Event.ControlId == "Quad_ToggleSettingsMenu") {
					SendCustomEvent("ToggleSettingsMenu", []);
				} else if (Event.ControlId == "Quad_ToggleUplayOverlay") {
					if (ManiaPlanet.UserMgr.MainUserProfile != Null) {
						ManiaPlanet.UserMgr.Uplay_OpenOverlay(ManiaPlanet.UserMgr.MainUserProfile.Id, CUserV2Manager::EUplayOverlaySection::Home);
					}
				}
			}


			else if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
				switch (Event.CustomEventType) {
					case "SetColor" : {
						if (!Event.CustomEventData.existskey(0)) continue;
						declare SplitColor = TL::Split(" ", Event.CustomEventData[0]);
						PrimaryColor = <1., 1., 1.>;

						if (SplitColor.count != 3) continue;
						for (I, 0, 2) PrimaryColor[I] = TL::ToReal(SplitColor[I]);

						Quad_ToggleSettingsMenu	.BgColorFocus = PrimaryColor * .75;
					}
				}
			}
		}
			
		if (DisableForceShowSystemBar > 0 && Now > DisableForceShowSystemBar) {
			DisableForceShowSystemBar = -1;
			ParentApp.ToolBarForceShow = False;
		}
	}
}

--></script>
</manialink>
	""";
}

Void Build() {
	G_ManialinkLayerWidget = UILayerCreate();
	G_ManialinkLayerWidget.Type = CUILayer::EUILayerType::ManiaplanetPlugin;
	G_ManialinkLayerWidget.ManialinkPage = Private_CreateManialink();
}

CUILayer GetLayer() {
	return G_ManialinkLayerWidget;
}

Void SetColor(Vec3 _Color) {
	LayerCustomEvent(G_ManialinkLayerWidget, "SetColor", [_Color.X^" "^_Color.Y^" "^_Color.Z]);
}
