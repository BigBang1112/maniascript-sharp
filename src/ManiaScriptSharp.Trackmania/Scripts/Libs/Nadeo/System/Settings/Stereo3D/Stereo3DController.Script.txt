
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Settings/Stereo3D/Stereo3DAPI.Script.txt"		as Stereo3DAPI
#Include "Libs/Nadeo/System/Settings/Layer_SettingsTabs.Script.txt"			as LayerSettingsTabs
#Include "Libs/Nadeo/System/Settings/Stereo3D/Layer_Stereo3D.Script.txt"	as LayerStereo3D

#Const C_LibStereo3DController_SettingsTabId		"Stereo3D"

Void Private_UpdateStereo3DSettings() {
	declare SettingsList = Stereo3DAPI::GetSettingsList();

	LayerStereo3D::SetSettingsList(SettingsList);
}

Text GetTabId() {
	return C_LibStereo3DController_SettingsTabId;
}

Void SetVisibility(Boolean _Visible) {
	LayerStereo3D::SetVisibility(_Visible);
}

Void Show(Integer _Direction) {
	LayerStereo3D::ShowPage(_Direction);
}

Void Hide(Integer _Direction) {
	LayerStereo3D::HidePage(_Direction);
}

Void SetLayerColor(Vec3 _Color) {
	LayerStereo3D::SetColor(_Color);
}

CUILayer GetLayer() {
	return LayerStereo3D::GetLayer();
}

Void Load() {
	LayerStereo3D::Build();

	LayerSettingsTabs::CreateTab(C_LibStereo3DController_SettingsTabId, _("|Settings|Stereoscopy"), "tab-vr.dds");

	Private_UpdateStereo3DSettings();
}

Void Loop() {
	Stereo3DAPI::Loop();

	foreach (Event in PendingEvents) {
		if (Event.Type == CManiaAppEvent::EType::LayerCustomEvent) {
			if (Event.CustomEventLayer != LayerStereo3D::GetLayer()) continue;

			switch (Event.CustomEventType) {
				case "SetSettingValue" : {
					if (!Event.CustomEventData.existskey(1)) continue;

					declare SettingId = Event.CustomEventData[0];
					declare NewValue = Event.CustomEventData[1];

				}
			}
		}
	}

	if (Stereo3DAPI::PendingUpdate()) {
		Private_UpdateStereo3DSettings();
	}
}
