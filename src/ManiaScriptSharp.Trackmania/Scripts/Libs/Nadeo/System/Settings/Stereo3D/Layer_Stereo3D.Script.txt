
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Settings/SettingsStructs.Script.txt"	as Structs
#Include "Libs/Nadeo/System/Settings/Layer_CommonBase.Script.txt"	as LayerBase

declare CUILayer G_ManialinkLayerStereo3D;

Text Private_CreateManialink() {
	LayerBase::InitSettingsCards();

	return LayerBase::CreateLayerMaialink("Settings:Stereo3D", """

<frame id="Frame_PageStereo3DSettings" pos="0 0">
	<frame pos="0 -7">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="General" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(3)}}}
	</frame>

	<frame pos="0 -39">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="3D Glasses" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(5)}}}
	</frame>

	<frame pos="0 -86.5">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Virtual Reality" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(4)}}}
	</frame>
</frame>

""", "", """
SetScrollPageHeight(120.);

declare SSettingModel[] LayerStereo3D_SettingsList for Page;
declare Integer LayerStereo3D_LastSettingsUpdate for Page;

declare PrevSettingsUpdate = -1;
""", """

if (PrevSettingsUpdate != LayerStereo3D_LastSettingsUpdate) {
	PrevSettingsUpdate = LayerStereo3D_LastSettingsUpdate;
	FormatSettingCards(LayerStereo3D_SettingsList);
}

	""", "");
}

Void Private_CustomEvent(Text _Name, Text[] _Params) {
	LayerCustomEvent(G_ManialinkLayerStereo3D, _Name, _Params);
}

Void Build() {
	G_ManialinkLayerStereo3D = UILayerCreate();
	G_ManialinkLayerStereo3D.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerStereo3D.ManialinkPage = Private_CreateManialink();
}

CUILayer GetLayer() {
	return G_ManialinkLayerStereo3D;
}

Void SetVisibility(Boolean _Visible) {
	G_ManialinkLayerStereo3D.IsVisible = _Visible;
	Private_CustomEvent("SetPageVisibility", [TL::ToText(_Visible)]);
}

Void ShowPage(Integer _Direction) {
	Private_CustomEvent("ShowPage", [TL::ToText(_Direction)]);
}

Void HidePage(Integer _Direction) {
	Private_CustomEvent("HidePage", [TL::ToText(_Direction)]);
}

Void SetColor(Vec3 _Color) {
	Private_CustomEvent("SetColor", [_Color.X^" "^_Color.Y^" "^_Color.Z]);
}

Void SetSettingsList(Structs::SSettingModel[] _SettingsList) {
	declare Structs::SSettingModel[] LayerStereo3D_SettingsList for G_ManialinkLayerStereo3D.LocalPage;
	declare Integer LayerStereo3D_LastSettingsUpdate for G_ManialinkLayerStereo3D.LocalPage;

	LayerStereo3D_SettingsList = _SettingsList;
	LayerStereo3D_LastSettingsUpdate = Now;
}
