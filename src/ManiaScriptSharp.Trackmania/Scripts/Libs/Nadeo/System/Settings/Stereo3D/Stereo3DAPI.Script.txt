
#Include "MathLib" as ML
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Settings/SettingsStructs.Script.txt" as Structs

#Const C_LibStereo3DAPI_AvailableSettings """
<settings>
	<setting type="Bool" id="StereoEnabled" name="Enable stereoscopy" />
	<setting type="Enum" id="StereoMode" name="Stereoscopy mode" />
	<setting type="Action" id="StereoReset" name="Restore default settings" />

	<setting type="Real" id="StereoDisplaySizeCM" name="Display size (centimeters)" />
	<setting type="Real" id="StereoDisplaySizeIN" name="Display size (inches)" />
	<setting type="Real" id="StereoStrength" name="Strength" min="0" max="100" />
	<setting type="Real" id="StereoConvergance" name="Convergance" min="0" max="10" />
	<setting type="Real" id="StereoColor" name="Saturation" min="0" max="1" />

	<setting type="Action" id="StereoDetectHMD" name="Detect head mounted device" />
	<setting type="Real" id="StereoWorldScale" name="World scale" min="0" max="1" />
	<setting type="Bool" id="StereoRecommended" name="Use recommended settings" />
	<setting type="Bool" id="StereoResetOnRace" name="Reset tracking on race start" />
</settings>
"""

declare Structs::SSettingModel[] G_LibStereo3DAPI_AvailableSettings;
declare Boolean G_LibStereo3DAPI_PendingSettingsUpdate;

Structs::SSettingModel[] Private_GetSettingsList() {
	if (G_LibStereo3DAPI_AvailableSettings.count > 0) {
		return G_LibStereo3DAPI_AvailableSettings;
	}

	declare Document <=> Xml.Parse_Xml(C_LibStereo3DAPI_AvailableSettings);
	
	foreach (Node in Document.Root.Children) {
		declare Structs::SSettingModel Setting;

		Setting.Id				= Node.GetAttributeText("id", "");
		Setting.Type			= Node.GetAttributeText("type", "");
		Setting.Name			= Node.GetAttributeText("name", "");
		Setting.ValueRangeMin	= Node.GetAttributeReal("min", 0.);
		Setting.ValueRangeMax	= Node.GetAttributeReal("max", 1.);
		Setting.Info			= Node.GetAttributeText("info", "");

		switch (Setting.Id) {
			case "StereoMode" : {
				Setting.AvailableOptions = ["Anaglyph", "LeftRight", "LeftRight2", "LineEven", "LineOdd", "HMD"];
				Setting.OptionsLabels = ["Anaglyph" => _("Anaglyph"), "LeftRight" => _("Left | Right"), "LeftRight2" => _("Right | Left"), "LineEven" => _("Polarized (even)"), "LineOdd" => _("Polarized (odd)"), "HMD" => _("Virtual Reality")];
			}
		}

		G_LibStereo3DAPI_AvailableSettings.add(Setting);
	}

	Xml.Parse_Destroy(Document);
	return G_LibStereo3DAPI_AvailableSettings;
}

Structs::SSettingModel[] GetSettingsList() {
	declare SettingsList = Private_GetSettingsList();

	for (I, 0, SettingsList.count - 1) {
		if (SettingsList[I].Type == "Enum") {
			if (!SettingsList[I].AvailableOptions.exists(SettingsList[I].CurrentValueText)) {
				SettingsList[I].CurrentValueText = SettingsList[I].AvailableOptions[0];
			}
		}

		SettingsList[I].IsDisabled = True;
	}

	return SettingsList;
}

Structs::SSettingModel GetSetting(Text _SettingId) {
	declare Structs::SSettingModel TargetSetting;

	foreach (Setting in GetSettingsList()) {
		if (Setting.Id == _SettingId) {
			TargetSetting = Setting;
			break;
		}
	}

	return TargetSetting;
}

Void Loop() {
	
}

Boolean PendingUpdate() {
	declare IsUpdate = G_LibStereo3DAPI_PendingSettingsUpdate;
	G_LibStereo3DAPI_PendingSettingsUpdate = False;
	return IsUpdate;
}
