
#Include "MathLib" as ML
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Settings/Inputs/InputsAPI.Script.txt"		as InputsAPI
#Include "Libs/Nadeo/System/Settings/Layer_SettingsTabs.Script.txt"		as LayerSettingsTabs
#Include "Libs/Nadeo/System/Settings/Inputs/Layer_Inputs.Script.txt"	as LayerInputs

#Const C_LibInputsController_SettingsTabId		"Inputs"

Void Private_UpdateInputsSettings() {
	if (ManiaPlanet.UserMgr.MainUserProfile == Null) return;

	declare SettingsList = InputsAPI::GetSettingsList();

	LayerInputs::SetSettingsList(SettingsList);
	LayerInputs::SetInputCategories(InputsAPI::GetGroupSizes(), InputsAPI::GetGroupNames());
}

Text GetTabId() {
	return C_LibInputsController_SettingsTabId;
}

Void SetVisibility(Boolean _Visible) {
	LayerInputs::SetVisibility(_Visible);
}

Void Show(Integer _Direction) {
	Private_UpdateInputsSettings();
	LayerInputs::ShowPage(_Direction);
}

Void Hide(Integer _Direction) {
	LayerInputs::HidePage(_Direction);
}

Void SetLayerColor(Vec3 _Color) {
	LayerInputs::SetColor(_Color);
}

CUILayer GetLayer() {
	return LayerInputs::GetLayer();
}

Void Load() {
	LayerInputs::Build();

	LayerSettingsTabs::CreateTab(C_LibInputsController_SettingsTabId, _("|Settings|Input"), "tab-inputs.dds");

	Private_UpdateInputsSettings();
}

Void Loop() {
	InputsAPI::Loop();

	foreach (Event in PendingEvents) {
		declare UserSettings <=> ManiaPlanet.UserMgr.MainUserProfile;

		if (Event.Type == CManiaAppEvent::EType::LayerCustomEvent) {
			if (Event.CustomEventLayer != LayerInputs::GetLayer()) continue;

			switch (Event.CustomEventType) {
				case "GlobalUnbindAll" : {
					InputsAPI::UnbindAllCurrent();
					wait(ManiaPlanet.Dialog_IsFinished);
					Private_UpdateInputsSettings();
				}

				case "GlobalUnbindOne" : {
					InputsAPI::UnbindOneButton();
					wait(ManiaPlanet.Dialog_IsFinished);
					Private_UpdateInputsSettings();
				}

				case "GlobalResetBinds" : {
					InputsAPI::RestoreDefaultsCurrent();
					wait(ManiaPlanet.Dialog_IsFinished);
					Private_UpdateInputsSettings();
				}

				case "EditAction" : {
					if (!Event.CustomEventData.existskey(0)) continue;
					declare ActionId = TL::ToInteger(Event.CustomEventData[0]);
					if (ActionId < 0 || !ManiaPlanet.Dialog_IsFinished) continue;

					ManiaPlanet.Dialog_BindInput(ActionId, Null);
					wait(ManiaPlanet.Dialog_IsFinished);
					Private_UpdateInputsSettings();
				}

				case "SetSettingValue" : {
					if (!Event.CustomEventData.existskey(1)) continue;

					declare Setting = InputsAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;

					declare NewValue = ML::Clamp(TL::ToReal(Event.CustomEventData[1]), Setting.ValueRangeMin, Setting.ValueRangeMax);
					declare VehicleSettings <=> InputsAPI::GetCurVehicleSettings();

					switch (Setting.Id) {
						case "MouseSensitivity"		: UserSettings.Inputs_MouseSensitivityDefault_Raw	= NewValue;
						case "MouseAcceleration"		: UserSettings.Inputs_MouseAccel										= NewValue;
						case "MouseScaleVertical"	: UserSettings.Inputs_MouseScaleY									= NewValue;
						case "MouseScaleFreeLook"	: UserSettings.Inputs_MouseScaleFreeLook						= NewValue;
						case "MouseScaleLaser"			: UserSettings.Inputs_MouseSensitivityLaser_Raw		= NewValue;
						
						case "AnalogSensitivity"		: if (VehicleSettings != Null) VehicleSettings.AnalogSensitivity		= NewValue;
						case "AnalogDeadZone"			: if (VehicleSettings != Null) VehicleSettings.AnalogDeadZone			= NewValue;
						case "AnalogRumbleIntens"	: if (VehicleSettings != Null) VehicleSettings.RumbleIntensity			= NewValue;
						case "AnalogCenterSpring"	: if (VehicleSettings != Null) VehicleSettings.CenterSpringIntensity	= NewValue;
					}

					Private_UpdateInputsSettings();
				}

				case "StopSliderDragging" : {
					Private_UpdateInputsSettings();
				}

				case "SettingEnum" : {
					if (!Event.CustomEventData.existskey(1)) continue;

					declare Setting = InputsAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;

					declare Direction = TL::ToInteger(Event.CustomEventData[1]);
					declare Key = Setting.AvailableOptions.keyof(Setting.CurrentValueText) + Direction;

					if (!Setting.AvailableOptions.existskey(Key)) continue;
					declare NewValue = Setting.AvailableOptions[Key];

					switch (Setting.Id) {
						case "AnalogCurVehicle" : {
							InputsAPI::SetSelectedVehicleId(NewValue);
						}

						case "GlobalCurDevice" : {
							InputsAPI::SetSelectedDeviceId(TL::ToInteger(NewValue));
						}

						case "MouseReleaseKey" : {
							switch (NewValue) {
								case "LeftAlt"		: UserSettings.Inputs_MouseReleaseKey = CUserV2Profile::EInputMouseReleaseKey::LeftAlt;
								case "RightAlt"		: UserSettings.Inputs_MouseReleaseKey = CUserV2Profile::EInputMouseReleaseKey::RightAlt;
								case "LeftCtrl"		: UserSettings.Inputs_MouseReleaseKey = CUserV2Profile::EInputMouseReleaseKey::LeftCtrl;
								case "RightCtrl"	: UserSettings.Inputs_MouseReleaseKey = CUserV2Profile::EInputMouseReleaseKey::RightCtrl;
							}
						}
					}

					Private_UpdateInputsSettings();
				}

				case "ToggleBoolean" : {
					if (!Event.CustomEventData.existskey(0)) continue;

					declare Setting = InputsAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;

					declare NewValue = !Setting.CurrentValueBoolean;
					declare VehicleSettings <=> InputsAPI::GetCurVehicleSettings();

					switch (Setting.Id) {
						case "MouseInvertVertical"	: UserSettings.Inputs_MouseLookInvertY = NewValue;
						case "AnalogWheelSensi"		: if (VehicleSettings != Null) VehicleSettings.AnalogSteerV2 = NewValue;
					}

					Private_UpdateInputsSettings();
				}

				case "SliderIncrement" : {
					if (!Event.CustomEventData.existskey(1)) continue;

					declare Setting = InputsAPI::GetSetting(Event.CustomEventData[0]);
					if (Setting.Id == "") continue;
					if (Setting.Type != "Real") continue;

					declare Direction = TL::ToInteger(Event.CustomEventData[1]);
					declare NewValue = ML::Clamp(Setting.CurrentValueReal + (Setting.ValueStep * Direction), Setting.ValueRangeMin, Setting.ValueRangeMax);

					switch (Setting.Id) {
						case "MouseSensitivity"		: UserSettings.Inputs_MouseSensitivityDefault_Raw	= NewValue;
						case "MouseAcceleration"		: UserSettings.Inputs_MouseAccel										= NewValue;
						case "MouseScaleVertical"	: UserSettings.Inputs_MouseScaleY									= NewValue;
						case "MouseScaleFreeLook"	: UserSettings.Inputs_MouseScaleFreeLook						= NewValue;
						case "MouseScaleLaser"			: UserSettings.Inputs_MouseSensitivityLaser_Raw		= NewValue;
					}

					Private_UpdateInputsSettings();
				}
			}
		}
	}

	if (InputsAPI::PendingUpdate()) {
		Private_UpdateInputsSettings();
	}
}
