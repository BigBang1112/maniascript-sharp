
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Settings/SettingsStructs.Script.txt"	as Structs
#Include "Libs/Nadeo/System/Settings/Layer_CommonBase.Script.txt"	as LayerBase


#Const C_ImgBase "file://Media/Manialinks/Common/Settings/"

declare CUILayer G_ManialinkLayerGraphics;

Text Private_CreateManialink() {
	LayerBase::InitSettingsCards();

	return LayerBase::CreateLayerMaialink("Settings:Graphics", """

<frame id="Frame_PageGraphicsSettings" pos="0 0">
	<frame pos="0 -7">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Display" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(11)}}}
	</frame>

	<frame pos="0 -97.5">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Quality" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(12)}}}
	</frame>

	<frame pos="0 -197.5">
		<label class="Bold" pos="3 5" textsize="2" textprefix="$t" text="Synchronisation" translate="1" />
		<quad class="Separator" pos="0 2" />
		{{{LayerBase::AddSettingsCards(2)}}}
	</frame>

</frame>

""", "", """
SetScrollPageHeight(220.);

declare SSettingModel[] LayerGraphics_SettingsList for Page;
declare Integer LayerGraphics_LastSettingsUpdate for Page;

declare PrevSettingsUpdate = -1;
""", """

foreach (Event in Input.PendingEvents) {
	if (!ListenToGamepadInputs()) break;

	if (Event.Type == CInputEvent::EType::PadButtonPress) {
		if (Event.Button == CInputEvent::EButton::Up || Event.Button == CInputEvent::EButton::LeftStick_Up) {
			PrevSetting(LayerGraphics_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerGraphics_LastSettingsUpdate = Now;
		}

		else if (Event.Button == CInputEvent::EButton::Down || Event.Button == CInputEvent::EButton::LeftStick_Down) {
			NextSetting(LayerGraphics_SettingsList);
			G_CommonBase_UseSettingHover = True;
			LayerGraphics_LastSettingsUpdate = Now;
		}
	}
}

if (PrevSettingsUpdate != LayerGraphics_LastSettingsUpdate) {
	PrevSettingsUpdate = LayerGraphics_LastSettingsUpdate;

	FormatSettingCards(LayerGraphics_SettingsList);
}

	""", "");
}

Void Private_CustomEvent(Text _Name, Text[] _Params) {
	LayerCustomEvent(G_ManialinkLayerGraphics, _Name, _Params);
}

Void Build() {
	G_ManialinkLayerGraphics = UILayerCreate();
	G_ManialinkLayerGraphics.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerGraphics.ManialinkPage = Private_CreateManialink();
}

CUILayer GetLayer() {
	return G_ManialinkLayerGraphics;
}

Void SetVisibility(Boolean _Visible) {
    G_ManialinkLayerGraphics.IsVisible = _Visible;
	Private_CustomEvent("SetPageVisibility", [TL::ToText(_Visible)]);
}

Void ShowPage(Integer _Direction) {
	Private_CustomEvent("ShowPage", [TL::ToText(_Direction)]);
}

Void HidePage(Integer _Direction) {
	Private_CustomEvent("HidePage", [TL::ToText(_Direction)]);
}

Void SetColor(Vec3 _Color) {
	Private_CustomEvent("SetColor", [_Color.X^" "^_Color.Y^" "^_Color.Z]);
}

Void SetSettingsList(Structs::SSettingModel[] _SettingsList) {
	declare Structs::SSettingModel[] LayerGraphics_SettingsList for G_ManialinkLayerGraphics.LocalPage;
	declare Integer LayerGraphics_LastSettingsUpdate for G_ManialinkLayerGraphics.LocalPage;

	LayerGraphics_SettingsList = _SettingsList;
	LayerGraphics_LastSettingsUpdate = Now;
}
