
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Links/TitleInfoStructs.Script.txt"	as Structs
#Include "Libs/Nadeo/System/Styles/CommonStyles.Script.txt"		as Styles

#Const C_ImgBase "file://Media/Manialinks/Common/Links/"

declare CUILayer G_ManialinkLayerTitleLinks;

Text Private_CreateManialink() {
	declare LinkLabels = "";

	for (I, 0, 6) {
		declare X = I * 7.;
		LinkLabels ^= "<label id='Label_TitleLinkIcon#"^I^"' class='Text Link Focus' halign='center' textsize='2' pos='"^X^"' ScriptEvents='1' hidden='1' />";
	}

	return """
<manialink version="3" name="Links:Window">
<stylesheet>
	{{{Styles::Get()}}}
	<style class="Separator" size="116 .25" halign="center" valign="center" bgcolor="FFF" opacity=".25" />
</stylesheet>

<frame id="Frame_WindowClip" pos="98 90" size="124 154" halign="center" z-index="10">
<frame id="Frame_TitleLinksWindow" pos="-60 60">
	<quad size="120 1" class="ColorFg2" />
	<quad class="ColorFg" pos="-4.5 4.5" size="64.5 5" scale="2" style="Bgs1InRace" substyle="BgButtonGlow" />
	<label id="Label_TopArrow" pos="60 2" halign="center" textsize="3" text="" />

	<frame pos="0 -1" size="32 32" id="Frame_CurrentTitleMaker">
		<frame pos="8 -8" rot="-45">
			<label class="Text" pos="0 1.5" size="18" halign="center" textsize=".875" textemboss="1" textprefix="$t" text="Made with" translate="1" />
			<label id="Label_CurrentTitleMaker" class="Bold" pos="0 -1.5" size="24" halign="center" textsize="1" textemboss="1" />
			<quad class="ColorFg2" size="32 8" halign="center" valign="center" />
		</frame>
	</frame>

	<frame pos="60 -3">
		<quad id="Quad_CurrentTitleLogo" pos="0 -12" size="56 24" halign="center" valign="center" />
		<quad id="Quad_ManiaPlanetLogo" pos="0 -12" size="72 18" halign="center" valign="center" style="ManiaPlanetLogos" substyle="ManiaPlanetLogoWhite" />
		<label id="Label_CurrentTitlePunchline" size="116" class="Bold" halign="center" valign="top" textsize="1.75" textemboss="1" autonewline="1" />
		<label id="Label_CurrentTitleDetails" size="116" class="Text" halign="center" textsize=".75" textcolor="888" />
		<label id="Label_CurrentTitleDescription" size="116" class="Text" halign="center" valign="top" textsize="1" autonewline="1" />

		<frame id="Frame_CurrentTitleLinks">
			<frame pos="0 -2">
				<label id="Label_LinksFrameHeader"  class="Text" halign="center" textsize=".75" textprefix="$t" />
				<label id="Label_LinksFrameHeader2" class="Text" halign="center" textsize=".75" textprefix="$t" />
			</frame>
			<frame id="Frame_TitleLinksOffset" pos="0 -6.5">
				{{{LinkLabels}}}
			</frame>
		</frame>
	</frame>

	<quad id="Quad_WindowFooterMetalBar" size="120.125 1" style="Bgs1InRace" substyle="BgMetalBar" />
	<quad class="ColorBg" pos="0 -26" size="120 10" image="{{{C_ImgBase}}}image-grad.dds" />
	<quad id="Quad_WindowBackgroundColor" class="ColorBg" pos="0 -36" size="120" image="{{{C_ImgBase}}}background.dds" ScriptEvents="1" />
	<quad id="Quad_CurrentTitleBackground" pos="60 -1" size="120 35" halign="center" keepratio="Clip" />
	<quad pos="60 -1" size="120 35" halign="center" bgcolor="777" />
	<quad id="Quad_WindowBackgroundShadow" pos="-2.5 1.5" size="250 1" scale=".5" style="Bgs1InRace" substyle="BgButtonShadow" opacity=".5" />
</frame>
</frame>

<script><!--

#Include "MathLib" as ML
#Include "TextLib" as TL
#Include "TimeLib" as Time

{{{Structs::IntoManialink()}}}

declare Vec3 G_ColorPrimary;
declare Vec3 G_ColorSecondary;

Void Nothing() {

}

main() {
	declare Frame_WindowClip				<=> (Page.GetFirstChild("Frame_WindowClip")					as CMlFrame);
	declare Frame_TitleLinksWindow			<=> (Page.GetFirstChild("Frame_TitleLinksWindow")			as CMlFrame);
	declare Label_TopArrow					<=> (Page.GetFirstChild("Label_TopArrow")					as CMlLabel);
	declare Quad_CurrentTitleLogo			<=> (Page.GetFirstChild("Quad_CurrentTitleLogo")			as CMlQuad);
	declare Quad_ManiaPlanetLogo			<=> (Page.GetFirstChild("Quad_ManiaPlanetLogo")				as CMlQuad);
	declare Label_CurrentTitlePunchline		<=> (Page.GetFirstChild("Label_CurrentTitlePunchline")		as CMlLabel);
	declare Label_CurrentTitleDetails		<=> (Page.GetFirstChild("Label_CurrentTitleDetails")		as CMlLabel);
	declare Label_CurrentTitleDescription	<=> (Page.GetFirstChild("Label_CurrentTitleDescription")	as CMlLabel);
	declare Quad_CurrentTitleBackground		<=> (Page.GetFirstChild("Quad_CurrentTitleBackground")		as CMlQuad);
	declare Frame_CurrentTitleMaker			<=> (Page.GetFirstChild("Frame_CurrentTitleMaker")			as CMlFrame);
	declare Label_CurrentTitleMaker			<=> (Page.GetFirstChild("Label_CurrentTitleMaker")			as CMlLabel);
	declare Frame_CurrentTitleLinks			<=> (Page.GetFirstChild("Frame_CurrentTitleLinks")			as CMlFrame);
	declare Frame_TitleLinksOffset			<=> (Page.GetFirstChild("Frame_TitleLinksOffset")			as CMlFrame);
	declare Label_LinksFrameHeader			<=> (Page.GetFirstChild("Label_LinksFrameHeader")			as CMlLabel);
	declare Label_LinksFrameHeader2			<=> (Page.GetFirstChild("Label_LinksFrameHeader2")			as CMlLabel);

	declare Quad_WindowBackgroundColor	<=> (Page.GetFirstChild("Quad_WindowBackgroundColor")	as CMlQuad);
	declare Quad_WindowBackgroundShadow	<=> (Page.GetFirstChild("Quad_WindowBackgroundShadow")	as CMlQuad);
	declare Quad_WindowFooterMetalBar	<=> (Page.GetFirstChild("Quad_WindowFooterMetalBar")	as CMlQuad);

	declare CMlLabel[] Label_TitleLinkIcons;

	for (I, 0, 6) {
		Label_TitleLinkIcons.add((Page.GetFirstChild("Label_TitleLinkIcon#"^I) as CMlLabel));
	}

	declare Boolean				LayerTitleLinks_IsVisible	for Page;
	declare STitleOnlineInfo	LayerTitleLinks_TitleInfo	for Page;

	declare UpdateColorScheme = True;
	declare UpdateTitleInfo = True;
	declare WindowHideOffset = 0.;

	declare PrevLayerIsVisible = False;

	G_ColorPrimary		= <1., 1., 1.>;
	G_ColorSecondary	= <1., 1., 1.>;

	while (True) {
		yield;

		{{{Styles::Sounds()}}}

		foreach (Event in PendingEvents) {
			if (Event.Type == CMlScriptEvent::Type::MouseClick) {
				if (Event.Control.HasClass("Link")) {
					declare Text ExternalURL for Event.Control;
					OpenLink(ExternalURL, CMlScript::LinkType::ExternalBrowser);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::MouseOver) {
				if (Event.Control.HasClass("Link")) {
					declare Duration = 125;
					declare Easing = CAnimManager::EAnimManagerEasing::SineOut;

					AnimMgr.Flush(Event.Control);
					AnimMgr.Flush(Label_LinksFrameHeader);
					AnimMgr.Flush(Label_LinksFrameHeader2);

					AnimMgr.Add(Event.Control, "<label scale='1.125' textcolor='"^TL::ColorToText(G_ColorPrimary)^"' />", Duration, Easing);
					Label_LinksFrameHeader.Opacity = 0.;
					Label_LinksFrameHeader2.Opacity = 1.;

					declare Text ExternalURL for Event.Control;

					if (ExternalURL == LayerTitleLinks_TitleInfo.TitlePageURL) {
						Label_LinksFrameHeader2.Value = _("Title page");
					} else if (ExternalURL == LayerTitleLinks_TitleInfo.DownloadURL) {
						Label_LinksFrameHeader2.Value = _("Direct download link");
					} else if (TL::Find("facebook.com/", ExternalURL, True, True)) {
						Label_LinksFrameHeader2.Value = "Facebook";
					} else if (TL::Find("youtube.com/", ExternalURL, True, True)) {
						Label_LinksFrameHeader2.Value = "YouTube";
					} else if (TL::Find("twitter.com/", ExternalURL, True, True)) {
						Label_LinksFrameHeader2.Value = "Twitter";
					} else if (TL::Find("forum.maniaplanet.com", ExternalURL, True, True)) {
						Label_LinksFrameHeader2.Value = _("Maniaplanet Forum");
					} else if (TL::Find("blog.maniaplanet.com", ExternalURL, True, True)) {
						Label_LinksFrameHeader2.Value = _("Maniaplanet Blog");
					} else if (TL::Find("doc.maniaplanet.com", ExternalURL, True, True)) {
						Label_LinksFrameHeader2.Value = _("Technical documentation");
					} else {
						Label_LinksFrameHeader2.Value = _("Official website");
					}
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::MouseOut) {
				if (Event.Control.HasClass("Link")) {
					declare Duration = 125;
					declare Easing = CAnimManager::EAnimManagerEasing::SineIn;

					AnimMgr.Flush(Event.Control);
					AnimMgr.Flush(Label_LinksFrameHeader);
					AnimMgr.Flush(Label_LinksFrameHeader2);

					AnimMgr.Add(Event.Control, "<label scale='1' textcolor='FFF' />", Duration, Easing);
					AnimMgr.Add(Label_LinksFrameHeader, "<label opacity='1' />", Now + Duration, Duration, Easing);
					AnimMgr.Add(Label_LinksFrameHeader2, "<label opacity='0' />", Now + Duration, Duration, Easing);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
				switch (Event.CustomEventType) {
					case "SetPosition" : {
						if (!Event.CustomEventData.existskey(0)) continue;
						declare TargetPosition = TL::ToReal(Event.CustomEventData[0]);
						declare ClampedPosition = ML::Clamp(TargetPosition, -98., 98.);

						Frame_WindowClip.RelativePosition_V3.X = ClampedPosition;
						Label_TopArrow.RelativePosition_V3.X = TargetPosition - ClampedPosition - Frame_TitleLinksWindow.RelativePosition_V3.X;
					}
					case "SetColors" : {
						if (!Event.CustomEventData.existskey(1)) continue;

						declare SplitColorPrimary	= TL::Split(" ", Event.CustomEventData[0]);
						declare SplitColorSecondary	= TL::Split(" ", Event.CustomEventData[1]);

						if (SplitColorPrimary.count != 3 && SplitColorSecondary.count != 3) continue;

						for (I, 0, 2) {
							G_ColorPrimary		[I] = TL::ToReal(SplitColorPrimary[I]);
							G_ColorSecondary	[I] = TL::ToReal(SplitColorSecondary[I]);
						}

						UpdateColorScheme = True;
					}
					case "UpdateTitleInfo" : {
						UpdateTitleInfo = True;
					}
				}
			}
		}

		if (PrevLayerIsVisible != LayerTitleLinks_IsVisible) {
			PrevLayerIsVisible = LayerTitleLinks_IsVisible;

			AnimMgr.Flush(Frame_TitleLinksWindow);

			declare Pos = <Frame_TitleLinksWindow.RelativePosition_V3.X, -8.>;
			declare Easing = CAnimManager::EAnimManagerEasing::SineOut;

			if (!LayerTitleLinks_IsVisible) {
				Pos.Y += WindowHideOffset;
				Easing = CAnimManager::EAnimManagerEasing::SineIn;
			}

			AnimMgr.Add(Frame_TitleLinksWindow, "<frame pos='"^Pos.X^" "^Pos.Y^"' />", 200, Easing);
		}

		if (!LayerTitleLinks_IsVisible) continue;

		if (UpdateColorScheme) {
			UpdateColorScheme = False;
			UpdateTitleInfo = True;

			Label_TopArrow.TextColor = G_ColorPrimary;
			Label_LinksFrameHeader2.TextColor = G_ColorPrimary;

			Page.GetClassChildren("ColorBg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = G_ColorSecondary;
				}
			}

			Page.GetClassChildren("ColorFg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = G_ColorPrimary;
				}
			}

			Page.GetClassChildren("ColorFg2", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).BgColor = G_ColorPrimary;
				}
			}
		}

		if (UpdateTitleInfo) {
			UpdateTitleInfo = False;

			declare WindowTotalHeight = 1. + Quad_CurrentTitleBackground.Size.Y - 11.;
			declare TitleInfo = LayerTitleLinks_TitleInfo;
			declare DetailsLabelValue = "";
			declare Text[] TitleURLs;

			if (TitleInfo.UID == "") {
				DetailsLabelValue = TL::Compose(_("Build %1"), "$<$FFF" ^ ManiaPlanet.System.ExeVersion ^ "$>");
				Label_LinksFrameHeader.Value = TL::Compose(_("Find more about $<%1$>"), "Maniaplanet");
				TitleInfo.BackgroundURL = "{{{C_ImgBase}}}maniaplanet.dds";

				TitleURLs = [
					"https://www.maniaplanet.com/",
					"https://forum.maniaplanet.com/",
					"https://blog.maniaplanet.com/",
					"https://doc.maniaplanet.com/"
				];
			}

			else {
				Label_LinksFrameHeader.Value = TL::Compose(_("Find more about $<%1$>"), TitleInfo.Name);

				DetailsLabelValue = TL::Compose(
					_("Crafted by %1 — Last updated %2 ago — Costs %3 Planets"),
					"$<$FFF$h[profile?" ^ TitleInfo.AuthorLogin ^ "]" ^ TitleInfo.AuthorNickName ^ "$>",
					"$<$FFF" ^ Time::FormatDelta(Time::GetCurrent(), TL::ToText(TitleInfo.UpdatedAt), Time::EDurationFormats::Full) ^ "$>",
					"$<$FFF" ^ TitleInfo.PlanetsCost ^ "$>"
				);

				TitleURLs.add(TitleInfo.TitlePageURL);
				foreach (URL in TitleInfo.ExternalURLs) TitleURLs.add(URL);
				TitleURLs.add(TitleInfo.DownloadURL);
			}

			Quad_CurrentTitleLogo.Visible = TitleInfo.UID != "";
			Quad_ManiaPlanetLogo.Visible = !Quad_CurrentTitleLogo.Visible;
			Quad_CurrentTitleLogo.ImageUrl = TitleInfo.LogoURL;
			Quad_CurrentTitleBackground.ImageUrl = TitleInfo.BackgroundURL;

			Label_CurrentTitlePunchline.Visible = TitleInfo.Punchline != "";

			if (Label_CurrentTitlePunchline.Visible) {
				WindowTotalHeight += 1.;
				Label_CurrentTitlePunchline.RelativePosition_V3.Y = -WindowTotalHeight;
				Label_CurrentTitlePunchline.Value = TL::Compose("$<$%1$>  $<%2$>  $<$%1$>", TL::ColorToText(G_ColorPrimary), TitleInfo.Punchline);
				WindowTotalHeight += Label_CurrentTitlePunchline.ComputeHeight(Label_CurrentTitlePunchline.Value) + 1.;
			}

			Label_CurrentTitleDetails.Visible = DetailsLabelValue != "";

			if (Label_CurrentTitleDetails.Visible) {
				WindowTotalHeight += 2.;
				Label_CurrentTitleDetails.RelativePosition_V3.Y = -WindowTotalHeight;
				Label_CurrentTitleDetails.Value = DetailsLabelValue;
				WindowTotalHeight += 2.;
			}

			Label_CurrentTitleDescription.Visible = TitleInfo.Description != "";

			if (Label_CurrentTitleDescription.Visible) {
				WindowTotalHeight += 1.5;
				Label_CurrentTitleDescription.RelativePosition_V3.Y = -WindowTotalHeight;
				Label_CurrentTitleDescription.Value = TL::GetTranslatedText(TitleInfo.Description);
				WindowTotalHeight += Label_CurrentTitleDescription.ComputeHeight(Label_CurrentTitleDescription.Value) + 2.;
			}

			Frame_CurrentTitleLinks.Visible = TitleURLs.count > 0;

			if (Frame_CurrentTitleLinks.Visible) {
				Frame_CurrentTitleLinks.RelativePosition_V3.Y = -WindowTotalHeight;
				Frame_TitleLinksOffset.RelativePosition_V3.X = (TitleURLs.count - 1) * -3.5;
				WindowTotalHeight += 10.;

				foreach (I => Label in Label_TitleLinkIcons) {
					Label.Visible = TitleURLs.existskey(I);
					if (!Label.Visible) continue;

					declare Text ExternalURL for Label;
					ExternalURL = TitleURLs[I];

					if (ExternalURL == TitleInfo.TitlePageURL) {
						Label.Value = "";
					} else if (ExternalURL == TitleInfo.DownloadURL) {
						Label.Value = "";
					} else if (TL::Find("facebook.com/", ExternalURL, True, True)) {
						Label.Value = "";
					} else if (TL::Find("youtube.com/", ExternalURL, True, True)) {
						Label.Value = "";
					} else if (TL::Find("twitter.com/", ExternalURL, True, True)) {
						Label.Value = "";
					} else if (TL::Find("forum.maniaplanet.com", ExternalURL, True, True)) {
						Label.Value = "";
					} else if (TL::Find("blog.maniaplanet.com", ExternalURL, True, True)) {
						Label.Value = "";
					} else if (TL::Find("doc.maniaplanet.com", ExternalURL, True, True)) {
						Label.Value = "";
					} else {
						Label.Value = "";
					}

					Label.TextColor = <1., 1., 1.>;
					Label.RelativeScale = 1.;
				}
			}

			WindowTotalHeight = ML::Max(WindowTotalHeight + 4., Quad_CurrentTitleBackground.Size.Y + 1.);

			Quad_WindowBackgroundColor.Size.Y = WindowTotalHeight + Quad_WindowBackgroundColor.RelativePosition_V3.Y;
			Quad_WindowBackgroundShadow.Size.Y = WindowTotalHeight * 2. + 10.;
			Quad_WindowFooterMetalBar.RelativePosition_V3.Y = -WindowTotalHeight;

			Frame_CurrentTitleMaker.Visible = TitleInfo.TitleMakerUID != "";

			if (Frame_CurrentTitleMaker.Visible) {
				Label_CurrentTitleMaker.Value = TitleInfo.TitleMakerName;
			}

			WindowHideOffset = WindowTotalHeight + 10.;
		}
	}
}

--></script>
</manialink>
	""";
}

Void Build() {
	G_ManialinkLayerTitleLinks = UILayerCreate();
	G_ManialinkLayerTitleLinks.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerTitleLinks.ManialinkPage = Private_CreateManialink();
}

Void SetVisibility(Boolean _IsVisible) {
	declare Boolean LayerTitleLinks_IsVisible for G_ManialinkLayerTitleLinks.LocalPage;
	LayerTitleLinks_IsVisible = _IsVisible;
}

Void SetPosition(Real _PositionX) {
	LayerCustomEvent(G_ManialinkLayerTitleLinks, "SetPosition", [TL::ToText(_PositionX)]);
}

Void SetColors(Vec3 _Primary, Vec3 _Secondary) {
	LayerCustomEvent(G_ManialinkLayerTitleLinks, "SetColors", [_Primary.X^" "^_Primary.Y^" "^_Primary.Z, _Secondary.X^" "^_Secondary.Y^" "^_Secondary.Z]);
}

Void SetTitleInfo(Structs::STitleOnlineInfo _TitleInfo) {
	declare Structs::STitleOnlineInfo LayerTitleLinks_TitleInfo for G_ManialinkLayerTitleLinks.LocalPage;
	LayerTitleLinks_TitleInfo = _TitleInfo;
	LayerCustomEvent(G_ManialinkLayerTitleLinks, "UpdateTitleInfo", []);
}
