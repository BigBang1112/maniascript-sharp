
#Include "Libs/Nadeo/System/Links/TitleInfoStructs.Script.txt" as Structs

declare Integer G_LibTitleInfo_PrevTitleInfoUpdate;

Structs::STitleOnlineInfo GetCurTitleInfo() {
	declare Structs::STitleOnlineInfo LibLinksController_CurTitleInfo for ManiaPlanet;
	return LibLinksController_CurTitleInfo;
}

Boolean PendingUpdate() {
	declare Integer LibLinksController_LastTitleInfoUpdate for ManiaPlanet;
	if (G_LibTitleInfo_PrevTitleInfoUpdate == LibLinksController_LastTitleInfoUpdate) return False;
	G_LibTitleInfo_PrevTitleInfoUpdate = LibLinksController_LastTitleInfoUpdate;
	return True;
}
