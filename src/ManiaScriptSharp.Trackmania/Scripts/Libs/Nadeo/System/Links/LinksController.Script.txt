
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Links/TitleInfoStructs.Script.txt"	as Structs
#Include "Libs/Nadeo/System/Links/LinksAPI.Script.txt"			as LinksAPI
#Include "Libs/Nadeo/System/Links/Layer_Widget.Script.txt"		as LayerWidget
#Include "Libs/Nadeo/System/Links/Layer_TitleLinks.Script.txt"	as LayerTitleLinks

#Include "Libs/Nadeo/System/Styles/ColorScheme.Script.txt" as ColorScheme

#Const C_LibLinksController_WidgetPositionX		113.

declare Boolean G_LibLinksController_LinksMenuIsVisible;
declare Integer G_ColorScheme_PreviousChangeCount;

Void SetVisibility(Boolean _IsVisible) {
	G_LibLinksController_LinksMenuIsVisible = _IsVisible;
	ToolBarForceShow = _IsVisible;

	LayerTitleLinks::SetVisibility(_IsVisible);
}

Void UpdateLayerTitleLinks() {
	declare TitleInfo = LinksAPI::GetCurTitleInfo();
	declare IsAvailable = True; // TitleInfo.UID != "";

	LayerWidget::SetAvailability(IsAvailable);
	LayerTitleLinks::SetTitleInfo(TitleInfo);

	if (!IsAvailable) {
		SetVisibility(False);
	}
}

Void Load() {
	LinksAPI::Load();
	LayerWidget::Build();
	LayerTitleLinks::Build();

	yield;

	LayerWidget::SetPosition(C_LibLinksController_WidgetPositionX);
	LayerTitleLinks::SetPosition(C_LibLinksController_WidgetPositionX);

	SetVisibility(False);
	UpdateLayerTitleLinks();
}

Void Loop() {

	if(	PendingEvents.count == 0 &&
			Http.PendingEvents.count == 0)
			return;
				
	LinksAPI::Loop();

	foreach (Event in PendingEvents) {
		if (Event.Type == CManiaAppEvent::EType::LayerCustomEvent) {
			switch (Event.CustomEventType) {
				case "ToggleLinksMenu" : {
					SetVisibility(!G_LibLinksController_LinksMenuIsVisible);
				}
			}
		}
	}

	if (LinksAPI::PendingUpdate()) {
		UpdateLayerTitleLinks();
	}
	declare ColorScheme_LatestChangeCount = ColorScheme::GetLatestChangeCount();
	if (G_ColorScheme_PreviousChangeCount != ColorScheme_LatestChangeCount) {
		declare PrimaryColor	= ColorScheme::GetPrimaryColor();
		declare SecondaryColor	= ColorScheme::GetSecondaryColor();
		
		LayerWidget::SetColor(PrimaryColor);
		LayerTitleLinks::SetColors(PrimaryColor, SecondaryColor);
        G_ColorScheme_PreviousChangeCount = ColorScheme_LatestChangeCount;
	}
}
