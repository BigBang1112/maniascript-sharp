
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Styles/CommonStyles.Script.txt" as Styles

declare CUILayer G_ManialinkLayerWidget;

Text Private_CreateManialink() {
	return """
<manialink version="3" name="Links:Widget">
<stylesheet>
	{{{Styles::Get()}}}
</stylesheet>

<frame id="Frame_LinksWidget" pos="104 90" hidden="1">
	<label id="Label_LinksMenuIcon" class="Text" pos="0 -3.5" textsize="2.5" halign="center" text="" opacity=".5" />
	<quad id="Quad_ToggleLinksMenu" halign="center" size="6 6" ScriptEvents="1" hidden="1" />
</frame>

<script><!--

#Include "TextLib" as TL

main() {
	declare Frame_LinksWidget		<=> (Page.GetFirstChild("Frame_LinksWidget")	as CMlFrame);
	declare Label_LinksMenuIcon		<=> (Page.GetFirstChild("Label_LinksMenuIcon")	as CMlLabel);
	declare Quad_ToggleLinksMenu	<=> (Page.GetFirstChild("Quad_ToggleLinksMenu")	as CMlQuad);

	while (True) {
		yield;

		foreach (Event in PendingEvents) {
			if (Event.Type == CMlScriptEvent::Type::MouseClick) {
				if (Event.ControlId == "Quad_ToggleLinksMenu") {
					SendCustomEvent("ToggleLinksMenu", []);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
				switch (Event.CustomEventType) {
					case "SetPosition" : {
						if (!Event.CustomEventData.existskey(0)) continue;
						Frame_LinksWidget.RelativePosition_V3.X = TL::ToReal(Event.CustomEventData[0]);
						Frame_LinksWidget.Visible = True;
					}
					case "SetColor" : {
						if (!Event.CustomEventData.existskey(0)) continue;
						declare SplitColor = TL::Split(" ", Event.CustomEventData[0]);
						declare Color = <1., 1., 1.>;

						if (SplitColor.count != 3) continue;
						for (I, 0, 2) Color[I] = TL::ToReal(SplitColor[I]) * .75;

						Quad_ToggleLinksMenu.BgColorFocus = Color;
					}
					case "SetAvailability" : {
						if (!Event.CustomEventData.existskey(0)) continue;
						Quad_ToggleLinksMenu.Visible = Event.CustomEventData[0] == "True";
						Label_LinksMenuIcon.Opacity = 1.;

						if (!Quad_ToggleLinksMenu.Visible) {
							Label_LinksMenuIcon.Opacity *= .5;
						}
					}
				}
			}
		}
	}
}

--></script>
</manialink>
	""";
}

Void Build() {
	G_ManialinkLayerWidget = UILayerCreate();
	G_ManialinkLayerWidget.Type = CUILayer::EUILayerType::ManiaplanetPlugin;
	G_ManialinkLayerWidget.ManialinkPage = Private_CreateManialink();
}

Void SetPosition(Real _PositionX) {
	LayerCustomEvent(G_ManialinkLayerWidget, "SetPosition", [TL::ToText(_PositionX)]);
}

Void SetColor(Vec3 _Color) {
	LayerCustomEvent(G_ManialinkLayerWidget, "SetColor", [_Color.X^" "^_Color.Y^" "^_Color.Z]);
}

Void SetAvailability(Boolean _Available) {
	LayerCustomEvent(G_ManialinkLayerWidget, "SetAvailability", [TL::ToText(_Available)]);
}
