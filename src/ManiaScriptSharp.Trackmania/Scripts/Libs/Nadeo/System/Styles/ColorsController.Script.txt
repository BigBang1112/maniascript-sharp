
#Include "ColorLib" as CL
#Include "Libs/Nadeo/System/Links/TitleInfo.Script.txt" as TitleInfo

#Const C_LibColorsController_DefaultColorScheme [
	<.25, .75, 1.>, <0., .5, 1.>
]

#Const C_LibColorsController_PredefniedColorSchemes [
	"Darkness"		=> [<.8, .9, 1.>,		<0., 0., 0.>],
	"Ignorance"		=> [<1., .25, .5>,		<1., 0., .375>],
	"Stadium"		=> [<.25, 1., .75>,		<0., 1., .5>],
	"Storm"			=> [<1., .5, .25>,		<1., .25, 0.>],
	"Toxines"		=> [<.75, 1., .25>,		<0., .25, 0.>],
	"Obsidian"		=> [<.5, .25, 1.>,		<.1, 0., .25>],
	"Iceberg"		=> [<.75, 1., 1.>,		<.5, 1., 1.>],
	"Frosted"		=> [<.25, .75, 1.>,		<0., .1, .25>]
]

#Const C_LibColorsController_ContextWithoutColor [
	CManiaPlanet::EContext::MenuStartUp,
	CManiaPlanet::EContext::MenuManiaPlanet,
	CManiaPlanet::EContext::Unknown
]

#Const C_LibColorsController_Pattern [28, 96, 70, 70, 142, 37, 123, 132]

declare Vec3	G_LibColorsController_PrimaryColor;
declare Vec3	G_LibColorsController_SecondaryColor;
declare Boolean	G_LibColorsController_CurTitleHasColor;
declare Vec3	G_LibColorsController_CurTitleColor;
declare Text	G_LibColorsController_PrevColorScheme;
declare Boolean	G_LibColorsController_PrevUseTitleColor;
declare Vec3	G_LibColorsController_PrevTrailsColor;
declare Integer	G_LibColorsController_Streak;
declare Integer	G_LibColorsController_SettingsLatestChangeCount;

declare CManiaPlanet::EContext G_LibColorsController_PrevActiveContext;

Void Private_UpdateCurrentColors() {

	declare Text SelectedScheme;
	declare Boolean UseTitleColor;
	if (ManiaPlanet.PersistentIsReady) {
		declare persistent Text		Persistent_ColorsController_SelectedScheme	for ManiaPlanet;
		declare persistent Boolean	Persistent_ColorsController_UseTitleColor	for ManiaPlanet = True;
		SelectedScheme = Persistent_ColorsController_SelectedScheme;
		UseTitleColor = Persistent_ColorsController_UseTitleColor;
	}

	if (G_LibColorsController_CurTitleHasColor && UseTitleColor) {
		declare TitleColorPrimary = CL::RgbToHsv(G_LibColorsController_CurTitleColor);

		G_LibColorsController_PrimaryColor		= CL::HsvToRgb(<TitleColorPrimary.X, TitleColorPrimary.Y, 1.>);
		G_LibColorsController_SecondaryColor	= G_LibColorsController_CurTitleColor;

		for (I, 0, 2) G_LibColorsController_SecondaryColor[I] *= .75;
	}

	else if (SelectedScheme == "Trails") {
		G_LibColorsController_PrimaryColor		= LocalUser.Color;
		G_LibColorsController_SecondaryColor	= LocalUser.Color;

		for (I, 0, 2) G_LibColorsController_SecondaryColor[I] *= .75;
	}

	else if (C_LibColorsController_PredefniedColorSchemes.existskey(SelectedScheme)) {
		declare ColorScheme = C_LibColorsController_PredefniedColorSchemes[SelectedScheme];

		G_LibColorsController_PrimaryColor		= ColorScheme[0];
		G_LibColorsController_SecondaryColor	= ColorScheme[1];
	}

	else {
		G_LibColorsController_PrimaryColor		= C_LibColorsController_DefaultColorScheme[0];
		G_LibColorsController_SecondaryColor	= C_LibColorsController_DefaultColorScheme[1];
	}

	declare Vec3 LibColorsController_PrimaryColor	for ManiaPlanet;
	declare Vec3 LibColorsController_SecondaryColor	for ManiaPlanet;

	LibColorsController_PrimaryColor	= G_LibColorsController_PrimaryColor;
	LibColorsController_SecondaryColor	= G_LibColorsController_SecondaryColor;
    
	declare Integer LibColorsController_LatestChangeCount for ManiaPlanet;
    LibColorsController_LatestChangeCount += 1;
    
}

Void Private_UpdateTitleColor() {
	declare TitleInfo = TitleInfo::GetCurTitleInfo();

	G_LibColorsController_CurTitleHasColor	= !C_LibColorsController_ContextWithoutColor.exists(ManiaPlanet.ActiveContext) && TitleInfo.ColorPrimary != <0., 0., 0.> && TitleInfo.ColorPrimary != <0., 8/15., 1.>;
	G_LibColorsController_CurTitleColor		= TitleInfo.ColorPrimary;

	Private_UpdateCurrentColors();
}

Vec3 GetPrimaryColor() {
	return G_LibColorsController_PrimaryColor;
}

Vec3 GetSecondaryColor() {
	return G_LibColorsController_SecondaryColor;
}

Void SetColorScheme(Text _ColorSchemeName) {
	if (!ManiaPlanet.PersistentIsReady) return;
	declare persistent Text Persistent_ColorsController_SelectedScheme for ManiaPlanet;
	Persistent_ColorsController_SelectedScheme = _ColorSchemeName;
	Private_UpdateCurrentColors();
}

Void SetUseTitleColor(Boolean _UseTitleColor) {
	if (!ManiaPlanet.PersistentIsReady) return;
	declare persistent Boolean Persistent_ColorsController_UseTitleColor for ManiaPlanet;
	Persistent_ColorsController_UseTitleColor = _UseTitleColor;
	Private_UpdateCurrentColors();
}

Void Load() {
	Private_UpdateCurrentColors();


	declare Text[] LibColorsController_ColorSchemes for ManiaPlanet;
	LibColorsController_ColorSchemes.clear();

	foreach (Key => Colors in C_LibColorsController_PredefniedColorSchemes) {
		LibColorsController_ColorSchemes.add(Key);
	}

	if (ManiaPlanet.PersistentIsReady) {
		declare persistent Text		Persistent_ColorsController_SelectedScheme	for ManiaPlanet;
		declare persistent Boolean	Persistent_ColorsController_UseTitleColor	for ManiaPlanet = True;
		G_LibColorsController_PrevColorScheme	= Persistent_ColorsController_SelectedScheme;
		G_LibColorsController_PrevUseTitleColor	= Persistent_ColorsController_UseTitleColor;
	}

	declare Boolean LibColorsController_SecretEnabled for ManiaPlanet;
	LibColorsController_SecretEnabled = False;
	G_LibColorsController_Streak = 0;
}

Void Loop() {

	declare Integer ColorsController_SettingsLatestChangeCount for ManiaPlanet;
	if(	PendingEvents.count == 0 && 
        G_LibColorsController_SettingsLatestChangeCount == ColorsController_SettingsLatestChangeCount)
			return;
            
    G_LibColorsController_SettingsLatestChangeCount = ColorsController_SettingsLatestChangeCount;      
				
	if (TitleInfo::PendingUpdate()) {
		Private_UpdateTitleColor();
	}

	if (G_LibColorsController_PrevActiveContext != ManiaPlanet.ActiveContext) {
		G_LibColorsController_PrevActiveContext = ManiaPlanet.ActiveContext;
		Private_UpdateTitleColor();
	}

	if (G_LibColorsController_PrevTrailsColor != LocalUser.Color) {
		G_LibColorsController_PrevTrailsColor = LocalUser.Color;
		Private_UpdateCurrentColors();
	}

	if (ManiaPlanet.PersistentIsReady) {
		declare persistent Text		Persistent_ColorsController_SelectedScheme	for ManiaPlanet;
		declare persistent Boolean	Persistent_ColorsController_UseTitleColor	for ManiaPlanet = True;

		if (G_LibColorsController_PrevColorScheme != Persistent_ColorsController_SelectedScheme) {
			G_LibColorsController_PrevColorScheme = Persistent_ColorsController_SelectedScheme;
			Private_UpdateCurrentColors();
		}

		if (G_LibColorsController_PrevUseTitleColor != Persistent_ColorsController_UseTitleColor) {
			G_LibColorsController_PrevUseTitleColor = Persistent_ColorsController_UseTitleColor;
			Private_UpdateCurrentColors();
		}
	}

	// Don't look
	foreach (Event in PendingEvents) {
		if (Event.Type == CManiaAppEvent::EType::KeyPress) {
			if (G_LibColorsController_Streak < 0) continue;

			declare Target = C_LibColorsController_Pattern[G_LibColorsController_Streak];

			if (Event.KeyCode != Target) {
				G_LibColorsController_Streak = 0;
				continue;
			}

			G_LibColorsController_Streak += 1;

			declare Boolean LibColorsController_SecretEnabled for ManiaPlanet;
			LibColorsController_SecretEnabled = G_LibColorsController_Streak == C_LibColorsController_Pattern.count;

			if (LibColorsController_SecretEnabled) {
				G_LibColorsController_Streak = -1;
			}
		}
	}
}
