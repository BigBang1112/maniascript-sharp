		Text GetExternalReportText() {
			if (G_CurrentCategory == {{{C_Category_Game}}}) {

				return _("Report on the forum");

		Void UpdateButtons() {
			
			declare CMlFrame	Frame_Buttons_Mouse			<=> (Page.GetFirstChild("buttons-mouse")		as CMlFrame);
			declare CMlFrame	Frame_Buttons_Pad			<=> (Page.GetFirstChild("inputs-pad")			as CMlFrame);
			
			declare CMlQuad 	Button_PrevPage 			<=> (Page.GetFirstChild("button-playerlist-prevpage") as CMlQuad);
			declare CMlQuad 	Button_NextPage 			<=> (Page.GetFirstChild("button-playerlist-nextpage") as CMlQuad);
			declare CMlQuad 	Button_PrevPage_Pad 		<=> (Page.GetFirstChild("button-playerlist-prevpage-pad") as CMlQuad);
			declare CMlQuad 	Button_NextPage_Pad 		<=> (Page.GetFirstChild("button-playerlist-nextpage-pad") as CMlQuad);
			
			if (G_CurPadModel == "") {
				Frame_Buttons_Mouse	.Show();
				Frame_Buttons_Pad	.Hide();
				Button_PrevPage_Pad	.Hide();
				Button_NextPage_Pad	.Hide();
			
				declare CMlLabel	Button_Send					<=> (Page.GetFirstChild("button-send")			as CMlLabel);
				declare CMlLabel	Button_ExternalReport		<=> (Page.GetFirstChild("button-externalreport")as CMlLabel);
			
				Button_ExternalReport.Value = GetExternalReportText();
			
				switch (G_CurrentStep) {
					case {{{C_Step_PickType}}} : {
						Button_Send				.Hide();
						Button_ExternalReport	.Show();
					}
					case {{{C_Step_PickPlayer}}} : {
						Button_Send				.Hide();
						Button_ExternalReport	.Show();
						
						if (G_PlayersList_CurrentIndex < {{{C_PlayersPerPage}}})
							Button_PrevPage.Hide();
						else
							Button_PrevPage.Show();
							
						if (G_PlayersList_CurrentIndex / {{{C_PlayersPerPage}}} == (G_PlayerIds.count - 1) / {{{C_PlayersPerPage}}})
							Button_NextPage.Hide();
						else
							Button_NextPage.Show();
					}
					case {{{C_Step_PickReason}}} : {
						Button_Send				.Hide();
						Button_ExternalReport	.Show();
					}
					case {{{C_Step_Send}}} : {
						if (G_RequestSent || G_SendStatus == {{{C_SendStatus_RequestSucceeded}}})
							Button_Send.Hide();
						else
							Button_Send.Show();
						if (G_SendStatus == {{{C_SendStatus_RequestNotSent}}})
							Button_ExternalReport.Show();
						else
							Button_ExternalReport.Hide();
					}
				}
			} else {
				Frame_Buttons_Mouse	.Hide();
				Frame_Buttons_Pad	.Show();
				Button_PrevPage		.Hide();
				Button_NextPage		.Hide();
				
				declare CMlFrame	Frame_Buttons_Pad_0			<=> (Page.GetFirstChild("input-pad-0")			as CMlFrame);
				declare CMlFrame	Frame_Buttons_Pad_1			<=> (Page.GetFirstChild("input-pad-1")			as CMlFrame);
				declare CMlFrame	Frame_Buttons_Pad_2			<=> (Page.GetFirstChild("input-pad-2")			as CMlFrame);
			
				declare Text Icon_Select;
				declare Text Icon_Cancel;
				declare Text Icon_Additionnal;
				if (G_CurPadModel == "PS4") {
					Icon_Select 	= "{{{C_ButtonsPath}}}PS4/Cross.dds";
					Icon_Cancel 	= "{{{C_ButtonsPath}}}PS4/Circle.dds";
					Icon_Additionnal= "{{{C_ButtonsPath}}}PS4/Triangle.dds";
					
					Button_PrevPage_Pad.ImageUrl = "{{{C_ButtonsPath}}}PS4/L1.dds";
					Button_NextPage_Pad.ImageUrl = "{{{C_ButtonsPath}}}PS4/R1.dds";
				} else {
					Icon_Select 	= "{{{C_ButtonsPath}}}XB1/A.dds";
					Icon_Cancel 	= "{{{C_ButtonsPath}}}XB1/B.dds";
					Icon_Additionnal= "{{{C_ButtonsPath}}}XB1/Y.dds";
					
					Button_PrevPage_Pad.ImageUrl = "{{{C_ButtonsPath}}}XB1/LB.dds";
					Button_NextPage_Pad.ImageUrl = "{{{C_ButtonsPath}}}XB1/RB.dds";
				}
				
			
				switch (G_CurrentStep) {
					case {{{C_Step_PickType}}} : {
						PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_0, Icon_Cancel		, _("|Infinitive|Quit"));
						PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_1, Icon_Select		, _("|Infinitive|Select"));
						PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_2, Icon_Additionnal	, GetExternalReportText());
					}
					case {{{C_Step_PickPlayer}}} : {
						PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_0, Icon_Cancel		, _("|Infinitive|Cancel"));
						PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_1, Icon_Select		, _("|Infinitive|Select"));
						PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_2, Icon_Additionnal	, GetExternalReportText());
						
						if (G_PlayersList_CurrentIndex < {{{C_PlayersPerPage}}})
							Button_PrevPage_Pad.Hide();
						else
							Button_PrevPage_Pad.Show();
							
						if (G_PlayersList_CurrentIndex / {{{C_PlayersPerPage}}} == (G_PlayerIds.count - 1) / {{{C_PlayersPerPage}}})
							Button_NextPage_Pad.Hide();
						else
							Button_NextPage_Pad.Show();
					}
					case {{{C_Step_PickReason}}} : {
						PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_0, Icon_Cancel		, _("|Infinitive|Cancel"));
						PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_1, Icon_Select		, _("|Infinitive|Select"));
						PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_2, Icon_Additionnal	, GetExternalReportText());
					}
					case {{{C_Step_Send}}} : {
					
						if (G_RequestSent || G_SendStatus == {{{C_SendStatus_RequestSucceeded}}})
							Frame_Buttons_Pad_1.Hide();
						else
							PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_1, Icon_Select		, _("Send"));
						if (G_SendStatus == {{{C_SendStatus_RequestNotSent}}}) {
							PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_2, Icon_Additionnal	, GetExternalReportText());
							PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_0, Icon_Cancel		, _("|Infinitive|Cancel"));
						} else {
							PadInputCard_SetTextAndIcon(Frame_Buttons_Pad_0, Icon_Cancel		, _("|Infinitive|Quit"));
							Frame_Buttons_Pad_2.Hide();
						}
					}
				}
			}
		}

		Void PlayersList_RefreshNames() {
			declare CMlFrame Frame_PlayersList <=> (Page.GetFirstChild("frame-player-list") as CMlFrame);
			
			declare Integer CurrentPage = G_PlayersList_CurrentIndex / {{{C_PlayersPerPage}}};
			for (I, 0, {{{C_PlayersPerPage}}} - 1) {
				declare Integer PlayerIndex = CurrentPage * {{{C_PlayersPerPage}}} + I;
			
				declare CMlFrame Frame_PlayerCard 	<=> (Frame_PlayersList.GetFirstChild("frame-player-"^I) as CMlFrame);
				if (PlayerIndex < G_PlayerNames.count) {
					Frame_PlayerCard.Show();
					declare CMlLabel Label_PlayerName	<=> (Frame_PlayerCard.GetFirstChild("label-name") as CMlLabel);
					Label_PlayerName.Value = G_PlayerNames[PlayerIndex];
				} else {
					Frame_PlayerCard.Hide();
				}
			}
			
			UpdateButtons();
		}
		
		Void PlayersList_RefreshSelection() {
			declare CMlFrame Frame_PlayersList <=> (Page.GetFirstChild("frame-player-list") as CMlFrame);
			
			declare Integer CurrentPage = G_PlayersList_CurrentIndex / {{{C_PlayersPerPage}}};
			for (I, 0, {{{C_PlayersPerPage}}} - 1) {
				declare Integer PlayerIndex = CurrentPage * {{{C_PlayersPerPage}}} + I;
			
				declare CMlFrame	Frame_PlayerCard 	<=> (Frame_PlayersList.GetFirstChild("frame-player-"^I) as CMlFrame);
				declare CMlQuad		Quad_Bg				<=> (Frame_PlayerCard.GetFirstChild("quad-card-bg")	as CMlQuad);
				if (PlayerIndex == G_PlayersList_CurrentIndex)
					Quad_Bg.Colorize = {{{C_Color_Selected}}};
				else
					Quad_Bg.Colorize = {{{C_Color_Default}}};
			}
		}

		Void PlayersList_Show() {
			declare CMlFrame Frame_PlayersList <=> (Page.GetFirstChild("frame-player-list") as CMlFrame);
			Frame_PlayersList.Show();
			
			declare Integer CurrentPage = (G_PlayersList_CurrentIndex - 1) / {{{C_PlayersPerPage}}};
			PlayersList_RefreshNames();
			PlayersList_RefreshSelection();
			
			declare CMlFrame Frame_PickedPlayer <=> (Page.GetFirstChild("frame-pickedplayer") as CMlFrame);
			Frame_PickedPlayer.Hide();
		}

		Void SetStep(Integer Step) {
			G_CurrentStep = Step;
			
			declare CMlFrame 	Frame_PickPlayer	<=> (Page.GetFirstChild("frame-pick-player") 	as CMlFrame);
			declare CMlFrame 	Frame_Subcat 		<=> (Page.GetFirstChild("frame-subcat") 		as CMlFrame);
			declare CMlFrame	Frame_Result		<=> (Page.GetFirstChild("frame-result")			as CMlFrame);
			declare CMlFrame	Frame_CreateReport	<=> (Page.GetFirstChild("frame-createreport")	as CMlFrame);
			
			
			switch (G_CurrentStep) {
				case {{{C_Step_PickType}}} : {
					declare Frame_Cat_Player 	<=> (Page.GetFirstChild("frame-cat-{{{C_Category_Player}}}") as CMlFrame);
					declare Quad_Bg_Player		<=> (Frame_Cat_Player.GetFirstChild("quad-card-bg") as CMlQuad);
					if (G_PlayerIds.count == 0) {
						Quad_Bg_Player.Colorize = {{{C_Color_GrayedOut}}};
						Quad_Bg_Player.ScriptEvents_Disable();
						if (G_CurrentCategory == {{{C_Category_Player}}})
							G_CurrentCategory = {{{C_Category_Game}}};
					} else {
						Quad_Bg_Player.Colorize = {{{C_Color_Default}}};
						Quad_Bg_Player.ScriptEvents_Restore();
					}
					
					Frame_CreateReport	.Show();
					Frame_PickPlayer	.Hide();
					Frame_Subcat		.Hide();
					Frame_Result		.Hide();
				}
				case {{{C_Step_PickPlayer}}} : {
					Frame_CreateReport	.Show();
					Frame_PickPlayer	.Show();
					Frame_Subcat		.Hide();
					Frame_Result		.Hide();
					
					//PlayersList_Show(CUser[]);
					if (G_PlayersList_CurrentIndex == -1)
						G_PlayersList_CurrentIndex = 0;
					PlayersList_Show();
				}
				case {{{C_Step_PickReason}}} : {
					Frame_CreateReport	.Show();
					Frame_Subcat		.Show();
					
					declare CMlFrame Frame_Subcat_Game 		<=> (Page.GetFirstChild("frame-subcat-game") as CMlFrame);
					declare CMlFrame Frame_Subcat_Player 	<=> (Page.GetFirstChild("frame-subcat-player") as CMlFrame);
					declare CMlFrame Frame_Subcat_Content 	<=> (Page.GetFirstChild("frame-subcat-content") as CMlFrame);
					
					G_CurrentSubCategory = 0;
					
					Frame_Subcat_Game	.Hide();
					Frame_Subcat_Player	.Hide();
					Frame_Subcat_Content.Hide();
					Frame_Result		.Hide();
					Frame_PickPlayer	.Hide();
					
					switch (G_CurrentCategory) {
						case {{{C_Category_Game}}} 		: Frame_Subcat_Game		.Show();
						case {{{C_Category_Player}}} 	: { Frame_Subcat_Player	.Show(); Frame_PickPlayer.Show(); }
						case {{{C_Category_Content}}} 	: Frame_Subcat_Content	.Show();
					}
					
					if (G_CurrentCategory == {{{C_Category_Player}}}) {
						Frame_Subcat.RelativePosition_V3.Y = -12.;
					} else {
						Frame_Subcat.RelativePosition_V3.Y = 2.;
					}
					UpdateSubcategories();
				}
				case {{{C_Step_Send}}} : {
					UpdateSendStatus({{{C_SendStatus_RequestNotSent}}});
				}
			}
			
			UpdateButtons();
		}

		Void ChangeCategory(Integer _Diff) {
			declare Boolean IsPlayerCatAvailable = (G_PlayerIds.count != 0);
			
			G_CurrentCategory += _Diff;
			if (G_CurrentCategory == {{{C_Category_Player}}} && !IsPlayerCatAvailable) 
				G_CurrentCategory += _Diff;
			G_CurrentCategory = ML::Clamp(G_CurrentCategory, {{{C_Category_Game}}}, {{{C_Category_Content}}});
		
			UpdateCategories();
		}
		
		Void ChangePlayerPage(Integer _Diff) {
			if (_Diff == -1 && G_PlayersList_CurrentIndex < {{{C_PlayersPerPage}}})
				return;
			if (_Diff == 1 && (G_PlayersList_CurrentIndex / {{{C_PlayersPerPage}}} == (G_PlayerIds.count - 1) / {{{C_PlayersPerPage}}}))
				return;
		
			G_PlayersList_CurrentIndex = ML::Clamp(G_PlayersList_CurrentIndex + _Diff * {{{C_PlayersPerPage}}}, 0, G_PlayerIds.count - 1);
			PlayersList_RefreshNames();
			PlayersList_RefreshSelection();
		}
		
		Void ChangePlayer(Integer _DiffX, Integer _DiffY) {
			declare IndexInPage = G_PlayersList_CurrentIndex % {{{C_PlayersPerPage}}};
			
			declare Integer CurLine		= IndexInPage / {{{C_PlayerList_Columns}}};
			declare Integer CurColumn	= IndexInPage % {{{C_PlayerList_Columns}}};
			CurLine 	= ML::Clamp(CurLine 	- _DiffY, 0, {{{C_PlayerList_Lines}}} - 1);
			if (CurColumn == {{{C_PlayerList_Columns}}} - 1 && _DiffX == 1)
				ChangePlayerPage(1);
			if (CurColumn == 0 && _DiffX == -1)
				ChangePlayerPage(-1);
				
			CurColumn	= ML::Clamp(CurColumn	+ _DiffX, 0, {{{C_PlayerList_Columns}}} - 1);
			
			G_PlayersList_CurrentIndex = ML::Clamp((G_PlayersList_CurrentIndex / {{{C_PlayersPerPage}}}) * {{{C_PlayersPerPage}}} + CurLine * {{{C_PlayerList_Columns}}} + CurColumn, 0, G_PlayerIds.count - 1);
			
			PlayersList_RefreshSelection();
		}
		
		Void ChangeSubcategory(Integer _DiffX, Integer _DiffY) {
			declare Integer Line	= G_CurrentSubCategory / 3;	
			declare Integer Column 	= G_CurrentSubCategory % 3;
		
		
		
			declare Integer NewIndex = G_CurrentSubCategory;
			if (NewIndex > 2 && _DiffY == 1)
				NewIndex -= 3;
			if (NewIndex <= 2 && _DiffY == -1)
				NewIndex += 3;
			if (NewIndex%3 > 0 && _DiffX == -1)
				NewIndex -= 1;
			if (NewIndex%3 < 2 && _DiffX == 1)
				NewIndex += 1;
				
			switch (G_CurrentCategory) {
				case {{{C_Category_Game}}} 		: NewIndex = ML::Clamp(NewIndex, {{{C_Subcat_Game_First}}}		, {{{C_Subcat_Game_Last}}});
				case {{{C_Category_Player}}}	: NewIndex = ML::Clamp(NewIndex, {{{C_Subcat_Player_First}}}	, {{{C_Subcat_Player_Last}}});
				case {{{C_Category_Content}}} 	: NewIndex = ML::Clamp(NewIndex, {{{C_Subcat_Content_First}}}	, {{{C_Subcat_Content_Last}}});
			}
		
			if (G_CurrentSubCategory != NewIndex) {
				G_CurrentSubCategory = NewIndex;
				UpdateSubcategories();
			}
		}
		
		Void ToggleScreenshot() {
			G_SendScreenshot = !G_SendScreenshot;
			UpdateScreenshot();
		}
		
		Void Step_Next() {
			switch (G_CurrentStep) {
				case {{{C_Step_PickType}}} : {
					if (G_CurrentCategory == {{{C_Category_Player}}}) {
						G_PlayersList_CurrentIndex = 0;
						SetStep({{{C_Step_PickPlayer}}});
					} else
						SetStep({{{C_Step_PickReason}}});
				}
				case {{{C_Step_PickPlayer}}} : {
					PlayersList_ShowPickedPlayer(G_PlayerNames[G_PlayersList_CurrentIndex]);
					SetStep({{{C_Step_PickReason}}});
				}
				case {{{C_Step_PickReason}}} : {
					SetStep({{{C_Step_Send}}});
				}
				case {{{C_Step_Send}}} : {
					if (G_SendStatus != {{{C_SendStatus_RequestSucceeded}}})
						SendReport();
				}
			}
		}
		
		Void OnCardClicked(CMlControl ClickedControl) {
			declare Text Id = ClickedControl.Parent.ControlId;
			declare Text[] SplitControlId = TL::Split("-", Id);
			switch (SplitControlId[1]) {
				case "pickedplayer": {				
					SetStep({{{C_Step_PickPlayer}}}); //on réouvre la liste des joueurs
				}
				case "player" : {
					declare Integer Page = G_PlayersList_CurrentIndex / {{{C_PlayersPerPage}}};
					declare Integer Index = TL::ToInteger(SplitControlId[2]);
					G_PlayersList_CurrentIndex = Page * {{{C_PlayersPerPage}}} + Index;
					Step_Next();
				}
				case "subcat" : {
					declare Integer Index = TL::ToInteger(SplitControlId[2]);
					if (Index != G_CurrentSubCategory) {
						G_CurrentSubCategory = Index;
						UpdateSubcategories();
					}
					G_CurrentStep = {{{C_Step_PickReason}}};
					Step_Next();
				}
				case "cat" : {
					declare Integer Index = TL::ToInteger(SplitControlId[2]);
					if (Index != G_CurrentCategory) {
						G_CurrentCategory = Index;
						UpdateCategories();
					}
					G_CurrentStep = {{{C_Step_PickType}}};
					Step_Next();
				}
			}
		}
		
		Void Cancel() {
			if (G_CurrentStep == {{{C_Step_PickType}}} || (G_CurrentStep == {{{C_Step_Send}}} && G_SendStatus != {{{C_SendStatus_RequestNotSent}}}))
				Quit();
			else if (G_CurrentStep == {{{C_Step_PickReason}}} && G_CurrentCategory != {{{C_Category_Player}}})
				SetStep({{{C_Step_PickType}}});
			else
				SetStep(G_CurrentStep - 1);
		}
		
		Void OnDirectionalButton(Integer X, Integer Y) {
			switch (G_CurrentStep) {
				case {{{C_Step_PickType}}} : {
					ChangeCategory(X);
				}
				case {{{C_Step_PickPlayer}}} : {
					ChangePlayer(X, Y);
				}
				case {{{C_Step_PickReason}}} : {
					ChangeSubcategory(X, Y);
				}
			}
		}

						} else if (Event.CustomEventType == "ReportError") {
							UpdateSendStatus({{{C_SendStatus_RequestFailed}}});
						} else if (Event.CustomEventType == "ShowGamepadButtons") {
							declare Text PadModel = Event.CustomEventData[0];
							if (G_CurPadModel != PadModel) {
								G_CurPadModel = PadModel;
								UpdateButtons();
							}
						}
					} else if (Event.Type == CMlScriptEvent::Type::MouseClick) {
						if (Event.ControlId == "button-close") {
							Quit();
						} else if (Event.ControlId == "quad-card-bg") {
							OnCardClicked(Event.Control);
						} else if (Event.ControlId == "quad-screenshot") {
							ToggleScreenshot();
						} else if (Event.ControlId == "button-send") {
							SendReport();
						} else if (Event.ControlId == "button-externalreport") {
							QuitToWebBrowser();
						}  else if (Event.ControlId == "button-report-technical") {
							QuitToWebBrowserForum();
						}  else if (Event.ControlId == "button-report-players") {
							QuitToWebBrowserSupport();
						} else if (Event.ControlId == "button-playerlist-nextpage") {
							assert(G_CurrentStep == {{{C_Step_PickPlayer}}});
							ChangePlayerPage(1);
						} else if (Event.ControlId == "button-playerlist-prevpage") {
							assert(G_CurrentStep == {{{C_Step_PickPlayer}}});
							ChangePlayerPage(-1);
						}
					}
				}
				
				foreach (Event in Input.PendingEvents) {
					if (Event.Type == CInputEvent::EType::PadButtonPress && Frame_GlobalPageContainer.Visible) {
						switch (Event.Button) {
							case CInputEvent::EButton::A: 		Step_Next();
							case CInputEvent::EButton::B: 		Cancel();
							case CInputEvent::EButton::L1:		{
								if (G_CurrentStep == {{{C_Step_PickPlayer}}})
									ChangePlayerPage(-1);
							}
							case CInputEvent::EButton::R1:		{
								if (G_CurrentStep == {{{C_Step_PickPlayer}}})
									ChangePlayerPage(1);
							}
							case CInputEvent::EButton::Down:	OnDirectionalButton(0, -1);
							case CInputEvent::EButton::Up:		OnDirectionalButton(0, 1);
							case CInputEvent::EButton::Left:	OnDirectionalButton(-1, 0);
							case CInputEvent::EButton::Right:	OnDirectionalButton(1, 0);
							case CInputEvent::EButton::Y:		{
								if (G_CurrentStep != {{{C_Step_Send}}} || G_SendStatus == {{{C_SendStatus_RequestNotSent}}}) {
									QuitToWebBrowser();
								}
							}
						}
					}
				}
				
				yield;
			}
		}
	
	--></script>
</manialink>
	""";
}

	G_ManialinkLayerReport = UILayerCreate();
	G_ManialinkLayerReport.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerReport.ManialinkPage = Private_CreateManialink();
}

	LayerCustomEvent(G_ManialinkLayerReport, "SetVisible", [TL::ToText(_IsVisible)]);
}

	return G_ManialinkLayerReport;
