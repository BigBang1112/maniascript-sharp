/** Set gauge ratio with value cut fix
 *
 *	@param	_Gauge		The gauge to set ratio
 *	@param	_Ratio		Ratio to set
 */
Void SetGaugeRatio(CMlGauge _Gauge, Real _Ratio) {
	if (_Gauge == Null) return;
	if (_Ratio <= 0.) { _Gauge.Ratio = 0.; return; }
	if (_Ratio >= 1.) { _Gauge.Ratio = 1.; return; }
	declare GaugeCutOff = 4 / _Gauge.Size.X;
	_Gauge.Ratio = ML::Clamp(_Ratio * (1 - GaugeCutOff) + GaugeCutOff, 0., 1.);
}

Text FormatPlanetsValue(Integer _NbPlanets) {
	declare NbPlanets = TL::ToText(_NbPlanets);
	declare Length = TL::Length(NbPlanets);
	declare OutputValue = "";

	for (I, 0, Length - 1) {
		declare Char = TL::SubString(NbPlanets, Length - 1 - I, 1);
		declare IsSpace = I % 3 == 0 && I != 0;

		if (IsSpace) OutputValue = " " ^ OutputValue;
		OutputValue = Char ^ OutputValue;
	}

	return OutputValue;
}

Void FormatChannelCard(CMlFrame _Frame, SChannelLiveInfo _ChannelInfo, Integer _Order) {
	if (_Frame == Null) return;

	declare Frame_ChannelOnline			<=> (_Frame.GetFirstChild("Frame_ChannelOnline")		as CMlFrame);

	declare Quad_ChannelLogoBadge		<=> (_Frame.GetFirstChild("Quad_ChannelLogoBadge")		as CMlQuad);

	declare Quad_CurrentEpisodeImage	<=> (_Frame.GetFirstChild("Quad_CurrentEpisodeImage")	as CMlQuad);
	declare Frame_ProgramInfo			<=> (_Frame.GetFirstChild("Frame_ProgramInfo")			as CMlFrame);
	declare Label_EpisodeNumber			<=> (_Frame.GetFirstChild("Label_EpisodeNumber")		as CMlLabel);
	declare Label_EpisodeDescription	<=> (_Frame.GetFirstChild("Label_EpisodeDescription")	as CMlLabel);

	declare Label_ChannelStartTime		<=> (_Frame.GetFirstChild("Label_ChannelStartTime")		as CMlLabel);

	// Override defaults
	switch (_Order) {
		case 0 : {
			Label_ChannelDisplayName.Value = "ShootMania";
			Quad_ChannelLogoBadge.ImageUrl = "{{{C_ImgBase}}}sm.dds";
		}
		case 1 : {
			Label_ChannelDisplayName.Value = "TrackMania²";
			Quad_ChannelLogoBadge.ImageUrl = "{{{C_ImgBase}}}tm.dds";
		}
	}

	declare NowTimestamp = TL::ToInteger(Time::GetCurrent());
	declare SEpisode CurrentEpisode;
	declare SEpisode[] NextEpisodes;

	foreach (Episode in _ChannelInfo.Episodes) {
		if (NowTimestamp >= Episode.TimeSlot_EndTimestamp) continue;

		if (CurrentEpisode.Episode_InstanceId == 0) {
			CurrentEpisode = Episode;
		} else {
			NextEpisodes.add(Episode);
		}
	}

	Frame_ChannelOnline.Visible = CurrentEpisode.Episode_InstanceId != 0;

	if (Frame_ChannelOnline.Visible) {
		Quad_CurrentEpisodeImage.ImageUrl = CurrentEpisode.Program_CardImageURL;

		declare InfoTotalHeight = 20.;

		Label_ChannelNbPlayers.Value = _ChannelInfo.OnlinePlayers ^ " ";
		Label_EpisodeNumber.Value = "S" ^ TL::FormatInteger(CurrentEpisode.Episode_Season, 2) ^ " E" ^ TL::FormatInteger(CurrentEpisode.Episode_Number, 2);
		Label_EpisodeDescription.Value = CurrentEpisode.Episode_Description;
		Label_ChannelStartTime.Value = Time::FormatDate(TL::ToText(CurrentEpisode.TimeSlot_StartTimestamp), Time::EDateFormats::TimeShort);

		Label_EpisodeDescription.Visible = Label_EpisodeDescription.Value != "";
		Label_EpisodeNumber.Visible = Label_EpisodeDescription.Visible || CurrentEpisode.Episode_Season > 1 || CurrentEpisode.Episode_Number > 1;

		if (Label_EpisodeNumber.Visible) InfoTotalHeight += 7.;
		if (Label_EpisodeDescription.Visible) InfoTotalHeight += Label_EpisodeDescription.ComputeHeight(Label_EpisodeDescription.Value);

		Frame_ProgramInfo.RelativePosition_V3.Y = InfoTotalHeight / 2;
	}

	for (I, 0, 2) {
		declare Frame <=> (_Frame.GetFirstChild("Frame_UpcomingEpisode#"^I) as CMlFrame);
		Frame.Visible = NextEpisodes.existskey(I);
		if (!Frame.Visible) continue;
		FormatUpcomingProgram(Frame, NextEpisodes[I]);
	}

	for (I, 0, 1) {
		declare Frame <=> (_Frame.GetFirstChild("Frame_ActionButton#"^I) as CMlFrame);
		declare Label_ActionButtonText	<=> (Frame.GetFirstChild("Label_ActionButtonText")	as CMlLabel);
		declare Quad_ActionButtonCard	<=> (Frame.GetFirstChild("Quad_ActionButtonCard")	as CMlQuad);

		declare Text ButtonActionName for Quad_ActionButtonCard;
		declare Text ButtonActionParam for Quad_ActionButtonCard;

		switch (I) {
			case 0 : {
				Label_ActionButtonText.Value = TL::Compose("► %1", _("Play"));
				ButtonActionName = "JoinChannel";
				ButtonActionParam = _ChannelInfo.JoinLink;
			}
			case 1 : {
				Label_ActionButtonText.Value = TL::Compose(" %1", _("Schedule"));
				ButtonActionName = "OpenChannelGrid";
				ButtonActionParam = _ChannelInfo.UID;
			}
		}
	}

	declare Text ChannelJoinLink for Quad_CurrentEpisodeImage;
	ChannelJoinLink = _ChannelInfo.JoinLink;

	declare Text CardChannelId for Quad_ChannelBannerCard;
	CardChannelId = _ChannelInfo.UID;

	declare CMlFrame ParentFrame for Quad_CurrentEpisodeImage;
	ParentFrame <=> _Frame;
}

Void FocusLiveEpisode(CMlFrame _Frame) {
	if (_Frame == Null) return;

	declare Quad_CurrentEpisodeImage	<=> (_Frame.GetFirstChild("Quad_CurrentEpisodeImage")	as CMlQuad);

	declare Duration = 200;
	declare Easing = CAnimManager::EAnimManagerEasing::SineOut;

	AnimMgr.Flush(Quad_CurrentEpisodeImage);
	AnimMgr.Add(Quad_CurrentEpisodeImage, "<quad scale='1.05' />", Duration, Easing);

	Page.GetClassChildren("OpacityAnim", _Frame, True);

	foreach (Control in Page.GetClassChildren_Result) {
		AnimMgr.Flush(Control);
		AnimMgr.Add(Control, "<control opacity='1' />", Duration, Easing);
	}
}
