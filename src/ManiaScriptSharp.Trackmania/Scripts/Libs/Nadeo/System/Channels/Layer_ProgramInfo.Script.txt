				if (Event.ControlId == "Quad_FeedbackButton") {

					OpenLink(FeedbackURL, CMlScript::LinkType::ExternalBrowser);
				}
			}
		}

		if (!PageIsVisible) continue;

		if (PrevRequestStatus != LayerStoreList_RequestStatusCode) {
			PrevRequestStatus = LayerStoreList_RequestStatusCode;

			//Frame_PendingDataRequest.Visible = LayerStoreList_RequestStatusCode / 100 != 2;

			PrevColorsUpdate = LayerProgramInfo_ColorUpdate;
			PrevProgramUpdate = -1;

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = LayerProgramInfo_ColorSecondary;
				}
			}

			Page.GetClassChildren("ColorBg2", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).BgColor = LayerProgramInfo_ColorSecondary;
				}
			}

			Page.GetClassChildren("ColorFg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {

					(Control as CMlQuad).Colorize = LayerProgramInfo_ColorPrimary;
				}
			}

			Page.GetClassChildren("ColorFg2", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).BgColor = LayerProgramInfo_ColorPrimary;
				}
			}
		}

		if (PrevProgramUpdate != LayerProgramInfo_LastUpdate) {
			PrevProgramUpdate = LayerProgramInfo_LastUpdate;

			UpdateScheduleCards = True;
			Frame_ScheduleCardsList.ScrollMax.Y = ML::Max(0., LayerProgramInfo_ProgramInfo.Episodes.count * 8 - Frame_ScheduleCardsList.Size.Y);

			Label_HeaderProgramName.Value = TL::StripFormatting(LayerProgramInfo_ProgramInfo.Program_Name);
			Quad_ProgramCardImage.ImageUrl = LayerProgramInfo_ProgramInfo.Program_CardImageURL;
			Quad_ProgramBackground.ImageUrl = Quad_ProgramCardImage.ImageUrl;

			declare Text[] Attributes_Keys;
			declare Text[] Attributes_Values;

			Attributes_Keys.add(_("Created by"));
			Attributes_Values.add(LayerProgramInfo_ProgramInfo.Program_AuthorName);

			Attributes_Keys.add(_("Players last 7 days"));
			Attributes_Values.add(TL::ToText(LayerProgramInfo_ProgramInfo.Program_PlayersLast7Days));

			Attributes_Keys.add(_("Hours played last 7 days"));
			Attributes_Values.add(TL::ToText(LayerProgramInfo_ProgramInfo.Program_PlaytimeLast7Days));

			foreach (I => Frame in Frame_KeyValuePair) {
				Frame.Visible = Attributes_Keys.existskey(I);
				if (!Frame.Visible) continue;

				declare Label_Attribute	<=> (Frame.GetFirstChild("Label_Attribute")	as CMlLabel);
				declare Label_Value		<=> (Frame.GetFirstChild("Label_Value")		as CMlLabel);

				Label_Attribute	.Value = Attributes_Keys[I];
				Label_Value		.Value = Attributes_Values[I];

				Label_Attribute.TextColor = LayerProgramInfo_ColorPrimary;
			}

			Frame_FeedbackButton.Visible = Http.IsValidUrl(LayerProgramInfo_ProgramInfo.Program_FeedbackURL);

			declare Text FeedbackURL for Quad_FeedbackButton;
			FeedbackURL = LayerProgramInfo_ProgramInfo.Program_FeedbackURL;
		}

		if (PrevScrollOffsetY != Frame_ScheduleCardsList.ScrollOffset.Y) {
			PrevScrollOffsetY = Frame_ScheduleCardsList.ScrollOffset.Y;
			Frame_ScheduleListOffset.RelativePosition_V3.Y = -PrevScrollOffsetY;
			UpdateScheduleCards = True;
		}

		if (UpdateScheduleCards) {
			UpdateScheduleCards = False;

			declare Offset = ML::FloorInteger(Frame_ScheduleCardsList.ScrollOffset.Y / 8.);

			foreach (I => Frame in Frame_ScheduleCards) {
				Frame.Visible = LayerProgramInfo_ProgramInfo.Episodes.existskey(Offset + I);
				if (!Frame.Visible) continue;

				declare Episode = LayerProgramInfo_ProgramInfo.Episodes[Offset + I];

				declare Label_NameOfDay	<=> (Frame.GetFirstChild("Label_NameOfDay")	as CMlLabel);
				declare Label_EpisodeNb	<=> (Frame.GetFirstChild("Label_EpisodeNb")	as CMlLabel);
				declare Label_StartTime	<=> (Frame.GetFirstChild("Label_StartTime")	as CMlLabel);
				declare Label_EndTime	<=> (Frame.GetFirstChild("Label_EndTime")	as CMlLabel);

				declare EpisodeWeekDay = (Episode.TimeSlot_StartTimestamp % 604800 / 86400 + 3) % 7;

				switch (Episode.TimeSlot_Day) {
					case 0 : Label_NameOfDay.Value = _("Monday");
					case 1 : Label_NameOfDay.Value = _("Tuesday");
					case 2 : Label_NameOfDay.Value = _("Wednesday");
					case 3 : Label_NameOfDay.Value = _("Thursday");
					case 4 : Label_NameOfDay.Value = _("Friday");
					case 5 : Label_NameOfDay.Value = _("Saturday");
					case 6 : Label_NameOfDay.Value = _("Sunday");
				}

				declare EpisodeStartTime = Episode.TimeSlot_StartIndex * 30000;
				declare EpisodeEndTime = EpisodeStartTime + Episode.TimeSlot_Duration * 1000 / 60;

				Label_EpisodeNb.Value = "S" ^ TL::FormatInteger(Episode.Episode_Season, 2) ^ " E" ^ TL::FormatInteger(Episode.Episode_Number, 2);
				Label_StartTime.Value = TL::TimeToText(EpisodeStartTime);
				Label_EndTime.Value = TL::TimeToText(EpisodeEndTime);
			}
		}
	}
}

}

Void Build() {
	G_ManialinkLayer = UILayerCreate();
	G_ManialinkLayer.Type = CUILayer::EUILayerType::ManiaplanetMenu;
	G_ManialinkLayer.AttachId = "Maniaplanet_Store";
	G_ManialinkLayer.ManialinkPage = Private_CreateManialinkLayer();
	G_ManialinkLayer.IsVisible = False;
}

Void SetVisibility(Boolean _Visible) {
	G_ManialinkLayer.IsVisible = _Visible;
}

Void SetColors(Vec3 _Primary, Vec3 _Secondary) {
	declare Vec3 LayerProgramInfo_ColorPrimary		for G_ManialinkLayer.LocalPage;
	declare Vec3 LayerProgramInfo_ColorSecondary	for G_ManialinkLayer.LocalPage;
	declare Integer LayerProgramInfo_ColorUpdate	for G_ManialinkLayer.LocalPage;

	LayerProgramInfo_ColorPrimary = _Primary;
	LayerProgramInfo_ColorSecondary = _Secondary;
	LayerProgramInfo_ColorUpdate = Now;
}

Void SetDisplayedProgramInfo(Structs::SProgramInfo _ProgramInfo) {
	declare Structs::SProgramInfo LayerProgramInfo_ProgramInfo for G_ManialinkLayer.LocalPage;
	declare Integer LayerProgramInfo_LastUpdate for G_ManialinkLayer.LocalPage;

	LayerProgramInfo_ProgramInfo = _ProgramInfo;
	LayerProgramInfo_LastUpdate = Now;
}

Void SetRequestStatus(Integer _StatusCode) {
	declare Integer LayerStoreList_RequestStatusCode for G_ManialinkLayer.LocalPage;
	LayerStoreList_RequestStatusCode = _StatusCode;
}
