	}

	return """

	</frame>
</frame3d>

<script><!--

#Include "MathLib" as ML
#Include "TextLib" as TL

Void Nothing() {

}

main() {
	declare Vec3	LayerOptions_ColorPrimary	for Page;
	declare Vec3	LayerOptions_ColorSecondary	for Page;
	declare Integer	LayerOptions_ColorUpdate	for Page;

	declare PrevColorsUpdate = -1;
	
	for (I, 0, 5) {

		declare Quad_LargeButtonCard <=> (Frame.GetFirstChild("Quad_LargeButtonCard") as CMlQuad);

		declare Integer ActivableScriptActionId for Quad_LargeButtonCard;
		ActivableScriptActionId = I;

		switch (I) {
			case 0 : {
				Label_LargeButtonIcon.Value = "";
				Label_LargeButtonName.Value = _("Settings");
				Label_LargeButtonDesc.Value = _("Customize your profile, change audio and display settings, rebind your keys.");
			}
			case 1 : {
				Label_LargeButtonIcon.Value = "";
				Label_LargeButtonName.Value = _("Player Page");
				Label_LargeButtonDesc.Value = _("Manage your online shared content and additional account settings through the Player Page.");
			}
			case 2 : {

				Label_LargeButtonIcon.Value = "";
				Label_LargeButtonName.Value = _("Add Stations");

			}
			case 5 : {
				Label_LargeButtonIcon.Value = "";
				Label_LargeButtonName.Value = _("Edit content packs");
				Label_LargeButtonDesc.Value = _("Create and manage own content packages, including titles.");

		yield;

			if (Event.Type == CMlScriptEvent::Type::MouseClick) {
				if (Event.ControlId == "Quad_LargeButtonCard") {
					declare Integer ActivableScriptActionId for Event.Control;

					switch (ActivableScriptActionId) {
						case 0 : {
							SendCustomEvent("OpenSettings", []);
						}
						case 1 : {
							OpenLink("https://www.maniaplanet.com/account", CMlScript::LinkType::ExternalBrowser);

Void Build() {
	G_ManialinkLayer = UILayerCreate();
	G_ManialinkLayer.Type = CUILayer::EUILayerType::ManiaplanetMenu;
	G_ManialinkLayer.AttachId = "Maniaplanet_Store";
	G_ManialinkLayer.ManialinkPage = Private_CreateManialinkLayer();
	G_ManialinkLayer.IsVisible = False;
}

Void SetVisibility(Boolean _Visible) {
	G_ManialinkLayer.IsVisible = _Visible;
}

Void SetColors(Vec3 _Primary, Vec3 _Secondary) {
	declare Vec3 LayerOptions_ColorPrimary		for G_ManialinkLayer.LocalPage;
	declare Vec3 LayerOptions_ColorSecondary	for G_ManialinkLayer.LocalPage;
	declare Integer LayerOptions_ColorUpdate	for G_ManialinkLayer.LocalPage;

	LayerOptions_ColorPrimary = _Primary;
	LayerOptions_ColorSecondary = _Secondary;
	LayerOptions_ColorUpdate = Now;
}
