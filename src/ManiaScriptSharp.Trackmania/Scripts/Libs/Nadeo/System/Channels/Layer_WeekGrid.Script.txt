
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Channels/Structs.Script.txt"	as Structs
#Include "Libs/Nadeo/System/Styles/CommonStyles.Script.txt"	as Styles

#Const C_ImgBase "file://Media/Manialinks/Common/Channels/"

#Const C_LibLayerWeekGrid_GridWidth		183.5
#Const C_LibLayerWeekGrid_GridColumns	7
#Const C_LibLayerWeekGrid_GridRows		48

#Const C_LibLayerWeekGrid_ScrollBarVisibleHeight	81.5

declare CUILayer G_ManialinkLayer;

Text Private_CreateManialinkLayer() {
	declare FrameTotalSize = <C_LibLayerWeekGrid_GridWidth - (C_LibLayerWeekGrid_GridColumns - 1) * .5, 0.>;
	declare GridItemSize = <FrameTotalSize.X / C_LibLayerWeekGrid_GridColumns, 0.>;

	GridItemSize.Y = GridItemSize.X * 1.3 / 4.8;
	FrameTotalSize.Y = (GridItemSize.Y + .5) * C_LibLayerWeekGrid_GridRows;

	declare Text ChannelCardQuads;

	for (I, 0, C_LibLayerWeekGrid_GridColumns * C_LibLayerWeekGrid_GridRows - 1) {
		declare Slot = I % C_LibLayerWeekGrid_GridRows;
		declare Day = I / C_LibLayerWeekGrid_GridRows;
		declare Pos = <(GridItemSize.X + .5) * Day, (GridItemSize.Y + .5) * -Slot>;

		ChannelCardQuads ^= "<quad id='Quad_GridItem#"^I^"' class='GridItem Focus' pos='"^Pos.X^" "^Pos.Y^"' keepratio='Clip' ScriptEvents='1' /><quad class='GridBg' pos='"^Pos.X^" "^Pos.Y^"' />";
	}

	declare Text HourLabels;

	for (I, 0, C_LibLayerWeekGrid_GridRows) {
		declare Hours = (I / 2) % 24;
		declare Minutes = I % 2 * 30;
		declare Y = I * -(GridItemSize.Y + .5) + .25;
		declare Scale = 1. + (I % 2) * -.25;
		declare FormattedText = Hours ^ ":" ^ TL::FormatInteger(Minutes, 2);

		HourLabels ^= "<label class='Text' pos='0 "^Y^"' halign='right' size='10' scale='"^Scale^"' text='"^FormattedText^"' />";
	}

	declare Text DayLabels;

	for (I, 0, C_LibLayerWeekGrid_GridColumns - 1) {
		declare X = I * (GridItemSize.X + .5);
		declare Text FormattedText;

		switch (I) {
			case 0 : FormattedText = _("Monday");
			case 1 : FormattedText = _("Tuesday");
			case 2 : FormattedText = _("Wednesday");
			case 3 : FormattedText = _("Thursday");
			case 4 : FormattedText = _("Friday");
			case 5 : FormattedText = _("Saturday");
			case 6 : FormattedText = _("Sunday");
		}

		DayLabels ^= "<label class='Text' pos='"^X^"' halign='center' size='"^(GridItemSize.X - 2.)^"' text='"^FormattedText^"' />";
	}

	return """
<manialink version="3" name="Channels:WeekGrid">
<stylesheet>
	{{{Styles::Get()}}}
	<style class="GridItem" size="{{{GridItemSize.X^" "^GridItemSize.Y}}}" />
	<style class="GridBg" size="{{{GridItemSize.X^" "^GridItemSize.Y}}}" bgcolor="000" opacity=".25" />
</stylesheet>

<frame3d valign="center" halign="center" size="200 120" scale="1.6" style3d="Floating">
	<frame id="Frame_WindowFrame" pos="-100 56">
		<quad size="200 1" class="ColorFg2" />
		<quad class="ColorFg" pos="-4.5 4.5" size="104.5 5" scale="2" style="Bgs1InRace" substyle="BgButtonGlow" colorize="FFF" />
		
		<frame id="Frame_WindowHeaderBox" pos="0 -7">
			<label class="Text" pos="6 .5" halign="center" valign="center" textsize="3" text="" />
			<label id="Label_HeaderChannelName" class="Bold" pos="11 -.325" textsize="3" textprefix="$t" text="Channel schedule" translate="1" />

			<frame id="Frame_HeaderButtonBack" pos="167">
				<label class="Bold" halign="center" pos="15 -.25" size="26" textsize="2" textprefix="$t" text="{{{TL::Compose(" %1", _("Back"))}}}" />
				<quad class="Focus" valign="center" size="30 6" bgcolor="FFF2" bgcolorfocus="FFF4" scriptaction="OpenFrontpage" ScriptEvents="1" />
			</frame>

			<quad size="200 12" valign="center" bgcolor="000" opacity=".25" />
		</frame>

		<frame id="Frame_PendingDataRequest" pos="0 -13" hidden="1">
			<frame id="Frame_RequestErrorCode" pos="100 -49" hidden="1">
				<quad pos="-50 12" size="100 1" class="ColorFg2" />
				<quad class="ColorFg" pos="-54.5 16.5" size="54.5 5" scale="2" style="Bgs1InRace" substyle="BgButtonGlow" colorize="FFF" />

				<label class="Bold" pos="0 6" halign="center" textsize="2" text="Request failed" translate="1" />
				<label id="Label_RequestErrorCode" class="Text" pos="0 1" halign="center" textsize="1" text="Error code: 400" />

				<frame id="Frame_HeaderButtonBack" pos="-15 -6">
					<label class="Bold" halign="center" pos="15 -.25" size="26" textsize="2" textprefix="$t" text="{{{TL::Compose("? %1", _("Refresh"))}}}" />
					<quad class="Focus" valign="center" size="30 6" bgcolor="FFF2" bgcolorfocus="FFF4" scriptaction="OpenChannelGrid" ScriptEvents="1" />
				</frame>

				<quad pos="-50 -12" size="100.125 1" style="Bgs1InRace" substyle="BgMetalBar" />
				<quad class="ColorBg" size="100 24" halign="center" valign="center" image="{{{C_ImgBase}}}background.dds" colorize="FFF" opacity=".95" />
			</frame>

			<quad size="200 98" bgcolor="000" opacity=".5" ScriptEvents="1" />
		</frame>

		<frame pos="2 -18">
			<frame pos="6 -4">
				<frame pos="0 1.5" size="8 85" halign="right">
					<frame id="Frame_HourLabelsOffset">
						<frame pos="0 -1.5">
							{{{HourLabels}}}
						</frame>
					</frame>
				</frame>
			</frame>
			<frame pos="{{{7.5 + GridItemSize.X / 2}}} -1.5">
				<frame id="Frame_CurrentDayHighlight">
					<quad pos="0 2.75" size="{{{GridItemSize.X + 1.}}} 87.5" halign="center" style="Bgs1InRace" substyle="BgColorContour" />
					<label id="Label_CurrentDayName" class="Text" pos="0 0" size="{{{GridItemSize.X}}} 4" halign="center" textcolor="000" />
					<quad pos="0 0" size="{{{GridItemSize.X}}} 5" halign="center" valign="center" bgcolor="FFF" />
				</frame>
				{{{DayLabels}}}
			</frame>
			<frame id="Frame_ChannelGridQuads" pos="7.5 -4" size="{{{C_LibLayerWeekGrid_GridWidth}}} 81.5" scroll="1" scrollmax="0 {{{FrameTotalSize.Y - 81.5}}}" scrollgrid="0 {{{GridItemSize.Y + .5}}}" scrollgridsnap="1">
				<quad id="Quad_CurrentHourLine" size="{{{C_LibLayerWeekGrid_GridWidth}}} .5" valign="center" bgcolor="F00" />
				{{{ChannelCardQuads}}}
				<quad size="{{{C_LibLayerWeekGrid_GridWidth^" "^FrameTotalSize.Y}}}" ScriptEvents="1" />
			</frame>
			<frame id="Frame_PageScrollingBar" pos="194.5 -4">
				<quad id="Quad_ScrollBarDragHandle" class="Focus" size="3" halign="center" ScriptEvents="1" />
				<quad id="Quad_ScrollBarHandle" size="1 50" halign="center" bgcolor="FFF" />
				<quad size=".25 {{{C_LibLayerWeekGrid_ScrollBarVisibleHeight}}}" halign="center" bgcolor="FFF" opacity=".25" />
			</frame>
		</frame>

		<quad pos="0 -111" size="200.125 1" style="Bgs1InRace" substyle="BgMetalBar" />
		<quad class="ColorBg" pos="0 -1" size="200 110" image="{{{C_ImgBase}}}background.dds" colorize="FFF" opacity=".95" ScriptEvents="1" />
		<quad pos="0 -1" size="200 110" style="Bgs1" substyle="BgDialogBlur" opacity="0" />
		<quad pos="-2.5 1.5" size="410 232" scale=".5" style="Bgs1InRace" substyle="BgButtonShadow" opacity=".5" />
	</frame>
</frame3d>

<script><!--

#Include "MathLib" as ML
#Include "TextLib" as TL
#Include "TimeLib" as Time

{{{Structs::IntoManialink()}}}

declare Vec3 G_ColorPrimary;
declare Vec3 G_ColorSecondary;

Void Nothing() {

}

main() {
	declare Label_HeaderChannelName		<=> (Page.GetFirstChild("Label_HeaderChannelName")	as CMlLabel);
	declare Frame_ChannelGridQuads		<=> (Page.GetFirstChild("Frame_ChannelGridQuads")	as CMlFrame);
	declare Frame_HourLabelsOffset		<=> (Page.GetFirstChild("Frame_HourLabelsOffset")	as CMlFrame);
	declare Quad_ScrollBarHandle		<=> (Page.GetFirstChild("Quad_ScrollBarHandle")		as CMlQuad);
	declare Quad_ScrollBarDragHandle	<=> (Page.GetFirstChild("Quad_ScrollBarDragHandle")	as CMlQuad);

	declare Quad_CurrentHourLine		<=> (Page.GetFirstChild("Quad_CurrentHourLine")			as CMlQuad);
	declare Frame_CurrentDayHighlight	<=> (Page.GetFirstChild("Frame_CurrentDayHighlight")	as CMlFrame);
	declare Label_CurrentDayName		<=> (Page.GetFirstChild("Label_CurrentDayName")			as CMlLabel);

	declare CMlQuad[] Quad_GridItems;

	for (I, 0, {{{C_LibLayerWeekGrid_GridColumns * C_LibLayerWeekGrid_GridRows - 1}}}) {
		Quad_GridItems.add((Page.GetFirstChild("Quad_GridItem#"^I) as CMlQuad));
	}

	declare Frame_PendingDataRequest	<=> (Page.GetFirstChild("Frame_PendingDataRequest")	as CMlFrame);
	declare Frame_RequestErrorCode		<=> (Page.GetFirstChild("Frame_RequestErrorCode")	as CMlFrame);
	declare Label_RequestErrorCode		<=> (Page.GetFirstChild("Label_RequestErrorCode")	as CMlLabel);

	declare SGridEpisode[]	LayerWeekGrid_GridEpisodes	for Page;
	declare Text			LayerWeekGrid_ChannelUID	for Page;
	declare Integer			LayerWeekGrid_LastUpdate	for Page;

	declare Integer LayerStoreList_RequestStatusCode for Page;

	declare UpdateColorScheme = True;
	declare UpdateNavigationBox = True;

	declare Boolean	IsScrollHovered;
	declare Boolean	IsScrollDragging;
	declare Real	ScrollDragStartPositionY;
	declare Real	ScrollDragStartOffset;
	declare Boolean	PrevMouseLeftButton;

	declare PrevTitleUpdate = -1;
	declare PrevScrollOffsetY = -1.;
	declare PrevRequestStatus = -1;
	declare PrevPageIsVisible = False;

	declare CurrentDayProgress = 0.;
	declare CurrentWeekDay = 0;

	declare PrevLocalDate = -1;
	declare PrevDayProgress = -1.;
	declare PrevWeekDay = -1;

	G_ColorPrimary		= <1., 1., 1.>;
	G_ColorSecondary	= <1., 1., 1.>;

	PageAlwaysUpdateScript = True;

	while (True) {
		yield;

		{{{Styles::Sounds()}}}

		foreach (Event in PendingEvents) {
			if (Event.Type == CMlScriptEvent::Type::MouseOut) {
				if (Event.ControlId == "Quad_ScrollBarDragHandle") {
					IsScrollHovered = False;
				}
			}
			else if (Event.Type == CMlScriptEvent::Type::MouseOver) {
				if (Event.ControlId == "Quad_ScrollBarDragHandle") {
					IsScrollHovered = True;
				}
			}
			else if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
				switch (Event.CustomEventType) {
					case "SetColors" : {
						if (!Event.CustomEventData.existskey(1)) continue;

						declare SplitColorPrimary	= TL::Split(" ", Event.CustomEventData[0]);
						declare SplitColorSecondary	= TL::Split(" ", Event.CustomEventData[1]);

						if (SplitColorPrimary.count != 3 && SplitColorSecondary.count != 3) continue;

						for (I, 0, 2) {
							G_ColorPrimary		[I] = TL::ToReal(SplitColorPrimary[I]);
							G_ColorSecondary	[I] = TL::ToReal(SplitColorSecondary[I]);
						}

						UpdateColorScheme = True;
					}
				}
			}
		}

		if (PrevPageIsVisible != PageIsVisible) {
			PrevPageIsVisible = PageIsVisible;

			CurrentDayProgress = System.CurrentLocalDate % 86400 / 86400.;
			declare HalfVisibleDayPart = (Frame_ChannelGridQuads.Size.Y / 2.) / (Frame_ChannelGridQuads.ScrollMax.Y + Frame_ChannelGridQuads.Size.Y);

			Frame_ChannelGridQuads.ScrollOffset.Y = (Frame_ChannelGridQuads.ScrollMax.Y + Frame_ChannelGridQuads.Size.Y) * (CurrentDayProgress - HalfVisibleDayPart);
		}

		if (!PageIsVisible) continue;

		if (PrevLocalDate != System.CurrentLocalDate) {
			PrevLocalDate = System.CurrentLocalDate;

			CurrentDayProgress = System.CurrentLocalDate % 86400 / 86400.;
			CurrentWeekDay = (System.CurrentLocalDate % 604800 / 86400 + 3) % 7;
		}

		if (PrevDayProgress != CurrentDayProgress) {
			PrevDayProgress = CurrentDayProgress;

			Quad_CurrentHourLine.RelativePosition_V3.Y = -CurrentDayProgress * (Frame_ChannelGridQuads.Size.Y + Frame_ChannelGridQuads.ScrollMax.Y);
		}

		if (PrevWeekDay != CurrentWeekDay) {
			PrevWeekDay = CurrentWeekDay;

			Frame_CurrentDayHighlight.RelativePosition_V3.X = {{{GridItemSize.X + .5}}} * CurrentWeekDay;

			switch (CurrentWeekDay) {
				case 0 : Label_CurrentDayName.Value = _("Monday");
				case 1 : Label_CurrentDayName.Value = _("Tuesday");
				case 2 : Label_CurrentDayName.Value = _("Wednesday");
				case 3 : Label_CurrentDayName.Value = _("Thursday");
				case 4 : Label_CurrentDayName.Value = _("Friday");
				case 5 : Label_CurrentDayName.Value = _("Saturday");
				case 6 : Label_CurrentDayName.Value = _("Sunday");
			}
		}

		if (PrevRequestStatus != LayerStoreList_RequestStatusCode) {
			PrevRequestStatus = LayerStoreList_RequestStatusCode;

			//Frame_PendingDataRequest.Visible = LayerStoreList_RequestStatusCode / 100 != 2;
			Frame_PendingDataRequest.Visible = LayerStoreList_RequestStatusCode >= 400;
			Frame_RequestErrorCode.Visible = LayerStoreList_RequestStatusCode >= 400;
			Label_RequestErrorCode.Value = TL::Compose(_("Error code: %1"), TL::ToText(LayerStoreList_RequestStatusCode));
		}

		if (PrevTitleUpdate != LayerWeekGrid_LastUpdate) {
			PrevTitleUpdate = LayerWeekGrid_LastUpdate;
			UpdateNavigationBox = True;
		}

		if (UpdateColorScheme) {
			UpdateColorScheme = False;

			Page.GetClassChildren("ColorBg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = G_ColorSecondary;
				}
			}

			Page.GetClassChildren("ColorBg2", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).BgColor = G_ColorSecondary;
				}
			}

			Page.GetClassChildren("ColorFg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = G_ColorPrimary;
				}
			}

			Page.GetClassChildren("ColorFg2", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).BgColor = G_ColorPrimary;
				}
			}
		}

		if (PrevMouseLeftButton != Input.MouseLeftButton) {
			PrevMouseLeftButton = Input.MouseLeftButton;

			if (Input.MouseLeftButton) {
				if (IsScrollHovered) {
					IsScrollDragging = True;
					ScrollDragStartPositionY = Input.MousePos.Y;
					ScrollDragStartOffset = Frame_ChannelGridQuads.ScrollOffset.Y;
				}
			}

			else {
				if (IsScrollDragging) {
					IsScrollDragging = False;
				}
			}
		}

		if (Input.MouseLeftButton) {
			if (IsScrollDragging) {
				declare ScrollBarPageHeight = Frame_ChannelGridQuads.ScrollMax.Y + Frame_ChannelGridQuads.Size.Y;
				declare PositionDiff = ScrollDragStartPositionY - Input.MousePos.Y;
				declare OffsetDiff = PositionDiff * (ScrollBarPageHeight / Frame_ChannelGridQuads.Size.Y);

				Frame_ChannelGridQuads.ScrollOffset.Y = ML::Clamp(ScrollDragStartOffset + OffsetDiff, 0., ScrollBarPageHeight - Frame_ChannelGridQuads.Size.Y);
			}
		}

		if (UpdateNavigationBox) {
			UpdateNavigationBox = False;

			declare Text ChannelName;

			switch (LayerWeekGrid_ChannelUID) {
				case "trackmania" : ChannelName = "Trackmania²";
				case "shootmania" : ChannelName = "Shootmania";
				default : ChannelName = _("Channel");
			}

			//L16N: "ChannelName schedule"
			Label_HeaderChannelName.Value = TL::Compose(_("$<%1$> schedule"), ChannelName);

			foreach (Quad in Quad_GridItems) {
				Quad.Visible = False;
			}

			foreach (Episode in LayerWeekGrid_GridEpisodes) {
				declare I = Episode.Day * 48 + Episode.Start;
				declare Quad <=> Quad_GridItems[I];

				Quad.ImageUrl = Episode.ProgramGridURL;
				Quad.Size.Y = {{{GridItemSize.Y}}} * Episode.Length + .5 * (Episode.Length - 1);
				Quad.Visible = True;

				declare Integer ProgramID for Quad;
				//ProgramID = Episode.ProgramID;
			}
		}

		if (PrevScrollOffsetY != Frame_ChannelGridQuads.ScrollOffset.Y) {
			PrevScrollOffsetY = Frame_ChannelGridQuads.ScrollOffset.Y;

			declare ScrollBarPageHeight		= Frame_ChannelGridQuads.ScrollMax.Y + Frame_ChannelGridQuads.Size.Y;
			declare ScrollBarVisibleHeight	= Frame_ChannelGridQuads.Size.Y;
			declare ScrollBarCurOffset		= ML::Clamp(Frame_ChannelGridQuads.ScrollOffset.Y, 0., ScrollBarPageHeight - ScrollBarVisibleHeight);

			Quad_ScrollBarHandle.Size.Y = ScrollBarVisibleHeight * (ScrollBarVisibleHeight / ScrollBarPageHeight);
			Quad_ScrollBarHandle.RelativePosition_V3.Y = (ScrollBarCurOffset / ScrollBarPageHeight) * -ScrollBarVisibleHeight;

			Quad_ScrollBarDragHandle.Size.Y = Quad_ScrollBarHandle.Size.Y;
			Quad_ScrollBarDragHandle.RelativePosition_V3.Y = Quad_ScrollBarHandle.RelativePosition_V3.Y;
		}

		Frame_HourLabelsOffset.RelativePosition_V3.Y = Frame_ChannelGridQuads.ScrollAnimOffset.Y;
	}
}

--></script>
</manialink>
	""";
}

Void Build() {
	G_ManialinkLayer = UILayerCreate();
	G_ManialinkLayer.Type = CUILayer::EUILayerType::ManiaplanetMenu;
	G_ManialinkLayer.AttachId = "Maniaplanet_Store";
	G_ManialinkLayer.ManialinkPage = Private_CreateManialinkLayer();
	G_ManialinkLayer.IsVisible = False;
}

Void SetVisibility(Boolean _Visible) {
	G_ManialinkLayer.IsVisible = _Visible;
}

Void SetColors(Vec3 _Primary, Vec3 _Secondary) {
	LayerCustomEvent(G_ManialinkLayer, "SetColors", [_Primary.X^" "^_Primary.Y^" "^_Primary.Z, _Secondary.X^" "^_Secondary.Y^" "^_Secondary.Z]);
}

Void SetPageContents(Structs::SGridEpisode[] _GridEpisodes, Text _ChannelUID) {
	declare Structs::SGridEpisode[] LayerWeekGrid_GridEpisodes for G_ManialinkLayer.LocalPage;
	declare Integer LayerWeekGrid_LastUpdate for G_ManialinkLayer.LocalPage;
	declare Text LayerWeekGrid_ChannelUID for G_ManialinkLayer.LocalPage;

	LayerWeekGrid_GridEpisodes = _GridEpisodes;
	LayerWeekGrid_ChannelUID = _ChannelUID;
	LayerWeekGrid_LastUpdate = Now;
}

Void SetRequestStatus(Integer _StatusCode) {
	declare Integer LayerStoreList_RequestStatusCode for G_ManialinkLayer.LocalPage;
	LayerStoreList_RequestStatusCode = _StatusCode;
}
