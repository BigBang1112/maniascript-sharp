Text FormatPlanetsValue(Integer _NbPlanets) {
	declare NbPlanets = TL::ToText(_NbPlanets);
	declare Length = TL::Length(NbPlanets);
	declare OutputValue = "";

	for (I, 0, Length - 1) {
		declare Char = TL::SubString(NbPlanets, Length - 1 - I, 1);
		declare IsSpace = I % 3 == 0 && I != 0;

		if (IsSpace) OutputValue = " " ^ OutputValue;
		OutputValue = Char ^ OutputValue;
	}

	return OutputValue;
}

Boolean Title_IsInStation(Text _TitleUid) {
	return ManiaPlanet.TitleIdsInstalled.exists(_TitleUid);
}

Boolean Title_IsOnDisk(Text _TitleUid) {
	foreach(Title in ManiaPlanet.TitlesAvailable) {
		if (Title.TitleId == _TitleUid) {
			return True;
		}
	}
	return False;
}
