Text FormatPlanetsValue(Integer _NbPlanets) {
	declare NbPlanets = TL::ToText(_NbPlanets);
	declare Length = TL::Length(NbPlanets);
	declare OutputValue = "";

	for (I, 0, Length - 1) {
		declare Char = TL::SubString(NbPlanets, Length - 1 - I, 1);
		declare IsSpace = I % 3 == 0 && I != 0;

		if (IsSpace) OutputValue = " " ^ OutputValue;
		OutputValue = Char ^ OutputValue;
	}

	return OutputValue;
}

Void FormatTitleFrame(CMlFrame _Frame, STitleOnlineInfo _TitleInfo) {
	if (_Frame == Null) return;

	declare Quad_TitleInfoImage			<=> (_Frame.GetFirstChild("Quad_TitleInfoImage")		as CMlQuad);
	declare Label_TitlePackName			<=> (_Frame.GetFirstChild("Label_TitlePackName")		as CMlLabel);
	declare Label_TitlePackAuthor		<=> (_Frame.GetFirstChild("Label_TitlePackAuthor")		as CMlLabel);
	declare Label_TitlePackPlayers		<=> (_Frame.GetFirstChild("Label_TitlePackPlayers")		as CMlLabel);
	declare Quad_TitleInfoButtonCard	<=> (_Frame.GetFirstChild("Quad_TitleInfoButtonCard")	as CMlQuad);

	declare Text LayerStoreList_Result_SortingOrder for Page;

	Quad_TitleInfoImage.ImageUrl = _TitleInfo.CardURL;
	Label_TitlePackName.Value = _TitleInfo.Name;
	//Label_TitlePackPlayers.Visible = LayerStoreList_Result_SortingOrder != "lastUpdate";

	switch (LayerStoreList_Result_SortingOrder) {
		case "onlinePlayers"	: Label_TitlePackPlayers.Value = FormatPlanetsValue(_TitleInfo.PlayersNbOnline) ^ " ";
		case "lastUpdate"		: Label_TitlePackPlayers.Value = Time::FormatDelta(Time::GetCurrent(), TL::ToText(_TitleInfo.UpdatedAt), Time::EDurationFormats::Abbreviated) ^ " ";
		case "playersLast24h"	: Label_TitlePackPlayers.Value = FormatPlanetsValue(_TitleInfo.PlayersNbLast24h) ^ " ";
		case "registrations"	: Label_TitlePackPlayers.Value = FormatPlanetsValue(_TitleInfo.PlayersNbRegistered) ^ " ";
	}

	//if (LayerStoreList_Result_SortingOrder == "lastUpdate") {
	//	Label_TitlePackAuthor.Value = TL::Compose(_("Updated %1 ago"), Time::FormatDelta(Time::GetCurrent(), TL::ToText(_TitleInfo.UpdatedAt), Time::EDurationFormats::Full));
	//} else {
		Label_TitlePackAuthor.Value = TL::Compose(_("by %1"), _TitleInfo.AuthorNickName);
	//}

	declare Text TitleUID for Quad_TitleInfoButtonCard;
	TitleUID = _TitleInfo.UID;
}
