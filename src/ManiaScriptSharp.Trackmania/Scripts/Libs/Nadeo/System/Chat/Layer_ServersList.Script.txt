	}

	return """
<manialink version="3" name="Chat:ServersList">
<stylesheet>
	{{{Styles::Get()}}}

	<quad size="133 1" class="ColorFg2" />

	<quad pos="0 -1" size="2 137" style="Bgs1InRace" substyle="BgGradLeft" opacity=".125" />

	<frame pos="0 -1">
		<frame pos="6 -6">
			<quad class="btn-servers" size="5 5" />
			<label class="Bold" pos="5 -.325" textsize="3" textprefix="$t" text="Servers with friends" translate="1" />

	<frame id="Frame_ServersListLegends" pos="3 -15.625">
		<label id="Label_SortByNickname" class="Text" pos="2" size="70" textsize=".875" textprefix="$t" text="Name" translate="1" ScriptEvents="1" />
		<label id="Label_SortByGameMode" class="Text" pos="89" size="20" halign="center" textsize=".875" textprefix="$t" text="Game mode" translate="1" ScriptEvents="1" />
		<label id="Label_SortByPlayers" class="Text" pos="107" size="12" halign="center" textsize=".875" textprefix="$t" text="Players" translate="1" ScriptEvents="1" />
		<label id="Label_SortByFriends" class="Text" pos="120.5" size="10" halign="center" textsize=".875" textprefix="$t" text="Friends" translate="1" ScriptEvents="1" />
	</frame>

	<frame pos="3 -18">
		<framemodel id="FrameModel_ServerCard">
			<frame pos="0 -3.5">
				<quad id="Quad_ServerPrivate" pos="2" size="4 4" hidden="1" />
				<label id="Label_ServerNickname" class="Text" pos="2 -.125" size="70" />

				<frame pos="78">
					<quad class="SeparatorV" />
					<label id="Label_ServerModeName" class="Text" pos="11 -.125" halign="center" textsize="1.25" />
				</frame>

				<frame pos="100">
					<quad class="SeparatorV" />
					<label id="Label_ServerNbPlayers" class="Bold" pos="7.1 .125" halign="right" textsize="2" />
					<label id="Label_ServerMaxPlayers" class="Bold" pos="6.9 -1.25" halign="left" textsize=".75" opacity=".25" />
				</frame>

				<frame pos="114">
					<quad class="SeparatorV" />
					<label id="Label_ServerFriends" class="Bold" pos="6.5 -.375" halign="center" textsize="2.5" />
				</frame>
			</frame>
			<quad id="Quad_ServerListCard" class="Focus" size="127 7" bgcolor="FFF1" bgcolorfocus="FFF2" ScriptEvents="1" />
		</framemodel>
		{{{ServersCards}}}
	</frame>

	<frame id="Frame_ListPageSwitch" pos="66.5 -134">
		<label id="Label_ListCurPageNb" class="Bold" halign="center" textsize="1.25" text="1 / 1" />
		<quad id="Quad_ListPagePrev" pos="-8" class="Arrow" substyle="ArrowPrev" hidden="1" ScriptEvents="1" />
		<quad id="Quad_ListPageNext" pos="8" class="Arrow" substyle="ArrowNext" hidden="1" ScriptEvents="1" />
		<quad pos="-8" class="Arrow" substyle="ArrowDisabled" />
		<quad pos="8" class="Arrow" substyle="ArrowDisabled" />
	</frame>

	<quad pos="66.5 -69.5" size="60 60" halign="center" valign="center" image="{{{C_ImgBase}}}decoration-logo.dds" opacity=".05" />

	<quad pos="0 -138" size="133.125 1" style="Bgs1InRace" substyle="BgMetalBar" />
	<quad class="ColorBg" pos="0 -1" size="133 137" image="{{{C_ImgBase}}}background.dds" opacity=".95" ScriptEvents="1" />
	<quad pos="0 -1" size="133 137" style="Bgs1" substyle="BgDialogBlur" opacity="0" />
	<quad pos="-2.5 1.5" size="276 284" scale=".5" style="Bgs1InRace" substyle="BgButtonShadow" opacity=".5" />
</frame>
</frame>
</frame>

<script><!--

#Include "MathLib" as ML
#Include "TextLib" as TL

#Struct SServerInfo {
	Text	Login;
	Text	Name;
	Text	JoinLink;
	Text	GameMode;
	Text	ModeStatus;
	Text[]	Contacts;
	Integer	NbPlayers;
	Integer	NbPlayersMax;
	Boolean	IsLobby;
	Boolean	Private;
}

Void FormatServerCard(CMlFrame _Frame, SServerInfo _ServerInfo) {
	if (_Frame == Null) return;

	declare Label_ServerNickname	<=> (_Frame.GetFirstChild("Label_ServerNickname")	as CMlLabel);
	declare Label_ServerModeName	<=> (_Frame.GetFirstChild("Label_ServerModeName")	as CMlLabel);
	declare Label_ServerNbPlayers	<=> (_Frame.GetFirstChild("Label_ServerNbPlayers")	as CMlLabel);
	declare Label_ServerMaxPlayers	<=> (_Frame.GetFirstChild("Label_ServerMaxPlayers")	as CMlLabel);
	declare Label_ServerFriends		<=> (_Frame.GetFirstChild("Label_ServerFriends")	as CMlLabel);
	declare Quad_ServerPrivate		<=> (_Frame.GetFirstChild("Quad_ServerPrivate")		as CMlQuad);
	declare Quad_ServerListCard		<=> (_Frame.GetFirstChild("Quad_ServerListCard")	as CMlQuad);


	Label_ServerNickname.Value = _ServerInfo.Name;
	Label_ServerModeName.Value = _ServerInfo.GameMode;
	Label_ServerNbPlayers.Value = TL::ToText(_ServerInfo.NbPlayers);
	Label_ServerMaxPlayers.Value = "/" ^ _ServerInfo.NbPlayersMax;
	Label_ServerFriends.Value = TL::ToText(_ServerInfo.Contacts.count);
	Quad_ServerPrivate.Visible = _ServerInfo.Private;

	Label_ServerNickname.RelativePosition_V3.X = Quad_ServerPrivate.RelativePosition_V3.X;

	if (Quad_ServerPrivate.Visible) {
		Label_ServerNickname.RelativePosition_V3.X += Quad_ServerPrivate.Size.X + 1.;
	}

	declare Text ButtonServerJoinLink for Quad_ServerListCard;
	ButtonServerJoinLink = _ServerInfo.JoinLink;
}

main() {
	declare Frame_WindowClip		<=> (Page.GetFirstChild("Frame_WindowClip")			as CMlFrame);
	declare Frame_ServersListWindow	<=> (Page.GetFirstChild("Frame_ServersListWindow")	as CMlFrame);
	declare Frame_PositionVertical	<=> (Page.GetFirstChild("Frame_PositionVertical")	as CMlFrame);

	declare Label_SortByNickname	<=> (Page.GetFirstChild("Label_SortByNickname")		as CMlLabel);
	declare Label_SortByGameMode	<=> (Page.GetFirstChild("Label_SortByGameMode")		as CMlLabel);
	declare Label_SortByPlayers		<=> (Page.GetFirstChild("Label_SortByPlayers")		as CMlLabel);
	declare Label_SortByFriends		<=> (Page.GetFirstChild("Label_SortByFriends")		as CMlLabel);

	declare Frame_ListPageSwitch	<=> (Page.GetFirstChild("Frame_ListPageSwitch")		as CMlFrame);
	declare Label_ListCurPageNb		<=> (Page.GetFirstChild("Label_ListCurPageNb")		as CMlLabel);
	declare Quad_ListPagePrev		<=> (Page.GetFirstChild("Quad_ListPagePrev")		as CMlQuad);
	declare Quad_ListPageNext		<=> (Page.GetFirstChild("Quad_ListPageNext")		as CMlQuad);

	declare CMlFrame[] Frame_ServerCards;

	for (I, 0, {{{C_NbServersDisplayedInList - 1}}}) {
		Frame_ServerCards.add((Page.GetFirstChild("Frame_ServerCard#"^I) as CMlFrame));
	}

	declare persistent Persistent_ChatMP4_ServersSortingMethod for Page = 0;

	declare Boolean			LayerServersList_IsVisible			for Page;
	declare Real			LayerServersList_PositionOffset		for Page;
	declare Boolean			LayerServersList_IsExtended			for Page;
	declare SServerInfo[]	LayerServersList_ServersList		for Page;
	declare Vec3			LayerServersList_BackgroundColor	for Page;
	declare Vec3			LayerServersList_HighlightsColor	for Page;

	declare UpdateColorScheme = True;
	declare UpdateServersList = True;
	declare HideLayerAfter = -1;

	declare ServersListCurPage = 0;
	declare ServersListNbPages = 0;

	declare PrevLayerIsVisible = False;
	declare PrevLayerIsExtended = False;
	declare PrevPositionOffset = -1.;

	while (True) {
		yield;

		foreach (Event in PendingEvents) {
			if (Event.Type == CMlScriptEvent::Type::MouseClick) {
				if (Event.ControlId == "Quad_ServerListCard") {
					declare Text ButtonServerJoinLink for Event.Control;
					OpenLink(ButtonServerJoinLink, CMlScript::LinkType::ManialinkBrowser);
				}

				else if (Event.Control == Quad_ListPagePrev) {
					if (ServersListCurPage <= 0) continue;
					ServersListCurPage -= 1;
					UpdateServersList = True;
				}

				else if (Event.Control == Quad_ListPageNext) {
					if (ServersListCurPage >= ServersListNbPages - 1) continue;
					ServersListCurPage += 1;
					UpdateServersList = True;
				}

				else if (Event.Control == Label_SortByNickname) {
					Persistent_ChatMP4_ServersSortingMethod = 2;
					UpdateServersList = True;
				}

				else if (Event.Control == Label_SortByGameMode) {
					Persistent_ChatMP4_ServersSortingMethod = 3;
					UpdateServersList = True;
				}

				else if (Event.Control == Label_SortByPlayers) {
					Persistent_ChatMP4_ServersSortingMethod = 0;
					UpdateServersList = True;
				}

				else if (Event.Control == Label_SortByFriends) {
					Persistent_ChatMP4_ServersSortingMethod = 1;
					UpdateServersList = True;
				}

				if (Event.Control.HasClass("Focus")) {
					Audio.PlaySoundEvent(CAudioManager::ELibSound::Valid, 1, 0.);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::MouseOver) {
				if (Event.Control.HasClass("Focus")) {
					Audio.PlaySoundEvent(CAudioManager::ELibSound::Focus, 1, 0.);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
				switch (Event.CustomEventType) {
					case "UpdateServersList" : UpdateServersList = True;
					case "UpdateColorScheme" : UpdateColorScheme = True;
				}
			}
		}

		if (PrevLayerIsVisible != LayerServersList_IsVisible) {
			PrevLayerIsVisible = LayerServersList_IsVisible;

			AnimMgr.Flush(Frame_PositionVertical);

			declare Pos = <Frame_PositionVertical.RelativePosition_V3.X, 0.>;
			declare Easing = CAnimManager::EAnimManagerEasing::SineOut;

			if (!LayerServersList_IsVisible) {
				Pos.Y += 150.;
				Easing = CAnimManager::EAnimManagerEasing::SineIn;
				HideLayerAfter = Now + 200;
			} else {
				HideLayerAfter = -1;
			}

			AnimMgr.Add(Frame_PositionVertical, "<frame pos='"^Pos.X^" "^Pos.Y^"' />", 200, Easing);
		}

		if (HideLayerAfter > 0 && Now >= HideLayerAfter) {
			HideLayerAfter = -1;
			SendCustomEvent("HideLayerServersList", []);
		}

		if (!LayerServersList_IsVisible) continue;

		if (PrevPositionOffset != LayerServersList_PositionOffset) {
			PrevPositionOffset = LayerServersList_PositionOffset;

			Frame_WindowClip.RelativePosition_V3.X = LayerServersList_PositionOffset;
		}

		if (PrevLayerIsExtended != LayerServersList_IsExtended) {
			PrevLayerIsExtended = LayerServersList_IsExtended;

			AnimMgr.Flush(Frame_ServersListWindow);

			declare Pos = <0., -6.>;
			declare Easing = CAnimManager::EAnimManagerEasing::SineOut;

			if (!LayerServersList_IsExtended) {
				Pos.X -= 140.;
				Easing = CAnimManager::EAnimManagerEasing::SineIn;
			}

			AnimMgr.Add(Frame_ServersListWindow, "<frame pos='"^Pos.X^" "^Pos.Y^"' />", 125, Easing);
		}

		Frame_ServersListWindow.Visible = Frame_ServersListWindow.RelativePosition_V3.X > -140.;
		if (!LayerServersList_IsExtended || !Frame_ServersListWindow.Visible) continue;

		if (UpdateColorScheme) {
			UpdateColorScheme = False;
			UpdateServersList = True;

			Page.GetClassChildren("ColorBg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = LayerServersList_BackgroundColor;
				}
			}

			Page.GetClassChildren("ColorFg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = LayerServersList_HighlightsColor;
				}
			}

			Page.GetClassChildren("ColorFg2", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).BgColor = LayerServersList_HighlightsColor;
				}
			}
		}

		if (UpdateServersList) {
			UpdateServersList = False;

			// Biggest hackathon in this lib so far
			foreach (I => Label in [Label_SortByPlayers, Label_SortByFriends, Label_SortByNickname, Label_SortByGameMode]) {
				if (I == Persistent_ChatMP4_ServersSortingMethod) {
					Label.TextColor = LayerServersList_HighlightsColor;
				} else {
					Label.TextColor = <1., 1., 1.>;
				}
			}

			declare SServerInfo[Text] ServersList_ByLogin;

			foreach (ServerInfo in LayerServersList_ServersList) {
				ServersList_ByLogin[ServerInfo.Login] = ServerInfo;
			}

			declare SServerInfo[]	SortedServersList;
			declare Integer[Text]	SortedServersList_Count;
			declare Text[Text]		SortedServersList_Text;

			foreach (Login => ServerInfo in ServersList_ByLogin) {
				switch (Persistent_ChatMP4_ServersSortingMethod) {
					case 0 : SortedServersList_Count[Login]	= -ServerInfo.NbPlayers;
					case 1 : SortedServersList_Count[Login]	= -ServerInfo.Contacts.count;
					case 2 : SortedServersList_Text[Login]	= TL::StripFormatting(ServerInfo.Name);
					case 3 : SortedServersList_Text[Login]	= TL::StripFormatting(ServerInfo.GameMode);
				}
			}

			SortedServersList_Count = SortedServersList_Count.sort();
			SortedServersList_Text = SortedServersList_Text.sort();

			foreach (Login => Count in SortedServersList_Count) {
				SortedServersList.add(ServersList_ByLogin[Login]);
			}

			foreach (Login => Value in SortedServersList_Text) {
				SortedServersList.add(ServersList_ByLogin[Login]);
			}

			// Pagination
			ServersListNbPages = (SortedServersList.count - 1) / Frame_ServerCards.count + 1;
			ServersListCurPage = ML::Clamp(ServersListCurPage, 0, ServersListNbPages - 1);

			Frame_ListPageSwitch.Visible = ServersListNbPages > 1;
			Quad_ListPagePrev.Visible = ServersListCurPage > 0;
			Quad_ListPageNext.Visible = ServersListCurPage < ServersListNbPages - 1;
			Label_ListCurPageNb.Value = (ServersListCurPage + 1) ^ " / " ^ ServersListNbPages;

			declare Offset = ServersListCurPage * Frame_ServerCards.count;

			// Do ze frames
			foreach (I => Frame in Frame_ServerCards) {
				Frame.Visible = SortedServersList.existskey(Offset + I);
				if (!Frame.Visible) continue;
				FormatServerCard(Frame, SortedServersList[Offset + I]);
			}
		}
	}
}
--></script>
</manialink>
	""";
}

Void Build() {
	G_ManialinkLayerServersList = UILayerCreate();
	G_ManialinkLayerServersList.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerServersList.ManialinkPage = Private_CreateManialink();
	G_ManialinkLayerServersList.IsVisible = False;
}

Void SetVisibility(Boolean _IsVisible) {
	declare Boolean LayerServersList_IsVisible for G_ManialinkLayerServersList.LocalPage;
	LayerServersList_IsVisible = _IsVisible;

	if (_IsVisible) {
		G_ManialinkLayerServersList.IsVisible = True;
	}
}

Void HideLayer() {
	G_ManialinkLayerServersList.IsVisible = False;
}

Void SetIsExtended(Boolean _IsExtended) {
	declare Boolean LayerServersList_IsExtended for G_ManialinkLayerServersList.LocalPage;
	LayerServersList_IsExtended = _IsExtended;
}

Void SetWindowOffset(Real _Offset) {
	declare Real LayerServersList_PositionOffset for G_ManialinkLayerServersList.LocalPage;
	LayerServersList_PositionOffset = _Offset;
}

Void SetColorScheme(Vec3 _Background, Vec3 _Highlights) {
	declare Vec3 LayerServersList_BackgroundColor for G_ManialinkLayerServersList.LocalPage;
	declare Vec3 LayerServersList_HighlightsColor for G_ManialinkLayerServersList.LocalPage;
	LayerServersList_BackgroundColor = _Background;
	LayerServersList_HighlightsColor = _Highlights;
	LayerCustomEvent(G_ManialinkLayerServersList, "UpdateColorScheme", []);
}

Void UpdateServersList(ChatServersInfo::SServerInfo[] _ServersInfo) {
	declare ChatServersInfo::SServerInfo[] LayerServersList_ServersList for G_ManialinkLayerServersList.LocalPage;
	LayerServersList_ServersList = _ServersInfo;

	LayerCustomEvent(G_ManialinkLayerServersList, "UpdateServersList", []);
}
