
#Include "TextLib" as TL
#Include "Libs/Nadeo/System/Styles/CommonStyles.Script.txt" as Styles

#Const C_ImgBase "file://Media/Manialinks/Common/Chat/"

declare CUILayer G_ManialinkLayerModals;

Text Private_CreateManialink() {
	return """
<manialink version="3" name="Chat:Modals">
<stylesheet>
	{{{Styles::Get()}}}
</stylesheet>

<frame id="Frame_ModalWindow" z-index="10" hidden="1">
	<quad pos="-60" size="120 1" class="ColorFg2" bgcolor="0F0" />
	<quad class="ColorFg" pos="-64.5 4.5" size="64.5 5" scale="2" style="Bgs1InRace" substyle="BgButtonGlow" colorize="0F0" />

	<quad id="Quad_ModalImage" halign="center" size="18 18" />
	<label id="Label_ModalTitle" class="Bold" halign="center" valign="top" size="116" textsize="3" autonewline="1" />
	<label id="Label_ModalText" class="Text" halign="center" valign="top" size="116" autonewline="1" />

	<frame id="Frame_Transfer" hidden="1">
		<entry id="Entry_TransferPlanets" pos="-4 -2.655" class="Bold" halign="right" size="30 5" focusareacolor1="0000" focusareacolor2="FFF2" textsize="2.5" default="100" ScriptEvents="1" />
		<quad pos="0 -2.5" halign="center" valign="center" size="5 5" style="ManiaPlanetLogos" substyle="IconPlanets" />
		<label id="Label_TextPlanets" class="Bold" pos="4 -2.75" textsize="2" textprefix="$t" text="Planets" translate="1" />

		<frame>
			<entry id="Entry_TransferTitle" pos="0 -8" class="Text" halign="center" size="68 4" focusareacolor1="0000" focusareacolor2="FFF2" textsize="2" default="Planets transfer" ScriptEvents="1" />
			<label class="Text" pos="0 -11.5" halign="center" size="70" textsize=".5" textprefix="$t" text="Transaction label" translate="1" />
		</frame>
	</frame>

	<frame id="Frame_ModalActions">
		<framemodel id="FrameModel_ModalButton">
			<label id="Label_ModalButtonText" class="Bold" pos="20 -3.75" halign="center" valign="center2" size="36" textsize="1.75" textprefix="$t" />
			<quad id="Quad_ModalButtonCard" class="Focus" size="40 7" bgcolor="FFF2" bgcolorfocus="FFF4" ScriptEvents="1" />
		</framemodel>

		<frameinstance modelid="FrameModel_ModalButton" id="Frame_ModalButton#0" hidden="1" />
		<frameinstance modelid="FrameModel_ModalButton" id="Frame_ModalButton#1" hidden="1" />
	</frame>

	<quad id="Quad_ModalBgMetal" pos="-60 -138" size="120.125 1" style="Bgs1InRace" substyle="BgMetalBar" />
	<quad id="Quad_ModalBgColor" class="ColorBg" pos="-60 -1" size="120 137" image="{{{C_ImgBase}}}background.dds" opacity=".95" ScriptEvents="1" />
	<quad id="Quad_ModalBgBlur" pos="-60 -1" size="120 137" style="Bgs1" substyle="BgModalBlur" opacity="0" />
	<quad id="Quad_ModalBgShadow" pos="-62.5 1.5" size="250 284" scale=".5" style="Bgs1InRace" substyle="BgButtonShadow" opacity=".5" />
</frame>

<quad id="Quad_ModalBgWinBlur" class="Fullscreen" z-index="5" style="Bgs1" substyle="BgDialogBlur" ScriptEvents="1" hidden="1" />

<script><!--

#Include "MathLib" as ML
#Include "TextLib" as TL

Void ShowModal(Text _Title, Text _Text, Text _ImageURL, Boolean _IsTransfer, Text[] _Actions) {
	declare Frame_ModalWindow	<=> (Page.GetFirstChild("Frame_ModalWindow")	as CMlFrame);
	declare Quad_ModalBgWinBlur	<=> (Page.GetFirstChild("Quad_ModalBgWinBlur")	as CMlQuad);
	declare Quad_ModalImage		<=> (Page.GetFirstChild("Quad_ModalImage")		as CMlQuad);
	declare Label_ModalTitle	<=> (Page.GetFirstChild("Label_ModalTitle")		as CMlLabel);
	declare Label_ModalText		<=> (Page.GetFirstChild("Label_ModalText")		as CMlLabel);
	declare Frame_Transfer		<=> (Page.GetFirstChild("Frame_Transfer")		as CMlFrame);
	declare Frame_ModalActions	<=> (Page.GetFirstChild("Frame_ModalActions")	as CMlFrame);
	declare Quad_ModalBgMetal	<=> (Page.GetFirstChild("Quad_ModalBgMetal")	as CMlQuad);
	declare Quad_ModalBgColor	<=> (Page.GetFirstChild("Quad_ModalBgColor")	as CMlQuad);
	declare Quad_ModalBgBlur	<=> (Page.GetFirstChild("Quad_ModalBgBlur")		as CMlQuad);
	declare Quad_ModalBgShadow	<=> (Page.GetFirstChild("Quad_ModalBgShadow")	as CMlQuad);

	declare FrameModalHeight = 5.;

	Quad_ModalImage.Visible = Http.IsValidUrl(_ImageURL);
	Quad_ModalImage.ImageUrl = _ImageURL;

	if (Quad_ModalImage.Visible) {
		Quad_ModalImage.RelativePosition_V3.Y = -FrameModalHeight;
		FrameModalHeight += Quad_ModalImage.Size.Y + 2.;
	}

	Label_ModalTitle.Visible = _Title != "";
	Label_ModalTitle.Value = _Title;

	if (Label_ModalTitle.Visible) {
		Label_ModalTitle.RelativePosition_V3.Y = -FrameModalHeight;
		FrameModalHeight += Label_ModalTitle.ComputeHeight(Label_ModalTitle.Value) + 1.;
	}

	Label_ModalText.Visible = _Text != "";
	Label_ModalText.Value = _Text;

	if (Label_ModalText.Visible) {
		Label_ModalText.RelativePosition_V3.Y = -FrameModalHeight;
		FrameModalHeight += Label_ModalText.ComputeHeight(Label_ModalText.Value) + 3.;
	}

	Frame_Transfer.Visible = _IsTransfer;

	if (Frame_Transfer.Visible) {
		Frame_Transfer.RelativePosition_V3.Y = -FrameModalHeight;
		FrameModalHeight += 17.;
	}

	Frame_ModalActions.Visible = _Actions.count > 0;

	if (Frame_ModalActions.Visible) {
		declare ButtonsTotalWidth = 0.;

		for (I, 0, 1) {
			declare Frame <=> (Page.GetFirstChild("Frame_ModalButton#"^I) as CMlFrame);
			Frame.Visible = _Actions.existskey(I);
			if (!Frame.Visible) continue;

			declare ActionName = _Actions[I];

			declare Label_ModalButtonText	<=> (Frame.GetFirstChild("Label_ModalButtonText")	as CMlLabel);
			declare Quad_ModalButtonCard	<=> (Frame.GetFirstChild("Quad_ModalButtonCard")	as CMlQuad);

			switch (ActionName) {
				case "ConfirmRemoveFriend"		: Label_ModalButtonText.Value = _("Remove friend");
				case "ConfirmDeleteHistory"		: Label_ModalButtonText.Value = _("Delete");
				case "ConfirmPlanetsTransfer"	: Label_ModalButtonText.Value = _("Confirm transfer");
				case "CancelModal"				: Label_ModalButtonText.Value = _("Cancel");
			}

			declare Text ButtonModalAction for Quad_ModalButtonCard;
			ButtonModalAction = ActionName;

			if (I > 0) ButtonsTotalWidth += 4.;

			Frame.RelativePosition_V3.X = ButtonsTotalWidth;
			ButtonsTotalWidth += Quad_ModalButtonCard.Size.X;
		}

		Frame_ModalActions.RelativePosition_V3 = <ButtonsTotalWidth / -2., -FrameModalHeight>;
		FrameModalHeight += 11.;
	}

	Quad_ModalBgMetal.RelativePosition_V3.Y = -FrameModalHeight;
	Quad_ModalBgColor.Size.Y = FrameModalHeight;
	Quad_ModalBgBlur.Size.Y = Quad_ModalBgColor.Size.Y;
	Quad_ModalBgShadow.Size.Y = Quad_ModalBgColor.Size.Y * 2. + 10.;
	FrameModalHeight += 1.;
	Frame_ModalWindow.RelativePosition_V3.Y = FrameModalHeight / 2.;

	Audio.PlaySoundEvent(CAudioManager::ELibSound::ShowDialog, 0, 0.);
	Frame_ModalWindow.Show();
	Quad_ModalBgWinBlur.Show();
	Page.LinksInhibited = True;
}

Void HideModal() {
	declare Frame_ModalWindow	<=> (Page.GetFirstChild("Frame_ModalWindow")	as CMlFrame);
	declare Quad_ModalBgWinBlur	<=> (Page.GetFirstChild("Quad_ModalBgWinBlur")	as CMlQuad);

	if (!Frame_ModalWindow.Visible) return;

	Audio.PlaySoundEvent(CAudioManager::ELibSound::HideDialog, 0, 0.);
	Frame_ModalWindow.Hide();
	Quad_ModalBgWinBlur.Hide();
	Page.LinksInhibited = False;
}

main() {
	declare Vec3 LayerModals_BackgroundColor for Page;
	declare Vec3 LayerModals_HighlightsColor for Page;

	declare UpdateColorScheme = True;

	while (True) {
		yield;

		foreach (Event in PendingEvents) {
			if (Event.Type == CMlScriptEvent::Type::MouseClick) {
				if (Event.ControlId == "Quad_ModalButtonCard") {
					declare Text ButtonModalAction for Event.Control;
					declare Entry_TransferPlanets	<=> (Page.GetFirstChild("Entry_TransferPlanets")	as CMlEntry);
					declare Entry_TransferTitle		<=> (Page.GetFirstChild("Entry_TransferTitle")		as CMlEntry);

					SendCustomEvent(ButtonModalAction, [Entry_TransferPlanets.Value, Entry_TransferTitle.Value]);
					HideModal();
				}

				else if (Event.ControlId == "Quad_ModalBgWinBlur") {
					HideModal();
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::MouseOver) {
				if (Event.Control.HasClass("Focus")) {
					Audio.PlaySoundEvent(CAudioManager::ELibSound::Focus, 1, 0.);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
				switch (Event.CustomEventType) {
					case "UpdateColorScheme" : UpdateColorScheme = True;
					case "ShowModal" : {
						if (!Event.CustomEventData.existskey(3)) continue;

						declare ModalTitle		= Event.CustomEventData[0];
						declare ModalText		= Event.CustomEventData[1];
						declare ModalImageURL	= Event.CustomEventData[2];
						declare ModalIsTransfer	= Event.CustomEventData[3] == "True";
						declare ModalActions	= TL::Split("|", Event.CustomEventData[4]);

						ShowModal(ModalTitle, ModalText, ModalImageURL, ModalIsTransfer, ModalActions);
					}
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::EntrySubmit) {
				if (Event.ControlId == "Entry_TransferPlanets") {
					declare Entry <=> (Event.Control as CMlEntry);
					Entry.Value = TL::ToText(ML::Clamp(TL::ToInteger(Entry.Value), 0, ManiaPlanet.AccountPlanets));
				}

				else if (Event.ControlId == "Entry_TransferTitle") {
					declare Entry <=> (Event.Control as CMlEntry);

					Entry.Value = TL::StripFormatting(Entry.Value);
					Entry.Value = TL::Replace(Entry.Value, "\n", " ");
					Entry.Value = TL::Join(" ", TL::Split(" ", Entry.Value));
					Entry.Value = TL::SubString(Entry.Value, 0, 128);
				}
			}
		}

		if (UpdateColorScheme) {
			UpdateColorScheme = False;

			Page.GetClassChildren("ColorBg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = LayerModals_BackgroundColor;
				}
			}

			Page.GetClassChildren("ColorFg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = LayerModals_HighlightsColor;
				}
			}

			Page.GetClassChildren("ColorFg2", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).BgColor = LayerModals_HighlightsColor;
				}
			}
		}
	}
}

--></script>
</manialink>
	""";
}

Void Build() {
	G_ManialinkLayerModals = UILayerCreate();
	G_ManialinkLayerModals.Type = CUILayer::EUILayerType::Normal;
	G_ManialinkLayerModals.ManialinkPage = Private_CreateManialink();
}

Void SetColorScheme(Vec3 _Background, Vec3 _Highlights) {
	declare Vec3 LayerModals_BackgroundColor for G_ManialinkLayerModals.LocalPage;
	declare Vec3 LayerModals_HighlightsColor for G_ManialinkLayerModals.LocalPage;
	LayerModals_BackgroundColor = _Background;
	LayerModals_HighlightsColor = _Highlights;
	LayerCustomEvent(G_ManialinkLayerModals, "UpdateColorScheme", Text[]);
}

Void ShowModal(Text _Title, Text _Text, Text _ImageURL, Boolean _IsTransfer, Text[] _Actions) {
	if (_Title == "" || _Text == "" || _Actions.count <= 0) return;

	LayerCustomEvent(G_ManialinkLayerModals, "ShowModal", [_Title, _Text, _ImageURL, TL::ToText(_IsTransfer), TL::Join("|", _Actions)]);
}
