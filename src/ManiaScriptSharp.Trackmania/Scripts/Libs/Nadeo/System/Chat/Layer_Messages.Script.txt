CChatRoom GetRoomFromLogin(Text _Login) {
	if (ManiaPlanet.ChatManager.Squad != Null && ManiaPlanet.ChatManager.Squad.RoomId == _Login) {
		return ManiaPlanet.ChatManager.Squad;
	}

	foreach (Room in ManiaPlanet.ChatManager.Channels) {
		if (Room.RoomId == _Login) return Room;
	}
	
	return Null;
}

Text GetRoomName(Text _Login) {
	if (TL::Find("@squad", _Login, False, False)) return _("Squad");

	switch (TL::Split("@", _Login)[0]) {
		case "trackmania" : return "TrackMania channel";
		case "shootmania" : return "ShootMania channel";
	}

	return _Login;
}

Text GetRoomIcon(Text _Login) {
	switch (TL::Split("@", _Login)[0]) {
		case "trackmania" : return "{{{C_ImgBase}}}channel-trackmania.dds";
		case "shootmania" : return "{{{C_ImgBase}}}channel-shootmania.dds";
	}
	
	return "{{{C_ImgBase}}}squad.dds";
}

Text FinalizeImageURL(Text _URL) {
	declare Suffix = TL::SubString(_URL, TL::Length(_URL) - 4, 4);
	declare URL = _URL;
	declare Suffixes = [".jpg", ".png", ".tga", ".dds", "webp", "webm", "jpeg"];

	if (!Suffixes.exists(Suffix)) URL ^= "?.jpg";

	return URL;
}

Void UpdateConversationDetails(Text _Login) {
	if (_Login == "") return;

	declare Quad_ContactPresence	<=> (Page.GetFirstChild("Quad_ContactPresence")		as CMlQuad);
	declare Quad_ContactPresence2	<=> (Page.GetFirstChild("Quad_ContactPresence2")	as CMlQuad);
	declare Quad_ContactAvatar		<=> (Page.GetFirstChild("Quad_ContactAvatar")		as CMlQuad);
	declare Frame_ContactNameStatus	<=> (Page.GetFirstChild("Frame_ContactNameStatus")	as CMlFrame);
	declare Label_ContactNickname	<=> (Page.GetFirstChild("Label_ContactNickname")	as CMlLabel);
	declare Label_ContactLogin		<=> (Page.GetFirstChild("Label_ContactLogin")		as CMlLabel);
	declare Label_ContactStatus		<=> (Page.GetFirstChild("Label_ContactStatus")		as CMlLabel);
	declare Frame_ContactServer		<=> (Page.GetFirstChild("Frame_ContactServer")		as CMlFrame);
	declare Label_ContactServerName	<=> (Page.GetFirstChild("Label_ContactServerName")	as CMlLabel);
	declare Quad_ContactJoinGame	<=> (Page.GetFirstChild("Quad_ContactJoinGame")		as CMlQuad);
	declare Quad_ContactSpecGame	<=> (Page.GetFirstChild("Quad_ContactSpecGame")		as CMlQuad);
	declare Quad_ContactIsMuted		<=> (Page.GetFirstChild("Quad_ContactIsMuted")		as CMlQuad);

	declare Contact				<=> ManiaPlanet.ChatManager.GetContactFromLogin(_Login);
	declare ChatRoom			<=> GetRoomFromLogin(_Login);
	declare ContactNickname		= _Login;
	declare ContactAvatarURL	= "{{{C_ImgBase}}}default-avatar.dds";
	declare ContactPresence		= 0;
	declare ContactStatusText	= "";
	declare ContactServerLogin	= "";
	declare ContactServerName	= "";
	declare ContactServerLink	= "";

	if (_Login == LocalUser.Login) {
		ContactNickname		= LocalUser.Name;
		ContactAvatarURL	= LocalUser.AvatarUrl;
		ContactStatusText	= ManiaPlanet.ChatManager.Presence_Status;
		ContactServerLogin	= ManiaPlanet.ChatManager.CurrentServerLogin;
		ContactServerName	= ManiaPlanet.ChatManager.CurrentServerName;
		ContactServerLink	= ManiaPlanet.ChatManager.CurrentServerJoinLink;
	}

	if (Contact != Null) {
		ContactNickname		= Contact.Name;
		ContactAvatarURL	= Contact.AvatarUrl;
		ContactStatusText	= Contact.PresenceStatus;
		ContactServerLogin	= Contact.CurrentServerLogin;
		ContactServerName	= Contact.CurrentServerName;
		ContactServerLink	= Contact.CurrentServerJoinLink;

		switch (Contact.PresenceShow) {
			case CChatContact::EPresenceShow::Available		: ContactPresence = 1;
			case CChatContact::EPresenceShow::Away			: ContactPresence = 2;
			case CChatContact::EPresenceShow::ExtendedAway	: ContactPresence = 2;
			case CChatContact::EPresenceShow::DoNotDisturb	: ContactPresence = 3;
			case CChatContact::EPresenceShow::WantToChat	: ContactPresence = 1;
			default											: ContactPresence = 0;
		}
	}

	// Chat room exists
	else if (ChatRoom != Null) {
		ContactNickname 	= GetRoomName(_Login);
		ContactAvatarURL	= GetRoomIcon(_Login);
		ContactPresence		= 1;

		if (ChatRoom.Members.count == 0) {
			ContactStatusText = _("1 member");
		} else {
			ContactStatusText = TL::Compose(_("%1 members"), TL::ToText(ChatRoom.Members.count + 1));
		}
	}

	Quad_ContactAvatar.ImageUrl = ContactAvatarURL;
	Label_ContactNickname.Value = ContactNickname;
	Label_ContactNickname.Size.X = ML::Min(Label_ContactNickname.ComputeWidth(Label_ContactNickname.Value), 50.);

	Label_ContactLogin.Visible = Contact != Null && TL::StripFormatting(ContactNickname) != _Login;

	if (Label_ContactLogin.Visible) {
		Label_ContactLogin.Value = "(" ^ _Login ^ ")";
		Label_ContactLogin.RelativePosition_V3.X = Label_ContactNickname.RelativePosition_V3.X + Label_ContactNickname.Size.X + 1.5;
	}

	// Contact presence color
	Quad_ContactPresence.Visible = ContactPresence != 0;
	Quad_ContactPresence.BgColor = G_PresenceColors[ContactPresence];
	Quad_ContactPresence2.Visible = Quad_ContactPresence.Visible;
	Quad_ContactPresence2.Colorize = Quad_ContactPresence.BgColor;

	Frame_ContactServer.Visible = ContactServerLogin != "";

	// Contact on a server
	if (ContactServerLogin != "") {
		Label_ContactServerName.Value = ContactServerName;

		foreach (Quad in [Quad_ContactJoinGame, Quad_ContactSpecGame]) {
			declare Text ButtonServerLogin for Quad;
			ButtonServerLogin = ContactServerLogin;
		}
	}

	// Contact status text
	if (ContactStatusText != "") {
		Label_ContactStatus.Value = ContactStatusText;
	}

	else switch (ContactPresence) {
		case -1 : Label_ContactStatus.Value = _("|Presence|Connecting");
		case 1 : Label_ContactStatus.Value = _("|Presence|Online");
		case 2 : Label_ContactStatus.Value = _("|Presence|Away");
		case 3 : Label_ContactStatus.Value = _("|Presence|Busy");
		default : Label_ContactStatus.Value = _("|Presence|Offline");
	}

	Label_ContactStatus.TextColor = G_PresenceColors[ContactPresence];

	// Muted icon
	declare Boolean LayerMessages_IsContactMuted for Page;
	Quad_ContactIsMuted.Visible = LayerMessages_IsContactMuted;

	// Positioning
	Frame_ContactNameStatus.RelativePosition_V3.Y = -13.;

	if (Frame_ContactServer.Visible) {
		Frame_ContactNameStatus.RelativePosition_V3.Y += 2.;
	}
}

Void AddTimeSeparator(Text _Time) {
	if (_Time == "") return;

	declare Frame <=> (Page.GetFirstChild("Frame_TimeSeparator#"^G_NbDisplayedTimeSeparators) as CMlFrame);
	if (Frame == Null) return;

	G_MessagesTotalHeight += 4.;

	declare Label_TimestampText <=> (Frame.GetFirstChild("Label_TimestampText") as CMlLabel);
	declare Quad_TimestampLineL <=> (Frame.GetFirstChild("Quad_TimestampLineL") as CMlQuad);
	declare Quad_TimestampLineR <=> (Frame.GetFirstChild("Quad_TimestampLineR") as CMlQuad);

	Frame.RelativePosition_V3.Y = G_MessagesTotalHeight;
	Label_TimestampText.Value = _Time;
	Quad_TimestampLineL.Size.X = 62.5 - (Label_TimestampText.ComputeWidth(Label_TimestampText.Value) / 2);
	Quad_TimestampLineR.Size.X = Quad_TimestampLineL.Size.X;
	Frame.Show();

	G_NbDisplayedTimeSeparators += 1;
}

Vec3 CL_IntegerToRgb(Integer _Color) {
    if (_Color <= 0) return <0., 0., 0.>;

    return <
        ((_Color / 256) / 256) / 255.,
        ((_Color / 256) % 256) / 255.,
        (_Color % 256) / 255.
    >;
}

Void FormatMessageFrame(Integer _Order, CMlFrame _Frame, CChatEntry _MessagesGroup, CChatMessage _Message, Boolean _ShowDetails) {
	if (_Frame == Null || _Frame == _MessagesGroup || _Frame == _Message) return;

	declare Frame_MessageDetails	<=> (_Frame.GetFirstChild("Frame_MessageDetails")	as CMlFrame);
	declare Label_MessageText		<=> (_Frame.GetFirstChild("Label_MessageText")		as CMlLabel);
	declare Frame_EmbedsContainer	<=> (_Frame.GetFirstChild("Frame_EmbedsContainer")	as CMlFrame);
	declare Quad_MsgRightClickArea	<=> (_Frame.GetFirstChild("Quad_MsgRightClickArea")	as CMlQuad);

	declare Boolean LayerMessages_DisplayEmbeds	for Page;
	declare Boolean LayerMessages_OutgoingLeft	for Page;
	declare Boolean LayerMessages_DisplayImages	for Page;

	declare MessageBody_Formatted = TL::Replace(_Message.Body, "$<$l[", "$<$" ^ TL::ColorToText(G_PresenceColors[1]) ^ "$l[");
	declare MessageHeight = 0.;
	declare CMlControl[] MessageAlignedContents;
	declare SChatMessageEmbed[] MessageEmbeds;

	if (_Message.Metadata != "" && _Message.Body == "") log("Works");

	if (LayerMessages_DisplayEmbeds && _Message.Metadata != "") {
		MessageEmbeds.fromjson(_Message.Metadata);
	}

	Frame_MessageDetails	.Visible = _ShowDetails;
	Label_MessageText		.Visible = MessageBody_Formatted != "";
	Frame_EmbedsContainer	.Visible = LayerMessages_DisplayEmbeds && MessageEmbeds.count > 0;

	// Format the sender info
	if (Frame_MessageDetails.Visible) {
		declare Quad_SenderAvatar		<=> (_Frame.GetFirstChild("Quad_SenderAvatar")		as CMlQuad);
		declare Label_SenderNickname	<=> (_Frame.GetFirstChild("Label_SenderNickname")	as CMlLabel);
		declare Label_MessageDate		<=> (_Frame.GetFirstChild("Label_MessageDate")		as CMlLabel);

		Quad_SenderAvatar.ImageUrl		= _MessagesGroup.AvatarUrl;
		Label_SenderNickname.Value		= _MessagesGroup.Nickname;
		Label_MessageDate.Value			= _MessagesGroup.Date;

		Label_SenderNickname.Size.X = Label_SenderNickname.ComputeWidth(Label_SenderNickname.Value);
		Label_MessageDate.RelativePosition_V3.X = ML::Abs(Label_SenderNickname.RelativePosition_V3.X) + Label_SenderNickname.Size.X + 1;

		MessageAlignedContents.add(Quad_SenderAvatar);
		MessageAlignedContents.add(Label_SenderNickname);
		MessageAlignedContents.add(Label_MessageDate);

		MessageHeight += 4.5;
	}

	// Format the message text
	if (Label_MessageText.Visible) {
		Label_MessageText.RelativePosition_V3.Y	= -MessageHeight;
		Label_MessageText.Value					= MessageBody_Formatted;

		MessageHeight += 3.5 + Label_MessageText.ComputeHeight(Label_MessageText.Value) - Label_MessageText.ComputeHeight("");

		MessageAlignedContents.add(Label_MessageText);
	}

	// Format the message embeds
	if (Frame_EmbedsContainer.Visible) {
		MessageAlignedContents.add(Frame_EmbedsContainer);

		for (I, 0, {{{C_LibLayerMessages_NbEmbedsModels - 1}}}) {
			declare Frame <=> (_Frame.GetFirstChild("Frame_MessageEmbed#"^I) as CMlFrame);
			Frame.Visible = MessageEmbeds.existskey(I);
			if (!Frame.Visible) continue;

			declare Embed = MessageEmbeds[I];
			Frame.RelativePosition_V3.Y = -MessageHeight;

			declare Quad_EmbedColor1		<=> (Frame.GetFirstChild("Quad_EmbedColor1")		as CMlQuad);
			declare Quad_EmbedColor2		<=> (Frame.GetFirstChild("Quad_EmbedColor2")		as CMlQuad);
			declare Quad_EmbedColorOff		<=> (Frame.GetFirstChild("Quad_EmbedColorOff")		as CMlQuad);
			declare Frame_EmbedContents		<=> (Frame.GetFirstChild("Frame_EmbedContents")		as CMlFrame);
			declare Label_EmbedAuthor		<=> (Frame.GetFirstChild("Label_EmbedAuthor")		as CMlLabel);
			declare Label_EmbedTitle		<=> (Frame.GetFirstChild("Label_EmbedTitle")		as CMlLabel);
			declare Label_EmbedDescription	<=> (Frame.GetFirstChild("Label_EmbedDescription")	as CMlLabel);
			declare Quad_EmbedImage			<=> (Frame.GetFirstChild("Quad_EmbedImage")			as CMlQuad);
			declare Label_EmbedProvider		<=> (Frame.GetFirstChild("Label_EmbedProvider")		as CMlLabel);
			declare Frame_PlanetsTransfer	<=> (Frame.GetFirstChild("Frame_PlanetsTransfer")	as CMlFrame);

			MessageAlignedContents.add(Frame);
			MessageAlignedContents.add(Frame_EmbedContents);
			MessageAlignedContents.add(Quad_EmbedImage);

			declare EmbedHeight = 0.;
			declare EmbedIsTransfer = Embed.Type == 8;

			declare EmbedImageURL = Embed.ImageURL;
			if (Embed.ImageProxy != "") EmbedImageURL = Embed.ImageProxy;
			
			declare EmbedTitle = Embed.Title;
			if (Http.IsValidUrl(Embed.URL)) EmbedTitle = "$<$l[" ^ Embed.URL ^ "]" ^ Embed.Title ^ "$>";

			Label_EmbedAuthor		.Visible = !EmbedIsTransfer && Embed.Author != "";
			Label_EmbedTitle		.Visible = !EmbedIsTransfer && Embed.Title != "";
			Label_EmbedDescription	.Visible = !EmbedIsTransfer && Embed.Description != "";
			Quad_EmbedImage			.Visible = !EmbedIsTransfer && LayerMessages_DisplayImages && Http.IsValidUrl(EmbedImageURL);
			Label_EmbedProvider		.Visible = !EmbedIsTransfer && Embed.Provider != "";
			Frame_PlanetsTransfer	.Visible = EmbedIsTransfer;

			foreach (Label in [Label_EmbedProvider, Label_EmbedAuthor, Label_EmbedTitle, Label_EmbedDescription]) {
				if (!Label.Visible) continue;
				if (Label != Label_EmbedDescription) EmbedHeight += .125;
				Label.RelativePosition_V3.Y = -EmbedHeight;
				
				switch (Label.ControlId) {
					case "Label_EmbedProvider"		: Label.Value = Embed.Provider;
					case "Label_EmbedAuthor"		: Label.Value = Embed.Author;
					case "Label_EmbedTitle"			: Label.Value = EmbedTitle;
					case "Label_EmbedDescription"	: Label.Value = Embed.Description;
				}

				Label.Size.Y = Label.ComputeHeight(Label.Value);
				EmbedHeight += Label.Size.Y + .5;

				MessageAlignedContents.add(Label);
			}

			if (Quad_EmbedImage.Visible) {
				Quad_EmbedImage.RelativePosition_V3.Y = -EmbedHeight;
				Quad_EmbedImage.ImageUrl = FinalizeImageURL(EmbedImageURL);
				EmbedHeight += Quad_EmbedImage.Size.Y;
			}

			if (Frame_PlanetsTransfer.Visible) {
				Frame_PlanetsTransfer.RelativePosition_V3.Y = -EmbedHeight;

				declare Label_TransferSender	<=> (Frame.GetFirstChild("Label_TransferSender")	as CMlLabel);
				declare Label_TransferTitle		<=> (Frame.GetFirstChild("Label_TransferTitle")		as CMlLabel);
				declare Label_TransferPlanets	<=> (Frame.GetFirstChild("Label_TransferPlanets")	as CMlLabel);
				declare Label_TransferUID		<=> (Frame.GetFirstChild("Label_TransferUID")		as CMlLabel);
				declare Quad_TransferIcon		<=> (Frame.GetFirstChild("Quad_TransferIcon")		as CMlQuad);
				declare Quad_TransferBackround	<=> (Frame.GetFirstChild("Quad_TransferBackround")	as CMlQuad);

				EmbedHeight += Quad_TransferBackround.Size.Y;

				declare IsOutgoing = Embed.Sender == LocalUser.Login;

				if (Embed.Recipient == LocalUser.Login) {
					Label_TransferSender.Value = TL::Compose(_("You received Planets from $<%1$>"), _MessagesGroup.Nickname);
				} else {
					Label_TransferSender.Value = _("You sent your Planets");
				}

				Label_TransferTitle.Value = Embed.Title;
				Label_TransferPlanets.Value = TL::Compose(_("%1 Planets"), TL::ToText(Embed.Planets));
				Label_TransferUID.Value = TL::Compose(_("Transaction #%1"), Embed.UID);

				MessageAlignedContents.add(Frame_PlanetsTransfer);
				MessageAlignedContents.add(Label_TransferSender);
				MessageAlignedContents.add(Label_TransferTitle);
				MessageAlignedContents.add(Label_TransferPlanets);
				MessageAlignedContents.add(Label_TransferUID);
				MessageAlignedContents.add(Quad_TransferIcon);
				MessageAlignedContents.add(Quad_TransferBackround);
			}
 
			Quad_EmbedColor1.Visible = Embed.Color != 0;
			Quad_EmbedColor2.Visible = Quad_EmbedColor1.Visible;

			if (Quad_EmbedColor1.Visible) {
				Quad_EmbedColor1.BgColor = CL_IntegerToRgb(Embed.Color);
				Quad_EmbedColor2.Colorize = Quad_EmbedColor1.BgColor;
			}

			Quad_EmbedColorOff.Size.Y = EmbedHeight;
			Quad_EmbedColor1.Size.Y = EmbedHeight;
			Quad_EmbedColor2.Size.Y = EmbedHeight + 4.5;

			if (EmbedHeight > 0) MessageHeight += 2.;
			MessageHeight += EmbedHeight;
		}
	}

	G_MessagesTotalHeight += MessageHeight;

	if (_Order == 0) G_MessageAnimationDistance = MessageHeight;

	_Frame.RelativePosition_V3.Y = G_MessagesTotalHeight;
	Quad_MsgRightClickArea.Size.Y = MessageHeight;

	// Flip outgoing messages to the right side
	declare MessageAlignDirection = CMlControl::AlignHorizontal::Left;
	_Frame.RelativePosition_V3.X = 0.;

	if (!LayerMessages_OutgoingLeft && _MessagesGroup.Direction == CChatEntry::EDirection::Sent) {
		MessageAlignDirection = CMlControl::AlignHorizontal::Right;
		_Frame.RelativePosition_V3.X = 131.;
	}

	foreach (Control in MessageAlignedContents) {
		Control.RelativePosition_V3.X	= ML::Abs(Control.RelativePosition_V3.X);
		Control.HorizontalAlign			= MessageAlignDirection;

		if (Control.HorizontalAlign == CMlControl::AlignHorizontal::Right) {
			Control.RelativePosition_V3.X *= -1.;
		}
	}

	declare Integer AreaMessageId for Quad_MsgRightClickArea;
	//AreaMessageId = _Message.MessageId;
}

Void ShowRightClickMenu_Contact(Vec2 _Position, CChatContact _Contact) {
	if (_Contact == Null) return;

	declare Boolean LayerMessages_IsContactMuted for Page;
	declare IsSelf = _Contact.Login == LocalUser.Login;
	declare Text[] Options;

	if (!IsSelf && _Contact.CurrentServerLogin != "") {
		Options.add("FriendJoinServer");
		Options.add("FriendSpectateServer");
	}

	Options.add("FriendCopyLogin");

	if (!IsSelf) {
		//Options.add("FriendSendMail");
		Options.add("FriendInviteToSquad");
		Options.add("FriendSendPlanets");

		if (LayerMessages_IsContactMuted) {
			Options.add("FriendMuteTurnOff");
		} else  {
			Options.add("FriendMuteTurnOn");
		}
	}

	//Options.add("FriendDeleteHistory");
	Options.add("FriendRemove");
	//Options.add("FriendBlock");

	ShowRightClickMenu(_Position, Options);
}

Void ShowRightClickMenu_ChatRoom(Vec2 _Position, CChatRoom _ChatRoom) {
	if (_ChatRoom == Null) return;

	declare Text[] Options;

	Options.add("FriendCopyLogin");
	Options.add("ChatRoomLeave");

	ShowRightClickMenu(_Position, Options);
}

Void ShowRightClickMenu_Common(Vec2 _Position, Text _Login) {
	declare Contact		<=> ManiaPlanet.ChatManager.GetContactFromLogin(_Login);
	declare ChatRoom	<=> GetRoomFromLogin(_Login);

	if (Contact != Null)	ShowRightClickMenu_Contact(_Position, Contact);
	if (ChatRoom != Null)	ShowRightClickMenu_ChatRoom(_Position, ChatRoom);
}

main() {
	declare Frame_WindowClip		<=> (Page.GetFirstChild("Frame_WindowClip")			as CMlFrame);
	declare Frame_PositionVertical	<=> (Page.GetFirstChild("Frame_PositionVertical")	as CMlFrame);
	declare Frame_MessagesWindow	<=> (Page.GetFirstChild("Frame_MessagesWindow")		as CMlFrame);
	declare Quad_ShowFriendOptions	<=> (Page.GetFirstChild("Quad_ShowFriendOptions")	as CMlQuad);
	declare Frame_MessagesContainer	<=> (Page.GetFirstChild("Frame_MessagesContainer")	as CMlFrame);
	declare Frame_MessagesAnim		<=> (Page.GetFirstChild("Frame_MessagesAnim")		as CMlFrame);
	declare Frame_MessagesOffset	<=> (Page.GetFirstChild("Frame_MessagesOffset")		as CMlFrame);

	declare Entry_TypeMessage		<=> (Page.GetFirstChild("Entry_TypeMessage")		as CMlEntry);

	declare Quad_ImagePreviewPic	<=> (Page.GetFirstChild("Quad_ImagePreviewPic")		as CMlQuad);
	declare Quad_ImagePreviewClose	<=> (Page.GetFirstChild("Quad_ImagePreviewClose")	as CMlQuad);
	declare Quad_ImagePreviewBg		<=> (Page.GetFirstChild("Quad_ImagePreviewBg")		as CMlQuad);

	declare Frame_ChatScrollBar		<=> (Page.GetFirstChild("Frame_ChatScrollBar")		as CMlFrame);
	declare Quad_ScrollBarHandle	<=> (Page.GetFirstChild("Quad_ScrollBarHandle")		as CMlQuad);
	declare Quad_ScrollBarArea		<=> (Page.GetFirstChild("Quad_ScrollBarArea")		as CMlQuad);

	declare CMlFrame[] Frame_MessageItems;

	for (I, 0, {{{C_LibLayerMessages_NbMessageModels - 1}}}) {
		Frame_MessageItems.add((Page.GetFirstChild("Frame_MessageItem#"^I) as CMlFrame));
	}

	declare Boolean	LayerMessages_IsVisible			for Page;
	declare Real	LayerMessages_PositionOffset	for Page;
	declare Boolean	LayerMessages_IsExtended		for Page;
	declare Ident	LayerMessages_ChatHistoryId		for Page;
	declare Text	LayerMessages_ConversationLogin	for Page;
	declare Integer LayerMessages_LastUpdate		for Page;
	declare Boolean	LayerMessages_NextUpdateAnim	for Page;
	declare Vec3	LayerMessages_BackgroundColor	for Page;
	declare Vec3	LayerMessages_HighlightsColor	for Page;
	declare Integer	LayerMessages_CurrentOffset		for Page;

	declare UpdateColorScheme = True;
	declare UpdateConversation = "";
	declare UpdateMessages = True;
	declare HideLayerAfter = -1;

	declare ConversationOffset = Integer[Text];
	declare MessagesOffset = 0;
	declare NbMessagesTotal = 0;

	declare DisplayImagePreview = False;

	declare HoveredMessageId = -1;
	declare LastHoveredMessageId = -1;

	declare HoveringUserCard = False;

	declare HoveringScrollBar = False;
	declare DraggingScrollBar = False;
	declare ScrollBarOriginOffset = 0;
	declare ScrollBarOriginPositionY = 0.;

	declare PrevLayerIsVisible = False;
	declare PrevPositionOffset = -1.;
	declare PrevLayerIsExtended = False;
	declare PrevDisplayImagePreview = False;
	declare PrevMessagesUpdate = -1;
	declare PrevIsMouseLeftButton = False;
	declare PrevIsMouseRightButton = False;
	declare PrevMessagesOffset = -1;
	declare PrevConversationLogin = "";
	declare PrevPositionVertical = -1.;
	declare PrevPositionHorizontal = -1.;
	declare PrevScrollOffset = 0.;

	while (True) {
		yield;

		if (PrevIsMouseLeftButton != Input.MouseLeftButton) {
			PrevIsMouseLeftButton = Input.MouseLeftButton;

			if (PrevIsMouseLeftButton) {
				HideRightClickMenu();

				if (HoveringScrollBar) {
					DraggingScrollBar = True;
					ScrollBarOriginOffset = ML::FloorInteger(-Frame_MessagesContainer.ScrollOffset.Y);
					ScrollBarOriginPositionY = Input.MousePos.Y;
				}
			}

			else {
				DraggingScrollBar = False;
			}
		}

		if (Input.MouseLeftButton && DraggingScrollBar) {
			declare PositionDiffY = Input.MousePos.Y - ScrollBarOriginPositionY;
			declare StepSize = (Quad_ScrollBarArea.Size.Y - Quad_ScrollBarHandle.Size.Y) / NbMessagesTotal;

			Frame_MessagesContainer.ScrollOffset.Y = -ML::ToReal(ML::Clamp(ScrollBarOriginOffset + ML::NearestInteger(PositionDiffY / StepSize), 0, NbMessagesTotal - 1));
		}

		foreach (Event in PendingEvents) {
			if (Event.Type == CMlScriptEvent::Type::MouseClick) {
				if (Event.ControlId == "Quad_EmbedImage") {
					DisplayImagePreview = True;
					Quad_ImagePreviewPic.ImageUrl = (Event.Control as CMlQuad).ImageUrl;
				}

				else if (Event.ControlId == "Quad_ImagePreviewClose") {
					DisplayImagePreview = False;
				}

				else if (Event.ControlId == "Quad_RightMenuCard") {
					declare Text ActionName for Event.Control;

					SendCustomEvent(ActionName, [G_CurConversationLogin]);
				}

				else if (Event.ControlId == "Quad_ContactJoinGame") {
					SendCustomEvent("FriendJoinServer_Info", [G_CurConversationLogin]);
					SendCustomEvent("HideServerInfo", []);
				}

				else if (Event.ControlId == "Quad_ContactSpecGame") {
					SendCustomEvent("FriendSpectateServer", [G_CurConversationLogin]);
				}

				else if (Event.ControlId == "Quad_ShowFriendOptions") {
					ShowRightClickMenu_Common(Event.Control.AbsolutePosition_V3, G_CurConversationLogin);
				}

				if (Event.Control.HasClass("Focus")) {
					Audio.PlaySoundEvent(CAudioManager::ELibSound::Valid, 1, 0.);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::MouseOut) {
				if (Event.ControlId == "Quad_MsgRightClickArea") {
					HoveredMessageId = -1;
				}

				else if (Event.ControlId == "Quad_ContactJoinGame" || Event.ControlId == "Quad_ContactSpecGame") {
					SendCustomEvent("HideServerInfo", []);
				}

				else if (Event.ControlId == "Quad_ConversationCard") {
					HoveringUserCard = False;
				}

				else if (Event.ControlId == "Quad_ScrollBarHandle") {
					HoveringScrollBar = False;
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::MouseOver) {
				if (Event.ControlId == "Quad_MsgRightClickArea") {
					declare Integer AreaMessageId for Event.Control;
					HoveredMessageId = AreaMessageId;
					LastHoveredMessageId = AreaMessageId;
				}

				else if (Event.ControlId == "Quad_ConversationCard") {
					HoveringUserCard = True;
				}

				else if (Event.ControlId == "Quad_ContactJoinGame" || Event.ControlId == "Quad_ContactSpecGame") {
					declare Text ButtonServerLogin for Event.Control;
					declare Pos = Event.Control.AbsolutePosition_V3.X^" "^Event.Control.AbsolutePosition_V3.Y;
					SendCustomEvent("ShowServerInfo", [ButtonServerLogin, Pos]);
				}

				else if (Event.ControlId == "Quad_ScrollBarHandle") {
					HoveringScrollBar = True;
				}

				if (Event.Control.HasClass("Focus")) {
					Audio.PlaySoundEvent(CAudioManager::ELibSound::Focus, 1, 0.);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::EntrySubmit) {
				if (Event.Control == Entry_TypeMessage) {
					if (Entry_TypeMessage.Value == "") continue;
					SendCustomEvent("SendMessage", [Entry_TypeMessage.Value]);
					Entry_TypeMessage.Value = "";
					Entry_TypeMessage.StartEdition();
					MessagesOffset = 0;
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
				switch (Event.CustomEventType) {
					case "UpdateContactCard" : {
						if (!Event.CustomEventData.existskey(0)) continue;
						UpdateConversation = Event.CustomEventData[0];
						G_CurConversationLogin = UpdateConversation;
					}
					case "UpdateColorScheme" : UpdateColorScheme = True;
				}
			}
		}

		if (PrevLayerIsVisible != LayerMessages_IsVisible) {
			PrevLayerIsVisible = LayerMessages_IsVisible;

			AnimMgr.Flush(Frame_PositionVertical);

			declare Pos = <Frame_PositionVertical.RelativePosition_V3.X, 0.>;
			declare Easing = CAnimManager::EAnimManagerEasing::SineOut;

			if (!LayerMessages_IsVisible) {
				Pos.Y += 150.;
				Easing = CAnimManager::EAnimManagerEasing::SineIn;
				HideRightClickMenu();
				HideLayerAfter = Now + 200;
			} else {
				HideLayerAfter = -1;
			}

			AnimMgr.Add(Frame_PositionVertical, "<frame pos='"^Pos.X^" "^Pos.Y^"' hidden='"^!LayerMessages_IsVisible^"' />", 200, Easing);
		}

		if (HideLayerAfter > 0 && Now >= HideLayerAfter) {
			HideLayerAfter = -1;
			SendCustomEvent("HideLayerMessages", []);
		}

		if (!LayerMessages_IsVisible) continue;

		if (PrevPositionOffset != LayerMessages_PositionOffset) {
			PrevPositionOffset = LayerMessages_PositionOffset;

			Frame_WindowClip.RelativePosition_V3.X = LayerMessages_PositionOffset;
		}

		if (PrevLayerIsExtended != LayerMessages_IsExtended) {
			PrevLayerIsExtended = LayerMessages_IsExtended;

			AnimMgr.Flush(Frame_MessagesWindow);

			declare Pos = <0., -6.>;
			declare Easing = CAnimManager::EAnimManagerEasing::SineOut;

			if (!LayerMessages_IsExtended) {
				Pos.X -= 140.;
				Easing = CAnimManager::EAnimManagerEasing::SineIn;
				HideRightClickMenu();
			}

			AnimMgr.Add(Frame_MessagesWindow, "<frame pos='"^Pos.X^" "^Pos.Y^"' hidden='"^!LayerMessages_IsVisible^"' />", 125, Easing);
		}

		if (PrevPositionHorizontal != Frame_MessagesWindow.RelativePosition_V3.Y) {
			PrevPositionHorizontal = Frame_MessagesWindow.RelativePosition_V3.Y;

			Frame_MessagesWindow.Visible = PrevPositionHorizontal > -139.;
		}

		Frame_MessagesWindow.Visible = Frame_MessagesWindow.RelativePosition_V3.X > -140.;
		if (!LayerMessages_IsExtended || !Frame_MessagesWindow.Visible) continue;

		if (PrevMessagesUpdate != LayerMessages_LastUpdate) {
			PrevMessagesUpdate = LayerMessages_LastUpdate;
			UpdateMessages = True;
		}

		if (UpdateColorScheme) {
			UpdateColorScheme = False;
			UpdateConversation = G_CurConversationLogin;
			UpdateMessages = True;

			G_PresenceColors = C_DefaultPresenceColors;
			G_PresenceColors[1] = LayerMessages_HighlightsColor;
			Quad_ShowFriendOptions.Colorize = LayerMessages_HighlightsColor;

			Page.GetClassChildren("ColorBg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = LayerMessages_BackgroundColor;
				}
			}

			Page.GetClassChildren("ColorFg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = LayerMessages_HighlightsColor;
				}
			}

			Page.GetClassChildren("ColorFg2", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).BgColor = LayerMessages_HighlightsColor;
				}
			}
		}

		if (UpdateConversation != "") {
			UpdateConversationDetails(UpdateConversation);
			UpdateConversation = "";
		}

		if (PrevConversationLogin != LayerMessages_ConversationLogin) {
			PrevConversationLogin = LayerMessages_ConversationLogin;
			UpdateMessages = True;

			if (!ConversationOffset.existskey(LayerMessages_ConversationLogin)) {
				ConversationOffset[LayerMessages_ConversationLogin] = 0;
			}

			MessagesOffset = ConversationOffset[LayerMessages_ConversationLogin];
			Frame_MessagesContainer.ScrollMax.Y = -ML::ToReal(MessagesOffset);
			Frame_MessagesContainer.ScrollOffset.Y = Frame_MessagesContainer.ScrollMax.Y;
		}

		/*
		if (PrevMessagesOffset != MessagesOffset) {
			PrevMessagesOffset = MessagesOffset;
			UpdateMessages = True;

			ConversationOffset[LayerMessages_ConversationLogin] = MessagesOffset;
			LayerMessages_CurrentOffset = MessagesOffset;

			SendCustomEvent("MessagesScrollingOffsetChange", [TL::ToText(LayerMessages_CurrentOffset)]);
		}
		*/

		if (PrevScrollOffset != Frame_MessagesContainer.ScrollOffset.Y) {
			PrevScrollOffset = Frame_MessagesContainer.ScrollOffset.Y;

			UpdateMessages = True;
			MessagesOffset = -ML::FloorInteger(PrevScrollOffset);

			ConversationOffset[LayerMessages_ConversationLogin] = MessagesOffset;
			LayerMessages_CurrentOffset = MessagesOffset;

			SendCustomEvent("MessagesScrollingOffsetChange", [TL::ToText(LayerMessages_CurrentOffset)]);

			Frame_MessagesOffset.RelativePosition_V3.Y = -PrevScrollOffset;
		}

		if (UpdateMessages) {
			UpdateMessages = False;

			G_MessagesTotalHeight = 0.;
			G_PrevMessageSender = "";
			G_PrevMessageDateText = "";
			G_MessageAnimationDistance = 0.;

			ClearTimeSeparators();

			declare LastMessageFrameOrder = 0;

			if (LayerMessages_ChatHistoryId != NullId && ManiaPlanet.ChatManager.Histories.existskey(LayerMessages_ChatHistoryId)) {
				declare History <=> ManiaPlanet.ChatManager.Histories[LayerMessages_ChatHistoryId];

				declare DisplayedMessagesLimitReached = False;
				declare CurrentMessageOrder = 0;

				NbMessagesTotal = 0;

				foreach (MessagesGroup in History.Entries) {
					NbMessagesTotal += MessagesGroup.Messages.count;
				}

				foreach (MessagesGroup in History.Entries) {
					foreach (I => Message in MessagesGroup.Messages) {
						if (CurrentMessageOrder >= MessagesOffset) {
							if (!Frame_MessageItems.existskey(LastMessageFrameOrder) || G_MessagesTotalHeight > Frame_MessagesContainer.Size.Y) {
								DisplayedMessagesLimitReached = True;
								continue;
							}

							if (!DisplayedMessagesLimitReached) {
								declare Frame <=> Frame_MessageItems[LastMessageFrameOrder];
								declare IsFirstInGroup = I == MessagesGroup.Messages.count - 1;
						 
								FormatMessageFrame(LastMessageFrameOrder, Frame, MessagesGroup, Message, IsFirstInGroup);
								Frame.Visible = True;

								LastMessageFrameOrder += 1;
							}
						}

						CurrentMessageOrder += 1;
					}

					if (DisplayedMessagesLimitReached) break;
				}

				Frame_MessagesContainer.ScrollMax.Y = -ML::ToReal(NbMessagesTotal - 1);

				Frame_ChatScrollBar.Visible = NbMessagesTotal > LastMessageFrameOrder;

				if (Frame_ChatScrollBar.Visible) {
					Quad_ScrollBarHandle.RelativePosition_V3.Y = (Quad_ScrollBarArea.Size.Y - Quad_ScrollBarHandle.Size.Y) * (ML::ToReal(MessagesOffset) / (NbMessagesTotal - 1));
				}
			}
 
 			for (I, LastMessageFrameOrder, Frame_MessageItems.count - 1) {
 				Frame_MessageItems[I].Visible = False;
 			}
 			
			AnimMgr.Flush(Frame_MessagesAnim);

			if (LayerMessages_NextUpdateAnim) {
				Frame_MessagesAnim.RelativePosition_V3.Y -= G_MessageAnimationDistance;
				AnimMgr.Add(Frame_MessagesAnim, "<frame pos='0 0' />", 250, CAnimManager::EAnimManagerEasing::SineOut);
				LayerMessages_NextUpdateAnim = False;
			}

			else {
				Frame_MessagesAnim.RelativePosition_V3.Y = 0.;
			}
		}

		if (PrevDisplayImagePreview != DisplayImagePreview) {
			PrevDisplayImagePreview = DisplayImagePreview;

			Quad_ImagePreviewClose.Visible = DisplayImagePreview;
			Page.LinksInhibited = DisplayImagePreview;

			declare Easing = CAnimManager::EAnimManagerEasing::Linear;
			declare Opacity = 0.;

			if (DisplayImagePreview) Opacity = 1.;
			
			AnimMgr.Flush(Quad_ImagePreviewPic);
			AnimMgr.Flush(Quad_ImagePreviewBg);

			AnimMgr.Add(Quad_ImagePreviewPic, "<quad opacity='"^Opacity^"' />", 125, Easing);
			AnimMgr.Add(Quad_ImagePreviewBg, "<quad opacity='"^(Opacity * .75)^"' />", 125, Easing);
		}

		if (PrevIsMouseRightButton != Input.MouseRightButton) {
			PrevIsMouseRightButton = Input.MouseRightButton;

			if (PrevIsMouseRightButton) {
				if (HoveringUserCard) {
					ShowRightClickMenu_Common(Input.MousePos, G_CurConversationLogin);
				}
			}
		}
	}
}

--></script>
</manialink>
	""";
}
