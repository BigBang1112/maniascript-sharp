CChatRoom GetRoomFromLogin(Text _Login) {
	if (ManiaPlanet.ChatManager.Squad != Null && ManiaPlanet.ChatManager.Squad.RoomId == _Login) {
		return ManiaPlanet.ChatManager.Squad;
	}

	foreach (Room in ManiaPlanet.ChatManager.Channels) {
		if (Room.RoomId == _Login) return Room;
	}

	return Null;
}

Void AnimOpacity(CMlControl _Control) {
	AnimMgr.Flush(_Control);

	switchtype (_Control) {
		case CMlLabel : (_Control as CMlLabel).Opacity = 0.;
		case CMlQuad : (_Control as CMlQuad).Opacity = 0.;
	}

	declare Opacity = 1.;

	switch (_Control.ControlId) {
		case "Quad_ContactIsSelected" : Opacity = .25;
		case "Quad_ContactPresence3" : Opacity = .125;
	}

	AnimMgr.Add(_Control, "<control opacity='"^Opacity^"' />", 125, CAnimManager::EAnimManagerEasing::SineIn);
}

Text GetRoomName(Text _Login) {
	if (TL::Find("@squad", _Login, False, False)) return _("Squad");

	switch (TL::Split("@", _Login)[0]) {
		case "trackmania" : return "TrackMania channel";
		case "shootmania" : return "ShootMania channel";
	}

	return _Login;
}

Text GetRoomIcon(Text _Login) {
	switch (TL::Split("@", _Login)[0]) {
		case "trackmania" : return "{{{C_ImgBase}}}channel-trackmania.dds";
		case "shootmania" : return "{{{C_ImgBase}}}channel-shootmania.dds";
	}

	return "{{{C_ImgBase}}}squad.dds";
}

Void FormatContactItem(Integer _Order, CMlFrame _Frame, Text _Login, Boolean _AnimFade) {
	if (_Frame == Null) return;

	declare Quad_ContactPresence	<=> (_Frame.GetFirstChild("Quad_ContactPresence")	as CMlQuad);
	declare Quad_ContactPresence2	<=> (_Frame.GetFirstChild("Quad_ContactPresence2")	as CMlQuad);
	declare Quad_ContactAvatar		<=> (_Frame.GetFirstChild("Quad_ContactAvatar")		as CMlQuad);
	declare Label_ContactNickname	<=> (_Frame.GetFirstChild("Label_ContactNickname")	as CMlLabel);
	declare Label_ContactStatus		<=> (_Frame.GetFirstChild("Label_ContactStatus")	as CMlLabel);
	declare Quad_ContactIsSelected	<=> (_Frame.GetFirstChild("Quad_ContactIsSelected")	as CMlQuad);
	declare Quad_ContactActionCard	<=> (_Frame.GetFirstChild("Quad_ContactActionCard")	as CMlQuad);
	declare Quad_GroupedServer		<=> (_Frame.GetFirstChild("Quad_GroupedServer")		as CMlQuad);

	declare Quad_ContactJoinGame	<=> (_Frame.GetFirstChild("Quad_ContactJoinGame")	as CMlQuad);
	declare Quad_ContactAccept		<=> (_Frame.GetFirstChild("Quad_ContactAccept")		as CMlQuad);
	declare Quad_ContactCancel		<=> (_Frame.GetFirstChild("Quad_ContactCancel")		as CMlQuad);
	declare Quad_ContactInvite		<=> (_Frame.GetFirstChild("Quad_ContactInvite")		as CMlQuad);

	declare Frame_MessagesCount		<=> (_Frame.GetFirstChild("Frame_MessagesCount")	as CMlFrame);
	declare Label_MessagesCount		<=> (_Frame.GetFirstChild("Label_MessagesCount")	as CMlLabel);
	declare Quad_MessagesCount		<=> (_Frame.GetFirstChild("Quad_MessagesCount")		as CMlQuad);

	declare Text			LayerContacts_ActiveConversation	for Page;
	declare Integer			LayerContacts_ContactsListMode		for Page;
	declare Integer[Text]	LayerContacts_NbUnreadMessages		for Page;

	declare Contact				<=> ManiaPlanet.ChatManager.GetContactFromLogin(_Login);
	declare ChatRoom			<=> GetRoomFromLogin(_Login);
	declare ContactLogin		= _Login;
	declare ContactNickname		= "";
	declare ContactAvatarURL	= "{{{C_ImgBase}}}default-avatar.dds";
	declare ContactPresence		= 0;
	declare ContactStatusText	= "";
	declare ContactServerLogin	= "";
	declare ContactServerName	= "";
	declare ContactServerLink	= "";
	declare ContactNbUnreads	= 0;

	declare IsSquadInvite = TL::StartsWith("squad:", _Login);

	if (LayerContacts_NbUnreadMessages.existskey(ContactLogin)) {
		ContactNbUnreads = LayerContacts_NbUnreadMessages[ContactLogin];
	}

	// Grab the data from the server
	foreach (User in ManiaPlanet.CurrentServerPlayers) {
		if (User.Login != _Login) continue;

		ContactLogin		= User.Login;
		ContactNickname		= User.Name;
		ContactAvatarURL	= User.AvatarUrl;
		ContactStatusText	= _("On this server");
		ContactServerLogin	= ManiaPlanet.ChatManager.CurrentServerLogin;
		ContactServerName	= ManiaPlanet.ChatManager.CurrentServerName;
		ContactServerLink	= ManiaPlanet.ChatManager.CurrentServerJoinLink;
		ContactPresence		= 1;
	}

	// That's us!
	if (_Login == LocalUser.Login) {
		declare Integer LayerContacts_LocalUserPresence for Page;

		ContactLogin		= LocalUser.Login;
		ContactNickname		= LocalUser.Name;
		ContactAvatarURL	= LocalUser.AvatarUrl;
		ContactStatusText	= ManiaPlanet.ChatManager.Presence_Status;
		ContactServerLogin	= ManiaPlanet.ChatManager.CurrentServerLogin;
		ContactServerName	= ManiaPlanet.ChatManager.CurrentServerName;
		ContactServerLink	= ManiaPlanet.ChatManager.CurrentServerJoinLink;
		ContactPresence		= LayerContacts_LocalUserPresence;
	}

	// Chat contact exists
	else if (Contact != Null) {
		ContactLogin		= Contact.Login;
		ContactNickname		= Contact.Name;
		ContactAvatarURL	= Contact.AvatarUrl;
		ContactStatusText	= Contact.PresenceStatus;
		ContactServerLogin	= Contact.CurrentServerLogin;
		ContactServerName	= Contact.CurrentServerName;
		ContactServerLink	= Contact.CurrentServerJoinLink;

		switch (Contact.PresenceShow) {
			case CChatContact::EPresenceShow::Available		: ContactPresence = 1;
			case CChatContact::EPresenceShow::Away			: ContactPresence = 2;
			case CChatContact::EPresenceShow::ExtendedAway	: ContactPresence = 2;
			case CChatContact::EPresenceShow::DoNotDisturb	: ContactPresence = 3;
			case CChatContact::EPresenceShow::WantToChat	: ContactPresence = 1;
			default											: ContactPresence = 0;
		}

		// Hackathon
		//if (LayerContacts_ContactsListMode == 3 && !Contact.IsBuddyInManiaPlanet) {
		//	ContactPresence = 1;
		//}

		if (Contact.Subscription_From == CChatContact::ESubscription::Pending) {
			ContactStatusText = _("|Chat|Incoming invite");
		}

		if (Contact.Subscription_To == CChatContact::ESubscription::Pending) {
			ContactStatusText = _("|Chat|Outgoing invite");
		}
	}

	// Chat room exists
	else if (ChatRoom != Null) {
		ContactLogin		= ChatRoom.RoomId;
		ContactNickname 	= GetRoomName(ContactLogin);
		ContactAvatarURL	= GetRoomIcon(ContactLogin);
		ContactPresence		= 1;

		if (ChatRoom.Members.count == 0) {
			ContactStatusText = _("1 member");
		} else {
			ContactStatusText = TL::Compose(_("%1 members"), TL::ToText(ChatRoom.Members.count + 1));
		}
	}

	// Squad invite
	else if (IsSquadInvite) {
		declare Sender <=> ManiaPlanet.ChatManager.GetContactFromLogin(TL::Replace(ContactLogin, "squad:", ""));

		if (Sender != Null) {
			ContactNickname = Sender.Name;
		} else {
			ContactNickname = _("Squad");
		}

		ContactAvatarURL	= "{{{C_ImgBase}}}squad.dds";
		ContactStatusText	= _("|Chat|Invites to their squad");
		ContactPresence		= 1;
	}

	Label_ContactNickname.Value = ContactNickname;

	if (Quad_ContactAvatar.Image == Null) {
		Quad_ContactAvatar.ImageUrl = ContactAvatarURL;
	}

	// Unreads counter
	Frame_MessagesCount.Visible = ContactNbUnreads > 0;

	if (Frame_MessagesCount.Visible) {
		Label_MessagesCount.Value = TL::ToText(ContactNbUnreads);
		Quad_MessagesCount.Size.X = Label_MessagesCount.ComputeWidth(Label_MessagesCount.Value) + 1.;
	}

	Quad_GroupedServer.Visible = G_PrevContactServer != "" && G_PrevContactServer == ContactServerLogin;
	Quad_GroupedServer.BgColor = G_PresenceColors[1];
	G_PrevContactServer = ContactServerLogin;

	// Status text visibility
	Label_ContactStatus.Visible = ContactServerLogin != "" || ContactStatusText != "" || (ContactPresence != 0 && ContactPresence != 1);
	if (Label_ContactStatus.Visible) Label_ContactNickname.RelativePosition_V3.Y = 1.5;
	else Label_ContactNickname.RelativePosition_V3.Y = 0.;

	// Contact on a server
	if (ContactServerLogin != "") {
		Label_ContactStatus.Value = ContactServerName;
		Label_ContactStatus.TextColor = <1., 1., 1.>;
	}

	// Contact status text
	else {
		if (ContactStatusText != "") {
			Label_ContactStatus.Value = ContactStatusText;
		}

		else switch (ContactPresence) {
			case -1 : Label_ContactStatus.Value = _("|Presence|Connecting");
			case 1 : Label_ContactStatus.Value = _("|Presence|Online");
			case 2 : Label_ContactStatus.Value = _("|Presence|Away");
			case 3 : Label_ContactStatus.Value = _("|Presence|Busy");
			default : Label_ContactStatus.Value = _("|Presence|Offline");
		}

		Label_ContactStatus.TextColor = G_PresenceColors[ContactPresence];
	}

	//if (TL::Length(Label_ContactStatus.Value) > 48) {
	//	Label_ContactStatus.Value = TL::SubString(Label_ContactStatus.Value, 0, 48) ^ "$z...";
	//}

	// Contact presence color
	Quad_ContactPresence.Visible = ContactPresence != 0;
	Quad_ContactPresence.BgColor = G_PresenceColors[ContactPresence];
	Quad_ContactPresence2.Visible = Quad_ContactPresence.Visible;
	Quad_ContactPresence2.Colorize = Quad_ContactPresence.BgColor;
	Quad_ContactIsSelected.BgColor = Quad_ContactPresence.BgColor;

	Quad_ContactJoinGame.Visible = ContactServerLogin != "" && ContactLogin != LocalUser.Login && ManiaPlanet.ChatManager.CurrentServerLogin != ContactServerLogin;
	Quad_ContactAccept.Visible = IsSquadInvite || (Contact != Null && Contact.Subscription_From == CChatContact::ESubscription::Pending);
	Quad_ContactCancel.Visible = IsSquadInvite || (Contact != Null && (Contact.Subscription_From == CChatContact::ESubscription::Pending || Contact.Subscription_To == CChatContact::ESubscription::Pending));
	Quad_ContactInvite.Visible = ContactLogin != LocalUser.Login && !Quad_ContactCancel.Visible && (Contact == Null || Contact.Subscription_To == CChatContact::ESubscription::None) && LayerContacts_ContactsListMode != 0;

	Quad_ContactIsSelected.Visible = ContactLogin == LayerContacts_ActiveConversation && _Frame.ControlId != "Frame_LocalUserCard";

	declare Text ButtonServerJoinLink for Quad_ContactJoinGame;
	declare Text ServerLogin for Quad_ContactJoinGame;
	ButtonServerJoinLink = ContactServerLink;
	ServerLogin = ContactServerLogin;

	foreach (Quad in [Quad_ContactActionCard, Quad_ContactAccept, Quad_ContactCancel, Quad_ContactInvite]) {
		declare Text ActionContactLogin for Quad;
		ActionContactLogin = ContactLogin;
	}

	if (_AnimFade) {
		Page.GetClassChildren("Anim", _Frame, True);

		foreach (Control in Page.GetClassChildren_Result) {
			AnimOpacity(Control);
		}
	}
}

Real GetCardPosition(Integer _Order) {
	declare SListGroup[] LayerContacts_ContactsGroups for Page;
	declare TotalPositionsSkipped = 0;
	declare HeaderHeights = 0.;

	foreach (Group in LayerContacts_ContactsGroups) {
		TotalPositionsSkipped += Group.Size;
		HeaderHeights += -4.;
		if (TotalPositionsSkipped > _Order) break;
	}

	return HeaderHeights + _Order * -8.;
}

	declare Quad_ContactsListScrollBar	<=> (Page.GetFirstChild("Quad_ContactsListScrollBar")	as CMlQuad);

	declare CMlFrame[] Frame_ContactsListItems;
	declare CMlLabel[] Label_ContactListHeaders;

	for (I, 0, {{{C_NbContactsDisplayedInList - 1}}}) {
		Frame_ContactsListItems.add((Page.GetFirstChild("Frame_ContactsListItem#"^I) as CMlFrame));
		Label_ContactListHeaders.add((Page.GetFirstChild("Label_ContactListHeader#"^I) as CMlLabel));
	}

	declare CMlFrame[] Frame_PresenceOptions;
	for (I, 0, 3) Frame_PresenceOptions.add((Page.GetFirstChild("Frame_PresenceOption#"^I) as CMlFrame));

	declare CMlFrame[] Frame_WindowTabs;

	for (I, 0, 4) {
		declare Frame <=> (Page.GetFirstChild("Frame_WindowTab#"^I) as CMlFrame);
		declare Quad_WindowTabIcon <=> (Frame.GetFirstChild("Quad_WindowTabIcon") as CMlQuad);

		switch (I) {
			case 0 : {
				Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}presence-orb.dds";
				Quad_WindowTabIcon.ImageUrlFocus = Quad_WindowTabIcon.ImageUrl;
				Quad_LocalUserPresence <=> Quad_WindowTabIcon;
			}
			case 1 : {
				Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-messages.dds";
				Quad_WindowTabIcon.ImageUrlFocus = "{{{C_ImgBase}}}buddy-mode-messages-focus.dds";
			}
			case 2 : {
				Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-invites.dds";
				Quad_WindowTabIcon.ImageUrlFocus = "{{{C_ImgBase}}}buddy-mode-invites-focus.dds";
			}
			case 3 : {
				Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-serverlist.dds";
				Quad_WindowTabIcon.ImageUrlFocus = "{{{C_ImgBase}}}buddy-mode-serverlist-focus.dds";
			}
			/*
			case 4 : {
				Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mailbox.dds";
				Quad_WindowTabIcon.ImageUrlFocus = "{{{C_ImgBase}}}buddy-mailbox-focus.dds";
			}
			*/
			case 4 : {
				Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-settings.dds";
				Quad_WindowTabIcon.ImageUrlFocus = "{{{C_ImgBase}}}buddy-mode-settings-focus.dds";
			}
		}

		declare Integer TabOrder for Quad_WindowTabIcon;
		TabOrder = I;

		Frame_WindowTabs.add(Frame);
	}

	declare Boolean			LayerContacts_IsVisible				for Page;
	declare Real			LayerContacts_PositionOffset		for Page;
	declare Integer			LayerContacts_LocalUserPresence		for Page;
	declare Text[]			LayerContacts_ContactsList			for Page;
	declare Text			LayerContacts_ActiveConversation	for Page;
	declare Integer			LayerContacts_ContactsListMode		for Page;
	declare Integer			LayerContacts_MessagesListMode		for Page;
	declare Integer			LayerContacts_NbFriendInvites		for Page;
	declare Vec3			LayerContacts_BackgroundColor		for Page;
	declare Vec3			LayerContacts_HighlightsColor		for Page;
	declare Integer			LayerContacts_ColorLatestChange		for Page;
	declare Integer[Text]	LayerContacts_NbUnreadMessages		for Page;
	declare SListGroup[]	LayerContacts_ContactsGroups		for Page;

	declare UpdateWindowTabs = True;
	declare TotalNbUnreads = 0;

	declare PresenceSelectVisible = False;
	declare UpdatePresenceOptions = True;

	declare UpdateContactsList = True;
	declare NextListUpdateIsAnim = True;
	declare HideLayerAfter = -1;

	declare PrevLayerIsVisible = False;
	declare PrevPositionOffset = -1.;
	declare PrevContactsListMode = -1;
	declare PrevPresenceSelectVisible = False;
	declare PrevNbOldMailMessages = -1;
	declare PrevEntrySearchContactsValue = "";
	declare PrevColorChange = 0;
	declare PrevScrollOffsetY = -1.;

	while (True) {
		yield;

		foreach (Event in PendingEvents) {
			if (Event.Type == CMlScriptEvent::Type::MouseClick) {
				if (Event.ControlId == "Quad_WindowTabIcon") {
					declare Integer TabOrder for Event.Control;

					switch (TabOrder) {
						case 0 : {
							PresenceSelectVisible = !PresenceSelectVisible;
						}

						case 1 : {
							Entry_SearchContacts.Value = "";
							SendCustomEvent("SearchContacts", [""]);
							SendCustomEvent("SetChatMode", [TL::ToText(TabOrder)]);
						}

						/*
						case 4 : {
							OpenLink("#mailto=", ::LinkType::ManialinkBrowser);
						}
						*/

						default : SendCustomEvent("SetChatMode", [TL::ToText(TabOrder)]);
					}
				}

				else if (Event.ControlId == "Quad_PresenceOptionCard") {
					declare Integer PresenceId for Event.Control;
					SendCustomEvent("SetPresence", [TL::ToText(PresenceId)]);
					PresenceSelectVisible = False;
				}

				else if (Event.ControlId == "Quad_SearchButtonClear") {
					Entry_SearchContacts.Value = "";
					SendCustomEvent("SearchContacts", [""]);
				}

				else if (Event.ControlId == "Quad_CloseChatRoomView") {
					SendCustomEvent("SetConversation", [""]);
				}

				else if (Event.ControlId == "Quad_ContactActionCard") {
					declare Text ActionContactLogin for Event.Control;
					declare Contact = ManiaPlanet.ChatManager.GetContactFromLogin(ActionContactLogin);
					declare IsFriend = Contact != Null && Contact.IsBuddyInManiaPlanet;

					if (Frame_LocalUserCard.Controls.exists(Event.Control)) {
						SendCustomEvent("OpenProfileSettings", Text[]);
					}

					else if (LayerContacts_ContactsListMode == 0 || LayerContacts_ContactsListMode == 2 || IsFriend) {
						SendCustomEvent("SetConversation", [ActionContactLogin]);
					}
				}

				else if (Event.ControlId == "Quad_ContactJoinGame") {
					declare Text ButtonServerJoinLink for Event.Control;
					OpenLink(ButtonServerJoinLink, CMlScript::LinkType::ManialinkBrowser);
					SendCustomEvent("CloseChatWindow", Text[]);
					SendCustomEvent("HideServerInfo", Text[]);
				}

				else if (Event.ControlId == "Quad_ContactInvite") {
					declare Text ActionContactLogin for Event.Control;
					SendCustomEvent("AddNewContact", [ActionContactLogin]);
				}

				else if (Event.ControlId == "Quad_ContactAccept") {
					declare Text ActionContactLogin for Event.Control;
					SendCustomEvent("AcceptInvite", [ActionContactLogin]);
				}

				else if (Event.ControlId == "Quad_ContactCancel") {
					declare Text ActionContactLogin for Event.Control;
					SendCustomEvent("RemoveContact", [ActionContactLogin]);
				}

				else if (Event.ControlId == "Quad_AddFriendButton") {
					if (Entry_AddFriendEntry.Value == "") continue;
					SendCustomEvent("AddNewContact", [Entry_AddFriendEntry.Value]);
					Entry_AddFriendEntry.Value = "";
				}

				if (Event.Control.HasClass("Focus")) {
					Audio.PlaySoundEvent(CAudioManager::ELibSound::Valid, 1, 0.);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::MouseOut) {
				if (Event.ControlId == "Quad_ContactJoinGame") {
					SendCustomEvent("HideServerInfo", Text[]);
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::MouseOver) {
				if (Event.ControlId == "Quad_ContactJoinGame") {
					declare Text ServerLogin for Event.Control;
					declare Pos = Event.Control.AbsolutePosition_V3.X^" "^Event.Control.AbsolutePosition_V3.Y;
					SendCustomEvent("ShowServerInfo", [ServerLogin, Pos]);
				}

				if (Event.Control.HasClass("Focus")) {
					Audio.PlaySoundEvent(CAudioManager::ELibSound::Focus, 1, 0.);
				}
			}

			else if (Event.Type == CMlEvent::Type::EntrySubmit) {
				if (Event.Control == Entry_SearchContacts) {
					SendCustomEvent("SearchContacts", [Entry_SearchContacts.Value]);
				}

				else if (Event.Control == Entry_AddFriendEntry) {
					if (Entry_AddFriendEntry.Value == "") continue;
					SendCustomEvent("AddNewContact", [Entry_AddFriendEntry.Value]);
					Entry_AddFriendEntry.Value = "";
				}
			}

			else if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
				switch (Event.CustomEventType) {
					case "UpdateContactsList" : {
						UpdateContactsList = True;
						UpdateWindowTabs = True;
					}
				}
			}
		}

		if (PrevLayerIsVisible != LayerContacts_IsVisible) {
			PrevLayerIsVisible = LayerContacts_IsVisible;

			AnimMgr.Flush(Frame_ContactsList);

			declare Pos = <Frame_ContactsList.RelativePosition_V3.X, -6.>;
			declare Easing = CAnimManager::EAnimManagerEasing::SineOut;

			if (!LayerContacts_IsVisible) {
				Pos.Y += 150.;
				Easing = CAnimManager::EAnimManagerEasing::SineIn;
				PresenceSelectVisible = False;
				HideLayerAfter = Now + 200;
			} else {
				HideLayerAfter = -1;
			}

			AnimMgr.Add(Frame_ContactsList, "<frame pos='"^Pos.X^" "^Pos.Y^"' />", 200, Easing);
		}

		if (HideLayerAfter > 0 && Now >= HideLayerAfter) {
			HideLayerAfter = -1;
			SendCustomEvent("HideLayerContacts", Text[]);
		}

		if (PrevPresenceSelectVisible != PresenceSelectVisible) {
			PrevPresenceSelectVisible = PresenceSelectVisible;

			if (PresenceSelectVisible) {
				AnimMgr.Flush(Frame_SetUserPresence);
				AnimMgr.Add(Frame_SetUserPresence, "<frame scale='1' />", 125, CAnimManager::EAnimManagerEasing::SineOut);

			} else {
				AnimMgr.Flush(Frame_SetUserPresence);
				AnimMgr.Add(Frame_SetUserPresence, "<frame scale='0' />", 125, CAnimManager::EAnimManagerEasing::SineIn);
			}

			Page.LinksInhibited = PresenceSelectVisible;
		}

		if (!LayerContacts_IsVisible) continue;

		if (PrevPositionOffset != LayerContacts_PositionOffset) {
			PrevPositionOffset = LayerContacts_PositionOffset;

			Frame_WindowClip.RelativePosition_V3.X = LayerContacts_PositionOffset;
		}

		if (PrevColorChange != LayerContacts_ColorLatestChange) {
			PrevColorChange = LayerContacts_ColorLatestChange;

			UpdatePresenceOptions = True;
			UpdateWindowTabs = True;
			UpdateContactsList = True;

			G_PresenceColors = C_DefaultPresenceColors;
			G_PresenceColors[1] = LayerContacts_HighlightsColor;

			Page.GetClassChildren("ColorBg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = LayerContacts_BackgroundColor;
				}
			}

			Page.GetClassChildren("ColorFg", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).Colorize = LayerContacts_HighlightsColor;
				}
			}

			Page.GetClassChildren("ColorFg2", Page.MainFrame, True);

			foreach (Control in Page.GetClassChildren_Result) {
				if (Control is CMlQuad) {
					(Control as CMlQuad).BgColor = LayerContacts_HighlightsColor;
				}
			}
		}

		if (PrevNbOldMailMessages != ManiaPlanet.Messenger_MessagesCount) {
			PrevNbOldMailMessages = ManiaPlanet.Messenger_MessagesCount;
			UpdateWindowTabs = True;
		}

		if (UpdateWindowTabs) {
			UpdateWindowTabs = False;

			TotalNbUnreads = 0;

			foreach (Login => Count in LayerContacts_NbUnreadMessages) {
				TotalNbUnreads += Count;
			}

			foreach (I => Frame in Frame_WindowTabs) {
				declare Quad_WindowTabIcon		<=> (Frame.GetFirstChild("Quad_WindowTabIcon")		as CMlQuad);
				declare Frame_NbNotifications	<=> (Frame.GetFirstChild("Frame_NbNotifications")	as CMlFrame);
				declare Label_NbNotifications	<=> (Frame.GetFirstChild("Label_NbNotifications")	as CMlLabel);
				declare Quad_NbNotifications	<=> (Frame.GetFirstChild("Quad_NbNotifications")	as CMlQuad);

				declare NbNotifications = 0;

				switch (I) {
					case 1 : {
						if (LayerContacts_ContactsListMode != 1) {
							Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-messages-on.dds";
						} else {
							Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-messages.dds";
						}

						NbNotifications = TotalNbUnreads;
					}
					case 2 : {
						if (LayerContacts_ContactsListMode == 1) {
							Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-invites-on.dds";
						} else {
							Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-invites.dds";
						}

						NbNotifications = LayerContacts_NbFriendInvites;
					}
					case 3 : {
						if (LayerContacts_MessagesListMode == 1) {
							Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-serverlist-on.dds";
						} else {
							Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-serverlist.dds";
						}
					}
					case 4 : {
						//NbNotifications = ManiaPlanet.Messenger_MessagesCount;
					}
					case 5 : {
						Quad_WindowTabIcon.ImageUrl = "{{{C_ImgBase}}}buddy-mode-settings.dds";
						Quad_WindowTabIcon.ImageUrlFocus = "{{{C_ImgBase}}}buddy-mode-settings-focus.dds";
					}
				}

				Frame_NbNotifications.Visible = NbNotifications > 0;

				if (Frame_NbNotifications.Visible) {
					Label_NbNotifications.Value = TL::ToText(NbNotifications);
					Quad_NbNotifications.Size.X = Label_NbNotifications.ComputeWidth(Label_NbNotifications.Value) + .5;
				}
			}
		}

		if (UpdatePresenceOptions) {
			UpdatePresenceOptions = False;

			foreach (I => Frame in Frame_PresenceOptions) {
				declare Quad_PresenceOptionColor	<=> (Frame.GetFirstChild("Quad_PresenceOptionColor")	as CMlQuad);
				declare Label_PresenceOptionName	<=> (Frame.GetFirstChild("Label_PresenceOptionName")	as CMlLabel);
				declare Quad_PresenceOptionCard		<=> (Frame.GetFirstChild("Quad_PresenceOptionCard")		as CMlQuad);

				declare Integer PresenceId for Quad_PresenceOptionCard;
				PresenceId = I + 1;
				if (I == 3) PresenceId = 0;

				Quad_PresenceOptionColor.Colorize = G_PresenceColors[PresenceId];
				Label_PresenceOptionName.TextColor = Quad_PresenceOptionColor.Colorize;

				switch (PresenceId) {
					case 0 : Label_PresenceOptionName.Value = _("Offline");
					case 1 : Label_PresenceOptionName.Value = _("Online");
					case 2 : Label_PresenceOptionName.Value = _("Away");
					case 3 : Label_PresenceOptionName.Value = _("Busy");
				}
			}
		}

		if (PrevScrollOffsetY != Frame_ContactsListContainer.ScrollOffset.Y) {
			PrevScrollOffsetY = Frame_ContactsListContainer.ScrollOffset.Y;
			UpdateContactsList = True;
		}

		if (UpdateContactsList) {
			UpdateContactsList = False;

			G_PrevContactServer = "";

			if (PrevContactsListMode != LayerContacts_ContactsListMode) {
				PrevContactsListMode = LayerContacts_ContactsListMode;
				Frame_ContactsListContainer.ScrollOffset.Y = 0.;
				NextListUpdateIsAnim = True;
			}

			FormatContactItem(0, Frame_LocalUserCard, LocalUser.Login, False);

			G_PrevContactServer = "";

			Quad_LocalUserPresence.Colorize = G_PresenceColors[LayerContacts_LocalUserPresence];

			Frame_SearchContacts.Visible	= LayerContacts_ContactsListMode == 0 || LayerContacts_ContactsListMode == 2;
			Frame_AddNewFriend.Visible		= LayerContacts_ContactsListMode == 1;
			Frame_CurrentChatRoom.Visible	= LayerContacts_ContactsListMode == 3;

			if (Frame_CurrentChatRoom.Visible) {
				declare ChatRoom <=> GetRoomFromLogin(LayerContacts_ActiveConversation);
				if (ChatRoom != Null) {
					Quad_ChatRoomAvatar.ImageUrl	= GetRoomIcon(ChatRoom.RoomId);
					Label_ChatRoomName.Value		= GetRoomName(ChatRoom.RoomId);
					Label_ChatRoomStatus.Value		= TL::Compose(_("%1 members"), TL::ToText(ChatRoom.Members.count + 1));
				}
			}

			// GROUPS
			declare TotalPreviousHeight = 0.;

			foreach (I => Label in Label_ContactListHeaders) {
				Label.Visible = LayerContacts_ContactsGroups.existskey(I);
				if (!Label.Visible) continue;
				declare Group = LayerContacts_ContactsGroups[I];

				Label.Value = TL::Compose("%1 (%2)", Group.Name, TL::ToText(Group.Size));
				Label.RelativePosition_V3.Y = -TotalPreviousHeight + -2.125;
				TotalPreviousHeight += 4. + Group.Size * 8.;
			}

			// TOTAL SCROLL HEIGHT
			declare TotalHeight = LayerContacts_ContactsList.count * 8. + LayerContacts_ContactsGroups.count * 4.;

			Frame_ContactsListContainer.ScrollMax.Y = ML::Max(0., TotalHeight - Frame_ContactsListContainer.Size.Y);
			Quad_ContactsListHoverBox.RelativePosition_V3.Y = -Frame_ContactsListContainer.ScrollOffset.Y;
			Quad_ContactsListHoverBox.Size.Y = Frame_ContactsListContainer.Size.Y;

			// SCROLL BAR
			Quad_ContactsListScrollBar.Visible = Frame_ContactsListContainer.ScrollMax.Y > 0;

			if (Quad_ContactsListScrollBar.Visible) {
				Quad_ContactsListScrollBar.Size.Y = Frame_ContactsListContainer.Size.Y * (Frame_ContactsListContainer.Size.Y / TotalHeight);
				Quad_ContactsListScrollBar.RelativePosition_V3.Y = -(Frame_ContactsListContainer.ScrollOffset.Y / Frame_ContactsListContainer.ScrollMax.Y) * (Frame_ContactsListContainer.Size.Y - Quad_ContactsListScrollBar.Size.Y);
			}

			// OFFSET
			declare Offset = 0;
			declare SectionOriginPos = 0.;

			foreach (Group in LayerContacts_ContactsGroups) {
				declare Break = False;

				for (I, 0, Group.Size - 1) {
					if (SectionOriginPos >= Frame_ContactsListContainer.ScrollOffset.Y - 12.) {
						Break = True;
						break;
					}

					SectionOriginPos += 8.;
					Offset += 1;
				}

				if (Break) break;

				SectionOriginPos += 4.;
			}

			Label_ContactListError.Visible = LayerContacts_ContactsList.count <= 0;

			if (Label_ContactListError.Visible) {
				if (LayerContacts_LocalUserPresence <= 0) {
					Label_ContactListError.Value = _("Come online to see your available chats and pending invites.");
				}

				else switch (LayerContacts_ContactsListMode) {
					case 0 : Label_ContactListError.Value = _("Your friends and group chats will show up here. Join multiplayer and find some friends online!");
					case 1 : Label_ContactListError.Value = _("You don't have pending friend requests right now.");
					case 2 : Label_ContactListError.Value = _("No friends matching your search have been found.");
					// case 3 : Label_ContactListError.Value = _("");
				}
			}

			foreach (I => Frame in Frame_ContactsListItems) {
				declare Key = I + Offset;
				Frame.Visible = LayerContacts_ContactsList.existskey(Key);
				if (!Frame.Visible) continue;

				declare ContactLogin = LayerContacts_ContactsList[Key];
				FormatContactItem(I, Frame, ContactLogin, NextListUpdateIsAnim);
				Frame.RelativePosition_V3.Y = GetCardPosition(Key);
			}

			if (NextListUpdateIsAnim) {
				foreach (Label in Label_ContactListHeaders) {
					AnimOpacity(Label);
				}
			}

			NextListUpdateIsAnim = False;
		}

		if (PrevEntrySearchContactsValue != Entry_SearchContacts.Value) {
			PrevEntrySearchContactsValue = Entry_SearchContacts.Value;
			
			Quad_SearchButtonGo.Visible = Entry_SearchContacts.Value == "";
			Quad_SearchButtonClear.Visible = !Quad_SearchButtonGo.Visible;
		}
	}
}

--></script>
</manialink>
	""";
}
