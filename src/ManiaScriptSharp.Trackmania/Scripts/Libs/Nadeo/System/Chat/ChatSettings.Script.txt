
#Const C_LibChatSettings_DefaultCustomStatusText			""
#Const C_LibChatSettings_DefaultUseRichPresenceStatus		True
#Const C_LibChatSettings_DefaultNotificationMessageLevel	2
#Const C_LibChatSettings_DefaultNotificationOnFriendOnline	True
#Const C_LibChatSettings_DefaultNotificationOnServerJoin	True
#Const C_LibChatSettings_DefaultDetectOutgoingEmbeds		True
#Const C_LibChatSettings_DefaultDisplayEmbedsInMessages		True
#Const C_LibChatSettings_DefaultDisplayEmbedImages			False
#Const C_LibChatSettings_DefaultOutgoingMessagesLeftSide	False
#Const C_LibChatSettings_DefaultChatWindowLeftSide			True

#Const C_LibChatSettings_MessageLevel_Nobody	0
#Const C_LibChatSettings_MessageLevel_Friends	1
#Const C_LibChatSettings_MessageLevel_Squad		2
#Const C_LibChatSettings_MessageLevel_Everyone	3

#Struct SChatPluginSettings {
	Boolean IsInitialized;

	Text CustomStatusText;
	Boolean UseRichPresenceStatus;

	Integer NotificationMessageLevel;
	Boolean NotificationOnFriendOnline;
	Boolean NotificationOnServerJoin;

	Boolean DetectOutgoingEmbeds;
	Boolean DisplayEmbedsInMessages;
	Boolean DisplayEmbedImages;

	Boolean OutgoingMessagesLeftSide;
	Boolean ChatWindowLeftSide;
}

Void Private_TriggerUpdateInChat() {
	declare Boolean LibChatSettings_IsPendingUpdate for ManiaPlanet;
	LibChatSettings_IsPendingUpdate = True;
}

SChatPluginSettings GetSettings() {
	if (!ManiaPlanet.PersistentIsReady) {
		declare SChatPluginSettings Default;
                Default.CustomStatusText           = C_LibChatSettings_DefaultCustomStatusText;
                Default.UseRichPresenceStatus      = C_LibChatSettings_DefaultUseRichPresenceStatus;
                Default.NotificationMessageLevel   = C_LibChatSettings_DefaultNotificationMessageLevel;
                Default.NotificationOnFriendOnline = C_LibChatSettings_DefaultNotificationOnFriendOnline;
                Default.NotificationOnServerJoin   = C_LibChatSettings_DefaultNotificationOnServerJoin;
                Default.DisplayEmbedsInMessages    = C_LibChatSettings_DefaultDisplayEmbedsInMessages;
                Default.DisplayEmbedImages         = C_LibChatSettings_DefaultDisplayEmbedImages;
                Default.DetectOutgoingEmbeds       = C_LibChatSettings_DefaultDetectOutgoingEmbeds;
                Default.OutgoingMessagesLeftSide   = C_LibChatSettings_DefaultOutgoingMessagesLeftSide;
                Default.ChatWindowLeftSide         = C_LibChatSettings_DefaultChatWindowLeftSide;
		return Default;
	}

	declare persistent SChatPluginSettings Persistent_LibChatSettings_Settings2 for ManiaPlanet;

	// Initialize settings 
	if (!Persistent_LibChatSettings_Settings2.IsInitialized) {
		Persistent_LibChatSettings_Settings2.IsInitialized = True;

		Persistent_LibChatSettings_Settings2.CustomStatusText			= C_LibChatSettings_DefaultCustomStatusText;
		Persistent_LibChatSettings_Settings2.UseRichPresenceStatus		= C_LibChatSettings_DefaultUseRichPresenceStatus;
		Persistent_LibChatSettings_Settings2.NotificationMessageLevel	= C_LibChatSettings_DefaultNotificationMessageLevel;
		Persistent_LibChatSettings_Settings2.NotificationOnFriendOnline	= C_LibChatSettings_DefaultNotificationOnFriendOnline;
		Persistent_LibChatSettings_Settings2.NotificationOnServerJoin	= C_LibChatSettings_DefaultNotificationOnServerJoin;
		Persistent_LibChatSettings_Settings2.DisplayEmbedsInMessages	= C_LibChatSettings_DefaultDisplayEmbedsInMessages;
		Persistent_LibChatSettings_Settings2.DisplayEmbedImages			= C_LibChatSettings_DefaultDisplayEmbedImages;
		Persistent_LibChatSettings_Settings2.DetectOutgoingEmbeds		= C_LibChatSettings_DefaultDetectOutgoingEmbeds;
		Persistent_LibChatSettings_Settings2.OutgoingMessagesLeftSide	= C_LibChatSettings_DefaultOutgoingMessagesLeftSide;
		Persistent_LibChatSettings_Settings2.ChatWindowLeftSide			= C_LibChatSettings_DefaultChatWindowLeftSide;
	}

	return Persistent_LibChatSettings_Settings2;
}

Void SaveSettings(SChatPluginSettings _Settings) {
	declare persistent SChatPluginSettings Persistent_LibChatSettings_Settings2 for ManiaPlanet;
	Persistent_LibChatSettings_Settings2 = _Settings;
}

Void ResetSettings() {
	declare persistent SChatPluginSettings Persistent_LibChatSettings_Settings2 for ManiaPlanet;
	Persistent_LibChatSettings_Settings2.IsInitialized = False;

	// Forces settings initialization (aka reset)
	declare NewSettings = GetSettings();
}

Text CustomStatusText() {
	return GetSettings().CustomStatusText;
}

	return GetSettings().UseRichPresenceStatus;
}

	return GetSettings().NotificationMessageLevel;
}

Boolean NotifyOnFriendMessage() {
	return NotificationMessageLevel() >= C_LibChatSettings_MessageLevel_Friends;
}

Boolean NotifyOnSquadMessage() {
	return NotificationMessageLevel() >= C_LibChatSettings_MessageLevel_Squad;
}

Boolean NotifyOnAnyMessage() {
	return NotificationMessageLevel() >= C_LibChatSettings_MessageLevel_Everyone;
}

Void SetCustomStatusText(Text _NewValue) {
	declare Settings = GetSettings();
	Settings.CustomStatusText = _NewValue;
	SaveSettings(Settings);
	Private_TriggerUpdateInChat();
}
