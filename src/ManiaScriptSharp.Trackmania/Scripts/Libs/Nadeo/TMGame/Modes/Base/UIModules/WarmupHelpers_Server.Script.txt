#Const WarmupHelpers::C_Mode_TimeAttack as C_Mode_TimeAttack
#Const WarmupHelpers::C_Mode_Rounds as C_Mode_Rounds

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Set the warmup status
 *
 *	@param	_IsActive				The warmup status
 */
Void SetWarmupActive(Boolean _IsActive) {
	declare netwrite Boolean Net_Race_WarmupHelpers_IsWarmupActive for Teams[0];
	Net_Race_WarmupHelpers_IsWarmupActive = _IsActive;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Set the current warmup end time
 *
 *	@param	_EndTime				The current warmup end time
 */
Void SetCurrentWarmupEndTime(Integer _EndTime) {
	declare netwrite Integer Net_Race_WarmupHelpers_CurrentWarmUpEndTime for Teams[0];
	Net_Race_WarmupHelpers_CurrentWarmUpEndTime = _EndTime;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the mode used by the warm up
Void SetWarmUpMode(Integer _Mode) {
	declare netwrite Integer Net_Race_WarmupHelpers_WarmUpMode for Teams[0] = C_Mode_TimeAttack;
	Net_Race_WarmupHelpers_WarmUpMode = _Mode;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set warm-up round progression
Void SetWarmUpProgression(Integer _Current, Integer _Total) {
	declare netwrite Int2 Net_Race_WarmupHelpers_WarmUpProgression for Teams[0] = <0, 0>;
	Net_Race_WarmupHelpers_WarmUpProgression = <_Current, _Total>;
}
