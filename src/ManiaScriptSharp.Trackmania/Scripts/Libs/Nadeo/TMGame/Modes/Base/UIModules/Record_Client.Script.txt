Void SetMedal(K_Controls _Controls, Integer _Medal) {
	declare CMlFrame Frame_MedalBanner <=> (_Controls.Frame_Medal.GetFirstChild("frame-medal-banner") as CMlFrame);
	declare CMlFrame MedalStack_Current = (_Controls.Frame_Medal.GetFirstChild("medal-stack-current") as CMlFrame);
	declare CMlLabel Label_MedalName <=> (_Controls.Frame_Medal.GetFirstChild("label-medal-name") as CMlLabel);
	declare CMlQuad Quad_MedalNext <=> (_Controls.Frame_Medal.GetFirstChild("quad-medal-next") as CMlQuad);
	declare CMlLabel Label_MedalTime <=> (_Controls.Frame_Medal.GetFirstChild("label-medal-time") as CMlLabel);
	declare CMlQuad Quad_Medal <=> (_Controls.Frame_Medal.GetFirstChild("quad-medal") as CMlQuad);
	declare Integer MedalTime = -1;
	declare Integer MedalNext = Medals::C_ScoreMgrMedal_None;

	{{{MedalStack::P}}}SetMedalsNumber(MedalStack_Current, _Medal); //@TODO @QG remove if needed (perma hidden for now)
	SetMedalImage(Quad_Medal, Frame_MedalBanner, _Medal);
	MedalStack_Current.Hide();

	switch (_Medal) {
		case Medals::C_ScoreMgrMedal_None: {
			if (Map != Null) {
				MedalTime = Map.MapInfo.TMObjective_BronzeTime;
				MedalNext = Medals::C_ScoreMgrMedal_Bronze;
			}
		}
		case Medals::C_ScoreMgrMedal_Bronze: {
			if (Map != Null) {
				MedalTime = Map.MapInfo.TMObjective_SilverTime;
				MedalNext = Medals::C_ScoreMgrMedal_Silver;
			}
		}
		case Medals::C_ScoreMgrMedal_Silver: {
			if (Map != Null) {
				MedalTime = Map.MapInfo.TMObjective_GoldTime;
				MedalNext = Medals::C_ScoreMgrMedal_Gold;
			}
		}
	}

	if (MedalTime >= 0) {
		Label_MedalName.Value = "{{{Text_NextMedal}}}";
		Tools::FitLabelValue(Label_MedalName, 2., 0.5, 0.25);
		Quad_MedalNext.ChangeImageUrl(GetMedalImage(MedalNext));
		Quad_MedalNext.Visible = True;
		if (Map != Null) {
			Label_MedalTime.Value = CampaignStruct::GetMedalScoreText(MedalTime, Map.MapType);
		} else {
			Label_MedalTime.Value = "";
		}
	} else {
		Label_MedalName.Value = "";
		Quad_MedalNext.Visible = False;
		Label_MedalTime.Value = "";
	}

Void Focus(CMlControl _From, CMlControl _To, K_State _State, K_Controls _Controls, Boolean _IsMouse, Boolean _ForceFocusUpdate) {
	declare Boolean IsNavigationControlUnfocusPossible = !(_State.IsControllerNavigationEnabled && (_To == Null || {{{Checkbox::P}}}IsCheckbox(_To)));

	if (_From != Null) {
		declare CMlControl NavigationControl = GetParentNavigationControl(_From);
		if ({{{Navigation::P}}}IsNavigationControl(NavigationControl) && IsNavigationControlUnfocusPossible) { //< Always keep a record focused if controller is enabled
			if (IsRecordFocusPossible(NavigationControl, _State, _IsMouse, _ForceFocusUpdate)) {
				{{{Navigation::P}}}Unfocus(This, NavigationControl);
				if (TL::StartsWith("button-record-", NavigationControl.ControlId)) {
					FocusRecord(NavigationControl as CMlFrame, _State.SpectatorTargetAccountId, False);
					if (!_IsMouse) FocusToggleGhost((NavigationControl as CMlFrame).GetFirstChild("button-toggle-ghost") as CMlQuad, False);
				} else if ({{{Scroll::P}}}IsScroll(NavigationControl)) {
					{{{Scroll::P}}}Unfocus(This, (NavigationControl as CMlFrame));
				} else if ({{{ExpendableButton::P}}}IsButton(NavigationControl)) {
					{{{ExpendableButton::P}}}Unfocus(This, (NavigationControl as CMlFrame));
				}
			}
		}
		if ({{{Checkbox::P}}}IsCheckbox(_From)) {
			{{{Checkbox::P}}}Unfocus(This, _From as CMlFrame);
		} else if (_From.ControlId == "button-toggle-ghost") {
			FocusToggleGhost(_From as CMlQuad, False);
		}
	}

	if (_To != Null) {
		declare CMlControl NavigationControl = GetParentNavigationControl(_To);
		if ({{{Navigation::P}}}IsNavigationControl(NavigationControl)) {
			if (IsRecordFocusPossible(NavigationControl, _State, _IsMouse, _ForceFocusUpdate)) {
				{{{Navigation::P}}}Focus(This, NavigationControl);
				if (TL::StartsWith("button-record-", NavigationControl.ControlId)) {
					FocusRecord(NavigationControl as CMlFrame, _State.SpectatorTargetAccountId, True);
					UpdateControllerInfo(_State, _Controls, NavigationControl);
				} else if ({{{Scroll::P}}}IsScroll(NavigationControl)) {
					{{{Scroll::P}}}Focus(This, (NavigationControl as CMlFrame));
				} else if ({{{ExpendableButton::P}}}IsButton(NavigationControl)) {
					{{{ExpendableButton::P}}}Focus(This, (NavigationControl as CMlFrame));
				}
			}
		}
		if ({{{Checkbox::P}}}IsCheckbox(_To)) {
			{{{Checkbox::P}}}Focus(This, _To as CMlFrame);
		} else if (_To.ControlId == "button-toggle-ghost") {
			FocusToggleGhost(_To as CMlQuad, True);
		}
	}
}
Void Focus(CMlControl _From, CMlControl _To, K_State _State, K_Controls _Controls, Boolean _IsMouse) {
	Focus(_From, _To, _State, _Controls, _IsMouse, False);
}

	switch (_Control.ControlId) {
		case "button-back-to-race": SendCustomEvent("{{{C_Event_SpectateGhost}}}", [_State.SpectatorTargetAccountId]);
	}
}

Boolean UpdateCelebration(K_Controls _Controls, K_Celebration _Celebration) {
	declare CMlLabel Label_NewMedal <=> (_Controls.Frame_Celebration.GetFirstChild("label-new-medal") as CMlLabel);
	declare CMlLabel Label_NewScore <=> (_Controls.Frame_Celebration.GetFirstChild("label-new-score") as CMlLabel);
	declare CMlLabel Label_NewScoreDiff <=> (_Controls.Frame_Celebration.GetFirstChild("label-new-score-difference") as CMlLabel);
	declare CMlQuad Quad_CelebrationBg <=> (_Controls.Frame_Celebration.GetFirstChild("quad-celebration-background") as CMlQuad);

	declare NewTimeDiff = False;

	declare Boolean SortByAscendingOrder = True;
	if (Map != Null && TitleControl::IsStuntMapType(Map.MapType)) SortByAscendingOrder = False;

	if (
		Map != Null &&
		_Celebration.IsActive &&
		_Celebration.BestScore >= 0 &&
		(
			_Celebration.PrevScore < 0 ||
			(
				(SortByAscendingOrder && _Celebration.BestScore < _Celebration.PrevScore) ||
				(!SortByAscendingOrder && _Celebration.BestScore > _Celebration.PrevScore)
			)
		)
	) {
		Label_NewScore.Value = CampaignStruct::GetMedalScoreText(_Celebration.BestScore, Map.MapType);
		Label_NewScoreDiff.Hide();
		if (_Celebration.PrevScore >= 0) {
			declare ScoreDiff = "";
			declare Integer Diff = _Celebration.BestScore - _Celebration.PrevScore;
			switch (Map.MapType) {
				case Const::C_MapType_Stunt, Const::C_MapType_Stunt_NoConcat: {
					ScoreDiff = "+" ^ TL::GetTranslatedText(TL::Compose(Const::C_Text_RecordPoints, TL::FormatInteger(Diff, 1, True)));
				}
				case Const::C_MapType_Platform, Const::C_MapType_Platform_NoConcat: {
					if (Diff == -1) ScoreDiff = "-" ^ TL::GetTranslatedText(Const::C_Text_RecordRespawn1);
					else ScoreDiff = TL::GetTranslatedText(TL::Compose(Const::C_Text_RecordRespawns, TL::ToText(Diff)));
				}
				default: {
					ScoreDiff = TL::TimeToText(_Celebration.BestScore - _Celebration.PrevScore, True, True);
				}
			}
			Label_NewScoreDiff.SetText(ScoreDiff);
			NewTimeDiff = True;
		}
		_Controls.Frame_CelebrationTime.Visible = True;
	} else {
		_Controls.Frame_CelebrationTime.Visible = False;
	}

	if (
		_Celebration.IsActive &&
		Medals::ABetterThanB(
			Medals::ConvertFromScoreMgrMedal(_Celebration.BestMedal),
			Medals::ConvertFromScoreMgrMedal(_Celebration.PrevMedal)
		)
	) {
		Label_NewMedal.Value = Medals::GetScoreMgrMedalName(_Celebration.BestMedal);
		_Controls.Frame_CelebrationMedal.Visible = True;
	} else {
		_Controls.Frame_CelebrationMedal.Visible = False;
	}

	if (_Controls.Frame_CelebrationTime.Visible && _Controls.Frame_CelebrationMedal.Visible) {
		_Controls.Frame_CelebrationTime.RelativePosition_V3.Y = {{{-PosY_Celebration}}};
		_Controls.Frame_CelebrationMedal.RelativePosition_V3.Y = {{{PosY_Celebration}}};
	} else {
		_Controls.Frame_CelebrationTime.RelativePosition_V3.Y = 0.;
		_Controls.Frame_CelebrationMedal.RelativePosition_V3.Y = 0.;
	}

	if (_Controls.Frame_CelebrationTime.Visible || _Controls.Frame_CelebrationMedal.Visible) {
		UIModules_PrestigeEarned_Common::BlockPrestigeCelebration(UI, "{{{Record::C_Id}}}");
		AnimMgr.Add(Quad_CelebrationBg, "<a opacity=\"0.7\" hidden=\"0\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		if (_Controls.Frame_CelebrationTime.Visible) {
			// Label anim
			Label_NewScore.RelativePosition_V3 = <140., -9.>;
			Label_NewScore.Hide();
			AnimMgr.Flush(Label_NewScore);
			AnimMgr.Add(Label_NewScore, "<a pos=\"0 -9\" hidden=\"0\" />", Now + 50, 150, CAnimManager::EAnimManagerEasing::QuadIn);
			if (NewTimeDiff) {
				Label_NewScoreDiff.RelativePosition_V3 = <140., -19.>;
				Label_NewScoreDiff.Hide();
				AnimMgr.Flush(Label_NewScoreDiff);
				AnimMgr.Add(Label_NewScoreDiff, "<a pos=\"0 -19\" hidden=\"0\" />", Now + 100, 150, CAnimManager::EAnimManagerEasing::QuadIn);
			}
		}
		if (_Controls.Frame_CelebrationMedal.Visible) {
			// Label anim
			Label_NewMedal.RelativePosition_V3 = <140., -9.>;
			Label_NewMedal.Hide();
			AnimMgr.Flush(Label_NewMedal);
			AnimMgr.Add(Label_NewMedal, "<a pos=\"0 -9\" hidden=\"0\" />", Now, 150, CAnimManager::EAnimManagerEasing::QuadIn);

			// Medal anim
			declare CMlFrame Frame_MedalBannerCelebration <=> (_Controls.Frame_CelebrationMedal.GetFirstChild("frame-medal-banner-celebration") as CMlFrame);
			declare CMlQuad Quad_BannerGlow <=> (_Controls.Frame_CelebrationMedal.GetFirstChild("quad-medal-banner-glow") as CMlQuad);

			declare GlowStartTime = Now;

			declare WaitTime1 = 250;
			declare PunchTime = 125;
			declare WaitTime2 = 1200;
			declare TravelTime = 200;
			declare TotalDuration = 3000;

			declare BgTravelEasing = CAnimManager::EAnimManagerEasing::QuadIn ;
			declare PunchEasing = CAnimManager::EAnimManagerEasing::QuadIn ;

			declare FadeEasing = CAnimManager::EAnimManagerEasing::QuadOut ;
			declare GlowEasing = CAnimManager::EAnimManagerEasing::Linear ;

			declare Text MedalGainSound;
			switch (_Celebration.BestMedal) {
				case Medals::C_ScoreMgrMedal_Bronze: {
					MedalGainSound = "{{{C_Sound_MedalGainBronze}}}";
				}
				case Medals::C_ScoreMgrMedal_Silver: {
					MedalGainSound = "{{{C_Sound_MedalGainSilver}}}";
				}
				case Medals::C_ScoreMgrMedal_Gold: {
					MedalGainSound = "{{{C_Sound_MedalGainGold}}}";
				}
				case Medals::C_ScoreMgrMedal_Author: {
					MedalGainSound = "{{{C_Sound_MedalGainAuthor}}}";
				}
			}

			Quad_Medal.RelativeRotation = ML::Rand(2.,4.);
			Quad_Medal.RelativeScale = 1.5;
			Quad_Medal.RelativePosition_V3 = <5., 55.>;
			Quad_Medal.Hide();

			Frame_MedalBannerCelebration.RelativeScale = 1.;
			StartTime += WaitTime0;
			AnimMgr.Flush(Frame_MedalBannerCelebration);

			AnimMgr.Add(Frame_MedalBannerCelebration, "<a rot=\"-80\" />", StartTime + 40, 40, CAnimManager::EAnimManagerEasing::Linear);

Void SetFrameRecord(K_State _State, CMlFrame _Frame_Record, K_TMGame_Record_Record _Record, Integer[Text] _RecordsStatus, Integer _RecordType) {
	if (_Frame_Record == Null) return;

	declare CMlFrame LoadingSpinner_Rank <=> (_Frame_Record.GetFirstChild("loading-spinner-rank") as CMlFrame);
	declare CMlFrame LoadingSpinner_ToggleGhost <=> (_Frame_Record.GetFirstChild("loading-spinner-toggle-ghost") as CMlFrame);
	declare CMlFrame PlayerName_Name = (_Frame_Record.GetFirstChild("playername-name") as CMlFrame);
	declare CMlLabel Label_Rank <=> (_Frame_Record.GetFirstChild("label-rank") as CMlLabel);
	declare CMlLabel Label_Score <=> (_Frame_Record.GetFirstChild("label-score") as CMlLabel);
	declare CMlQuad Button_ToggleGhost <=> (_Frame_Record.GetFirstChild("button-toggle-ghost") as CMlQuad);
	declare CMlQuad Quad_Bg <=> (_Frame_Record.GetFirstChild("quad-bg") as CMlQuad);
	declare CMlQuad Quad_Icon <=> (_Frame_Record.GetFirstChild("quad-icon") as CMlQuad);
	declare CMlQuad Quad_Medal <=> (_Frame_Record.GetFirstChild("quad-medal") as CMlQuad);

	declare K_TMGame_Record_Record RecordClient_PlayerRecord for _Frame_Record = C_TMGame_Record_Record_Null;
	RecordClient_PlayerRecord = _Record;

	declare Boolean CanPlayAgainstRecords = (
		{{{UserStore_ML::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayRecords) &&
		!TL::StartsWith(C_FakeAccountId_MedalGhost, RecordClient_PlayerRecord.AccountId) &&
		!_State.GhostChoiceAccountIdList.exists(RecordClient_PlayerRecord.AccountId) &&
		RecordClient_PlayerRecord.AccountId != "" &&
		RecordClient_PlayerRecord.Score >= 0
	);

	// Update rank label
	Label_Rank.Visible = _Record.Rank > 0 && _RecordType != C_RecordType_Medals;
	{{{LoadingSpinner::P}}}SetIsLoading(This, LoadingSpinner_Rank, _Record.Rank <= 0 && _RecordType != C_RecordType_Medals);
	if (Label_Rank.Visible) {
		Label_Rank.SetText(Tools::FormatTop(_Record.Rank, False, 10000));
		Tools::FitLabelValue(Label_Rank, 1., 0.25, 0.25);
	}

	// Update medal and icon quad
	Quad_Icon.Visible = _RecordType == C_RecordType_Medals && _Record.Rank < 0 && _Record.AccountId == LocalUser.WebServicesUserId;
	Quad_Medal.Visible = _RecordType == C_RecordType_Medals && _Record.Rank >= 0;
	if (Quad_Medal.Visible) {
		Quad_Medal.ChangeImageUrl(GetMedalImage(Medals::C_ScoreMgrMedal_Author - _Record.Rank));
	}

	// Update time label
	if ({{{UserStore_ML::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayRecords)) {
		Label_Score.RelativePosition_V3.X = {{{Grid_Record.Pos[C_Col_Score]}}};
	} else {
		Label_Score.RelativePosition_V3.X = {{{Grid_Record.Pos[C_Col_ToggleGhost] + (Grid_Record.Size[C_Col_ToggleGhost] * 0.5)}}};
	}
	if (Map != Null) {
		if (Quad_Medal.Visible) {
			Label_Score.Value = CampaignStruct::GetMedalScoreText(Medals::GetScoreFromMedal(Map, Medals::C_ScoreMgrMedal_Author - _Record.Rank), Map.MapType, True);
		} else {
			declare CampaignStruct::K_ScoreTextOptions Options = CampaignStruct::C_ScoreTextOptions_Null;
			Options.EnableAbbreviation = True;
			Options.EnableHiddenScore = True;
			Label_Score.Value = CampaignStruct::GetMedalScoreText(_Record.Score, Map.MapType, Options);
		}
		Tools::FitLabelValue(Label_Score, 1., 0.5, 0.1);
	} else {
		Label_Score.Value = "";
	}

	// Update player name
	if ({{{UserStore_ML::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayRecords)) {
		{{{PlayerName::P}}}SetSize(This, PlayerName_Name, <{{{Grid_Record.Size[C_Col_Name]}}}, {{{Size_RecordRow.Y - 2}}}>);
	} else {
		{{{PlayerName::P}}}SetSize(This, PlayerName_Name, <{{{Grid_Record.Size[C_Col_Name] + Grid_Record.Size[C_Col_ToggleGhost]}}}, {{{Size_RecordRow.Y - 2}}}>);
	}
	if (Quad_Medal.Visible) {
		{{{PlayerName::P}}}SetName(This, PlayerName_Name, Medals::GetScoreMgrMedalName(Medals::C_ScoreMgrMedal_Author - _Record.Rank), "", False, False);
	} else {
		{{{PlayerName::P}}}SetAccountId(This, PlayerName_Name, _Record.AccountId, False);
	}

	// Update the button to toggle the ghost
	if ((_Record.Rank <= 0 && _Record.AccountId != LocalUser.WebServicesUserId) || !CanPlayAgainstRecords) {
		Button_ToggleGhost.Visible = False;
		{{{LoadingSpinner::P}}}StopAnimation(This, LoadingSpinner_ToggleGhost);
	} else if (!_RecordsStatus.existskey(_Record.AccountId)) {
		Button_ToggleGhost.Colorize = <1., 1., 1.>;
		Button_ToggleGhost.ImageUrl = "{{{Image_Add}}}";
		Button_ToggleGhost.Opacity = 0.5;
		Button_ToggleGhost.Visible = True;
		{{{LoadingSpinner::P}}}StopAnimation(This, LoadingSpinner_ToggleGhost);
	} else if (_RecordsStatus[_Record.AccountId] == {{{C_RecordStatus_Loading}}}) {
		Button_ToggleGhost.Visible = False;
		{{{LoadingSpinner::P}}}StartAnimation(This, LoadingSpinner_ToggleGhost);
	} else if (_RecordsStatus[_Record.AccountId] == {{{C_RecordStatus_Loaded}}} || _RecordsStatus[_Record.AccountId] == {{{C_RecordStatus_Followed}}}) {
		Button_ToggleGhost.Colorize = {{{Stylesheet::GetColorRGB(Stylesheet::C_Color_AccentOne)}}};
		Button_ToggleGhost.ImageUrl = "{{{Image_Remove}}}";
		Button_ToggleGhost.Opacity = 1.;
		Button_ToggleGhost.Visible = True;
		{{{LoadingSpinner::P}}}StopAnimation(This, LoadingSpinner_ToggleGhost);
	}

	// Save player record for report system in Profile
	if (Map != Null && Map.MapInfo != Null)	{
		declare Text RecordClient_PlayerRecordDetail for _Frame_Record = "";
		RecordClient_PlayerRecordDetail = ReportStruct::FormatRecordDetail(_Record.DisplayName, _Record.Score, Map.MapInfo.MapType);

		declare Text RecordClient_PlayerRecordId for _Frame_Record = "";
		declare netread Text[Text] Net_TMGame_Record_SeasonIds for Teams[0] = [];
		RecordClient_PlayerRecordId = ReportStruct::FormatRecordId(_Record.AccountId, Net_TMGame_Record_SeasonIds.get(Map.MapInfo.MapUid, ""), Map.MapInfo.MapUid);
	}

	FocusRecord(_Frame_Record, _State.SpectatorTargetAccountId, {{{Navigation::P}}}IsFocused(This, _Frame_Record, "{{{C_NavGroupName}}}")); //< Keep the focus if the record was focused before the black screen
}

K_TMGame_Record_Records SetRecords(
	K_State _State,
	K_Controls _Controls,
	K_TMGame_Record_Records[] _ZonesRecords,
	Boolean _IsLoading,
	Integer[Text] _RecordsStatus,
	Integer[] _RecordsErrorCodes,
	Boolean _MapAvailaibleOnNadeoServices,
	Boolean _CanViewLeaderboards
) {
	declare CMlFrame Frame_Loading <=> (_Controls.Frame_Records.GetFirstChild("frame-loading") as CMlFrame);
	declare CMlFrame LoadingSpinner_Records <=> (_Controls.Frame_Records.GetFirstChild("loading-spinner-records") as CMlFrame);
	declare CMlFrame Frame_RankingsContainer <=> (_Controls.Frame_Records.GetFirstChild("frame-rankings-container") as CMlFrame);
	declare Boolean CanPlayAgainstRecords = {{{UserStore_ML::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayRecords);

	declare K_TMGame_Record_Records ZoneRecords;
	if (_ZonesRecords.existskey(_State.SelectedZone)) {
		ZoneRecords = _ZonesRecords[_State.SelectedZone];
	} else if (_ZonesRecords.count > 0) {
		ZoneRecords = _ZonesRecords[0];
	}

	declare Integer TMGame_Record_RecordsSelectedZone for ClientUI = C_RecordType_None;
	if (_ZonesRecords.count <= 0) {
		TMGame_Record_RecordsSelectedZone = C_RecordType_None;
	} else if (ZoneRecords.Type == C_RecordType_Club) {
		TMGame_Record_RecordsSelectedZone = C_RecordType_Club;
	} else if (ZoneRecords.Type == C_RecordType_ClubVIP) {
		TMGame_Record_RecordsSelectedZone = C_RecordType_ClubVIP;
	} else if (ZoneRecords.Type == C_RecordType_GlobalVIP) {
		TMGame_Record_RecordsSelectedZone = C_RecordType_GlobalVIP;
	} else if (ZoneRecords.Type == C_RecordType_Medals) {
		TMGame_Record_RecordsSelectedZone = C_RecordType_Medals;
	} else {
		TMGame_Record_RecordsSelectedZone = C_RecordType_Zones;
	}

	Frame_Loading.Visible = _IsLoading;
	Frame_RankingsContainer.Visible = !_IsLoading && _MapAvailaibleOnNadeoServices && ZoneRecords.Records.count > 0;
	_Controls.Frame_StandardRequired.Visible = !_IsLoading && ZoneRecords.Records.count <= 0 && !HasRecordPermission(ZoneRecords.Type);
	_Controls.Frame_MissingPrivilege.Visible = !_IsLoading && !_CanViewLeaderboards && !_Controls.Frame_StandardRequired.Visible;
	_Controls.Frame_MapNotAvailable.Visible = !_IsLoading && !_MapAvailaibleOnNadeoServices && !_Controls.Frame_MissingPrivilege.Visible && !_Controls.Frame_StandardRequired.Visible;
	_Controls.Frame_NoRecords.Visible = !_IsLoading && _MapAvailaibleOnNadeoServices && ZoneRecords.Records.count <= 0 && !_Controls.Frame_StandardRequired.Visible;
	_Controls.Frame_ScrollRanking.Visible = (Frame_RankingsContainer.Visible && (ZoneRecords.Type == C_RecordType_ClubVIP || ZoneRecords.Type == C_RecordType_GlobalVIP));
	_Controls.Frame_Ranking.Visible = (Frame_RankingsContainer.Visible && !_Controls.Frame_ScrollRanking.Visible);
	_Controls.Frame_Zone.Visible = !_IsLoading && _MapAvailaibleOnNadeoServices && _ZonesRecords.count >= 1;
	_Controls.Frame_TogglePB.Visible = (
		CanPlayAgainstRecords &&
		(_Controls.Frame_ScrollRanking.Visible || _Controls.Frame_Ranking.Visible) &&

Text SaveSelectedZone(K_TMGame_Record_Records[] _ZonesRecords, Integer _SelectedZone) {
	if (_ZonesRecords.existskey(_SelectedZone)) {
		if (_ZonesRecords[_SelectedZone].Type == C_RecordType_Club) {
			return C_FakeZone_Club;
		} else if (_ZonesRecords[_SelectedZone].Type == C_RecordType_ClubVIP) {
			return C_FakeZone_ClubVIP;
		} else if (_ZonesRecords[_SelectedZone].Type == C_RecordType_GlobalVIP) {
			return C_FakeZone_GlobalVIP;
		} else if (_ZonesRecords[_SelectedZone].Type == C_RecordType_Medals) {
			return C_FakeZone_Medals;
		} else {
			return _ZonesRecords[_SelectedZone].ZoneName;
		}
	}
	return "";
}

Integer LoadSelectedZone(K_TMGame_Record_Records[] _ZonesRecords, Text _SavedZone) {
	foreach (Key => ZoneRecords in _ZonesRecords) {
		if (_SavedZone == C_FakeZone_Club) {
			if (ZoneRecords.Type == C_RecordType_Club) return Key;
		} else if (_SavedZone == C_FakeZone_ClubVIP) {
			if (ZoneRecords.Type == C_RecordType_ClubVIP) return Key;
		} else if (_SavedZone == C_FakeZone_GlobalVIP) {
			if (ZoneRecords.Type == C_RecordType_GlobalVIP) return Key;
		} else if (_SavedZone == C_FakeZone_Medals) {
			if (ZoneRecords.Type == C_RecordType_Medals) return Key;
		} else if (_SavedZone == ZoneRecords.ZoneName) {
			return Key;
		}
	}
	return 0;
}

Void ToggleGhost(K_State _State, CMlControl _Control) {
	declare K_TMGame_Record_Record RecordClient_PlayerRecord for _Control = C_TMGame_Record_Record_Null;
	if (!_State.GhostChoiceAccountIdList.exists(RecordClient_PlayerRecord.AccountId)) {
		if (RecordClient_PlayerRecord.AccountId == _State.SpectatorTargetAccountId && InputPlayer != Null && InputPlayer.SpawnStatus != CSmPlayer::ESpawnStatus::Spawned)	{
			SendCustomEvent("{{{C_Event_SpectateGhost}}}", [RecordClient_PlayerRecord.AccountId, ""^RecordClient_PlayerRecord.Rank, ""^RecordClient_PlayerRecord.Score]);
		} else {
			SendCustomEvent("{{{C_Event_ToggleGhost}}}", [RecordClient_PlayerRecord.AccountId, ""^RecordClient_PlayerRecord.Rank, ""^RecordClient_PlayerRecord.Score]);
		}
	}
}

K_State ChangeZone(
	K_State _State,
	K_Controls _Controls,
	K_TMGame_Record_Records[] _ZonesRecords,
	Boolean _IsLoading,
	Integer[Text] _RecordsStatus,
	Integer[] _RecordsErrorCodes,
	Boolean _MapAvailaibleOnNadeoServices,
	Boolean _CanViewLeaderboards,
	Boolean _IsMouse,
	Integer _ZoneShift
) {
	declare K_State State = _State;

	// Update current zone
	State.SelectedZone = SelectZone(_Controls, _ZonesRecords, State.SelectedZone + _ZoneShift);
	State.CurrentZoneRecords = SetRecords(State, _Controls, _ZonesRecords, _IsLoading, _RecordsStatus, _RecordsErrorCodes, _MapAvailaibleOnNadeoServices, _CanViewLeaderboards);

	// Update focus
	if (!_IsMouse || State.IsControllerNavigationEnabled) {
		if (State.CurrentZoneRecords.Type == C_RecordType_ClubVIP || State.CurrentZoneRecords.Type == C_RecordType_GlobalVIP) { //< Focus scroll if VIP records
			Focus({{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroupName}}}"), _Controls.Scroll_Ranking, State, _Controls, _IsMouse, True);
		} else {
			declare CMlControl ControlToFocus;
			if (_Controls.Frame_Ranking.Controls.count > 0) ControlToFocus = _Controls.Frame_Ranking.Controls[0]; //< Set a default focus in case there is no local user's record
			foreach (Key => Control in _Controls.Frame_Ranking.Controls) { //< Try to focus the local user's record
				if (State.CurrentZoneRecords.Records.existskey(Key) && State.CurrentZoneRecords.Records[Key].AccountId == LocalUser.WebServicesUserId) {
					ControlToFocus = Control;
					break;
				}
			}
			Focus({{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroupName}}}"), ControlToFocus, State, _Controls, _IsMouse, True);
		}
	}

	return State;
}

/// Get the scope of the current state
K_Scope GetScope(K_State _State) {
	declare K_Scope Scope = K_Scope {
		Type = _State.ScopeNotSeason,
		Id = ""
	};
	if (_State.SeasonIds.existskey(_State.CurrentMapUid)) {
		Scope.Type = _State.ScopeSeason;
		Scope.Id = _State.SeasonIds[_State.CurrentMapUid];
	}
	return Scope;
}

K_State AddRecordsError(K_State _State, Integer _ErrorCode) {
	declare K_State State = _State;
	if (!State.RecordsErrorCodes.exists(_ErrorCode)) {
		State.RecordsErrorCodes.add(_ErrorCode);
	}
	return State;
}

/// Send the records status to the manialink UI
Void SendRecordsStatusToML(K_State _State) {
	declare Integer[Text] TMGame_Record_RecordsStatus for ClientUI;
	declare Integer TMGame_Record_RecordsStatusUpdate for ClientUI;

	TMGame_Record_RecordsStatus = [];

	foreach (AccountId => Record in _State.LoadingRecordGhosts) {
		TMGame_Record_RecordsStatus[AccountId] = C_RecordStatus_Loading;
	}
	foreach (AccountId => Record in _State.LoadedRecordGhosts) {
		TMGame_Record_RecordsStatus[AccountId] = C_RecordStatus_Loaded;
	}
	if (_State.SpectatorTargetRecordGhost.AccountId != "") {
		TMGame_Record_RecordsStatus[_State.SpectatorTargetRecordGhost.AccountId] = C_RecordStatus_Followed;
	}

	TMGame_Record_RecordsStatusUpdate += 1;
}

/// Show/hide the PB ghost on the client
K_State DisplayPBGhost(K_State _State, Boolean _DisplayPBGhost) {
	declare K_State State = _State;

	// @Todo Once this has been playtested extensively we might consider
	// saving this value in a persistent variable to remember
	// the choice of the user
	State.DisplayPBGhost = _DisplayPBGhost;

	if (GhostMgr != Null) {
		// Remove previous PB ghost
		if (State.PBGhostInstanceId != NullId) {
			GhostMgr.Ghost_Remove(State.PBGhostInstanceId);
			State.PBGhostInstanceId = NullId;
		}
		if (State.PBCpSyncedGhostInstanceId != NullId) {
			GhostMgr.Ghost_Remove(State.PBCpSyncedGhostInstanceId);
			State.PBCpSyncedGhostInstanceId = NullId;
		}

		// Add new PBGhost
		if (
			State.PBGhostEnabled &&
			(State.DisplayPBGhost || (_State.SpectatorTargetRecordGhost.AccountId == "" && _State.LoadedRecordGhosts.count <= 0)) &&
			State.PBGhostId != NullId &&
			DataFileMgr.Ghosts.existskey(State.PBGhostId)
		) {
			declare CGhost PBGhost <=> DataFileMgr.Ghosts[State.PBGhostId];
			State.PBGhostInstanceId = GhostMgr.Ghost_Add(PBGhost, True);
			if (ModeConst::C_EnableCPSyncedGhost) {
				State.PBCpSyncedGhostInstanceId = GhostMgr.Ghost_AddWaypointSynced(PBGhost, True);
			}
		}
	}

	SendPBGhostVisibilityToModeAndML(State);

	return State;
}

	declare K_State State = _State;

	declare K_RecordGhost[] Records;
	if (State.LoadingRecordGhosts.existskey(_AccountId)) Records.add(State.LoadingRecordGhosts[_AccountId]);
	if (State.LoadedRecordGhosts.existskey(_AccountId)) Records.add(State.LoadedRecordGhosts[_AccountId]);

	foreach (Record in Records) {
		Task::Destroy(Record.Task_RetrieveRecords);
		Task::Destroy(Record.Task_RetrieveGhost);
		if (Record.GhostInstanceId != NullId) {
			GhostMgr.Ghost_Remove(Record.GhostInstanceId);
		}
		if (Record.GhostId != NullId && DataFileMgr.Ghosts.existskey(Record.GhostId)) {
			DataFileMgr.Ghost_Release(Record.GhostId);
		}
	}

	State.LoadingRecordGhosts.removekey(_AccountId);
	State.LoadedRecordGhosts.removekey(_AccountId);

	State = DisplayPBGhost(State, State.DisplayPBGhost);

	return State;
}
K_State ReleaseAllRecordGhosts(K_State _State) {
	declare K_State State = _State;

	declare TmpLoadingRecordGhosts = State.LoadingRecordGhosts;
	foreach (AccountId => Record in TmpLoadingRecordGhosts) {
		State = ReleaseRecordGhost(State, AccountId);
	}
	State.LoadingRecordGhosts = [];

	declare TmpLoadedRecordGhosts = State.LoadedRecordGhosts;
	foreach (AccountId => Record in TmpLoadedRecordGhosts) {
		State = ReleaseRecordGhost(State, AccountId);
	}
	State.LoadedRecordGhosts = [];

	SendRecordsStatusToML(State);
	State = DisplayPBGhost(State, State.DisplayPBGhost);

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Retrieve a player record ghost
K_State RetrieveRecordGhost(K_State _State, Text _AccountId, Integer _Medal) {
	declare K_State State = _State;

	// Release previous record if any
	State = ReleaseRecordGhost(State, _AccountId);

	declare K_RecordGhost Record = K_RecordGhost {
		AccountId = _AccountId,
		Medal = _Medal
	};
	declare K_Scope Scope;
	if (C_ForceRecordScope != "") {
		Scope = K_Scope {
			Type = C_ForceRecordScope,
			Id = ""
		};
	} else {
		Scope = GetScope(_State);
	}

	Record.Task_RetrieveRecords = Task::DestroyAndCreate(
		Record.Task_RetrieveRecords,
		ScoreMgr,
		ScoreMgr.Map_GetPlayerListRecordList(MainUser::GetMainUserId(), [Record.AccountId], State.CurrentMapUid, Scope.Type, Scope.Id, State.ModeName, State.CustomData)
	);

	State.LoadingRecordGhosts[Record.AccountId] = Record;
	SendRecordsStatusToML(State);

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if we can retrieve the records
Boolean CanRetrieveRecords(K_State _State) {
	return (
		_State.RecordsEnabled && //< The mode enabled the records
		_State.CurrentMapUid != "" && //< A map is loaded
		_State.PlayerIsDisplayingRecords && //< The records are displayed on the player screen
		_State.RecordsVisibilityFromSettings != CUserV2Profile::EDisplayRecords::Hide &&
		(
			(_State.CurrentMedal >= C_MedalToUnlockClubRecords && _State.RecordsClubId != ClubStruct::C_ClubNullId) ||
			_State.CurrentMedal >= C_MedalToUnlockMedalRecords ||
			_State.CurrentMedal >= C_MedalToUnlockWorldRecords
		)
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the records are dirty and we should update them
Boolean RecordsAreDirty(K_State _State) {
	return (
		_State.RecordsAreDirty.count > 0 ||
		_State.RecordsLoadedForMapUid != _State.CurrentMapUid
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Mark all types of records as dirty
K_State SetAllRecordsDirty(K_State _State) {
	declare K_State State = _State;

	foreach (RecordType in C_RecordTypes) {
		if (!State.RecordsAreDirty.exists(RecordType)) {
			if ((RecordType != C_RecordType_Club && RecordType != C_RecordType_ClubVIP) || State.RecordsClubId != ClubStruct::C_ClubNullId) {
				State.RecordsAreDirty.add(RecordType);
			}
		}
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the user has the permission to retrive the records of a given type
Boolean Private_HasGamePermission(Integer _RecordType) {
	return (
		G_RecordPermission.existskey(_RecordType) && (
			G_RecordPermission[_RecordType] == "" ||
			UserStore_MA::HasGamePermission(PrivilegeMgr, G_RecordPermission[_RecordType])
		)
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the game mode is compatible with the record of a given type
Boolean Private_IsGameModeCompatibleWithRecord(Integer _RecordType) {
	if (Map != Null) {
		if (TitleControl::IsRaceMapType(Map.MapType)) return True;
		if (TitleControl::IsStuntMapType(Map.MapType)) return True;
		if (TitleControl::IsPlatformMapType(Map.MapType)) return _RecordType == C_RecordType_Medals;
	}
	return False;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the given type of records must be requested or not
Boolean IsRecordRequested(K_State _State, Integer _RecordType) {
	return (
		_State.RecordsAreDirty.exists(_RecordType) &&
		Private_HasGamePermission(_RecordType) &&
		Private_IsGameModeCompatibleWithRecord(_RecordType) && (
			_State.RecordsSelectedZone == C_RecordType_None ||
			_State.RecordsSelectedZone == _RecordType ||
			_State.RecordsNeedInit.exists(_RecordType)
		)
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/* Start requests to get the map records
 *
 *	@param	_RaceTime								If negative get the player's best time
 *																	If positive get an estimation of the ranking for the given race time
 */
K_State RetrieveRecords(K_State _State, Integer _RaceTime) {
	declare K_State State = _State;
	declare Boolean HasStartedRequest = False;

	// C_RecordType_Zones
	if (IsRecordRequested(State, C_RecordType_Zones)) {
		if (State.RecordsVisibilityFromSettings == CUserV2Profile::EDisplayRecords::Always || State.CurrentMedal >= C_MedalToUnlockWorldRecords) {
			State.RequestZonesTopRecords = Http::DestroyAndCreate(State.RequestZonesTopRecords, LeaderboardAPI::GetMapTopRankings(LeaderboardAPI::C_GroupUid_PersonalBest, State.CurrentMapUid, _RaceTime));
			State.RequestZonesSurroundingRecords = Http::DestroyAndCreate(State.RequestZonesSurroundingRecords, LeaderboardAPI::GetSurroundingRankings(LeaderboardAPI::C_GroupUid_PersonalBest, State.CurrentMapUid, Record::C_BeforeMe, Record::C_AfterMe, _RaceTime));
			State.RecordsNeedInit.remove(C_RecordType_Zones);
			HasStartedRequest = True;
		}
	}

	// C_RecordType_Club
	if (IsRecordRequested(State, C_RecordType_Club) && State.RecordsClubId != ClubStruct::C_ClubNullId) {
		if (State.RecordsVisibilityFromSettings == CUserV2Profile::EDisplayRecords::Always || State.CurrentMedal >= C_MedalToUnlockClubRecords) {
			State.RequestClubTopRecords = Http::DestroyAndCreate(State.RequestClubTopRecords, LeaderboardAPI::GetMapTopRankingsInClub(LeaderboardAPI::C_GroupUid_PersonalBest, State.CurrentMapUid, State.RecordsClubId, _RaceTime));
			State.RequestClubSurroundingRecords = Http::DestroyAndCreate(State.RequestClubSurroundingRecords, LeaderboardAPI::GetMapSurroundingRankingsInClub(LeaderboardAPI::C_GroupUid_PersonalBest, State.CurrentMapUid, State.RecordsClubId, Record::C_BeforeMe, Record::C_AfterMe, _RaceTime));
			State.RecordsNeedInit.remove(C_RecordType_Club);
			HasStartedRequest = True;
		}
	}

	// C_RecordType_ClubVIP
	if (IsRecordRequested(State, C_RecordType_ClubVIP) && State.RecordsClubId != ClubStruct::C_ClubNullId) {
		if (State.RecordsVisibilityFromSettings == CUserV2Profile::EDisplayRecords::Always || State.CurrentMedal >= C_MedalToUnlockClubRecords) {
			State.RequestClubVIPList = Http::DestroyAndCreate(State.RequestClubVIPList, ClubAPI::GetVIPListOnMap(State.RecordsClubId, State.CurrentMapUid, LeaderboardAPI::C_GroupUid_PersonalBest));
			State.Task_GetClubVIPRecords = Task::DestroyIfInitialized(State.Task_GetClubVIPRecords);
			State.RecordsNeedInit.remove(C_RecordType_ClubVIP);
			HasStartedRequest = True;
		}
	}

	// C_RecordType_GlobalVIP
	if (IsRecordRequested(State, C_RecordType_GlobalVIP)) {
		if (State.RecordsVisibilityFromSettings == CUserV2Profile::EDisplayRecords::Always || State.CurrentMedal >= C_MedalToUnlockGlobalVIPRecords) {
			State.RequestGlobalVIPList = Http::DestroyAndCreate(State.RequestGlobalVIPList, ClubAPI::GetGlobalVIPListOnMap(State.CurrentMapUid, LeaderboardAPI::C_GroupUid_PersonalBest));
			State.Task_GetGlobalVIPRecords = Task::DestroyIfInitialized(State.Task_GetGlobalVIPRecords);
			State.RecordsNeedInit.remove(C_RecordType_GlobalVIP);
			HasStartedRequest = True;
		}
	}

	// C_RecordType_Medals
	if (IsRecordRequested(State, C_RecordType_Medals)) {
		if (State.RecordsVisibilityFromSettings != CUserV2Profile::EDisplayRecords::Hide || State.CurrentMedal >= C_MedalToUnlockMedalRecords) {
			LeaderboardStore::Action_GetPlayersCloseToMedals(This, State.CurrentMapUid);
			State.RecordsNeedInit.remove(C_RecordType_Medals);
			HasStartedRequest = True;
		}
	}

	// Retrieve map information from NadeoServices if necessary
	if (State.MapAvailaibleOnNadeoServices == C_MapAvailable_NotChecked) {
		State.Task_GetMapFromUid = Task::DestroyAndCreate(
			State.Task_GetMapFromUid,
			DataFileMgr,
			DataFileMgr.Map_NadeoServices_GetFromUid(MainUser::GetMainUserId(), State.CurrentMapUid)
		);
		HasStartedRequest = True;
	}

	if (HasStartedRequest) {
		State.RequestRecordsInProgress = True;
		State.RecordsErrorCodes = [];
		State.ResponseZonesTopRecords = LeaderboardAPI::K_ResponseFromGetMapTopRankings {};
		State.ResponseZonesSurroundingRecords = LeaderboardAPI::K_ResponseFromGetSurroundingRankings {};
		State.ResponseClubTopRecords = LeaderboardAPI::K_ResponseFromGetMapTopRankingsInClub {};
		State.ResponseClubSurroundingRecords = LeaderboardAPI::K_ResponseFromGetMapSurroundingRankingsInClub {};
		State.ResponseClubVIPRecords = [];
		State.ResponseGlobalVIPRecords = [];
		State.LoadingRecords = (State.RecordsLoadedForMapUid != State.CurrentMapUid); //< Display the loading in the UI only if there are no prior records.
		State.RecordsLoadingForMapUid = State.CurrentMapUid;
		SendRecordsToML(State);
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Update the records
K_State SetRecordsFromResponses(K_State _State) {
	declare K_State State = _State;
	declare Record::K_TMGame_Record_Record[Text][Text] TmpRanking;

	// The record `Position` attribute is rounded when above 10 000.
	// This means that several players can have the same rank.
	// Do not use it as an array key anymore. Instead use the account id.

	// Add the new zones records
	if (State.ResponseZonesTopRecords.TopRankings.count > 0 || State.ResponseZonesSurroundingRecords.TopsByZone.count > 0) {
		foreach (ZoneRecords in State.ResponseZonesTopRecords.TopRankings) {
			if (!TmpRanking.existskey(ZoneRecords.ZoneName)) {
				TmpRanking[ZoneRecords.ZoneName] = [];
			}
			foreach (Record in ZoneRecords.Tops) {
				if (Record.Position >= 1 && Record.Position <= Record::C_RecordsNb) {
					TmpRanking[ZoneRecords.ZoneName][Record.AccountId] = Record::K_TMGame_Record_Record {
						Rank = Record.Position,
						AccountId = Record.AccountId,
						DisplayName = UserStore_MA::GetUserMgrPlayerName(Record.AccountId),
						Score = Record.Score
					};
					if (TmpRanking[ZoneRecords.ZoneName].count >= Record::C_RecordsNb) break;
				}
			}
		}
		foreach (ZoneName => ZoneRecords in State.ResponseZonesSurroundingRecords.TopsByZone) {
			if (!TmpRanking.existskey(ZoneName)) {
				TmpRanking[ZoneName] = [];
			}
			foreach (Record in ZoneRecords) {
				TmpRanking[ZoneName][Record.AccountId] = Record::K_TMGame_Record_Record {
					Rank = Record.Position,
					AccountId = Record.AccountId,
					DisplayName = UserStore_MA::GetUserMgrPlayerName(Record.AccountId),
					Score = Record.Score
				};
				if (Record.AccountId == LocalUser.WebServicesUserId && Record.Score >= 0) {
					if (State.RecordsPlayerTime < 0 || Record.Score < State.RecordsPlayerTime) {
						State.RecordsPlayerTime = Record.Score;
					}
				}
			}
		}

		State.ResponseZonesTopRecords = LeaderboardAPI::K_ResponseFromGetMapTopRankings {};
		State.ResponseZonesSurroundingRecords = LeaderboardAPI::K_ResponseFromGetSurroundingRankings {};
		State.RecordsAreDirty.remove(C_RecordType_Zones);
	}
	// Restore the old zones records
	else {
		foreach (ZoneRecords in State.ZonesRecords) {
			if (ZoneRecords.Type == C_RecordType_Zones) {
				if (!TmpRanking.existskey(ZoneRecords.ZoneName)) {
					TmpRanking[ZoneRecords.ZoneName] = [];
				}
				foreach (Record in ZoneRecords.Records) {
					TmpRanking[ZoneRecords.ZoneName][Record.AccountId] = Record;
				}
			}
		}
	}

	// Add the new global VIP records
	if (State.ResponseGlobalVIPRecords.count > 0) {
		State.ResponseGlobalVIPRecords = State.ResponseGlobalVIPRecords.sort();
		if (!TmpRanking.existskey(C_FakeZone_GlobalVIP)) {
			TmpRanking[C_FakeZone_GlobalVIP] = [];
		}
		declare Integer Rank = 1;
		foreach (AccountId => Score in State.ResponseGlobalVIPRecords) {
			TmpRanking[C_FakeZone_GlobalVIP][AccountId] = Record::K_TMGame_Record_Record {
				Rank = Rank,
				AccountId = AccountId,
				DisplayName = UserStore_MA::GetUserMgrPlayerName(AccountId),
				Score = Score
			};
			Rank += 1;
		}

		State.ResponseGlobalVIPRecords = [];
		State.RecordsAreDirty.remove(C_RecordType_GlobalVIP);
	}
	// Restore the old global VIP records
	else {
		foreach (ZoneRecords in State.ZonesRecords) {
			if (ZoneRecords.Type == C_RecordType_GlobalVIP) {
				if (!TmpRanking.existskey(C_FakeZone_GlobalVIP)) {
					TmpRanking[C_FakeZone_GlobalVIP] = [];
				}
				foreach (Record in ZoneRecords.Records) {
					TmpRanking[C_FakeZone_GlobalVIP][Record.AccountId] = Record;
				}
			}
		}
	}

	// Add the new club VIP records
	if (State.ResponseClubVIPRecords.count > 0) {
		State.ResponseClubVIPRecords = State.ResponseClubVIPRecords.sort();
		if (!TmpRanking.existskey(C_FakeZone_ClubVIP)) {
			TmpRanking[C_FakeZone_ClubVIP] = [];
		}
		declare Integer Rank = 1;
		foreach (AccountId => Score in State.ResponseClubVIPRecords) {
			TmpRanking[C_FakeZone_ClubVIP][AccountId] = Record::K_TMGame_Record_Record {
				Rank = Rank,
				AccountId = AccountId,
				DisplayName = UserStore_MA::GetUserMgrPlayerName(AccountId),
				Score = Score
			};
			Rank += 1;
		}

		State.ResponseClubVIPRecords = [];
		State.RecordsAreDirty.remove(C_RecordType_ClubVIP);
	}
	// Restore the old club VIP records
	else {
		foreach (ZoneRecords in State.ZonesRecords) {
			if (ZoneRecords.Type == C_RecordType_ClubVIP) {
				if (!TmpRanking.existskey(C_FakeZone_ClubVIP)) {
					TmpRanking[C_FakeZone_ClubVIP] = [];
				}
				foreach (Record in ZoneRecords.Records) {
					TmpRanking[C_FakeZone_ClubVIP][Record.AccountId] = Record;
				}
			}
		}
	}

	// Add the new club records
	if (State.ResponseClubTopRecords.Tops.count > 0 || State.ResponseClubSurroundingRecords.Tops.count > 0) {
		if (!TmpRanking.existskey(C_FakeZone_Club)) {
			TmpRanking[C_FakeZone_Club] = [];
		}
		foreach (Record in State.ResponseClubTopRecords.Tops) {
			if (Record.Position >= 1 && Record.Position <= Record::C_RecordsNb) {
				TmpRanking[C_FakeZone_Club][Record.AccountId] = Record::K_TMGame_Record_Record {
					Rank = Record.Position,
					AccountId = Record.AccountId,
					DisplayName = UserStore_MA::GetUserMgrPlayerName(Record.AccountId),
					Score = Record.Score
				};
				if (TmpRanking[C_FakeZone_Club].count >= Record::C_RecordsNb) break;
			}
		}
		foreach (Record in State.ResponseClubSurroundingRecords.Tops) {
			TmpRanking[C_FakeZone_Club][Record.AccountId] = Record::K_TMGame_Record_Record {
				Rank = Record.Position,
				AccountId = Record.AccountId,
				DisplayName = UserStore_MA::GetUserMgrPlayerName(Record.AccountId),
				Score = Record.Score
			};
			if (Record.AccountId == LocalUser.WebServicesUserId && Record.Score >= 0) {
				if (State.RecordsPlayerTime < 0 || Record.Score < State.RecordsPlayerTime) {
					State.RecordsPlayerTime = Record.Score;
				}
			}
		}

		State.ResponseClubTopRecords = LeaderboardAPI::K_ResponseFromGetMapTopRankingsInClub {};
		State.ResponseClubSurroundingRecords = LeaderboardAPI::K_ResponseFromGetMapSurroundingRankingsInClub {};
		State.RecordsAreDirty.remove(C_RecordType_Club);
	}
	// Restore the old club records
	else {
		foreach (ZoneRecords in State.ZonesRecords) {
			if (ZoneRecords.Type == C_RecordType_Club) {
				if (!TmpRanking.existskey(C_FakeZone_Club)) {
					TmpRanking[C_FakeZone_Club] = [];
				}
				foreach (Record in ZoneRecords.Records) {
					TmpRanking[C_FakeZone_Club][Record.AccountId] = Record;
				}
			}
		}
	}

	// Add the new medals records
	if (Private_HasGamePermission(C_RecordType_Medals)) {
		if (State.RecordsVisibilityFromSettings != CUserV2Profile::EDisplayRecords::Hide || State.CurrentMedal >= C_MedalToUnlockMedalRecords) {
			declare Integer Score;
			declare Integer Time;
			declare Integer Points;
			declare Integer RespawnCount;
			declare K_Scope Scope;
			if (C_ForceRecordScope != "") {
				Scope = K_Scope {
					Type = C_ForceRecordScope,
					Id = ""
				};
			} else {
				Scope = GetScope(State);
			}
			ScoreMgr.Map_GetRecordFullScore(
				MainUser::GetMainUserId(),
				Map.MapInfo.MapUid,
				Scope.Type,
				Scope.Id,
				State.ModeName,
				State.CustomData,
				Time,
				Points,
				RespawnCount
			);
			Score = Time;
			if (Map != Null) {
				switch (Map.MapType) {
					case Const::C_MapType_Stunt, Const::C_MapType_Stunt_NoConcat: Score = Points;
					case Const::C_MapType_Platform, Const::C_MapType_Platform_NoConcat: Score = RespawnCount;
				}
			}
			declare Record::K_TMGame_Record_Record LocalUserRecord = Record::K_TMGame_Record_Record {
				Score = Score,
				DisplayName = UserStore_MA::GetUserMgrPlayerName(LocalUser.WebServicesUserId),
				AccountId = LocalUser.WebServicesUserId,
				Rank = -1
			};

			if (!TmpRanking.existskey(C_FakeZone_Medals)) TmpRanking[C_FakeZone_Medals] = [];
			if (LeaderboardStore::GetPlayersCloseToMedals(This, State.CurrentMapUid).count > 0) {
				for (Medal, Medals::C_ScoreMgrMedal_Bronze, Medals::C_ScoreMgrMedal_Author, -1) {
					if (State.RecordsVisibilityFromSettings == CUserV2Profile::EDisplayRecords::Progressive && Medal >= Medals::C_ScoreMgrMedal_Author && State.CurrentMedal < Medals::C_ScoreMgrMedal_Gold) continue; //< Display author time when gold medal is reached
					if (State.CurrentMedal == Medal) TmpRanking[C_FakeZone_Medals][C_LocalUserRecordIndex] = LocalUserRecord; //< Add the LocalUser's personal best with a custom index
					declare Player = LeaderboardStore::GetPlayersCloseToMedals(This, State.CurrentMapUid).get(Medal, CampaignStruct::LibCampaignStruct_K_PlayerCloseToMedal {
						AccountId = C_FakeAccountId_MedalGhost^Medal,
						Score = Medals::GetScoreFromMedal(Map, Medal)
					});

					// Add the ghost of the medal if it exists, otherwise just add the time of the medal as a fake ghost
					TmpRanking[C_FakeZone_Medals][Player.AccountId] = Record::K_TMGame_Record_Record {
						Rank = Medals::C_ScoreMgrMedal_Author - Medal,
						AccountId = Player.AccountId,
						DisplayName = UserStore_MA::GetUserMgrPlayerName(Player.AccountId),
						Score = Player.Score
					};
				}
				State.RecordsAreDirty.remove(C_RecordType_Medals);
			} else {
				for (Medal, Medals::C_ScoreMgrMedal_Bronze, Medals::C_ScoreMgrMedal_Author, -1) {
					if (State.RecordsVisibilityFromSettings == CUserV2Profile::EDisplayRecords::Progressive && Medal >= Medals::C_ScoreMgrMedal_Author && State.CurrentMedal < Medals::C_ScoreMgrMedal_Gold) continue; //< Display author time when gold medal is reached
					if (State.CurrentMedal == Medal) TmpRanking[C_FakeZone_Medals][C_LocalUserRecordIndex] = LocalUserRecord; //< Add the LocalUser's personal best
					if (!TmpRanking[C_FakeZone_Medals].existskey(C_FakeAccountId_MedalGhost^Medal)) {
						TmpRanking[C_FakeZone_Medals][C_FakeAccountId_MedalGhost^Medal] = Record::K_TMGame_Record_Record {
							Rank = Medals::C_ScoreMgrMedal_Author - Medal,
							AccountId = C_FakeAccountId_MedalGhost^Medal,
							DisplayName = Medals::GetScoreMgrMedalName(Medal),
							Score = Medals::GetScoreFromMedal(Map, Medal)
						};
					}
				}
			}

			// Add the local player record to the end if it's not already there
			if (LocalUserRecord.Score > 0) {
				if (!TmpRanking[C_FakeZone_Medals].existskey(LocalUser.WebServicesUserId) && !TmpRanking[C_FakeZone_Medals].existskey(C_LocalUserRecordIndex)) {
					TmpRanking[C_FakeZone_Medals][LocalUser.WebServicesUserId] = LocalUserRecord;
				}
			}
		}
	}

	// Create the new records
	State.ZonesRecords = [];
	declare Text[] RecordAccountIds;
	declare Integer RecordsMaxNb = Record::C_RecordsNb + Record::C_BeforeMe + 1 + Record::C_AfterMe;
	foreach (ZoneName => Records in TmpRanking) {
		declare Record::K_TMGame_Record_Records ZoneRecords = Record::K_TMGame_Record_Records {
			ZoneName = ZoneName,
			WorstScore = -1,
			IsFull = False,
			Type = C_RecordType_Zones
		};

		if (ZoneName == C_FakeZone_Club) {
			ZoneRecords.ZoneName = State.RecordsClubName;
			ZoneRecords.Type = C_RecordType_Club;
		} else if (ZoneName == C_FakeZone_ClubVIP) {
			ZoneRecords.ZoneName = _("|Club|Club VIP");
			ZoneRecords.Type = C_RecordType_ClubVIP;
		} else if (ZoneName == C_FakeZone_GlobalVIP) {
			ZoneRecords.ZoneName = _("|Club|Global VIP");
			ZoneRecords.Type = C_RecordType_GlobalVIP;
		} else if (ZoneName == C_FakeZone_Medals) {
			ZoneRecords.ZoneName = _("|Ghosts|Medals");
			ZoneRecords.Type = C_RecordType_Medals;
		}

		declare Integer PlayerRank = -1;
		foreach (Record in Records) {
			// Sometimes the ranks returned by the top and surrounding records API collides.
			// We avoid that here by rewriting the rank to be always above the number of players
			// displayed before in the ranking.
			// Check issue #2303 for more details.
			declare Record::K_TMGame_Record_Record RecordWithFixedRank = Record;
			if (ZoneRecords.Type != C_RecordType_Medals) {
				RecordWithFixedRank.Rank = ML::Max(RecordWithFixedRank.Rank, ZoneRecords.Records.count + 1); //< Rank must be superior to the number of players already in the ranking
			}

			ZoneRecords.Records.add(RecordWithFixedRank);
			if (!RecordAccountIds.exists(RecordWithFixedRank.AccountId)) {
				RecordAccountIds.add(RecordWithFixedRank.AccountId);
			}
			if (RecordWithFixedRank.Score >= 0 && (ZoneRecords.WorstScore < 0 || RecordWithFixedRank.Score > ZoneRecords.WorstScore)) {
				ZoneRecords.WorstScore = RecordWithFixedRank.Score;
			}
			if (RecordWithFixedRank.AccountId == LocalUser.WebServicesUserId) {
				PlayerRank = RecordWithFixedRank.Rank;
			}
		}

		ZoneRecords.IsFull = (
			ZoneRecords.Records.count >= Record::C_RecordsNb &&
			PlayerRank >= 1 &&
			(
				ZoneRecords.Records.count >= RecordsMaxNb ||
				ZoneRecords.Records.count >= PlayerRank + Record::C_AfterMe
			)
		);

		State.ZonesRecords.add(ZoneRecords);
	}

	// Remove old records ghost
	declare TmpLoadingRecordGhosts = State.LoadingRecordGhosts;
	foreach (AccountId => Record in TmpLoadingRecordGhosts) {
		if (!RecordAccountIds.exists(AccountId)) {
			State = ReleaseRecordGhost(State, AccountId);
		}
	}

	declare TmpLoadedRecordGhosts = State.LoadedRecordGhosts;
	foreach (AccountId => Record in TmpLoadedRecordGhosts) {
		if (!RecordAccountIds.exists(AccountId)) {
			State = ReleaseRecordGhost(State, AccountId);
		}
	}

	return State;
}

/// Start a task to retrieve the display name of the records
K_State RetrieveDisplayName(K_State _State) {
	if (UserMgr == Null) return _State;

	declare K_State State = _State;

	declare Text[] AccountIds;
	foreach (ZoneRecords in State.ZonesRecords) {
		foreach (Record in ZoneRecords.Records) {
			if (UserStore_MA::GetUserMgrPlayerName(Record.AccountId) == "" && !AccountIds.exists(Record.AccountId)) {
				AccountIds.add(Record.AccountId);
			}
		}
	}

	if (AccountIds.count > 0) {
		if (State.TaskIdRetrieveDisplayName != NullId && UserMgr.TaskResults.existskey(State.TaskIdRetrieveDisplayName)) {
			UserMgr.TaskResult_Release(State.TaskIdRetrieveDisplayName);
			State.TaskIdRetrieveDisplayName = NullId;
		}

		declare Task = UserMgr.RetrieveDisplayName(MainUser::GetMainUserId(), AccountIds);
		if (Task == Null) {
			State.TaskIdRetrieveDisplayName = NullId;
		} else {
			State.TaskIdRetrieveDisplayName = Task.Id;
		}
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the display names in the records
K_State UpdateDisplayName(K_State _State) {
	if (UserMgr == Null) return _State;

	declare K_State State = _State;

	declare TmpRecords = State.ZonesRecords;
	foreach (ZoneKey => ZoneRecords in TmpRecords) {
		foreach (RecordKey => Record in ZoneRecords.Records) {
			if (Record.DisplayName == "") {
				State.ZonesRecords[ZoneKey].Records[RecordKey].DisplayName = UserStore_MA::GetUserMgrPlayerName(Record.AccountId);
			}
		}
	}

	return State;
}

/// Set the PB ghost on the client
K_State SetPBGhost(K_State _State, CGhost _Ghost, Boolean _CanCelebrate) {
	if (_Ghost == Null || _Ghost.Id == _State.PBGhostId) return _State;

	declare K_State State = _State;
	declare CGhost Ghost = _Ghost;

	if (State.PBGhostEnabled) {
		//L16N [Record] Best time done by the player.
		Ghost.Nickname = _("Personal best");
		//L16N [Record] Best time done by the player. PB stands for Personnal Best. The translation must be 3 or less letters because it's displayed on the back of the car (as the trigram).
		Ghost.Trigram = _("|Personal best|PB");

		if (GhostMgr != Null) {
			if (State.PBGhostInstanceId != NullId) {
				GhostMgr.Ghost_Remove(State.PBGhostInstanceId);
				State.PBGhostInstanceId = NullId;
			}
			if (State.PBCpSyncedGhostInstanceId != NullId) {
				GhostMgr.Ghost_Remove(State.PBCpSyncedGhostInstanceId);
				State.PBCpSyncedGhostInstanceId = NullId;
			}
		}
	}
	if (State.PBGhostEnabled || State.CelebratePB) {
		if (State.PBGhostId != NullId && DataFileMgr.Ghosts.existskey(State.PBGhostId)) {
			DataFileMgr.Ghost_Release(State.PBGhostId);
		}
		State.PBGhostId = Ghost.Id;
	}

	declare Boolean SortByAscendingOrder = True;
	State.Celebration.PrevScore = State.Celebration.BestScore;
	State.Celebration.BestScore = Ghost.Result.Time;
	if (Map != Null) {
		switch (Map.MapType) {
			case Const::C_MapType_Stunt, Const::C_MapType_Stunt_NoConcat: {
				State.Celebration.BestScore = Ghost.Result.Score;
				SortByAscendingOrder = False;
			}
			case Const::C_MapType_Platform, Const::C_MapType_Platform_NoConcat: {
				State.Celebration.BestScore = Ghost.Result.NbRespawns;
			}
		}
	}
	if (
		_CanCelebrate &&
		(
			State.Celebration.PrevScore < 0 ||
			(
				(SortByAscendingOrder && State.Celebration.BestScore < State.Celebration.PrevScore) ||
				(!SortByAscendingOrder && State.Celebration.BestScore > State.Celebration.PrevScore)
			)
		)
		) {
		State.RequestCelebration = True;
	}

	State = DisplayPBGhost(State, State.DisplayPBGhost);

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Remove the client PB ghost
K_State ResetPBGhost(K_State _State) {
	declare K_State State = _State;

	if (State.TaskIdGetRecordGhost != NullId && ScoreMgr.TaskResults.existskey(State.TaskIdGetRecordGhost)) {
		ScoreMgr.TaskResult_Release(State.TaskIdGetRecordGhost);
	}
	State.TaskIdGetRecordGhost = NullId;
	if (GhostMgr != Null) {
		if (State.PBGhostInstanceId != NullId) {
			GhostMgr.Ghost_Remove(State.PBGhostInstanceId);
			State.PBGhostInstanceId = NullId;
		}
		if (State.PBCpSyncedGhostInstanceId != NullId) {
			GhostMgr.Ghost_Remove(State.PBCpSyncedGhostInstanceId);
			State.PBCpSyncedGhostInstanceId = NullId;
		}
	}
	if (State.PBGhostId != NullId && DataFileMgr.Ghosts.existskey(State.PBGhostId)) {
		DataFileMgr.Ghost_Release(State.PBGhostId);
	}
	State.PBGhostId = NullId;
	State.Celebration.PrevScore = -1;
	State.Celebration.BestScore = -1;

	State = DisplayPBGhost(State, State.DisplayPBGhost);

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the PB Ghost of the client on the given map
K_State RetrievePBGhost(K_State _State) {
	declare K_State State = ResetPBGhost(_State);

	if ((State.PBGhostEnabled || State.CelebratePB) && State.CurrentMapUid != "" && State.ModeName != "") {
		declare K_Scope Scope = GetScope(State);
		if (Scope.Type != "") {
			declare CTaskResult_Ghost TaskGetRecordGhost = ScoreMgr.Map_GetRecordGhost_v2(MainUser::GetMainUserId(), State.CurrentMapUid, Scope.Type, Scope.Id, State.ModeName, State.CustomData);
			if (TaskGetRecordGhost != Null) {
				State.TaskIdGetRecordGhost = TaskGetRecordGhost.Id;
			}
		}
	}

	return State;
}

	declare Integer TMGame_Record_CurrentMedal for ClientUI = Medals::C_ScoreMgrMedal_None;

	State.CurrentMedal = _Medal;

	State.Celebration.PrevMedal = State.Celebration.BestMedal;

	return State;

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the state
K_State Update(
	K_State _State,
	CMap _Map,
	Text _ScopeSeason,
	Text _ScopeNotSeason,
	Text _ModeName,
	Text _CustomData,
	Text[Text] _SeasonIds
) {
	declare K_State State = _State;

	if (_Map == Null && State.CurrentMapUid != "") {
		State = SetCurrentMapUid(State, "");
		State.MapAvailaibleOnNadeoServices = C_MapAvailable_NotChecked;
	} else if (_Map != Null && State.CurrentMapUid != _Map.MapInfo.MapUid) {
		State = SetCurrentMapUid(State, _Map.MapInfo.MapUid);
		State.MapAvailaibleOnNadeoServices = C_MapAvailable_NotChecked;
	}

	State.ScopeSeason = _ScopeSeason;
	State.ScopeNotSeason = _ScopeNotSeason;
	State.ModeName = _ModeName;
	State.CustomData = _CustomData;
	State.SeasonIds = _SeasonIds;
	State.RequestCelebration = False;
	State.Celebration.IsActive = False;

	declare Boolean RetrieveRecords = False;
	if (State.RecordsLoadedForMapUid != State.CurrentMapUid || State.ReloadRecords) {
		State.ReloadRecords = False;
		declare PinnedClub = ClubStore::GetUserPinnedClub();
		State.RecordsClubId = PinnedClub.Id;
		State.RecordsClubName = PinnedClub.Name;
		if (State.RecordsLoadedForMapUid != "") {
			State = ResetRecords(State);
		}
		// Retrieve the player's record one time at the loading
		// of the map. Otherwise the player would notify the server
		// and force an update in the other players mania app
		// every time they improved their time even if it was
		// not a new record.
		// So do not use CanRetrieveRecords(State) here.
		if (
			State.RecordsEnabled &&
			State.CurrentMapUid != ""
		) {
			State = SetAllRecordsDirty(State);
			RetrieveRecords = True;
		}
	}

	declare K_Celebration TMGame_Record_Celebration for ClientUI;
	TMGame_Record_Celebration = State.Celebration;

	State = RetrievePBGhost(State);
	State = RetrieveMedal(State, False);
	// Retrieve records after updating the medal
	// because they are a condition to start
	// the records requests
	if (RetrieveRecords) {
		State = RetrieveRecords(State, State.RecordsPlayerTime);
	}

	return State;
}

/// Process the pending records requests
Void Private_ProcessRecordsRequests() {
	declare K_State TMGame_Record_State for UI;

	if (TMGame_Record_State.RequestRecordsInProgress) {
		if (
			Http::IsInitialized(TMGame_Record_State.RequestZonesTopRecords) ||
			Http::IsInitialized(TMGame_Record_State.RequestZonesSurroundingRecords) ||
			Http::IsInitialized(TMGame_Record_State.RequestClubTopRecords) ||
			Http::IsInitialized(TMGame_Record_State.RequestClubSurroundingRecords) ||
			Http::IsInitialized(TMGame_Record_State.RequestClubVIPList) ||
			Http::IsInitialized(TMGame_Record_State.RequestGlobalVIPList) ||
			Task::IsInitialized(TMGame_Record_State.Task_GetClubVIPRecords) ||
			Task::IsInitialized(TMGame_Record_State.Task_GetGlobalVIPRecords) ||
			Task::IsInitialized(TMGame_Record_State.Task_GetMapFromUid) ||
			LeaderboardStore::IsRetrievingPlayersCloseToMedals(This)
		) {
			// Requests for C_RecordType_Zones
			if (Http::IsInitialized(TMGame_Record_State.RequestZonesTopRecords)) {
				TMGame_Record_State.RequestZonesTopRecords = Http::Update(TMGame_Record_State.RequestZonesTopRecords);
				if (!Http::IsRunning(TMGame_Record_State.RequestZonesTopRecords)) {
					if (Http::IsSuccess(TMGame_Record_State.RequestZonesTopRecords)) {
						TMGame_Record_State.ResponseZonesTopRecords = LeaderboardAPI::GetResponseFromGetMapTopRankings(TMGame_Record_State.RequestZonesTopRecords);
						TMGame_Record_State.RecordsLoadedForMapUid = TMGame_Record_State.RecordsLoadingForMapUid;
					} else {
						declare NLSError = Http::GetNLSError(TMGame_Record_State.RequestZonesTopRecords);
						if (NLSError.Error == C_NLSError_MapNotFound) {
							TMGame_Record_State = AddRecordsError(TMGame_Record_State, C_RecordsError_MapNotFound);
						} else {
							TMGame_Record_State = AddRecordsError(TMGame_Record_State, C_RecordsError_CouldNotLoadZonesTop);
						}
					}
					TMGame_Record_State.RequestZonesTopRecords = Http::Destroy(TMGame_Record_State.RequestZonesTopRecords);
				}
			}
			if (Http::IsInitialized(TMGame_Record_State.RequestZonesSurroundingRecords)) {
				TMGame_Record_State.RequestZonesSurroundingRecords = Http::Update(TMGame_Record_State.RequestZonesSurroundingRecords);
				if (!Http::IsRunning(TMGame_Record_State.RequestZonesSurroundingRecords)) {
					if (Http::IsSuccess(TMGame_Record_State.RequestZonesSurroundingRecords)) {
						TMGame_Record_State.ResponseZonesSurroundingRecords = LeaderboardAPI::GetResponseFromGetSurroundingRankings(TMGame_Record_State.RequestZonesSurroundingRecords);
						TMGame_Record_State.RecordsLoadedForMapUid = TMGame_Record_State.RecordsLoadingForMapUid;
					} else {
						TMGame_Record_State = AddRecordsError(TMGame_Record_State, C_RecordsError_CouldNotLoadZonesSurround);
					}
					TMGame_Record_State.RequestZonesSurroundingRecords = Http::Destroy(TMGame_Record_State.RequestZonesSurroundingRecords);
				}
			}

			// Requests for C_RecordType_Club
			if (Http::IsInitialized(TMGame_Record_State.RequestClubTopRecords)) {
				TMGame_Record_State.RequestClubTopRecords = Http::Update(TMGame_Record_State.RequestClubTopRecords);
				if (!Http::IsRunning(TMGame_Record_State.RequestClubTopRecords)) {
					if (Http::IsSuccess(TMGame_Record_State.RequestClubTopRecords)) {
						TMGame_Record_State.ResponseClubTopRecords = LeaderboardAPI::GetResponseFromGetMapTopRankingsInClub(TMGame_Record_State.RequestClubTopRecords);
						TMGame_Record_State.RecordsLoadedForMapUid = TMGame_Record_State.RecordsLoadingForMapUid;
					} else {
						declare NLSError = Http::GetNLSError(TMGame_Record_State.RequestClubTopRecords);
						if (NLSError.Error == C_NLSError_MapNotFound) {
							TMGame_Record_State = AddRecordsError(TMGame_Record_State, C_RecordsError_MapNotFound);
						} else {
							TMGame_Record_State = AddRecordsError(TMGame_Record_State, C_RecordsError_CouldNotLoadClubTop);
						}
					}
					TMGame_Record_State.RequestClubTopRecords = Http::Destroy(TMGame_Record_State.RequestClubTopRecords);
				}
			}
			if (Http::IsInitialized(TMGame_Record_State.RequestClubSurroundingRecords)) {
				TMGame_Record_State.RequestClubSurroundingRecords = Http::Update(TMGame_Record_State.RequestClubSurroundingRecords);
				if (!Http::IsRunning(TMGame_Record_State.RequestClubSurroundingRecords)) {
					if (Http::IsSuccess(TMGame_Record_State.RequestClubSurroundingRecords)) {
						TMGame_Record_State.ResponseClubSurroundingRecords = LeaderboardAPI::GetResponseFromGetMapSurroundingRankingsInClub(TMGame_Record_State.RequestClubSurroundingRecords);
						TMGame_Record_State.RecordsLoadedForMapUid = TMGame_Record_State.RecordsLoadingForMapUid;
					} else {
						TMGame_Record_State = AddRecordsError(TMGame_Record_State, C_RecordsError_CouldNotLoadClubSurround);
					}
					TMGame_Record_State.RequestClubSurroundingRecords = Http::Destroy(TMGame_Record_State.RequestClubSurroundingRecords);
				}
			}

			// Requests and tasks for C_RecordType_ClubVIP
			if (Http::IsInitialized(TMGame_Record_State.RequestClubVIPList)) {
				TMGame_Record_State.RequestClubVIPList = Http::Update(TMGame_Record_State.RequestClubVIPList);
				if (!Http::IsRunning(TMGame_Record_State.RequestClubVIPList)) {
					if (Http::IsSuccess(TMGame_Record_State.RequestClubVIPList)) {
						declare Text[] VIPAccountIds = ClubAPI::GetResponseFromGetVIPListOnMap(TMGame_Record_State.RequestClubVIPList);
						if (VIPAccountIds.count > 0) {
							declare K_Scope Scope = GetScope(TMGame_Record_State);
							if (C_ForceRecordScope != "") Scope = K_Scope { Type = C_ForceRecordScope };
							TMGame_Record_State.Task_GetClubVIPRecords = Task::DestroyAndCreate(
								TMGame_Record_State.Task_GetClubVIPRecords,
								ScoreMgr,
								ScoreMgr.Map_GetPlayerListRecordList(
									MainUser::GetMainUserId(),
									VIPAccountIds.slice(0, 200), //< Limit the number of account ids requested. see: /trackmania-next/tmnext/-/issues/3499

									Scope.Id,

									TMGame_Record_State.CustomData
								)
							);
						}
					} else {

					}
					TMGame_Record_State.RequestClubVIPList = Http::Destroy(TMGame_Record_State.RequestClubVIPList);
				}
			}
			if (Task::IsInitialized(TMGame_Record_State.Task_GetClubVIPRecords)) {
				TMGame_Record_State.Task_GetClubVIPRecords = Task::Update(TMGame_Record_State.Task_GetClubVIPRecords);

							TMGame_Record_State.ResponseClubVIPRecords[Record.AccountId] = Record.Time;

								switch (Map.MapType) {
									case Const::C_MapType_Stunt, Const::C_MapType_Stunt_NoConcat: {
										TMGame_Record_State.ResponseClubVIPRecords[Record.AccountId] = Record.Score;
									}
									case Const::C_MapType_Platform, Const::C_MapType_Platform_NoConcat: {
										TMGame_Record_State.ResponseClubVIPRecords[Record.AccountId] = Record.RespawnCount;
									}
								}
							}
						}
					} else {
						TMGame_Record_State = AddRecordsError(TMGame_Record_State, C_RecordsError_CouldNotGetClubVIPRecords);
					}
					TMGame_Record_State.Task_GetClubVIPRecords = Task::Destroy(TMGame_Record_State.Task_GetClubVIPRecords);
				}
			}

			// Requests for C_RecordType_GlobalVIP
			if (Http::IsInitialized(TMGame_Record_State.RequestGlobalVIPList)) {
				TMGame_Record_State.RequestGlobalVIPList = Http::Update(TMGame_Record_State.RequestGlobalVIPList);
				if (!Http::IsRunning(TMGame_Record_State.RequestGlobalVIPList)) {
					if (Http::IsSuccess(TMGame_Record_State.RequestGlobalVIPList)) {
						declare Text[] VIPAccountIds = ClubAPI::GetResponseFromGetVIPListOnMap(TMGame_Record_State.RequestGlobalVIPList);
						if (VIPAccountIds.count > 0) {
							declare K_Scope Scope = GetScope(TMGame_Record_State);
							if (C_ForceRecordScope != "") Scope = K_Scope { Type = C_ForceRecordScope };
							TMGame_Record_State.Task_GetGlobalVIPRecords = Task::DestroyAndCreate(
								TMGame_Record_State.Task_GetGlobalVIPRecords,
								ScoreMgr,
								ScoreMgr.Map_GetPlayerListRecordList(
									MainUser::GetMainUserId(),
									VIPAccountIds.slice(0, 200), //< Limit the number of account ids requested. see: /trackmania-next/tmnext/-/issues/3499
									TMGame_Record_State.CurrentMapUid,
									Scope.Type,
									Scope.Id,
									TMGame_Record_State.ModeName,
									TMGame_Record_State.CustomData
								)
							);
						}
					} else {
						TMGame_Record_State = AddRecordsError(TMGame_Record_State, C_RecordsError_CouldNotGetGlobalVIPList);
					}
					TMGame_Record_State.RequestGlobalVIPList = Http::Destroy(TMGame_Record_State.RequestGlobalVIPList);
				}
			}
			if (Task::IsInitialized(TMGame_Record_State.Task_GetGlobalVIPRecords)) {
				TMGame_Record_State.Task_GetGlobalVIPRecords = Task::Update(TMGame_Record_State.Task_GetGlobalVIPRecords);
				if (!Task::IsRunning(TMGame_Record_State.Task_GetGlobalVIPRecords)) {
					declare CTaskResult_MapRecordList SourceTask = Task::GetSourceTask_MapRecordList(TMGame_Record_State.Task_GetGlobalVIPRecords);
					if (Task::IsSuccess(TMGame_Record_State.Task_GetGlobalVIPRecords) && SourceTask != Null) {
						TMGame_Record_State.RecordsLoadedForMapUid = TMGame_Record_State.RecordsLoadingForMapUid;
						foreach (Record in SourceTask.MapRecordList) {
							TMGame_Record_State.ResponseGlobalVIPRecords[Record.AccountId] = Record.Time;
							if (Map != Null) {
								switch (Map.MapType) {
									case Const::C_MapType_Stunt, Const::C_MapType_Stunt_NoConcat: {
										TMGame_Record_State.ResponseGlobalVIPRecords[Record.AccountId] = Record.Score;
									}
									case Const::C_MapType_Platform, Const::C_MapType_Platform_NoConcat: {
										TMGame_Record_State.ResponseGlobalVIPRecords[Record.AccountId] = Record.RespawnCount;
									}
								}
							}
						}
					} else {
						TMGame_Record_State = AddRecordsError(TMGame_Record_State, C_RecordsError_CouldNotGetGlobalVIPRecords);
					}
					TMGame_Record_State.Task_GetGlobalVIPRecords = Task::Destroy(TMGame_Record_State.Task_GetGlobalVIPRecords);
				}
			}

			// Task to retrieve map information from NadeoServices
			if (Task::IsInitialized(TMGame_Record_State.Task_GetMapFromUid)) {
				TMGame_Record_State.Task_GetMapFromUid = Task::Update(TMGame_Record_State.Task_GetMapFromUid);
				if (!Task::IsRunning(TMGame_Record_State.Task_GetMapFromUid)) {
					if (Task::IsSuccess(TMGame_Record_State.Task_GetMapFromUid)) {
						TMGame_Record_State.MapAvailaibleOnNadeoServices = C_MapAvailable_Available;
					} else {
						TMGame_Record_State.MapAvailaibleOnNadeoServices = C_MapAvailable_NotAvailable;
					}
					TMGame_Record_State.Task_GetMapFromUid = Task::Destroy(TMGame_Record_State.Task_GetMapFromUid);
				}
			}
		} else {
			TMGame_Record_State = SetRecordsFromResponses(TMGame_Record_State);
			TMGame_Record_State = RetrieveDisplayName(TMGame_Record_State);
			TMGame_Record_State.RequestRecordsInProgress = False;
			TMGame_Record_State.RecordsLoadingForMapUid = "";
			TMGame_Record_State.LoadingRecords = False;
			SendRecordsToML(TMGame_Record_State);
		}
	}
}

/// Process the mania app events
Void Private_ProcessEvents() {
	declare K_State TMGame_Record_State for UI;

	foreach (Event in PendingEvents) {
		// Update PB and medal if player improve their time
		if (Event.PlaygroundType == CManiaAppPlaygroundEvent::Type::RecordUpdated) {
			if (Event.Ghost != Null) {
				if (TMGame_Record_State.PBGhostEnabled || TMGame_Record_State.CelebratePB) {
					declare Boolean IsBetter = True;
					if (TMGame_Record_State.PBGhostId != NullId && DataFileMgr.Ghosts.existskey(TMGame_Record_State.PBGhostId)) {
						declare CGhost PBGhost = DataFileMgr.Ghosts[TMGame_Record_State.PBGhostId];
						IsBetter = (
							Event.Ghost.Result.Checkpoints.count > 0 &&
							Event.Ghost.Result.Checkpoints.count == PBGhost.Result.Checkpoints.count &&
							Event.Ghost.Result.Checkpoints[Event.Ghost.Result.Checkpoints.count - 1] < PBGhost.Result.Checkpoints[PBGhost.Result.Checkpoints.count - 1]
						);
					}
					if (IsBetter) {
						TMGame_Record_State = SetPBGhost(TMGame_Record_State, Event.Ghost, TMGame_Record_State.CelebratePB);
					}
				}
				if (TMGame_Record_State.MedalEnabled || TMGame_Record_State.CelebrateMedal || TMGame_Record_State.RecordsEnabled) {
					TMGame_Record_State = RetrieveMedal(TMGame_Record_State, TMGame_Record_State.CelebrateMedal);
				}
				if (TMGame_Record_State.RecordsEnabled) {
					if (Event.Ghost.Result.Checkpoints.count > 0 && Event.Ghost.Result.Checkpoints[Event.Ghost.Result.Checkpoints.count - 1] >= 0) {
						if (TMGame_Record_State.RecordsPlayerTime < 0 || Event.Ghost.Result.Checkpoints[Event.Ghost.Result.Checkpoints.count - 1] < TMGame_Record_State.RecordsPlayerTime) {
							TMGame_Record_State.RecordsPlayerTime = Event.Ghost.Result.Checkpoints[Event.Ghost.Result.Checkpoints.count - 1];
							TMGame_Record_State = SetAllRecordsDirty(TMGame_Record_State);
							SendCustomEvent(Record::C_Event_NewRecord, [""^TMGame_Record_State.RecordsPlayerTime]);
							if (CanRetrieveRecords(TMGame_Record_State)) {
								TMGame_Record_State = RetrieveRecords(TMGame_Record_State, TMGame_Record_State.RecordsPlayerTime);
							}
						}
					}
				}
			}
		} else if (Event.PlaygroundType == CManiaAppPlaygroundEvent::Type::LayerCustomEvent) {
			switch (Event.CustomEventType) {
				case C_Event_SpectateGhost: {
					assert(Event.CustomEventData.count >= 1, "'C_Event_SpectateGhost' needs at least 1 parameter: 'AccountId'.");
					declare Text AccountId = Event.CustomEventData[0];
					declare Boolean IsHiddenScore = False;
					if (Event.CustomEventData.count >= 3) {
						IsHiddenScore = TL::ToInteger(Event.CustomEventData[2]) < 0;
					}

					if (!TL::StartsWith(C_FakeAccountId_MedalGhost, AccountId)) {
						// Remove the ghost added by this UI to avoid conflicts with the ghost to spectate that will be added by the game mode
						if (TMGame_Record_State.LoadedRecordGhosts.existskey(AccountId)) {
							TMGame_Record_State = ReleaseRecordGhost(TMGame_Record_State, AccountId);
						}

						// Ask the game mode to spectate the given ghost
						if (Event.CustomEventData.count >= 2 && TMGame_Record_State.RecordsSelectedZone == C_RecordType_Medals) {
							SendCustomEvent(Record::C_Event_Spectate, [AccountId, ""^Medals::C_ScoreMgrMedal_Author - TL::ToInteger(Event.CustomEventData[1]), ""^IsHiddenScore]);
						} else {
							SendCustomEvent(Record::C_Event_Spectate, [AccountId, ""^Medals::C_ScoreMgrMedal_None, ""^IsHiddenScore]);
						}
					}
				}
				case C_Event_TogglePB: {
					TMGame_Record_State = DisplayPBGhost(TMGame_Record_State, !TMGame_Record_State.DisplayPBGhost);
				}
				case C_Event_ToggleGhost: {
					assert(Event.CustomEventData.count >= 1, "'C_Event_ToggleGhost' needs at least 1 parameter: 'AccountId'.");
					declare Text AccountId = Event.CustomEventData[0];
					declare Boolean IsHiddenScore = False;
					if (Event.CustomEventData.count >= 3) {
						IsHiddenScore = TL::ToInteger(Event.CustomEventData[2]) < 0;
					}

					if (!TL::StartsWith(C_FakeAccountId_MedalGhost, AccountId)) {
						if (TMGame_Record_State.LoadedRecordGhosts.existskey(AccountId)) {
							TMGame_Record_State = ReleaseRecordGhost(TMGame_Record_State, AccountId);
							SendRecordsStatusToML(TMGame_Record_State);
						} else if (!TMGame_Record_State.LoadingRecordGhosts.existskey(AccountId) && !IsHiddenScore) {
							if (UserStore_MA::HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayRecords)) {
								if (Event.CustomEventData.count >= 2 && TMGame_Record_State.RecordsSelectedZone == C_RecordType_Medals) {
									TMGame_Record_State = RetrieveRecordGhost(TMGame_Record_State, AccountId, Medals::C_ScoreMgrMedal_Author - TL::ToInteger(Event.CustomEventData[1]));
								} else {
									TMGame_Record_State = RetrieveRecordGhost(TMGame_Record_State, AccountId, Medals::C_ScoreMgrMedal_None);
								}
							}
						}
					}
				}
			}
		}
	}
}

/// Update the library
Void Yield() {
	declare K_State TMGame_Record_State for UI;

	// Update the library only if the GhostMgr is available
	if (GhostMgr != Null) {
		declare Boolean ForceUpdate = False;

		// Enable or disable record from the server
		declare netread Integer Net_TMGame_Record_SetupUpdate for Playground.Teams[0] = -1;
		if (
			TMGame_Record_State.SetupUpdate != Net_TMGame_Record_SetupUpdate ||
			TMGame_Record_State.RecordsVisibilityFromSettings != UserMgr.MainUserProfile.Interface_DisplayRecords
		) {
			declare netread Boolean Net_TMGame_Record_PBGhostEnabled for Playground.Teams[0] = False;
			declare netread Boolean Net_TMGame_Record_CelebratePB for Playground.Teams[0] = False;
			declare netread Boolean Net_TMGame_Record_MedalEnabled for Playground.Teams[0] = False;
			declare netread Boolean Net_TMGame_Record_CelebrateMedal for Playground.Teams[0] = False;
			declare netread Boolean Net_TMGame_Record_DisplayRecords for Playground.Teams[0] = False;
			declare netread Integer Net_TMGame_Record_ForceMapUpdate for Playground.Teams[0] = -1;
			TMGame_Record_State.SetupUpdate = Net_TMGame_Record_SetupUpdate;
			TMGame_Record_State.PBGhostEnabled = (Net_TMGame_Record_PBGhostEnabled && UserStore_MA::HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_ViewPBGhostMultiplayer));
			TMGame_Record_State.CelebratePB = Net_TMGame_Record_CelebratePB;
			TMGame_Record_State.MedalEnabled = Net_TMGame_Record_MedalEnabled;
			TMGame_Record_State.CelebrateMedal = Net_TMGame_Record_CelebrateMedal;
			TMGame_Record_State.RecordsEnabled = UpdateRecordsEnabled(Net_TMGame_Record_DisplayRecords, TMGame_Record_State.CanViewLeaderboards);

			if (TMGame_Record_State.ForceMapUpdate != Net_TMGame_Record_ForceMapUpdate) {
				TMGame_Record_State.ForceMapUpdate = Net_TMGame_Record_ForceMapUpdate;
				TMGame_Record_State.ReloadRecords = True;
			}

			if (TMGame_Record_State.RecordsVisibilityFromSettings != UserMgr.MainUserProfile.Interface_DisplayRecords) {
				TMGame_Record_State.RecordsVisibilityFromSettings = UserMgr.MainUserProfile.Interface_DisplayRecords;
				if (TMGame_Record_State.RecordsVisibilityFromSettings != CUserV2Profile::EDisplayRecords::Hide) {
					TMGame_Record_State.ReloadRecords = True;
				}
			}

			ForceUpdate = True;
		}

		// Check leaderboard privilege
		if (!Privileges::Check_IsComplete(TMGame_Record_State.LeaderboardPrivilegeCheck)) {
			TMGame_Record_State.LeaderboardPrivilegeCheck = Privileges::Check_Update(TMGame_Record_State.LeaderboardPrivilegeCheck);
			if (Privileges::Check_IsComplete(TMGame_Record_State.LeaderboardPrivilegeCheck)) {
				TMGame_Record_State.LeaderboardPrivilegeCheck = Privileges::Check_Destroy(TMGame_Record_State.LeaderboardPrivilegeCheck);
				if (TMGame_Record_State.CanViewLeaderboards != Privileges::Check_HasPrivilege(TMGame_Record_State.LeaderboardPrivilegeCheck)) {
					TMGame_Record_State.CanViewLeaderboards = Privileges::Check_HasPrivilege(TMGame_Record_State.LeaderboardPrivilegeCheck);

					declare netread Boolean Net_TMGame_Record_DisplayRecords for Playground.Teams[0] = False;
					TMGame_Record_State.RecordsEnabled = UpdateRecordsEnabled(Net_TMGame_Record_DisplayRecords, TMGame_Record_State.CanViewLeaderboards);
					SendRecordsToML(TMGame_Record_State);
					ForceUpdate = True;
				}
			}
		}

		// Setup or map updated
		if (
			ForceUpdate || (
				(
					(Map == Null && TMGame_Record_State.CurrentMapUid != "") ||
					(Map != Null && TMGame_Record_State.CurrentMapUid != Map.MapInfo.MapUid)
				) && (
					TMGame_Record_State.PBGhostEnabled ||
					TMGame_Record_State.CelebratePB ||
					TMGame_Record_State.MedalEnabled ||
					TMGame_Record_State.CelebrateMedal ||
					TMGame_Record_State.RecordsEnabled
				)
			)
		) {
			declare netread Text Net_TMGame_Record_ScopeSeason for Playground.Teams[0] = "";
			declare netread Text Net_TMGame_Record_ScopeNotSeason for Playground.Teams[0] = "";
			declare netread Text Net_TMGame_Record_ModeName for Playground.Teams[0] = "";
			declare netread Text Net_TMGame_Record_CustomData for Playground.Teams[0] = "";
			declare netread Text[Text] Net_TMGame_Record_SeasonIds for Playground.Teams[0] = [];
			TMGame_Record_State = Update(
				TMGame_Record_State,
				Map,
				Net_TMGame_Record_ScopeSeason,
				Net_TMGame_Record_ScopeNotSeason,
				Net_TMGame_Record_ModeName,
				Net_TMGame_Record_CustomData,
				Net_TMGame_Record_SeasonIds
			);
		}

		// PB ghost retrieved
		if (
			TMGame_Record_State.TaskIdGetRecordGhost != NullId &&
			ScoreMgr.TaskResults.existskey(TMGame_Record_State.TaskIdGetRecordGhost) &&
			!ScoreMgr.TaskResults[TMGame_Record_State.TaskIdGetRecordGhost].IsProcessing
		) {
			declare TaskGetRecordGhost = (ScoreMgr.TaskResults[TMGame_Record_State.TaskIdGetRecordGhost] as CTaskResult_Ghost);
			if (TaskGetRecordGhost.HasSucceeded) {
				TMGame_Record_State = SetPBGhost(TMGame_Record_State, TaskGetRecordGhost.Ghost, False);
			}
			ScoreMgr.TaskResult_Release(TaskGetRecordGhost.Id);
			TMGame_Record_State.TaskIdGetRecordGhost = NullId;
		}

		// Trigger celebration screen
		if (
			TMGame_Record_State.RequestCelebration &&
			UI.UISequence == CUIConfig::EUISequence::Finish &&
			UI.ScoreTableVisibility != CUIConfig::EVisibility::ForcedVisible
		) {
			TMGame_Record_State.RequestCelebration = False;
			TMGame_Record_State.Celebration.IsActive = True;
			declare K_Celebration TMGame_Record_Celebration for ClientUI;
			TMGame_Record_Celebration = TMGame_Record_State.Celebration;
		} else if (
			TMGame_Record_State.Celebration.IsActive && (
				UI.UISequence != CUIConfig::EUISequence::Finish ||
				UI.ScoreTableVisibility == CUIConfig::EVisibility::ForcedVisible
			)
		) {
			TMGame_Record_State.Celebration.IsActive = False;
			declare K_Celebration TMGame_Record_Celebration for ClientUI;
			TMGame_Record_Celebration = TMGame_Record_State.Celebration;
		}
	}

	if (ScoreMgr != Null && UserMgr != Null && DataFileMgr != Null) {
		// Initialize the manialink related variables only once the manialink is initialized properly
		declare Boolean TMGame_Record_ManialinkIsInitialized for ClientUI = False;
		if (TMGame_Record_State.ManialinkIsInitialized != TMGame_Record_ManialinkIsInitialized) {
			TMGame_Record_State.ManialinkIsInitialized = TMGame_Record_ManialinkIsInitialized;

			if (TMGame_Record_State.ManialinkIsInitialized) {
				declare Boolean TMGame_Record_PlayerIsDisplayingRecords for ClientUI = False;
				declare Integer TMGame_Record_RecordsSelectedZone for ClientUI = C_RecordType_None;
				TMGame_Record_State.PlayerIsDisplayingRecords = TMGame_Record_PlayerIsDisplayingRecords;
				TMGame_Record_State.RecordsSelectedZone = TMGame_Record_RecordsSelectedZone;
			}
		}

		// Player started displaying the records
		if (TMGame_Record_State.ManialinkIsInitialized) {
			declare Boolean TMGame_Record_PlayerIsDisplayingRecords for ClientUI = False;
			declare Integer TMGame_Record_RecordsSelectedZone for ClientUI = C_RecordType_None;
			if (
				TMGame_Record_State.PlayerIsDisplayingRecords != TMGame_Record_PlayerIsDisplayingRecords ||
				TMGame_Record_State.RecordsSelectedZone != TMGame_Record_RecordsSelectedZone
			) {
				TMGame_Record_State.PlayerIsDisplayingRecords = TMGame_Record_PlayerIsDisplayingRecords;
				TMGame_Record_State.RecordsSelectedZone = TMGame_Record_RecordsSelectedZone;
				if (CanRetrieveRecords(TMGame_Record_State) && RecordsAreDirty(TMGame_Record_State)) {
					TMGame_Record_State = RetrieveRecords(TMGame_Record_State, TMGame_Record_State.RecordsPlayerTime);
				}
			}
		}

		// New record by another player
		declare netread Integer Net_TMGame_Record_NewRecordsUpdate for Playground.Teams[0];
		if (TMGame_Record_State.RecordsServerUpdate != Net_TMGame_Record_NewRecordsUpdate) {
			TMGame_Record_State.RecordsServerUpdate = Net_TMGame_Record_NewRecordsUpdate;

			declare netread Integer[Text][Integer] Net_TMGame_Record_NewRecords for UI;
			declare netwrite Integer[] Net_TMGame_Record_ReceivedRecords for UI;

			// Remove already received records
			declare Integer[] TmpReceivedRecords = Net_TMGame_Record_ReceivedRecords;
			foreach (Key in TmpReceivedRecords) {
				if (!Net_TMGame_Record_NewRecords.existskey(Key)) {
					Net_TMGame_Record_ReceivedRecords.remove(Key);
				}
			}

			// Receive new records
			declare Integer[Text] NewZonesRecordTime;
			declare Text[] LocalUserZones = TL::Split("|", LocalUser.ZonePath);
			foreach (Key => ZonesNewRecord in Net_TMGame_Record_NewRecords) {
				if (!Net_TMGame_Record_ReceivedRecords.exists(Key)) {
					Net_TMGame_Record_ReceivedRecords.add(Key);
					foreach (Zone => Time in ZonesNewRecord) {
						if (Time >= 0 && LocalUserZones.exists(Zone) && (!NewZonesRecordTime.existskey(Zone) || Time < NewZonesRecordTime[Zone])) {
							NewZonesRecordTime[Zone] = Time;
						}
					}
				}
			}

			// Check if the best new record is better than one
			// of the time in the local records
			// If so request an update of the records to retrieve
			// the new record from the API
			if (TMGame_Record_State.RecordsEnabled && NewZonesRecordTime.count > 0) {
				if (TMGame_Record_State.ZonesRecords.count <= 0) {
					TMGame_Record_State = SetAllRecordsDirty(TMGame_Record_State);
				} else {
					foreach (ZoneRecords in TMGame_Record_State.ZonesRecords) {
						declare Boolean HasRecordsForZone = NewZonesRecordTime.existskey(ZoneRecords.ZoneName);
						if (
							(
								ZoneRecords.Type == C_RecordType_Club ||
								ZoneRecords.Type == C_RecordType_ClubVIP ||
								ZoneRecords.Type == C_RecordType_GlobalVIP ||
								HasRecordsForZone
							) && (
								!ZoneRecords.IsFull ||
								(HasRecordsForZone && NewZonesRecordTime[ZoneRecords.ZoneName] <= ZoneRecords.WorstScore)
							)
						) {
							if (
								ZoneRecords.Type == C_RecordType_Club &&
								TMGame_Record_State.RecordsClubId != ClubStruct::C_ClubNullId &&
								!TMGame_Record_State.RecordsAreDirty.exists(C_RecordType_Club)
							) {
								TMGame_Record_State.RecordsAreDirty.add(C_RecordType_Club);
							}
							if (
								ZoneRecords.Type == C_RecordType_ClubVIP &&
								TMGame_Record_State.RecordsClubId != ClubStruct::C_ClubNullId &&
								!TMGame_Record_State.RecordsAreDirty.exists(C_RecordType_ClubVIP)
							) {
								TMGame_Record_State.RecordsAreDirty.add(C_RecordType_ClubVIP);
							}
							if (
								ZoneRecords.Type == C_RecordType_GlobalVIP &&
								TMGame_Record_State.RecordsClubId != ClubStruct::C_ClubNullId &&
								!TMGame_Record_State.RecordsAreDirty.exists(C_RecordType_GlobalVIP)
							) {
								TMGame_Record_State.RecordsAreDirty.add(C_RecordType_GlobalVIP);
							}
							if (ZoneRecords.Type == C_RecordType_Zones && !TMGame_Record_State.RecordsAreDirty.exists(C_RecordType_Zones)) {
								TMGame_Record_State.RecordsAreDirty.add(C_RecordType_Zones);
							}
							if (
								TMGame_Record_State.RecordsAreDirty.count >= 5 &&
								TMGame_Record_State.RecordsAreDirty.exists(C_RecordType_Club) &&
								TMGame_Record_State.RecordsAreDirty.exists(C_RecordType_ClubVIP) &&
								TMGame_Record_State.RecordsAreDirty.exists(C_RecordType_GlobalVIP) &&
								TMGame_Record_State.RecordsAreDirty.exists(C_RecordType_Medals) &&
								TMGame_Record_State.RecordsAreDirty.exists(C_RecordType_Zones)
							) break;
						}
					}
				}
				if (CanRetrieveRecords(TMGame_Record_State) && RecordsAreDirty(TMGame_Record_State)) {
					TMGame_Record_State = RetrieveRecords(TMGame_Record_State, TMGame_Record_State.RecordsPlayerTime);
				}
			}
		}

		// Process records requests
		Private_ProcessRecordsRequests();

		// Get display name of the players in the records
		if (TMGame_Record_State.TaskIdRetrieveDisplayName != NullId) {
			if (UserMgr.TaskResults.existskey(TMGame_Record_State.TaskIdRetrieveDisplayName)) {
				if (!UserMgr.TaskResults[TMGame_Record_State.TaskIdRetrieveDisplayName].IsProcessing) {
					if (UserMgr.TaskResults[TMGame_Record_State.TaskIdRetrieveDisplayName].HasSucceeded) {
						TMGame_Record_State = UpdateDisplayName(TMGame_Record_State);
						SendRecordsToML(TMGame_Record_State);
					}
					UserMgr.TaskResult_Release(TMGame_Record_State.TaskIdRetrieveDisplayName);
				}
			} else {
				TMGame_Record_State.TaskIdRetrieveDisplayName = NullId;
			}
		}

		// Load records selected by the player
		if (TMGame_Record_State.LoadingRecordGhosts.count > 0) {
			declare TmpLoadingRecordGhosts = TMGame_Record_State.LoadingRecordGhosts;
			foreach (AccountId => LoadingRecord in TmpLoadingRecordGhosts) {
				if (Task::IsInitialized(LoadingRecord.Task_RetrieveRecords)) {
					declare Task::K_Task Task_RetrieveRecords_Updated = Task::Update(LoadingRecord.Task_RetrieveRecords);
					if (!Task::IsRunning(Task_RetrieveRecords_Updated)) {
						declare CTaskResult_MapRecordList SourceTask = Task::GetSourceTask_MapRecordList(Task_RetrieveRecords_Updated);
						if (Task::IsSuccess(Task_RetrieveRecords_Updated) && SourceTask != Null && SourceTask.MapRecordList.count >= 1) {
							declare CMapRecord MapRecord <=> SourceTask.MapRecordList[0];
							TMGame_Record_State.LoadingRecordGhosts[AccountId].Task_RetrieveGhost = Task::DestroyAndCreate(
								TMGame_Record_State.LoadingRecordGhosts[AccountId].Task_RetrieveGhost,
								DataFileMgr,
								DataFileMgr.Ghost_Download(MapRecord.FileName, MapRecord.ReplayUrl)
							);
						}
						TMGame_Record_State.LoadingRecordGhosts[AccountId].Task_RetrieveRecords = Task::Destroy(Task_RetrieveRecords_Updated);
					} else {
						TMGame_Record_State.LoadingRecordGhosts[AccountId].Task_RetrieveRecords = Task_RetrieveRecords_Updated;
					}
				} else if (Task::IsInitialized(LoadingRecord.Task_RetrieveGhost)) {
					declare Task::K_Task Task_RetrieveGhost_Updated = Task::Update(LoadingRecord.Task_RetrieveGhost);
					if (!Task::IsRunning(Task_RetrieveGhost_Updated)) {
						declare CTaskResult_Ghost SourceTask = Task::GetSourceTask_Ghost(Task_RetrieveGhost_Updated);
						if (Task::IsSuccess(Task_RetrieveGhost_Updated) && SourceTask != Null && SourceTask.Ghost != Null) {
							declare Text DisplayName = UserStore_MA::GetUserMgrPlayerName(LoadingRecord.AccountId);
							if (DisplayName != "") SourceTask.Ghost.Nickname = DisplayName;
							switch (TMGame_Record_State.LoadingRecordGhosts[AccountId].Medal) {
								case Medals::C_ScoreMgrMedal_Bronze: SourceTask.Ghost.OverrideZoneIcon_ImageUrl = Const::C_ImageUrl_Medal_Bronze;
								case Medals::C_ScoreMgrMedal_Silver: SourceTask.Ghost.OverrideZoneIcon_ImageUrl = Const::C_ImageUrl_Medal_Silver;
								case Medals::C_ScoreMgrMedal_Gold: SourceTask.Ghost.OverrideZoneIcon_ImageUrl = Const::C_ImageUrl_Medal_Gold;
							}
							TMGame_Record_State.LoadingRecordGhosts[AccountId].GhostId = SourceTask.Ghost.Id;
							TMGame_Record_State.LoadingRecordGhosts[AccountId].GhostInstanceId = GhostMgr.Ghost_Add(SourceTask.Ghost, True);
						}
						TMGame_Record_State.LoadingRecordGhosts[AccountId].Task_RetrieveGhost = Task::Destroy(Task_RetrieveGhost_Updated);
					} else {
						TMGame_Record_State.LoadingRecordGhosts[AccountId].Task_RetrieveGhost = Task_RetrieveGhost_Updated;
					}
				} else {
					if (TMGame_Record_State.LoadingRecordGhosts[AccountId].GhostId != NullId) {
						TMGame_Record_State.LoadedRecordGhosts[AccountId] = TMGame_Record_State.LoadingRecordGhosts[AccountId];
					}
					TMGame_Record_State.LoadingRecordGhosts.removekey(AccountId);
					SendRecordsStatusToML(TMGame_Record_State);
					TMGame_Record_State = DisplayPBGhost(TMGame_Record_State, TMGame_Record_State.DisplayPBGhost);
				}
			}
		}

		// Set spectator record account id
		declare netread Text Net_TMGame_Record_SpectatorTargetAccountId for Playground.Teams[0] = "";
		if (TMGame_Record_State.SpectatorTargetRecordGhost.AccountId != Net_TMGame_Record_SpectatorTargetAccountId) {
			TMGame_Record_State = ReleaseRecordGhost(TMGame_Record_State, TMGame_Record_State.SpectatorTargetRecordGhost.AccountId);
			TMGame_Record_State.SpectatorTargetRecordGhost = K_RecordGhost {
				AccountId = Net_TMGame_Record_SpectatorTargetAccountId
			};

			SendSpectatorTargetAccountIdToML(TMGame_Record_State);
			SendRecordsStatusToML(TMGame_Record_State);
		}
	}

	Private_ProcessEvents();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Unload the library
Void Unload() {
	// Wait for the manialink to be initialized before
	declare Boolean TMGame_Record_ManialinkIsInitialized for ClientUI = False;
	TMGame_Record_ManialinkIsInitialized = False;

	// Save the state in a declare for variable so data survive a script restart
	// The ghost instances are not removed between script restarts
	// so we must keep their id to remove them manually
