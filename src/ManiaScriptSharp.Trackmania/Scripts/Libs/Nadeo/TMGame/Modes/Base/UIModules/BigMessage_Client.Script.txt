/// Update the message visibility
Boolean SetMessageVisibility(K_Controls _Controls, Boolean _IsVisible) {
	AnimMgr.Flush(_Controls.Label_Message);
	AnimMgr.Flush(_Controls.Frame_Message);

	// Prepare controls for the upcoming animation
	if (_IsVisible) {
		_Controls.Frame_Message.Hide();
		_Controls.Label_Message.Opacity = 0.;
		_Controls.Frame_Message.RelativeScale = 1.2;
		Tools::FirstPartyLogoMatchLuminance(_Controls.Label_Message);
		Tools::FitLabelValue(_Controls.Label_Message, {{{C_TextSize}}}, 3., 0.5);
		{{{ExpendableAsset::P}}}SetWidth(This, _Controls.Asset_Background, ML::Min(_Controls.Label_Message.Size.X, _Controls.Label_Message.ComputeWidth(_Controls.Label_Message.Value)) + 24., False);
		{{{ExpendableAsset::P}}}SetOpacity(This, _Controls.Asset_Background, 0., False);
		{{{ExpendableAsset::P}}}RedrawAsset(This, _Controls.Asset_Background, False);
	}

	// Animations
	if (_IsVisible) {
		AnimMgr.Add(_Controls.Label_Message, "<anim opacity=\"1.\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		AnimMgr.Add(_Controls.Frame_Message, "<anim scale=\"1.\" hidden=\"0\" />", 250, CAnimManager::EAnimManagerEasing::BackOut);
		{{{ExpendableAsset::P}}}SetOpacity(This, _Controls.Asset_Background, 0.7);
	} else {
		AnimMgr.Add(_Controls.Label_Message, "<anim opacity=\"0.\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		AnimMgr.Add(_Controls.Frame_Message, "<anim scale=\"1.2\" hidden=\"1\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		{{{ExpendableAsset::P}}}SetOpacity(This, _Controls.Asset_Background, 0.);
	}

	return _IsVisible;
}
