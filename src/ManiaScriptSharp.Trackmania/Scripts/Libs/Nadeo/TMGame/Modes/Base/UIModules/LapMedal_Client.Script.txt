/// Get the manialink template
Text Private_GetTemplate() {
	declare Layout::K_Element[] ElementList = [
		Layout::K_Element {
			Id = "container-root",
			Direction = Layout::C_Direction_TopToBottom,
			Width = Layout::NewSizingFixed(C_Private_UISize.X),

			ChildGap = 1.

		Layout::K_Element {
			ParentId = "container-root",
			Id = "title",
			Width = Layout::NewSizingGrow(),
			Height = Layout::NewSizingFixed(5.)
		},
		Layout::K_Element {
			ParentId = "container-root",
			Id = "counter",
			Width = Layout::NewSizingGrow(),
			Height = Layout::NewSizingFixed(5.)
		},
		Layout::K_Element {
			ParentId = "container-root",
			Id = "container-medal-grid",
			Direction = Layout::C_Direction_LeftToRight,
			Width = Layout::NewSizingGrow(),
			Height = Layout::NewSizingGrow(),
			ChildGap = 1.
		}
	];
	for (Medal, Medals::C_ScoreMgrMedal_Bronze, Medals::C_ScoreMgrMedal_Author) {
		ElementList.add(Layout::K_Element {
			ParentId = "container-medal-grid",
			Id = Layout::NewId("container-medal-column", Medal),
			Direction = Layout::C_Direction_TopToBottom,
			Width = Layout::NewSizingGrow(),
			Height = Layout::NewSizingGrow(),
			ChildGap = 1.
		});
		for (Line, 1, C_Private_MedalLineNb) {
			ElementList.add(Layout::K_Element {
				ParentId = Layout::NewId("container-medal-column", Medal),
				Id = Layout::NewId("container-medal-line", Medal, Line),
				Direction = Layout::C_Direction_LeftToRight,
				Width = Layout::NewSizingGrow(),
				Height = Layout::NewSizingGrow(),
				ChildGap = 0.
			});
			ElementList.add(Layout::K_Element {
				ParentId = Layout::NewId("container-medal-line", Medal, Line),
				Id = Layout::NewId("medal-icon", Medal, Line),
				Width = Layout::NewSizingGrow(),
				Height = Layout::NewSizingGrow()
			});
			ElementList.add(Layout::K_Element {
				ParentId = Layout::NewId("container-medal-line", Medal, Line),
				Id = Layout::NewId("medal-lap", Medal, Line),
				Width = Layout::NewSizingGrow(),
				Height = Layout::NewSizingGrow()
			});
		}
	}
	declare Layout::K_Layout Layout = Layout::NewLayout(<320., 160.>, "container-root", ElementList);

	declare Vec2 Size_Title = Layout::GetSize(Layout, "title");
	declare Vec2 Pos_Title = Layout::GetPosition(Layout, "title") + <Size_Title.X * 0.5, Size_Title.Y * -0.5>;
	declare Vec2 Size_Counter = Layout::GetSize(Layout, "counter");
	declare Vec2 Pos_Counter = Layout::GetPosition(Layout, "counter") + <Size_Counter.X * 0.5, Size_Counter.Y * -0.5>;

	declare Text MedalIconListML;
	declare Text MedalLapListML;
	declare Integer MedalZIndex;
	for (Medal, Medals::C_ScoreMgrMedal_Bronze, Medals::C_ScoreMgrMedal_Author) {
		for (Line, 1, C_Private_MedalLineNb) {
			declare Vec2 Size_MedalIcon = Layout::GetSize(Layout, Layout::NewId("medal-icon", Medal, Line));
			declare Vec2 Pos_MedalIcon = Layout::GetPosition(Layout, Layout::NewId("medal-icon", Medal, Line)) + <Size_MedalIcon.X * 0.5, Size_MedalIcon.Y * -0.5>;
			declare Vec2 Size_MedalLap = Layout::GetSize(Layout, Layout::NewId("medal-lap", Medal, Line));
			declare Vec2 Pos_MedalLap = Layout::GetPosition(Layout, Layout::NewId("medal-lap", Medal, Line)) + <Size_MedalLap.X * 0.5, Size_MedalLap.Y * -0.5>;

			MedalIconListML ^=
"""
<quad pos="{{{Pos_MedalIcon.X}}} {{{Pos_MedalIcon.Y}}}" z-index="{{{MedalZIndex}}}" size="{{{Size_MedalIcon.X}}} {{{Size_MedalIcon.Y}}}" halign="center" valign="center" keepratio="fit" opacity="{{{C_Private_Opacity_Locked}}}" />
"""
			;
			MedalLapListML ^=
"""
<label pos="{{{Pos_MedalLap.X}}} {{{Pos_MedalLap.Y}}}" z-index="{{{MedalZIndex}}}" size="{{{Size_MedalLap.X}}} {{{Size_MedalLap.Y}}}" halign="center" valign="center2" textfont="{{{Stylesheet::GetFont(Stylesheet::C_Font_Weight_SemiBold)}}}" textprefix="$t$i" textsize="{{{C_Private_TextSize_MedalLap.X}}}" />
"""
			;
		}
	}

	return
"""
<frame z-index="-2" id="{{{C_Private_Id_Frame_Global}}}">
	<frame {{{UIModules::CustomizableUIModule(UIModules_LapMedal_Common::C_UIModuleConfig)}}}>
		<frame size="{{{C_Private_UISize.X}}} {{{C_Private_UISize.Y}}}" halign="right">
			<frame id="{{{C_Private_Id_Frame_AnimVisible}}}">
				<quad z-index="0" size="{{{C_Private_UISize.X}}} {{{C_Private_UISize.Y}}}" bgcolor="{{{Stylesheet::GetColorHex6(Stylesheet::C_Color_BGDark)}}}" opacity="0.6" />
				<label id="{{{C_Private_Id_Label_Title}}}" pos="{{{Pos_Title.X}}} {{{Pos_Title.Y}}}" z-index="1" size="{{{Size_Title.X}}} {{{Size_Title.Y}}}" halign="center" valign="center2" textfont="{{{Stylesheet::GetFont(Stylesheet::C_Font_Weight_ExtraBold)}}}" textcolor="{{{Stylesheet::GetColorHex6(Stylesheet::C_Color_TextLight)}}}" textprefix="$t$i" textsize="{{{C_Private_TextSize_Title.X}}}" text="{{{C_Private_Text_LapMedals}}}" />

					{{{MedalIconListML}}}
				</frame>
				<frame id="{{{C_Private_Id_Frame_MedalLapList}}}" z-index="4">
					{{{MedalLapListML}}}

		</frame>
	</frame>

"""

}

Void Private_UpdateVisibility(CSmMlScriptIngame _Context, K_Private_Instance _Instance) {
	declare Boolean IsVisible = (
		_Instance.IsModuleVisible &&
		_Instance.LapCount >= 0 && (
			_Instance.DisplaySpeedLimit >= 0. ||
			_Instance.DisplayTimeLimit >= 0
		)
	);

	_Context.AnimMgr.Flush(_Instance.Controls.Frame_AnimVisible);
	if (IsVisible) {
		_Context.AnimMgr.Add(_Instance.Controls.Frame_AnimVisible, "<anim pos=\""^-C_Private_UISize.X^" 0\" hidden=\"0\" />", C_Private_VisibleAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);

		_Context.AnimMgr.Add(_Instance.Controls.Frame_AnimVisible, "<anim pos=\"0 0\" hidden=\"1\" />", C_Private_VisibleAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
	}
}

/// Send the lap count to the manialink
Void Private_SendLapCount(CManiaAppPlaygroundCommon _Context, K_Private_Instance _Instance) {
	declare CUILayer Layer = UIModules::GetModuleLayer(GetId());
	if (Layer != Null) {
		declare Integer TMGame_ModesBaseUIModulesLapMedal_LapCount for Layer.LocalPage = -1;
		TMGame_ModesBaseUIModulesLapMedal_LapCount = _Instance.LapCount;
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the current map uid
K_Private_Instance Private_SetMapUid(CManiaAppPlaygroundCommon _Context, K_Private_Instance _Instance, CMap _Map) {
	declare K_Private_Instance Instance = _Instance;

	if (_Map == Null || _Map.MapInfo == Null) {
		Instance.MapUid = "";
		Instance.LapCount = -1;
	} else if (Instance.MapUid != _Map.MapInfo.MapUid) {
		Instance.MapUid = _Map.MapInfo.MapUid;
		Instance.LapCount = -1;

		if (_Context.DataFileMgr != Null && _Context.UserMgr != Null && _Context.UserMgr.MainUser != Null) {
			Instance.Task_GetZenCount = Task::DestroyAndCreate(
				_Context.UserMgr,
				_Context.DataFileMgr,
				Instance.Task_GetZenCount,
				_Context.DataFileMgr.Map_NadeoServices_GetZenCount(_Context.UserMgr.MainUser.Id, Instance.MapUid)
			);
		}
	}

	Private_SendLapCount(_Context, Instance);

	return Instance;
}

/// ManiaApp update
Void Yield(CManiaAppPlaygroundCommon _Context) {
	declare K_Private_Instance TMGame_ModesBaseUIModulesLapMedal_Instance as Instance for _Context = K_Private_Instance {};

	if (
		(_Context.Map != Null && _Context.Map.MapInfo != Null && Instance.MapUid != _Context.Map.MapInfo.MapUid) ||
		((_Context.Map == Null || _Context.Map.MapInfo == Null) && Instance.MapUid != "")
	) {
		Instance = Private_SetMapUid(_Context, Instance, _Context.Map);
	}

	if (
		Task::IsProcessing(Instance.Task_GetZenCount) &&
		_Context.UserMgr != Null &&
		_Context.DataFileMgr != Null
	) {
		Instance.Task_GetZenCount = Task::Update(_Context.UserMgr, _Context.DataFileMgr, Instance.Task_GetZenCount);
		if (!Task::IsProcessing(Instance.Task_GetZenCount)) {
			declare CWebServicesTaskResult_Natural SourceTask = Task::GetTaskResult_Natural(_Context.DataFileMgr, Instance.Task_GetZenCount);
			if (Task::IsSuccess(Instance.Task_GetZenCount) && SourceTask != Null) {
				Instance.LapCount = SourceTask.Value;
			} else {
				Instance.LapCount = -1;
			}
			Task::Destroy(_Context.UserMgr, _Context.DataFileMgr, Instance.Task_GetZenCount);
			Private_SendLapCount(_Context, Instance);
		}
	}

	if (
		Task::IsProcessing(Instance.Task_IncrementZenCount) &&
		_Context.UserMgr != Null &&
		_Context.DataFileMgr != Null
	) {
		Instance.Task_IncrementZenCount = Task::Update(_Context.UserMgr, _Context.DataFileMgr, Instance.Task_IncrementZenCount);
		if (!Task::IsProcessing(Instance.Task_IncrementZenCount)) {
			declare CWebServicesTaskResult_Natural SourceTask = Task::GetTaskResult_Natural(_Context.DataFileMgr, Instance.Task_IncrementZenCount);
			if (Task::IsSuccess(Instance.Task_IncrementZenCount) && SourceTask != Null && Instance.MapUid == Instance.IncrementMapUid) {
				Instance.LapCount = SourceTask.Value;
			}
			Task::Destroy(_Context.UserMgr, _Context.DataFileMgr, Instance.Task_IncrementZenCount);
			Private_SendLapCount(_Context, Instance);
		}
	}

	foreach (Event in _Context.PendingEvents) {
		if (
			Event.PlaygroundType == CManiaAppPlaygroundEvent::Type::OnPlayerTriggerWaypoint &&
			(Event.IsNewLap || Event.IsFinish) &&
			_Context.DataFileMgr != Null &&
			_Context.UserMgr != Null &&
			_Context.UserMgr.MainUser != Null &&
			_Context.Map != Null &&
			_Context.Map.MapInfo != Null &&
			_Context.Map.MapInfo.MapUid != ""
		) {
			Instance.IncrementMapUid = _Context.Map.MapInfo.MapUid;
			Instance.Task_IncrementZenCount = Task::DestroyAndCreate(
				_Context.UserMgr,
				_Context.DataFileMgr,
				Instance.Task_IncrementZenCount,
				_Context.DataFileMgr.Map_NadeoServices_IncrZenCount(_Context.UserMgr.MainUser.Id, _Context.Map.MapInfo.MapUid)
			);
		}
	}
}
