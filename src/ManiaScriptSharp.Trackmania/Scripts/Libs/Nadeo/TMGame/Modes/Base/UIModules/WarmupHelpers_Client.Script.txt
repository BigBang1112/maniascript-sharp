/** ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
 * Those variables for Teams[0] are set directly in
 * the Warmup.Script.txt (via WarmupHelpers_Server.Script.txt)
 */
Boolean IsWarmupActive(CTeam _Team) {
	declare netread Boolean Net_Race_WarmupHelpers_IsWarmupActive for _Team = False;
	return Net_Race_WarmupHelpers_IsWarmupActive;
}

Integer CurrentWarmupEndTime(CTeam _Team) {
	declare netread Integer Net_Race_WarmupHelpers_CurrentWarmUpEndTime for _Team = 0;
	return Net_Race_WarmupHelpers_CurrentWarmUpEndTime;
}

Integer GetWarmUpMode(CTeam _Team) {
	declare netread Integer Net_Race_WarmupHelpers_WarmUpMode for _Team = C_Mode_TimeAttack;
	return Net_Race_WarmupHelpers_WarmUpMode;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get warm-up round progression
Int2 GetWarmUpProgression(CTeam _Team) {
	declare netread Int2 Net_Race_WarmupHelpers_WarmUpProgression for _Team = <0, 0>;
	return Net_Race_WarmupHelpers_WarmUpProgression;
}
