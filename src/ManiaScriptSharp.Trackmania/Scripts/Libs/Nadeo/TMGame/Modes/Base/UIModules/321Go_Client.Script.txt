/**
 *  UI module: 321Go client side
 */
#Const Version    "1.0.0"
#Const ScriptName "Libs/Nadeo/TMGame/Modes/Base/UIModules/321Go_Client.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Utils/ManiaView2.Script.txt" as MV

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Id "UIModule_Race_321Go"
#Const C_Type CUILayer::EUILayerType::Normal

#Const C_SoundsTime [1500, 1000, 500, 0] //< Times before the start at which the sound are played
#Const C_SoundStartDelay 250

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Public
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Get the module manialink
 *
 *  @return                           The module manialink
 */
Text GetML() {
	return MV::Create(
C_Id, 3,
"""
""",
"""
#Include "Libs/Nadeo/TMGame/Modes/MV_Utils.Script.txt" as MV_Utils

#Const C_SoundsTime {{{dump(C_SoundsTime)}}}
#Const C_SoundStartDelay {{{dump(C_SoundStartDelay)}}}
""",
"""
***MainInit***
***
declare Boolean DisplayModule;
declare Integer[] SoundsToPlay;
declare Integer RaceStartTime;
***

***MainStart***
***
DisplayModule = True;
SoundsToPlay = [];
RaceStartTime = -1;
***

***MainLoop***
***
declare Owner <=> MV_Utils::GetOwner(This);
if (Owner != Null) {
	declare netread Boolean Net_Race_321Go_IsVisible for Owner;
	if (DisplayModule != Net_Race_321Go_IsVisible) {
		DisplayModule = Net_Race_321Go_IsVisible;
	}
}

if (Owner == Null && DisplayModule) {
	DisplayModule = False;
}

if (
	Owner != Null && (
		Owner.SpawnStatus == CSmPlayer::ESpawnStatus::Spawning ||
		Owner.SpawnStatus == CSmPlayer::ESpawnStatus::Spawned
	)
) {
	if (RaceStartTime != Owner.StartTime) {
		RaceStartTime = Owner.StartTime;
		if (RaceStartTime >= 0) {
			// Add a delay to avoid missing the first sound of the countdown
			// because of a long script update
			declare Integer TimeBeforeStart = Owner.StartTime - GameTime + C_SoundStartDelay;
			foreach (SoundTime in C_SoundsTime) {
				if (SoundTime <= TimeBeforeStart) {
					SoundsToPlay.add(SoundTime);
				}
			}
		}
	}
	if (SoundsToPlay.count > 0) {
		declare Integer TimeBeforeStart = Owner.StartTime - GameTime;
		if (TimeBeforeStart <= SoundsToPlay[0]) {
			Audio.PlaySoundEvent(CAudioManager::ELibSound::Countdown, SoundsToPlay.count - 1, 0.);
			SoundsToPlay.removekey(0);
		}
	}
}
***
""",
[],
[]
	);
}
