// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries

#Include "TextLib" as TL

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_ScoreTextOptions {
	Boolean EnableAbbreviation;
	Boolean EnableHiddenScore;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Type of scores
#Const C_ScoreType_Time 0
#Const C_ScoreType_Points 1
#Const C_ScoreType_Respawns 2

#Const C_Private_HiddenScoreTime -1

#Const C_ScoreTextOptions_Null K_ScoreTextOptions {
	EnableAbbreviation = False,
	EnableHiddenScore = False
}

//L16N Points abreviation. %1 will be replaced with the points. e.g. "42 pts"
#Const C_Text_Pts _("|Points|%1 pts")
//L16N Display the number of points of the player record, in this case more than 1 so "point" is plural, %1 will be replaced by the number of points, e.g "52 points"
#Const C_Text_RecordPoints _("%1 points")
//L16N Display the number of points of the player record, in this case 1 so "point" is singular
#Const C_Text_RecordPoint _("1 point")
//L16N Display the number of points of the player record, in this case there is no record set
#Const C_Text_RecordNoPoints "-"

//L16N Display the number of respawns of the player record, in this case 0 so "0 respawns"
#Const C_Text_RecordRespawn0 _("0 respawns")
//L16N Display the number of respawns of the player record, in this case 1 so "respawn" is singular
#Const C_Text_RecordRespawn1 _("1 respawn")
//L16N Display the number of respawns of the player record, in more than 1 so "respawn" is plural, %1 will be replaced by the number of respawns, e.g "15 respawns"
#Const C_Text_RecordRespawns _("%1 respawns")
//L16N Display the number of respawns of the player record, in this case there is no record set
#Const C_Text_RecordNoRespawns "-"

//L16N Player times that are better than a certain value will be hidden and replaced with this message on the leaderboards. e.g. instead of "1. PlayerName 01:23.456" players will see "1. PlayerName Secret". Try to stay under 8 characters if possible.
#Const C_Private_Text_HiddenScoreTime _("|RankingTop|Secret")
#Const C_Private_Text_EmptyScoreTime "--:--.---"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Public
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Get the value of the text displayed for the score (can be a time, points or respawns)
Text GetScoreText(Integer _Score, Integer _ScoreType, K_ScoreTextOptions _Options) {
	declare Text ScoreText = "";

	switch (_ScoreType) {
		case C_ScoreType_Time: {
			if (_Options.EnableHiddenScore && _Score == C_Private_HiddenScoreTime) {
				ScoreText = C_Private_Text_HiddenScoreTime;
			} else if (_Score <= 0) {
				ScoreText = C_Private_Text_EmptyScoreTime;
			} else {
				ScoreText = TL::TimeToText(_Score, True, True);
			}
		}
		case C_ScoreType_Points: {
			if (_Score <= 0) {
				ScoreText = C_Text_RecordNoPoints;
			} else if (_Score == 1) {
				ScoreText = C_Text_RecordPoint;
			} else if (_Options.EnableAbbreviation) {
				ScoreText = TL::Compose(C_Text_Pts, TL::FormatInteger(_Score, 1, True));
			} else {
				ScoreText = TL::Compose(C_Text_RecordPoints, TL::FormatInteger(_Score, 1, True));
			}
		}
		case C_ScoreType_Respawns: {
			if (_Options.EnableAbbreviation) {
				ScoreText = TL::FormatInteger(_Score, 1, True);
			} else if (_Score < 0) {
				ScoreText = C_Text_RecordNoRespawns;
			} else if (_Score == 0) {
				ScoreText = C_Text_RecordRespawn0;
			} else if (_Score == 1) {
				ScoreText = C_Text_RecordRespawn1;
			} else {
				ScoreText = TL::Compose(C_Text_RecordRespawns, TL::FormatInteger(_Score, 1, True));
			}
		}
		default: {
			ScoreText = ""^_Score;
		}
	}

	return ScoreText;
}
Text GetScoreText(Integer _Score, Integer _ScoreType, Boolean _EnableAbbreviation) {
	declare K_ScoreTextOptions Options = C_ScoreTextOptions_Null;
	Options.EnableAbbreviation = _EnableAbbreviation;
	return GetScoreText(_Score, _ScoreType, Options);
}
Text GetScoreText(Integer _Score, Integer _ScoreType) {
	return GetScoreText(_Score, _ScoreType, C_ScoreTextOptions_Null);
}
