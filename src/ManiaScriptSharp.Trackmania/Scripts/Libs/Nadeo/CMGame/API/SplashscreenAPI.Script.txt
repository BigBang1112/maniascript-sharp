/** Start a request to get the splashscreen to display
 *
 * @deprecated Use `Request_GetSplashscreenList()` instead
 *
 * @param _TargetSubscription    The current player subscription
 * @param _Locale                The language of the client
 * @param _ZoneId                The current player zone id
 * @return                       The request
 */
Http::K_Request GetSplashscreen(Text _TargetSubscription, Text _Locale, Text _ZoneId) {
	declare Text[Text] Query;

	if (_Locale != "") Query[C_QueryParameter_Locale] = _Locale;
	if (_ZoneId != "") Query[C_QueryParameter_ZoneId] = _ZoneId;
	if (C_GameSubscriptionPlans.exists(_TargetSubscription)) {
		Query[C_QueryParameter_TargetSubscription] = _TargetSubscription;
	}

	return Http::CreateGet(Config::Get().APIBaseUrl^C_API_Splashscreen^C_Route_GetSplashscreen^Http::CreateQueryString(Query), C_Headers);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Parse the response of the GetSplashscreen request
 *
 * @deprecated Use `Response_GetSplashscreenList()` instead
 *
 * @param _Request    The GetSplashscreen request
 * @return            The parsed request
 */
SplashscreenStruct::K_Splashscreen GetResponseFromGetSplashscreen(Http::K_Request _Request) {
	declare K_HttpResponseFromGetSplashscreen HttpResponseFromGetSplashscreen;
	HttpResponseFromGetSplashscreen.fromjson(Http::GetResult(_Request));

	return SplashscreenStruct::K_Splashscreen {
		Uid = HttpResponseFromGetSplashscreen.uid,
		ActionType = HttpResponseFromGetSplashscreen.actionType,
		ActionIdentifier = HttpResponseFromGetSplashscreen.actionIdentifier,
		Url = HttpResponseFromGetSplashscreen.splashscreen16x9Url,
		IsVideo = HttpResponseFromGetSplashscreen.video,
		IsLooping = HttpResponseFromGetSplashscreen.loop
	};
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Start a request to indicate to the API that the given splashscreen has been displayed to the player.
 *
 * @deprecated Use `Request_PostCreateImpression()` instead
 *
 * @param _SplashscreenUid    The uid of the displayed splashscreen
 * @param _ViewTime           The time spend by the player on the splashscreen (in milliseconds)
 * @param _IsClicked          True if the player has clicked on the splashscreen, False otherwise.
 * @return                    The request
 */
Http::K_Request PostCreateImpression(Text _SplashscreenUid, Integer _ViewTime, Boolean _IsClicked) {
	declare K_HttpPostCreateImpression HttpPostCreateImpression = K_HttpPostCreateImpression {
		splashscreenUid = _SplashscreenUid,
		viewTime = _ViewTime,
		clicked = _IsClicked
	};
	return Http::CreatePost(Config::Get().APIBaseUrl^C_API_Splashscreen^C_Route_PostCreateImpression, HttpPostCreateImpression.tojson(), C_Headers);
}
Http::K_Request Request_PostCreateImpression(CHttpManager _Http, Text _SplashscreenUid, Integer _ViewTime, Boolean _IsClicked) {
	declare K_HttpPostCreateImpression HttpPostCreateImpression = K_HttpPostCreateImpression {
		splashscreenUid = _SplashscreenUid,
		viewTime = _ViewTime,
		clicked = _IsClicked
	};
	return Http::CreatePost(_Http, Config::Get().APIBaseUrl^C_API_Splashscreen^C_Route_PostCreateImpression, HttpPostCreateImpression.tojson(), C_Headers);
}

/// Start a request to get the splashscreen to display
Http::K_Request Request_GetSplashscreenList(CHttpManager _Http, Text _TargetSubscription, Text _Locale, Text _ZoneId) {
	declare Text[Text] Query;
	if (_Locale != "") Query[C_QueryParameter_Locale] = _Locale;
	if (_ZoneId != "") Query[C_QueryParameter_ZoneId] = _ZoneId;
	if (C_GameSubscriptionPlans.exists(_TargetSubscription)) {
		Query[C_QueryParameter_TargetSubscription] = _TargetSubscription;
	}

	declare Text Route = Http::InjectRouteParameters(C_Route_GetSplashscreenList, [C_RouteParameter_SplashscreenType => C_SplashscreenType_Splashscreen]);

	return Http::CreateGet(_Http, Config::Get().APIBaseUrl^C_API_Splashscreen^Route^Http::CreateQueryString(Query), C_Headers);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the response to the `GetSplashscreen` request
SplashscreenStruct::K_Splashscreen[] Response_GetSplashscreenList(Http::K_Request _Request) {
	declare K_Private_HttpResponseGetSplashscreenList HttpResponseGetSplashscreenList;
	HttpResponseGetSplashscreenList.fromjson(Http::GetResult(_Request));

	declare SplashscreenStruct::K_Splashscreen[] SplashscreenList;
	for (Splashscreen in HttpResponseGetSplashscreenList.list) {
		SplashscreenList.add(SplashscreenStruct::K_Splashscreen {
			Uid = Splashscreen.uid,
			ActionType = Splashscreen.actionType,
			ActionIdentifier = Splashscreen.actionIdentifier,
			Url = Splashscreen.mediaUrl,
			IsVideo = Splashscreen.video,
			IsLooping = Splashscreen.loop
		});
	}

	return SplashscreenList;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the response to the `GetLoadingScreenList` request
K_ResponseGetLoadingScreenList Response_GetLoadingScreenList(Http::K_Request _Request) {
	declare K_Private_HttpResponseGetSplashscreenList HttpResponseGetLoadingScreenList;
	HttpResponseGetLoadingScreenList.fromjson(Http::GetResult(_Request));

	declare K_ResponseGetLoadingScreenList ResponseGetLoadingScreenList = C_ResponseGetLoadingScreenList_Null;

	ResponseGetLoadingScreenList.RelativeUpdateTime = HttpResponseGetLoadingScreenList.relativeNextRequest;

	for (LoadingScreen in HttpResponseGetLoadingScreenList.list) {
		ResponseGetLoadingScreenList.List.add(SplashscreenStruct::K_LoadingScreen {
			Uid = LoadingScreen.uid,
			MediaUrl = LoadingScreen.mediaUrl,
			Message = LoadingScreen.text,
			DisplayPercentage = LoadingScreen.displayPercentage
		});
	}

	return ResponseGetLoadingScreenList;
}
