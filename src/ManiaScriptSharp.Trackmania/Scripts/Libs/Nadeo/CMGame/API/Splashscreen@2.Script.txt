/**
 * Interact with the Splashscreen web API
 */
#Const Version "2.0.0"
#Const ScriptName "Libs/Nadeo/CMGame/API/Splashscreen@2.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Utils/Config.Script.txt" as Config
#Include "Libs/Nadeo/CMGame/Utils/Http@2.Script.txt" as Http
#Include "Libs/Nadeo/CMGame/Shared/Splashscreen.Script.txt" as SplashscreenShared

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_Private_ResponseGetSplashscreen {
	Text uid;
	Text actionType;
	Boolean loop;
	Text actionIdentifier;
	Text splashscreen16x9Url;
	Boolean video;
}

#Struct K_Private_DataPostCreateImpression {
	Text splashscreenUid;
	Integer viewTime; //< In milliseconds
	Boolean clicked;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Route_GetSplashscreen "/api/token/splashscreen"
#Const C_Route_PostCreateImpression "/api/token/splashscreen/create-impression"

#Const C_QueryParameter_TargetSubscription "targetSubscription"
#Const C_QueryParameter_Locale "locale"
#Const C_QueryParameter_ZoneId "zoneId"

#Const SplashscreenShared::C_GameSubscriptionPlans as C_GameSubscriptionPlans

#Const C_DefaultRequestOptions Http::K_RequestOptions {
	Headers = [
		"Accept" => "application/json",
		"Content-Type" => "application/json"
	]
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Start a request to get the splashscreen to display
 *
 * @param _TargetSubscription    The current player subscription
 * @param _Locale                The language of the client
 * @param _ZoneId                The current player zone id
 * @return                       The request
 */
Http::K_Request Request_GetSplashscreen(
	CSystemPlatform _System,
	CHttpManager _HttpMgr,
	Text _TargetSubscription,
	Text _Locale,
	Text _ZoneId
) {
	declare Text[Text] Query;
	if (_Locale != "") Query[C_QueryParameter_Locale] = _Locale;
	if (_ZoneId != "") Query[C_QueryParameter_ZoneId] = _ZoneId;
	if (C_GameSubscriptionPlans.exists(_TargetSubscription)) {
		Query[C_QueryParameter_TargetSubscription] = _TargetSubscription;
	}

	return Http::CreateGet(
			_System,
		_HttpMgr,
		Config::Get().APIBaseUrl^C_Route_GetSplashscreen^Http::CreateQueryString(Query),
		C_DefaultRequestOptions
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the response to the `GetSplashscreen` request
SplashscreenShared::K_Splashscreen Response_GetSplashscreen(Http::K_Request _Request) {
	declare K_Private_ResponseGetSplashscreen Http_GetSplashscreen;
	Http_GetSplashscreen.fromjson(Http::GetResult(_Request));
	return SplashscreenShared::K_Splashscreen {
		Uid = Http_GetSplashscreen.uid,
		ActionType = Http_GetSplashscreen.actionType,
		ActionIdentifier = Http_GetSplashscreen.actionIdentifier,
		Url = Http_GetSplashscreen.splashscreen16x9Url,
		IsVideo = Http_GetSplashscreen.video,
		IsLooping = Http_GetSplashscreen.loop
	};
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Start a request to increment the number of impression of a splashscreen
 *
 * @param _SplashscreenUid    The uid of the displayed splashscreen
 * @param _ViewTime           The time spent by the player on the splashscreen (in milliseconds)
 * @param _IsClicked          True if the player has clicked on the splashscreen, False otherwise.
 * @return                    The request
 */
Http::K_Request Request_PostCreateImpression(
	CSystemPlatform _System,
	CHttpManager _HttpMgr,
	Text _SplashscreenUid,
	Integer _ViewTime,
	Boolean _IsClicked
) {
	declare K_Private_DataPostCreateImpression DataPostCreateImpression = K_Private_DataPostCreateImpression {
		splashscreenUid = _SplashscreenUid,
		viewTime = _ViewTime,
		clicked = _IsClicked
	};
	return Http::CreatePost(
		_System,
		_HttpMgr,
		Config::Get().APIBaseUrl^C_Route_PostCreateImpression,
		DataPostCreateImpression.tojson(),
		C_DefaultRequestOptions
	);
}
