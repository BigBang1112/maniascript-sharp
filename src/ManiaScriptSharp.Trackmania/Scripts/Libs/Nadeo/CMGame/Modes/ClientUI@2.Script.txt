/**
 * Control the ClientUI values
 * Several scripts can try to modify the ClientUI values
 * This can lead to races conditions difficult to debug
 * To avoid that, any modification of a ClientUI value
 * must go through this library
 */
#Const Version "2.1.0"
#Const ScriptName "Libs/Nadeo/CMGame/Modes/ClientUI@2.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constant
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_UISequenceMode_FinishRace 1
#Const C_UISequenceMode_WatchRecordWithMenuNavigation 2
#Const C_UISequenceMode_SystemUIInteraction 3

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the applied UI sequence
Void Private_ApplyUISequenceModes(CUIConfig _ClientUI, Text[][Integer] _Modes) {
	if (_Modes.count <= 0) {
		_ClientUI.UISequence = CUIConfig::EUISequence::None;
	} else if (_Modes.existskey(C_UISequenceMode_FinishRace)) {
		_ClientUI.UISequence = CUIConfig::EUISequence::Finish;
	} else if (
		_Modes.existskey(C_UISequenceMode_WatchRecordWithMenuNavigation) ||
		_Modes.existskey(C_UISequenceMode_SystemUIInteraction)
	) {
		_ClientUI.UISequence = CUIConfig::EUISequence::UIInteraction;
	} else {
		_ClientUI.UISequence = CUIConfig::EUISequence::None;
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Enable or disable an UI sequence mode
Void SetUISequenceModeEnabled(CUIConfig _ClientUI, Text _RequesterId, Integer _Mode, Boolean _Enabled) {
	declare Text[][Integer] CMGame_ModesClientUI2_ActiveUISequenceModes for _ClientUI = [];
	if (_Enabled) {
		if (!CMGame_ModesClientUI2_ActiveUISequenceModes.existskey(_Mode)) {
			CMGame_ModesClientUI2_ActiveUISequenceModes[_Mode] = [];
		}
		if (!CMGame_ModesClientUI2_ActiveUISequenceModes[_Mode].exists(_RequesterId)) {
			CMGame_ModesClientUI2_ActiveUISequenceModes[_Mode].add(_RequesterId);
		}
	} else if (CMGame_ModesClientUI2_ActiveUISequenceModes.existskey(_Mode)) {
		CMGame_ModesClientUI2_ActiveUISequenceModes[_Mode].remove(_RequesterId);
		if (CMGame_ModesClientUI2_ActiveUISequenceModes[_Mode].count <= 0) {
			CMGame_ModesClientUI2_ActiveUISequenceModes.removekey(_Mode);
		}
	}

	Private_ApplyUISequenceModes(_ClientUI, CMGame_ModesClientUI2_ActiveUISequenceModes);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the given UI sequence mode is enabled
Boolean IsUISequenceModeEnabled(CUIConfig _ClientUI, Text _RequesterId, Integer _Mode) {
	declare Text[][Integer] CMGame_ModesClientUI2_ActiveUISequenceModes for _ClientUI = [];
	return (
		CMGame_ModesClientUI2_ActiveUISequenceModes.existskey(_Mode) &&
		CMGame_ModesClientUI2_ActiveUISequenceModes[_Mode].exists(_RequesterId)
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize the library
Void Initialize(CUIConfig _ClientUI) {
	declare Text[][Integer] CMGame_ModesClientUI2_ActiveUISequenceModes for _ClientUI = [];
	CMGame_ModesClientUI2_ActiveUISequenceModes = [];
	Private_ApplyUISequenceModes(_ClientUI, CMGame_ModesClientUI2_ActiveUISequenceModes);
}
