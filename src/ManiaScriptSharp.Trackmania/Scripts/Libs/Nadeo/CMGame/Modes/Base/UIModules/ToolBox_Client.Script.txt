/**
 * UI module: Toolbox (client side)
 */
#Const Version		"1.0.0"
#Const ScriptName	"Libs/Nadeo/CMGame/Modes/Base/UIModules/Toolbox_Client.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Components/Toolbox@2.Script.txt" as Toolbox
#Include "Libs/Nadeo/CMGame/Utils/Config.Script.txt" as Config
#Include "Libs/Nadeo/CMGame/Utils/ManiaView@3.Script.txt" as MV

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_ComponentName "CMGame_Modes_UIModules_Toolbox"
#Const C_ComponentClass "cmgame-modes-uimodules-toolbox"

#Const C_Id_ToolboxInstance "cmgame-mode-uimodules-toolbox_toolbox-instance"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Private functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the component template
Text Private_Template() {
	return """<CMGameToolbox2 id="{{{C_Id_ToolboxInstance}}}"/>""";
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Manialink key press event hook
Void Private_Hook_PageKeyPress(CMlScript _This, CMlScriptEvent _Event) {
	if (_This.Input != Null && _This.Input.IsKeyPressed(142)) { //< Y (keycode 142) is pressed
		if (_Event.KeyName == "F3" && Config::HasFeature(Config::C_Feature_DebugToolbox)) {
			Toolbox::OpenClose(_This);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Public functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the component
MV::K_Component Component() {
	return MV::K_Component {
		Name = C_ComponentName,
		Path = ScriptName,
		Class = C_ComponentClass,
		Template = Private_Template(),
		Hooks = [
			MV::C_Hook_PageOnEvent_KeyPress => """{{{C_ComponentName}}}::Private_Hook_PageKeyPress"""
		],
		Components = [
			Toolbox::Component()
		]
	};
}
