/**
 *	This library is deprecated and is just proxy to the new version
 *	Use `Libs/Nadeo/CMGame/Modes/ClientUI@2.Script.txt` directly instead
 *
 *	Control the ClientUI values
 *	Several scripts can try to modify the ClientUI values
 *	This can lead to races conditions difficult to debug
 *	To avoid that, any modification of a ClientUI value
 *	must go through this library
 */
#Const Version		"1.2.1"
#Const ScriptName	"Libs/Nadeo/CMGame/Modes/ClientUI.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Modes/ClientUI@2.Script.txt" as ClientUI

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constant
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const ClientUI::C_UISequenceMode_FinishRace as C_UISequenceMode_FinishRace
#Const ClientUI::C_UISequenceMode_WatchRecordWithMenuNavigation as C_UISequenceMode_WatchRecordWithMenuNavigation
#Const ClientUI::C_UISequenceMode_SystemUIInteraction as C_UISequenceMode_SystemUIInteraction

#Const C_Private_RequesterId "cmgame-modes-clientui-v1"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
Void SetUISequenceModeEnabled(Integer _Mode, Boolean _Enabled) {
	ClientUI::SetUISequenceModeEnabled(ClientUI, C_Private_RequesterId, _Mode, _Enabled);
}

Boolean IsUISequenceModeEnabled(Integer _Mode) {
	return ClientUI::IsUISequenceModeEnabled(ClientUI, C_Private_RequesterId, _Mode);
}

Void Unload() {
	ClientUI::Initialize(ClientUI);
}

Void Load() {
	Unload();
}
