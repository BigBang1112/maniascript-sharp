/// Get the control type
Text Private_GetControlType(CMlControl _Control) {
	switchtype (_Control) {
		case CMlFrame: {
			if (MVShared::IsComponent(_Control)) {
				return MVShared::GetComponentName(_Control);
			} else {
				return "frame";
			}
		}
		case CMlLabel: return "label";
		case CMlQuad: return "quad";
		case CMlArrow: return "arrow";
		case CMlCamera: return "camera";
		case CMlColorPicker: return "colorpicker";
		case CMlFileEntry: return "fileentry"; //< Must be above CMlEntry
		case CMlEntry: return "entry";
		case CMlGauge: return "gauge";
		case CMlGraph: return "graph";
		case CMlMediaPlayer: return "media"; //< Not possible to differentiate "video" from "audio"
		case CMlMinimap: return "minimap";
		case CMlSlider: return "slider";
		case CMlTextEdit: return "textedit";
		case CMlTimeLine: return "timeline";
	}

	assert(False, """[Stylesheet@3] Unknown control type""");

	return "";
}

	if (_Control.DataAttributeExists(C_DataAttribute_Class)) {

	return [];
}

/// Check if a selector match a control
Boolean Private_SelectorMatchControl(K_Selector _Selector, CMlControl _Control) {
	if (_Control == Null) return False;

	if (_Selector.Control != "" && _Selector.Control != Private_GetControlType(_Control)) {
		return False;
	}

	if (_Selector.Classes.count > 0 && !_Selector.Classes.containsonly(Private_GetControlClasses(_Control))) {
		return False;
	}

	if (_Selector.Id != "" && _Selector.Id != _Control.ControlId) {
		return False;
	}

	return True;
}
Boolean Private_SelectorsMatchControl(K_Selector[] _Selectors, CMlControl _Control) {
	if (_Control == Null || _Selectors.count <= 0) {
		return False;
	} else if (_Selectors.count == 1) {
		return Private_SelectorMatchControl(_Selectors[0], _Control);
	}

	declare CMlControl TargetControl <=> _Control;
	declare Integer SelectorKey = _Selectors.count - 1;
	declare Integer Combinator = C_Combinator_Null;
	declare Boolean IsMatching = True;

	while (_Selectors.existskey(SelectorKey)) {
		declare K_Selector Selector = _Selectors[SelectorKey];

		switch (Combinator) {
			case C_Combinator_Null, C_Combinator_Child: {
				if (TargetControl == Null) {
					return False;
				} else if (Private_SelectorMatchControl(Selector, TargetControl)) {
					if (Selector.Combinator == C_Combinator_Child || Selector.Combinator == C_Combinator_Descendant) {
						TargetControl <=> TargetControl.Parent;
					}
					Combinator = Selector.Combinator;
					SelectorKey -= 1;
					IsMatching = True;
				} else {
					return False;
				}
			}
			case C_Combinator_Descendant: {
				if (TargetControl == Null) {
					return False;
				} else if (Private_SelectorMatchControl(Selector, TargetControl)) {
					if (Selector.Combinator == C_Combinator_Child || Selector.Combinator == C_Combinator_Descendant) {
						TargetControl <=> TargetControl.Parent;
					}
					Combinator = Selector.Combinator;
					SelectorKey -= 1;
					IsMatching = True;
				} else {
					TargetControl <=> TargetControl.Parent;
					IsMatching = False;
				}
			}
			default: {
				return False;
			}
		}
	}

	return IsMatching;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Compute a selector specificity
 * Loosely based on CSS specification
 * https://www.w3.org/TR/CSS21/cascade.html#specificity
 */
K_Private_Specificity Private_ComputeSelectorSpecificity(K_Selector _Selector) {
	declare K_Private_Specificity Specificity = K_Private_Specificity { A = 0, B = 0, C = 0 };
	if (_Selector.Id != "") Specificity.A += 1;
	Specificity.B += _Selector.Classes.count;
	if (_Selector.Control != "") Specificity.C += 1;
	return Specificity;
}
K_Rule Private_ComputeSelectorSpecificity(K_Rule _Rule) {
	declare K_Rule Rule = _Rule;
	if (Rule.Selectors.count == 1) {
		Rule.Private_Specificity = Private_ComputeSelectorSpecificity(Rule.Selectors[0]);
	} else {
		Rule.Private_Specificity = K_Private_Specificity { A = 0, B = 0, C = 0 };
		for (Selector in Rule.Selectors) {
			declare K_Private_Specificity Specificity = Private_ComputeSelectorSpecificity(Selector);
			Rule.Private_Specificity.A += Specificity.A;
			Rule.Private_Specificity.B += Specificity.B;
			Rule.Private_Specificity.C += Specificity.C;
		}
	}
	return Rule;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Sort rules by their specificity
K_Rule[] Private_SortRulesBySpecificity(K_Rule[] _Rules) {
	declare K_Rule[][Integer][Integer][Integer] RulesToSort;

	for (Rule in _Rules) {
		if (RulesToSort.existskey(Rule.Private_Specificity.A)) {
			if (RulesToSort[Rule.Private_Specificity.A].existskey(Rule.Private_Specificity.B)) {
				if (RulesToSort[Rule.Private_Specificity.A][Rule.Private_Specificity.B].existskey(Rule.Private_Specificity.C)) {
					RulesToSort[Rule.Private_Specificity.A][Rule.Private_Specificity.B][Rule.Private_Specificity.C].add(Rule);
				} else {
					RulesToSort[Rule.Private_Specificity.A][Rule.Private_Specificity.B][Rule.Private_Specificity.C] = [Rule];
				}
			} else {
				RulesToSort[Rule.Private_Specificity.A][Rule.Private_Specificity.B] = [Rule.Private_Specificity.C => [Rule]];
			}
		} else {
			RulesToSort[Rule.Private_Specificity.A] = [Rule.Private_Specificity.B => [Rule.Private_Specificity.C => [Rule]]];
		}
	}

	declare K_Rule[] RulesSorted;
	for (RulesA in RulesToSort.sortkeyreverse()) {
		for (RulesB in RulesA.sortkeyreverse()) {
			for (RulesC in RulesB.sortkeyreverse()) {
				for (Rule in RulesC) {
					RulesSorted.addfirst(Rule);
				}
			}
		}
	}

	return RulesSorted;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Compile the stylesheets included in the page
Void Compile(CMlPage _Page) {
	declare K_Stylesheet CMGame_UtilsStylesheet3_CompiledStylesheet as CompiledStylesheet for _Page = C_Stylesheet_Null;
	declare Integer PreviousUpdateCounter = CompiledStylesheet.UpdateCounter;
	CompiledStylesheet = C_Stylesheet_Null;
	CompiledStylesheet.UpdateCounter = PreviousUpdateCounter + 1;

	declare K_Stylesheet[Text] CMGame_UtilsStylesheet3_IncludedStylesheets for _Page = [];
	declare Integer[Text] SelectorHashToRuleKey = [];

	// Compute specificity and merge duplicate selectors
	for (Stylesheet in CMGame_UtilsStylesheet3_IncludedStylesheets) {
		for (RuleWithoutSpecifity in Stylesheet.Rules) {
			declare K_Rule Rule = Private_ComputeSelectorSpecificity(RuleWithoutSpecifity);
			declare Text SelectorsHash = Private_GetSelectorsHash(Rule.Selectors);
			if (SelectorHashToRuleKey.existskey(SelectorsHash)) {
				declare K_Rule ExistingRule = CompiledStylesheet.Rules[SelectorHashToRuleKey[SelectorsHash]];
				for (PropertyName => PropertyValue in Rule.Properties) {
					ExistingRule.Properties[PropertyName] = PropertyValue;
				}
				CompiledStylesheet.Rules[SelectorHashToRuleKey[SelectorsHash]] = ExistingRule;
			} else {
				SelectorHashToRuleKey[SelectorsHash] = CompiledStylesheet.Rules.count;
				CompiledStylesheet.Rules.add(Rule);
			}
		}
	}

	// Sort all rules by specificity
	CompiledStylesheet.Rules = Private_SortRulesBySpecificity(CompiledStylesheet.Rules);
}
Void Compile(CMlScript _Context) {
	Compile(_Context.Page);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the properties applied to a control
Text[Text] GetControlProperties(CMlPage _Page, CMlControl _Control) {
	declare K_Stylesheet CMGame_UtilsStylesheet3_CompiledStylesheet as CompiledStylesheet for _Page = C_Stylesheet_Null;
	declare K_Private_ControlProperties CMGame_UtilsStylesheet3_ControlProperties as ControlProperties for _Control = C_ControlProperties_Null;

	if (ControlProperties.IsCached && ControlProperties.UpdateCounter == CompiledStylesheet.UpdateCounter) {
		return ControlProperties.Properties;
	}


	ControlProperties.Properties = [];
	ControlProperties.UpdateCounter = CompiledStylesheet.UpdateCounter;
	ControlProperties.IsCached = True;

	declare Text[] KeepControlProperties;
	if (_Control.DataAttributeExists(C_DataAttribute_KeepControlProperties)) {
		KeepControlProperties = TL::Split(" ", _Control.DataAttributeGet(C_DataAttribute_KeepControlProperties));
	}

	for (Rule in CompiledStylesheet.Rules) {
		if (Private_SelectorsMatchControl(Rule.Selectors, _Control)) {
			for (PropertyName => PropertyValue in Rule.Properties) {
				if (!KeepControlProperties.exists(PropertyName)) {
					ControlProperties.Properties[PropertyName] = PropertyValue;
				}
			}
		}
	}

	return ControlProperties.Properties;
}
Text[Text] GetControlProperties(CMlScript _Context, CMlControl _Control) {
	return GetControlProperties(_Context.Page, _Control);
}

/// Apply properties on a control
Void ApplyControlProperties(CMlControl _Control, Text[Text] _ControlProperties) {
	for (PropertyName => PropertyValue in _ControlProperties) {
		switch (PropertyName) {
			case C_Prop_Size: _Control.Size = Cast::ToVec2(PropertyValue);
			case C_Prop_HAlign: _Control.HorizontalAlign = Cast::ToHAlign(PropertyValue);
			case C_Prop_VAlign: _Control.VerticalAlign = Cast::ToVAlign(PropertyValue);
			case C_Prop_Hidden: _Control.Visible = !Cast::ToBoolean(PropertyValue);
			case C_Prop_Pos: _Control.RelativePosition_V3 = Cast::ToVec2(PropertyValue);
			case C_Prop_ZIndex: _Control.ZIndex = TL::ToReal(PropertyValue);
			case C_Prop_Scale: _Control.RelativeScale = TL::ToReal(PropertyValue);
			case C_Prop_Rotation: _Control.RelativeRotation = TL::ToReal(PropertyValue);
			default: {
				switchtype (_Control as TypedControl) {
					case CMlLabel: {
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_Style: TypedControl.Style = PropertyValue;
							case C_Prop_Substyle: TypedControl.Substyle = PropertyValue;
							case C_Prop_Blend: TypedControl.Blend = Cast::ToLabelBlendMode(PropertyValue);
							case C_Prop_TTSAltText: TypedControl.TTS_AltText = PropertyValue;
							case C_Prop_TTSAltTextTranslate: TypedControl.TTS_AltText_Translate = Cast::ToBoolean(PropertyValue);
							case C_Prop_TextFont: TypedControl.TextFont = PropertyValue;
							case C_Prop_TextValue: TypedControl.Value = PropertyValue;
							case C_Prop_TextMaxLine: TypedControl.MaxLine = TL::ToInteger(PropertyValue);
							case C_Prop_TextLineSpacing: TypedControl.LineSpacing = TL::ToReal(PropertyValue);
							case C_Prop_TextItalicSlope: TypedControl.ItalicSlope = TL::ToReal(PropertyValue);
							case C_Prop_TextAppendEllipsis: TypedControl.AppendEllipsis = Cast::ToBoolean(PropertyValue);
							case C_Prop_TextAutoNewLine: TypedControl.AutoNewLine = Cast::ToBoolean(PropertyValue);
							case C_Prop_TextPrefix: TypedControl.TextPrefix = PropertyValue;
							case C_Prop_TextColor: TypedControl.TextColor = CL::HexToRgb(PropertyValue);
							case C_Prop_TextSize: TypedControl.TextSizeReal = TL::ToReal(PropertyValue);
							case C_Prop_TextSizeFit: MLHelpers::FitLabelValue(TypedControl, Cast::ToVec3(PropertyValue));
							case C_Prop_TextFilterProfanities: TypedControl.FilterProfanities = Cast::ToFilterProfanities(PropertyValue);
						}
					}
					case CMlQuad: {
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_Style: TypedControl.Style = PropertyValue;
							case C_Prop_Substyle: TypedControl.Substyle = PropertyValue;
							case C_Prop_Blend: TypedControl.Blend = Cast::ToQuadBlendMode(PropertyValue);
							case C_Prop_TTSAltText: TypedControl.TTS_AltText = PropertyValue;
							case C_Prop_TTSAltTextTranslate: TypedControl.TTS_AltText_Translate = Cast::ToBoolean(PropertyValue);
							case C_Prop_QuadImageUrl: TypedControl.ImageUrl = PropertyValue;
							case C_Prop_QuadImageUrlFocus: TypedControl.ImageUrlFocus = PropertyValue;
							case C_Prop_QuadAlphaMaskUrl: TypedControl.AlphaMaskUrl = PropertyValue;
							case C_Prop_QuadSuperSample: TypedControl.SuperSample = Cast::ToBoolean(PropertyValue);
							case C_Prop_QuadColorize: TypedControl.Colorize = CL::HexToRgb(PropertyValue);
							case C_Prop_QuadModulateColor: TypedControl.ModulateColor = CL::HexToRgb(PropertyValue);
							case C_Prop_QuadBgColor: TypedControl.BgColor = CL::HexToRgb(PropertyValue);
							case C_Prop_QuadBgColorFocus: TypedControl.BgColorFocus = CL::HexToRgb(PropertyValue);
							case C_Prop_QuadKeepRatio: TypedControl.KeepRatio = Cast::ToKeepRatioMode(PropertyValue);
						}
					}
					case CMlFileEntry: { //< Must be above CMlEntry
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_TextValue: TypedControl.Value = PropertyValue;
							case C_Prop_TextFormat: TypedControl.TextFormat = Cast::ToEntryFormat(PropertyValue);
							case C_Prop_TextColor: TypedControl.TextColor = CL::HexToRgb(PropertyValue);
							case C_Prop_TextSize: TypedControl.TextSizeReal = TL::ToReal(PropertyValue);
							case C_Prop_TextMaxLength: TypedControl.MaxLength = TL::ToInteger(PropertyValue);
						}
					}
					case CMlEntry: {
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_TextValue: TypedControl.Value = PropertyValue;
							case C_Prop_TextFormat: TypedControl.TextFormat = Cast::ToEntryFormat(PropertyValue);
							case C_Prop_TextColor: TypedControl.TextColor = CL::HexToRgb(PropertyValue);
							case C_Prop_TextSize: TypedControl.TextSizeReal = TL::ToReal(PropertyValue);
							case C_Prop_TextMaxLength: TypedControl.MaxLength = TL::ToInteger(PropertyValue);
						}
					}
					case CMlTextEdit: {
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_TextValue: TypedControl.Value = PropertyValue;
							case C_Prop_TextMaxLine: TypedControl.MaxLine = TL::ToInteger(PropertyValue);
							case C_Prop_TextAutoNewLine: TypedControl.AutoNewLine = Cast::ToBoolean(PropertyValue);
							case C_Prop_TextLineSpacing: TypedControl.LineSpacing = TL::ToReal(PropertyValue);
							case C_Prop_TextColor: TypedControl.TextColor = CL::HexToRgb(PropertyValue);
							case C_Prop_TextSize: TypedControl.TextSizeReal = TL::ToReal(PropertyValue);
							case C_Prop_TextFormat: TypedControl.TextFormat = Cast::ToTextEditFormat(PropertyValue);
						}
					}
				}
			}
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add a stylesheet class to the control
Void AddControlClasses(CMlPage _Page, CMlControl _Control, Text[] _Classes) {
	declare Text[] ControlClasses = Private_GetControlClasses(_Control);
	declare Boolean Modified = False;
	for (Class in _Classes) {
		if (!ControlClasses.exists(Class)) {
			ControlClasses.add(Class);
			Modified = True;
		}
	}
	if (Modified) {
		declare K_Private_ControlProperties CMGame_UtilsStylesheet3_ControlProperties as ControlProperties for _Control = C_ControlProperties_Null;
		ControlProperties.IsCached = False;
		Private_SetControlClasses(_Control, ControlClasses);
		ApplyControlProperties(_Page, _Control);
	}
}
Void AddControlClasses(CMlScript _Context, CMlControl _Control, Text[] _Classes) {
	AddControlClasses(_Context.Page, _Control, _Classes);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Remove a style class from the control
Void RemoveControlClasses(CMlPage _Page, CMlControl _Control, Text[] _Classes) {
	declare Text[] ControlClasses = Private_GetControlClasses(_Control);
	declare Boolean Modified = False;
	for (Class in _Classes) {
		if (ControlClasses.remove(Class)) {
			Modified = True;
		}
	}
	if (Modified) {
		declare K_Private_ControlProperties CMGame_UtilsStylesheet3_ControlProperties as ControlProperties for _Control = C_ControlProperties_Null;
		ControlProperties.IsCached = False;
		Private_SetControlClasses(_Control, ControlClasses);
		ApplyControlProperties(_Page, _Control);
	}
}
Void RemoveControlClasses(CMlScript _Context, CMlControl _Control, Text[] _Classes) {
	RemoveControlClasses(_Context.Page, _Control, _Classes);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
