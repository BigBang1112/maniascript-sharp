/**
 * Common interface for asynchronous operation
 */
#Const Version "1.0.0"
#Const ScriptName "Libs/Nadeo/CMGame/Utils/Promise.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Utils/Error.Script.txt" as Error

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_Promise {
	Integer Status;
	Text Value;
	Error::K_Error Error;
	Text[Text] Metadata;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Status_Null 0
#Const C_Status_Pending 1
#Const C_Status_Fulfilled 2
#Const C_Status_Rejected 3

#Const C_DefaultResolveValue ""

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Create a new promise
K_Promise Create() {
	return K_Promise {
		Status = C_Status_Pending,
		Value = "",
		Error = Error::GetDefaultError(),
		Metadata = []
	};
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Fulfill a promise
K_Promise Resolve(K_Promise _Promise, Text _Value) {
	declare K_Promise Promise = _Promise;
	Promise.Status = C_Status_Fulfilled;
	Promise.Value = _Value;
	return Promise;
}
K_Promise Resolve(K_Promise _Promise) {
	return Resolve(_Promise, C_DefaultResolveValue);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Reject a promise
K_Promise Reject(K_Promise _Promise, Error::K_Error _Error) {
	declare K_Promise Promise = _Promise;
	Promise.Status = C_Status_Rejected;
	Promise.Error = _Error;
	return Promise;
}
K_Promise Reject(K_Promise _Promise) {
	return Reject(_Promise, Error::GetError(Error::C_Type_ManiaScript, Error::C_Code_ManiaScript_PromiseRejectedWithoutError));
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the promise is pending
Boolean IsPending(K_Promise _Promise) {
	return _Promise.Status == C_Status_Pending;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if all promises are pending
Boolean AreAllPending(K_Promise[] _Promises) {
	if (_Promises.count <= 0) return False;
	for (Promise in _Promises) {
		if (!IsPending(Promise)) return False;
	}
	return True;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the promise is settled (fulfilled or rejected)
Boolean IsSettled(K_Promise _Promise) {
	return (
		_Promise.Status == C_Status_Fulfilled ||
		_Promise.Status == C_Status_Rejected
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if all promises are settled (fulfilled or rejected)
Boolean AreAllSettled(K_Promise[] _Promises) {
	if (_Promises.count <= 0) return True;
	for (Promise in _Promises) {
		if (!IsSettled(Promise)) return False;
	}
	return True;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the promise is fulfilled
Boolean IsFulfilled(K_Promise _Promise) {
	return _Promise.Status == C_Status_Fulfilled;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if all promises are fulfilled
Boolean AreAllFulfilled(K_Promise[] _Promises) {
	if (_Promises.count <= 0) return False;
	for (Promise in _Promises) {
		if (!IsFulfilled(Promise)) return False;
	}
	return True;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the promise is rejected
Boolean IsRejected(K_Promise _Promise) {
	return _Promise.Status == C_Status_Rejected;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if all promises are rejected
Boolean AreAllRejected(K_Promise[] _Promises) {
	if (_Promises.count <= 0) return False;
	for (Promise in _Promises) {
		if (!IsRejected(Promise)) return False;
	}
	return True;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the resolved value
Text GetValue(K_Promise _Promise) {
	return _Promise.Value;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the resolved values of several promises
Text[] GetValues(K_Promise[] _Promises) {
	declare Text[] Values;
	for (Promise in _Promises) {
		Values.add(GetValue(Promise));
	}
	return Values;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the rejected error
Error::K_Error GetError(K_Promise _Promise) {
	return _Promise.Error;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the rejected errors of several promises
Error::K_Error[] GetErrors(K_Promise[] _Promises) {
	declare Error::K_Error[] Errors;
	for (Promise in _Promises) {
		Errors.add(GetError(Promise));
	}
	return Errors;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add metadata on a promise
K_Promise SetMetadata(K_Promise _Promise, Text[Text] _Metadata) {
	declare K_Promise Promise = _Promise;
	foreach (Key => Value in _Metadata) {
		Promise.Metadata[Key] = Value;
	}
	return Promise;
}
K_Promise SetMetadata(K_Promise _Promise, Text _Key, Text _Value) {
	declare K_Promise Promise = _Promise;
	Promise.Metadata[_Key] = _Value;
	return Promise;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Remove all metadata from a promise
K_Promise ResetMetadata(K_Promise _Promise) {
	declare K_Promise Promise = _Promise;
	Promise.Metadata = [];
	return Promise;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get metadata from a promise
Text[Text] GetMetadata(K_Promise _Promise) {
	return _Promise.Metadata;
}
Text GetMetadata(K_Promise _Promise, Text _Key, Text _DefaultValue) {
	return _Promise.Metadata.get(_Key, _DefaultValue);
}
Text GetMetadata(K_Promise _Promise, Text _Key) {
	return GetMetadata(_Promise, _Key, "");
}
