/// Check if the text is translated
Boolean IsTranslatedText(Text _Text) {
	// The compose character also act as a translate character
	return TL::StartsWith(C_TranslateCharacter, _Text) || TL::StartsWith(C_ComposeCharacter, _Text);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** If `InputPlayer` is not spawned and they are not spectating anyone (cam7 or no other player is spawned)
 * then `GUIPlayer` will be `Null`. In this case we want to get the `InputPlayer` instead of `Null`.
 */
CSmPlayer GetTargetPlayer(CSmMlScriptIngame _Context) {
	if (_Context.GUIPlayer != Null) return _Context.GUIPlayer;
	return _Context.InputPlayer;
}
