// MARK: Boolean

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Cast a Text to a Boolean
Boolean ToBoolean(Text _Source) {
	declare Text Value = TL::ToLowerCase(_Source);
	return (
		Value != "0" &&
		Value != "-0" &&
		Value != "null" &&
		Value != "nullid" &&
		Value != "false" &&
		Value != ""
	);
}

}

// MARK: Vec2

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Cast a Text to a Vec2
Vec2 ToVec2(Text _Source) {
	declare Text[] Values = TL::Split(" ", TL::Trim(_Source));

	declare Vec2 Value = <0., 0.>;
	if (Values.existskey(0)) {
		Value.X = TL::ToReal(Values[0]);
	}
	if (Values.existskey(1)) {
		Value.Y = TL::ToReal(Values[1]);
	}

	return Value;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Cast a Vec2 to a Text
Text ToText(Vec2 _Source) {
	return _Source.X^" "^_Source.Y;
}

// MARK: Real[]

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Cast a Text to a Real[]
Real[] ToRealArray(Text _Source) {
	declare Real[] Array;
	declare Text[] Values = TL::Split(" ", TL::Trim(_Source));

	foreach (Value in Values) {
		Array.add(TL::ToReal(Value));
	}

	return Array;
}

// MARK: TextSizeFit

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Convert a text-size-fit Text attribute to a Vec3 compatible with the FitLabelValue() function
Vec3 ToTextSizeFitVec3(Text _Source) {
	declare Text[] Values = TL::Split(" ", TL::Trim(_Source));

	declare Vec3 TextSizeFit = <0., 0., 0.>;

	if (Values.existskey(0)) {
		TextSizeFit.X = TL::ToReal(Values[0]);
	}
	if (Values.existskey(1)) {
		TextSizeFit.Y = TL::ToReal(Values[1]);
	} else {
		TextSizeFit.Y = TextSizeFit.X;
	}
	if (Values.existskey(2)) {
		TextSizeFit.Z = TL::ToReal(Values[2]);
	}

	return TextSizeFit;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Convert a text-size-fit Text attribute to a Vec2 compatible with the FitLabelValue() function
Vec2 ToTextSizeFitVec2(Text _Source) {
	declare Text[] Values = TL::Split(" ", TL::Trim(_Source));

	declare Vec2 TextSizeFit = <0., 0.>;

	if (Values.existskey(0)) {
		TextSizeFit.X = TL::ToReal(Values[0]);
	}
	if (Values.existskey(1)) {
		TextSizeFit.Y = TL::ToReal(Values[1]);
	} else {
		TextSizeFit.Y = TextSizeFit.X;
	}

	return TextSizeFit;
}
