/**
 *	Http authorization header
 */
#Const Version		"1.0.2"
#Const ScriptName	"Libs/Nadeo/CMGame/Utils/HttpServerAuthorizationHeader.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constant
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Token_Null ""

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add the Live authorization header to headers
Text[Text] GetHeadersWithLiveAuthorization(Text[Text] _Headers) {
	declare Text[Text] Headers = _Headers;
	declare Text LibCommonHttpServerAuthorizationHeader_LiveToken for This = C_Token_Null;
	Headers["Authorization"] = "nadeo_v1 t="^LibCommonHttpServerAuthorizationHeader_LiveToken;
	return Headers;
}
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the Live services token
Void SetLiveToken(Text _LiveToken) {
	declare Text LibCommonHttpServerAuthorizationHeader_LiveToken for This = C_Token_Null;
	LibCommonHttpServerAuthorizationHeader_LiveToken = _LiveToken;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add the Club authorization header to headers
Text[Text] GetHeadersWithClubAuthorization(Text[Text] _Headers) {
	log("WARNING [HttpServerAuthorizationHeader] Club tokens are deprecated. Use Live tokens instead.");
	return GetHeadersWithLiveAuthorization(_Headers);
}
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the Club services token
// @mslint-disable-next-line no-unused-variables -- For backward compatibility reasons we cannot remove the deprecated `_ClubToken` parameter
Void SetClubToken(Text _ClubToken) {
	log("WARNING [HttpServerAuthorizationHeader] Club tokens are deprecated. Use Live tokens instead.");
}
