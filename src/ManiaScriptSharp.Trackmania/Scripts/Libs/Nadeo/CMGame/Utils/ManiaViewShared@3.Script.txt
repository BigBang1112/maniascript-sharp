#Const Version "3.0.0"
#Const ScriptName "Libs/Nadeo/CMGame/Utils/ManiaViewShared@3.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Class_ComponentInstance "cmgame-mv3-component-instance"
#Const C_Attribute_ComponentName "cmgame-mv3-component-name"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if a control is a component
Boolean IsComponent(CMlControl _Control) {
	return _Control.HasClass(C_Class_ComponentInstance);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get a component name
Text GetComponentName(CMlControl _Control) {
	return _Control.DataAttributeGet(C_Attribute_ComponentName);
}
