// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the component instance to which the element belongs
CMlFrame GetComponentInstanceThatOwnsControl(CMlControl _Control) {
	if (_Control == Null) return Null;

	// Check if the owner is already cached
	declare Boolean CMGame_MV3_ComponentInstanceOwnerIsCached for _Control = False;
	declare CMlFrame CMGame_MV3_ComponentInstanceOwnerCache for _Control = Null;
	if (CMGame_MV3_ComponentInstanceOwnerIsCached) {
		return CMGame_MV3_ComponentInstanceOwnerCache;
	}

	// Find the owner
	declare CMlFrame Control = _Control.Parent;
	declare CMlFrame Owner = Null;
	while (Control != Null && Owner == Null) {
		if (IsComponent(Control)) {
			Owner = Control;
		} else {
			Control = Control.Parent;
		}
	}

	// Save owner in cache
	CMGame_MV3_ComponentInstanceOwnerCache = Owner;
	CMGame_MV3_ComponentInstanceOwnerIsCached = True;

	return Owner;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Turn on/off a component instance hook
Void EnableHook(CMlFrame _ComponentInstance, Text _HookName, Boolean _Enabled) {
	if (!_ComponentInstance.DataAttributeExists(C_Attribute_ComponentHooks)) return;
	declare Text[] ComponentHooks = TL::Split(";", _ComponentInstance.DataAttributeGet(C_Attribute_ComponentHooks));
	if (!ComponentHooks.exists(_HookName)) return;

	declare Text ComponentName = GetComponentName(_ComponentInstance);

	if (_Enabled) {
		if (G_ComponentHooksComponentInstances.existskey(_HookName)) {
			if (G_ComponentHooksComponentInstances[_HookName].existskey(ComponentName)) {
				if (!G_ComponentHooksComponentInstances[_HookName][ComponentName].exists(_ComponentInstance)) {
					G_ComponentHooksComponentInstances[_HookName][ComponentName].add(_ComponentInstance);
				}
			} else {
				G_ComponentHooksComponentInstances[_HookName][ComponentName] = [_ComponentInstance];
			}
		} else {
			G_ComponentHooksComponentInstances[_HookName] = [ComponentName => [_ComponentInstance]];
		}
	} else if (
		G_ComponentHooksComponentInstances.existskey(_HookName) &&
		G_ComponentHooksComponentInstances[_HookName].existskey(ComponentName)
	) {
		G_ComponentHooksComponentInstances[_HookName][ComponentName].remove(_ComponentInstance);
	}
}
Void EnableHook(CMlFrame _ComponentInstance, Text _HookName) {
	EnableHook(_ComponentInstance, _HookName, True);
}
Void DisableHook(CMlFrame _ComponentInstance, Text _HookName) {
	EnableHook(_ComponentInstance, _HookName, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Reset all component instances with a hook
Void Private_ResetComponentHooksComponentInstances() {
	G_ComponentHooksComponentInstances = [];
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add a hook to a component instance
Void Private_AddComponentInstanceHook(CMlFrame _ComponentInstance, Text _HookName, Text _ComponentName) {
	if (G_ComponentHooksComponentInstances.existskey(_HookName)) {
		if (G_ComponentHooksComponentInstances[_HookName].existskey(_ComponentName)) {
			G_ComponentHooksComponentInstances[_HookName][_ComponentName].add(_ComponentInstance);
		} else {
			G_ComponentHooksComponentInstances[_HookName][_ComponentName] = [_ComponentInstance];
		}
	} else {
		G_ComponentHooksComponentInstances[_HookName] = [_ComponentName => [_ComponentInstance]];
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get all component instances with the given hook
CMlFrame[] Private_GetComponentInstancesWithHook(Text _HookName, Text _ComponentName) {
	return G_ComponentHooksComponentInstances.get(_HookName, []).get(_ComponentName, []);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Mark a component instance to be redrawn at the end of the frame
Void MarkForRedraw(CMlFrame _ComponentInstance, Text[] _Tags) {
	if (_ComponentInstance == Null) return;
	declare K_Redraw Redraw = G_ComponentInstancesToRedraw.get(_ComponentInstance.Id, K_Redraw { Component = _ComponentInstance });
	for (Tag in _Tags) {
		if (Tag != "" && !Redraw.Tags.exists(Tag)) {
			Redraw.Tags.add(Tag);
		}
	}
	G_ComponentInstancesToRedraw[_ComponentInstance.Id] = Redraw;
}
Void MarkForRedraw(CMlFrame _ComponentInstance, Text _Tag) {
	MarkForRedraw(_ComponentInstance, [_Tag]);
}
Void MarkForRedraw(CMlFrame _ComponentInstance) {
	MarkForRedraw(_ComponentInstance, "");
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the given component instance is marked for redraw
Boolean IsMarkedForRedraw(CMlFrame _ComponentInstance) {
	return (
		_ComponentInstance != Null &&
		G_ComponentInstancesToRedraw.existskey(_ComponentInstance.Id)
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Get the redraw tags from a component and remove it from the queue
 * This allow the user to manually redraw a component
 */
Text[] DoManualRedraw(CMlFrame _ComponentInstance) {
	if (_ComponentInstance == Null || !G_ComponentInstancesToRedraw.existskey(_ComponentInstance.Id)) return [];
	declare Text[] Tags = G_ComponentInstancesToRedraw[_ComponentInstance.Id].Tags;
	G_ComponentInstancesToRedraw.removekey(_ComponentInstance.Id);
	return Tags;
}

/// Check if one of the given redraw tag is present
Boolean HasRedrawTag(Text[] _TagsInWhichToSearch, Text _InitTag, Text[] _TagsToFind) {
	return (
		_TagsInWhichToSearch.exists(_InitTag) ||
		_TagsInWhichToSearch.containsoneof(_TagsToFind)
	);
}
Boolean HasRedrawTag(Text[] _TagsInWhichToSearch, Text _InitTag, Text _TagToFind) {
	return HasRedrawTag(_TagsInWhichToSearch, _InitTag, [_TagToFind]);
}
Boolean HasRedrawTag(Text[] _TagsInWhichToSearch, Text[] _TagsToFind) {
	return _TagsInWhichToSearch.containsoneof(_TagsToFind);
}
Boolean HasRedrawTag(Text[] _TagsInWhichToSearch, Text _TagToFind) {
	return HasRedrawTag(_TagsInWhichToSearch, [_TagToFind]);
}

/// Find all components in a manialink xml
Text[] Private_FindComponents(Text _ComponentName, Text _Manialink) {
	return TL::RegexFind(
		"""<\s*{{{_ComponentName}}}\b\s*(?:{{{C_Regex_TagWord}}}\s*=\s*".*?"\s*)*\/?>""",
		_Manialink,
		"g"
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Regex match a component in a manialink xml and capture its attributes
K_MatchedComponent Private_MatchComponent(Text _ComponentName, Text _Manialink) {
	declare Text[] Matches = TL::RegexMatch(
		"""<\s*{{{_ComponentName}}}\b\s*((?:{{{C_Regex_TagWord}}}\s*=\s*".*?"\s*)*)\/?>""",
		_Manialink,
		""
	);

	declare K_MatchedComponent MatchedComponent = K_MatchedComponent {
		Matched = Matches.count > 0,
		Name = _ComponentName
	};

	if (Matches.count > 0) MatchedComponent.Xml = Matches[0];
	if (Matches.count > 1) MatchedComponent.Attributes = Matches[1];

	return MatchedComponent;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Parse the attributes of an xml node
Text[Text] Private_ParseAttributes(Text _RawAttributes) {
	if (TL::Trim(_RawAttributes) == "") return [];

	declare Text[Text] ParsedAttributes;
	declare Text[] RawAttributes = TL::RegexFind(TL::Trim(""" {{{C_Regex_TagWord}}}\s*=\s*".*?" """), _RawAttributes, "g");
	declare Text[] DuplicatedAttributes;
	for (RawAttribrute in RawAttributes) {
		declare Text[] Matches = TL::RegexMatch(TL::Trim(""" ({{{C_Regex_TagWord}}})\s*=\s*"(.*?)" """), RawAttribrute, "");
		if (Matches.count >= 3) {
			declare Text AttributeName = Matches[1];
			for (DirectiveShortcut => DirectiveFull in C_DefaultDirectives) {
				if (TL::StartsWith(DirectiveShortcut, AttributeName)) {
					AttributeName = DirectiveFull ^ TL::SubText(AttributeName, TL::Length(DirectiveShortcut), TL::Length(AttributeName));
				}
			}
			if (ParsedAttributes.existskey(AttributeName)) {
				DuplicatedAttributes.add(Matches[1]);
			}
			ParsedAttributes[AttributeName] = Matches[2];
		}
	}

	assert(
		DuplicatedAttributes.count <= 0,
		"""[ManiaView@3] Attribute `{{{TL::Join("`, `", DuplicatedAttributes)}}}` is duplicated on control with id `{{{ParsedAttributes.get("id", "")}}}`"""
	);

	return ParsedAttributes;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Convert an attributes array to a node attributes string
Text Private_OutputAttributes(Text[Text] _Attributes) {
	declare Text Attributes;

	for (Name => Value in _Attributes) {
		Attributes ^= """{{{Name}}}="{{{Value}}}" """;
	}

	return TL::Trim(Attributes);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Find nodes with mixins
 *	When `_SlowFind` is disabled, only a fast check is done on each node
 *	This fast check can return false positive but they will be filtered out later anyway
 *	When `_SlowFind` is enabled, an accurate check is done but it will fail on longer nodes
 *	because of the backtracking stack limit in the C++ regex engine
 */
Text[] Private_FindMixins(Text _Manialink, Boolean _SlowFind) {
	declare Text[] Result;

	declare Text[] Nodes = TL::RegexFind("""<\s*{{{C_Regex_TagWord}}}\b\s*(?:{{{C_Regex_TagWord}}}\s*=\s*".*?"\s*)*\/?>""", _Manialink, "g");
	for (Node in Nodes) {
		declare Boolean FastHasMixin = TL::RegexFind("""\$\$|@@|mv-mixin-\w+:""", Node, "").count > 0;
		if (FastHasMixin) {
			if (_SlowFind) {
				declare Boolean HasMixin = TL::RegexFind(
					"""<\s*{{{C_Regex_TagWord}}}(?:[^>\/]|\/(?!>))*(?:\$\$|@@|mv-mixin-\w+:){{{C_Regex_TagWord}}}="(?:[^"\\]|\\["\\nrt])*"(?:[^>\/]|\/(?!>))*\/?>""",
					Node,
					"g"
				).count > 0;
				if (HasMixin) {
					Result.add(Node);
				}
			} else {
				Result.add(Node);
			}
		}
	}
	return Result;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Regex match a mixin in a manialink xml
K_MatchedMixin Private_MatchMixin(Text _Manialink) {
	declare Text[] Matches = TL::RegexMatch(
		"""(<\s*{{{C_Regex_TagWord}}}\b\s*)((?:{{{C_Regex_TagWord}}}\s*=\s*".*?"\s*)*)(\/?>)""",
		_Manialink,
		""
	);

	declare K_MatchedMixin MatchedMixin = K_MatchedMixin {
		Matched = Matches.count >= 4,
		SlotStart = "",
		SlotEnd = ""
	};

	if (MatchedMixin.Matched) {
		MatchedMixin.Xml = Matches[0];
		MatchedMixin.Prepend = TL::Trim(Matches[1]);
		MatchedMixin.Attributes = TL::Trim(Matches[2]);
		MatchedMixin.Append = TL::Trim(Matches[3]);
	}

	return MatchedMixin;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Sort the tree of components from root to leafs
K_ManialinkPageCompilation Private_CompileStep_SortComponents(K_ManialinkPageCompilation _Compilation) {
	declare K_ManialinkPageCompilation Compilation = _Compilation;

	declare K_Component[] Components = [Compilation.PageComponent];
	Compilation.SortedComponents = [];
	while (Components.count > 0) {
		declare K_Component Component = Components[0];
		assert(Component.Name != "", """[ManiaView@3] Component with path `{{{Component.Path}}}` does not have a name""");
		assert(Component.Class != "", """[ManiaView@3] Component with path `{{{Component.Path}}}` does not have a class""");

		declare Text[] EventNames;
		for (EventName => Event in Component.DefineEvents) {
			assert(!EventNames.exists(EventName), """[ManiaView@3] Event `{{{EventName}}}` in component `{{{Component.Name}}}` already defined""");
			assert(Event.Get != "", """[ManiaView@3] Event `{{{EventName}}}` of component `{{{Component.Name}}}` is missing a `Get` function in `DefineEvents`""");
			assert(Event.Reset != "", """[ManiaView@3] Event `{{{EventName}}}` of component `{{{Component.Name}}}` is missing a `Reset` function in `DefineEvents`""");
			EventNames.add(EventName);
		}

		if (Compilation.SortedComponents.existskey(Component.Name)) {
			assert(
				Compilation.SortedComponents[Component.Name].Path == Component.Path,
				"""[ManiaView@3] The `{{{Component.Name}}}` component already exists with a different path : `{{{Compilation.SortedComponents[Component.Name].Path}}}` != `{{{Component.Path}}}`"""
			);
			assert(
				Compilation.SortedComponents[Component.Name].Class == Component.Class,
				"""[ManiaView@3] The `{{{Component.Name}}}` component already exists with a different class : `{{{Compilation.SortedComponents[Component.Name].Class}}}` != `{{{Component.Class}}}`"""
			);

			// If a component uses another component that is already loaded
			// remove it from the list and add it at the end
			// This component must be concataneted before the one that uses it
			Compilation.SortedComponents.removekey(Component.Name);
		}
		Compilation.SortedComponents[Component.Name] = Component;
		Components.add(Component.Components);
		Components.removekey(0);
	}

	declare Integer Priority = 0;
	Compilation.ComponentsPriorities = [];
	for (Component in reverse Compilation.SortedComponents) {
		Compilation.ComponentsPriorities[Component.Name] = Priority;
		Priority += 1;
	}

	return Compilation;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Sort the used mixins
K_ManialinkPageCompilation Private_CompileStep_SortMixins(K_ManialinkPageCompilation _Compilation) {
	declare K_ManialinkPageCompilation Compilation = _Compilation;

	Compilation.SortedMixins = [];
	Compilation.MixinsProps = [];
	Compilation.MixinsEvents = [];
	for (Component in Compilation.SortedComponents) {
		for (ComponentMixin in Component.Mixins) {
			// Mixin already exists, check if there is a conflict
			if (Compilation.SortedMixins.existskey(ComponentMixin.Name)) {
				declare K_Mixin SortedMixin = Compilation.SortedMixins[ComponentMixin.Name];
				assert(
					SortedMixin.Path == ComponentMixin.Path,
					"""[ManiaView@3] The `{{{ComponentMixin.Name}}}` mixin already exists with a different path : `{{{ComponentMixin.Path}}}` != `{{{SortedMixin.Path}}}`"""
				);
				assert(
					SortedMixin.Class == ComponentMixin.Class,
					"""[ManiaView@3] The `{{{ComponentMixin.Name}}}` mixin already exists with a different class : `{{{ComponentMixin.Class}}}` != `{{{SortedMixin.Class}}}`"""
				);
			}
			// Add new mixin
			else {
				assert(ComponentMixin.Name != "", """[ManiaView@3] Mixin with path `{{{ComponentMixin.Path}}}` does not have a name""");
				assert(ComponentMixin.Class != "", """[ManiaView@3] Mixin with path `{{{ComponentMixin.Path}}}` does not have a class""");
				Compilation.SortedMixins[ComponentMixin.Name] = ComponentMixin;
				for (Prop in ComponentMixin.DefineProps) {
					assert(!Compilation.MixinsProps.existskey(Prop), """[ManiaView@3] Prop `{{{Prop}}}` in mixin `{{{ComponentMixin.Name}}}` already defined in mixin `{{{Compilation.MixinsProps[Prop]}}}`""");
					Compilation.MixinsProps[Prop] = ComponentMixin.Name;
				}
				for (EventName => Event in ComponentMixin.DefineEvents) {
					assert(!Compilation.MixinsEvents.existskey(EventName), """[ManiaView@3] Event `{{{EventName}}}` in mixin `{{{ComponentMixin.Name}}}` already defined in mixin `{{{Compilation.MixinsEvents[EventName]}}}`""");
					assert(Event.Get != "", """[ManiaView@3] Event `{{{EventName}}}` of mixin `{{{ComponentMixin.Name}}}` is missing a `Get` function in `DefineEvents`""");
					assert(Event.Reset != "", """[ManiaView@3] Event `{{{EventName}}}` of mixin `{{{ComponentMixin.Name}}}` is missing a `Reset` function in `DefineEvents`""");
					Compilation.MixinsEvents[EventName] = ComponentMixin.Name;
				}
			}
		}
	}

	return Compilation;
}

/// Bind user functions to their requested component hooks
K_ManialinkPageCompilation Private_CompileStep_BindComponentsHooks(K_ManialinkPageCompilation _Compilation) {
	declare K_ManialinkPageCompilation Compilation = _Compilation;

	declare Text[Text] ML_Hooks_ComponentsMouseEvents = [];
	Compilation.ML_Hooks_Components = [];
	for (Component in reverse Compilation.SortedComponents) {
		// Bind functions to hooks
		declare Text ComponentInitFunctionCalls = "";
		declare Text ComponentDrawFunctionCalls = "";
		declare Text[Text] ComponentMouseEventsFunctionCalls = [];
		declare Text[Text] ComponentFunctionCalls = [];
		for (HookName => HookFunction in Component.Hooks) {
			if (Compilation.DefinedComponentHooks.existskey(HookName)) {
				declare Text HookFunctionCall = Private_CallBoundFunction(HookFunction, Compilation.DefinedComponentHooks[HookName]);
				if (HookName == C_Hook_ComponentInit) {
					ComponentInitFunctionCalls ^= """{{{HookFunctionCall}}};""";
				} else if (HookName == C_Hook_ComponentDraw) {
					ComponentDrawFunctionCalls ^= """{{{HookFunctionCall}}};""";
				} else if (
					HookName == C_Hook_ComponentOnEvent_TriggerZoneMouseClick ||
					HookName == C_Hook_ComponentOnEvent_TriggerZoneMouseRightClick ||
					HookName == C_Hook_ComponentOnEvent_TriggerZoneMouseOver ||
					HookName == C_Hook_ComponentOnEvent_TriggerZoneMouseOut
				) {
					ComponentMouseEventsFunctionCalls[HookName] = ComponentMouseEventsFunctionCalls.get(HookName, "") ^ """{{{HookFunctionCall}}};""";
				} else if (
					HookName == C_Hook_ComponentMainStart ||
					HookName == C_Hook_ComponentMainEnd ||
					HookName == C_Hook_ComponentBeforeYield ||
					HookName == C_Hook_ComponentAfterYield
				) {
					ComponentFunctionCalls[HookName] = ComponentFunctionCalls.get(HookName, "") ^ """{{{HookFunctionCall}}};""";
				} else {
					declare Text DebugTuningmarkOpen = "";
					declare Text DebugTuningmarkClose = "";
					if (C_Debug_EnableTuningMark) {
						DebugTuningmarkOpen = """tuningmark("+{{{Component.Name}}}@{{{HookName}}}");""";
						DebugTuningmarkClose = """tuningmark("-{{{Component.Name}}}@{{{HookName}}}");""";
					}

					Compilation.ML_Hooks ^=
"""
***{{{HookName}}}***
***
{{{DebugTuningmarkOpen}}}
{{{HookFunctionCall}}};
{{{DebugTuningmarkClose}}}
***
""";
				}
			}
		}

		// Bind functions to the component init hook
		if (ComponentInitFunctionCalls != "") {
			declare Text Test = "if";
			if (Compilation.ML_Hooks_ComponentsInit != "") Test = "else if";
			Compilation.ML_Hooks_ComponentsInit ^=
"""
			{{{Test}}} (ComponentInstance != Null && ComponentInstance.HasClass({{{dump(Component.Class)}}})) {
				{{{ComponentInitFunctionCalls}}}
			}
""";
		}

		// Bind functions to the component draw hook
		if (ComponentDrawFunctionCalls != "") {
			declare Text Test = "if";
			if (Compilation.ML_Hooks_ComponentsDraw != "") Test = "else if";
			Compilation.ML_Hooks_ComponentsDraw ^=
"""
		{{{Test}}} (ComponentInstance != Null && ComponentInstance.HasClass({{{dump(Component.Class)}}})) {
			{{{ComponentDrawFunctionCalls}}}
		}
""";
		}

		// Bind functions to the component mouse event hooks
		for (HookName => ComponentMouseEventFunctionCalls in ComponentMouseEventsFunctionCalls) {
			if (ComponentMouseEventFunctionCalls != "") {
				declare Text ML_Hook_ComponentsMouseEvents = ML_Hooks_ComponentsMouseEvents.get(HookName, "");
				declare Text Test = "if";
				if (ML_Hook_ComponentsMouseEvents != "") Test = "else if";
				ML_Hook_ComponentsMouseEvents ^=
"""
					{{{Test}}} (ComponentInstance != Null && ComponentInstance.HasClass({{{dump(Component.Class)}}})) {
						{{{ComponentMouseEventFunctionCalls}}}
					}
""";
				ML_Hooks_ComponentsMouseEvents[HookName] = ML_Hook_ComponentsMouseEvents;
			}
		}
		for (HookName => ML_Hook_ComponentsMouseEvents in ML_Hooks_ComponentsMouseEvents) {
			if (ML_Hook_ComponentsMouseEvents != "") {
				Compilation.ML_Hooks_ComponentsMouseEvents[HookName] =
	"""
					if (Event.Control != Null && Event.Control.HasClass({{{C_IncludeName}}}::C_Class_TriggerZone)) {
						declare CMlFrame ComponentInstance = {{{C_IncludeName}}}::GetComponentInstanceThatOwnsControl(Event.Control);
						{{{ML_Hook_ComponentsMouseEvents}}}
					}
	""";
			}
		}

		// Bind functions to the component hooks
		for (HookName => ComponentHookFunctionCalls in ComponentFunctionCalls) {
			if (ComponentHookFunctionCalls != "") {
				declare Text DebugTuningmarkOpen = "";
				declare Text DebugTuningmarkClose = "";
				if (C_Debug_EnableTuningMark) {
					DebugTuningmarkOpen = """tuningmark("+{{{Component.Name}}}@{{{HookName}}}");""";
					DebugTuningmarkClose = """tuningmark("-{{{Component.Name}}}@{{{HookName}}}");""";
				}

				declare Text ML_Hook_Components = Compilation.ML_Hooks_Components.get(HookName, "");
				ML_Hook_Components ^=
"""
	{{{DebugTuningmarkOpen}}}
	for (ComponentInstance in {{{C_IncludeName}}}::Private_GetComponentInstancesWithHook("{{{HookName}}}", "{{{Component.Name}}}")) {
		{{{ComponentHookFunctionCalls}}}
	}
	{{{DebugTuningmarkClose}}}
""";
				Compilation.ML_Hooks_Components[HookName] = ML_Hook_Components;
			}
		}
	}

	return Compilation;
}

/// Bind user functions to their requested mixin hooks
K_ManialinkPageCompilation Private_CompileStep_BindMixinsHooks(K_ManialinkPageCompilation _Compilation) {
	declare K_ManialinkPageCompilation Compilation = _Compilation;

	Compilation.ML_Hooks_Mixins = [];
	for (Mixin in reverse Compilation.SortedMixins) {
		// Bind functions to hooks
		declare Text MixinInitHookFunctionCalls = "";
		declare Text[Text] MixinFunctionCalls = [];
		for (HookName => HookFunction in Mixin.Hooks) {
			if (Compilation.DefinedMixinHooks.existskey(HookName)) {
				declare Text HookFunctionCall = Private_CallBoundFunction(HookFunction, Compilation.DefinedMixinHooks[HookName]);
				if (HookName == C_Hook_MixinInit) {
					MixinInitHookFunctionCalls ^= """{{{HookFunctionCall}}};""";
				} else if (
					HookName == C_Hook_MixinMainStart ||
					HookName == C_Hook_MixinMainEnd ||
					HookName == C_Hook_MixinBeforeYield ||
					HookName == C_Hook_MixinAfterYield
				) {
					MixinFunctionCalls[HookName] = MixinFunctionCalls.get(HookName, "") ^ """{{{HookFunctionCall}}};""";
				} else {
					declare Text DebugTuningmarkOpen = "";
					declare Text DebugTuningmarkClose = "";
					if (C_Debug_EnableTuningMark) {
						DebugTuningmarkOpen = """tuningmark("+{{{Mixin.Name}}}@{{{HookName}}}");""";
						DebugTuningmarkClose = """tuningmark("-{{{Mixin.Name}}}@{{{HookName}}}");""";
					}

					Compilation.ML_Hooks ^=
"""
***{{{HookName}}}***
***
{{{DebugTuningmarkOpen}}}
{{{HookFunctionCall}}};
{{{DebugTuningmarkClose}}}
***
""";
				}
			}
		}

		// Bind functions to the mixin init hook
		if (MixinInitHookFunctionCalls != "") {
			Compilation.ML_Hooks_MixinsInit ^=
"""
		if (MixinInstance.HasClass({{{dump(Mixin.Class)}}})) {
			{{{MixinInitHookFunctionCalls}}}
		}
""";
		}

		// Bind functions to the mixin hooks
		for (HookName => MixinHookFunctionCalls in MixinFunctionCalls) {
			if (MixinHookFunctionCalls != "") {
				declare Text DebugTuningmarkOpen = "";
				declare Text DebugTuningmarkClose = "";
				if (C_Debug_EnableTuningMark) {
					DebugTuningmarkOpen = """tuningmark("+{{{Mixin.Name}}}@{{{HookName}}}");""";
					DebugTuningmarkClose = """tuningmark("-{{{Mixin.Name}}}@{{{HookName}}}");""";
				}

				declare Text ML_Hook_Mixins = Compilation.ML_Hooks_Mixins.get(HookName, "");
				ML_Hook_Mixins ^=
"""
		for (MixinInstance in {{{C_Prefix}}}G_MixinHooksMixinInstances.get("{{{HookName}}}", []).get("{{{Mixin.Name}}}", [])) {
			{{{DebugTuningmarkOpen}}}
			{{{MixinHookFunctionCalls}}}
			{{{DebugTuningmarkClose}}}
		}
""";
				Compilation.ML_Hooks_Mixins[HookName] = ML_Hook_Mixins;
			}
		}
	}

	return Compilation;
}

/// Inject the page component instance in the manialink page
K_ManialinkPageCompilation Private_CompileStep_InjectPageComponentInstance(K_ManialinkPageCompilation _Compilation) {
	if (_Compilation.PageComponent.Template == "") return _Compilation;

	declare K_ManialinkPageCompilation Compilation = _Compilation;

	declare Text ComponentHooks = "";
	for (HookName => HookFunctionCall in _Compilation.PageComponent.Hooks) {
		if (
			HookName == C_Hook_ComponentMainStart ||
			HookName == C_Hook_ComponentMainEnd ||
			HookName == C_Hook_ComponentBeforeYield ||
			HookName == C_Hook_ComponentAfterYield
		) {
			if (ComponentHooks != "") ComponentHooks ^= ";";
			ComponentHooks ^= HookName;
		}
	}
	declare Text XmlComponentHooks = "";
	if (ComponentHooks != "") {
		XmlComponentHooks = "data-"^C_Attribute_ComponentHooks^"=\""^ComponentHooks^"\"";
	}

	Compilation.ML_Templates ^=
"""
{{{_Compilation.PageComponent.Slots.get(C_Slot_ComponentStart, "")}}}
<frameinstance
	modelid="{{{TL::MLEncode(_Compilation.PageComponent.Name)}}}"
	class="{{{TL::MLEncode(_Compilation.PageComponent.Class)}}} {{{C_Class_ComponentInstance}}}"
	z-index="{{{_Compilation.ZIndex}}}"
	data-{{{C_Attribute_ComponentName}}}="{{{TL::MLEncode(_Compilation.PageComponent.Name)}}}"
	{{{XmlComponentHooks}}}
/>
{{{_Compilation.PageComponent.Slots.get(C_Slot_ComponentEnd, "")}}}
""";

	return Compilation;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Parse mixin instance xml and generate code for them
K_ComponentCompilation Private_CompileStep_CompileMixinTemplate(
	K_ComponentCompilation _ComponentCompilation,
	K_ManialinkPageCompilation _Compilation
) {
	declare K_ComponentCompilation ComponentCompilation = _ComponentCompilation;
	declare K_EventDefinition[Text] AllDefinedEvents;
	declare Text[] XmlMixins = Private_FindMixins(ComponentCompilation.Template, False);
	for (XmlMixin in XmlMixins) {
		declare K_MatchedMixin MatchedMixin = Private_MatchMixin(XmlMixin);
		if (MatchedMixin.Matched) {
			declare Text[Text] ParsedAttributes = Private_ParseAttributes(MatchedMixin.Attributes);
			declare Text[] MixinClasses;

			// Find all props and events in the mixin instance
			declare Text[Text] ControlEventBindings;
			for (AttributeName => AttributeValue in ParsedAttributes) {
				declare Text[] Matches = TL::RegexMatch("""^({{{C_DefaultDirectives["mv-mixin-set:"]}}}|{{{C_DefaultDirectives["mv-mixin-on:"]}}})({{{C_Regex_TagWord}}})$""", AttributeName, "");
				if (Matches.count >= 3) {
					declare Text MixinName = "";
					if (Matches[1] == C_DefaultDirectives["mv-mixin-set:"]) {
						assert(
							_Compilation.MixinsProps.existskey(Matches[2]),
							"""[ManiaView@3] Unknown mixin prop `{{{Matches[2]}}}` on control with id `{{{ParsedAttributes.get("id", "")}}}`"""
						);
						MixinName = _Compilation.MixinsProps[Matches[2]];
					} else if (Matches[1] == C_DefaultDirectives["mv-mixin-on:"]) {
						declare Text EventType = Matches[2];
						assert(
							_Compilation.MixinsEvents.existskey(EventType),
							"""[ManiaView@3] Unknown mixin event `{{{EventType}}}` on control with id `{{{ParsedAttributes.get("id", "")}}}`"""
						);

						MixinName = _Compilation.MixinsEvents[EventType];
						declare Text EventTypeBindingKey = Private_GetEventTypeBindingKey(MixinName, EventType);
						assert(
							!ControlEventBindings.existskey(EventTypeBindingKey),
							"""[ManiaView@3] A function is already bound to mixin event `{{{EventType}}}` on control with id `{{{ParsedAttributes.get("id", "")}}}`"""
						);

						ControlEventBindings[EventTypeBindingKey] = AttributeValue;
						if (!AllDefinedEvents.existskey(EventType)) {
							AllDefinedEvents[EventType] = _Compilation.SortedMixins.get(MixinName, K_Mixin {}).DefineEvents.get(EventType, K_EventDefinition {});
						}
					}
					if (MixinName != "") {
						declare K_Mixin Mixin = _Compilation.SortedMixins.get(MixinName, K_Mixin {});
						if (Mixin.Class != "" && !MixinClasses.exists(Mixin.Class)) {
							MixinClasses.add(Mixin.Class);
						}

						ParsedAttributes["data-"^C_Attribute_MixinName] = Mixin.Name;

						// Mark mixins with a mixin level hook
						if (Mixin.Hooks.count > 0) {
							declare Text[] MixinHooks = TL::Split(";", ParsedAttributes.get("data-"^C_Attribute_MixinHooks, ""));
							for (HookName => HookFunctionCall in Mixin.Hooks) {
								if (
									(
										HookName == C_Hook_MixinMainStart ||
										HookName == C_Hook_MixinMainEnd ||
										HookName == C_Hook_MixinBeforeYield ||
										HookName == C_Hook_MixinAfterYield
									) &&
									!MixinHooks.exists(HookName)
								) {
									MixinHooks.add(HookName);
								}
							}
							if (MixinHooks.count > 0) {
								ParsedAttributes["data-"^C_Attribute_MixinHooks] = TL::Join(";", MixinHooks);
							}
						}

						// Inject mixin slots
						MatchedMixin.SlotStart = Mixin.Slots.get(C_Slot_MixinStart, "");
						MatchedMixin.SlotEnd = Mixin.Slots.get(C_Slot_MixinEnd, "");
					}
				}
			}

			// Bind functions to the events
			if (ControlEventBindings.count > 0) {
				declare Text Class = ParsedAttributes.get("class", "");
				declare Text EventBindingsClass = C_Class_EventBindings^"-"^Private_GetComponentCompilationUniqueEventBindingId();
				Class ^= " " ^ EventBindingsClass;
				ParsedAttributes["class"] = Class;

				for (EventTypeBindingKey => EventFunction in ControlEventBindings) {
					if (ComponentCompilation.MixinsEventBindings.existskey(EventTypeBindingKey)) {
						ComponentCompilation.MixinsEventBindings[EventTypeBindingKey].add([EventFunction, EventBindingsClass]);
					} else {
						ComponentCompilation.MixinsEventBindings[EventTypeBindingKey] = [[EventFunction, EventBindingsClass]];
					}
				}
			}

			// Add the mixin instance class to the node
			if (MixinClasses.count > 0) {
				declare Text Class = ParsedAttributes.get("class", "");
				if (Class != "") Class ^= " ";
				Class ^= C_Class_MixinInstance ^ " " ^ TL::Join(" ", MixinClasses);
				ParsedAttributes["class"] = Class;
			}

			// Rewrite mixin props in the xml
			ComponentCompilation.Template = TL::Replace(
				ComponentCompilation.Template,
				MatchedMixin.Xml,
				"""{{{MatchedMixin.SlotStart}}}{{{MatchedMixin.Prepend}}} {{{Private_OutputAttributes(ParsedAttributes)}}} {{{MatchedMixin.Append}}}{{{MatchedMixin.SlotEnd}}}"""
			);
		}
	}
	return ComponentCompilation;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Parse component instance xml and generate code for them
K_ComponentCompilation Private_CompileStep_CompileComponentTemplate(
	K_ComponentCompilation _ComponentCompilation,
	K_Component _Component
) {
	declare K_ComponentCompilation ComponentCompilation = _ComponentCompilation;
	declare Text[] XmlComponents = Private_FindComponents(_Component.Name, ComponentCompilation.Template);
	for (XmlComponent in XmlComponents) {
		declare K_MatchedComponent MatchedComponent = Private_MatchComponent(_Component.Name, XmlComponent);
		if (MatchedComponent.Matched) {
			declare Text[Text] Attributes = Private_ParseAttributes(MatchedComponent.Attributes);

			// Add the component instance class and name to the node
			declare Text Class = Attributes.get("class", "");
			if (Class != "") Class ^= " ";
			Class ^= _Component.Class ^ " " ^ C_Class_ComponentInstance;
			Attributes["data-"^C_Attribute_ComponentName] = _Component.Name;

			// Mark components with a component level hook
			declare Text ComponentHooks = Attributes.get("data-"^C_Attribute_ComponentHooks, "");
			for (HookName => HookFunctionCall in _Component.Hooks) {
				if (
					HookName == C_Hook_ComponentMainStart ||
					HookName == C_Hook_ComponentMainEnd ||
					HookName == C_Hook_ComponentBeforeYield ||
					HookName == C_Hook_ComponentAfterYield
				) {
					if (ComponentHooks != "") ComponentHooks ^= ";";
					ComponentHooks ^= HookName;
				}
			}
			if (ComponentHooks != "") {
				Attributes["data-"^C_Attribute_ComponentHooks] = ComponentHooks;
			}

			// Find events binding
			declare Text[Text] ComponentEventBindings;
			for (AttributeName => AttributeValue in Attributes) {
				declare Text[] Matches = TL::RegexMatch("""^{{{C_DefaultDirectives["mv-on:"]}}}({{{C_Regex_TagWord}}})$""", AttributeName, "");
				if (Matches.count >= 2) {
					assert(
						_Component.DefineEvents.existskey(Matches[1]),
						"""[ManiaView@3] Unknown event `{{{Matches[1]}}}` on component with id `{{{Attributes.get("id", "")}}}`"""
					);
					ComponentEventBindings[Private_GetEventTypeBindingKey(_Component.Name, Matches[1])] = AttributeValue;
				}
			}
			// Bind functions to the events
			if (ComponentEventBindings.count > 0) {
				declare Text EventBindingsClass = C_Class_EventBindings^"-"^Private_GetComponentCompilationUniqueEventBindingId();
				Class ^= " " ^ EventBindingsClass;

				for (EventTypeBindingKey => EventFunction in ComponentEventBindings) {
					if (ComponentCompilation.ComponentsEventBindings.existskey(EventTypeBindingKey)) {
						ComponentCompilation.ComponentsEventBindings[EventTypeBindingKey].add([EventFunction, EventBindingsClass]);
					} else {
						ComponentCompilation.ComponentsEventBindings[EventTypeBindingKey] = [[EventFunction, EventBindingsClass]];
					}
				}
			}

			// Validate props
			for (AttributeName => AttributeValue in Attributes) {
				declare Text[] Matches = TL::RegexMatch("""^{{{C_DefaultDirectives["mv-set:"]}}}({{{C_Regex_TagWord}}})$""", AttributeName, "");
				if (Matches.count >= 2) {
					assert(
						_Component.DefineProps.exists(Matches[1]),
						"""[ManiaView@3] Unknown prop `{{{Matches[1]}}}` on component with id `{{{Attributes.get("id", "")}}}`"""
					);
				}
			}

			Attributes["class"] = Class;

			// Replace component node by a frameinstance
			ComponentCompilation.Template = TL::Replace(
				ComponentCompilation.Template,
				MatchedComponent.Xml,
				"""{{{_Component.Slots.get(C_Slot_ComponentStart, "")}}}<frameinstance modelid="{{{TL::MLEncode(_Component.Name)}}}" {{{Private_OutputAttributes(Attributes)}}} />{{{_Component.Slots.get(C_Slot_ComponentEnd, "")}}}"""
			);
		}
	}

	return ComponentCompilation;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Parse component instance xml and generate code for them
K_ComponentCompilation Private_CompileStep_CompileComponentTemplates(K_ManialinkPageCompilation _Compilation, K_Component _Component) {
	declare K_ComponentCompilation ComponentCompilation = Private_GetComponentCompilationCache(_Component.Name);

	if (!ComponentCompilation.IsCompiled) {
		ComponentCompilation.Template = _Component.Template;

		// Mixins must be processed before components
		if (_Component.Mixins.count > 0) {
			ComponentCompilation = Private_CompileStep_CompileMixinTemplate(ComponentCompilation, _Compilation);
		}

		for (UsedComponent in _Component.Components) {
			ComponentCompilation = Private_CompileStep_CompileComponentTemplate(ComponentCompilation, UsedComponent);
		}

		ComponentCompilation.IsCompiled = True;
		Private_SetComponentCompilationCache(_Component.Name, ComponentCompilation);
	}

	return ComponentCompilation;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Parse component instances xml and generate code for them
K_ManialinkPageCompilation Private_CompileStep_CompileComponentsTemplates(K_ManialinkPageCompilation _Compilation) {
	declare K_ManialinkPageCompilation Compilation = _Compilation;

	Compilation.CompiledComponents = [];
	for (Component in Compilation.SortedComponents) {
		Compilation.CompiledComponents[Component.Name] = Private_CompileStep_CompileComponentTemplates(_Compilation, Component);

		for (EventTypeBindingKey => EventBindings in Compilation.CompiledComponents[Component.Name].ComponentsEventBindings) {
			if (Compilation.ComponentsEventBindings.existskey(EventTypeBindingKey)) {
				Compilation.ComponentsEventBindings[EventTypeBindingKey].add(EventBindings);
			} else {
				Compilation.ComponentsEventBindings[EventTypeBindingKey] = EventBindings;
			}
		}

		for (EventTypeBindingKey => EventBindings in Compilation.CompiledComponents[Component.Name].MixinsEventBindings) {
			if (Compilation.MixinsEventBindings.existskey(EventTypeBindingKey)) {
				Compilation.MixinsEventBindings[EventTypeBindingKey].add(EventBindings);
			} else {
				Compilation.MixinsEventBindings[EventTypeBindingKey] = EventBindings;
			}
		}
	}

	return Compilation;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Concatenate components manialink parts
K_ManialinkPageCompilation Private_CompileStep_ConcatenateComponents(K_ManialinkPageCompilation _Compilation) {
	declare K_ManialinkPageCompilation Compilation = _Compilation;

	for (Component in reverse Compilation.SortedComponents) {
		declare K_ComponentCompilation CompiledComponent = Compilation.CompiledComponents.get(Component.Name, K_ComponentCompilation {});

		if (CompiledComponent.Template != "") {
			Compilation.ML_Templates ^=
"""
<framemodel id="{{{TL::MLEncode(Component.Name)}}}">
{{{CompiledComponent.Template}}}
</framemodel>
""";
		}

		// Auto include component inside the manialink script
		if (Component.Path != "") {
			Compilation.ML_Script_Directives ^=
"""
#Include "{{{Component.Path}}}" as {{{Component.Name}}}
""";
		}

		Compilation.ML_Script_Directives ^= Component.Script.Directives;
		Compilation.ML_Script_DeclarationsTop ^= Component.Script.DeclarationsTop;
		Compilation.ML_Script_Declarations ^= Component.Script.Declarations;
		Compilation.ML_Script_DeclarationsBottom ^= Component.Script.DeclarationsBottom;

		// Generate the components events processing code
		for (EventType => EventDefinition in Component.DefineEvents) {
			declare Text[][] EventBindings = Compilation.ComponentsEventBindings.get(Private_GetEventTypeBindingKey(Component.Name, EventType), []);
			declare Text ML_EventBindings = "";
			for (EventBinding in EventBindings) {
				declare Text HasClassTest = "if";
				if (ML_EventBindings != "") {
					HasClassTest = "else if";
				}
				ML_EventBindings ^=
"""
			{{{HasClassTest}}} (Event.Component.HasClass("{{{EventBinding[1]}}}")) {
				{{{Private_CallBoundFunction(EventBinding[0], "Event")}}};
			}
""";
			}

			if (ML_EventBindings != "") {
				Compilation.ML_ProcessComponentsEvents ^=
"""
	for (Event in {{{Private_CallBoundFunction(EventDefinition.Get, "This")}}}) {
		if (Event.Component != Null) {
			{{{ML_EventBindings}}}
		}
	}
""";
			}
			Compilation.ML_ProcessComponentsEvents ^=
"""
	{{{Private_CallBoundFunction(EventDefinition.Reset, "This")}}};
""";
		}
	}

	return Compilation;
}

/// Concatenate mixins manialink parts
K_ManialinkPageCompilation Private_CompileStep_ConcatenateMixins(K_ManialinkPageCompilation _Compilation) {
	declare K_ManialinkPageCompilation Compilation = _Compilation;

	for (Mixin in reverse Compilation.SortedMixins) {
		// Auto include mixin inside the manialink script
		if (Mixin.Path != "") {
			Compilation.ML_Script_Directives ^=
"""
#Include "{{{Mixin.Path}}}" as {{{Mixin.Name}}}
""";
		}

		Compilation.ML_Script_Directives ^= Mixin.Script.Directives;
		Compilation.ML_Script_DeclarationsTop ^= Mixin.Script.DeclarationsTop;
		Compilation.ML_Script_Declarations ^= Mixin.Script.Declarations;
		Compilation.ML_Script_DeclarationsBottom ^= Mixin.Script.DeclarationsBottom;

		// Generate the mixins events processing code
		for (EventType => EventDefinition in Mixin.DefineEvents) {
			declare Text EventTypeBindingKey = Private_GetEventTypeBindingKey(Mixin.Name, EventType);
			declare Text[][] EventBindings = Compilation.MixinsEventBindings.get(EventTypeBindingKey, []);
			declare Text ML_EventBindings = "";
			for (EventBinding in EventBindings) {
				declare Text HasClassTest = "if";
				if (ML_EventBindings != "") {
					HasClassTest = "else if";
				}
				ML_EventBindings ^=
"""
			{{{HasClassTest}}} (Event.Control.HasClass("{{{EventBinding[1]}}}")) {
				{{{Private_CallBoundFunction(EventBinding[0], "Event")}}};
			}
""";
			}

			if (ML_EventBindings != "") {
				Compilation.ML_ProcessMixinsEvents ^=

		if (Event.Control != Null) {

	{{{Private_CallBoundFunction(EventDefinition.Reset, "This")}}};
""";
		}
	}

	return Compilation;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check that the given slot id is valid
Void Private_IsValidSlotId(Text _SlotId, Integer _ElementType, Text _ElementName) {
	declare Text Message = "";
	switch (_ElementType) {
		case C_Type_Component: Message = "component";
		case C_Type_Mixin: Message = "mixin";
	}
	assert(
		(
			_SlotId == C_Slot_Styles ||
			_SlotId == C_Slot_PageStart ||
			_SlotId == C_Slot_PageEnd || (
				_ElementType == C_Type_Component && (
					_SlotId == C_Slot_ComponentStart ||
					_SlotId == C_Slot_ComponentEnd
				)
			) || (
				_ElementType == C_Type_Mixin && (
					_SlotId == C_Slot_MixinStart ||
					_SlotId == C_Slot_MixinEnd
				)
			)
		),
		"""[ManiaView@3] Unknown slot `{{{_SlotId}}}` in {{{Message}}} `{{{_ElementName}}}`"""
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Concatenate components and mixins slots
K_ManialinkPageCompilation Private_CompileStep_ConcatenateSlots(K_ManialinkPageCompilation _Compilation) {
	declare K_ManialinkPageCompilation Compilation = _Compilation;

	Compilation.ML_Slots = [];

	for (Component in reverse Compilation.SortedComponents) {
		for (SlotId => SlotContent in Component.Slots) {
			Private_IsValidSlotId(SlotId, C_Type_Component, Component.Name);

			if (SlotId == C_Slot_Styles || SlotId == C_Slot_PageStart || SlotId == C_Slot_PageEnd) {
				if (Compilation.ML_Slots.existskey(SlotId)) {
					Compilation.ML_Slots[SlotId] ^= "\n"^SlotContent;
				} else {
					Compilation.ML_Slots[SlotId] = SlotContent;
				}
			}
		}
	}

	for (Mixin in reverse Compilation.SortedMixins) {
		for (SlotId => SlotContent in Mixin.Slots) {
			Private_IsValidSlotId(SlotId, C_Type_Mixin, Mixin.Name);

			if (SlotId == C_Slot_Styles || SlotId == C_Slot_PageStart || SlotId == C_Slot_PageEnd) {
				if (Compilation.ML_Slots.existskey(SlotId)) {
					Compilation.ML_Slots[SlotId] ^= "\n"^SlotContent;
				} else {
					Compilation.ML_Slots[SlotId] = SlotContent;
				}
			}
		}
	}

	declare Text Slot_Styles = Compilation.ML_Slots.get(C_Slot_Styles, "");
	if (Slot_Styles != "") {
		Compilation.ML_Slots[C_Slot_Styles] =
"""
	<stylesheet>
		{{{Slot_Styles}}}
	</stylesheet>
""";
	}

	return Compilation;
}

/// Generate the manialink page
Text Private_CompileStep_GenerateManialinkPage(K_ManialinkPageCompilation _Compilation) {
	return
"""
<manialink version="{{{_Compilation.ManialinkVersion}}}" name="{{{TL::MLEncode(_Compilation.PageComponent.Name)}}}">
	{{{_Compilation.ML_Slots.get(C_Slot_Styles, "")}}}
	{{{_Compilation.ML_Slots.get(C_Slot_PageStart, "")}}}
	{{{_Compilation.ML_Templates}}}
	{{{_Compilation.ML_Slots.get(C_Slot_PageEnd, "")}}}
	<script><!--
#Include "TextLib" as {{{C_Prefix}}}TL
#Include "{{{ScriptName}}}" as {{{C_IncludeName}}}
#Include "Libs/Nadeo/CMGame/Utils/Stylesheet@4.Script.txt" as {{{C_Prefix}}}Stylesheet4
{{{_Compilation.ML_Script_Directives}}}

#Const {{{C_Prefix}}}C_ComponentsPriorities {{{dump(_Compilation.ComponentsPriorities.sort())}}}

declare CMlFrame[][Text] {{{C_Prefix}}}G_ComponentInstances;
declare CMlControl[] {{{C_Prefix}}}G_MixinInstances;
declare CMlControl[][Text][Text] {{{C_Prefix}}}G_MixinHooksMixinInstances;

{{{_Compilation.ML_Script_DeclarationsTop}}}

Void {{{C_Prefix}}}ProcessHook_ComponentMainStart() {
	{{{_Compilation.ML_Hooks_Components.get(C_Hook_ComponentMainStart, "")}}}
}

Void {{{C_Prefix}}}ProcessHook_ComponentMainEnd() {
	{{{_Compilation.ML_Hooks_Components.get(C_Hook_ComponentMainEnd, "")}}}
}

Void {{{C_Prefix}}}ProcessHook_ComponentBeforeYield() {
	{{{_Compilation.ML_Hooks_Components.get(C_Hook_ComponentBeforeYield, "")}}}
}

Void {{{C_Prefix}}}ProcessHook_ComponentAfterYield() {
	{{{_Compilation.ML_Hooks_Components.get(C_Hook_ComponentAfterYield, "")}}}
}

Void {{{C_Prefix}}}ProcessComponentsEvents() {
	+++{{{C_Hook_PageBeforeComponentsEvents}}}+++
	{{{_Compilation.ML_ProcessComponentsEvents}}}
	+++{{{C_Hook_PageAfterComponentsEvents}}}+++
}

Void {{{C_Prefix}}}ProcessHook_MixinMainStart() {
	{{{_Compilation.ML_Hooks_Mixins.get(C_Hook_MixinMainStart, "")}}}
}

Void {{{C_Prefix}}}ProcessHook_MixinMainEnd() {
	{{{_Compilation.ML_Hooks_Mixins.get(C_Hook_MixinMainEnd, "")}}}
}

Void {{{C_Prefix}}}ProcessHook_MixinBeforeYield() {
	{{{_Compilation.ML_Hooks_Mixins.get(C_Hook_MixinBeforeYield, "")}}}
}

Void {{{C_Prefix}}}ProcessHook_MixinAfterYield() {
	{{{_Compilation.ML_Hooks_Mixins.get(C_Hook_MixinAfterYield, "")}}}
}

Void {{{C_Prefix}}}ProcessMixinsEvents() {
	+++{{{C_Hook_PageBeforeMixinsEvents}}}+++
	{{{_Compilation.ML_ProcessMixinsEvents}}}
	+++{{{C_Hook_PageAfterMixinsEvents}}}+++
}

Void {{{C_Prefix}}}ProcessManialinkEvents() {
	+++{{{C_Hook_PageBeforeManialinkEvents}}}+++
	for (Event in PendingEvents) {
		+++{{{C_Hook_PageOnEvent}}}+++
		switch (Event.Type) {
			case CMlScriptEvent::Type::KeyPress: {
				+++{{{C_Hook_PageOnEvent_KeyPress}}}+++
			}
			case CMlScriptEvent::Type::MouseClick: {
				+++{{{C_Hook_PageOnEvent_MouseClick}}}+++
				{{{_Compilation.ML_Hooks_ComponentsMouseEvents.get(C_Hook_ComponentOnEvent_TriggerZoneMouseClick, "")}}}
			}
			case CMlScriptEvent::Type::MouseRightClick: {
				+++{{{C_Hook_PageOnEvent_MouseRightClick}}}+++
				{{{_Compilation.ML_Hooks_ComponentsMouseEvents.get(C_Hook_ComponentOnEvent_TriggerZoneMouseRightClick, "")}}}
			}
			case CMlScriptEvent::Type::MouseOver: {
				+++{{{C_Hook_PageOnEvent_MouseOver}}}+++
				{{{_Compilation.ML_Hooks_ComponentsMouseEvents.get(C_Hook_ComponentOnEvent_TriggerZoneMouseOver, "")}}}
			}
			case CMlScriptEvent::Type::MouseOut: {
				+++{{{C_Hook_PageOnEvent_MouseOut}}}+++
				{{{_Compilation.ML_Hooks_ComponentsMouseEvents.get(C_Hook_ComponentOnEvent_TriggerZoneMouseOut, "")}}}
			}
			case CMlScriptEvent::Type::EntrySubmit: {
				+++{{{C_Hook_PageOnEvent_EntrySubmit}}}+++
			}
			case CMlScriptEvent::Type::MenuNavigation: {
				+++{{{C_Hook_PageOnEvent_MenuNavigation}}}+++
			}
			case CMlScriptEvent::Type::PluginCustomEvent: {
				+++{{{C_Hook_PageOnEvent_PluginCustomEvent}}}+++
			}
		}
	}
	+++{{{C_Hook_PageAfterManialinkEvents}}}+++
}

Void {{{C_Prefix}}}RedrawComponents() {
	// We could request a new redraw in the draw hook calls
	// So first we need to store the list of components requesting a redraw
	// in a temporary variable, then reset the list
	// and only after that loop on the list stored in the temporary variable
	declare {{{C_IncludeName}}}::K_Redraw[Ident] ComponentInstancesToRedraw = {{{C_IncludeName}}}::GetComponentInstancesToRedraw();
	{{{C_IncludeName}}}::ResetComponentInstancesToRedraw();
	for (ComponentInstanceId => Redraw in ComponentInstancesToRedraw) {
		declare CMlFrame ComponentInstance = Redraw.Component;
		declare Text[] Tags = Redraw.Tags;
		{{{_Compilation.ML_Hooks_ComponentsDraw}}}
	}
}

Void {{{C_Prefix}}}Yield() {
	+++{{{C_Hook_PageBeforeYield}}}+++
	{{{C_Prefix}}}ProcessHook_ComponentBeforeYield();
	{{{C_Prefix}}}ProcessHook_MixinBeforeYield();
	{{{C_Prefix}}}ProcessComponentsEvents();
	{{{C_Prefix}}}ProcessMixinsEvents();
	{{{C_Prefix}}}RedrawComponents();
	yield;
	// Process C++ ML events before anything else so it can generate data used by the following steps
	{{{C_Prefix}}}ProcessManialinkEvents();
	+++{{{C_Hook_PageAfterYield}}}+++
	{{{C_Prefix}}}ProcessHook_ComponentAfterYield();
	{{{C_Prefix}}}ProcessHook_MixinAfterYield();
}

{{{_Compilation.ML_Script_Declarations}}}
{{{_Compilation.ML_Hooks}}}

Void {{{C_Prefix}}}InitializeComponents() {
	{{{C_Prefix}}}G_ComponentInstances = [];
	for (ComponentName => ComponentPriority in {{{C_Prefix}}}C_ComponentsPriorities) {
		{{{C_Prefix}}}G_ComponentInstances[ComponentName] = [];
	}
	Page.GetClassChildren({{{dump(C_Class_ComponentInstance)}}}, Page.MainFrame, True);
	for (Control in Page.GetClassChildren_Result) {
		declare CMlFrame ComponentInstance = (Control as CMlFrame);
		declare Text ComponentName = {{{C_IncludeName}}}::GetComponentName(ComponentInstance);
		if (!{{{C_Prefix}}}G_ComponentInstances.existskey(ComponentName)) {
			{{{C_Prefix}}}G_ComponentInstances[ComponentName] = [];
		}
		{{{C_Prefix}}}G_ComponentInstances[ComponentName].add(ComponentInstance);
	}
	{{{C_IncludeName}}}::Private_ResetComponentHooksComponentInstances();
	for (ComponentName => ComponentInstances in {{{C_Prefix}}}G_ComponentInstances) {
		for (ComponentInstance in ComponentInstances) {
			if (ComponentInstance.DataAttributeExists({{{dump(C_Attribute_ComponentHooks)}}})) {
				for (HookName in {{{C_Prefix}}}TL::Split(";", ComponentInstance.DataAttributeGet({{{dump(C_Attribute_ComponentHooks)}}}))) {
					{{{C_IncludeName}}}::Private_AddComponentInstanceHook(ComponentInstance, HookName, ComponentName);
				}
			}
			{{{_Compilation.ML_Hooks_ComponentsInit}}}
			+++{{{C_Hook_ComponentAllInit}}}+++
		}
	}
}

Void {{{C_Prefix}}}InitializeMixins() {
	{{{C_Prefix}}}G_MixinInstances = [];
	Page.GetClassChildren({{{dump(C_Class_MixinInstance)}}}, Page.MainFrame, True);
	for (Control in Page.GetClassChildren_Result) {
		declare CMlControl MixinInstance = Control;
		{{{C_Prefix}}}G_MixinInstances.add(MixinInstance);
	}
	for (MixinInstance in {{{C_Prefix}}}G_MixinInstances) {
		if (
			MixinInstance.DataAttributeExists({{{dump(C_Attribute_MixinHooks)}}}) &&
			MixinInstance.DataAttributeExists({{{dump(C_Attribute_MixinName)}}})
		) {
			declare Text[] MixinHooks = {{{C_Prefix}}}TL::Split(";", MixinInstance.DataAttributeGet({{{dump(C_Attribute_MixinHooks)}}}));
			declare Text MixinName = MixinInstance.DataAttributeGet({{{dump(C_Attribute_MixinName)}}});
			for (HookName in MixinHooks) {
				if ({{{C_Prefix}}}G_MixinHooksMixinInstances.existskey(HookName)) {
					if ({{{C_Prefix}}}G_MixinHooksMixinInstances[HookName].existskey(MixinName)) {
						{{{C_Prefix}}}G_MixinHooksMixinInstances[HookName][MixinName].add(MixinInstance);
					} else {
						{{{C_Prefix}}}G_MixinHooksMixinInstances[HookName][MixinName] = [MixinInstance];
					}
				} else {
					{{{C_Prefix}}}G_MixinHooksMixinInstances[HookName] = [MixinName => [MixinInstance]];
				}
			}
		}
		{{{_Compilation.ML_Hooks_MixinsInit}}}
		+++{{{C_Hook_MixinAllInit}}}+++
	}
}

{{{_Compilation.ML_Script_DeclarationsBottom}}}

main() {
	+++{{{C_Hook_PageStylesheetInit}}}+++
	{{{C_Prefix}}}Stylesheet4::ApplyStylesheet(Page);

	+++{{{C_Hook_PageMainInit}}}+++

	{{{C_Prefix}}}InitializeComponents();
	{{{C_Prefix}}}InitializeMixins();

	+++{{{C_Hook_PageMainStart}}}+++
	{{{C_Prefix}}}ProcessHook_ComponentMainStart();
	{{{C_Prefix}}}ProcessHook_MixinMainStart();

	while (True) {
		{{{C_Prefix}}}Yield();
	}

	+++{{{C_Hook_PageMainEnd}}}+++
	{{{C_Prefix}}}ProcessHook_ComponentMainEnd();
	{{{C_Prefix}}}ProcessHook_MixinMainEnd();
}
--></script>
</manialink>
""";
}

/// Build a component as a manialink page
Text ManialinkPage(K_Component _PageComponent, Integer _ManialinkVersion, Real _ZIndex) {
	declare K_ManialinkPageCompilation Compilation = K_ManialinkPageCompilation {
		ML_Templates = "",
		ML_Hooks = "",
		ML_Hooks_Components = [],
		ML_Hooks_ComponentsMouseEvents = [],
		ML_Hooks_ComponentsInit = "",
		ML_Hooks_ComponentsDraw = "",
		ML_Hooks_Mixins = [],
		ML_Hooks_MixinsInit = "",
		ML_ProcessComponentsEvents = "",
		ML_ProcessMixinsEvents = "",
		ML_Script_Directives = "",
		ML_Script_DeclarationsTop = "",
		ML_Script_Declarations = "",
		ML_Script_DeclarationsBottom = "",
		DefinedComponentHooks = Private_GetDefaultComponentHooks(),
		DefinedMixinHooks = Private_GetDefaultMixinHooks(),
		PageComponent = _PageComponent,
		SortedComponents = [],
		CompiledComponents = [],
		ComponentsPriorities = [],
		ComponentsEventBindings = [],
		SortedMixins = [],
		MixinsProps = [],
		MixinsEvents = [],
		MixinsEventBindings = [],
		ManialinkVersion = _ManialinkVersion,
		ZIndex = _ZIndex
	};

	Compilation = Private_CompileStep_SortComponents(Compilation);
	Compilation = Private_CompileStep_SortMixins(Compilation);
	Compilation = Private_CompileStep_CompileComponentsTemplates(Compilation);
	Compilation = Private_CompileStep_BindComponentsHooks(Compilation);
	Compilation = Private_CompileStep_BindMixinsHooks(Compilation);
	Compilation = Private_CompileStep_ConcatenateComponents(Compilation);
	Compilation = Private_CompileStep_ConcatenateMixins(Compilation);
	Compilation = Private_CompileStep_ConcatenateSlots(Compilation);
	Compilation = Private_CompileStep_InjectPageComponentInstance(Compilation);
	return Private_CompileStep_GenerateManialinkPage(Compilation);
}
Text ManialinkPage(K_Component _PageComponent, Integer _ManialinkVersion) {
	return ManialinkPage(_PageComponent, _ManialinkVersion, C_DefaultZIndex);
}
Text ManialinkPage(K_Component _PageComponent, Real _ZIndex) {
	return ManialinkPage(_PageComponent, C_DefaultManialinkVersion, _ZIndex);
}
Text ManialinkPage(K_Component _PageComponent) {
	return ManialinkPage(_PageComponent, C_DefaultManialinkVersion, C_DefaultZIndex);
