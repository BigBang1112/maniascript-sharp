/**
 * Synchronize different scripts
 */
#Const Version "1.0.0"
#Const ScriptName "Libs/Nadeo/CMGame/Utils/Synchro.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Status_Null 0
#Const C_Status_Initializing 1
#Const C_Status_Initialized 2

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set title menu initialization status
Void SetTitleMenuInitializationStatus(CSystemPlatform _System, Integer _Status) {
	declare Integer CMGame_UtilsSynchro_TitleMenuInitializationStatus for _System = C_Status_Null;
	CMGame_UtilsSynchro_TitleMenuInitializationStatus = _Status;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the title menu is initialized
Boolean IsTitleMenuInitialized(CSystemPlatform _System) {
	declare Integer CMGame_UtilsSynchro_TitleMenuInitializationStatus for _System = C_Status_Null;
	return CMGame_UtilsSynchro_TitleMenuInitializationStatus == C_Status_Initialized;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Start a synchronization point
Void Start(CSystemPlatform _System, Text _Id) {
	declare Text[] CMGame_UtilsSynchro_Ids for _System = [];
	if (!CMGame_UtilsSynchro_Ids.exists(_Id)) {
		CMGame_UtilsSynchro_Ids.add(_Id);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Complete a synchronization point
Void Complete(CSystemPlatform _System, Text _Id) {
	declare Text[] CMGame_UtilsSynchro_Ids for _System = [];
	CMGame_UtilsSynchro_Ids.remove(_Id);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if a synchronization point is complete
Boolean IsComplete(CSystemPlatform _System, Text _Id) {
	declare Text[] CMGame_UtilsSynchro_Ids for _System = [];
	return !CMGame_UtilsSynchro_Ids.exists(_Id);
}
