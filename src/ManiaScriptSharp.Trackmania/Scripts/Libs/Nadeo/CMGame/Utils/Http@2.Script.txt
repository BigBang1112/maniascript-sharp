/// Check if there are already too many requests
Boolean IsSaturated(CHttpManager _HttpMgr) {
	return (_HttpMgr.SlotsAvailable <= C_MinimumAvailableSlotsNb);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if an http request result status code is considered successfull
Boolean IsHttpStatusCodeSuccess(Integer _StatusCode) {
	return (_StatusCode >= 200 && _StatusCode < 300);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Create a query string from an array
 * ["key" => "value, "key2" => "value2"]
 * -> "?key=value&key2=value2"
 * ["key3" => ["valueA", "valueB"], "key4" => ["valueC"]]
 * -> "?key3[]=valueA&key3[]=valueB&key4[]=valueC"
 */
Text CreateQueryString(Text[Text] _Query, Text[][Text] _QueryArray) {
	declare Text QueryString = "";

	for (Key => Value in _Query) {
		if (QueryString == "") QueryString ^= "?";
		else QueryString ^= "&";
		QueryString ^= TL::URLEncode(Key)^"="^TL::URLEncode(Value);
	}
	for (Key => Values in _QueryArray) {
		for (Value in Values) {
			if (QueryString == "") QueryString ^= "?";
			else QueryString ^= "&";
			QueryString ^= TL::URLEncode(Key)^"[]="^TL::URLEncode(Value);
		}
	}

	return QueryString;
}
Text CreateQueryString(Text[Text] _Query) {
	return CreateQueryString(_Query, []);
}
Text CreateQueryString(Text[][Text] _QueryArray) {
	return CreateQueryString([], _QueryArray);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Create an headers string from an array
 * ["key" => "value", "key2" => "value2"]
 * -> "key: value\nkey2: value2"
 */
Text CreateHeadersString(Text _AppendTo, Text[Text] _Headers) {
	declare Text HeadersString = _AppendTo;

	foreach (Key => Value in _Headers) {
		if (HeadersString != "") HeadersString ^= "\n";
		HeadersString ^= Key^": "^Value;
	}

	return HeadersString;
}
Text CreateHeadersString(Text[Text] _Headers) {
	return CreateHeadersString("", _Headers);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Inject the given parameters into the route
 * _Route => "/path/:Param1/to/:Param2"
 * _Parameters => ["Param1" => "aaa", "Param2" => "bbb"]
 * => "/path/aaa/to/bbb"
 */
Text InjectRouteParameters(Text _Route, Text[Text] _Parameters) {
	declare Text Route = _Route;
	foreach (Key => Value in _Parameters) {
		Route = TL::Replace(Route, ":"^Key, Value);
	}
	return Route;
}

/// Destroy a request
K_Request Destroy(CHttpManager _HttpMgr, K_Request _Request) {
	if (
		_Request.Status == C_Status_Null ||
		_Request.Status == C_Status_Destroyed
	) {
		return _Request;
	}

	declare K_Request Request = _Request;

	if (Request.HttpRequestId != NullId && _HttpMgr.Requests.existskey(Request.HttpRequestId)) {
		_HttpMgr.Destroy(_HttpMgr.Requests[Request.HttpRequestId]);
	}

	Request.Status = C_Status_Destroyed;

	return Request;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get letter value for the hash function
Integer Private_GetLetterValue(Text _Letter) {
	if (C_HashConversionTable.exists(TL::ToLowerCase(_Letter))) {
		return C_HashConversionTable.keyof(TL::ToLowerCase(_Letter)) + 1;
	}
	return C_HashConversionTable.keyof("") + 1;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Combine letters for the hash function
Text Private_CombineLetters(Text _Letter1, Text _Letter2) {
	return C_HashConversionTable.get((Private_GetLetterValue(_Letter1) + Private_GetLetterValue(_Letter2)) % (C_HashConversionTable.count + 1), "");
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// `TL::SubText()` crash the game if `_Start` is after the end of the `_Text`
Text Private_SafeSubText(Text _Text, Integer _Start, Integer _Length) {
	if (_Start >= TL::Length(_Text)) return "";
	return TL::SubText(_Text, _Start, _Length);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get an identifier for similar requests
Text Private_GetRequestHash(Text _Url, Text _Data, Text _File, Text _Headers) {
	// Hash the headers to avoid leaking an authentication token to other script contexts
	// Collision is not a huge problem, so keep it simple
	declare Text HeadersHash = "";
	declare Integer HeadersLength = TL::Length(_Headers);
	declare Integer Start = 0;
	declare Integer WordLength = 128;
	while (Start < HeadersLength) {
		declare Text Word = Private_SafeSubText(_Headers, Start, WordLength);
		if (HeadersHash == "") {
			HeadersHash = Word;
		} else {
			declare Text CombinedHeadersHash = "";
			for (I, 0, WordLength - 1) {
				CombinedHeadersHash ^= Private_CombineLetters(
					Private_SafeSubText(Word, I, 1),
					Private_SafeSubText(HeadersHash, I, 1)
				);
			}
			HeadersHash = CombinedHeadersHash;
		}
		Start += WordLength;
	}

	return TL::Join("|", [_Url, _Data, _File, HeadersHash]);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Removed the passed timers
Integer[Text] Private_CleanDebounceTimers(Integer[Text] _DebounceTimers, Integer _MillisecondsSinceInit) {
	declare Integer[Text] DebounceTimers;
	for (RequestHash => Timer in _DebounceTimers) {
		if (Timer >= _MillisecondsSinceInit) {
			DebounceTimers[RequestHash] = Timer;
		}
	}
	return DebounceTimers;
}

/// Create a new request
K_Request Private_Create(
	CSystemPlatform _System,
	CHttpManager _HttpMgr,
	Text _Url,
	K_RequestOptions _RequestOptions,
	Boolean _IgnoreDebounce
) {
	declare K_Request Request = C_Request_Null;
	declare Boolean CMGame_UtilsHttp2_OfflineModeEnabled for _System = C_Default_OfflineModeEnabled;
	declare Boolean IsLocalFileSystemRequest = TL::StartsWith("file://", _Url);

	Request.Url = _Url;
	Request.Options = _RequestOptions;
	Request.Promise = Promise::Create();

	if (CMGame_UtilsHttp2_OfflineModeEnabled && !IsLocalFileSystemRequest) {
		Request.Status = C_Status_Destroyed;
		Request.Promise = Promise::Reject(Request.Promise, Error::GetError(Error::C_Type_Http, Error::C_Code_Http_OfflineModeEnabled));
	} else if (IsSaturated(_HttpMgr)) {
		Request.Status = C_Status_WaitingSlot;
	} else {
		// Setup headers
		declare Text DefaultHeaders;
		if (!_RequestOptions.DisableAutomaticAuthorizationHeader) {
			DefaultHeaders ^= HttpAuthorizationHeader::GetLive();
		}
		declare Text Headers = CreateHeadersString(DefaultHeaders, _RequestOptions.Headers);

		/** Debounce system
		 * To prevent the same source from spamming the same request in short intervals,
		 * delay it for a small amount of time. This way, if the same source tries
		 * to create the same request, the previous request that was still waiting in the debounce queue
		 * will be destroyed before it is started. Note, however, that there is no batching of debounced requests.
		 * This means that once the debounce timer is reached, all debounced requests in the queue will fire at once.
		 * This is important to avoid errors with non-idempotent requests (typically post requests)
		 */
		declare Integer[Text] CMGame_UtilsHttp2_DebounceTimers for _System = [];
		declare Integer MillisecondsSinceInit = TiL::GetMillisecondsSinceInit();
		declare Text RequestHash = Private_GetRequestHash(_Url, _RequestOptions.Data, _RequestOptions.File, Headers);
		if (!_IgnoreDebounce && !IsLocalFileSystemRequest) {
			CMGame_UtilsHttp2_DebounceTimers = Private_CleanDebounceTimers(CMGame_UtilsHttp2_DebounceTimers, MillisecondsSinceInit);
			if (CMGame_UtilsHttp2_DebounceTimers.existskey(RequestHash)) {
				Request.Status = C_Status_WaitingDebounce;
				Request.DebounceTimer = CMGame_UtilsHttp2_DebounceTimers[RequestHash];
			}
		}

		if (Request.Status != C_Status_WaitingDebounce) {
			CMGame_UtilsHttp2_DebounceTimers[RequestHash] = MillisecondsSinceInit + C_DebounceDuration;

			// Create the http request
			declare CHttpRequest HttpRequest;
			if (_RequestOptions.FakeResponse.Enabled) {
				Request.IsFake = True;
				Request.ResponseDelayEndTime = Now + _RequestOptions.FakeResponse.Delay;
				Request.Result = _RequestOptions.FakeResponse.Result;
				Request.ResponseStatusCode = _RequestOptions.FakeResponse.StatusCode;
			} else {
				switch (_RequestOptions.Method) {
					case C_Method_Get: {
						HttpRequest <=> _HttpMgr.CreateGet(_Url, C_EnableCppCache, Headers);
					}
					case C_Method_Post: {
						assert(
							_RequestOptions.Data == "" || _RequestOptions.File == "",
							"""[Http@2] Post requests may use either `Data` or `File`, not both"""
						);
						if (_RequestOptions.File != "") {
							HttpRequest <=> _HttpMgr.CreatePostFile(_Url, _RequestOptions.File, Headers);
						} else {
							HttpRequest <=> _HttpMgr.CreatePost(_Url, _RequestOptions.Data, Headers);
						}
					}
					case C_Method_Put: {
						HttpRequest <=> _HttpMgr.CreatePut(_Url, _RequestOptions.Data, Headers);
					}
				}
			}

			// Setup the request struct
			if (HttpRequest != Null) {
				Request.HttpRequestId = HttpRequest.Id;
			}
			Request.Status = C_Status_Processing;
		}
	}

	return Request;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the complete status of the task
Integer Private_GetCompleteStatus(Boolean _IsExternalUpdate) {
	if (_IsExternalUpdate) return C_Status_Complete;
	return C_Status_WaitingExternalComplete;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update a request
K_Request Private_Update(CSystemPlatform _System, CHttpManager _HttpMgr, K_Request _Request, Boolean _IsExternalUpdate) {
	if (
		_Request.Status != C_Status_WaitingSlot &&
		_Request.Status != C_Status_WaitingDebounce &&
		_Request.Status != C_Status_Processing &&
		_Request.Status != C_Status_WaitingExternalComplete
	) {
		return _Request;
	}

	declare K_Request Request = _Request;

	switch (Request.Status) {
		case C_Status_WaitingSlot: {
			if (!IsSaturated(_HttpMgr)) {
				declare Text[Text] Metadata = GetMetadata(Request);
				Request = Destroy(_HttpMgr, Request);
				Request = Private_Create(_System, _HttpMgr, Request.Url, Request.Options, False);
				Request = SetMetadata(Request, Metadata);
			}
		}
		case C_Status_WaitingDebounce: {
			if (TiL::GetMillisecondsSinceInit() >= Request.DebounceTimer) {
				declare Text[Text] Metadata = GetMetadata(Request);
				Request = Destroy(_HttpMgr, Request);
				Request = Private_Create(_System, _HttpMgr, Request.Url, Request.Options, True);
				Request = SetMetadata(Request, Metadata);
			}
		}
		case C_Status_Processing: {
			if (Request.IsFake) {
				if (Now >= Request.ResponseDelayEndTime) {
					Request.Status = Private_GetCompleteStatus(_IsExternalUpdate);
					if (IsHttpStatusCodeSuccess(Request.ResponseStatusCode)) {
						Request.Promise = Promise::Resolve(Request.Promise);
					} else {
						Request.Promise = Promise::Reject(Request.Promise, Error::GetError(Error::C_Type_Http, ""^Request.ResponseStatusCode));
					}
				}
			} else {
				declare CHttpRequest HttpRequest <=> _HttpMgr.Requests.get(Request.HttpRequestId, Null);
				if (HttpRequest == Null) {
					Request.Status = Private_GetCompleteStatus(_IsExternalUpdate);
					Request.Promise = Promise::Reject(Request.Promise, Error::GetError(Error::C_Type_Http, Error::C_Code_Http_RequestNotFound));
				} else if (HttpRequest.IsCompleted) {
					Request.Status = Private_GetCompleteStatus(_IsExternalUpdate);
					Request.Result = HttpRequest.Result;
					Request.ResponseStatusCode = HttpRequest.StatusCode;
					if (IsHttpStatusCodeSuccess(Request.ResponseStatusCode)) {
						Request.Promise = Promise::Resolve(Request.Promise);
					} else {
						Request.Promise = Promise::Reject(Request.Promise, Error::GetError(Error::C_Type_Http, ""^Request.ResponseStatusCode));
					}
				}
			}
		}
		case C_Status_WaitingExternalComplete: {
			if (_IsExternalUpdate) {
				Request.Status = C_Status_Complete;
			}
		}
	}

	return Request;
}
K_Request Update(CSystemPlatform _System, CHttpManager _HttpMgr, K_Request _Request) {
	return Private_Update(_System, _HttpMgr, _Request, True);
}

/// Destroy a previous request and create a new one
K_Request DestroyAndCreate(CHttpManager _HttpMgr, K_Request _PreviousRequest, K_Request _NewRequest) {
	Destroy(_HttpMgr, _PreviousRequest);

}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the request is processing
Boolean IsProcessing(K_Request _Request) {
	return (
		_Request.Status == C_Status_WaitingSlot ||
		_Request.Status == C_Status_WaitingDebounce ||
		_Request.Status == C_Status_Processing ||
		_Request.Status == C_Status_WaitingExternalComplete
	);
}
