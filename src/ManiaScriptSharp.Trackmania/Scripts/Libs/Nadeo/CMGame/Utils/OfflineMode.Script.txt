/**
 * Offline mode management for the title
 */
#Const Version "1.1.0"
#Const ScriptName "Libs/Nadeo/CMGame/Utils/OfflineMode.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Utils/Http@2.Script.txt" as Http2
#Include "Libs/Nadeo/CMGame/Utils/Http@3.Script.txt" as Http3

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constant
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Default_IsEnabled False

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Enable or disable the offline mode
Void Enable(CSystemPlatform _System, Boolean _IsEnabled) {
	declare Boolean CMGame_UtilsOfflineMode_IsEnabled for _System = C_Default_IsEnabled;
	CMGame_UtilsOfflineMode_IsEnabled = _IsEnabled;

	Http2::Protected_SetOfflineMode(_System, CMGame_UtilsOfflineMode_IsEnabled);
	Http3::Protected_SetOfflineMode(_System, CMGame_UtilsOfflineMode_IsEnabled);
}
Void Enable(CSystemPlatform _System) {
	Enable(_System, True);
}
Void Disable(CSystemPlatform _System) {
	Enable(_System, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the offline mode is enabled
Boolean IsEnabled(CSystemPlatform _System) {
	declare Boolean CMGame_UtilsOfflineMode_IsEnabled for _System = C_Default_IsEnabled;
	return CMGame_UtilsOfflineMode_IsEnabled;
}
