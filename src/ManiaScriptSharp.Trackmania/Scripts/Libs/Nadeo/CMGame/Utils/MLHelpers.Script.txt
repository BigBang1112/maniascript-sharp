/// Convert a Vec2 to a `pos` attribute
Text Pos(Vec2 _Pos) {
	return "pos=\""^_Pos.X^" "^_Pos.Y^"\"";
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Convert a Vec3 to a `pos` and `z-index` attribute
Text Pos(Vec3 _Pos) {
	return "pos=\""^_Pos.X^" "^_Pos.Y^"\" z-index=\""^_Pos.Z^"\"";
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Convert a Vec2 to a `size` attribute
Text Size(Vec2 _Size) {
	return "size=\""^_Size.X^" "^_Size.Y^"\"";
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Enable or disable translation on a label
Text Translate(Boolean _IsTranslated) {
	if (_IsTranslated) {
		return "translate=\"1\" data-"^C_Private_DataAttributeTranslate^"=\"1\"";
	}
	return "translate=\"0\" data-"^C_Private_DataAttributeTranslate^"=\"0\"";
}

/// Skew position on X axis
Real SkewX(Real _OffsetY, Real _Angle) {
	return _OffsetY * ML::Tan(ML::DegToRad(_Angle));
}
Real SkewX(Real _PosX, Real _PosY, Real _Angle) {
	return _PosX + SkewX(_PosY, _Angle);
}
Vec2 SkewX(Vec2 _Pos, Real _Angle) {
	return <_Pos.X + SkewX(_Pos.Y, _Angle), _Pos.Y>;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Skew position on X axis and keep Y pos
 *
 *	@return 	The skewed x with the untouched y for ML 'pos' parameter
 *	@usage		pos="{{{MLHelpers::Skew(_PosX, _PosY, _Angle)}}}"
 */
Text Skew(Real _PosX, Real _PosY, Real _Angle) {
	return SkewX(_PosX, _PosY, _Angle)^" "^_PosY;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Keep the original ratio of the asset depending on the horizontal or vertical size
 *
 *	@return 	The X and Y size depending on the asset ratio and the given size parameter
 */
Vec2 KeepRatioFromX(Real _AssetSizeX, Real _AssetSizeY, Real _SizeX) {
	return <_SizeX, _SizeX*_AssetSizeY/_AssetSizeX>;
}
Vec2 KeepRatioFromX(Vec2 _AssetSize, Real _SizeX) {
	return <_SizeX, _SizeX*_AssetSize.Y/_AssetSize.X>;
}
Vec2 KeepRatioFromY(Real _AssetSizeX, Real _AssetSizeY, Real _SizeY) {
	return <_SizeY*_AssetSizeX/_AssetSizeY, _SizeY>;
}
Vec2 KeepRatioFromY(Vec2 _AssetSize, Real _SizeY) {
	return <_SizeY*_AssetSize.X/_AssetSize.Y, _SizeY>;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Keep the original ratio of the asset depending on the horizontal or vertical size
 *
 *	@return 	The ML formatted text with X and Y size depending on the asset ratio and the given size parameter
 *	@usage		size="{{{MLHelpers::KeepRatioFromX(_AssetSizeX, _AssetSizeY, _SizeX)}}}"
 */
Text MLKeepRatioFromX(Real _AssetSizeX, Real _AssetSizeY, Real _SizeX) {
	declare Vec2 SizeWithRatio = KeepRatioFromX(_AssetSizeX, _AssetSizeY, _SizeX);
	return SizeWithRatio.X^" "^SizeWithRatio.Y;
}
Text MLKeepRatioFromX(Vec2 _AssetSize, Real _SizeX) {
	declare Vec2 SizeWithRatio = KeepRatioFromX(_AssetSize.X, _AssetSize.Y, _SizeX);
	return SizeWithRatio.X^" "^SizeWithRatio.Y;
}
Text MLKeepRatioFromY(Real _AssetSizeX, Real _AssetSizeY, Real _SizeY) {
	declare Vec2 SizeWithRatio = KeepRatioFromY(_AssetSizeX, _AssetSizeY, _SizeY);
	return SizeWithRatio.X^" "^SizeWithRatio.Y;
}
Text MLKeepRatioFromY(Vec2 _AssetSize, Real _SizeY) {
	declare Vec2 SizeWithRatio = KeepRatioFromY(_AssetSize.X, _AssetSize.Y, _SizeY);
	return SizeWithRatio.X^" "^SizeWithRatio.Y;
}

/// Get all controls that support the opacity parameter in a frame
CMlControl[] GetControlsWithOpacitySupport(CMlFrame _Frame) {
	declare CMlControl[] ControlsWithOpacitySupport;
	foreach (Control in _Frame.Controls) {
		switchtype (Control) {
			case CMlEntry: {
				ControlsWithOpacitySupport.add(Control);
			}
			case CMlFrame: {
				declare CMlControl[] TmpControls = GetControlsWithOpacitySupport(Control as CMlFrame);
				foreach (Control in TmpControls) {
					ControlsWithOpacitySupport.add(Control);
				}
			}
			case CMlLabel: {
				ControlsWithOpacitySupport.add(Control);
			}
			case CMlQuad: {
				ControlsWithOpacitySupport.add(Control);
			}
			case CMlTextEdit: {
				ControlsWithOpacitySupport.add(Control);
			}
		}
	}
	return ControlsWithOpacitySupport;
}

/// Set the color of a quad taking its color mode into account
Void SetQuadColor(CMlQuad _Quad, Integer _ColorMode, Vec3 _Color) {
	// Initialize all properties properly here first
	// Modifying one of them afterward can also have an effect on the others.
	// eg: modifying `ModulateColor` will also change the value of `Colorize`.
	_Quad.Colorize = C_DefaultColorize;
	_Quad.ModulateColor = C_DefaultModulateColor;
	_Quad.BgColor = C_DefaultBgColor;

	switch (_ColorMode) {
		case C_ColorMode_Colorize: _Quad.Colorize = _Color;
		case C_ColorMode_Modulate: _Quad.ModulateColor = _Color;
		case C_ColorMode_Background: _Quad.BgColor = _Color;
	}
}

/// Get a control position relative to the top-left of the control
Vec2 GetAnchorPosition(Vec2 _Size, CMlControl::AlignHorizontal _HAlign, CMlControl::AlignVertical _VAlign) {
	declare Vec2 Position = <0., 0.>;

	switch (_HAlign) {
		case CMlControl::AlignHorizontal::Left: Position.X = 0.;
		case CMlControl::AlignHorizontal::HCenter: Position.X = _Size.X * 0.5;
		case CMlControl::AlignHorizontal::Right: Position.X = _Size.X * 1.;
	}
	switch (_VAlign) {
		case CMlControl::AlignVertical::Top: Position.Y = 0.;
		case CMlControl::AlignVertical::VCenter: Position.Y = _Size.Y * -0.5;
		case CMlControl::AlignVertical::VCenter2: Position.Y = _Size.Y * -0.5;
		case CMlControl::AlignVertical::Bottom: Position.Y = _Size.Y * -1.;
	}

	return Position;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Set the alignment of a control without changing its HAlign and VAlign properties
 * eg: you have a 50x10 control top-left aligned
 * you want a bottom-right alignment instead without changing its HAlign and VAlign properties
 * This function will move the control to <-50., 10.>
 *
 * 0,0              -50,10
 * @--------        @--------
 * |       |   =>   |       |
 * ---------        ---------
 *    50,-10              0,0
 */
Void AlignContent(CMlControl _Control, Vec2 _Size, CMlControl::AlignHorizontal _HAlign, CMlControl::AlignVertical _VAlign) {
	_Control.RelativePosition_V3 = -GetAnchorPosition(_Size, _HAlign, _VAlign);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Set the alignment of a control but keep its original position
 * eg: you have a 50x10 control top-left aligned
 * you want a bottom-right alignment instead while keeping the control at the same position
 * This function will change the HAlign and VAlign properties of the component and move it to <50., -10.>
 *
 * 0,0              0,0
 * @--------        ---------
 * |       |   =>   |       |
 * ---------        --------@
 *    50,-10           50,-10
 */
Void AlignControl(CMlControl _Control_Align, Vec2 _Size, CMlControl::AlignHorizontal _HAlign, CMlControl::AlignVertical _VAlign) {
	_Control_Align.RelativePosition_V3 = GetAnchorPosition(_Size, _HAlign, _VAlign);
	_Control_Align.HorizontalAlign = _HAlign;
	_Control_Align.VerticalAlign = _VAlign;
}

/// Check if the text wrapping is enabled
Boolean TextWrappingEnabled(CMlLabel _Label_Target) {
	return _Label_Target.AutoNewLine || _Label_Target.MaxLine != -1;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if a label was explicitely marked as translated with the `Translate()` function
Boolean Private_IsLabelPossiblyTranslated(CMlLabel _Label) {
	return (
		!_Label.DataAttributeExists(C_Private_DataAttributeTranslate) ||
		Cast::ToBoolean(_Label.DataAttributeGet(C_Private_DataAttributeTranslate))
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Get the height of the label
 *
 * Unlike `ComputeWidth()`, `ComputeHeight()` does not work with automatic text translation
 * The function returns the height of the original text and not the one of its translation
 * To work around that we must use `TextLib::GetTranslatedText()` to get the translated text
 * and pass it to `ComputeHeight()` instead of the original text
 *
 * At the moment, `CMlLabel::ComputeHeight()` does not return the correct height at first
 * if the label did not have a `textsize="n"` attribute in the manialink xml
 * When the `textsize` attribute is not set in the manialink, then `TextSizeReal = -1` in the script.
 * `TextSizeReal` can be changed after but `ComputeHeight()` will return an incorrect value at first.
 * Only after a `yield` the value will be correct for all future changes.
 * Example:
 * // Label.TextSizeReal == -1.
 * Label.ComputeHeight(); //< wrong value
 * Label.TextSizeReal = 2.;
 * Label.ComputeHeight(); //< wrong value
 * yield;
 * Label.ComputeHeight(); //< right value
 * Label.TextSizeReal = 4.;
 * Label.ComputeHeight(); //< right value
 */
Real ComputeHeight(CMlLabel _Label_Target, Text _Value) {
	if (Private_IsLabelPossiblyTranslated(_Label_Target)) {
		return _Label_Target.ComputeHeight(TL::GetTranslatedText(_Value));
	}
	return _Label_Target.ComputeHeight(_Value);
}
Real ComputeHeight(CMlLabel _Label_Target) {
	return ComputeHeight(_Label_Target, _Label_Target.Value);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Get the width of a label
 *
 * When computing the width a translated text that contains hidden context on a label with disabled translation
 * the hidden context is taken into account in the width computation but not displayed afterward
 * For example computing the width of `_("|HiddenContext|Text to translate")`
 * will return the width of `|HiddenContext|Text to translate` but display `Text to translate`
 */
Real ComputeWidth(CMlLabel _Label_Target, Text _Value) {
	if (Private_IsLabelPossiblyTranslated(_Label_Target)) {
		return _Label_Target.ComputeWidth(_Value);
	}
	if (TL::StartsWith(C_Private_TranslateCharacter^"|", _Value)) {
		declare Text[] MatchList = TL::RegexMatch("""^{{{C_Private_TranslateCharacter}}}\|[^|]*\|(.*)""", _Value, "");
		if (MatchList.count > 0) {
			return _Label_Target.ComputeWidth(MatchList[1], False);
		}
	}
	return _Label_Target.ComputeWidth(_Value, False);
}
Real ComputeWidth(CMlLabel _Label_Target) {
	return ComputeWidth(_Label_Target, _Label_Target.Value);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Check if the text can fit inside the label
 *
 * Text wrapping on the label is enabled if `AutoNewLine = True` or `MaxLine != -1`
 * The number of lines displayed by the label will be limited only if `MaxLine > 0`
 *
 * `ComputeWidth()` behavior:
 * The function will split the text on explicit new line character (`\n`)
 * and return the width of the longest line without taking text wrapping into account
 *
 * `ComputeHeight()` behavior:
 * If text wrapping is disabled the function will return the height of one line of text
 * If text wrapping is enabled the function will return the height of all lines of text, taking text wrapping  into account
 *
 * If text wrapping is enabled, we only need to check the height of the text
 * because the text is wrapped and cannot exceed the label width
 * If text wrapping is disabled, we need to check the width and the height of the text
 */
Boolean IsValueFittingLabel(CMlLabel _Label_Target) {
	return (
		(TextWrappingEnabled(_Label_Target) || ComputeWidth(_Label_Target, _Label_Target.Value) <= _Label_Target.Size.X) &&
		ComputeHeight(_Label_Target, _Label_Target.Value) <= _Label_Target.Size.Y
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Make the label text fit inside the label size without being cut
 * `_TextSizeReal.X` is the initial size
 * `_TextSizeReal.Y` is the target size
 *   if X > Y the text size will decrease until the text fit inside the label
 *   if X < Y the text size will increase as long as the text fit inside the label
 * Automatically calculate the step size by dividing the distance between X and Y by two each time
 */
Void FitLabelValue(CMlLabel _Label_Target, Vec2 _TextSizeReal, Integer _StepsNb) {
	_Label_Target.TextSizeReal = _TextSizeReal.X;

	// Stop after setting the initial size if:
	// - the target size is identical to the initial size
	// - we will reduce the text size but the value is already fitting label
	// - we will increase the text size but the value is already not fitting the label
	if (
		_TextSizeReal.X == _TextSizeReal.Y ||
		(_TextSizeReal.X > _TextSizeReal.Y && IsValueFittingLabel(_Label_Target)) ||
		(_TextSizeReal.X < _TextSizeReal.Y && !IsValueFittingLabel(_Label_Target))
	) return;

	// Divide the step size by 2 after each step
	// Save the last known fitting text size during the process
	declare Integer StepsNb = 0;
	declare Real StepSize = ML::Abs(_TextSizeReal.X - _TextSizeReal.Y) / 2.;
	declare Real FittingTextSizeReal = ML::Min(_TextSizeReal.X, _TextSizeReal.Y);
	while (StepsNb < _StepsNb) {
		if (IsValueFittingLabel(_Label_Target)) {
			FittingTextSizeReal = ML::Max(_Label_Target.TextSizeReal, FittingTextSizeReal);
			_Label_Target.TextSizeReal += StepSize;
		} else {
			_Label_Target.TextSizeReal -= StepSize;
		}
		StepsNb += 1;
		StepSize /= 2.;
	}
	// Keep the biggest fitting text size
	if (!IsValueFittingLabel(_Label_Target) || FittingTextSizeReal > _Label_Target.TextSizeReal) {
		_Label_Target.TextSizeReal = FittingTextSizeReal;
	}
}
Void FitLabelValue(CMlLabel _Label_Target, Vec2 _TextSizeReal) {
	FitLabelValue(_Label_Target, _TextSizeReal, C_Private_FitLabelAutoStepsNb);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Make the label text fit inside the label size without being cut
 * `_TextSizeReal.X` is the initial size
 * `_TextSizeReal.Y` is the target size
 *   if X > Y the text size will decrease until the text fit inside the label
 *   if X < Y the text size will increase as long as the text fit inside the label
 * `_TextSizeReal.Z` is the step size
 */
Void FitLabelValue(CMlLabel _Label_Target, Vec3 _TextSizeReal) {
	if (_TextSizeReal.Z <= 0.) {
		return FitLabelValue(_Label_Target, <_TextSizeReal.X, _TextSizeReal.Y>);
	}

	_Label_Target.TextSizeReal = _TextSizeReal.X;

	// Stop after setting the initial size if:
	// - the target size is identical to the initial size
	// - we will reduce the text size but the value is already fitting label
	// - we will increase the text size but the value is already not fitting the label
	if (
		_TextSizeReal.X == _TextSizeReal.Y ||
		(_TextSizeReal.X > _TextSizeReal.Y && IsValueFittingLabel(_Label_Target)) ||
		(_TextSizeReal.X < _TextSizeReal.Y && !IsValueFittingLabel(_Label_Target))
	) return;

	// Maximum number of steps warning
	declare Real MaxSteps = ML::Abs(_TextSizeReal.X - _TextSizeReal.Y) / _TextSizeReal.Z;
	if (MaxSteps > C_Private_FitLabelMaxSteps) {
		log("""WARNING [MLHelpers] FitLabelValue() too many steps: {{{MaxSteps}}} > {{{C_Private_FitLabelMaxSteps}}} | TextSizeReal: {{{_TextSizeReal}}} | Label id: '{{{_Label_Target.ControlId}}}""");
	}

	if (_TextSizeReal.X > _TextSizeReal.Y) {
		while (_Label_Target.TextSizeReal > _TextSizeReal.Y && !IsValueFittingLabel(_Label_Target)) {
			_Label_Target.TextSizeReal = ML::Max(_Label_Target.TextSizeReal - _TextSizeReal.Z, _TextSizeReal.Y);
		}
	} else {
		declare Real FittingTextSizeReal = _TextSizeReal.Y;
		while (_Label_Target.TextSizeReal < _TextSizeReal.Y && IsValueFittingLabel(_Label_Target)) {
			FittingTextSizeReal = _Label_Target.TextSizeReal;
			_Label_Target.TextSizeReal = ML::Min(_Label_Target.TextSizeReal + _TextSizeReal.Z, _TextSizeReal.Y);
		}
		if (!IsValueFittingLabel(_Label_Target)) {
			_Label_Target.TextSizeReal = FittingTextSizeReal;
		}
	}
}

/// Compute the width of a label value taking the label size.x into account
Real ComputeConstrainedLabelValueWidth(CMlLabel _Label, Text _Value, Boolean _Translated) {
	return ML::Min(_Label.ComputeWidth(_Value, _Translated), _Label.Size.X);
}
Real ComputeConstrainedLabelValueWidth(CMlLabel _Label, Boolean _Translated) {
	return ComputeConstrainedLabelValueWidth(_Label, _Label.Value, _Translated);
}
Real ComputeConstrainedLabelValueWidth(CMlLabel _Label, Text _Value) {
	return ML::Min(ComputeWidth(_Label, _Value), _Label.Size.X);
}
Real ComputeConstrainedLabelValueWidth(CMlLabel _Label) {
	return ComputeConstrainedLabelValueWidth(_Label, _Label.Value);
}

/// Create an xml node
Text CreateXmlNode(Text _Name, Text[Text] _Attributes) {
	declare Text Attributes = "";
	for (AttributeName => AttributeValue in _Attributes) {
		Attributes ^= """{{{TL::MLEncode(AttributeName)}}}="{{{TL::MLEncode(AttributeValue)}}}" """;
	}
	return """<{{{TL::MLEncode(_Name)}}} {{{Attributes}}}/>""";
}
