#Const C_Collection_Stadium "Stadium"
#Const C_Collection_BlueBay "BlueBay"
#Const C_Collection_GreenCoast "GreenCoast"
#Const C_Collection_RedIsland "RedIsland"
#Const C_Collection_WhiteShore "WhiteShore"

#Const C_PlayableCollectionList [
	"Stadium",
	"BlueBay",
	"GreenCoast",
	"RedIsland",
	"WhiteShore"
]

#Const C_VistaName_Stadium "Stadium"
#Const C_VistaName_BlueBay "Blue Bay"
#Const C_VistaName_GreenCoast "Green Coast"
#Const C_VistaName_RedIsland "Red Island"
#Const C_VistaName_WhiteShore "White Shore"

#Const C_Mood_Sunrise "Sunrise"
#Const C_Mood_Day "Day"
#Const C_Mood_Sunset "Sunset"
#Const C_Mood_Night "Night"
#Const C_Mood_StadiumSunrise "48x48Screen155Sunrise"
#Const C_Mood_StadiumDay "48x48Screen155Day"
#Const C_Mood_StadiumSunset "48x48Screen155Sunset"
#Const C_Mood_StadiumNight "48x48Screen155Night"
#Const C_Mood_BlueBaySunrise "Sunrise64"
#Const C_Mood_BlueBayDay "Day64"
#Const C_Mood_BlueBaySunset "Sunset64"
#Const C_Mood_BlueBayNight "Night64"
#Const C_Mood_GreenCoastSunrise "Sunrise64"
#Const C_Mood_GreenCoastDay "Day64"
#Const C_Mood_GreenCoastSunset "Sunset64"
#Const C_Mood_GreenCoastNight "Night64"

#Const C_SkinType_CarSport "Models/CarSport"

#Const C_PlayerModel_CarSport "CarSport"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
Text GetCollectionMood(Text _Collection, Text _Mood) {
	switch (_Collection) {
		case C_Collection_Stadium: {
			switch (_Mood) {
				case C_Mood_Sunrise: return C_Mood_StadiumSunrise;
				case C_Mood_Day: return C_Mood_StadiumDay;
				case C_Mood_Sunset: return C_Mood_StadiumSunset;
				case C_Mood_Night: return C_Mood_StadiumNight;
				default: return C_Mood_StadiumDay;
			}
		}
		case C_Collection_BlueBay: {
			switch (_Mood) {
				case C_Mood_Sunrise: return C_Mood_BlueBaySunrise;
				case C_Mood_Day: return C_Mood_BlueBayDay;
				case C_Mood_Sunset: return C_Mood_BlueBaySunset;
				case C_Mood_Night: return C_Mood_BlueBayNight;
				default: return C_Mood_BlueBayDay;
			}
		}
		case C_Collection_GreenCoast: {
			switch (_Mood) {
				case C_Mood_Sunrise: return C_Mood_GreenCoastSunrise;
				case C_Mood_Day: return C_Mood_GreenCoastDay;
				case C_Mood_Sunset: return C_Mood_GreenCoastSunset;
				case C_Mood_Night: return C_Mood_GreenCoastNight;
				default: return C_Mood_GreenCoastDay;
			}
		}
	}

	switch (_Mood) {
		case C_Mood_Sunrise: return C_Mood_Sunrise;
		case C_Mood_Day: return C_Mood_Day;
		case C_Mood_Sunset: return C_Mood_Sunset;
		case C_Mood_Night: return C_Mood_Night;
	}
	return C_Mood_Day;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the given collection is playable in Trackmania
Boolean IsPlayable(Text _Collection) {
	return C_PlayableCollectionList.exists(_Collection);
}
