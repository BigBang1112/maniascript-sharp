/**
 * Utility functions related to script contexts
 */
#Const Version "1.0.0"
#Const ScriptName "Libs/Nadeo/CMGame/Utils/Context.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the parent mania app instance from a manialink script
CManiaApp GetParentApp(CMlScript _This) {
	switchtype (_This as Script) {
		case CAnyEditorPluginLayer: return Null;
		case CManiaAppTitleLayer: return Script.ParentApp;
		case CMapEditorPluginLayer: return Script.Editor;
		case CMlBrowser: return Script.ParentApp;
		case CMlManiaPlanet: return Script.ParentApp;
		case CMlScriptIngame: return Script.ParentApp;
		case CMlStation: return Script.ParentApp;
		default: assert(False, "Context not supported");
	}
	return Null;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the http manager instance for the given context
CHttpManager GetHttpManager(CNod _Context) {
	switchtype (_Context as Context) {
		case CManiaApp: return Context.Http;
		case CMlScript: return Context.Http;
		case CMode: return Context.Http;
		case CServerPlugin: return Context.Http;
		default: assert(False, "Context not supported");
	}
	return Null;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the system instance for the given context
CSystemPlatform GetSystem(CNod _Context) {
	switchtype (_Context as Context) {
		case CManiaApp: return Context.System;
		case CManiaPlanet: return Context.System;
		case CMlScript: return Context.System;
		case CMode: return Context.System;
		case CServerPlugin: return Context.System;
		default: assert(False, "Context not supported");
	}
	return Null;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the maniaplanet instance for the given context
CManiaPlanet GetManiaPlanet(CNod _Context) {
	switchtype (_Context as Context) {
		case CManiaApp: return Context.ManiaPlanet;
		case CManiaAppTitleLayer: return Context.ParentApp.ManiaPlanet;
		case CMapEditorPluginLayer: return Context.Editor.ManiaPlanet;
		case CMlBrowser: return Context.ManiaPlanet;
		case CMlManiaPlanet: return Context.ManiaPlanet;
		case CMlScriptIngame: return Context.ParentApp.ManiaPlanet;
		case CMlStation: return Context.ManiaPlanet;
		default: assert(False, "Context not supported");
	}
	return Null;
}
