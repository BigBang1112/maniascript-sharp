/// Initialize the style list of a control
Text[] Private_InitializeStyleList(CMlControl _Control) {
	declare Text[] StyleList;
	if (_Control.DataAttributeExists(C_Private_DataAttribute_StyleList)) {
		StyleList = TL::Split(C_Private_Seperator_StyleList, _Control.DataAttributeGet(C_Private_DataAttribute_StyleList));
	}
	return StyleList;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize the disabled property list of a control
Text[] Private_InitializeDisabledPropertyList(CMlControl _Control) {
	declare Text[] DisabledPropertyList;
	if (_Control.DataAttributeExists(C_Private_DataAttribute_DisabledPropertyList)) {
		DisabledPropertyList = TL::Split(C_Private_Seperator_DisabledPropertyList, _Control.DataAttributeGet(C_Private_DataAttribute_DisabledPropertyList));
	}
	return DisabledPropertyList;
}

/// Create a style

	G_Stylesheet[_Style] = _PropertyList;

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the merge behavior of a control property
Void SetPropertyListMergeBehavior(CMlControl _Control, Integer[Text] _PropertyListMergeBehavior) {
	declare Integer[Text] CMGame_UtilsStylesheet4_PropertyListMergeBehavior for _Control = [];
	CMGame_UtilsStylesheet4_PropertyListMergeBehavior = _PropertyListMergeBehavior;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the properties applied on a control
Text[Text] GetPropertyList(CMlControl _Control) {
	if (G_Stylesheet.count <= 0) return [];

	declare Text[] CMGame_UtilsStylesheet4_StyleList for _Control = Private_InitializeStyleList(_Control);
	if (CMGame_UtilsStylesheet4_StyleList.count <= 0) return [];

	declare Text[] CMGame_UtilsStylesheet4_DisabledPropertyList for _Control = Private_InitializeDisabledPropertyList(_Control);
	declare Integer[Text] CMGame_UtilsStylesheet4_PropertyListMergeBehavior for _Control = [];

	declare Text[Text] PropertyList;
	for (Style in CMGame_UtilsStylesheet4_StyleList) {
		declare Text[Text] StylePropertyList = G_Stylesheet.get(Style, []);
		for (PropertyName => PropertyValue in StylePropertyList) {
			if (
				CMGame_UtilsStylesheet4_DisabledPropertyList.count == 0 ||
				!CMGame_UtilsStylesheet4_DisabledPropertyList.exists(PropertyName)
			) {
				switch (CMGame_UtilsStylesheet4_PropertyListMergeBehavior.get(PropertyName, C_MergeBehavior_Overwrite)) {
					case C_MergeBehavior_MergeWithSpace: {
						declare Text PreviousPropertyValue = PropertyList.get(PropertyName, "");
						if (PreviousPropertyValue == "") {
							PropertyList[PropertyName] = PropertyValue;
						} else {
							PropertyList[PropertyName] ^= " "^PropertyValue;
						}
					}
					default: { //< C_MergeBehavior_Overwrite
						PropertyList[PropertyName] = PropertyValue;
					}
				}
			}
		}
	}

	return PropertyList;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Apply a property list on a control
Void ApplyPropertyList(CMlControl _Control, Text[Text] _ControlProperties) {
	for (PropertyName => PropertyValue in _ControlProperties) {
		switch (PropertyName) {
			case C_Prop_Size: _Control.Size = Cast::ToVec2(PropertyValue);
			case C_Prop_HAlign: _Control.HorizontalAlign = Cast::ToHAlign(PropertyValue);
			case C_Prop_VAlign: _Control.VerticalAlign = Cast::ToVAlign(PropertyValue);
			case C_Prop_Hidden: _Control.Visible = !Cast::ToBoolean(PropertyValue);
			case C_Prop_Pos: _Control.RelativePosition_V3 = Cast::ToVec2(PropertyValue);
			case C_Prop_ZIndex: _Control.ZIndex = TL::ToReal(PropertyValue);
			case C_Prop_Scale: _Control.RelativeScale = TL::ToReal(PropertyValue);
			case C_Prop_Rotation: _Control.RelativeRotation = TL::ToReal(PropertyValue);
			default: {
				switchtype (_Control as TypedControl) {
					case CMlLabel: {
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_Style: TypedControl.Style = PropertyValue;
							case C_Prop_Substyle: TypedControl.Substyle = PropertyValue;
							case C_Prop_Blend: TypedControl.Blend = Cast::ToLabelBlendMode(PropertyValue);
							case C_Prop_TTSAltText: TypedControl.TTS_AltText = PropertyValue;
							case C_Prop_TTSAltTextTranslate: TypedControl.TTS_AltText_Translate = Cast::ToBoolean(PropertyValue);
							case C_Prop_TextFont: TypedControl.TextFont = PropertyValue;
							case C_Prop_TextValue: TypedControl.Value = PropertyValue;
							case C_Prop_TextMaxLine: TypedControl.MaxLine = TL::ToInteger(PropertyValue);
							case C_Prop_TextLineSpacing: TypedControl.LineSpacing = TL::ToReal(PropertyValue);
							case C_Prop_TextItalicSlope: TypedControl.ItalicSlope = TL::ToReal(PropertyValue);
							case C_Prop_TextAppendEllipsis: TypedControl.AppendEllipsis = Cast::ToBoolean(PropertyValue);
							case C_Prop_TextAutoNewLine: TypedControl.AutoNewLine = Cast::ToBoolean(PropertyValue);
							case C_Prop_TextPrefix: TypedControl.TextPrefix = PropertyValue;
							case C_Prop_TextColor: TypedControl.TextColor = CL::HexToRgb(PropertyValue);
							case C_Prop_TextSize: TypedControl.TextSizeReal = TL::ToReal(PropertyValue);
							case C_Prop_TextFilterProfanities: TypedControl.FilterProfanities = Cast::ToFilterProfanities(PropertyValue);
						}
					}
					case CMlQuad: {
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_Style: TypedControl.Style = PropertyValue;
							case C_Prop_Substyle: TypedControl.Substyle = PropertyValue;
							case C_Prop_Blend: TypedControl.Blend = Cast::ToQuadBlendMode(PropertyValue);
							case C_Prop_TTSAltText: TypedControl.TTS_AltText = PropertyValue;
							case C_Prop_TTSAltTextTranslate: TypedControl.TTS_AltText_Translate = Cast::ToBoolean(PropertyValue);
							case C_Prop_QuadImageUrl: TypedControl.ImageUrl = PropertyValue;
							case C_Prop_QuadImageUrlFocus: TypedControl.ImageUrlFocus = PropertyValue;
							case C_Prop_QuadAlphaMaskUrl: TypedControl.AlphaMaskUrl = PropertyValue;
							case C_Prop_QuadSuperSample: TypedControl.SuperSample = Cast::ToBoolean(PropertyValue);
							case C_Prop_QuadColorize: TypedControl.Colorize = CL::HexToRgb(PropertyValue);
							case C_Prop_QuadModulateColor: TypedControl.ModulateColor = CL::HexToRgb(PropertyValue);
							case C_Prop_QuadBgColor: TypedControl.BgColor = CL::HexToRgb(PropertyValue);
							case C_Prop_QuadBgColorFocus: TypedControl.BgColorFocus = CL::HexToRgb(PropertyValue);
							case C_Prop_QuadKeepRatio: TypedControl.KeepRatio = Cast::ToKeepRatioMode(PropertyValue);
						}
					}
					case CMlFileEntry: { //< Must be above CMlEntry
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_TextValue: TypedControl.Value = PropertyValue;
							case C_Prop_TextFormat: TypedControl.TextFormat = Cast::ToEntryFormat(PropertyValue);
							case C_Prop_TextColor: TypedControl.TextColor = CL::HexToRgb(PropertyValue);
							case C_Prop_TextSize: TypedControl.TextSizeReal = TL::ToReal(PropertyValue);
							case C_Prop_TextMaxLength: TypedControl.MaxLength = TL::ToInteger(PropertyValue);
						}
					}
					case CMlEntry: {
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_TextValue: TypedControl.Value = PropertyValue;
							case C_Prop_TextFormat: TypedControl.TextFormat = Cast::ToEntryFormat(PropertyValue);
							case C_Prop_TextColor: TypedControl.TextColor = CL::HexToRgb(PropertyValue);
							case C_Prop_TextSize: TypedControl.TextSizeReal = TL::ToReal(PropertyValue);
							case C_Prop_TextMaxLength: TypedControl.MaxLength = TL::ToInteger(PropertyValue);
						}
					}
					case CMlTextEdit: {
						switch (PropertyName) {
							case C_Prop_Opacity: TypedControl.Opacity = TL::ToReal(PropertyValue);
							case C_Prop_TextValue: TypedControl.Value = PropertyValue;
							case C_Prop_TextMaxLine: TypedControl.MaxLine = TL::ToInteger(PropertyValue);
							case C_Prop_TextAutoNewLine: TypedControl.AutoNewLine = Cast::ToBoolean(PropertyValue);
							case C_Prop_TextLineSpacing: TypedControl.LineSpacing = TL::ToReal(PropertyValue);
							case C_Prop_TextColor: TypedControl.TextColor = CL::HexToRgb(PropertyValue);
							case C_Prop_TextSize: TypedControl.TextSizeReal = TL::ToReal(PropertyValue);
							case C_Prop_TextFormat: TypedControl.TextFormat = Cast::ToTextEditFormat(PropertyValue);
						}
					}
				}
			}
		}
	}

	// The text size fit property must be applied after all the others because its result depends on them
	if (_Control is CMlLabel && _ControlProperties.existskey(C_Prop_TextSizeFit)) {
		MLHelpers::FitLabelValue((_Control as CMlLabel), Cast::ToTextSizeFitVec3(_ControlProperties[C_Prop_TextSizeFit]));
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Apply control styles
Void ApplyStyleList(CMlControl _Control) {
	ApplyPropertyList(_Control, GetPropertyList(_Control));
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Apply control styles on all controls target by the library
Void ApplyStylesheet(CMlPage _Page) {
	_Page.GetClassChildren(C_Class_Target, _Page.MainFrame, True);
	for (Control in _Page.GetClassChildren_Result) {
		ApplyStyleList(Control);
	}
}
Void ApplyStylesheet(CMlScript _Context) {
	ApplyStylesheet(_Context.Page);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the styles on a control
Boolean SetStyleList(CMlControl _Control, Text[] _StyleList) {
	declare Text[] CMGame_UtilsStylesheet4_StyleList for _Control = Private_InitializeStyleList(_Control);
	if (CMGame_UtilsStylesheet4_StyleList == _StyleList) return False;
	CMGame_UtilsStylesheet4_StyleList = _StyleList;
	return True;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add styles on a control
Boolean AddStyleList(CMlControl _Control, Text[] _StyleList) {
	declare Text[] CMGame_UtilsStylesheet4_StyleList for _Control = Private_InitializeStyleList(_Control);
	declare Boolean Modified = False;
	for (Style in _StyleList) {
		if (!CMGame_UtilsStylesheet4_StyleList.exists(Style)) {
			CMGame_UtilsStylesheet4_StyleList.add(Style);
			Modified = True;
		}
	}
	return Modified;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add style on a control
Boolean AddStyle(CMlControl _Control, Text _Style) {
	return AddStyleList(_Control, [_Style]);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Remove styles on a control
Boolean RemoveStyleList(CMlControl _Control, Text[] _StyleList) {
	declare Text[] CMGame_UtilsStylesheet4_StyleList for _Control = Private_InitializeStyleList(_Control);
	declare Boolean Modified = False;
	for (Style in _StyleList) {
		if (CMGame_UtilsStylesheet4_StyleList.remove(Style)) {
			Modified = True;
		}
	}
	return Modified;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Remove style on a control
Boolean RemoveStyle(CMlControl _Control, Text _Style) {
	return RemoveStyleList(_Control, [_Style]);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Inject a style list in the control manialink xml

	return """ data-{{{C_Private_DataAttribute_StyleList}}}="{{{TL::Join(C_Private_Seperator_StyleList, _StyleList)}}}" """;
}

/// Get the styles on a control
Text[] GetStyleList(CMlControl _Control) {
	declare Text[] CMGame_UtilsStylesheet4_StyleList for _Control = Private_InitializeStyleList(_Control);
	return CMGame_UtilsStylesheet4_StyleList;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the text size fit of a label
Vec3 GetTextSizeFit(CMlLabel _Label) {
	return Cast::ToTextSizeFitVec3(GetPropertyList(_Label).get(C_Prop_TextSizeFit, ""));
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the maximum text size of a label
Real GetTextSizeFitMax(CMlLabel _Label) {
	return GetTextSizeFit(_Label).X;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the maximum text size of a label
Void FitLabelValue(CMlLabel _Label) {
	MLHelpers::FitLabelValue(_Label, GetTextSizeFit(_Label));
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the control opacity
Real GetOpacity(CMlControl _Control, Real _DefaultValue) {
	declare Text[Text] PropertyList = GetPropertyList(_Control);
	if (PropertyList.existskey(C_Prop_Opacity)) {
		return Cast::ToReal(PropertyList[C_Prop_Opacity]);
	}
	return _DefaultValue;
}
