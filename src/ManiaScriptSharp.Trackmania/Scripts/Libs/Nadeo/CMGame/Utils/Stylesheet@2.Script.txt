/// Return a color structure from the given hexadecimal color
K_Color GetColorStruct(Text _Hex6Color) {
	return K_Color {
		Hex6 = _Hex6Color,
		RGB = CL::Hex6ToRgb(_Hex6Color)
	};
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the theme to use in the given context
K_Theme GetTheme(CSystemPlatform _System) {
	declare K_Theme CMGame_Stylesheet2_Theme for _System = C_Theme_Null;
	return CMGame_Stylesheet2_Theme;
}
K_Theme GetTheme(CManiaApp _ManiaApp) {
	declare K_Theme CMGame_Stylesheet2_Theme for _ManiaApp = C_Theme_Null;
	if (CMGame_Stylesheet2_Theme == C_Theme_Null) {
		return GetTheme(_ManiaApp.System);
	}
	return CMGame_Stylesheet2_Theme;
}
K_Theme GetTheme(CMlScript _Script) {
	declare K_Theme CMGame_Stylesheet2_Theme for _Script = C_Theme_Null;
	if (CMGame_Stylesheet2_Theme == C_Theme_Null) {
		declare CManiaApp ManiaApp <=> Context::GetParentApp(_Script);
		if (ManiaApp == Null) {
			return GetTheme(_Script.System);
		}
		return GetTheme(ManiaApp);
	}
	return CMGame_Stylesheet2_Theme;
}
K_Theme GetTheme(CMode _Mode) {
	declare K_Theme CMGame_Stylesheet2_Theme for _Mode = C_Theme_Null;
	if (CMGame_Stylesheet2_Theme == C_Theme_Null) {
		return GetTheme(_Mode.System);
	}
	return CMGame_Stylesheet2_Theme;
}
K_Theme GetTheme(CNod _Nod) {
	switchtype (_Nod as TypedNod) {
		case CSystemPlatform: return GetTheme(TypedNod);
		case CManiaApp: return GetTheme(TypedNod);
		case CMlScript: return GetTheme(TypedNod);
		case CMode: return GetTheme(TypedNod);
	}
	return GetTheme(System);
}
K_Theme GetTheme() {
	return GetTheme(This);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the skew angle for the given theme
Real GetSkewAngle(K_Theme _Theme) {
	assert(_Theme.Id != C_ThemeId_Null, "[CMGame/Stylesheet@2] You must define a theme with the `SetTheme()` function before trying to get the skew angle.");
	return _Theme.SkewAngle;
}
Real GetSkewAngle() {
	return GetSkewAngle(GetTheme());
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get a color in K_Color format for the given theme
K_Color GetColor(K_Theme _Theme, Integer _ColorId) {
	assert(_Theme.Id != C_ThemeId_Null, "[CMGame/Stylesheet@2] You must define a theme with the `SetTheme()` function before trying to get a color.");
	return _Theme.Colors.get(_ColorId, C_Color_Null);
}
K_Color GetColor(Integer _ColorId) {
	return GetColor(GetTheme(), _ColorId);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get a color in Hex6 format for the given theme
Text GetColorHex6(K_Theme _Theme, Integer _ColorId) {
	return GetColor(_Theme, _ColorId).Hex6;
}
Text GetColorHex6(Integer _ColorId) {
	return GetColorHex6(GetTheme(), _ColorId);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get a color in RGB format for the given theme
Vec3 GetColorRGB(K_Theme _Theme, Integer _ColorId) {
	return GetColor(_Theme, _ColorId).RGB;
}
Vec3 GetColorRGB(Integer _ColorId) {
	return GetColorRGB(GetTheme(), _ColorId);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get a font for the given theme
Text GetFont(K_Theme _Theme, Integer _FontId) {
	assert(_Theme.Id != C_ThemeId_Null, "[CMGame/Stylesheet@2] You must define a theme with the `SetTheme()` function before trying to get a font.");
	return _Theme.Fonts.get(_FontId, C_FontRaw_Default);
}
Text GetFont(Integer _FontId) {
	return GetFont(GetTheme(), _FontId);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Format all the font styles for the given theme
/// Call this function in the manialink <stylesheet> tag
Text GetLabelStyles() {
	declare Text FormattedLabelStyles;

	foreach (ThemeLabelStyles in GetTheme().LabelStyles) {
		FormattedLabelStyles ^= ThemeLabelStyles^"\n";
	}

	return FormattedLabelStyles;
}
