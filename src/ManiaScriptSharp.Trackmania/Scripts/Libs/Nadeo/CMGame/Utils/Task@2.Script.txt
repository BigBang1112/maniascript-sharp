/// Get the type of task manager
Integer Private_GetTaskMgrType(CNod _TaskMgr) {
	switchtype (_TaskMgr as TaskMgr) {
		case CDataFileMgr: return C_TaskMgrType_DataFileMgr;
		case CScoreMgr: return C_TaskMgrType_ScoreMgr;
		case CTrackingMgr: return C_TaskMgrType_TrackingMgr;
		case CUserV2Manager: return C_TaskMgrType_UserMgr;
		// case CPresenceMgr: return C_TaskMgrType_PresenceMgr;
		// case CManiaPlanet: return C_TaskMgrType_ManiaPlanet;
		case CRankingsZoneMgr: return C_TaskMgrType_ZoneMgr;
		case CWSNotificationManager: return C_TaskMgrType_NotificationMgr;
		case CMode: return C_TaskMgrType_Mode;
		case CPrivilegeMgr: return C_TaskMgrType_PrivilegeMgr;
		default: {
			assert(
				False,
				"""[Task@2] The provided `_TaskMgr` is not supported by the library"""
			);
		}
	}
	return C_TaskMgrType_Null;
}

/// Get a task result in a manager from its id
CTaskResult Private_GetTaskResult(CNod _TaskMgr, Ident _TaskResultId) {
	switchtype (_TaskMgr as TaskMgr) {
		case CDataFileMgr: return TaskMgr.TaskResults.get(_TaskResultId, Null);
		case CScoreMgr: return TaskMgr.TaskResults.get(_TaskResultId, Null);
		case CTrackingMgr: return TaskMgr.TaskResults.get(_TaskResultId, Null);
		case CUserV2Manager: return TaskMgr.TaskResults.get(_TaskResultId, Null);
		// case CPresenceMgr: return TaskMgr.TaskResults.get(_TaskResultId, Null);
		// case CManiaPlanet: return TaskMgr.MasterServer_MSTaskResults.get(_TaskResultId, Null);
		case CRankingsZoneMgr: return TaskMgr.TaskResults.get(_TaskResultId, Null);
		case CWSNotificationManager: return TaskMgr.TaskResults.get(_TaskResultId, Null);
		case CMode: return TaskMgr.TaskResults.get(_TaskResultId, Null);
		case CPrivilegeMgr: return TaskMgr.TaskResults.get(_TaskResultId, Null);
		default: {
			assert(
				False,
				"""[Task@2] The provided `_TaskMgr` is not supported by the library"""
			);
		}
	}
	return Null;
}
CTaskResult GetTaskResult(CNod _TaskMgr, K_Task _Task) {
	assert(
		Private_GetTaskMgrType(_TaskMgr) == _Task.TaskMgrType,
		"""[Task@2] The provided `_TaskMgr` does not match the one from the task"""
	);
	return Private_GetTaskResult(_TaskMgr, _Task.TaskResultId);
}

/// Destroy a task
K_Task Destroy(CUserV2Manager _UserMgr, CNod _TaskMgr, K_Task _Task) {
	if (
		_Task.Status == C_Status_Null ||
		_Task.Status == C_Status_Destroyed
	) {
		return _Task;
	}

	assert(
		Private_GetTaskMgrType(_TaskMgr) == _Task.TaskMgrType,
		"""[Task@2] The provided `_TaskMgr` does not match the one from the task"""
	);

	declare K_Task Task = _Task;

	if (_Task.TaskResultId != NullId) {
		switchtype (_TaskMgr as TaskMgr) {
			case CDataFileMgr: {
				if (TaskMgr.TaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.TaskResult_Release(_Task.TaskResultId);
				}
			}
			case CScoreMgr: {
				if (TaskMgr.TaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.TaskResult_Release(_Task.TaskResultId);
				}
			}
			case CTrackingMgr: {
				if (TaskMgr.TaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.TaskResult_Release(_Task.TaskResultId);
				}
			}
			case CUserV2Manager: {
				if (TaskMgr.TaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.TaskResult_Release(_Task.TaskResultId);
				}
			}
			/* case CPresenceMgr: {
				if (TaskMgr.TaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.TaskResult_Release(_Task.TaskResultId);
				}
			}
			case CManiaPlanet: {
				if (TaskMgr.MasterServer_MSTaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.MasterServer_ReleaseMSTaskResult(_Task.TaskResultId);
				}
			}
			*/
			case CRankingsZoneMgr: {
				if (TaskMgr.TaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.TaskResult_Release(_Task.TaskResultId);
				}
			}
			case CWSNotificationManager: {
				if (TaskMgr.TaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.TaskResult_Release(_Task.TaskResultId);
				}
			}
			case CMode: {
				if (TaskMgr.TaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.TaskResult_Release(_Task.TaskResultId);
				}
			}
			case CPrivilegeMgr: {
				if (TaskMgr.TaskResults.existskey(_Task.TaskResultId)) {
					TaskMgr.TaskResult_Release(_Task.TaskResultId);
				}
			}
			default: {
				assert(
					False,
					"""[Task@2] The provided `_TaskMgr` is not supported by the library"""
				);
			}
		}
	}

	if (
		Task.TaskShowSystemDialogId != NullId &&
		_UserMgr.TaskResults.existskey(Task.TaskShowSystemDialogId)
	) {
		_UserMgr.TaskResult_Release(Task.TaskShowSystemDialogId);
	}

	Task.TaskResultId = NullId;
	Task.TaskShowSystemDialogId = NullId;
	Task.Status = C_Status_Destroyed;

	return Task;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the complete status of the task
Integer Private_GetCompleteStatus(Boolean _IsExternalUpdate) {
	if (_IsExternalUpdate) return C_Status_Complete;
	return C_Status_WaitingExternalComplete;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update new task
K_Task Private_Update(CUserV2Manager _UserMgr, CNod _TaskMgr, K_Task _Task, Boolean _IsExternalUpdate) {
	if (
		_Task.Status != C_Status_Processing &&
		_Task.Status != C_Status_ErrorDialog &&
		_Task.Status != C_Status_WaitingExternalComplete
	) {
		return _Task;
	}

	assert(
		Private_GetTaskMgrType(_TaskMgr) == _Task.TaskMgrType,
		"""[Task@2] The provided `_TaskMgr` does not match the one from the task"""
	);

	declare K_Task Task = _Task;

	switch (Task.Status) {
		case C_Status_Processing: {
			declare CTaskResult TaskResult <=> Private_GetTaskResult(_TaskMgr, Task.TaskResultId);
			if (TaskResult == Null) {
				Task.Status = Private_GetCompleteStatus(_IsExternalUpdate);
			} else if (!TaskResult.IsProcessing) {
				if (TaskResult.HasSucceeded && !TaskResult.HasFailed && !TaskResult.IsCanceled) {
					Task.Status = Private_GetCompleteStatus(_IsExternalUpdate);
					Task.Promise = Promise::Resolve(Task.Promise);
				} else {
					Task.Promise = Promise::Reject(Task.Promise, Error::GetError(TaskResult.ErrorType, TaskResult.ErrorCode));
					if (Error::IsFromFirstParty(Promise::GetError(Task.Promise))) {
						declare CTaskResult TaskShowSystemDialog <=> _UserMgr.TaskError_ShowSystemDialog(_UserMgr.MainUser.Id, TaskResult.ErrorCode);
						if (TaskShowSystemDialog == Null) {
							Task.Status = Private_GetCompleteStatus(_IsExternalUpdate);
						} else {
							Task.TaskShowSystemDialogId = TaskShowSystemDialog.Id;
							Task.Status = C_Status_ErrorDialog;
						}
					} else {
						Task.Status = Private_GetCompleteStatus(_IsExternalUpdate);
					}
				}
			}
		}
		case C_Status_ErrorDialog: {
			declare CTaskResult ErrorTaskResult <=> Private_GetTaskResult(_UserMgr, Task.TaskShowSystemDialogId);
			if (ErrorTaskResult == Null) {
				Task.Status = Private_GetCompleteStatus(_IsExternalUpdate);
			} else if (!ErrorTaskResult.IsProcessing) {
				Task.Status = Private_GetCompleteStatus(_IsExternalUpdate);
			}
		}
		case C_Status_WaitingExternalComplete: {
			if (_IsExternalUpdate) {
				Task.Status = C_Status_Complete;
			}
		}
	}

	return Task;
}
K_Task Update(CUserV2Manager _UserMgr, CNod _TaskMgr, K_Task _Task) {
	return Private_Update(_UserMgr, _TaskMgr, _Task, True);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Create a new task
K_Task Create(CUserV2Manager _UserMgr, CNod _TaskMgr, CTaskResult _TaskResult) {
	assert(
		_TaskMgr != Null,
		"""[Task@2] The provided `_TaskMgr` cannot be `Null`"""
	);

	declare K_Task Task = C_Task_Null;
	if (_TaskResult != Null) {
		assert(
			Private_GetTaskResult(_TaskMgr, _TaskResult.Id) != Null,
			"""[Task@2] Cannot find the task result in the provided `_TaskMgr`"""
		);
		Task.TaskResultId = _TaskResult.Id;
	}
	Task.TaskMgrType = Private_GetTaskMgrType(_TaskMgr);
	Task.Status = C_Status_Processing;
	Task.Promise = Promise::Create();

	return Private_Update(_UserMgr, _TaskMgr, Task, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Destroy a previous task and create a new one
K_Task DestroyAndCreate(CUserV2Manager _UserMgr, CNod _TaskMgr, K_Task _PreviousTask, CTaskResult _TaskResult) {
	Destroy(_UserMgr, _TaskMgr, _PreviousTask);
	return Create(_UserMgr, _TaskMgr, _TaskResult);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Cancel a processing task
K_Task Cancel(CUserV2Manager _UserMgr, CNod _TaskMgr, K_Task _Task) {
	if (_Task.Status != C_Status_Processing) return _Task;

	declare K_Task Task = _Task;

	declare CTaskResult TaskResult <=> Private_GetTaskResult(_TaskMgr, Task.TaskResultId);
	if (TaskResult != Null) {
		TaskResult.Cancel();
		Task = Private_Update(_UserMgr, _TaskMgr, Task, False);
	}

	return Task;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the task is processing
Boolean IsProcessing(K_Task _Task) {
	return (
		_Task.Status == C_Status_Processing ||
		_Task.Status == C_Status_ErrorDialog ||
		_Task.Status == C_Status_WaitingExternalComplete
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the task is a success
Boolean IsSuccess(K_Task _Task) {
	return Promise::IsFulfilled(_Task.Promise);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the task error object
Error::K_Error GetError(K_Task _Task) {
	return Promise::GetError(_Task.Promise);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the task failed because of a first party error
Boolean IsErrorFromFirstParty(K_Task _Task) {
	return Error::IsFromFirstParty(Promise::GetError(_Task.Promise));
}
