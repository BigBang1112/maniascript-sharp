/**
 * A box represent a manialink element with margin, border and padding
 *
 * A box is composed of four areas that contain one another:
 * ┌─────────────────────────────────────────────┐
 * │                  Margin area                │
 * │┌───────────────────────────────────────────┐│
 * ││                 Border area               ││
 * ││┌─────────────────────────────────────────┐││
 * │││                Padding area             │││
 * │││┌───────────────────────────────────────┐│││
 * ││││               Content area            ││││
 * │││└───────────────────────────────────────┘│││
 * ││└─────────────────────────────────────────┘││
 * │└───────────────────────────────────────────┘│
 * └─────────────────────────────────────────────┘
 *
 * This is similar to the [CSS box model](https://developer.mozilla.org/en-US/docs/Learn_web_development/Core/Styling_basics/Box_model#what_is_the_css_box_model)
 * with [`box-sizing` set to `border-box`](https://developer.mozilla.org/en-US/docs/Learn_web_development/Core/Styling_basics/Box_model#the_alternative_css_box_model)
 */
#Const Version "1.1.2"

#Include "MathLib" as ML

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_Margin {
	Real Top = 0.;
	Real Right = 0.;
	Real Bottom = 0.;
	Real Left = 0.;
}
#Struct K_Border {
	Real Top = 0.;
	Real Right = 0.;
	Real Bottom = 0.;
	Real Left = 0.;
}
#Struct K_Padding {
	Real Top = 0.;
	Real Right = 0.;
	Real Bottom = 0.;
	Real Left = 0.;
}
#Struct K_Area {
	Vec2 Pos = <0., 0.>;
	Vec2 Size = <0., 0.>;
}
#Struct K_Box {
	K_Margin Margin;
	K_Border Border;
	K_Padding Padding;
	K_Area MarginArea;
	K_Area BorderArea;
	K_Area PaddingArea;
	K_Area ContentArea;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Margin_Null K_Margin {
	Top = 0.,
	Right = 0.,
	Bottom = 0.,
	Left = 0.
}
#Const C_Border_Null K_Border {
	Top = 0.,
	Right = 0.,
	Bottom = 0.,
	Left = 0.
}
#Const C_Padding_Null K_Padding {
	Top = 0.,
	Right = 0.,
	Bottom = 0.,
	Left = 0.
}
#Const C_Box_Null K_Box {
	Margin = K_Margin { Top = 0., Right = 0., Bottom = 0., Left = 0. },
	Border = K_Border { Top = 0., Right = 0., Bottom = 0., Left = 0. },
	Padding = K_Padding { Top = 0., Right = 0., Bottom = 0., Left = 0. },
	MarginArea = K_Area { Pos = <0., 0.>, Size = <0., 0.> },
	BorderArea = K_Area { Pos = <0., 0.>, Size = <0., 0.> },
	PaddingArea = K_Area { Pos = <0., 0.>, Size = <0., 0.> },
	ContentArea = K_Area { Pos = <0., 0.>, Size = <0., 0.> }
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Create a box
K_Box New(
	Vec2 _Size,
	K_Margin _MarginAbsolute,
	K_Margin _MarginRatio,
	K_Border _BorderAbsolute,
	K_Border _BorderRatio,
	K_Padding _PaddingAbsolute,
	K_Padding _PaddingRatio
) {
	declare K_Padding Padding = K_Padding {
		Top = ML::Max(0., _PaddingAbsolute.Top) + ML::Max(0., _Size.Y * _PaddingRatio.Top),
		Right = ML::Max(0., _PaddingAbsolute.Right) + ML::Max(0., _Size.X * _PaddingRatio.Right),
		Bottom = ML::Max(0., _PaddingAbsolute.Bottom) + ML::Max(0., _Size.Y * _PaddingRatio.Bottom),
		Left = ML::Max(0., _PaddingAbsolute.Left) + ML::Max(0., _Size.X * _PaddingRatio.Left)
	};
	declare K_Border Border = K_Border {
		Top = ML::Max(0., _BorderAbsolute.Top) + ML::Max(0., _Size.Y * _BorderRatio.Top),
		Right = ML::Max(0., _BorderAbsolute.Right) + ML::Max(0., _Size.X * _BorderRatio.Right),
		Bottom = ML::Max(0., _BorderAbsolute.Bottom) + ML::Max(0., _Size.Y * _BorderRatio.Bottom),
		Left = ML::Max(0., _BorderAbsolute.Left) + ML::Max(0., _Size.X * _BorderRatio.Left)
	};
	declare K_Margin Margin = K_Margin {
		Top = ML::Max(0., _MarginAbsolute.Top) + ML::Max(0., _Size.Y * _MarginRatio.Top),
		Right = ML::Max(0., _MarginAbsolute.Right) + ML::Max(0., _Size.X * _MarginRatio.Right),
		Bottom = ML::Max(0., _MarginAbsolute.Bottom) + ML::Max(0., _Size.Y * _MarginRatio.Bottom),
		Left = ML::Max(0., _MarginAbsolute.Left) + ML::Max(0., _Size.X * _MarginRatio.Left)
	};
	// The border area is expanded until it can contains the border and padding
	declare Vec2 BorderPaddingContentSize = <
		ML::Max(Border.Left + Padding.Left + Border.Right + Padding.Right, _Size.X),
		ML::Max(Border.Top + Padding.Top + Border.Bottom + Padding.Bottom, _Size.Y)
	>;
	return K_Box {
		Margin = Margin,
		Border = Border,
		Padding = Padding,
		MarginArea = K_Area {
			Pos = <0., 0.>,
			Size = <Margin.Left + BorderPaddingContentSize.X + Margin.Right, Margin.Top + BorderPaddingContentSize.Y + Margin.Bottom>
		},
		BorderArea = K_Area {
			Pos = <Margin.Left, -Margin.Top>,
			Size = BorderPaddingContentSize
		},
		PaddingArea = K_Area {
			Pos = <Margin.Left + Border.Left, -Margin.Top - Border.Top>,
			Size = <BorderPaddingContentSize.X - Border.Left - Border.Right, BorderPaddingContentSize.Y - Border.Top - Border.Bottom>
		},
		ContentArea = K_Area {
			Pos = <Margin.Left + Border.Left + Padding.Left, -Margin.Top - Border.Top - Padding.Top>,
			Size = <BorderPaddingContentSize.X - Border.Left - Padding.Left - Border.Right - Padding.Right, BorderPaddingContentSize.Y - Border.Top - Padding.Top - Border.Bottom - Padding.Bottom>
		}
	};
}
K_Box New(Vec2 _Size, K_Margin _MarginAbsolute, K_Margin _MarginRatio, K_Padding _PaddingAbsolute, K_Padding _PaddingRatio) {
	return New(_Size, _MarginAbsolute, _MarginRatio, C_Border_Null, C_Border_Null, _PaddingAbsolute, _PaddingRatio);
}
K_Box NewBox(Vec2 _Size, K_Margin _MarginAbsolute, K_Margin _MarginRatio, K_Border _BorderAbsolute, K_Border _BorderRatio, K_Padding _PaddingAbsolute, K_Padding _PaddingRatio) {
	return New(_Size, _MarginAbsolute, _MarginRatio, _BorderAbsolute, _BorderRatio, _PaddingAbsolute, _PaddingRatio);
}
K_Box NewBox(Vec2 _Size, K_Margin _MarginAbsolute, K_Margin _MarginRatio, K_Padding _PaddingAbsolute, K_Padding _PaddingRatio) {
	return New(_Size, _MarginAbsolute, _MarginRatio, _PaddingAbsolute, _PaddingRatio);
}

// Convert a real array to a margin struct following css syntax rule
K_Margin ToMargin(Real[] _Source) {
	if (_Source.count <= 0) return C_Margin_Null;

	declare K_Margin Margin = C_Margin_Null;
	switch (_Source.count) {
		case 1: {
			Margin.Top = _Source[0];
			Margin.Right = Margin.Top;
			Margin.Bottom = Margin.Top;
			Margin.Left = Margin.Right;
		}
		case 2: {
			Margin.Top = _Source[0];
			Margin.Right = _Source[1];
			Margin.Bottom = Margin.Top;
			Margin.Left = Margin.Right;
		}
		case 3: {
			Margin.Top = _Source[0];
			Margin.Right = _Source[1];
			Margin.Bottom = _Source[2];
			Margin.Left = Margin.Right;
		}
		default: {
			Margin.Top = _Source[0];
			Margin.Right = _Source[1];
			Margin.Bottom = _Source[2];
			Margin.Left = _Source[3];
		}
	}
	return Margin;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Convert a margin text property to a struct following css syntax rule
K_Margin ToMargin(Text _Source) {
	return ToMargin(Cast::ToRealArray(_Source));
}
