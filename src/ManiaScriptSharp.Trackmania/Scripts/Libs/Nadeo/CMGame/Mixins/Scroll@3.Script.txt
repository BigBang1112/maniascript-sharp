/**
 * Scroll mixin
 * ============
 *
 * Scroll through a list of elements
 *
 * Props
 * -----
 *
 * - `scroll-element-size` (`Vec2`): the size of one element
 * - `scroll-gutter-size` (`Vec2`): the size of the verical and horizontal gutter between the elements
 * - `scroll-grid-size` (`Int2`): the number of elements in a row and the number of rows to display in the scroll window
 * - `scroll-window-size` (`Vec2`): the size the scroll window
 * - `scroll-window-padding` (`Padding`): the padding of the scroll window
 * - `scroll-mouse-step-size` (`Vec2`): the size of one step of mousewheel scroll
 * - `scroll-direction` (`Integer`): the direction of the scroll
 * - `scroll-previous-line-visible-ratio` (`Real`): how much of the previous line is visible when scrolling to an element
 * - `scroll-next-line-visible-ratio` (`Real`): how much of the next line is visible when scrolling to an element
 * - `scroll-scrollbar-id` (`Text`): the id of the scrollbar to synchronize with the scroll
 * - `scroll-element-zone-class` (`Text`): space separated list of classes that trigger mouse events
 */

/** Convert a screen coordinate or size to a flex coordinate or size
 * The main axis is the axis running in the direction the scroll elements are laid out in (for example, as a row across the page, or a column down the page)
 * The cross axis is the axis running perpendicular to the direction the scroll elements are laid out in
 * ref: CSS flexbox
 */
K_Private_FlexVec2 Private_GetFlexPos(Vec2 _ScreenPos, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: return K_Private_FlexVec2 { Main = _ScreenPos.X, Cross = _ScreenPos.Y };
		case C_Direction_Column: return K_Private_FlexVec2 { Main = -_ScreenPos.Y, Cross = -_ScreenPos.X };
	}
	return K_Private_FlexVec2 { Main = _ScreenPos.X, Cross = _ScreenPos.Y };
}
K_Private_FlexVec2 Private_GetFlexSize(Vec2 _ScreenSize, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: return K_Private_FlexVec2 { Main = _ScreenSize.X, Cross = _ScreenSize.Y };
		case C_Direction_Column: return K_Private_FlexVec2 { Main = _ScreenSize.Y, Cross = _ScreenSize.X };
	}
	return K_Private_FlexVec2 { Main = _ScreenSize.X, Cross = _ScreenSize.Y };
}
K_Private_FlexInt2 Private_GetFlexSize(Int2 _ScreenSize, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: return K_Private_FlexInt2 { Main = _ScreenSize.X, Cross = _ScreenSize.Y };
		case C_Direction_Column: return K_Private_FlexInt2 { Main = _ScreenSize.Y, Cross = _ScreenSize.X };
	}
	return K_Private_FlexInt2 { Main = _ScreenSize.X, Cross = _ScreenSize.Y };
}
K_Private_FlexPadding Private_GetFlexPadding(Box::K_Padding _ScreenPadding, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: {
			return K_Private_FlexPadding {
				MainLeft = _ScreenPadding.Left,
				MainRight = _ScreenPadding.Right,
				CrossTop = _ScreenPadding.Top,
				CrossBottom = _ScreenPadding.Bottom
			};
		}
		case C_Direction_Column: {
			return K_Private_FlexPadding {
				MainLeft = _ScreenPadding.Top,
				MainRight = _ScreenPadding.Bottom,
				CrossTop = _ScreenPadding.Left,
				CrossBottom = _ScreenPadding.Right
			};
		}
	}
	return K_Private_FlexPadding {
		MainLeft = _ScreenPadding.Left,
		MainRight = _ScreenPadding.Right,
		CrossTop = _ScreenPadding.Top,
		CrossBottom = _ScreenPadding.Bottom
	};
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Convert a flex coordinate or size to a screen coordinate or size
Vec2 Private_GetScreenPos(K_Private_FlexVec2 _FlexPos, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: return <_FlexPos.Main, _FlexPos.Cross>;
		case C_Direction_Column: return <-_FlexPos.Cross, -_FlexPos.Main>;
	}
	return <_FlexPos.Main, _FlexPos.Cross>;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add `K_Private_FlexVec2`
K_Private_FlexVec2 Private_Add(K_Private_FlexVec2 _FlexVec2A, K_Private_FlexVec2 _FlexVec2B) {
	return K_Private_FlexVec2 {
		Main = _FlexVec2A.Main + _FlexVec2B.Main,
		Cross = _FlexVec2A.Cross + _FlexVec2B.Cross
	};
}
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add `K_Private_Vec2`
K_Private_Vec2 Private_Add(K_Private_Vec2 _VecA, K_Private_Vec2 _VecB) {
	return K_Private_Vec2 {
		Flex = Private_Add(_VecA.Flex, _VecB.Flex),
		Screen = _VecA.Screen + _VecB.Screen
	};
}

/// Find in which scroll element the given control is
CMlControl Private_FindScrollElement(CMlControl _Control) {
	if (_Control == Null) return Null;

	declare Boolean CMGame_MixinsScroll3_ScrollElementHasCache for _Control = False;
	declare CMlControl CMGame_MixinsScroll3_ScrollElementOwner for _Control <=> Null;
	if (CMGame_MixinsScroll3_ScrollElementHasCache) {
		return CMGame_MixinsScroll3_ScrollElementOwner;
	}

	declare CMlControl Control <=> _Control;
	while (!Control.HasClass(C_Class_Element) && Control.Parent != Null) {
		Control <=> Control.Parent;
	}

	CMGame_MixinsScroll3_ScrollElementHasCache = True;
	if (Control.HasClass(C_Class_Element)) {
		CMGame_MixinsScroll3_ScrollElementOwner <=> Control;
	} else {
		CMGame_MixinsScroll3_ScrollElementOwner <=> Null;
	}

	return CMGame_MixinsScroll3_ScrollElementOwner;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the scroll element with the requested key
K_Element Private_GetElementWithKey(K_Private_Instance _Instance, Integer _ElementKey) {
	for (Element in _Instance.Elements) {
		if (Element.Key == _ElementKey) {
			return Element;
		}
	}
	return C_Element_Null;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the current scroll offset
Vec2 Private_GetScrollOffset(K_Private_Instance _Instance) {
	if (_Instance.Animation.IsPlaying && _Instance.Animation.StartTime + _Instance.Animation.Duration <= Now) {
		return _Instance.Animation.OffsetStart + _Instance.Animation.OffsetDelta;
	}
	return _Instance.ScrollOffset;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the scrollbar range when the content of the scroll window change
Void Private_UpdateScrollBarRange(K_Private_Instance _Instance) {
	if (_Instance.Controls.Frame_ScrollBar != Null) {
		declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(_Instance), _Instance.Properties.Direction);
		declare Vec2 Range = <ML::Abs(ScrollOffsetFlex.Cross), ML::Abs(ScrollOffsetFlex.Cross) + _Instance.Properties.WindowSize.Flex.Cross>;
		Component_ScrollBar::SetRange(_Instance.Controls.Frame_ScrollBar, Range, _Instance.ScrollBarFullLength);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if an element is focused
Boolean Private_IsElementFocused(K_Private_Instance _Instance, Integer _ElementKey) {
	return (_Instance.IsFocused && _ElementKey == _Instance.FocusedElementKey);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the scroll focused element
K_Private_Instance Private_SetFocusedElementKey(CMlScript _This, K_Private_Instance _Instance, Integer _FocusedElementKey, Boolean _IsInit) {
	declare Integer FocusedElementKey = -1;
	if (_Instance.ElementsNb > 0) {
		FocusedElementKey = ML::Clamp(_FocusedElementKey, 0, _Instance.ElementsNb - 1);
	}

	if (!_IsInit && _Instance.FocusedElementKey == FocusedElementKey) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	declare Integer PreviouslyFocusedElementKey = Instance.FocusedElementKey;
	Instance.FocusedElementKey = FocusedElementKey;

	declare K_Private_State CMGame_MixinsScroll3_State as State for _This.Page = C_Private_State_Null;
	declare K_Element From = C_Element_Null;
	declare K_Element To = C_Element_Null;
	for (Key => Element in Instance.Elements) {
		// Update focused state
		if (Element.IsFocused != Private_IsElementFocused(Instance, Element.Key)) {
			Instance.Elements[Key].IsFocused = Private_IsElementFocused(Instance, Element.Key);
		}
		if (Element.Key == PreviouslyFocusedElementKey) {
			From = Instance.Elements[Key];
		}
		if (Element.Key == Instance.FocusedElementKey) {
			To = Instance.Elements[Key];
		}
	}
	State.CollectEvents.add(K_Event {
		Type = C_EventType_ElementFocusUpdated,
		Frame_Scroll = Instance.Controls.Frame_Scroll,
		From = From,
		To = To,
		IsMouse = False
	});

	return Instance;
}
K_Private_Instance Private_SetFocusedElementKey(CMlScript _This, K_Private_Instance _Instance, Integer _FocusedElementKey) {
	return Private_SetFocusedElementKey(_This, _Instance, _FocusedElementKey, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Scroll from a position to another
K_Private_Instance Private_ScrollFromOffsetToOffset(
	K_Private_Instance _Instance,
	K_Private_FlexVec2 _From,
	K_Private_FlexVec2 _To,
	Boolean _IsAnimated
) {
	if (_To.Cross == _From.Cross) return _Instance;

	// Assigning a new value to `ScrollOffset` like that does not work here:
	// `Instance.Controls.Frame_Scroll.ScrollOffset.Y = _To.Cross;`
	// `ScrollOffset` will simply ignore the assignment and will keep its original value
	// To get around this behaviour we create an animation with no duration
	// Assigning the value in the next yield is working

	declare K_Private_Instance Instance = _Instance;

	declare K_Private_FlexVec2 OffsetDelta = K_Private_FlexVec2 {
		Main = _To.Main - _From.Main,
		Cross = _To.Cross - _From.Cross
	};

	declare Integer AnimDuration = 0;
	if (_IsAnimated) AnimDuration = C_Private_AnimDuration_Scroll;

	Instance.Animation.IsPlaying = True;
	Instance.Animation.StartTime = Now;
	Instance.Animation.Duration = AnimDuration;
	Instance.Animation.OffsetStart = Private_GetScreenPos(_From, Instance.Properties.Direction);
	Instance.Animation.OffsetDelta = Private_GetScreenPos(OffsetDelta, Instance.Properties.Direction);

	return Instance;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Scroll to the given scroll ratio
K_Private_Instance Private_ScrollToRatio(K_Private_Instance _Instance, Real _Ratio, Boolean _IsAnimated) {
	declare K_Private_Instance Instance = _Instance;

	declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
	declare K_Private_FlexVec2 ScrollOffsetTargetFlex = ScrollOffsetFlex;
	ScrollOffsetTargetFlex.Cross = Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollMax, Instance.Properties.Direction).Cross * _Ratio;

	return Private_ScrollFromOffsetToOffset(Instance, ScrollOffsetFlex, ScrollOffsetTargetFlex, _IsAnimated);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Scroll to the focused elements
K_Private_Instance Private_ScrollToFocusedElementKey(K_Private_Instance _Instance, Integer _ElementKey, Boolean _IsAnimated) {
	declare K_Private_Instance Instance = _Instance;

	declare Integer FocusedElementLineIndex = 0;
	if (Instance.Properties.GridSize.Flex.Main > 0) {
		FocusedElementLineIndex = _ElementKey / Instance.Properties.GridSize.Flex.Main;
	}
	declare Vec2 FocusedElementVerticalRange = <Instance.Properties.WindowPadding.Flex.CrossTop + (FocusedElementLineIndex * (Instance.Properties.ElementSize.Flex.Cross + Instance.Properties.GutterSize.Flex.Cross)), 0.>;
	FocusedElementVerticalRange.Y = FocusedElementVerticalRange.X + Instance.Properties.ElementSize.Flex.Cross;

	// Try to show a bit of the elements on the line above or below the focused element
	declare Real PreviousLinePreviewSizeCross = 0.;
	if (Instance.Properties.PreviousLineVisibleRatio > 0.) {
		PreviousLinePreviewSizeCross = Instance.Properties.GutterSize.Flex.Cross + (Instance.Properties.ElementSize.Flex.Cross * Instance.Properties.PreviousLineVisibleRatio);
	}
	declare Real NextLinePreviewSizeCross = 0.;
	if (Instance.Properties.NextLineVisibleRatio > 0.) {
		NextLinePreviewSizeCross = Instance.Properties.GutterSize.Flex.Cross + (Instance.Properties.ElementSize.Flex.Cross * Instance.Properties.NextLineVisibleRatio);
	}

	declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
	declare K_Private_FlexVec2 ScrollOffsetTargetFlex = ScrollOffsetFlex;
	if (FocusedElementVerticalRange.X < ScrollOffsetFlex.Cross + PreviousLinePreviewSizeCross) {
		ScrollOffsetTargetFlex.Cross = ML::Max(0., FocusedElementVerticalRange.X - PreviousLinePreviewSizeCross);
	} else if (FocusedElementVerticalRange.Y > ScrollOffsetFlex.Cross + Instance.Properties.WindowSize.Flex.Cross - NextLinePreviewSizeCross) {
		ScrollOffsetTargetFlex.Cross = ML::Min(Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollMax, Instance.Properties.Direction).Cross, FocusedElementVerticalRange.Y - Instance.Properties.WindowSize.Flex.Cross + NextLinePreviewSizeCross);
	}

	return Private_ScrollFromOffsetToOffset(Instance, ScrollOffsetFlex, ScrollOffsetTargetFlex, _IsAnimated);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Scroll to the focused elements
K_Private_Instance Private_ScrollToFocusedElement(K_Private_Instance _Instance, Boolean _IsAnimated) {
	return Private_ScrollToFocusedElementKey(_Instance, _Instance.FocusedElementKey, _IsAnimated);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Scroll from a ratio to the focused elements
K_Private_Instance Private_ScrollFromRatioToFocusedElement(K_Private_Instance _Instance, Real _Ratio, Boolean _IsAnimated) {
	return Private_ScrollToFocusedElement(Private_ScrollToRatio(_Instance, _Ratio, False), _IsAnimated);
}

/// Set the scroll offset
K_Private_Instance Private_SetScrollOffset(
	CMlScript _This,
	K_Private_Instance _Instance,
	Vec2 _ScrollOffset,
	Vec2 _ScrollAnimOffset,
	Boolean _IsInit
) {
	if (!_IsInit && _Instance.ScrollOffset == _ScrollOffset && _Instance.ScrollAnimOffset == _ScrollAnimOffset) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	Instance.ScrollOffset = _ScrollOffset;
	Instance.ScrollAnimOffset = _ScrollAnimOffset;

	Instance.Controls.Quad_WindowBackground.RelativePosition_V3 = -Instance.ScrollAnimOffset;

	// Find the range of element keys visible in the scroll window
	declare Int2 ElementKeyRange = <0, 0>;
	declare K_Private_Vec2 ElementAndGutterSize = Private_Add(Instance.Properties.ElementSize, Instance.Properties.GutterSize);
	declare K_Private_FlexVec2 ScrollAnimOffsetFlex = Private_GetFlexPos(Instance.ScrollAnimOffset, Instance.Properties.Direction);
	if (Instance.ElementsNb > 0 && ElementAndGutterSize.Flex.Cross > 0.) {
		ElementKeyRange.X = ML::Clamp(ML::Max(0, ML::FloorInteger(ML::Max(0., ScrollAnimOffsetFlex.Cross - Instance.Properties.WindowPadding.Flex.CrossTop) / ElementAndGutterSize.Flex.Cross)) * Instance.Properties.GridSize.Flex.Main, 0, ML::Max(0, Instance.ElementsNb - 1));
		ElementKeyRange.Y = ML::Clamp(ElementKeyRange.X + (Instance.Properties.GridSize.Flex.Main * Instance.Properties.GridSize.Flex.Cross) + Instance.Properties.GridSize.Flex.Main - 1, ElementKeyRange.X, ML::Max(0, Instance.ElementsNb - 1));
	}

	// If the range changed, update the scroll
	if (_IsInit || Instance.ElementKeyRange != ElementKeyRange) {
		Instance.ElementKeyRange = ElementKeyRange;

		declare K_Event Event = K_Event {
			Type = C_EventType_ElementsUpdated,
			Frame_Scroll = Instance.Controls.Frame_Scroll,
			IsMouse = False
		};

		// Update the elements that left the scroll window
		declare Integer[] UpToDateElementKeys; // Keys of the elements that are already up to date
		declare Integer[] ReusableElementsArrayKeys; // Keys of the `Instance.Elements` that are outside of the scroll window and can be reused to display new elements that enter the scroll window. CAUTION: This is not the element key but the key of the element in the `Instance.Elements` array.
		for (Key => Element in Instance.Elements) {
			if (
				Instance.ElementsNb <= 0 ||
				Element.Key == C_Private_ElementKey_Null ||
				Element.Key < ElementKeyRange.X ||
				Element.Key > ElementKeyRange.Y ||
				UpToDateElementKeys.exists(Element.Key)
			) {
				if (Element.Key != C_Private_ElementKey_Null) {
					declare K_Element UpdatedElement = Instance.Elements[Key];
					UpdatedElement.Key = C_Private_ElementKey_Null;
					UpdatedElement.Control.Visible = False;

					declare Integer CMGame_MixinsScroll3_ElementKey for UpdatedElement.Control = C_Private_ElementKey_Null;
					CMGame_MixinsScroll3_ElementKey = UpdatedElement.Key;

					Instance.Elements[Key] = UpdatedElement;
				}
				ReusableElementsArrayKeys.add(Key);
			} else if (_IsInit) {
				ReusableElementsArrayKeys.add(Key);
			} else {
				UpToDateElementKeys.add(Element.Key);
			}
		}

		// Update the elements that entered the scroll window
		if (Instance.ElementsNb > 0) {
			for (ElementKey, ElementKeyRange.X, ElementKeyRange.Y) {
				if (!UpToDateElementKeys.exists(ElementKey) && ReusableElementsArrayKeys.count > 0) {
					declare Integer ElementsArrayKey = ReusableElementsArrayKeys[0];
					declare K_Element UpdatedElement = Instance.Elements[ElementsArrayKey];
					UpdatedElement.Key = ElementKey;
					UpdatedElement.Control.Visible = True;
					UpdatedElement.IsFocused = Private_IsElementFocused(Instance, UpdatedElement.Key);

					declare Integer CMGame_MixinsScroll3_ElementKey for UpdatedElement.Control = C_Private_ElementKey_Null;
					CMGame_MixinsScroll3_ElementKey = UpdatedElement.Key;

					declare K_Private_FlexVec2 RelativePositionFlex = K_Private_FlexVec2 { Main = 0., Cross = 0. };
					if (Instance.Properties.GridSize.Flex.Main > 0) {
						RelativePositionFlex.Main = Instance.Properties.WindowPadding.Flex.MainLeft + ((UpdatedElement.Key % Instance.Properties.GridSize.Flex.Main) * ElementAndGutterSize.Flex.Main);
					}
					if (Instance.Properties.GridSize.Flex.Cross > 0) {
						RelativePositionFlex.Cross = -Instance.Properties.WindowPadding.Flex.CrossTop - ((UpdatedElement.Key / Instance.Properties.GridSize.Flex.Main) * ElementAndGutterSize.Flex.Cross);
					}
					UpdatedElement.Control.RelativePosition_V3 = Private_GetScreenPos(RelativePositionFlex, Instance.Properties.Direction);

					Event.UpdatedElements.add(UpdatedElement);
					Instance.Elements[ElementsArrayKey] = UpdatedElement;
					ReusableElementsArrayKeys.removekey(0);
				}
			}
		}

		Event.Elements = Instance.Elements;

		declare K_Private_State CMGame_MixinsScroll3_State as State for _This.Page = C_Private_State_Null;
		State.CollectEvents.add(Event);
	}

	Private_UpdateScrollBarRange(Instance);

	return Instance;
}

	return Private_SetScrollOffset(_This, _Instance, _ScrollOffset, _ScrollAnimOffset, False);
}

/// Set the number of elements in the scroll
K_Private_Instance Private_SetElementsNb(
	CMlScript _This,
	K_Private_Instance _Instance,
	Integer _ElementsNb,
	Boolean _IsInit
) {
	if (!_IsInit && _Instance.ElementsNb == _ElementsNb) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	Instance.ElementsNb = _ElementsNb;
	Instance.ScrollBarFullLength = 0.;

	declare K_Private_FlexVec2 ScrollMaxFlex = K_Private_FlexVec2 { Main = 0., Cross = 0. };
	declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
	if (Instance.Properties.GridSize.Flex.Main > 0) {
		declare K_Private_Vec2 ElementAndGutterSize = Private_Add(Instance.Properties.ElementSize, Instance.Properties.GutterSize);
		declare Integer LinesNb = (Instance.ElementsNb + Instance.Properties.GridSize.Flex.Main - 1) / Instance.Properties.GridSize.Flex.Main;
		Instance.ScrollBarFullLength = ML::Max(0., (LinesNb * ElementAndGutterSize.Flex.Cross) - Instance.Properties.GutterSize.Flex.Cross + Instance.Properties.WindowPadding.Flex.CrossTop + Instance.Properties.WindowPadding.Flex.CrossBottom);
		ScrollMaxFlex.Cross = ML::Max(0., Instance.ScrollBarFullLength - Instance.Properties.WindowSize.Flex.Cross);
	}
	ScrollOffsetFlex.Cross = ML::Min(ScrollOffsetFlex.Cross, ScrollMaxFlex.Cross);
	Instance.Controls.Frame_Scroll.ScrollMax = Private_GetScreenPos(ScrollMaxFlex, Instance.Properties.Direction);
	Instance.Controls.Frame_Scroll.ScrollOffset = Private_GetScreenPos(ScrollOffsetFlex, Instance.Properties.Direction);

	// Add a bit of vertical scroll to be able to detect mouse vertical scrollwheel and convert it into horizontal scroll
	if (Instance.Properties.Direction == C_Direction_Column && Instance.Controls.Frame_Scroll.ScrollMax.Y == 0.) {
		Instance.Controls.Frame_Scroll.ScrollMin.Y = -Instance.Properties.MouseStepSize.Flex.Cross;
		Instance.Controls.Frame_Scroll.ScrollMax.Y = Instance.Properties.MouseStepSize.Flex.Cross;
	} else {
		Instance.Controls.Frame_Scroll.ScrollMin = <0., 0.>;
	}

	// Check that the currently focused element still exists
	Instance = Private_SetFocusedElementKey(_This, Instance, Instance.FocusedElementKey);

	// When the number of elements change, we must force an initialization of the scroll offset to refresh all elements
	Instance = Private_SetScrollOffset(_This, Instance, Private_GetScreenPos(ScrollOffsetFlex, Instance.Properties.Direction), Instance.Controls.Frame_Scroll.ScrollAnimOffset, True);

	return Instance;
}

}

/// Focus the next element in the line
K_Private_Instance Private_ApplyInput_NextElement(CMlScript _This, K_Private_Instance _Instance, NavigationInputs::K_Input _Input) {
	declare K_Private_Instance Instance = _Instance;

	if (
		Instance.Properties.GridSize.Flex.Main > 0 &&
		Instance.FocusedElementKey < Instance.ElementsNb - 1 &&
		Instance.FocusedElementKey % Instance.Properties.GridSize.Flex.Main < Instance.Properties.GridSize.Flex.Main - 1
	) {
		Instance = Private_SetFocusedElementKey(_This, Instance, Instance.FocusedElementKey + 1);
		Instance = Private_ScrollToFocusedElement(Instance, True);
	} else {
		Mixin_Navigation::ApplyEscapingNavigationInput(_This, False, Instance.Controls.Frame_Scroll, _Input.Type, _Input.IsFromMouse, _Input.IsActionAutoRepeat);
	}

	return Instance;

/// Focus the element in the next line
K_Private_Instance Private_ApplyInput_NextLine(CMlScript _This, K_Private_Instance _Instance, NavigationInputs::K_Input _Input) {
	declare K_Private_Instance Instance = _Instance;

	if (Instance.FocusedElementKey < Instance.ElementsNb - Instance.Properties.GridSize.Flex.Main) {
		Instance = Private_SetFocusedElementKey(_This, Instance, Instance.FocusedElementKey + Instance.Properties.GridSize.Flex.Main);
		Instance = Private_ScrollToFocusedElement(Instance, True);
	} else if (
		Instance.Properties.GridSize.Flex.Main > 0 &&
		Instance.FocusedElementKey < Instance.ElementsNb - 1 &&
		Instance.FocusedElementKey / Instance.Properties.GridSize.Flex.Main < (Instance.ElementsNb - 1) / Instance.Properties.GridSize.Flex.Main
	) {
		Instance = Private_SetFocusedElementKey(_This, Instance, Instance.ElementsNb - 1);
		Instance = Private_ScrollToFocusedElement(Instance, True);
	} else {
		// Scroll to the bottom when escaping the scroll
		declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
		declare K_Private_FlexVec2 ScrollMaxFlex = Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollMax, Instance.Properties.Direction);
		if (ScrollOffsetFlex.Cross < ScrollMaxFlex.Cross) {
			Instance = Private_ScrollToRatio(Instance, 1., True);
		}
		Mixin_Navigation::ApplyEscapingNavigationInput(_This, False, Instance.Controls.Frame_Scroll, _Input.Type, _Input.IsFromMouse, _Input.IsActionAutoRepeat);
	}

	return Instance;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Focus the element in the previous line
K_Private_Instance Private_ApplyInput_PreviousLine(CMlScript _This, K_Private_Instance _Instance, NavigationInputs::K_Input _Input) {
	declare K_Private_Instance Instance = _Instance;

	if (Instance.FocusedElementKey >= Instance.Properties.GridSize.Flex.Main) {
		Instance = Private_SetFocusedElementKey(_This, Instance, Instance.FocusedElementKey - Instance.Properties.GridSize.Flex.Main);
		Instance = Private_ScrollToFocusedElement(Instance, True);
	} else {
		// Scroll to the top when escaping the scroll
		declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
		if (ScrollOffsetFlex.Cross > 0.) {
			Instance = Private_ScrollToRatio(Instance, 0., True);
		}
		Mixin_Navigation::ApplyEscapingNavigationInput(_This, False, Instance.Controls.Frame_Scroll, _Input.Type, _Input.IsFromMouse, _Input.IsActionAutoRepeat);
	}

	return Instance;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Apply a directionnal input (up, right, down, left) to the scroll
K_Private_Instance Private_ApplyInput(CMlScript _This, K_Private_Instance _Instance, NavigationInputs::K_Input _Input) {
	declare K_Private_Instance Instance = _Instance;

	switch (_Input.Type) {
		case NavigationInputs::C_InputType_Up: {
			switch (Instance.Properties.Direction) {
				case C_Direction_Row: Instance = Private_ApplyInput_PreviousLine(_This, Instance, _Input);
				case C_Direction_Column: Instance = Private_ApplyInput_PreviousElement(_This, Instance, _Input);
			}
		}
		case NavigationInputs::C_InputType_Down: {
			switch (Instance.Properties.Direction) {
				case C_Direction_Row: Instance = Private_ApplyInput_NextLine(_This, _Instance, _Input);
				case C_Direction_Column: Instance = Private_ApplyInput_NextElement(_This, Instance, _Input);
			}
		}
		case NavigationInputs::C_InputType_Left: {
			switch (Instance.Properties.Direction) {
				case C_Direction_Row: Instance = Private_ApplyInput_PreviousElement(_This, Instance, _Input);
				case C_Direction_Column: Instance = Private_ApplyInput_PreviousLine(_This, Instance, _Input);
			}
		}
		case NavigationInputs::C_InputType_Right: {
			switch (Instance.Properties.Direction) {
				case C_Direction_Row: Instance = Private_ApplyInput_NextElement(_This, Instance, _Input);
				case C_Direction_Column: Instance = Private_ApplyInput_NextLine(_This, _Instance, _Input);
			}
		}
		case NavigationInputs::C_InputType_Select: {
			Private_SelectFocusedElement(_This, Instance, False);
		}
		default: {
			if (!_Input.IsFromMouse) {
				Private_RelayInputElementKey(_This, Instance, Instance.FocusedElementKey, _Input);
			}
		}
	}

	return Instance;
}

/// Initialiaze mixin instance
Void Private_Init(CMlScript _This, CMlControl _Control) {
	assert(_Control is CMlFrame, """[CMGame/Mixins/Scroll@3] The scroll mixin can only be applied to a frame. `#{{{_Control.ControlId}}}` is not a frame.""");

	declare K_Private_Instance CMGame_MixinsScroll3_Instance as Instance for _Control = C_Private_Instance_Null;
	Instance = C_Private_Instance_Null;
	Instance.IsScrollComponent = True;

	Instance.Controls.Frame_Scroll <=> (_Control as CMlFrame);

	// Sort the different types of controls in the scroll frame
	for (Control in Instance.Controls.Frame_Scroll.Controls) {
		if (Control.HasClass(C_Class_WindowBackground)) {
			assert(Control is CMlQuad, "[CMGame/Mixins/Scroll@3] Control with class `C_Class_WindowBackground` must be a quad.");
			assert(Instance.Controls.Quad_WindowBackground == Null, "[CMGame/Mixins/Scroll@3] You can only have one control with class `C_Class_WindowBackground` inside the scroll frame.");
			Instance.Controls.Quad_WindowBackground <=> (Control as CMlQuad);
		} else if (Control.HasClass(C_Class_Element)) {
			Control.Visible = False;
			Instance.Elements.add(K_Element {
				Key = C_Private_ElementKey_Null,
				Control = Control,
				IsFocused = False
			});
		} else if (!Control.HasClass(C_Class_CustomControl)) {
			assert(False, """[CMGame/Mixins/Scroll@3] Control with id `#{{{Control.ControlId}}}` does not have a valid class. Every control in the scroll frame must use one of the following class: `C_Class_Element`, `C_Class_WindowBackground`, `C_Class_CustomControl`.""");
		}
	}
	assert(Instance.Controls.Quad_WindowBackground != Null, "[CMGame/Mixins/Scroll@3] You must add a quad with the class `C_Class_WindowBackgorund` inside the scroll frame. This quad will be invisible and generate the scroll events.");

	// Initialize properties from the manialink
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_ElementSize)) {
		Instance.Properties.ElementSize.Screen = MV::GetMixinProp_Vec2(Instance.Controls.Frame_Scroll, C_Prop_ElementSize, Instance.Properties.ElementSize.Screen);
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_GutterSize)) {
		Instance.Properties.GutterSize.Screen = MV::GetMixinProp_Vec2(Instance.Controls.Frame_Scroll, C_Prop_GutterSize, Instance.Properties.GutterSize.Screen);
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_GridSize)) {
		Instance.Properties.GridSize.Screen = MV::GetMixinProp_Int2(Instance.Controls.Frame_Scroll, C_Prop_GridSize, Instance.Properties.GridSize.Screen);
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_WindowSize)) {
		Instance.Properties.WindowSize.Screen = MV::GetMixinProp_Vec2(Instance.Controls.Frame_Scroll, C_Prop_WindowSize, Instance.Properties.WindowSize.Screen);
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_WindowPadding)) {
		Instance.Properties.WindowPadding.Screen = MV::GetMixinProp_Padding(Instance.Controls.Frame_Scroll, C_Prop_WindowPadding, Instance.Properties.WindowPadding.Screen);
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_MouseStepSize)) {
		Instance.Properties.MouseStepSize.Screen = MV::GetMixinProp_Vec2(Instance.Controls.Frame_Scroll, C_Prop_MouseStepSize, Instance.Properties.MouseStepSize.Screen);
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_Direction)) {
		Instance.Properties.Direction = MV::GetMixinProp_Integer(Instance.Controls.Frame_Scroll, C_Prop_Direction, Instance.Properties.Direction);
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_PreviousLineVisibleRatio)) {
		Instance.Properties.PreviousLineVisibleRatio = MV::GetMixinProp_Real(Instance.Controls.Frame_Scroll, C_Prop_PreviousLineVisibleRatio, Instance.Properties.PreviousLineVisibleRatio);
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_NextLineVisibleRatio)) {
		Instance.Properties.NextLineVisibleRatio = MV::GetMixinProp_Real(Instance.Controls.Frame_Scroll, C_Prop_NextLineVisibleRatio, Instance.Properties.NextLineVisibleRatio);
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_ScrollBarId)) {
		Instance = Private_SetScrollBarId(_This, Instance, MV::GetMixinProp_Text(Instance.Controls.Frame_Scroll, C_Prop_ScrollBarId, ""));
	}
	if (MV::HasMixinProp(Instance.Controls.Frame_Scroll, C_Prop_ElementZoneClasses)) {
		declare Text[] ElementZoneClasses = TL::Split(" ", MV::GetMixinProp_Text(Instance.Controls.Frame_Scroll, C_Prop_ElementZoneClasses, ""));
		for (ElementZoneClass in ElementZoneClasses) {
			if (!Instance.Properties.ElementZoneClasses.exists(ElementZoneClass)) {
				Instance.Properties.ElementZoneClasses.add(ElementZoneClass);
			}
		}
	}

	// Find element zones that trigger mouse events
	declare Text[] ElementZoneClasses = Instance.Properties.ElementZoneClasses;
	if (!ElementZoneClasses.exists(C_Class_ElementZone)) {
		ElementZoneClasses.add(C_Class_ElementZone);
	}
	for (Element in Instance.Elements) {
		if (Element.Control is CMlFrame) {
			for (ElementZoneClass in ElementZoneClasses) {
				_This.Page.GetClassChildren(ElementZoneClass, (Element.Control as CMlFrame), True);
				for (Control in _This.Page.GetClassChildren_Result) {
					assert((Control is CMlQuad), """[CMGame/Mixins/Scroll@3] All element zone controls must be a `CMlQuad`.""");
					declare Boolean CMGame_MixinsScroll3_IsElementZone for Control = False;
					declare CMlFrame CMGame_MixinsScroll3_FrameScrollOwner for Control = Null;
					CMGame_MixinsScroll3_IsElementZone = True;
					CMGame_MixinsScroll3_FrameScrollOwner <=> Instance.Controls.Frame_Scroll;
				}
			}
		}
	}

	Instance.Controls.Quad_WindowBackground.Opacity = 0.;
	Instance.Controls.Frame_Scroll.ScrollActive = True;
	Instance.Controls.Frame_Scroll.ScrollMin = <0., 0.>;

	Instance = Private_SetElementSize(Instance, Instance.Properties.ElementSize.Screen, True);
	Instance = Private_SetGutterSize(Instance, Instance.Properties.GutterSize.Screen, True);
	Instance = Private_SetGridSize(Instance, Instance.Properties.GridSize.Screen, True);
	Instance = Private_SetWindowSize(Instance, Instance.Properties.WindowSize.Screen, True);
	Instance = Private_SetWindowPadding(Instance, Instance.Properties.WindowPadding.Screen, True);
	Instance = Private_SetMouseStepSize(Instance, Instance.Properties.MouseStepSize.Screen, True);

	Instance = Private_SetElementsNb(_This, Instance, Instance.ElementsNb, True);
	Instance = Private_SetFocusedElementKey(_This, Instance, Instance.FocusedElementKey, True);
	Instance = Private_SetIsFocused(_This, Instance, Instance.IsFocused, True);
	Instance = Private_SetScrollOffset(_This, Instance, Instance.Controls.Frame_Scroll.ScrollOffset, Instance.Controls.Frame_Scroll.ScrollAnimOffset, True);
	Instance = Private_SetIsNavigationEnabled(Instance, Instance.IsNavigationEnabled, True);
}

/// Update the page
Void Private_Hook_PageAfterYield(CMlScript _This) {
	if (_This.PageIsVisible) {
		declare K_Private_State CMGame_MixinsScroll3_State as State for _This.Page = C_Private_State_Null;
		for (Frame_Scroll in State.Frame_Scrolls) {
			declare K_Private_Instance CMGame_MixinsScroll3_Instance as Instance for Frame_Scroll = C_Private_Instance_Null;
			if (Instance.ScrollOffset != Frame_Scroll.ScrollOffset || Instance.ScrollAnimOffset != Frame_Scroll.ScrollAnimOffset) {
				// Convert mousewheel vertical scroll into horizontal scroll
				if (Instance.Properties.Direction == C_Direction_Column && Frame_Scroll.ScrollOffset.Y != 0.) {
					Frame_Scroll.ScrollOffset.X -= Frame_Scroll.ScrollOffset.Y;
					Frame_Scroll.ScrollOffset.Y = 0.;
				}
				Instance = Private_SetScrollOffset(_This, Instance, Frame_Scroll.ScrollOffset, Frame_Scroll.ScrollAnimOffset);
				// Abort animation if user scrolls manually
				Instance.Animation.IsPlaying = False;
			} else if (Instance.Animation.IsPlaying) {
				if (
					Now > Instance.Animation.StartTime &&
					Now < Instance.Animation.StartTime + Instance.Animation.Duration
				) {
					Instance.Controls.Frame_Scroll.ScrollOffset = <
						AL::EaseOutQuad(
							Now - Instance.Animation.StartTime,
							Instance.Animation.OffsetStart.X,
							Instance.Animation.OffsetDelta.X,
							Instance.Animation.Duration
						),
						AL::EaseOutQuad(
							Now - Instance.Animation.StartTime,
							Instance.Animation.OffsetStart.Y,
							Instance.Animation.OffsetDelta.Y,
							Instance.Animation.Duration
						)
					>;
					Instance = Private_SetScrollOffset(_This, Instance, Frame_Scroll.ScrollOffset, Frame_Scroll.ScrollAnimOffset);
				} else if (Now >= Instance.Animation.StartTime + Instance.Animation.Duration) {
					Instance.Animation.IsPlaying = False;
					Instance.Controls.Frame_Scroll.ScrollOffset = Instance.Animation.OffsetStart + Instance.Animation.OffsetDelta;
					Instance = Private_SetScrollOffset(_This, Instance, Frame_Scroll.ScrollOffset, Frame_Scroll.ScrollAnimOffset);
					Instance.Animation.IsPlaying = False;
				}
			}

			if (Instance.Controls.Frame_ScrollBar != Null && Component_ScrollBar::IsMouseDragging(_This, Instance.Controls.Frame_ScrollBar)) {
				declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollOffset, Instance.Properties.Direction);
				ScrollOffsetFlex.Cross = Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollMax, Instance.Properties.Direction).Cross * Component_ScrollBar::GetMouseDraggingScrollRatio(Instance.Controls.Frame_ScrollBar);
				Instance.Controls.Frame_Scroll.ScrollOffset = Private_GetScreenPos(ScrollOffsetFlex, Instance.Properties.Direction);
			}

			if (Instance.IsFocused && Instance.IsNavigationEnabled) {
				for (Input in NavigationInputs::GetInputList(_This)) {
					Instance = Private_ApplyInput(_This, Instance, Input);
				}
			}
		}
	}

	for (Event in _This.PendingEvents) {
		switch (Event.Type) {
			case CMlScriptEvent::Type::MouseOver: {
				if (Event.Control != Null && !Mixin_Navigation::MouseIsSleeping(_This)) {
					declare Boolean CMGame_MixinsScroll3_IsElementZone for Event.Control = False;
					if (CMGame_MixinsScroll3_IsElementZone) {
						declare CMlFrame CMGame_MixinsScroll3_FrameScrollOwner for Event.Control = Null;
						if (CMGame_MixinsScroll3_FrameScrollOwner != Null) {
							declare K_Private_Instance CMGame_MixinsScroll3_Instance as Instance for CMGame_MixinsScroll3_FrameScrollOwner = C_Private_Instance_Null;
							if (Instance.IsNavigationEnabled) {
								// Event.Control will be destroyed by the C++ at the end of the frame. Create a copy to keep a valid reference to the control after the next yield.
								declare CMlControl LocalCopyEventControl = Event.Control;

								if (Instance.IsFocused) {
									Instance = Private_FocusElement(_This, Instance, LocalCopyEventControl);
								} else {
									Private_RequestFocus(_This, Instance, LocalCopyEventControl, C_EventType_ElementFocusUpdated, True);
								}
							}
						}
					}
				}
			}
			case CMlScriptEvent::Type::MouseClick: {
				if (Event.Control != Null && !Mixin_Navigation::MouseIsSleeping(_This)) {
					declare Boolean CMGame_MixinsScroll3_IsElementZone for Event.Control = False;
					if (CMGame_MixinsScroll3_IsElementZone) {
						declare CMlFrame CMGame_MixinsScroll3_FrameScrollOwner for Event.Control = Null;
						if (CMGame_MixinsScroll3_FrameScrollOwner != Null) {
							declare K_Private_Instance CMGame_MixinsScroll3_Instance as Instance for CMGame_MixinsScroll3_FrameScrollOwner = C_Private_Instance_Null;
							if (Instance.IsNavigationEnabled) {
								// Event.Control will be destroyed by the C++ at the end of the frame. Create a copy to keep a valid reference to the control after the next yield.
								declare CMlControl LocalCopyEventControl = Event.Control;

								if (Instance.IsFocused) {
									Instance = Private_SelectElement(_This, Instance, LocalCopyEventControl, True);
								} else {
									Private_RequestFocus(_This, Instance, LocalCopyEventControl, C_EventType_ElementSelected, True);
								}
							}
						}
					}
				}
			}
		}
	}
}

/// Get the required number of elements to put in the scroll frame for the given grid size
Integer GetRequiredElementsNb(Int2 _GridSize, Integer _Direction) {
	declare K_Private_FlexInt2 GridSizeFlex = Private_GetFlexSize(_GridSize, _Direction);
	return ML::Max(0, (GridSizeFlex.Main * GridSizeFlex.Cross) + GridSizeFlex.Main);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the recommended window size to display all the elements of the grid
Vec2 GetRecommendedWindowSize(Vec2 _ElementSize, Vec2 _GutterSize, Int2 _GridSize, Box::K_Padding _WindowPadding) {
	declare Vec2 ElementAndGutterSize = _ElementSize + _GutterSize;
	return <
		_WindowPadding.Left + _WindowPadding.Right + (ElementAndGutterSize.X * _GridSize.X) - _GutterSize.X,
		_WindowPadding.Top + _WindowPadding.Bottom + (ElementAndGutterSize.Y * _GridSize.Y) - _GutterSize.Y
	>;
}
Vec2 GetRecommendedWindowSize(Vec2 _ElementSize, Vec2 _GutterSize, Int2 _GridSize) {
	return GetRecommendedWindowSize(_ElementSize, _GutterSize, _GridSize, Box::C_Padding_Null);
}

/// Process a `ScrollFocusRequested` event
Void PassOnScrollFocusRequested(CMlScript _This, K_Event _Event) {
	declare K_Private_Instance CMGame_MixinsScroll3_Instance as Instance for _Event.Frame_Scroll = C_Private_Instance_Null;
	switch (_Event.Private_RelayType) {
		case C_EventType_ElementSelected: {
			Instance = Private_SelectElementKey(_This, Instance, _Event.Element.Key, _Event.IsMouse);
		}
		case C_EventType_ElementInputRelayed: {
			Private_RelayInputElementKey(_This, Instance, _Event.Element.Key, _Event.Input);
		}
		case C_EventType_ElementFocusUpdated: {
			Instance = Private_FocusElementKey(_This, Instance, _Event.Element.Key);
		}
	}
}
