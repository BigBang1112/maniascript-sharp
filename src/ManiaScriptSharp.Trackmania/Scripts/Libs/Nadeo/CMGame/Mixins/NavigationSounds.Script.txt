/**
 * Handle the sound of the navigation in the menus
 *
 * Available props:
 * - nav-sounds-focus (Text): the id of the sound to play when the control is focused
 * - nav-sounds-unfocus (Text): the id of the sound to play when the control is unfocused
 * - nav-sounds-select (Text): the id of the sound to play when the control is selected
 */
#Const Version "1.0.1"
#Const ScriptName "Libs/Nadeo/CMGame/Mixins/NavigationSounds.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Menus/Audio@2.Script.txt" as Audio
#Include "Libs/Nadeo/CMGame/Mixins/Navigation@2.Script.txt" as Navigation
#Include "Libs/Nadeo/CMGame/Mixins/Scroll@2.Script.txt" as Scroll
#Include "Libs/Nadeo/CMGame/Utils/ManiaView@3.Script.txt" as MV

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_Private_NavigationSounds {
	Text SoundId_Focus;
	Text SoundId_Unfocus;
	Text SoundId_Select;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_MixinName "CMGame_Mixins_NavigationSounds"
#Const C_MixinClass "cmgame-mixins-navigation-sounds"

#Const C_Prop_Focus "nav-sounds-focus"
#Const C_Prop_Unfocus "nav-sounds-unfocus"
#Const C_Prop_Select "nav-sounds-select"

#Const C_Private_Action_Null 0
#Const C_Private_Action_Focus 1
#Const C_Private_Action_Unfocus 2
#Const C_Private_Action_Select 3

#Const C_Private_NavigationSounds_Null K_Private_NavigationSounds {
	SoundId_Focus = "",
	SoundId_Unfocus = "",
	SoundId_Select = ""
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Play the navigation sound of a control
Void Private_PlayNavigationSound(CMlScript _This, CMlControl _Control, Integer _Action) {
	if (_Control == Null) return;

	declare K_Private_NavigationSounds CMGame_NavigationSounds_NavigationSounds for _Control = C_Private_NavigationSounds_Null;
	switch (_Action) {
		case C_Private_Action_Focus: {
			if (CMGame_NavigationSounds_NavigationSounds.SoundId_Focus == "") {
				Audio::StartSound(_This, Audio::C_SoundId_Menu_Focus);
			} else {
				Audio::StartSound(_This, CMGame_NavigationSounds_NavigationSounds.SoundId_Focus);
			}
		}
		case C_Private_Action_Unfocus: {
			if (CMGame_NavigationSounds_NavigationSounds.SoundId_Unfocus != "") {
				Audio::StartSound(_This, CMGame_NavigationSounds_NavigationSounds.SoundId_Unfocus);
			}
		}
		case C_Private_Action_Select: {
			if (CMGame_NavigationSounds_NavigationSounds.SoundId_Select == "") {
				Audio::StartSound(_This, Audio::C_SoundId_Menu_Select);
			} else {
				Audio::StartSound(_This, CMGame_NavigationSounds_NavigationSounds.SoundId_Select);
			}
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialiaze navigation sounds mixin instance
Void Private_Hook_MixinInit(CMlControl _Control) {
	declare K_Private_NavigationSounds CMGame_NavigationSounds_NavigationSounds for _Control = C_Private_NavigationSounds_Null;
	CMGame_NavigationSounds_NavigationSounds = C_Private_NavigationSounds_Null;

	CMGame_NavigationSounds_NavigationSounds.SoundId_Focus = MV::GetMixinProp_Text(_Control, C_Prop_Focus, CMGame_NavigationSounds_NavigationSounds.SoundId_Focus);
	CMGame_NavigationSounds_NavigationSounds.SoundId_Unfocus = MV::GetMixinProp_Text(_Control, C_Prop_Unfocus, CMGame_NavigationSounds_NavigationSounds.SoundId_Unfocus);
	CMGame_NavigationSounds_NavigationSounds.SoundId_Select = MV::GetMixinProp_Text(_Control, C_Prop_Select, CMGame_NavigationSounds_NavigationSounds.SoundId_Select);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update navigation sounds mixin
Void Private_Hook_PageAfterYield(CMlScript _This) {
	for (Event in Navigation::GetPendingEvents(_This)) {
		switch (Event.Type) {
			case Navigation::C_EventType_Focus: {
				Private_PlayNavigationSound(_This, Event.Control, C_Private_Action_Focus);
			}
			case Navigation::C_EventType_Unfocus: {
				Private_PlayNavigationSound(_This, Event.Control, C_Private_Action_Unfocus);
			}
			case Navigation::C_EventType_Select: {
				Private_PlayNavigationSound(_This, Event.Control, C_Private_Action_Select);
			}
		}
	}

	foreach (Event in Scroll::GetPendingEvents(_This)) {
		switch (Event.Type) {
			case Scroll::C_EventType_Focus: {
				if (Event.From != Null) {
					Private_PlayNavigationSound(_This, Event.From, C_Private_Action_Unfocus);
				}
				if (Event.To != Null) {
					Private_PlayNavigationSound(_This, Event.To, C_Private_Action_Focus);
				}
			}
			case Scroll::C_EventType_Select: {
				if (Event.To != Null) {
					Private_PlayNavigationSound(_This, Event.To, C_Private_Action_Select);
				}
			}
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set a custom unfocus sound for a control
Void SetUnfocusSound(CMlControl _Control, Text _SoundId) {
	declare K_Private_NavigationSounds CMGame_NavigationSounds_NavigationSounds for _Control = C_Private_NavigationSounds_Null;
	CMGame_NavigationSounds_NavigationSounds.SoundId_Unfocus = _SoundId;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set a custom focus sound for a control
Void SetFocusSound(CMlControl _Control, Text _SoundId) {
	declare K_Private_NavigationSounds CMGame_NavigationSounds_NavigationSounds for _Control = C_Private_NavigationSounds_Null;
	CMGame_NavigationSounds_NavigationSounds.SoundId_Focus = _SoundId;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set a custom select sound for a control
Void SetSelectSound(CMlControl _Control, Text _SoundId) {
	declare K_Private_NavigationSounds CMGame_NavigationSounds_NavigationSounds for _Control = C_Private_NavigationSounds_Null;
	CMGame_NavigationSounds_NavigationSounds.SoundId_Select = _SoundId;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the navigation sounds mixin
MV::K_Mixin Mixin() {
	return MV::K_Mixin {
		Name = C_MixinName,
		Path = ScriptName,
		Class = C_MixinClass,
		Hooks = [
			MV::C_Hook_MixinInit => """{{{C_MixinName}}}::Private_Hook_MixinInit(MixinInstance)""",
			MV::C_Hook_PageAfterYield => """{{{C_MixinName}}}::Private_Hook_PageAfterYield"""
		],
		DefineProps = [
			C_Prop_Focus,
			C_Prop_Unfocus,
			C_Prop_Select
		]
	};
}
