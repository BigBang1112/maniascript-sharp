/**
 *	Mixin: ScrollClampedControl
 *
 *	A control that will be clamped to the top or bottom of a
 *	scroll instead of leaving the scroll clip
 *
 *	Available props:
 *	- scroll-clamped-control-type -> select how the clamped control will behave (C_ClampedControlType_Top, C_ClampedControlType_Bottom, C_ClampedControlType_Both [default])
 *	- scroll-clamped-control-valign -> vertical alignment of the clamped control (C_ClampedControlVAlign_Top [default], C_ClampedControlVAlign_Center, C_ClampedControlVAlign_Bottom)
 *	- scroll-clamped-control-height -> height of the clamped control
 */
#Const Version "1.0.0"
#Const ScriptName "Libs/Nadeo/CMGame/Mixins/ScrollClampedControl.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Utils/ManiaView@3.Script.txt" as MV

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constant
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_MixinName "CMGameScrollClampedControl"
#Const C_MixinClass "cmgame-mixins-scroll-clamped-control"

#Const C_ClampedControlType_Top 0
#Const C_ClampedControlType_Bottom 1
#Const C_ClampedControlType_Both 2

#Const C_ClampedControlVAlign_Top 0
#Const C_ClampedControlVAlign_Center 1
#Const C_ClampedControlVAlign_Bottom 2

#Const C_Prop_ClampedControlType "scroll-clamped-control-type"
#Const C_Prop_ClampedControlVAlign "scroll-clamped-control-valign"
#Const C_Prop_ClampedControlHeight "scroll-clamped-control-height"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the mixin
MV::K_Mixin Mixin() {
	return MV::K_Mixin {
		Name = C_MixinName,
		Path = ScriptName,
		Class = C_MixinClass,
		DefineProps = [
			C_Prop_ClampedControlType,
			C_Prop_ClampedControlVAlign,
			C_Prop_ClampedControlHeight
		]
	};
}
