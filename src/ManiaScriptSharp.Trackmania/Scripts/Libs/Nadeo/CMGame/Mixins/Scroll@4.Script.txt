/**
 * Scroll mixin
 * ============
 *
 * Scroll through a list of elements
 *
 * Props
 * -----
 *
 * - `scroll-element-size` (`Vec2`): the size of one element
 * - `scroll-gutter-size` (`Vec2`): the size of the verical and horizontal gutter between the elements
 * - `scroll-line-element-nb` (`Integer`): the number of elements in a line
 * - `scroll-window-size` (`Vec2`): the size the scroll window
 * - `scroll-window-padding` (`Padding`): the padding of the scroll window
 * - `scroll-mouse-step-size` (`Vec2`): the size of one step of mousewheel scroll
 * - `scroll-direction` (`Integer`): the direction of the scroll
 * - `scroll-previous-line-visible-ratio` (`Real`): how much of the previous line is visible when scrolling to an element
 * - `scroll-next-line-visible-ratio` (`Real`): how much of the next line is visible when scrolling to an element
 * - `scroll-previous-line-visible-length` (`Real`): how much content is visible before the focused element
 * - `scroll-next-line-visible-length` (`Real`): how much content is visible after the focused element
 * - `scroll-scrollbar-id` (`Text`): the id of the scrollbar to synchronize with the scroll
 */

/** Convert a screen coordinate or size to a flex coordinate or size
 * The main axis is the axis running in the direction the scroll elements are laid out in (for example, as a row across the page, or a column down the page)
 * The cross axis is the axis running perpendicular to the direction the scroll elements are laid out in
 * ref: CSS flexbox
 */
K_Private_FlexVec2 Private_GetFlexPos(Vec2 _ScreenPos, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: return K_Private_FlexVec2 { Main = _ScreenPos.X, Cross = _ScreenPos.Y };
		case C_Direction_Column: return K_Private_FlexVec2 { Main = -_ScreenPos.Y, Cross = -_ScreenPos.X };
	}
	return K_Private_FlexVec2 { Main = _ScreenPos.X, Cross = _ScreenPos.Y };
}
K_Private_FlexVec2 Private_GetFlexSize(Vec2 _ScreenSize, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: return K_Private_FlexVec2 { Main = _ScreenSize.X, Cross = _ScreenSize.Y };
		case C_Direction_Column: return K_Private_FlexVec2 { Main = _ScreenSize.Y, Cross = _ScreenSize.X };
	}
	return K_Private_FlexVec2 { Main = _ScreenSize.X, Cross = _ScreenSize.Y };
}
K_Private_FlexInt2 Private_GetFlexSize(Int2 _ScreenSize, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: return K_Private_FlexInt2 { Main = _ScreenSize.X, Cross = _ScreenSize.Y };
		case C_Direction_Column: return K_Private_FlexInt2 { Main = _ScreenSize.Y, Cross = _ScreenSize.X };
	}
	return K_Private_FlexInt2 { Main = _ScreenSize.X, Cross = _ScreenSize.Y };
}
K_Private_FlexPadding Private_GetFlexPadding(Struct_Box::K_Padding _ScreenPadding, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: {
			return K_Private_FlexPadding {
				MainLeft = _ScreenPadding.Left,
				MainRight = _ScreenPadding.Right,
				CrossTop = _ScreenPadding.Top,
				CrossBottom = _ScreenPadding.Bottom
			};
		}
		case C_Direction_Column: {
			return K_Private_FlexPadding {
				MainLeft = _ScreenPadding.Top,
				MainRight = _ScreenPadding.Bottom,
				CrossTop = _ScreenPadding.Left,
				CrossBottom = _ScreenPadding.Right
			};
		}
	}
	return K_Private_FlexPadding {
		MainLeft = _ScreenPadding.Left,
		MainRight = _ScreenPadding.Right,
		CrossTop = _ScreenPadding.Top,
		CrossBottom = _ScreenPadding.Bottom
	};
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Convert a flex coordinate or size to a screen coordinate or size
Vec2 Private_GetScreenPos(K_Private_FlexVec2 _FlexPos, Integer _Direction) {
	switch (_Direction) {
		case C_Direction_Row: return <_FlexPos.Main, _FlexPos.Cross>;
		case C_Direction_Column: return <-_FlexPos.Cross, -_FlexPos.Main>;
	}
	return <_FlexPos.Main, _FlexPos.Cross>;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add `K_Private_FlexVec2`
K_Private_FlexVec2 Private_Add(K_Private_FlexVec2 _FlexVec2A, K_Private_FlexVec2 _FlexVec2B) {
	return K_Private_FlexVec2 {
		Main = _FlexVec2A.Main + _FlexVec2B.Main,
		Cross = _FlexVec2A.Cross + _FlexVec2B.Cross
	};
}
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add `K_Private_Vec2`
K_Private_Vec2 Private_Add(K_Private_Vec2 _VecA, K_Private_Vec2 _VecB) {
	return K_Private_Vec2 {
		Flex = Private_Add(_VecA.Flex, _VecB.Flex),
		Screen = _VecA.Screen + _VecB.Screen
	};
}

/// Find in which scroll element the given control is
CMlControl Private_FindScrollElement(CMlControl _Control) {
	if (_Control == Null) return Null;

	declare Boolean CMGame_MixinsScroll4_ScrollElementHasCache for _Control = False;
	declare CMlControl CMGame_MixinsScroll4_ScrollElementOwner for _Control <=> Null;
	if (CMGame_MixinsScroll4_ScrollElementHasCache) {
		return CMGame_MixinsScroll4_ScrollElementOwner;
	}

	declare CMlControl Control <=> _Control;
	while (!Control.HasClass(C_Class_Element) && Control.Parent != Null) {
		Control <=> Control.Parent;
	}

	CMGame_MixinsScroll4_ScrollElementHasCache = True;
	if (Control.HasClass(C_Class_Element)) {
		CMGame_MixinsScroll4_ScrollElementOwner <=> Control;
	} else {
		CMGame_MixinsScroll4_ScrollElementOwner <=> Null;
	}

	return CMGame_MixinsScroll4_ScrollElementOwner;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the scroll element with the requested key
K_Element Private_GetElementWithKey(K_Private_Instance _Instance, Integer _ElementKey) {
	for (Element in _Instance.ElementList) {
		if (Element.Key == _ElementKey) {
			return Element;
		}
	}
	return C_Element_Null;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the current scroll offset
Vec2 Private_GetScrollOffset(K_Private_Instance _Instance) {
	if (_Instance.Animation.IsPlaying && _Instance.Animation.StartTime + _Instance.Animation.Duration <= Now) {
		return _Instance.Animation.OffsetStart + _Instance.Animation.OffsetDelta;
	}
	return _Instance.ScrollOffset;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the scrollbar range when the content of the scroll window change
Void Private_UpdateScrollBarRange(K_Private_Instance _Instance) {
	if (_Instance.Controls.Frame_ScrollBar != Null) {
		declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(_Instance), _Instance.Properties.Direction);
		declare Vec2 Range = <ML::Abs(ScrollOffsetFlex.Cross), ML::Abs(ScrollOffsetFlex.Cross) + _Instance.Properties.WindowSize.Flex.Cross>;
		Component_ScrollBar::SetRange(_Instance.Controls.Frame_ScrollBar, Range, _Instance.ScrollBarFullLength);
	}
}

/// Check if an element is focused
Boolean Private_IsElementFocused(K_Private_Instance _Instance, Integer _ElementKey) {
	return (_Instance.IsFocused && _ElementKey == _Instance.FocusedElementKey);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the scroll focused element
K_Private_Instance Private_SetFocusedElementKey(CMlScript _Context, K_Private_Instance _Instance, Integer _FocusedElementKey, Boolean _IsInit) {
	declare Integer FocusedElementKey = -1;
	if (_Instance.ElementListCount > 0) {
		FocusedElementKey = ML::Clamp(_FocusedElementKey, 0, _Instance.ElementListCount - 1);
	}

	if (!_IsInit && _Instance.FocusedElementKey == FocusedElementKey) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	declare Integer PreviouslyFocusedElementKey = Instance.FocusedElementKey;
	Instance.FocusedElementKey = FocusedElementKey;

	declare K_Private_State CMGame_MixinsScroll4_State as State for _Context.Page = C_Private_State_Null;
	declare K_Element From = C_Element_Null;
	declare K_Element To = C_Element_Null;
	for (Key => Element in Instance.ElementList) {
		// Update focused state
		if (Element.IsFocused != Private_IsElementFocused(Instance, Element.Key)) {
			Instance.ElementList[Key].IsFocused = Private_IsElementFocused(Instance, Element.Key);
		}
		if (Element.Key == PreviouslyFocusedElementKey) {
			From = Instance.ElementList[Key];
		}
		if (Element.Key == Instance.FocusedElementKey) {
			To = Instance.ElementList[Key];
		}
	}
	if (To.Control != Null && Instance.IsFocused) {
		Mixin_Navigation::SetFocus(To.Control, Mixin_Navigation::C_ActionInput_Null);
	}
	State.CollectEvents.add(K_Event {
		Type = C_EventType_ElementFocusUpdated,
		Frame_Scroll = Instance.Controls.Frame_Scroll,
		From = From,
		To = To
	});

	return Instance;
}
K_Private_Instance Private_SetFocusedElementKey(CMlScript _Context, K_Private_Instance _Instance, Integer _FocusedElementKey) {
	return Private_SetFocusedElementKey(_Context, _Instance, _FocusedElementKey, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Scroll from a position to another
K_Private_Instance Private_ScrollFromOffsetToOffset(
	K_Private_Instance _Instance,
	K_Private_FlexVec2 _From,
	K_Private_FlexVec2 _To,
	Boolean _IsAnimated
) {
	if (_To.Cross == _From.Cross) return _Instance;

	// Assigning a new value to `ScrollOffset` like that does not work here:
	// `Instance.Controls.Frame_Scroll.ScrollOffset.Y = _To.Cross;`
	// `ScrollOffset` will simply ignore the assignment and will keep its original value
	// To get around this behaviour we create an animation with no duration
	// Assigning the value in the next yield is working

	declare K_Private_Instance Instance = _Instance;

	declare K_Private_FlexVec2 OffsetDelta = K_Private_FlexVec2 {
		Main = _To.Main - _From.Main,
		Cross = _To.Cross - _From.Cross
	};

	declare Integer AnimDuration = 0;
	if (_IsAnimated) AnimDuration = C_Private_AnimDuration_Scroll;

	Instance.Animation.IsPlaying = True;
	Instance.Animation.StartTime = Now;
	Instance.Animation.Duration = AnimDuration;
	Instance.Animation.OffsetStart = Private_GetScreenPos(_From, Instance.Properties.Direction);
	Instance.Animation.OffsetDelta = Private_GetScreenPos(OffsetDelta, Instance.Properties.Direction);

	return Instance;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Scroll to the given scroll ratio
K_Private_Instance Private_ScrollToRatio(K_Private_Instance _Instance, Real _Ratio, Boolean _IsAnimated) {
	declare K_Private_Instance Instance = _Instance;

	declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
	declare K_Private_FlexVec2 ScrollOffsetTargetFlex = ScrollOffsetFlex;
	ScrollOffsetTargetFlex.Cross = Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollMax, Instance.Properties.Direction).Cross * _Ratio;

	return Private_ScrollFromOffsetToOffset(Instance, ScrollOffsetFlex, ScrollOffsetTargetFlex, _IsAnimated);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Scroll to the focused elements
K_Private_Instance Private_ScrollToFocusedElementKey(K_Private_Instance _Instance, Integer _ElementKey, Boolean _IsAnimated) {
	declare K_Private_Instance Instance = _Instance;

	declare Integer FocusedElementLineIndex = 0;
	if (Instance.Properties.LineElementNb > 0) {
		FocusedElementLineIndex = _ElementKey / Instance.Properties.LineElementNb;
	}
	declare Vec2 FocusedElementVerticalRange = <Instance.Properties.WindowPadding.Flex.CrossTop + (FocusedElementLineIndex * (Instance.Properties.ElementSize.Flex.Cross + Instance.Properties.GutterSize.Flex.Cross)), 0.>;
	FocusedElementVerticalRange.Y = FocusedElementVerticalRange.X + Instance.Properties.ElementSize.Flex.Cross;

	// Try to show a bit of the elements on the line above or below the focused element
	declare Real PreviousLinePreviewSizeCross = 0.;
	if (
		Instance.Properties.PreviousLineVisibleRatio > 0. ||
		Instance.Properties.PreviousLineVisibleLength > 0.
	) {
		PreviousLinePreviewSizeCross = Instance.Properties.GutterSize.Flex.Cross + (Instance.Properties.ElementSize.Flex.Cross * Instance.Properties.PreviousLineVisibleRatio) + Instance.Properties.PreviousLineVisibleLength;
	}
	declare Real NextLinePreviewSizeCross = 0.;
	if (
		Instance.Properties.NextLineVisibleRatio > 0. ||
		Instance.Properties.NextLineVisibleLength > 0.
	) {
		NextLinePreviewSizeCross = Instance.Properties.GutterSize.Flex.Cross + (Instance.Properties.ElementSize.Flex.Cross * Instance.Properties.NextLineVisibleRatio) + Instance.Properties.NextLineVisibleLength;
	}

	declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
	declare K_Private_FlexVec2 ScrollOffsetTargetFlex = ScrollOffsetFlex;
	if (FocusedElementVerticalRange.X < ScrollOffsetFlex.Cross + PreviousLinePreviewSizeCross) {
		ScrollOffsetTargetFlex.Cross = ML::Max(0., FocusedElementVerticalRange.X - PreviousLinePreviewSizeCross);
	} else if (FocusedElementVerticalRange.Y > ScrollOffsetFlex.Cross + Instance.Properties.WindowSize.Flex.Cross - NextLinePreviewSizeCross) {
		ScrollOffsetTargetFlex.Cross = ML::Min(Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollMax, Instance.Properties.Direction).Cross, FocusedElementVerticalRange.Y - Instance.Properties.WindowSize.Flex.Cross + NextLinePreviewSizeCross);
	}

	return Private_ScrollFromOffsetToOffset(Instance, ScrollOffsetFlex, ScrollOffsetTargetFlex, _IsAnimated);
}

/// Scroll to the focused elements
K_Private_Instance Private_ScrollToFocusedElement(K_Private_Instance _Instance, Boolean _IsAnimated) {
	return Private_ScrollToFocusedElementKey(_Instance, _Instance.FocusedElementKey, _IsAnimated);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Scroll until the element key is at the top of the scroll window
K_Private_Instance Private_ScrollToElementKeyTop(K_Private_Instance _Instance, Integer _ElementKey, Real _Offset, Boolean _IsAnimated) {
	declare K_Private_Instance Instance = _Instance;

	declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);

	declare Integer LineNb = 0;
	if (Instance.Properties.LineElementNb > 0) {
		LineNb = _ElementKey / Instance.Properties.LineElementNb;
	}
	declare Real ElementSizeCross = Instance.Properties.ElementSize.Flex.Cross + Instance.Properties.GutterSize.Flex.Cross;
	declare K_Private_FlexVec2 ScrollOffsetTargetFlex = ScrollOffsetFlex;
	ScrollOffsetTargetFlex.Cross = Instance.Properties.WindowPadding.Flex.CrossTop + (LineNb * ElementSizeCross) - _Offset;

	return Private_ScrollFromOffsetToOffset(Instance, ScrollOffsetFlex, ScrollOffsetTargetFlex, _IsAnimated);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the number of lines that can be visible at the same time in the scroll window
Integer Private_GetVisibleLinesNb(Real _WindowLength, Real _ElementLength, Real _GutterLength) {
	declare Real ElementAndGutterSize = _ElementLength + _GutterLength;
	if (ElementAndGutterSize == 0.) return 0;
	return ML::CeilingInteger((_WindowLength + _GutterLength) / ElementAndGutterSize) + 1;
}
Integer Private_GetVisibleLinesNb(Vec2 _WindowSize, Vec2 _ElementSize, Vec2 _GutterSize, Integer _Direction) {
	declare Integer LinesNb = 0;
	if (_Direction == C_Direction_Row) {
		LinesNb = Private_GetVisibleLinesNb(_WindowSize.Y, _ElementSize.Y, _GutterSize.Y);
	} else if (_Direction == C_Direction_Column) {
		LinesNb = Private_GetVisibleLinesNb(_WindowSize.X, _ElementSize.X, _GutterSize.X);
	}
	return LinesNb;
}

/// Set the scroll offset
K_Private_Instance Private_SetScrollOffset(
	CMlScript _Context,
	K_Private_Instance _Instance,
	Vec2 _ScrollOffset,
	Vec2 _ScrollAnimOffset,
	Boolean _IsInit
) {
	if (!_IsInit && _Instance.ScrollOffset == _ScrollOffset && _Instance.ScrollAnimOffset == _ScrollAnimOffset) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	Instance.ScrollOffset = _ScrollOffset;
	Instance.ScrollAnimOffset = _ScrollAnimOffset;

	Instance.Controls.Quad_WindowBackground.RelativePosition_V3 = -Instance.ScrollAnimOffset;

	// Find the range of element keys visible in the scroll window
	declare Int2 ElementKeyRange = <0, 0>;
	declare K_Private_Vec2 ElementAndGutterSize = Private_Add(Instance.Properties.ElementSize, Instance.Properties.GutterSize);
	declare K_Private_FlexVec2 ScrollAnimOffsetFlex = Private_GetFlexPos(Instance.ScrollAnimOffset, Instance.Properties.Direction);
	if (Instance.ElementListCount > 0 && ElementAndGutterSize.Flex.Cross > 0.) {
		ElementKeyRange.X = ML::Clamp(ML::Max(0, ML::FloorInteger(ML::Max(0., ScrollAnimOffsetFlex.Cross - Instance.Properties.WindowPadding.Flex.CrossTop) / ElementAndGutterSize.Flex.Cross)) * Instance.Properties.LineElementNb, 0, ML::Max(0, Instance.ElementListCount - 1));
		ElementKeyRange.Y = ML::Clamp(ElementKeyRange.X + ML::Max(0, (Private_GetVisibleLinesNb(Instance.Properties.WindowSize.Flex.Cross, Instance.Properties.ElementSize.Flex.Cross, Instance.Properties.GutterSize.Flex.Cross) * Instance.Properties.LineElementNb) - 1), ElementKeyRange.X, ML::Max(0, Instance.ElementListCount - 1));
	}

	// If the range changed, update the scroll
	if (_IsInit || Instance.ElementKeyRange != ElementKeyRange) {
		Instance.ElementKeyRange = ElementKeyRange;

		declare K_Event Event = K_Event {
			Type = C_EventType_ElementListUpdated,
			Frame_Scroll = Instance.Controls.Frame_Scroll
		};

		// Update the elements that left the scroll window
		declare Integer[] UpToDateElementKeys; // Keys of the elements that are already up to date
		declare Integer[] ReusableElementListArrayKeys; // Keys of the `Instance.ElementList` that are outside of the scroll window and can be reused to display new elements that enter the scroll window. CAUTION: This is not the element key but the key of the element in the `Instance.ElementList` array.
		for (Key => Element in Instance.ElementList) {
			if (
				Instance.ElementListCount <= 0 ||
				Element.Key == C_Private_ElementKey_Null ||
				Element.Key < ElementKeyRange.X ||
				Element.Key > ElementKeyRange.Y ||
				UpToDateElementKeys.exists(Element.Key)
			) {
				if (Element.Key != C_Private_ElementKey_Null) {
					declare K_Element UpdatedElement = Instance.ElementList[Key];
					UpdatedElement.Key = C_Private_ElementKey_Null;
					UpdatedElement.Control.Visible = False;

					declare Integer CMGame_MixinsScroll4_ElementKey for UpdatedElement.Control = C_Private_ElementKey_Null;
					CMGame_MixinsScroll4_ElementKey = UpdatedElement.Key;

					Instance.ElementList[Key] = UpdatedElement;
				}
				ReusableElementListArrayKeys.add(Key);
			} else if (_IsInit) {
				ReusableElementListArrayKeys.add(Key);
			} else {
				UpToDateElementKeys.add(Element.Key);
			}
		}

		// Update the elements that entered the scroll window
		if (Instance.ElementListCount > 0) {
			for (ElementKey, ElementKeyRange.X, ElementKeyRange.Y) {
				if (!UpToDateElementKeys.exists(ElementKey) && ReusableElementListArrayKeys.count > 0) {
					declare Integer ElementListArrayKey = ReusableElementListArrayKeys[0];
					declare K_Element UpdatedElement = Instance.ElementList[ElementListArrayKey];
					UpdatedElement.Key = ElementKey;
					UpdatedElement.Control.Visible = True;
					UpdatedElement.IsFocused = Private_IsElementFocused(Instance, UpdatedElement.Key);
					if (UpdatedElement.IsFocused && Instance.IsFocused) {
						Mixin_Navigation::SetFocus(UpdatedElement.Control, Mixin_Navigation::C_ActionInput_Null);
					}

					declare Integer CMGame_MixinsScroll4_ElementKey for UpdatedElement.Control = C_Private_ElementKey_Null;
					CMGame_MixinsScroll4_ElementKey = UpdatedElement.Key;

					declare K_Private_FlexVec2 RelativePositionFlex = K_Private_FlexVec2 { Main = 0., Cross = 0. };
					if (Instance.Properties.LineElementNb > 0) {
						RelativePositionFlex.Main = Instance.Properties.WindowPadding.Flex.MainLeft + ((UpdatedElement.Key % Instance.Properties.LineElementNb) * ElementAndGutterSize.Flex.Main);
						RelativePositionFlex.Cross = -Instance.Properties.WindowPadding.Flex.CrossTop - ((UpdatedElement.Key / Instance.Properties.LineElementNb) * ElementAndGutterSize.Flex.Cross);
					}
					UpdatedElement.Control.RelativePosition_V3 = Private_GetScreenPos(RelativePositionFlex, Instance.Properties.Direction);

					Event.UpdatedElementList.add(UpdatedElement);
					Instance.ElementList[ElementListArrayKey] = UpdatedElement;
					ReusableElementListArrayKeys.removekey(0);
				}
			}
		}

		Event.ElementList = Instance.ElementList;

		declare K_Private_State CMGame_MixinsScroll4_State as State for _Context.Page = C_Private_State_Null;
		State.CollectEvents.add(Event);
	}

	Private_UpdateScrollBarRange(Instance);

	return Instance;
}

/// Set the number of elements in the scroll
K_Private_Instance Private_SetElementListNb(
	CMlScript _Context,
	K_Private_Instance _Instance,
	Integer _ElementListCount,
	Boolean _IsInit
) {
	if (!_IsInit && _Instance.ElementListCount == _ElementListCount) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	Instance.ElementListCount = _ElementListCount;
	Instance.ScrollBarFullLength = 0.;

	declare K_Private_FlexVec2 ScrollMaxFlex = K_Private_FlexVec2 { Main = 0., Cross = 0. };
	declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
	if (Instance.Properties.LineElementNb > 0) {
		declare K_Private_Vec2 ElementAndGutterSize = Private_Add(Instance.Properties.ElementSize, Instance.Properties.GutterSize);
		declare Integer LinesNb = (Instance.ElementListCount + Instance.Properties.LineElementNb - 1) / Instance.Properties.LineElementNb;
		Instance.ScrollBarFullLength = ML::Max(0., (LinesNb * ElementAndGutterSize.Flex.Cross) - Instance.Properties.GutterSize.Flex.Cross + Instance.Properties.WindowPadding.Flex.CrossTop + Instance.Properties.WindowPadding.Flex.CrossBottom);
		ScrollMaxFlex.Cross = ML::Max(0., Instance.ScrollBarFullLength - Instance.Properties.WindowSize.Flex.Cross);
	}
	ScrollOffsetFlex.Cross = ML::Min(ScrollOffsetFlex.Cross, ScrollMaxFlex.Cross);
	Instance.Controls.Frame_Scroll.ScrollMax = Private_GetScreenPos(ScrollMaxFlex, Instance.Properties.Direction);
	Instance.Controls.Frame_Scroll.ScrollOffset = Private_GetScreenPos(ScrollOffsetFlex, Instance.Properties.Direction);

	// Add a bit of vertical scroll to be able to detect mouse vertical scrollwheel and convert it into horizontal scroll
	if (Instance.Properties.Direction == C_Direction_Column && Instance.Controls.Frame_Scroll.ScrollMax.Y == 0.) {
		Instance.Controls.Frame_Scroll.ScrollMin.Y = -Instance.Properties.MouseStepSize.Flex.Cross;
		Instance.Controls.Frame_Scroll.ScrollMax.Y = Instance.Properties.MouseStepSize.Flex.Cross;
	} else {
		Instance.Controls.Frame_Scroll.ScrollMin = <0., 0.>;
	}

	// Check that the currently focused element still exists
	Instance = Private_SetFocusedElementKey(_Context, Instance, Instance.FocusedElementKey);

	// When the number of elements change, we must force an initialization of the scroll offset to refresh all elements
	Instance = Private_SetScrollOffset(_Context, Instance, Private_GetScreenPos(ScrollOffsetFlex, Instance.Properties.Direction), Instance.Controls.Frame_Scroll.ScrollAnimOffset, True);

	return Instance;
}

/// Focus the next element in the line
K_Private_Instance Private_ApplyInput_NextElement(CMlScript _Context, K_Private_Instance _Instance, NavigationInputs::K_Input _Input) {
	declare K_Private_Instance Instance = _Instance;

	if (
		Instance.Properties.LineElementNb > 0 &&
		Instance.FocusedElementKey < Instance.ElementListCount - 1 &&
		Instance.FocusedElementKey % Instance.Properties.LineElementNb < Instance.Properties.LineElementNb - 1
	) {
		Instance = Private_SetFocusedElementKey(_Context, Instance, Instance.FocusedElementKey + 1);
		Instance = Private_ScrollToFocusedElement(Instance, True);
	} else {
		Mixin_Navigation::TriggerInput(Instance.Controls.Frame_Scroll, _Input.Type);
	}

	return Instance;
}

/// Focus the element in the next line
K_Private_Instance Private_ApplyInput_NextLine(CMlScript _Context, K_Private_Instance _Instance, NavigationInputs::K_Input _Input) {
	declare K_Private_Instance Instance = _Instance;

	if (Instance.FocusedElementKey < Instance.ElementListCount - Instance.Properties.LineElementNb) {
		Instance = Private_SetFocusedElementKey(_Context, Instance, Instance.FocusedElementKey + Instance.Properties.LineElementNb);
		Instance = Private_ScrollToFocusedElement(Instance, True);
	} else if (
		Instance.Properties.LineElementNb > 0 &&
		Instance.FocusedElementKey < Instance.ElementListCount - 1 &&
		Instance.FocusedElementKey / Instance.Properties.LineElementNb < (Instance.ElementListCount - 1) / Instance.Properties.LineElementNb
	) {
		Instance = Private_SetFocusedElementKey(_Context, Instance, Instance.ElementListCount - 1);
		Instance = Private_ScrollToFocusedElement(Instance, True);
	} else {
		// Scroll to the bottom when escaping the scroll
		declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
		declare K_Private_FlexVec2 ScrollMaxFlex = Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollMax, Instance.Properties.Direction);
		if (ScrollOffsetFlex.Cross < ScrollMaxFlex.Cross) {
			Instance = Private_ScrollToRatio(Instance, 1., True);
		}
		Mixin_Navigation::TriggerInput(Instance.Controls.Frame_Scroll, _Input.Type);
	}

	return Instance;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Focus the element in the previous line
K_Private_Instance Private_ApplyInput_PreviousLine(CMlScript _Context, K_Private_Instance _Instance, NavigationInputs::K_Input _Input) {
	declare K_Private_Instance Instance = _Instance;

	if (Instance.FocusedElementKey >= Instance.Properties.LineElementNb) {
		Instance = Private_SetFocusedElementKey(_Context, Instance, Instance.FocusedElementKey - Instance.Properties.LineElementNb);
		Instance = Private_ScrollToFocusedElement(Instance, True);
	} else {
		// Scroll to the top when escaping the scroll
		declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);
		if (ScrollOffsetFlex.Cross > 0.) {
			Instance = Private_ScrollToRatio(Instance, 0., True);
		}
		Mixin_Navigation::TriggerInput(Instance.Controls.Frame_Scroll, _Input.Type);
	}

	return Instance;
}

/// Apply a directionnal input (up, right, down, left) to the scroll
K_Private_Instance Private_ApplyInput(CMlScript _Context, K_Private_Instance _Instance, NavigationInputs::K_Input _Input) {
	declare K_Private_Instance Instance = _Instance;

	switch (_Input.Type) {
		case NavigationInputs::C_InputType_Up: {
			switch (Instance.Properties.Direction) {
				case C_Direction_Row: Instance = Private_ApplyInput_PreviousLine(_Context, Instance, _Input);
				case C_Direction_Column: Instance = Private_ApplyInput_PreviousElement(_Context, Instance, _Input);
			}
		}
		case NavigationInputs::C_InputType_Down: {
			switch (Instance.Properties.Direction) {
				case C_Direction_Row: Instance = Private_ApplyInput_NextLine(_Context, _Instance, _Input);
				case C_Direction_Column: Instance = Private_ApplyInput_NextElement(_Context, Instance, _Input);
			}
		}
		case NavigationInputs::C_InputType_Left: {
			switch (Instance.Properties.Direction) {
				case C_Direction_Row: Instance = Private_ApplyInput_PreviousElement(_Context, Instance, _Input);
				case C_Direction_Column: Instance = Private_ApplyInput_PreviousLine(_Context, Instance, _Input);
			}
		}
		case NavigationInputs::C_InputType_Right: {
			switch (Instance.Properties.Direction) {
				case C_Direction_Row: Instance = Private_ApplyInput_NextElement(_Context, Instance, _Input);
				case C_Direction_Column: Instance = Private_ApplyInput_NextLine(_Context, _Instance, _Input);
			}
		}
		default: {
			if (!_Input.IsFromMouse) {
				Private_RelayInputElementKey(_Context, Instance, Instance.FocusedElementKey, _Input);
			}
		}
	}

	return Instance;
}

/// Convert an array of prop values to a properties structure
K_Private_Properties Private_ToProperties(Text[Text] _PropValueList) {
	declare K_Private_Properties Properties = C_Private_Properties_Default;

	for (PropName => PropValue in _PropValueList) {
		switch (PropName) {
			case C_Prop_ElementSize: Properties.ElementSize.Screen = Cast::ToVec2(PropValue);
			case C_Prop_GutterSize: Properties.GutterSize.Screen = Cast::ToVec2(PropValue);
			case C_Prop_LineElementNb: Properties.LineElementNb = Cast::ToInteger(PropValue);
			case C_Prop_WindowSize: Properties.WindowSize.Screen = Cast::ToVec2(PropValue);
			case C_Prop_WindowPadding: Properties.WindowPadding.Screen = Struct_Box::ToPadding(PropValue);
			case C_Prop_MouseStepSize: Properties.MouseStepSize.Screen = Cast::ToVec2(PropValue);
			case C_Prop_Direction: Properties.Direction = Cast::ToInteger(PropValue);
			case C_Prop_PreviousLineVisibleRatio: Properties.PreviousLineVisibleRatio = Cast::ToReal(PropValue);
			case C_Prop_NextLineVisibleRatio: Properties.NextLineVisibleRatio = Cast::ToReal(PropValue);
			case C_Prop_PreviousLineVisibleLength: Properties.PreviousLineVisibleLength = Cast::ToReal(PropValue);
			case C_Prop_NextLineVisibleLength: Properties.NextLineVisibleLength = Cast::ToReal(PropValue);
			case C_Prop_ScrollBarId: Properties.ScrollBarId = PropValue;
		}
	}

	return Properties;
}

/// Initialiaze mixin instance
K_Private_Instance Private_InitializeNavigation(K_Private_Instance _Instance) {
	declare K_Private_Instance Instance = _Instance;

	for (Listener in Instance.NavigationListenerList) {
		EventEmitter::RemoveListener(Listener);
	}
	Instance.NavigationListenerList = [];

	Mixin_Navigation::NewGroup(Instance.Controls.Frame_Scroll, [Instance.Controls.Quad_WindowBackground]);
	Mixin_Navigation::NewControl_WithMouseActionList(Instance.Controls.Quad_WindowBackground, [
		Mixin_Navigation::NewAction(Mixin_Navigation::C_ActionInput_Up, Mixin_Navigation::C_ActionType_Null, Null),
		Mixin_Navigation::NewAction(Mixin_Navigation::C_ActionInput_Down, Mixin_Navigation::C_ActionType_Null, Null),
		Mixin_Navigation::NewAction(Mixin_Navigation::C_ActionInput_Left, Mixin_Navigation::C_ActionType_Null, Null),
		Mixin_Navigation::NewAction(Mixin_Navigation::C_ActionInput_Right, Mixin_Navigation::C_ActionType_Null, Null)
	]);

	declare CMlControl[] GroupControlList = [];
	for (Element in Instance.ElementList) {
		GroupControlList.add(Element.Control);
		Mixin_Navigation::NewControl(Element.Control);
		Instance.NavigationListenerList.add(
			EventEmitter::AddListener(Element.Control, [Mixin_Navigation::C_EventType_Focus, Mixin_Navigation::C_EventType_Select])
		);
	}
	if (GroupControlList.count > 0) {
		Mixin_Navigation::NewGroup(Instance.Controls.Quad_WindowBackground, GroupControlList);
	}

	Instance.NavigationListenerList.add(
		EventEmitter::AddListener(Instance.Controls.Frame_Scroll, [Mixin_Navigation::C_EventType_Focus, Mixin_Navigation::C_EventType_Unfocus])
	);

	Instance = Private_SetIsNavigationEnabled(Instance, Instance.IsNavigationEnabled, True);

	return Instance;
}

/// Initialiaze mixin instance
Void Private_Init(CMlScript _Context, CMlControl _Control) {
	assert(_Control is CMlFrame, """[CMGame/Mixins/Scroll@4] The scroll mixin can only be applied to a frame. `#{{{_Control.ControlId}}}` is not a frame.""");

	declare K_Private_Instance CMGame_MixinsScroll4_Instance as Instance for _Control = C_Private_Instance_Null;
	Instance = C_Private_Instance_Null;

	Instance.Controls.Frame_Scroll <=> (_Control as CMlFrame);

	// Sort the different types of controls in the scroll frame
	for (Control in Instance.Controls.Frame_Scroll.Controls) {
		if (Control.HasClass(C_Class_WindowBackground)) {
			assert(Control is CMlQuad, "[CMGame/Mixins/Scroll@4] Control with class `C_Class_WindowBackground` must be a quad.");
			assert(Instance.Controls.Quad_WindowBackground == Null, "[CMGame/Mixins/Scroll@4] You can only have one control with class `C_Class_WindowBackground` inside the scroll frame.");
			Instance.Controls.Quad_WindowBackground <=> (Control as CMlQuad);
		} else if (Control.HasClass(C_Class_Element)) {
			Control.Visible = False;
			Instance.ElementList.add(K_Element {
				Key = C_Private_ElementKey_Null,
				Control = Control,
				IsFocused = False
			});
		} else if (!Control.HasClass(C_Class_CustomControl)) {
			assert(False, """[CMGame/Mixins/Scroll@4] Control with id `#{{{Control.ControlId}}}` does not have a valid class. Every control in the scroll frame must use one of the following class: `C_Class_Element`, `C_Class_WindowBackground`, `C_Class_CustomControl`.""");
		}
	}
	assert(Instance.Controls.Quad_WindowBackground != Null, "[CMGame/Mixins/Scroll@4] You must add a quad with the class `C_Class_WindowBackgorund` inside the scroll frame. This quad will be invisible and generate the scroll events.");

	Instance.Properties = Private_ToProperties(MV::GetMixinPropValueList(Instance.Controls.Frame_Scroll, C_PropList));

	Instance.Controls.Quad_WindowBackground.Opacity = 0.;
	Instance.Controls.Frame_Scroll.ScrollActive = True;
	Instance.Controls.Frame_Scroll.ScrollMin = <0., 0.>;

	Instance = Private_InitializeNavigation(Instance);

	Instance = Private_SetScrollBarId(_Context, Instance, Instance.Properties.ScrollBarId);
	Instance = Private_SetElementSize(Instance, Instance.Properties.ElementSize.Screen, True);
	Instance = Private_SetGutterSize(Instance, Instance.Properties.GutterSize.Screen, True);
	Instance = Private_SetLineElementNb(Instance, Instance.Properties.LineElementNb, True);
	Instance = Private_SetWindowSize(Instance, Instance.Properties.WindowSize.Screen, True);
	Instance = Private_SetWindowPadding(Instance, Instance.Properties.WindowPadding.Screen, True);
	Instance = Private_SetMouseStepSize(Instance, Instance.Properties.MouseStepSize.Screen, True);

	Instance = Private_SetElementListNb(_Context, Instance, Instance.ElementListCount, True);
	Instance = Private_SetFocusedElementKey(_Context, Instance, Instance.FocusedElementKey, True);
	Instance = Private_SetIsFocused(_Context, Instance, Mixin_Navigation::IsFocused(Instance.Controls.Frame_Scroll), True);
	Instance = Private_SetScrollOffset(_Context, Instance, Instance.Controls.Frame_Scroll.ScrollOffset, Instance.Controls.Frame_Scroll.ScrollAnimOffset, True);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize the page
Void Private_Hook_PageMainInit(CMlScript _Context) {
	declare K_Private_State CMGame_MixinsScroll4_State as State for _Context.Page = C_Private_State_Null;
	State = C_Private_State_Null;

	_Context.Page.GetClassChildren(C_MixinClass, _Context.Page.MainFrame, True);
	for (Control in _Context.Page.GetClassChildren_Result) {
		declare CMlFrame Frame_Scroll = (Control as CMlFrame); //< Point to the frame instead of the array key
		State.Frame_Scrolls.add(Frame_Scroll);
	}
	for (Frame_Scroll in State.Frame_Scrolls) {
		Private_Init(_Context, Frame_Scroll);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the events
Void Private_Hook_PageBeforeManialinkEvents(CMlScript _Context) {
	declare K_Private_State CMGame_MixinsScroll4_State as State for _Context.Page = C_Private_State_Null;
	State.PendingEvents = State.CollectEvents;
	State.CollectEvents = [];
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the page
Void Private_Hook_PageAfterYield(CMlScript _Context) {
	if (_Context.PageIsVisible) {
		declare K_Private_State CMGame_MixinsScroll4_State as State for _Context.Page = C_Private_State_Null;
		for (Frame_Scroll in State.Frame_Scrolls) {
			declare K_Private_Instance CMGame_MixinsScroll4_Instance as Instance for Frame_Scroll = C_Private_Instance_Null;
			if (Instance.ScrollOffset != Frame_Scroll.ScrollOffset || Instance.ScrollAnimOffset != Frame_Scroll.ScrollAnimOffset) {
				// Send vertical mouse scroll event
				if (Instance.ScrollOffset.Y != Frame_Scroll.ScrollOffset.Y) {
					if (Instance.ScrollOffset.Y < Frame_Scroll.ScrollOffset.Y) {
						State.CollectEvents.add(K_Event {
							Type = C_EventType_MouseScrollDown,
							Frame_Scroll = Frame_Scroll,
							IsMouse = True
						});
					} else {
						State.CollectEvents.add(K_Event {
							Type = C_EventType_MouseScrollUp,
							Frame_Scroll = Frame_Scroll,
							IsMouse = True
						});
					}
				}
				// Convert mousewheel vertical scroll into horizontal scroll
				if (Instance.Properties.Direction == C_Direction_Column && Frame_Scroll.ScrollOffset.Y != 0.) {
					Frame_Scroll.ScrollOffset.X -= Frame_Scroll.ScrollOffset.Y;
					Frame_Scroll.ScrollOffset.Y = 0.;
				}
				Instance = Private_SetScrollOffset(_Context, Instance, Frame_Scroll.ScrollOffset, Frame_Scroll.ScrollAnimOffset);
				// Abort animation if user scrolls manually
				Instance.Animation.IsPlaying = False;
			} else if (Instance.Animation.IsPlaying) {
				if (
					Now > Instance.Animation.StartTime &&
					Now < Instance.Animation.StartTime + Instance.Animation.Duration
				) {
					Instance.Controls.Frame_Scroll.ScrollOffset = <
						AL::EaseOutQuad(
							Now - Instance.Animation.StartTime,
							Instance.Animation.OffsetStart.X,
							Instance.Animation.OffsetDelta.X,
							Instance.Animation.Duration
						),
						AL::EaseOutQuad(
							Now - Instance.Animation.StartTime,
							Instance.Animation.OffsetStart.Y,
							Instance.Animation.OffsetDelta.Y,
							Instance.Animation.Duration
						)
					>;
					Instance = Private_SetScrollOffset(_Context, Instance, Frame_Scroll.ScrollOffset, Frame_Scroll.ScrollAnimOffset);
				} else if (Now >= Instance.Animation.StartTime + Instance.Animation.Duration) {
					Instance.Animation.IsPlaying = False;
					Instance.Controls.Frame_Scroll.ScrollOffset = Instance.Animation.OffsetStart + Instance.Animation.OffsetDelta;
					Instance = Private_SetScrollOffset(_Context, Instance, Frame_Scroll.ScrollOffset, Frame_Scroll.ScrollAnimOffset);
					Instance.Animation.IsPlaying = False;
				}
			}

			if (Instance.Controls.Frame_ScrollBar != Null && Component_ScrollBar::IsMouseDragging(_Context, Instance.Controls.Frame_ScrollBar)) {
				declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollOffset, Instance.Properties.Direction);
				ScrollOffsetFlex.Cross = Private_GetFlexPos(Instance.Controls.Frame_Scroll.ScrollMax, Instance.Properties.Direction).Cross * Component_ScrollBar::GetMouseDraggingScrollRatio(Instance.Controls.Frame_ScrollBar);
				Instance.Controls.Frame_Scroll.ScrollOffset = Private_GetScreenPos(ScrollOffsetFlex, Instance.Properties.Direction);
			}

			if (Instance.IsFocused && Instance.IsNavigationEnabled) {
				for (Input in NavigationInputs::GetInputList(_Context)) {
					Instance = Private_ApplyInput(_Context, Instance, Input);
				}
			}

			for (Event in Mixin_Navigation::FlushListenerList(Instance.NavigationListenerList)) {
				if (Event.Target == Instance.Controls.Frame_Scroll) {
					switch (Event.Type) {
						case Mixin_Navigation::C_EventType_Focus: {
							Instance = Private_SetIsFocused(_Context, Instance, True);
						}
						case Mixin_Navigation::C_EventType_Unfocus: {
							Instance = Private_SetIsFocused(_Context, Instance, False);
						}
					}
				} else if (Event.Target != Null && Event.Target.HasClass(C_Class_Element)) {
					switch (Event.Type) {
						case Mixin_Navigation::C_EventType_Focus: {
							if (Event.InputType != Mixin_Navigation::C_ActionInput_Null) {
								Instance = Private_FocusElement(_Context, Instance, Event.Target);
							}
						}
						case Mixin_Navigation::C_EventType_Select: {
							Instance = Private_SelectElement(_Context, Instance, Event.Target);
						}
					}
				}
			}
		}
	}
}

/// Get the recommended window size to display all the elements of the grid
Vec2 GetRecommendedWindowSize(Vec2 _ElementSize, Vec2 _GutterSize, Int2 _GridSize, Struct_Box::K_Padding _WindowPadding) {
	declare Vec2 ElementAndGutterSize = _ElementSize + _GutterSize;
	return <
		_WindowPadding.Left + _WindowPadding.Right + (ElementAndGutterSize.X * _GridSize.X) - _GutterSize.X,
		_WindowPadding.Top + _WindowPadding.Bottom + (ElementAndGutterSize.Y * _GridSize.Y) - _GutterSize.Y
	>;
}
Vec2 GetRecommendedWindowSize(Vec2 _ElementSize, Vec2 _GutterSize, Int2 _GridSize) {
	return GetRecommendedWindowSize(_ElementSize, _GutterSize, _GridSize, Struct_Box::K_Padding {});
}

/// Get the current scroll ratio

	declare K_Private_FlexVec2 ScrollOffsetFlex = Private_GetFlexPos(Private_GetScrollOffset(Instance), Instance.Properties.Direction);

	return ScrollOffsetFlex.Cross / ScrollMaxFlex.Cross;
}

		]
	};
