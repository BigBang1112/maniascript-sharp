/**
 * Component ExpendableButton
 * ==========================
 *
 * Info
 * ----
 *
 * An horizontally expendable button
 *
 * Props
 * -----
 *
 * - `width-min` (`Real`): minimum width of the button
 *   Even if the button text is shorter, the button won't be narrower than this width.
 * - `width-max` (`Real`): maximum width of the button
 *   Even if the button text is longer, it will be reduced or cut so the button does not exceed this width.
 * - `width` (`Real`): forced width of the button
 *   This width will still be constrained by the minimum and maximum width.
 *   If the minimum width is 10, the maximum width 100, then if you set the width to 5 it will be clamped to 10.
 *   Same thing if you set the width to 200, it will be clamped to 100.
 * - `height` (`Real`): height of the button
 * - `halign` (`Text`): horizontal alignment of the button
 *   Can be `left`, `center` or `right`
 * - `valign` (`Text`): vertical alignment of the button
 *   Can be `top`, `center` or `bottom`
 * - `text` (`Text`): the text of the button
 * - `text-font` (`Text`): the font of the button
 * - `text-font-size` (`Real`): size of the button text
 * - `text-font-size-min` (`Real`): minimum text font size applied by the FitLabelValue function to the button text
 * - `text-font-size-step` (`Real`): size of the steps used by the FitLabelValue function on the button text
 * - `text-label-size` (`Vec2`): size of the label that display the button text
 * - `text-padding-ratio` (`Vec2`): percentage of padding around the button text (x is left/right side and y top/bottom side)
 * - `text-offset` (`Vec2`): offset of the label that display the button text
 * - `text-autonewline` (`Boolean`): enable text with multiple lines
 * - `text-maxline` (`Integer`): the maximum number of lines in the button text
 * - `text-halign` (`Text`): horizontal alignment of the text displayed in the button
 *   Can be `left`, `center` or `right`
 * - `text-valign` (`Text`): vertical alignment of the text displayed in the button
 *   Can be `top`, `center`, `center2` or `bottom`
 * - `text-prefix` (`Text`): the prefix of the text displayed in the button
 * - `text-tts-alt` (`Text`): alternative button text for the Text-To-Speech
 * - `text-tts-alt-translate` (`Boolean`): translate the alternative tts text
 * - `text-filter-profanities` (`Text`): set the profanities filter on the button text
 *   Can be `Tools::C_FilterProfanities_Never`, `Tools::C_FilterProfanities_Always` or `Tools::C_FilterProfanities_OnlyIfNotTranslated`
 * - `text-color-default` (`Text`): color of the button text in the default state
 * - `text-color-focus` (`Text`): color of the button text when focused
 * - `text-opacity-default` (`Real`): opacity of the button text in the default state
 * - `text-opacity-focus` (`Real`): opacity of the button text when focused
 * - `icon-image` (`Text`): url to the image used for the button icon
 * - `icon-size` (`Vec2`): size of the button icon
 *   If no size is given, the icon size will be computed from the button height.
 * - `icon-colormode` (`Integer`): color mode used for the button icon
 *   Can be `C_ColorMode_None` for no color,
 *   `C_ColorMode_Colorize` to use the `colorize` property of the quad,
 *   `C_ColorMode_Modulate` to use the `modulatecolor` property of the quad or
 *   `C_ColorMode_Background` to use the `bgcolor` property of the quad.
 * - `icon-color-default` (`Text`): color of the button icon when the button is unfocused
 * - `icon-color-focused` (`Text`): color of the button icon when the button is focused
 * - `icon-opacity-default` (`Real`): opacity of the button icon when the button is unfocused
 * - `icon-opacity-focused` (`Real`): opacity of the button icon when the button is focused
 * - `icon-align` (`Integer`): alignment of the icon in the button
 *   Use the `C_IconAlign_XXX` constants to set how the icon will be placed in the button
 *   `C_IconAlign_Left` display the icon on the left of the button
 *   `C_IconAlign_Right` display the icon on the right of the button
 *   `C_IconAlign_Center` display the icon in the middle of the button, it will overlap the button text if it has one
 *   `C_IconAlign_TextLeft` display the icon at the left of the button's text
 *   `C_IconAlign_TextRight` display the icon at the right of the button's text
 * - `icon-offset` (`Vec2`): apply an offset to the aligned position of the icon
 * - `icon-keepratio` (`Text`): the ratio mode applied on the icon
 * - `bg-color-default` (`Text`): color to apply to the background displayed when the button is in default state
 * - `bg-color-focus` (`Text`): color to apply to the background displayed when the button is focused
 * - `fg-color-default` (`Text`): color to apply to the foreground displayed when the button is in default state
 * - `fg-color-focus` (`Text`): color to apply to the foreground displayed when the button is focused
 * - `bg-opacity-default` (`Real`): opacity to apply to the background displayed when the button is in default state
 * - `bg-opacity-focus` (`Real`): opacity to apply to the background displayed when the button is focused
 * - `fg-opacity-default` (`Real`): opacity to apply to the foreground displayed when the button is in default state
 * - `fg-opacity-focus` (`Real`): opacity to apply to the foreground displayed when the button is focused
 * - `bg-styles-default` (`Text`): styles to apply to the background displayed when the button is in default state
 * - `bg-styles-focus` (`Text`): styles to apply to the background displayed when the button is focused
 * - `fg-styles-default` (`Text`): styles to apply to the foreground displayed when the button is in default state
 * - `fg-styles-focus` (`Text`): styles to apply to the foreground displayed when the button is focused
 * - `transition-duration` (`Integer`): duration of the props update transition animation
 * - `hitbox-skew` (`Real`): skew of the button hitbox
 * - `supersample` (`Boolean`): enable supersample on the background and foreground images
 * - `styles` (`Text`): styles to apply to the asset, new styles can be created with the `CreateStyle()` function
 *      When several styles are applied to an asset, the styles are merged together with the properties
 *      of the latest style overriding the properties of the earlier ones.
 *      For example if "style-a" is `{ A = 1, B = 2, C = 3 }`, "style-b" is `{ A = 11 , C = 33}` and
 *      "style-c" is `{ A = 111 }` and we set the asset style to `"style-a style-b style-c"`
 *      then the applied properties will be `{ A = 111, B = 2, C = 33 }`.
 */

/// Convert a Text icon alignment to an Integer icon alignment
Integer Private_TextToIconAlign(Text _IconAlign) {
	switch (TL::ToLowerCase(TL::Trim(_IconAlign))) {
		case "left", ""^C_IconAlign_Left: return C_IconAlign_Left;
		case "center", ""^C_IconAlign_Center: return C_IconAlign_Center;
		case "right", ""^C_IconAlign_Right: return C_IconAlign_Right;
		case "text-left", ""^C_IconAlign_TextLeft: return C_IconAlign_TextLeft;
		case "text-right", ""^C_IconAlign_TextRight: return C_IconAlign_TextRight;
	}
	return C_IconAlign_TextLeft;
}

/// Update the animated attribute of a control
Text[Text][CMlControl] Private_SetControlAnimAttribute(
	Text[Text][CMlControl] _ControlsAnimAttributes,
	CMlControl _Control,
	Text _AttributeName,
	Text _AttributeValue
) {
	declare Text[Text][CMlControl] ControlsAnimAttributes = _ControlsAnimAttributes;
	if (ControlsAnimAttributes.existskey(_Control)) {
		ControlsAnimAttributes[_Control][_AttributeName] = _AttributeValue;
	} else {
		ControlsAnimAttributes[_Control] = [_AttributeName =>_AttributeValue];
	}
	return ControlsAnimAttributes;
}

/// Set the color of a quad taking its color mode into account
Text[Text][CMlControl] Private_SetQuadColor(
	CMlQuad _Quad,
	Integer _ColorMode,
	Vec3 _Color,
	Boolean _IsAnimated,
	Text[Text][CMlControl] _ControlsAnimAttributes
) {
	declare Text[Text][CMlControl] ControlsAnimAttributes = _ControlsAnimAttributes;
	declare Vec3 PrevColor;

	switch (_ColorMode) {
		case C_ColorMode_Colorize: PrevColor = _Quad.Colorize;
		case C_ColorMode_Modulate: PrevColor = _Quad.ModulateColor;
		case C_ColorMode_Background: PrevColor = _Quad.BgColor;
	}

	// Initialize all properties properly here first
	// Modifying one of them afterward can also have an effect on the others.
	// eg: modifying `ModulateColor` will also change the value of `Colorize`.
	_Quad.Colorize = C_Default_Colorize;
	_Quad.ModulateColor = C_Default_ModulateColor;
	_Quad.BgColor = C_Default_BgColor;

	switch (_ColorMode) {
		case C_ColorMode_Colorize: {
			if (_IsAnimated) {
				_Quad.Colorize = PrevColor;
				ControlsAnimAttributes = Private_SetControlAnimAttribute(
					ControlsAnimAttributes,
					_Quad,
					"colorize",
					CL::RgbToHex6(_Color)
				);
			} else {
				_Quad.Colorize = _Color;
			}
		}
		case C_ColorMode_Modulate: {
			if (_IsAnimated) {
				_Quad.ModulateColor = PrevColor;
				ControlsAnimAttributes = Private_SetControlAnimAttribute(
					ControlsAnimAttributes,
					_Quad,
					"modulatecolor",
					CL::RgbToHex6(_Color)
				);
			} else {
				_Quad.ModulateColor = _Color;
			}
		}
		case C_ColorMode_Background: {
			if (_IsAnimated) {
				_Quad.BgColor = PrevColor;
				ControlsAnimAttributes = Private_SetControlAnimAttribute(
					ControlsAnimAttributes,
					_Quad,
					"bgcolor",
					CL::RgbToHex6(_Color)
				);
			} else {
				_Quad.BgColor = _Color;
			}
		}
	}

	return ControlsAnimAttributes;
}

/// Update the super sample property of an expendable asset component
Void Private_SetExpendableAssetSuperSample(CMlScript _This, CMlFrame _Asset, Integer _SuperSample) {
	switch (_SuperSample) {
		case C_SuperSample_Null: ExpendableAsset::ResetSuperSample(_This, _Asset);
		case C_SuperSample_On: ExpendableAsset::SetSuperSample(_Asset, True);
		case C_SuperSample_Off: ExpendableAsset::SetSuperSample(_Asset, False);
	}
}

/// Enable a property
Integer[] Private_EnableProperty(Integer[] _EnabledProperties, Integer _Property, Boolean _Enabled) {
	declare Integer[] EnabledProperties = _EnabledProperties;

	if (_Enabled) {
		if (!EnabledProperties.exists(_Property)) {
			EnabledProperties.add(_Property);
		}
	} else {
		EnabledProperties.remove(_Property);
	}

	return EnabledProperties;
}

		case C_Property_IconSize: Properties.IconSize = _SourceProperties.IconSize;

		case C_Property_IconColorDefault: Properties.IconColorDefault = _SourceProperties.IconColorDefault;

		case C_Property_IconOpacityDefault: Properties.IconOpacityDefault = _SourceProperties.IconOpacityDefault;

/// Merge a base properties struct with other enabled properties
K_Properties Private_MergeProperties(K_Properties _Base, K_Properties[] _PropertiesToMerge) {
	declare K_Properties MergedProperties = _Base;

	for (Properties in _PropertiesToMerge) {
		for (PropertyId in Properties.Enabled) {
			MergedProperties = Private_SetProperty(MergedProperties, PropertyId, Properties, C_UpdateMode_Merge);
		}
	}

	return MergedProperties;
}

/// Merge the base, styles and local properties of a component
K_Properties Private_ComputeMergedProperties(K_Instance _Instance, K_Global _Global) {
	declare K_Properties[] InstanceProperties = [_Instance.LocalProperties];
	for (Style in reverse _Instance.Styles) {
		if (Style != "" && _Global.StylesProperties.existskey(Style)) {
			InstanceProperties.addfirst(_Global.StylesProperties[Style]);
		}
	}
	return Private_MergeProperties(_Global.BaseProperties, InstanceProperties);
}

/// Set a property of the component
K_Instance Private_SetLocalProperty(K_Instance _Instance, Integer _PropertyId, K_Properties _Properties) {
	if (!Private_IsPropertyModified(_PropertyId, _Instance.LocalProperties, _Properties)) return _Instance;

	declare K_Instance Instance = _Instance;
	Instance.LocalProperties = Private_SetProperty(Instance.LocalProperties, _PropertyId, _Properties, C_UpdateMode_Override);
	// We know that the local properties will always override the base and styles properties
	// So we don't need to spend time recomputing the whole merged properties structure
	// We can just write the new value directly
	Instance.MergedProperties = Private_SetProperty(Instance.MergedProperties, _PropertyId, _Properties, C_UpdateMode_Override);
	MV::MarkForRedraw(Instance.Controls.Frame_ExpendableButton, Private_GetPropertyRedrawTags(_PropertyId));
	return Instance;
}

	declare K_Instance CMGame_ComponentsExpendableButton2_Instance for _Frame_ExpendableButton = C_Instance_Null;
	CMGame_ComponentsExpendableButton2_Instance = Private_SetLocalProperty(CMGame_ComponentsExpendableButton2_Instance, _PropertyId, _Properties);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Reset a property of the component
K_Instance Private_ResetLocalProperty(K_Instance _Instance, K_Global _Global, Integer _PropertyId) {
	if (!_Instance.LocalProperties.Enabled.exists(_PropertyId)) return _Instance;

	declare K_Instance Instance = _Instance;
	Instance.LocalProperties.Enabled = Private_EnableProperty(Instance.LocalProperties.Enabled, _PropertyId, False);
	Instance.MergedProperties = Private_ComputeMergedProperties(Instance, _Global);
	MV::MarkForRedraw(Instance.Controls.Frame_ExpendableButton, Private_GetPropertyRedrawTags(_PropertyId));
	return Instance;
}
Void Private_ResetLocalProperty(CMlScript _This, CMlFrame _Frame_ExpendableButton, Integer _PropertyId) {
	declare K_Global CMGame_ComponentsExpendableButton2_Global for _This.Page = C_Global_Null;
	declare K_Instance CMGame_ComponentsExpendableButton2_Instance for _Frame_ExpendableButton = C_Instance_Null;
	CMGame_ComponentsExpendableButton2_Instance = Private_ResetLocalProperty(CMGame_ComponentsExpendableButton2_Instance, CMGame_ComponentsExpendableButton2_Global, _PropertyId);
}

Void Private_Draw_LabelTextPosition(K_Instance _Instance, Vec2 _Size, Vec2 _Padding, Vec2 _IconSize, Real _IconGap, Real _TextWidth) {
	switch (_Instance.MergedProperties.TextHAlign) {
		case CMlControl::AlignHorizontal::Left, CMlControl::AlignHorizontal::None: {
			switch (_Instance.MergedProperties.IconAlign) {
				case C_IconAlign_Left: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Padding.X + _IconSize.X + _IconGap;
				}
				case C_IconAlign_Center: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Padding.X;
				}
				case C_IconAlign_Right: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Padding.X;
				}
				case C_IconAlign_TextLeft: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Padding.X + _IconSize.X + _IconGap;
				}
				case C_IconAlign_TextRight: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Padding.X;
				}
			}
		}
		case CMlControl::AlignHorizontal::HCenter: {
			switch (_Instance.MergedProperties.IconAlign) {
				case C_IconAlign_Left: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Padding.X + _IconSize.X + _IconGap + ((_Size.X - (_Padding.X * 2.) - _IconSize.X - _IconGap) * 0.5);
				}
				case C_IconAlign_Center: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Size.X * 0.5;
				}
				case C_IconAlign_Right: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Size.X - _Padding.X - _IconSize.X - _IconGap - ((_Size.X - (_Padding.X * 2.) - _IconSize.X - _IconGap) * 0.5);
				}
				case C_IconAlign_TextLeft: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Padding.X + ((_TextWidth + _IconSize.X + _IconGap) * 0.5) + ((_IconSize.X + _IconGap) * 0.5);
				}
				case C_IconAlign_TextRight: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Size.X - _Padding.X - ((_TextWidth + _IconSize.X + _IconGap) * 0.5) - ((_IconSize.X + _IconGap) * 0.5);
				}
			}
		}
		case CMlControl::AlignHorizontal::Right: {
			switch (_Instance.MergedProperties.IconAlign) {
				case C_IconAlign_Left: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Size.X - _Padding.X;
				}
				case C_IconAlign_Center: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Size.X - _Padding.X;
				}
				case C_IconAlign_Right: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Size.X - _Padding.X - _IconSize.X - _IconGap;
				}
				case C_IconAlign_TextLeft: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Size.X - _Padding.X;
				}
				case C_IconAlign_TextRight: {
					_Instance.Controls.Label_Text.RelativePosition_V3.X = _Size.X - _Padding.X - _IconSize.X - _IconGap;
				}
			}
		}
	}
	switch (_Instance.MergedProperties.TextVAlign) {
		case CMlControl::AlignVertical::Top, CMlControl::AlignVertical::None: {
			_Instance.Controls.Label_Text.RelativePosition_V3.Y = -_Padding.Y;
		}
		case CMlControl::AlignVertical::VCenter, CMlControl::AlignVertical::VCenter2: {
			_Instance.Controls.Label_Text.RelativePosition_V3.Y = _Size.Y * -0.5;
		}
		case CMlControl::AlignVertical::Bottom: {
			_Instance.Controls.Label_Text.RelativePosition_V3.Y = -_Size.Y + _Padding.Y;
		}
	}
	_Instance.Controls.Label_Text.RelativePosition_V3 += _Instance.MergedProperties.TextOffset;
}

/// Redraw the button size and alignment
Text[Text][CMlControl] Private_Draw_SizeAndAlignment(
	CMlScript _This,
	K_Instance _Instance,
	Text[] _Tags,
	Boolean _IsAnimated,
	Text[Text][CMlControl] _ControlsAnimAttributes
) {
	declare Text[Text][CMlControl] ControlsAnimAttributes = _ControlsAnimAttributes;

	if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_SizeAndAlignment])) {
		declare Vec2 Size = <0., ML::Max(0., _Instance.MergedProperties.Height)>;
		declare Vec2 Padding = <
			Size.Y * _Instance.MergedProperties.PaddingRatio.X, //< Always use the height to compute the padding
			Size.Y * _Instance.MergedProperties.PaddingRatio.Y
		>;
		declare Vec2 IconSize = <0., 0.>;
		declare Real IconGap = 0.;
		if (_Instance.MergedProperties.IconImageUrl != "") {
			if (_Instance.MergedProperties.IconSize == <0., 0.>) {
				declare Real IconHeight = ML::Max(0., Size.Y - (Padding.Y * 2.));
				IconSize = <IconHeight, IconHeight>;
			} else {
				IconSize = _Instance.MergedProperties.IconSize;
			}
			if (IconSize.X > 0.) {
				IconGap = Size.Y * C_IconTextGapRatio;
			}
		}
		if (_Instance.MergedProperties.Width >= 0.) {
			Size.X = _Instance.MergedProperties.Width;
		} else {
			Size.X = _Instance.Controls.Label_Text.ComputeWidth(_Instance.MergedProperties.TextValue, Tools::IsTranslatedText(_Instance.MergedProperties.TextValue)) + (Padding.X * 2.) + IconSize.X + IconGap;
		}
		if (_Instance.MergedProperties.WidthMin >= 0.) Size.X = ML::Max(Size.X, _Instance.MergedProperties.WidthMin);
		if (_Instance.MergedProperties.WidthMax >= 0.) Size.X = ML::Min(Size.X, _Instance.MergedProperties.WidthMax);
		declare Real TextWidth = ML::Max(0., Size.X - (Padding.X * 2.));
		if (_Instance.MergedProperties.IconAlign != C_IconAlign_Center) {
			TextWidth = ML::Max(0., TextWidth - IconSize.X - IconGap);
		}

		ExpendableAsset::SetSize(_Instance.Controls.Asset_BgDefault, Size);
		ExpendableAsset::SetSize(_Instance.Controls.Asset_BgFocus, Size);
		ExpendableAsset::SetSize(_Instance.Controls.Asset_FgDefault, Size);
		ExpendableAsset::SetSize(_Instance.Controls.Asset_FgFocus, Size);

		if (_Instance.MergedProperties.TextLabelSize == <0., 0.>) {
			_Instance.Controls.Label_Text.Size = <TextWidth, ML::Max(0., Size.Y - (Padding.Y * 2.))>;
		} else {
			_Instance.Controls.Label_Text.Size = _Instance.MergedProperties.TextLabelSize;
		}
		Private_Draw_LabelTextPosition(_Instance, Size, Padding, IconSize, IconGap, TextWidth);
		_Instance.Controls.Label_Text.HorizontalAlign = _Instance.MergedProperties.TextHAlign;
		_Instance.Controls.Label_Text.VerticalAlign = _Instance.MergedProperties.TextVAlign;

		if (_Instance.MergedProperties.TextFitLabelMin >= 0.) {
			Tools::FitLabelValue(
				_Instance.Controls.Label_Text,
				_Instance.MergedProperties.TextFontSize,
				_Instance.MergedProperties.TextFitLabelMin,
				_Instance.MergedProperties.TextFitLabelStep
			);
		}
		declare Real TextContentWidth = _Instance.Controls.Label_Text.ComputeWidth(_Instance.MergedProperties.TextValue, Tools::IsTranslatedText(_Instance.MergedProperties.TextValue));

		switch (_Instance.MergedProperties.IconAlign) {
			case C_IconAlign_Left: {
				_Instance.Controls.Quad_Icon.HorizontalAlign = CMlControl::AlignHorizontal::Left;
				_Instance.Controls.Quad_Icon.RelativePosition_V3.X = Padding.X;
			}
			case C_IconAlign_Center: {
				_Instance.Controls.Quad_Icon.HorizontalAlign = CMlControl::AlignHorizontal::HCenter;
				_Instance.Controls.Quad_Icon.RelativePosition_V3.X = Size.X * 0.5;
			}
			case C_IconAlign_Right: {
				_Instance.Controls.Quad_Icon.HorizontalAlign = CMlControl::AlignHorizontal::Right;
				_Instance.Controls.Quad_Icon.RelativePosition_V3.X = Size.X - Padding.X;
			}
			case C_IconAlign_TextLeft: {
				_Instance.Controls.Quad_Icon.HorizontalAlign = CMlControl::AlignHorizontal::Right;
				declare Real LabelShift = 0.;
				switch (_Instance.MergedProperties.TextHAlign) {
					case CMlControl::AlignHorizontal::HCenter: LabelShift = ML::Min(TextContentWidth, _Instance.Controls.Label_Text.Size.X) * 0.5;
					case CMlControl::AlignHorizontal::Right: LabelShift = ML::Min(TextContentWidth, _Instance.Controls.Label_Text.Size.X);
				}
				_Instance.Controls.Quad_Icon.RelativePosition_V3.X = _Instance.Controls.Label_Text.RelativePosition_V3.X - LabelShift - IconGap;
			}
			case C_IconAlign_TextRight: {
				_Instance.Controls.Quad_Icon.HorizontalAlign = CMlControl::AlignHorizontal::Left;
				declare Real LabelShift = 0.;
				switch (_Instance.MergedProperties.TextHAlign) {
					case CMlControl::AlignHorizontal::Left, CMlControl::AlignHorizontal::None: LabelShift = ML::Min(TextContentWidth, _Instance.Controls.Label_Text.Size.X);
					case CMlControl::AlignHorizontal::HCenter: LabelShift = ML::Min(TextContentWidth, _Instance.Controls.Label_Text.Size.X) * 0.5;
				}
				_Instance.Controls.Quad_Icon.RelativePosition_V3.X = _Instance.Controls.Label_Text.RelativePosition_V3.X + LabelShift + IconGap;
			}
		}
		_Instance.Controls.Quad_Icon.Size = IconSize;
		_Instance.Controls.Quad_Icon.RelativePosition_V3.Y = _Instance.Controls.Label_Text.RelativePosition_V3.Y;
		_Instance.Controls.Quad_Icon.RelativePosition_V3 += _Instance.MergedProperties.IconOffset;
		_Instance.Controls.Quad_Icon.VerticalAlign = _Instance.Controls.Label_Text.VerticalAlign;

		_Instance.Controls.Frame_HitboxClip.Size = Size;
		_Instance.Controls.Frame_HitboxClip.RelativePosition_V3.Y = -Size.Y;
		_Instance.Controls.Quad_Hitbox.RelativeRotation = _Instance.MergedProperties.HitboxSkew;
		if (_Instance.MergedProperties.HitboxSkew > 0.) {
			_Instance.Controls.Frame_HitboxClip.HorizontalAlign = CMlControl::AlignHorizontal::Left;
			_Instance.Controls.Frame_HitboxClip.RelativePosition_V3.X = 0.;
			_Instance.Controls.Quad_Hitbox.HorizontalAlign = CMlControl::AlignHorizontal::Left;
		} else {
			_Instance.Controls.Frame_HitboxClip.HorizontalAlign = CMlControl::AlignHorizontal::Right;
			_Instance.Controls.Frame_HitboxClip.RelativePosition_V3.X = Size.X;
			_Instance.Controls.Quad_Hitbox.HorizontalAlign = CMlControl::AlignHorizontal::Right;
		}

		declare Real AngleRad = ML::DegToRad(ML::Abs(_Instance.MergedProperties.HitboxSkew));
		if (ML::Tan(AngleRad) * Size.Y <= Size.X) {
			declare Real ParallelogramShiftX = (Size.X - (ML::Tan(AngleRad) * Size.Y));
			_Instance.Controls.Quad_Hitbox.Size = <
				ML::Cos(AngleRad) * ParallelogramShiftX,
				(Size.Y / ML::Cos(AngleRad)) + ParallelogramShiftX
			>;
		} else {
			declare Real Diagonal = ML::Sqrt(ML::Pow(Size.X, 2.) + ML::Pow(Size.Y, 2.));
			_Instance.Controls.Quad_Hitbox.Size = <Diagonal, Diagonal>;
			log("WARNING [ExpendableButton@2] The hitbox is too skewed and will overflow the button");
		}

		declare Vec2 AlignPosition = Tools::GetAlignPosition(Size, _Instance.MergedProperties.HAlign, _Instance.MergedProperties.VAlign);
		_This.AnimMgr.Flush(_Instance.Controls.Frame_ButtonAlign);
		if (_IsAnimated) {
			ControlsAnimAttributes = Private_SetControlAnimAttribute(
				ControlsAnimAttributes,
				_Instance.Controls.Frame_ButtonAlign,
				"pos",
				AlignPosition.X^" "^AlignPosition.Y
			);
		} else {
			_Instance.Controls.Frame_ButtonAlign.RelativePosition_V3 = AlignPosition;
		}

		declare Vec2 CMGame_ComponentsExpendableButton2_CacheSize for _Instance.Controls.Frame_ExpendableButton = <0., 0.>;
		CMGame_ComponentsExpendableButton2_CacheSize = Size;
	}

	return ControlsAnimAttributes;
}

/// Redraw the component background and foreground styles

			} else {
				ExpendableAsset::SetColor(_Instance.Controls.Asset_BgDefault, _Instance.MergedProperties.BgColorDefault);
			}

			if (_Instance.MergedProperties.BgOpacityDefault == C_Opacity_Null) {
				ExpendableAsset::ResetOpacity(_This, _Instance.Controls.Asset_BgDefault);
			} else {
				ExpendableAsset::SetOpacity(_Instance.Controls.Asset_BgDefault, _Instance.MergedProperties.BgOpacityDefault);

				if (_Instance.MergedProperties.BgOpacityFocus == C_Opacity_Null) {
					ExpendableAsset::ResetOpacity(_This, _Instance.Controls.Asset_BgFocus);
				} else {
					ExpendableAsset::SetOpacity(_Instance.Controls.Asset_BgFocus, _Instance.MergedProperties.BgOpacityFocus);
				}
			} else {
				ExpendableAsset::SetOpacity(_Instance.Controls.Asset_BgFocus, 0.);
			}

			Private_SetExpendableAssetSuperSample(_This, _Instance.Controls.Asset_BgFocus, _Instance.MergedProperties.SuperSample);
		}
	}
	if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_FgDefault])) {
		_Instance.Controls.Asset_FgDefault.Visible = _Instance.MergedProperties.FgStylesDefault.count > 0;

			if (_Instance.MergedProperties.FgColorDefault == C_Color_Null) {
				ExpendableAsset::ResetColor(_This, _Instance.Controls.Asset_FgDefault);
			} else {
				ExpendableAsset::SetColor(_Instance.Controls.Asset_FgDefault, _Instance.MergedProperties.FgColorDefault);
			}

			if (_Instance.MergedProperties.FgOpacityDefault == C_Opacity_Null) {
				ExpendableAsset::ResetOpacity(_This, _Instance.Controls.Asset_FgDefault);
			} else {
				ExpendableAsset::SetOpacity(_Instance.Controls.Asset_FgDefault, _Instance.MergedProperties.FgOpacityDefault);
			}

			Private_SetExpendableAssetSuperSample(_This, _Instance.Controls.Asset_FgDefault, _Instance.MergedProperties.SuperSample);
		}

		_Instance.Controls.Asset_FgFocus.Visible = _Instance.MergedProperties.FgStylesFocus.count > 0;

			ExpendableAsset::SetStyles(_This, _Instance.Controls.Asset_FgFocus, _Instance.MergedProperties.FgStylesFocus);

			if (_Instance.IsFocused) {
				if (_Instance.MergedProperties.FgOpacityFocus == C_Opacity_Null) {
					ExpendableAsset::ResetOpacity(_This, _Instance.Controls.Asset_FgFocus);
				} else {
					ExpendableAsset::SetOpacity(_Instance.Controls.Asset_FgFocus, _Instance.MergedProperties.FgOpacityFocus);
				}
			} else {

			}

			if (_Instance.MergedProperties.FgColorFocus == C_Color_Null) {

			} else {

/// Redraw the component state (default, focus)
Text[Text][CMlControl] Private_Draw_State(
	CMlScript _This,
	K_Instance _Instance,
	Text[] _Tags,
	Boolean _IsAnimated,
	Text[Text][CMlControl] _ControlsAnimAttributes
) {
	declare Text[Text][CMlControl] ControlsAnimAttributes = _ControlsAnimAttributes;

	if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_State])) {
		if (_Instance.IsFocused) {
			if (_IsAnimated) {
				ControlsAnimAttributes = Private_SetControlAnimAttribute(
					ControlsAnimAttributes,
					_Instance.Controls.Label_Text,
					"textcolor",
					CL::RgbToHex6(_Instance.MergedProperties.TextColorFocus)
				);
				ControlsAnimAttributes = Private_SetControlAnimAttribute(
					ControlsAnimAttributes,
					_Instance.Controls.Label_Text,
					"opacity",
					""^_Instance.MergedProperties.TextOpacityFocus
				);
				ControlsAnimAttributes = Private_SetControlAnimAttribute(
					ControlsAnimAttributes,
					_Instance.Controls.Quad_Icon,
					"opacity",
					""^_Instance.MergedProperties.IconOpacityFocus
				);
			} else {
				_Instance.Controls.Label_Text.TextColor = _Instance.MergedProperties.TextColorFocus;
				_Instance.Controls.Label_Text.Opacity = _Instance.MergedProperties.TextOpacityFocus;
				_Instance.Controls.Quad_Icon.Opacity = _Instance.MergedProperties.IconOpacityFocus;
			}
			ControlsAnimAttributes = Private_SetQuadColor(
				_Instance.Controls.Quad_Icon,
				_Instance.MergedProperties.IconColorMode,
				_Instance.MergedProperties.IconColorFocus,
				_IsAnimated,
				ControlsAnimAttributes
			);
			if (_Instance.MergedProperties.BgStylesFocus.count > 0) {
				if (_Instance.MergedProperties.BgOpacityFocus == C_Opacity_Null) {
					ExpendableAsset::ResetOpacity(_This, _Instance.Controls.Asset_BgFocus);
				} else {
					ExpendableAsset::SetOpacity(_Instance.Controls.Asset_BgFocus, _Instance.MergedProperties.BgOpacityFocus);
				}
			}
			if (_Instance.MergedProperties.FgStylesFocus.count > 0) {
				if (_Instance.MergedProperties.FgOpacityFocus == C_Opacity_Null) {
					ExpendableAsset::ResetOpacity(_This, _Instance.Controls.Asset_FgFocus);
				} else {
					ExpendableAsset::SetOpacity(_Instance.Controls.Asset_FgFocus, _Instance.MergedProperties.FgOpacityFocus);
				}
			}
		} else {
			if (_IsAnimated) {
				ControlsAnimAttributes = Private_SetControlAnimAttribute(
					ControlsAnimAttributes,
					_Instance.Controls.Label_Text,
					"textcolor",
					CL::RgbToHex6(_Instance.MergedProperties.TextColorDefault)
				);
				ControlsAnimAttributes = Private_SetControlAnimAttribute(
					ControlsAnimAttributes,
					_Instance.Controls.Label_Text,
					"opacity",
					""^_Instance.MergedProperties.TextOpacityDefault
				);
				ControlsAnimAttributes = Private_SetControlAnimAttribute(
					ControlsAnimAttributes,
					_Instance.Controls.Quad_Icon,
					"opacity",
					""^_Instance.MergedProperties.IconOpacityDefault
				);
			} else {
				_Instance.Controls.Label_Text.TextColor = _Instance.MergedProperties.TextColorDefault;
				_Instance.Controls.Label_Text.Opacity = _Instance.MergedProperties.TextOpacityDefault;
				_Instance.Controls.Quad_Icon.Opacity = _Instance.MergedProperties.IconOpacityDefault;
			}
			ControlsAnimAttributes = Private_SetQuadColor(
				_Instance.Controls.Quad_Icon,
				_Instance.MergedProperties.IconColorMode,
				_Instance.MergedProperties.IconColorDefault,
				_IsAnimated,
				ControlsAnimAttributes
			);
			if (_Instance.MergedProperties.BgStylesFocus.count > 0) {
				ExpendableAsset::SetOpacity(_Instance.Controls.Asset_BgFocus, 0.);
			}
			if (_Instance.MergedProperties.FgStylesFocus.count > 0) {
				ExpendableAsset::SetOpacity(_Instance.Controls.Asset_FgFocus, 0.);
			}
		}
	}

	return ControlsAnimAttributes;
}

/// Redraw the component
Void Private_Hook_ComponentDraw(CMlScript _This, CMlFrame _Frame_ExpendableButton, Text[] _Tags) {
	declare K_Instance CMGame_ComponentsExpendableButton2_Instance as Instance for _Frame_ExpendableButton = C_Instance_Null;

	declare Text[Text][CMlControl] ControlsAnimAttributes;
	declare Boolean IsAnimated = (
		Instance.MergedProperties.TransitionDuration > 0 &&
		!MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_SkipTransition])
	);

	_This.AnimMgr.Flush(Instance.Controls.Label_Text);

	if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init])) {
		MV::MarkForRedraw(Instance.Controls.Asset_BgDefault, ExpendableAsset::C_RedrawTag_Init);
		MV::MarkForRedraw(Instance.Controls.Asset_BgFocus, ExpendableAsset::C_RedrawTag_Init);
		MV::MarkForRedraw(Instance.Controls.Asset_FgDefault, ExpendableAsset::C_RedrawTag_Init);
		MV::MarkForRedraw(Instance.Controls.Asset_FgFocus, ExpendableAsset::C_RedrawTag_Init);
	}
	if (MV::HasRedrawTag(_Tags, [C_RedrawTag_SkipTransition])) {
		MV::MarkForRedraw(Instance.Controls.Asset_BgDefault, ExpendableAsset::C_RedrawTag_SkipTransition);
		MV::MarkForRedraw(Instance.Controls.Asset_BgFocus, ExpendableAsset::C_RedrawTag_SkipTransition);
		MV::MarkForRedraw(Instance.Controls.Asset_FgDefault, ExpendableAsset::C_RedrawTag_SkipTransition);
		MV::MarkForRedraw(Instance.Controls.Asset_FgFocus, ExpendableAsset::C_RedrawTag_SkipTransition);
	}

	if (
		MV::HasRedrawTag(_Tags, [
			C_RedrawTag_Init,
			C_RedrawTag_BaseProperties,
			C_RedrawTag_Styles,
			C_RedrawTag_TransitionDuration
		])
	) {
		ExpendableAsset::SetTransitionDuration(Instance.Controls.Asset_BgDefault, Instance.MergedProperties.TransitionDuration);
		ExpendableAsset::SetTransitionDuration(Instance.Controls.Asset_BgFocus, Instance.MergedProperties.TransitionDuration);
		ExpendableAsset::SetTransitionDuration(Instance.Controls.Asset_FgDefault, Instance.MergedProperties.TransitionDuration);
		ExpendableAsset::SetTransitionDuration(Instance.Controls.Asset_FgFocus, Instance.MergedProperties.TransitionDuration);
	}

	// Anything that modify the label properties must be updated before computing the size of the button
	// The width of the label depends on these properties
	Private_Draw_Text(Instance, _Tags);
	Private_Draw_Icon(Instance, _Tags);
	ControlsAnimAttributes = Private_Draw_SizeAndAlignment(_This, Instance, _Tags, IsAnimated, ControlsAnimAttributes);
	Private_Draw_BgFgStyles(_This, Instance, _Tags);
	ControlsAnimAttributes = Private_Draw_State(_This, Instance, _Tags, IsAnimated, ControlsAnimAttributes);

	if (IsAnimated && ControlsAnimAttributes.count > 0) {
		for (Control => AnimAttributes in ControlsAnimAttributes) {
			_This.AnimMgr.Add(
				Control,
				Tools::CreateXmlNode("anim", AnimAttributes),
				Instance.MergedProperties.TransitionDuration,
				CAnimManager::EAnimManagerEasing::Linear
			);
		}
	}
}
