/**
 * Component ExpendableAsset
 * =========================
 *
 * Info
 * ----
 *
 * Create quad with custom left and right edges
 * Used to create the slanted design of all UI elements
 *
 * Props
 * -----
 *
 * - `size` (`Vec2`): the size of the asset
 * - `color` (`Hex6`): the color of the asset
 * - `halign` (`Text`): horizontal alignment of the asset, can be `left`, `center` or `right`
 * - `valign` (`Text`): vertical alignment of the asset, can be `top`, `center` or `bottom`
 * - `opacity` (`Real`): the opacity of the asset
 * - `supersample` (`Real`): enable supersample on the asset quads when they are colorized to reduce aliasing
 * - `transition-duration` (`Integer`): duration of the props update transition animation
 * - `image-left` (`Text`): url of the image displayed in the left part of the asset
 * - `image-middle` (`Text`): url of the image displayed in the middle part of the asset
 * - `image-right` (`Text`): url of the image displayed in the right part of the asset
 * - `size-left` (`Vec2`): size of the image displayed in the left part of the asset
 * - `size-middle` (`Vec2`): size of the image displayed in the middle part of the asset
 * - `size-right` (`Vec2`): size of the image displayed in the right part of the asset
 * - `colormode-left` (`Text`): color mode of the image displayed in the left part of the asset
 * - `colormode-middle` (`Text`): color mode of the image displayed in the middle part of the asset
 * - `colormode-right` (`Text`): color mode of the image displayed in the right part of the asset
 * - `styles` (`Text`): styles to apply to the asset, new styles can be created with the `CreateStyle()` function
 *      When several styles are applied to an asset, the styles are merged together with the properties
 *      of the latest style overriding the properties of the earlier ones.
 *      For example if "style-a" is `{ A = 1, B = 2, C = 3 }`, "style-b" is `{ A = 11 , C = 33}` and
 *      "style-c" is `{ A = 111 }` and we set the asset style to `"style-a style-b style-c"`
 *      then the applied properties will be `{ A = 111, B = 2, C = 33 }`.
 */

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Enable a property
Integer[] Private_EnableProperty(Integer[] _EnabledProperties, Integer _Property, Boolean _Enabled) {
	declare Integer[] EnabledProperties = _EnabledProperties;

	if (_Enabled) {
		if (!EnabledProperties.exists(_Property)) {
			EnabledProperties.add(_Property);
		}
	} else {
		EnabledProperties.remove(_Property);
	}

	return EnabledProperties;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set and enable a property
K_Properties Private_SetProperty(K_Properties _TargetProperties, Integer _PropertyId, K_Properties _SourceProperties) {
	declare K_Properties Properties = _TargetProperties;

	Properties.Enabled = Private_EnableProperty(Properties.Enabled, _PropertyId, True);

	switch (_PropertyId) {
		case C_Property_Images: {
			for (Position => Image in _SourceProperties.Images) {
				Properties.Images[Position] = Image;
			}
		}
		case C_Property_Size: Properties.Size = _SourceProperties.Size;
		case C_Property_Color: Properties.Color = _SourceProperties.Color;
		case C_Property_HAlign: Properties.HAlign = _SourceProperties.HAlign;
		case C_Property_VAlign: Properties.VAlign = _SourceProperties.VAlign;
		case C_Property_Opacity: Properties.Opacity = _SourceProperties.Opacity;
		case C_Property_SuperSample: Properties.SuperSample = _SourceProperties.SuperSample;
		case C_Property_TransitionDuration: Properties.TransitionDuration = _SourceProperties.TransitionDuration;
	}

	return Properties;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Merge a base properties struct with other enabled properties
K_Properties Private_MergeProperties(K_Properties _Base, K_Properties[] _PropertiesToMerge) {
	declare K_Properties MergedProperties = _Base;

	for (Properties in _PropertiesToMerge) {
		for (PropertyId in Properties.Enabled) {
			MergedProperties = Private_SetProperty(MergedProperties, PropertyId, Properties);
		}
	}

	return MergedProperties;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the given property was modified
Boolean Private_IsPropertyModified(Integer _PropertyId, K_Properties _PrevProperties, K_Properties _NewProperties) {
	if (!_PrevProperties.Enabled.exists(_PropertyId)) return True;

	switch (_PropertyId) {
		case C_Property_Images: {
			for (Position in C_Positions) {
				if (_PrevProperties.Images.get(Position, C_Image_Null) != _NewProperties.Images.get(Position, C_Image_Null)) {
					return True;
				}
			}
		}
		case C_Property_Size: return _PrevProperties.Size != _NewProperties.Size;
		case C_Property_Color: return _PrevProperties.Color != _NewProperties.Color;
		case C_Property_HAlign: return _PrevProperties.HAlign != _NewProperties.HAlign;
		case C_Property_VAlign: return _PrevProperties.VAlign != _NewProperties.VAlign;
		case C_Property_Opacity: return _PrevProperties.Opacity != _NewProperties.Opacity;
		case C_Property_SuperSample: return _PrevProperties.SuperSample != _NewProperties.SuperSample;
		case C_Property_TransitionDuration: return _PrevProperties.TransitionDuration != _NewProperties.TransitionDuration;
	}

	return False;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Merge the base, styles and local properties of a component
K_Properties Private_ComputeMergedProperties(K_Instance _Instance, K_Global _Global) {
	declare K_Properties[] InstanceProperties = [_Instance.LocalProperties];
	for (Style in reverse _Instance.Styles) {
		if (Style != "" && _Global.StylesProperties.existskey(Style)) {
			InstanceProperties.addfirst(_Global.StylesProperties[Style]);
		}
	}
	return Private_MergeProperties(_Global.BaseProperties, InstanceProperties);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the redraw tags to trigger when modifying the given property
Text[] Private_GetPropertyRedrawTags(Integer _PropertyId) {
	switch (_PropertyId) {
		case C_Property_Images: return [C_RedrawTag_Images];
		case C_Property_Size: return [C_RedrawTag_Size];
		case C_Property_Color: return [C_RedrawTag_Color];
		case C_Property_HAlign: return [C_RedrawTag_Align];
		case C_Property_VAlign: return [C_RedrawTag_Align];
		case C_Property_Opacity: return [C_RedrawTag_Opacity];
		case C_Property_SuperSample: return [C_RedrawTag_SuperSample];
		case C_Property_TransitionDuration: return [C_RedrawTag_Null];
	}
	return [];
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set a property of the component
K_Instance Private_SetLocalProperty(K_Instance _Instance, Integer _PropertyId, K_Properties _Properties) {
	if (!Private_IsPropertyModified(_PropertyId, _Instance.LocalProperties, _Properties)) return _Instance;

	declare K_Instance Instance = _Instance;
	Instance.LocalProperties = Private_SetProperty(Instance.LocalProperties, _PropertyId, _Properties);
	// We know that the local properties will always override the base and styles properties
	// So we don't need to spend time recomputing the whole merged properties structure
	// We can just write the new value directly
	Instance.MergedProperties = Private_SetProperty(Instance.MergedProperties, _PropertyId, _Properties);
	MV::MarkForRedraw(Instance.Controls.Frame_ExpendableAsset, Private_GetPropertyRedrawTags(_PropertyId));
	return Instance;
}
Void Private_SetLocalProperty(CMlFrame _Frame_ExpendableAsset, Integer _PropertyId, K_Properties _Properties) {
	declare K_Instance CMGame_ComponentsExpendableAsset2_Instance for _Frame_ExpendableAsset = C_Instance_Null;
	CMGame_ComponentsExpendableAsset2_Instance = Private_SetLocalProperty(CMGame_ComponentsExpendableAsset2_Instance, _PropertyId, _Properties);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Reset a property of the component
K_Instance Private_ResetLocalProperty(K_Instance _Instance, K_Global _Global, Integer _PropertyId) {
	if (!_Instance.LocalProperties.Enabled.exists(_PropertyId)) return _Instance;

	declare K_Instance Instance = _Instance;
	Instance.LocalProperties.Enabled = Private_EnableProperty(Instance.LocalProperties.Enabled, _PropertyId, False);
	Instance.MergedProperties = Private_ComputeMergedProperties(Instance, _Global);
	MV::MarkForRedraw(Instance.Controls.Frame_ExpendableAsset, Private_GetPropertyRedrawTags(_PropertyId));
	return Instance;
}
Void Private_ResetLocalProperty(CMlScript _This, CMlFrame _Frame_ExpendableAsset, Integer _PropertyId) {
	declare K_Global CMGame_ComponentsExpendableAsset2_Global for _This.Page = C_Global_Null;
	declare K_Instance CMGame_ComponentsExpendableAsset2_Instance for _Frame_ExpendableAsset = C_Instance_Null;
	CMGame_ComponentsExpendableAsset2_Instance = Private_ResetLocalProperty(CMGame_ComponentsExpendableAsset2_Instance, CMGame_ComponentsExpendableAsset2_Global, _PropertyId);
}

/// Set the color of a quad taking its color mode into account
Text[Text] Private_SetQuadColor(
	CMlQuad _Quad,
	Integer _ColorMode,
	Vec3 _Color,
	Boolean _IsAnimated,
	Text[Text] _QuadsAnimAttributes
) {
	declare Text[Text] QuadsAnimAttributes = _QuadsAnimAttributes;
	declare Vec3 PrevColor;

	switch (_ColorMode) {
		case C_ColorMode_Colorize: PrevColor = _Quad.Colorize;
		case C_ColorMode_Modulate: PrevColor = _Quad.ModulateColor;
		case C_ColorMode_Background: PrevColor = _Quad.BgColor;
	}

	// Initialize all properties properly here first
	// Modifying one of them afterward can also have an effect on the others.
	// eg: modifying `ModulateColor` will also change the value of `Colorize`.
	_Quad.Colorize = C_Default_Colorize;
	_Quad.ModulateColor = C_Default_ModulateColor;
	_Quad.BgColor = C_Default_BgColor;

	switch (_ColorMode) {
		case C_ColorMode_Colorize: {
			if (_IsAnimated) {
				_Quad.Colorize = PrevColor;
				QuadsAnimAttributes["colorize"] = CL::RgbToHex6(_Color);
			} else {
				_Quad.Colorize = _Color;
			}
		}
		case C_ColorMode_Modulate: {
			if (_IsAnimated) {
				_Quad.ModulateColor = PrevColor;
				QuadsAnimAttributes["modulatecolor"] = CL::RgbToHex6(_Color);
			} else {
				_Quad.ModulateColor = _Color;
			}
		}
		case C_ColorMode_Background: {
			if (_IsAnimated) {
				_Quad.BgColor = PrevColor;
				QuadsAnimAttributes["bgcolor"] = CL::RgbToHex6(_Color);
			} else {
				_Quad.BgColor = _Color;
			}
		}
	}

	return QuadsAnimAttributes;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the size of the left or right quad
Vec2 Private_GetLeftRightQuadSize(K_Image _Image, K_Properties _AssetProperties) {
	if (_Image.Size.Y == 0.) return <0., 0.>;
	return <_AssetProperties.Size.Y * _Image.Size.X / _Image.Size.Y, _AssetProperties.Size.Y>;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Redraw the component
Void Private_Hook_ComponentDraw(CMlScript _This, CMlFrame _Frame_ExpendableAsset, Text[] _Tags) {
	declare K_Instance CMGame_ComponentsExpendableAsset2_Instance as Instance for _Frame_ExpendableAsset = C_Instance_Null;

	declare Boolean IsAnimated = (
		Instance.MergedProperties.TransitionDuration > 0 &&
		!MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_Images, C_RedrawTag_SkipTransition])
	);
	declare Text[Text][Integer] QuadsAnimAttributes;
	for (Position in C_Positions) {
		QuadsAnimAttributes[Position] = [];
	}
	for (Quad in Instance.Controls.Quads) {
		_This.AnimMgr.Flush(Quad);
	}

	if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_Images, C_RedrawTag_Size])) {
		declare CMlQuad Quad_Left <=> Instance.Controls.Quads[C_Position_Left];
		declare CMlQuad Quad_Middle <=> Instance.Controls.Quads[C_Position_Middle];
		declare CMlQuad Quad_Right <=> Instance.Controls.Quads[C_Position_Right];

		declare Vec2 QuadLeftSize = Private_GetLeftRightQuadSize(
			Instance.MergedProperties.Images.get(C_Position_Left, C_Image_Null),
			Instance.MergedProperties
		);
		declare Vec2 QuadRightSize = Private_GetLeftRightQuadSize(
			Instance.MergedProperties.Images.get(C_Position_Right, C_Image_Null),
			Instance.MergedProperties
		);
		declare Vec2 QuadMiddleSize = <
			ML::Max(0., Instance.MergedProperties.Size.X - QuadLeftSize.X - QuadRightSize.X),
			Instance.MergedProperties.Size.Y
		>;
		declare Vec2 QuadMiddlePos = <QuadLeftSize.X, Quad_Left.RelativePosition_V3.Y>;
		declare Vec2 QuadRightPos = <QuadLeftSize.X + QuadMiddleSize.X, Quad_Right.RelativePosition_V3.Y>;

		if (IsAnimated) {
			QuadsAnimAttributes[C_Position_Left]["size"] = QuadLeftSize.X^" "^QuadLeftSize.Y;
			QuadsAnimAttributes[C_Position_Right]["size"] = QuadRightSize.X^" "^QuadRightSize.Y;
			QuadsAnimAttributes[C_Position_Middle]["size"] = QuadMiddleSize.X^" "^QuadMiddleSize.Y;
			QuadsAnimAttributes[C_Position_Middle]["pos"] = QuadMiddlePos.X^" "^QuadMiddlePos.Y;
			QuadsAnimAttributes[C_Position_Right]["pos"] = QuadRightPos.X^" "^QuadRightPos.Y;
		} else {
			Quad_Left.Size = QuadLeftSize;
			Quad_Right.Size = QuadRightSize;
			Quad_Middle.Size = QuadMiddleSize;
			Quad_Middle.RelativePosition_V3 = QuadMiddlePos;
			Quad_Right.RelativePosition_V3 = QuadRightPos;
		}
	}

	if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_Images, C_RedrawTag_Size, C_RedrawTag_Color, C_RedrawTag_Opacity, C_RedrawTag_SuperSample])) {
		foreach (Position in C_Positions) {
			declare CMlQuad Quad <=> Instance.Controls.Quads.get(Position, Null);
			if (Quad != Null) {
				declare K_Image Image = Instance.MergedProperties.Images.get(Position, C_Image_Null);

				if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_Images])) {
					if (Image.Url != "" || Image.ColorMode == C_ColorMode_Background) {
						Quad.ImageUrl = Image.Url;
						Quad.Visible = True;
					} else {
						Quad.Visible = False;
					}
				}

				if (Quad.Visible) {
					if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_Images, C_RedrawTag_Color])) {
						QuadsAnimAttributes[Position] = Private_SetQuadColor(Quad, Image.ColorMode, Instance.MergedProperties.Color, IsAnimated, QuadsAnimAttributes[Position]);
					}

					if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_Images, C_RedrawTag_SuperSample])) {
						Quad.SuperSample = (Instance.MergedProperties.SuperSample && Image.Url != "" && Image.ColorMode == C_ColorMode_Colorize);
					}

					if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_Opacity])) {
						if (IsAnimated) {
							QuadsAnimAttributes[Position]["opacity"] = ""^Instance.MergedProperties.Opacity;
						} else {
							Quad.Opacity = Instance.MergedProperties.Opacity;
						}
					}
				}
			}
		}
	}

	if (MV::HasRedrawTag(_Tags, [C_RedrawTag_Init, C_RedrawTag_BaseProperties, C_RedrawTag_Styles, C_RedrawTag_Size, C_RedrawTag_Align])) {
		declare Vec2 AlignPosition = Tools::GetAlignPosition(Instance.MergedProperties.Size, Instance.MergedProperties.HAlign, Instance.MergedProperties.VAlign);
		_This.AnimMgr.Flush(Instance.Controls.Frame_Align);
		if (IsAnimated) {
			_This.AnimMgr.Add(Instance.Controls.Frame_Align, """<anim pos="{{{AlignPosition.X}}} {{{AlignPosition.Y}}}" />""", Instance.MergedProperties.TransitionDuration, CAnimManager::EAnimManagerEasing::Linear);
		} else {
			Instance.Controls.Frame_Align.RelativePosition_V3 = AlignPosition;
		}
	}

	if (IsAnimated) {
		for (Position => QuadAnimAttributes in QuadsAnimAttributes) {
			if (QuadAnimAttributes.count > 0) {
				declare CMlQuad Quad <=> Instance.Controls.Quads.get(Position, Null);
				if (Quad != Null) {
					_This.AnimMgr.Add(Quad, Tools::CreateXmlNode("anim", QuadAnimAttributes), Instance.MergedProperties.TransitionDuration, CAnimManager::EAnimManagerEasing::Linear);
				}
			}
		}
	}
}

/// Initialize the component
Void Private_Hook_ComponentInit(CMlScript _This, CMlFrame _Frame_ExpendableAsset) {
	declare K_Instance CMGame_ComponentsExpendableAsset2_Instance as Instance for _Frame_ExpendableAsset = C_Instance_Null;
	Instance = C_Instance_Null;

	Instance.Controls.Frame_ExpendableAsset <=> _Frame_ExpendableAsset;
	Instance.Controls.Frame_Align <=> (Instance.Controls.Frame_ExpendableAsset.GetFirstChild(C_Id_FrameAlign) as CMlFrame);
	Instance.Controls.Quads = [
		C_Position_Left => (Instance.Controls.Frame_ExpendableAsset.GetFirstChild(C_Id_QuadLeft) as CMlQuad),
		C_Position_Middle => (Instance.Controls.Frame_ExpendableAsset.GetFirstChild(C_Id_QuadCenter) as CMlQuad),
		C_Position_Right => (Instance.Controls.Frame_ExpendableAsset.GetFirstChild(C_Id_QuadRight) as CMlQuad)
	];

	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_Size)) {
		Instance = Private_SetLocalProperty(Instance, C_Property_Size, K_Properties { Size = MV::GetProp_Vec2(Instance.Controls.Frame_ExpendableAsset, C_Prop_Size, <0., 0.>) });
	}
	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_Color)) {
		Instance = Private_SetLocalProperty(Instance, C_Property_Color, K_Properties { Color = MV::GetProp_Color(Instance.Controls.Frame_ExpendableAsset, C_Prop_Color, <0., 0., 0.>) });
	}
	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_HAlign)) {
		Instance = Private_SetLocalProperty(Instance, C_Property_HAlign, K_Properties { HAlign = MV::GetProp_HAlign(Instance.Controls.Frame_ExpendableAsset, C_Prop_HAlign, CMlControl::AlignHorizontal::Left) });
	}
	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_VAlign)) {
		Instance = Private_SetLocalProperty(Instance, C_Property_VAlign, K_Properties { VAlign = MV::GetProp_VAlign(Instance.Controls.Frame_ExpendableAsset, C_Prop_VAlign, CMlControl::AlignVertical::Top) });
	}
	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_Opacity)) {
		Instance = Private_SetLocalProperty(Instance, C_Property_Opacity, K_Properties { Opacity = MV::GetProp_Real(Instance.Controls.Frame_ExpendableAsset, C_Prop_Opacity, 1.) });
	}
	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_SuperSample)) {
		Instance = Private_SetLocalProperty(Instance, C_Property_SuperSample, K_Properties { SuperSample = MV::GetProp_Boolean(Instance.Controls.Frame_ExpendableAsset, C_Prop_SuperSample, False) });
	}
	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_TransitionDuration)) {
		Instance = Private_SetLocalProperty(Instance, C_Property_TransitionDuration, K_Properties { TransitionDuration = MV::GetProp_Integer(Instance.Controls.Frame_ExpendableAsset, C_Prop_TransitionDuration, 0) });
	}
	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_Styles)) {
		Instance.Styles = TL::Split(" ", TL::Trim(MV::GetProp_Text(Instance.Controls.Frame_ExpendableAsset, C_Prop_Styles, "")));
	}

	declare K_Image[Integer] PropImages;
	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_ImageLeft)) {
		if (!PropImages.existskey(C_Position_Left)) PropImages[C_Position_Left] = C_Image_Null;
		PropImages[C_Position_Left].Url = MV::GetProp_Text(Instance.Controls.Frame_ExpendableAsset, C_Prop_ImageLeft, "");
	}
	if (MV::HasProp(Instance.Controls.Frame_ExpendableAsset, C_Prop_ImageMiddle)) {
		if (!PropImages.existskey(C_Position_Middle)) PropImages[C_Position_Middle] = C_Image_Null;
		PropImages[C_Position_Middle].Url = MV::GetProp_Text(Instance.Controls.Frame_ExpendableAsset, C_Prop_ImageMiddle, "");
