/**
 * Slice9 component
 * ================
 *
 * Resizable image that scale proportionally
 *
 * Props
 * -----
 *
 * - `size` (`Vec2`): size of the component
 * - `margin` (`Margin`): margin around the component
 * - `margin-ratio` (`Margin`): margin for the component proportional to the component size
 * - `border` (`Border`): border around the component
 * - `border-ratio` (`Border`): border for the component proportional to the component size
 * - `padding` (`Padding`): padding inside the component
 * - `padding-ratio` (`Padding`): padding for the component proportional to the component size
 * - `halign` (`Text`): horizontal alignment of the component, can be `left`, `center` or `right`
 * - `valign` (`Text`): vertical alignment of the component, can be `top`, `center` or `bottom`
 * - `is-visible` (`Boolean`): the visibility of the component
 * - `border-resize` (`Text`): control how the eight border slices are resized, can be `C_BorderResize_Stretch` or `C_BorderResize_Space`
 * - `border-flexible` (`Boolean`): control if the eight border slices can be scaled down for the component to fit in the requested size
 * - `(top|center|right)-(left|center|right)-size` (`Vec2`): the slice size
 * - `(top|center|right)-(left|center|right)-image-url` (`Text`): the slice image
 * - `(top|center|right)-(left|center|right)-color-mode` (`Text`): the slice color mode, can be `default`, `background`, `colorize` or `modulate`
 * - `(top|center|right)-(left|center|right)-color` (`Text`): the slice color
 * - `(top|center|right)-(left|center|right)-opacity` (`Real`): the slice opacity
 * - `(top|center|right)-(left|center|right)-mirror` (`Integer`): the slice mirror mode, can be `C_Mirror_None`, `C_Mirror_Horizontal`, `C_Mirror_Vertical` or `C_Mirror_HorizontalVertical`
 */

/// Check if the instance is dirty
Boolean Private_IsDirty(K_Private_Instance _Instance) {
	return (_Instance.IsPropertiesDirty || _Instance.IsBoxDirty);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the properties
K_Private_Instance Private_UpdateProperties(K_Private_Instance _Instance) {
	declare K_Private_Instance Instance = _Instance;

	Instance.IsPropertiesDirty = False;

	Instance.Properties = Private_GetDefaultProperties();
	Instance.Properties = Private_MergeProperties(Instance.Properties, Instance.StyleProperties);
	Instance.Properties = Private_MergeProperties(Instance.Properties, Instance.ControlProperties);

	return Instance;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the box
K_Private_Instance Private_UpdateBox(K_Private_Instance _Instance) {
	declare K_Private_Instance Instance = _Instance;

	Instance.IsBoxDirty = False;

	Instance.Box = Struct_Box::New(
		Instance.Properties.Size,
		Instance.Properties.Margin,
		Instance.Properties.MarginRatio,
		Instance.Properties.Border,
		Instance.Properties.BorderRatio,
		Instance.Properties.Padding,
		Instance.Properties.PaddingRatio
	);

	return Instance;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update dirty values
K_Private_Instance Private_UpdateDirty(K_Private_Instance _Instance) {
	if (!Private_IsDirty(_Instance)) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	if (Instance.IsPropertiesDirty) {
		Instance = Private_UpdateProperties(Instance);
	}
	if (Instance.IsBoxDirty) {
		Instance = Private_UpdateBox(Instance);
	}

	return Instance;
}

/// Set the slice mirror mode
Void Private_SetSliceMirror(CMlQuad _Quad_Slice, K_SliceProperties _SliceProperties) {
	switch (_SliceProperties.Mirror) {
		case C_Mirror_Vertical: {
			_Quad_Slice.RelativeRotation = 90.;
			_Quad_Slice.HorizontalAlign = CMlControl::AlignHorizontal::Left;
			_Quad_Slice.VerticalAlign = CMlControl::AlignVertical::Bottom;
		}
		case C_Mirror_Horizontal: {
			_Quad_Slice.RelativeRotation = -90.;
			_Quad_Slice.HorizontalAlign = CMlControl::AlignHorizontal::Right;
			_Quad_Slice.VerticalAlign = CMlControl::AlignVertical::Top;
		}
		case C_Mirror_HorizontalVertical, C_Mirror_VerticalHorizontal: {
			_Quad_Slice.RelativeRotation = 180.;
			_Quad_Slice.HorizontalAlign = CMlControl::AlignHorizontal::Right;
			_Quad_Slice.VerticalAlign = CMlControl::AlignVertical::Bottom;
		}
		default: {
			_Quad_Slice.RelativeRotation = 0.;
			_Quad_Slice.HorizontalAlign = CMlControl::AlignHorizontal::Left;
			_Quad_Slice.VerticalAlign = CMlControl::AlignVertical::Top;
		}
	}
}

/// Set the slice sizes in stretch mode
Void Private_SetSliceSizes_Stretch(K_Private_Instance _Instance) {
	declare Real[] LeftSizeXAll = [_Instance.Properties.SliceList[C_SliceId_TopLeft].Size.X, _Instance.Properties.SliceList[C_SliceId_CenterLeft].Size.X, _Instance.Properties.SliceList[C_SliceId_BottomLeft].Size.X];
	declare Real LeftSizeXMax = LeftSizeXAll.sortreverse()[0];
	declare Real[] RightSizeXAll = [_Instance.Properties.SliceList[C_SliceId_TopRight].Size.X, _Instance.Properties.SliceList[C_SliceId_CenterRight].Size.X, _Instance.Properties.SliceList[C_SliceId_BottomRight].Size.X];
	declare Real RightSizeXMax = RightSizeXAll.sortreverse()[0];
	declare Real[] TopSizeYAll = [_Instance.Properties.SliceList[C_SliceId_TopLeft].Size.Y, _Instance.Properties.SliceList[C_SliceId_TopCenter].Size.Y, _Instance.Properties.SliceList[C_SliceId_TopRight].Size.Y];
	declare Real TopSizeYMax = TopSizeYAll.sortreverse()[0];
	declare Real[] BottomSizeYAll = [_Instance.Properties.SliceList[C_SliceId_BottomLeft].Size.Y, _Instance.Properties.SliceList[C_SliceId_BottomCenter].Size.Y, _Instance.Properties.SliceList[C_SliceId_BottomRight].Size.Y];
	declare Real BottomSizeYMax = BottomSizeYAll.sortreverse()[0];

	declare Real LeftSizeX = 0.;
	declare Real RightSizeX = 0.;
	declare Real CenterSizeX = 0.;
	declare Real TopSizeY = 0.;
	declare Real BottomSizeY = 0.;
	declare Real CenterSizeY = 0.;

	if (_Instance.Box.ContentArea.Size.X >= LeftSizeXMax + RightSizeXMax) {
		LeftSizeX = LeftSizeXMax;
		RightSizeX = RightSizeXMax;
		CenterSizeX = _Instance.Box.ContentArea.Size.X - (LeftSizeXMax + RightSizeXMax);
	} else if (LeftSizeXMax + RightSizeXMax > 0.) {
		declare Real Ratio = _Instance.Box.ContentArea.Size.X / (LeftSizeXMax + RightSizeXMax);
		LeftSizeX = LeftSizeXMax * Ratio;
		RightSizeX = RightSizeXMax * Ratio;
	}
	if (_Instance.Box.ContentArea.Size.Y >= TopSizeYMax + BottomSizeYMax) {
		TopSizeY = TopSizeYMax;
		BottomSizeY = BottomSizeYMax;
		CenterSizeY = _Instance.Box.ContentArea.Size.Y - (TopSizeYMax + BottomSizeYMax);
	} else if (TopSizeYMax + BottomSizeYMax > 0.) {
		declare Real Ratio = _Instance.Box.ContentArea.Size.Y / (TopSizeYMax + BottomSizeYMax);
		TopSizeY = TopSizeYMax * Ratio;
		BottomSizeY = BottomSizeYMax * Ratio;
	}

	_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size = <LeftSizeX, TopSizeY>;
	_Instance.Controls.QuadList_Slice[C_SliceId_TopCenter].Size = <CenterSizeX, TopSizeY>;
	_Instance.Controls.QuadList_Slice[C_SliceId_TopRight].Size = <RightSizeX, TopSizeY>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].Size = <LeftSizeX, CenterSizeY>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterCenter].Size = <CenterSizeX, CenterSizeY>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterRight].Size = <RightSizeX, CenterSizeY>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomLeft].Size = <LeftSizeX, BottomSizeY>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomCenter].Size = <CenterSizeX, BottomSizeY>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomRight].Size = <RightSizeX, BottomSizeY>;

	_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].RelativePosition_V3 = <0., 0.>;
	_Instance.Controls.QuadList_Slice[C_SliceId_TopCenter].RelativePosition_V3 = <_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size.X, 0.>;
	_Instance.Controls.QuadList_Slice[C_SliceId_TopRight].RelativePosition_V3 = <_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size.X + _Instance.Controls.QuadList_Slice[C_SliceId_TopCenter].Size.X, 0.>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].RelativePosition_V3 = <0., -_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size.Y>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterCenter].RelativePosition_V3 = <_Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].Size.X, -_Instance.Controls.QuadList_Slice[C_SliceId_TopCenter].Size.Y>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterRight].RelativePosition_V3 = <_Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].Size.X + _Instance.Controls.QuadList_Slice[C_SliceId_CenterCenter].Size.X, -_Instance.Controls.QuadList_Slice[C_SliceId_TopRight].Size.Y>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomLeft].RelativePosition_V3 = <0., -_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size.Y - _Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].Size.Y>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomCenter].RelativePosition_V3 = <_Instance.Controls.QuadList_Slice[C_SliceId_BottomLeft].Size.X, -_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size.Y - _Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].Size.Y>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomRight].RelativePosition_V3 = <_Instance.Controls.QuadList_Slice[C_SliceId_BottomLeft].Size.X + _Instance.Controls.QuadList_Slice[C_SliceId_BottomCenter].Size.X, -_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size.Y - _Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].Size.Y>;
}

/// Set the slice sizes in space mode
Void Private_SetSliceSizes_Space(K_Private_Instance _Instance) {
	declare Real[] LeftSizeXAll = [_Instance.Properties.SliceList[C_SliceId_TopLeft].Size.X, _Instance.Properties.SliceList[C_SliceId_CenterLeft].Size.X, _Instance.Properties.SliceList[C_SliceId_BottomLeft].Size.X];
	declare Real LeftSizeXMax = LeftSizeXAll.sortreverse()[0];
	declare Real[] RightSizeXAll = [_Instance.Properties.SliceList[C_SliceId_TopRight].Size.X, _Instance.Properties.SliceList[C_SliceId_CenterRight].Size.X, _Instance.Properties.SliceList[C_SliceId_BottomRight].Size.X];
	declare Real RightSizeXMax = RightSizeXAll.sortreverse()[0];
	declare Real[] TopSizeYAll = [_Instance.Properties.SliceList[C_SliceId_TopLeft].Size.Y, _Instance.Properties.SliceList[C_SliceId_TopCenter].Size.Y, _Instance.Properties.SliceList[C_SliceId_TopRight].Size.Y];
	declare Real TopSizeYMax = TopSizeYAll.sortreverse()[0];
	declare Real[] BottomSizeYAll = [_Instance.Properties.SliceList[C_SliceId_BottomLeft].Size.Y, _Instance.Properties.SliceList[C_SliceId_BottomCenter].Size.Y, _Instance.Properties.SliceList[C_SliceId_BottomRight].Size.Y];
	declare Real BottomSizeYMax = BottomSizeYAll.sortreverse()[0];

	declare Real RatioSizeX = 1.;
	declare Real CenterSizeX = 0.;
	if (LeftSizeXMax + RightSizeXMax > _Instance.Box.ContentArea.Size.X && _Instance.Properties.BorderFlexible) {
		if (LeftSizeXMax + RightSizeXMax != 0.) {
			RatioSizeX = _Instance.Box.ContentArea.Size.X / (LeftSizeXMax + RightSizeXMax);
		} else {
			RatioSizeX = 0.;
		}
	} else {
		CenterSizeX = ML::Max(0., _Instance.Box.ContentArea.Size.X - LeftSizeXMax - RightSizeXMax);
	}

	declare Real RatioSizeY = 1.;
	declare Real CenterSizeY = 0.;
	if (TopSizeYMax + BottomSizeYMax > _Instance.Box.ContentArea.Size.Y && _Instance.Properties.BorderFlexible) {
		if (TopSizeYMax + BottomSizeYMax != 0.) {
			RatioSizeY = _Instance.Box.ContentArea.Size.Y / (TopSizeYMax + BottomSizeYMax);
		} else {
			RatioSizeY = 0.;
		}
	} else {
		CenterSizeY = ML::Max(0., _Instance.Box.ContentArea.Size.Y - TopSizeYMax - BottomSizeYMax);
	}

	// Scale both size with the smallest ratio to respect the images aspect ratio
	declare Real RatioSize = ML::Min(RatioSizeX, RatioSizeY);

	_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size = <_Instance.Properties.SliceList[C_SliceId_TopLeft].Size.X * RatioSize, _Instance.Properties.SliceList[C_SliceId_TopLeft].Size.Y * RatioSize>;
	_Instance.Controls.QuadList_Slice[C_SliceId_TopRight].Size = <_Instance.Properties.SliceList[C_SliceId_TopRight].Size.X * RatioSize, _Instance.Properties.SliceList[C_SliceId_TopRight].Size.Y * RatioSize>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomLeft].Size = <_Instance.Properties.SliceList[C_SliceId_BottomLeft].Size.X * RatioSize, _Instance.Properties.SliceList[C_SliceId_BottomLeft].Size.Y * RatioSize>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomRight].Size = <_Instance.Properties.SliceList[C_SliceId_BottomRight].Size.X * RatioSize, _Instance.Properties.SliceList[C_SliceId_BottomRight].Size.Y * RatioSize>;

	_Instance.Controls.QuadList_Slice[C_SliceId_TopCenter].Size = <CenterSizeX, _Instance.Properties.SliceList[C_SliceId_TopCenter].Size.Y * RatioSize>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterRight].Size = <_Instance.Properties.SliceList[C_SliceId_CenterRight].Size.X * RatioSize, CenterSizeY>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomCenter].Size = <CenterSizeX, _Instance.Properties.SliceList[C_SliceId_BottomCenter].Size.Y * RatioSize>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].Size = <_Instance.Properties.SliceList[C_SliceId_CenterLeft].Size.X * RatioSize, CenterSizeY>;

	_Instance.Controls.QuadList_Slice[C_SliceId_CenterCenter].Size = <CenterSizeX, CenterSizeY>;

	_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].RelativePosition_V3 = <
		(LeftSizeXMax * RatioSize) - _Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size.X,
		-((TopSizeYMax * RatioSize) - _Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size.Y)
	>;
	_Instance.Controls.QuadList_Slice[C_SliceId_TopCenter].RelativePosition_V3 = <
		_Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].RelativePosition_V3.X + _Instance.Controls.QuadList_Slice[C_SliceId_TopLeft].Size.X,
		-((TopSizeYMax * RatioSize) - _Instance.Controls.QuadList_Slice[C_SliceId_TopCenter].Size.Y)
	>;
	_Instance.Controls.QuadList_Slice[C_SliceId_TopRight].RelativePosition_V3 = <
		_Instance.Controls.QuadList_Slice[C_SliceId_TopCenter].RelativePosition_V3.X + _Instance.Controls.QuadList_Slice[C_SliceId_TopCenter].Size.X,
		-((TopSizeYMax * RatioSize) - _Instance.Controls.QuadList_Slice[C_SliceId_TopRight].Size.Y)
	>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].RelativePosition_V3 = <
		(LeftSizeXMax * RatioSize) - _Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].Size.X,
		-(TopSizeYMax * RatioSize)
	>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterCenter].RelativePosition_V3 = <
		_Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].RelativePosition_V3.X + _Instance.Controls.QuadList_Slice[C_SliceId_CenterLeft].Size.X,
		-(TopSizeYMax * RatioSize)
	>;
	_Instance.Controls.QuadList_Slice[C_SliceId_CenterRight].RelativePosition_V3 = <
		_Instance.Controls.QuadList_Slice[C_SliceId_CenterCenter].RelativePosition_V3.X + _Instance.Controls.QuadList_Slice[C_SliceId_CenterCenter].Size.X,
		-(TopSizeYMax * RatioSize)
	>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomLeft].RelativePosition_V3 = <
		(LeftSizeXMax * RatioSize) - _Instance.Controls.QuadList_Slice[C_SliceId_BottomLeft].Size.X,
		-(TopSizeYMax * RatioSize) - CenterSizeY
	>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomCenter].RelativePosition_V3 = <
		_Instance.Controls.QuadList_Slice[C_SliceId_BottomLeft].RelativePosition_V3.X + _Instance.Controls.QuadList_Slice[C_SliceId_BottomLeft].Size.X,
		-(TopSizeYMax * RatioSize) - CenterSizeY
	>;
	_Instance.Controls.QuadList_Slice[C_SliceId_BottomRight].RelativePosition_V3 = <
		_Instance.Controls.QuadList_Slice[C_SliceId_BottomCenter].RelativePosition_V3.X + _Instance.Controls.QuadList_Slice[C_SliceId_BottomCenter].Size.X,
		-(TopSizeYMax * RatioSize) - CenterSizeY
	>;

/// Redraw the component
Void Private_Hook_ComponentDraw(CMlFrame _Frame_Slice9, Text[] _Tags) {
	declare K_Private_Instance CMGame_ComponentsSlice9_Instance as Instance for _Frame_Slice9 = C_Private_Instance_Null;

	Instance = Private_UpdateDirty(Instance);

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Box])) {
		Instance.Controls.Frame_ContentArea.RelativePosition_V3 = Instance.Box.ContentArea.Pos;

		switch (Instance.Properties.BorderResize) {
			case C_BorderResize_Stretch: {
				Private_SetSliceSizes_Stretch(Instance);
			}
			default: { // C_BorderResize_Space
				Private_SetSliceSizes_Space(Instance);
			}
		}
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Box, C_Private_RedrawTag_Align])) {
		MLHelpers::AlignContent(Instance.Controls.Frame_Align, Instance.Box.MarginArea.Size, Instance.Properties.HAlign, Instance.Properties.VAlign);
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_IsVisible])) {
		Instance.Controls.Frame_Slice9.Visible = Instance.Properties.IsVisible;
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_SliceList])) {
		for (SliceId => Quad_Slice in Instance.Controls.QuadList_Slice) {
			Quad_Slice.Visible = Instance.Properties.SliceList[SliceId].Opacity > 0.;
			Quad_Slice.Opacity = Instance.Properties.SliceList[SliceId].Opacity;
			Quad_Slice.ImageUrl = Instance.Properties.SliceList[SliceId].ImageUrl;
			Private_SetSliceColor(Quad_Slice, Instance.Properties.SliceList[SliceId]);
			Private_SetSliceMirror(Quad_Slice, Instance.Properties.SliceList[SliceId]);
		}
	}
}
