/**
 * Component: Checkbox
 * ===========================
 *
 * The `Checkbox` component was designed to be used with the `Navigation` class.
 * With this component, the player will for example be able to check or uncheck a property or a parameter.
 * The mouse detection zone for the navigation will automatically adapt depending on :
 * - the size of the label (`text-size`),
 * - the size of the icon (`icon-size`),
 * - their respective position inside the component (`text-pos` and `icon-pos`)
 *
 * Events
 * ----------
 * - C_Event_Toggle                the event that is sent when the checkbox is toggled
 *
 * Parameters
 * ----------
 * - `halign` (`CMlControl::AlignHorizontal`): horizontal alignment of the checkbox
 *      Manialink parameters: `left`, `center` or `right`
 * - `valign` (`CMlControl::AlignVertical`): vertical alignment of the checkbox
 *      Manialink parameters: `top`, `center`, `center2` or `bottom`
 * - `size` (`Vec2`): size of the checkbox
 * - `opacity-focused` (`Real`): opacity of images and texts when the component is focused
 * - `opacity-unfocused` (`Real`): opacity of images and texts when the component is not focused
 * - `text` (`Text`): text displayed in the checkbox
 * - `text-mirror` (`Text`): mirrored text displayed in the checkbox (for switch-like checkbox)
 * - `text-prefix` (`Text`): prefix of the texts displayed in the checkbox
 * - `text-font` (`Text`): font of the texts displayed in the checkbox
 * - `text-color-unfocused` (`Vec3`): color of the texts when the component if not focused
 * - `text-color-focused` (`Vec3`): color of the texts when the component if focused
 * - `text-pos` (`Vec2`): position of the text in the component <[0;1], [0;1]> (centered in <0.5, 0.5>)
 * - `text-halign` (`CMlControl::AlignHorizontal`): horizontal alignment of the text displayed in the checkbox
 *      Manialink parameters: `left`, `center` or `right`
 * - `text-valign` (`CMlControl::AlignVertical`): vertical alignment of the text displayed in the checkbox
 *      Manialink parameters: `top`, `center`, `center2` or `bottom`
 * - `text-italicslope` (`Real`): italic slope of the texts displayed in the checkbox
 * - `text-fitlabel-enabled` (`Real`): fit the texts in the checkbox when it exceeds its width
 * - `text-size` (`Real`): size of the texts displayed in the checkbox
 * - `text-autonewline-enabled` (`Boolean`): enable text with multiple lines and better handle text containing '\n' characters
 * - `text-maxlines` (`Integer`): sets the maximum line of text in the checkbox
 * - `text-adaptablesize-enabled` (`Boolean`): enable an automatic resize of the texts displayed in the checkbox
 * - `icon-size` (`Vec2`): size of the icon displayed in the checkbox
 * - `icon-color-unfocused` (`Vec3`): color of the icon when the component is not focused
 * - `icon-color-focused` (`Vec3`): color of the icon when the component is focused
 * - `icon-pos` (`Vec2`): position of the icon diplayed in the checkbox <[0;1], [0;1]> (centered in <0.5, 0.5>)
 * - `icon-image-checked` (`Text`): url to the image used when the checkbox is checked
 * - `icon-image-unchecked` (`Text`): url to the image used when the checkbox is unchecked
 */

/// Update the colors and opacities of the component
Void Private_ApplyOpacityAndColor(CMlScript _This, CMlFrame _Frame_Checkbox, Boolean _IsAnimated) {
	declare K_Checkbox CMGame_Checkbox2_Checkbox for _Frame_Checkbox = C_Checkbox_Null;

	_This.AnimMgr.Flush(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox);
	_This.AnimMgr.Flush(CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox);
	_This.AnimMgr.Flush(CMGame_Checkbox2_Checkbox.Controls.Quad_Icon);

	if (CMGame_Checkbox2_Checkbox.IsLocked) {
		if (_IsAnimated) {
			_This.AnimMgr.Add(CMGame_Checkbox2_Checkbox.Controls.Quad_Icon, """<anim colorize="{{{CL::RgbToHex6(CMGame_Checkbox2_Checkbox.MergedProperties.IconColorUnfocused)}}}" opacity="0.5"/>""", C_FocusAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
			_This.AnimMgr.Add(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox, """<anim textcolor="{{{CL::RgbToHex6(CMGame_Checkbox2_Checkbox.MergedProperties.TextColorUnfocused)}}}" opacity="0.5"/>""", C_FocusAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
			if (CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue != "") {
				_This.AnimMgr.Add(CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox, """<anim textcolor="{{{CL::RgbToHex6(CMGame_Checkbox2_Checkbox.MergedProperties.TextColorUnfocused)}}}" opacity="0.5"/>""", C_FocusAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
			}
		} else {
			CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Colorize = CMGame_Checkbox2_Checkbox.MergedProperties.IconColorUnfocused;
			CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Opacity = 0.5;
			CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TextColor = CMGame_Checkbox2_Checkbox.MergedProperties.TextColorUnfocused;
			CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Opacity = 0.5;
			if (CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue != "") {
				CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.TextColor = CMGame_Checkbox2_Checkbox.MergedProperties.TextColorUnfocused;
				CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Opacity = 0.5;
			}
		}
	} else if (CMGame_Checkbox2_Checkbox.IsFocused) {
		if (_IsAnimated) {
			_This.AnimMgr.Add(CMGame_Checkbox2_Checkbox.Controls.Quad_Icon, """<anim colorize="{{{CL::RgbToHex6(CMGame_Checkbox2_Checkbox.MergedProperties.IconColorFocused)}}}" opacity="{{{CMGame_Checkbox2_Checkbox.MergedProperties.OpacityFocused}}}"/>""", C_FocusAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
			_This.AnimMgr.Add(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox, """<anim textcolor="{{{CL::RgbToHex6(CMGame_Checkbox2_Checkbox.MergedProperties.TextColorFocused)}}}" opacity="{{{CMGame_Checkbox2_Checkbox.MergedProperties.OpacityFocused}}}"/>""", C_FocusAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
			if (CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue != "") {
				_This.AnimMgr.Add(CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox, """<anim textcolor="{{{CL::RgbToHex6(CMGame_Checkbox2_Checkbox.MergedProperties.TextColorFocused)}}}" opacity="{{{CMGame_Checkbox2_Checkbox.MergedProperties.OpacityFocused}}}"/>""", C_FocusAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
			}
		} else {
			CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Colorize = CMGame_Checkbox2_Checkbox.MergedProperties.IconColorFocused;
			CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Opacity = CMGame_Checkbox2_Checkbox.MergedProperties.OpacityFocused;
			CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TextColor = CMGame_Checkbox2_Checkbox.MergedProperties.TextColorFocused;
			CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Opacity = CMGame_Checkbox2_Checkbox.MergedProperties.OpacityFocused;
			if (CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue != "") {
				CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.TextColor = CMGame_Checkbox2_Checkbox.MergedProperties.TextColorFocused;
				CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Opacity = CMGame_Checkbox2_Checkbox.MergedProperties.OpacityFocused;
			}
		}
	} else if (_IsAnimated) {
		_This.AnimMgr.Add(CMGame_Checkbox2_Checkbox.Controls.Quad_Icon, """<anim colorize="{{{CL::RgbToHex6(CMGame_Checkbox2_Checkbox.MergedProperties.IconColorUnfocused)}}}" opacity="{{{CMGame_Checkbox2_Checkbox.MergedProperties.OpacityUnfocused}}}"/>""", C_FocusAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
		_This.AnimMgr.Add(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox, """<anim textcolor="{{{CL::RgbToHex6(CMGame_Checkbox2_Checkbox.MergedProperties.TextColorUnfocused)}}}" opacity="{{{CMGame_Checkbox2_Checkbox.MergedProperties.OpacityUnfocused}}}"/>""", C_FocusAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
		if (CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue != "") {
			_This.AnimMgr.Add(CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox, """<anim textcolor="{{{CL::RgbToHex6(CMGame_Checkbox2_Checkbox.MergedProperties.TextColorUnfocused)}}}" opacity="{{{CMGame_Checkbox2_Checkbox.MergedProperties.OpacityUnfocused}}}"/>""", C_FocusAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
		}
	} else {
		CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Colorize = CMGame_Checkbox2_Checkbox.MergedProperties.IconColorUnfocused;
		CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Opacity = CMGame_Checkbox2_Checkbox.MergedProperties.OpacityUnfocused;
		CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TextColor = CMGame_Checkbox2_Checkbox.MergedProperties.TextColorUnfocused;
		CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Opacity = CMGame_Checkbox2_Checkbox.MergedProperties.OpacityUnfocused;
		if (CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue != "") {
			CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.TextColor = CMGame_Checkbox2_Checkbox.MergedProperties.TextColorUnfocused;
			CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Opacity = CMGame_Checkbox2_Checkbox.MergedProperties.OpacityUnfocused;
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// The component is focused by the Navigation component
Void Private_Focus(CMlScript _This, CMlFrame _Frame_Checkbox, Boolean _IsTTSFocused, Boolean _IsAnimated) {
	declare K_Checkbox CMGame_Checkbox2_Checkbox for _Frame_Checkbox = C_Checkbox_Null;
	if (CMGame_Checkbox2_Checkbox.IsLocked) return;

	CMGame_Checkbox2_Checkbox.IsFocused = True;
	Private_ApplyOpacityAndColor(_This, _Frame_Checkbox, _IsAnimated);
	if (_IsTTSFocused) CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TTS_Focus();
	else CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TTS_Unfocus();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// The component is unfocused by the Navigation component
Void Private_Unfocus(CMlScript _This, CMlFrame _Frame_Checkbox, Boolean _IsAnimated) {
	declare K_Checkbox CMGame_Checkbox2_Checkbox for _Frame_Checkbox = C_Checkbox_Null;
	if (CMGame_Checkbox2_Checkbox.IsLocked) return;

	CMGame_Checkbox2_Checkbox.IsFocused = False;
	Private_ApplyOpacityAndColor(_This, _Frame_Checkbox, _IsAnimated);
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TTS_Unfocus();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Enable or disable a properties from the `Enabled` array
Integer[] Private_EnableProperty(Integer[] _EnabledProperties, Integer _Property, Boolean _Enabled) {
	declare Integer[] EnabledProperties = _EnabledProperties;

	if (_Enabled) {
		if (!EnabledProperties.exists(_Property)) {
			EnabledProperties.add(_Property);
		}
	} else {
		EnabledProperties.remove(_Property);
	}

	return EnabledProperties;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set and enable a property
K_Properties Private_SetProperty(K_Properties _TargetProperties, Integer _PropertyId, K_Properties _SourceProperties) {
	declare K_Properties Properties = _TargetProperties;

	Properties.Enabled = Private_EnableProperty(Properties.Enabled, _PropertyId, True);

	switch (_PropertyId) {
		case C_Property_HAlign: Properties.HAlign = _SourceProperties.HAlign;
		case C_Property_VAlign: Properties.VAlign = _SourceProperties.VAlign;
		case C_Property_Size: Properties.Size = _SourceProperties.Size;
		case C_Property_OpacityFocused: Properties.OpacityFocused = _SourceProperties.OpacityFocused;
		case C_Property_OpacityUnfocused: Properties.OpacityUnfocused = _SourceProperties.OpacityUnfocused;
		case C_Property_TextValue: Properties.TextValue = _SourceProperties.TextValue;
		case C_Property_TextMirrorValue: Properties.TextMirrorValue = _SourceProperties.TextMirrorValue;
		case C_Property_TextPrefix: Properties.TextPrefix = _SourceProperties.TextPrefix;
		case C_Property_TextFont: Properties.TextFont = _SourceProperties.TextFont;
		case C_Property_TextColorUnfocused: Properties.TextColorUnfocused = _SourceProperties.TextColorUnfocused;
		case C_Property_TextColorFocused: Properties.TextColorFocused = _SourceProperties.TextColorFocused;
		case C_Property_TextPos: Properties.TextPos = _SourceProperties.TextPos;
		case C_Property_TextHAlign: Properties.TextHAlign = _SourceProperties.TextHAlign;
		case C_Property_TextVAlign: Properties.TextVAlign = _SourceProperties.TextVAlign;
		case C_Property_TextItalicSlope: Properties.TextItalicSlope = _SourceProperties.TextItalicSlope;
		case C_Property_TextFitLabelEnabled: Properties.TextFitLabelEnabled = _SourceProperties.TextFitLabelEnabled;
		case C_Property_TextSize: Properties.TextSize = _SourceProperties.TextSize;
		case C_Property_TextAutoNewLineEnabled: Properties.TextAutoNewLineEnabled = _SourceProperties.TextAutoNewLineEnabled;
		case C_Property_TextMaxLines: Properties.TextMaxLines = _SourceProperties.TextMaxLines;
		case C_Property_TextAdaptableSizeEnabled: Properties.TextAdaptableSizeEnabled = _SourceProperties.TextAdaptableSizeEnabled;
		case C_Property_IconSize: Properties.IconSize = _SourceProperties.IconSize;
		case C_Property_IconColorUnfocused: Properties.IconColorUnfocused = _SourceProperties.IconColorUnfocused;
		case C_Property_IconColorFocused: Properties.IconColorFocused = _SourceProperties.IconColorFocused;
		case C_Property_IconPos: Properties.IconPos = _SourceProperties.IconPos;
		case C_Property_IconImageChecked: Properties.IconImageChecked = _SourceProperties.IconImageChecked;
		case C_Property_IconImageUnchecked: Properties.IconImageUnchecked = _SourceProperties.IconImageUnchecked;
	}

	return Properties;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Merge a base properties with other enabled properties
K_Properties Private_MergeProperties(K_Properties _Base, K_Properties[] _PropertiesToMerge) {
	declare K_Properties MergedProperties = _Base;

	foreach (Properties in _PropertiesToMerge) {
		foreach (PropertyId in Properties.Enabled) {
			MergedProperties = Private_SetProperty(MergedProperties, PropertyId, Properties);
		}
	}

	return MergedProperties;
}

/// Redraw the component (apply size, color...)
Void Private_Redraw(CMlScript _This, CMlFrame _Frame_Checkbox) {
	declare K_State CMGame_Checkbox2_State for _This.Page = C_State_Null;
	declare K_Checkbox CMGame_Checkbox2_Checkbox for _Frame_Checkbox = C_Checkbox_Null;

	// Merge properties
	CMGame_Checkbox2_Checkbox.MergedProperties = Private_MergeProperties(
		CMGame_Checkbox2_State.DefaultProperties,
		[CMGame_Checkbox2_Checkbox.CustomizedProperties]
	);

	// Checkbox labels
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Value = CMGame_Checkbox2_Checkbox.MergedProperties.TextValue;
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TextSizeReal = CMGame_Checkbox2_Checkbox.MergedProperties.TextSize;
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TextPrefix = CMGame_Checkbox2_Checkbox.MergedProperties.TextPrefix;
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TextFont = CMGame_Checkbox2_Checkbox.MergedProperties.TextFont;
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.HorizontalAlign = CMGame_Checkbox2_Checkbox.MergedProperties.TextHAlign;
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.VerticalAlign = CMGame_Checkbox2_Checkbox.MergedProperties.TextVAlign;
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.ItalicSlope = CMGame_Checkbox2_Checkbox.MergedProperties.TextItalicSlope;
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.AutoNewLine = CMGame_Checkbox2_Checkbox.MergedProperties.TextAutoNewLineEnabled;
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.MaxLine = CMGame_Checkbox2_Checkbox.MergedProperties.TextMaxLines;
	if (CMGame_Checkbox2_Checkbox.MergedProperties.TextAdaptableSizeEnabled) {
		CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size = <
			ML::Min(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.ComputeWidth(TL::GetTranslatedText(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Value), False), CMGame_Checkbox2_Checkbox.MergedProperties.Size.X),
			CMGame_Checkbox2_Checkbox.MergedProperties.Size.Y
		>;
	} else {
		CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size = CMGame_Checkbox2_Checkbox.MergedProperties.Size;
	}

	if (CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue == "") {
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Value = "";
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Size = <0., 0.>;
	} else {
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Value = CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue;
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.TextSizeReal = CMGame_Checkbox2_Checkbox.MergedProperties.TextSize;
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.TextPrefix = CMGame_Checkbox2_Checkbox.MergedProperties.TextPrefix;
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.TextFont = CMGame_Checkbox2_Checkbox.MergedProperties.TextFont;
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.HorizontalAlign = Tools::MirrorHAlign(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.HorizontalAlign);
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.VerticalAlign = Tools::MirrorVAlign(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.VerticalAlign);
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.ItalicSlope = CMGame_Checkbox2_Checkbox.MergedProperties.TextItalicSlope;
		if (CMGame_Checkbox2_Checkbox.MergedProperties.TextAdaptableSizeEnabled) {
			CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Size = <
				ML::Min(CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.ComputeWidth(TL::GetTranslatedText(CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Value), False), CMGame_Checkbox2_Checkbox.MergedProperties.Size.X),
				CMGame_Checkbox2_Checkbox.MergedProperties.Size.Y
			>;
		} else {
			CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Size = CMGame_Checkbox2_Checkbox.MergedProperties.Size;
		}
	}

	// Text to speech
	if (CMGame_Checkbox2_Checkbox.IsChecked) {
		if (CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue != "") { //< switch button
			CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TTS_AltText = TL::Compose("%1 %2", C_TTS_SwitchButton, CMGame_Checkbox2_Checkbox.MergedProperties.TextValue);
		} else {
			CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TTS_AltText = TL::Compose("%1 %2", CMGame_Checkbox2_Checkbox.MergedProperties.TextValue, C_TTS_Checked);
		}
	} else {
		if (CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue != "") { //< switch button
			CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TTS_AltText = TL::Compose("%1 %2", C_TTS_SwitchButton, CMGame_Checkbox2_Checkbox.MergedProperties.TextMirrorValue);
		} else {
			CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TTS_AltText = TL::Compose("%1 %2", CMGame_Checkbox2_Checkbox.MergedProperties.TextValue, C_TTS_Unchecked);
		}
	}

	// Checkbox icon
	CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size = CMGame_Checkbox2_Checkbox.MergedProperties.IconSize;
	if (CMGame_Checkbox2_Checkbox.IsChecked && CMGame_Checkbox2_Checkbox.MergedProperties.IconImageChecked != "") {
		CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.ChangeImageUrl(CMGame_Checkbox2_Checkbox.MergedProperties.IconImageChecked);
		CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Visible = True;
	} else if (!CMGame_Checkbox2_Checkbox.IsChecked && CMGame_Checkbox2_Checkbox.MergedProperties.IconImageUnchecked != "") {
		CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.ChangeImageUrl(CMGame_Checkbox2_Checkbox.MergedProperties.IconImageUnchecked);
		CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Visible = True;
	} else {
		CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Visible = False;
	}

	// Position
	declare Vec2 SizeOfReference = <
		ML::Max(CMGame_Checkbox2_Checkbox.MergedProperties.Size.X, CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.X),
		ML::Max(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.Y, CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.Y)
	>;
	CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3 = <
		SizeOfReference.X * (CMGame_Checkbox2_Checkbox.MergedProperties.IconPos.X-0.5),
		SizeOfReference.Y * (CMGame_Checkbox2_Checkbox.MergedProperties.IconPos.Y-0.5)
	>;
	CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.RelativePosition_V3 = <
		SizeOfReference.X * (CMGame_Checkbox2_Checkbox.MergedProperties.TextPos.X-0.5),
		SizeOfReference.Y * (CMGame_Checkbox2_Checkbox.MergedProperties.TextPos.Y-0.5)
	>;
	if (CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Value != "") {
		CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.RelativePosition_V3 = <
			SizeOfReference.X * (1-CMGame_Checkbox2_Checkbox.MergedProperties.TextPos.X-0.5),
			SizeOfReference.Y * (1-CMGame_Checkbox2_Checkbox.MergedProperties.TextPos.Y-0.5)
		>;
	}

	// Get the center pos of the label depending on its alignment
	declare Vec2 Pos_LabelCenter = CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.RelativePosition_V3;
	switch (CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.HorizontalAlign) {
		case CMlControl::AlignHorizontal::Left: Pos_LabelCenter.X += CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.X/2;
		case CMlControl::AlignHorizontal::Right: Pos_LabelCenter.X -= CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.X/2;
	}
	switch (CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.VerticalAlign) {
		case CMlControl::AlignVertical::Top: Pos_LabelCenter.Y -= CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.Y/2;
		case CMlControl::AlignVertical::Bottom: Pos_LabelCenter.Y += CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.Y/2;
	}

	// Activation zone
	if (CMGame_Checkbox2_Checkbox.MergedProperties.TextAdaptableSizeEnabled) SizeOfReference = <ML::Max(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.X, CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.X), ML::Max(CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.Y, CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.Y)>;
	if (CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Value == "") {
		CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.Size = <
			ML::Max(SizeOfReference.X, CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.X/2 + ML::Abs(CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.X - Pos_LabelCenter.X) + CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.X/2),
			ML::Max(SizeOfReference.Y, CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.Y/2 + ML::Abs(CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.Y - Pos_LabelCenter.Y) + CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.Y/2)
		>;
		CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.RelativePosition_V3 = CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3;
		if (CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.X < Pos_LabelCenter.X) {
			CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.RelativePosition_V3.X = ((CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.X - CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.X*0.5) + (Pos_LabelCenter.X + CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.X*0.5))*0.5;
		} else if (CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.X > Pos_LabelCenter.X) {
			CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.RelativePosition_V3.X = ((Pos_LabelCenter.X - CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.X*0.5) + (CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.X + CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.X*0.5))*0.5;
		}
		if (CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.Y < Pos_LabelCenter.Y) {
			CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.RelativePosition_V3.Y = ((CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.Y - CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.Y*0.5) + (Pos_LabelCenter.Y + CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.Y*0.5))*0.5;
		} else if (CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.Y > Pos_LabelCenter.Y) {
			CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.RelativePosition_V3.Y = ((Pos_LabelCenter.Y - CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.Y*0.5) + (CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.Y + CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.Y*0.5))*0.5;
		}
	} else {
		declare Real Margin = ML::Abs(CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3.X - Pos_LabelCenter.X) - CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.X/2 - CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.X/2;
		CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.Size.X = ML::Max(SizeOfReference.X * 2, CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.Size.X + (2 * Margin) + (2 * CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.Size.X));
		CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.Size.Y = SizeOfReference.Y;
		CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.RelativePosition_V3 = CMGame_Checkbox2_Checkbox.Controls.Quad_Icon.RelativePosition_V3;
	}

	// Align
	Tools::Align(CMGame_Checkbox2_Checkbox.Controls.Frame_Align, CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.Size, CMGame_Checkbox2_Checkbox.MergedProperties.HAlign, CMGame_Checkbox2_Checkbox.MergedProperties.VAlign);
	CMGame_Checkbox2_Checkbox.Controls.Frame_Align.RelativePosition_V3 -= CMGame_Checkbox2_Checkbox.Controls.Quad_NavigationZone.RelativePosition_V3; //< Adjust the offset created by the label

	// Fit - Apply after every other label properties, especially its size
	if (CMGame_Checkbox2_Checkbox.MergedProperties.TextFitLabelEnabled) {
		Tools::FitLabelValue(
			CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox,
			CMGame_Checkbox2_Checkbox.MergedProperties.TextSize,
			CMGame_Checkbox2_Checkbox.MergedProperties.TextSize / Tools::C_FitLabelMaxSteps,
			CMGame_Checkbox2_Checkbox.MergedProperties.TextSize / Tools::C_FitLabelMaxSteps
		);
		if (CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox.Value != "") {
			Tools::FitLabelValue(
				CMGame_Checkbox2_Checkbox.Controls.Label_MirrorCheckbox,
				CMGame_Checkbox2_Checkbox.MergedProperties.TextSize,
				CMGame_Checkbox2_Checkbox.MergedProperties.TextSize / Tools::C_FitLabelMaxSteps,
				CMGame_Checkbox2_Checkbox.MergedProperties.TextSize / Tools::C_FitLabelMaxSteps
			);
		}
	}

	// Update focus state
	Private_ApplyOpacityAndColor(_This, _Frame_Checkbox, False);
}

/// Manialink page yield hook
Void Private_Hook_PageAfterYield(CMlScript _This) {
	// Manage navigation events
	foreach (Event in Navigation::GetPendingEvents(_This)) {
		switch (Event.Type) {
			case Navigation::C_EventType_Navigate: {
				switch (Event.Input) {
					case Navigation::C_InputType_Select, Navigation::C_InputType_MouseClick: {
						declare K_Checkbox CMGame_Checkbox2_Checkbox for Event.To = C_Checkbox_Null;
						if (CMGame_Checkbox2_Checkbox.IsCheckbox && !CMGame_Checkbox2_Checkbox.IsLocked && !Event.IsActionAutoRepeat) {
							CMGame_Checkbox2_Checkbox.IsChecked = !CMGame_Checkbox2_Checkbox.IsChecked;
							Private_Redraw(_This, Event.To as CMlFrame);

							// Trigger TTS focus after component update
							if (CMGame_Checkbox2_Checkbox.IsChecked && CMGame_Checkbox2_Checkbox.IsTTSFocusedOnCheck) {
								CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TTS_Focus();
							}

							// Send an event when the check state is updated
							declare K_Event_Toggle[] CMGame_Checkbox_Events_Toggle for _This.Page = [];
							CMGame_Checkbox_Events_Toggle.add(K_Event_Toggle {
								Component = Event.To as CMlFrame,
								IsChecked = CMGame_Checkbox2_Checkbox.IsChecked
							});
						}
					}
				}
			}
		}
	}
}

	declare K_Event_Toggle[] CMGame_Checkbox_Events_Toggle for _This.Page = [];

}

	Private_AssertValidComponent(_Frame_Checkbox);
	declare K_Checkbox CMGame_Checkbox2_Checkbox for _Frame_Checkbox = C_Checkbox_Null;
	return CMGame_Checkbox2_Checkbox.IsChecked;

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check or uncheck the checkbox
Boolean Check(CMlScript _This, CMlFrame _Frame_Checkbox, Boolean _IsChecked, Boolean _IsRedrawRequired) {
	Private_AssertValidComponent(_Frame_Checkbox);
	declare K_Checkbox CMGame_Checkbox2_Checkbox for _Frame_Checkbox = C_Checkbox_Null;
	if (CMGame_Checkbox2_Checkbox.IsLocked) return CMGame_Checkbox2_Checkbox.IsChecked;
	declare Boolean IsTTSFocused = CMGame_Checkbox2_Checkbox.IsChecked != _IsChecked && CMGame_Checkbox2_Checkbox.IsTTSFocusedOnCheck;

	// Send an event when the check state is updated
	if (CMGame_Checkbox2_Checkbox.IsChecked != _IsChecked) {
		declare K_Event_Toggle[] CMGame_Checkbox_Events_Toggle for _This.Page = [];
		CMGame_Checkbox_Events_Toggle.add(K_Event_Toggle {
			Component = _Frame_Checkbox,
			IsChecked = _IsChecked
		});
	}

	// Update the component
	CMGame_Checkbox2_Checkbox.IsChecked = _IsChecked;
	if (_IsRedrawRequired) Private_Redraw(_This, _Frame_Checkbox);

	// Trigger TTS focus after component update
	if (IsTTSFocused) {
		CMGame_Checkbox2_Checkbox.Controls.Label_Checkbox.TTS_Focus();
	}

	return CMGame_Checkbox2_Checkbox.IsChecked;
}
Boolean Check(CMlScript _This, CMlFrame _Frame_Checkbox, Boolean _IsChecked) {
	return Check(_This, _Frame_Checkbox, _IsChecked, True);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Invert the check state of the checkbox
Boolean Toggle(CMlScript _This, CMlFrame _Frame_Checkbox, Boolean _IsRedrawRequired) {
	Private_AssertValidComponent(_Frame_Checkbox);
	declare K_Checkbox CMGame_Checkbox2_Checkbox for _Frame_Checkbox = C_Checkbox_Null;
	return Check(_This, _Frame_Checkbox, !CMGame_Checkbox2_Checkbox.IsChecked, _IsRedrawRequired);
}
Boolean Toggle(CMlScript _This, CMlFrame _Frame_Checkbox) {
	return Toggle(_This, _Frame_Checkbox, True);
}
