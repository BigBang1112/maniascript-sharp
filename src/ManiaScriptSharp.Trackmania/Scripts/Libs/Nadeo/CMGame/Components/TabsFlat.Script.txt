/**
 * TabsFlat component
 * ===================
 *
 * Tabs component with a flat style
 *
 * Props
 * -----
 *
 * - `size` (`Vec2`): size of the component
 * - `margin` (`Margin`): margin around the component
 * - `margin-ratio` (`Margin`): margin for the component proportional to the component size
 * - `padding` (`Padding`): padding inside the component
 * - `padding-ratio` (`Padding`): padding for the component proportional to the component size
 * - `halign` (`Text`): horizontal alignment of the component, can be `left`, `center` or `right`
 * - `valign` (`Text`): vertical alignment of the component, can be `top`, `center` or `bottom`
 * - `is-visible` (`Boolean`): the visibility of the component
 * - `is-navigation-enabled` (`Boolean`): the state of the navigation in the component
 * - `tab-width` (`Integer`): how the tab width should be computed, can be:
 *   ~ `C_TabWidth_TextCompact` to use the length of the tab text
 *   ~ `C_TabWidth_TextFill` to use the length of the tab text but expand each tab proportionally to fill the full width of the component
 *   ~ `C_TabWidth_Same` to divide the full width of the component equally between each tab
 * - `tab-gap` (`Real`): width of the gap between the tabs
 * - `tab-padding` (`Real`): padding inside the tab
 * - `button-previous-tab` (`Integer`): input used to go to the previous tab
 * - `button-next-tab` (`Integer`): input used to go to the next tab
 * - `button-width` (`Real`): width of the previous and next tab buttons
 */

/// Get the component template
Text Private_Template() {
	declare Text ML_Tabs;
	for (I, 1, C_TabsMax) {
		ML_Tabs ^= """<frameinstance modelid="{{{C_Private_Id_Framemodel_Tab}}}" />""";
	}

	return
"""
<framemodel id="{{{C_Private_Id_Framemodel_Tab}}}">
	<quad id="{{{C_Private_Id_Quad_TabBackground}}}" class="{{{MV::C_Class_TriggerZone}}}" z-index="0" scriptevents="1" />
	<label id="{{{C_Private_Id_Label_TabValue}}}" z-index="1" halign="center" valign="center2" textfont="{{{Stylesheet::GetFont(Stylesheet::C_Font_SemiBold)}}}" />
</framemodel>
<frame id="{{{C_Private_Id_Frame_Align}}}">
	<frame id="{{{C_Private_Id_Frame_Content}}}" z-index="1">
		<label id="{{{C_Private_Id_Label_TabWidth}}}" z-index="1" hidden="1" />
		<CMGameButtonIcon3 id="{{{C_Private_Id_ButtonIcon_PreviousTab}}}" z-index="2" />
		<CMGameButtonIcon3 id="{{{C_Private_Id_ButtonIcon_NextTab}}}" z-index="2" />
		<frame id="{{{C_Private_Id_Frame_Tabs}}}" z-index="3">
			{{{ML_Tabs}}}
		</frame>
	</frame>
</frame>
"""
	;
}

/// Select a tab
K_Private_Instance Private_SelectTabId(K_Private_Instance _Instance, Integer _SelectedTabId, Boolean _IsInit) {
	declare Integer SelectedTabKey = -1;
	for (TabKey => Tab in _Instance.Tabs) {
		if (Tab.Id == _SelectedTabId) {
			SelectedTabKey = TabKey;
			break;
		}
	}
	if (!_IsInit && _Instance.SelectedTabId == _SelectedTabId && _Instance.SelectedTabKey == SelectedTabKey) return _Instance;

	declare K_Private_Instance Instance = _Instance;
	Instance.SelectedTabId = _SelectedTabId;
	Instance.SelectedTabKey = SelectedTabKey;
	MV::MarkForRedraw(Instance.Controls.Frame_TabsFlat, C_Private_RedrawTag_Tabs);

	return Instance;
}
K_Private_Instance Private_SelectTabId(K_Private_Instance _Instance, Integer _SelectedTabId) {
	return Private_SelectTabId(_Instance, _SelectedTabId, False);
}
K_Private_Instance Private_SelectPreviousTab(K_Private_Instance _Instance) {
	if (_Instance.Tabs.count <= 0) return _Instance;

	declare Integer SelectedTabKey = _Instance.SelectedTabKey - 1;
	if (SelectedTabKey < 0) SelectedTabKey = _Instance.Tabs.count - 1;
	if (!_Instance.Tabs.existskey(SelectedTabKey)) return _Instance;

	return Private_SelectTabId(_Instance, _Instance.Tabs[SelectedTabKey].Id);
}
K_Private_Instance Private_SelectNextTab(K_Private_Instance _Instance) {
	if (_Instance.Tabs.count <= 0) return _Instance;

	declare Integer SelectedTabKey = _Instance.SelectedTabKey + 1;
	if (SelectedTabKey > _Instance.Tabs.count - 1) SelectedTabKey = 0;
	if (!_Instance.Tabs.existskey(SelectedTabKey)) return _Instance;

	return Private_SelectTabId(_Instance, _Instance.Tabs[SelectedTabKey].Id);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Focus a tab
K_Private_Instance Private_FocusTabKey(K_Private_Instance _Instance, Integer _FocusedTabKey, Boolean _IsFocused, Boolean _IsInit) {
	declare Integer[] FocusedTabKeys = _Instance.FocusedTabKeys;
	if (_IsFocused) {
		if (!FocusedTabKeys.exists(_FocusedTabKey)) FocusedTabKeys.add(_FocusedTabKey);
	} else {
		FocusedTabKeys.remove(_FocusedTabKey);
	}
	if (!_IsInit && _Instance.FocusedTabKeys == FocusedTabKeys) return _Instance;

	declare K_Private_Instance Instance = _Instance;
	Instance.FocusedTabKeys = FocusedTabKeys;
	MV::MarkForRedraw(Instance.Controls.Frame_TabsFlat, C_Private_RedrawTag_TabsFocus);

	return Instance;
}
K_Private_Instance Private_FocusTabKey(K_Private_Instance _Instance, Integer _FocusedTabKey, Boolean _IsFocused) {
	return Private_FocusTabKey(_Instance, _FocusedTabKey, _IsFocused, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the tabs displayed by the component
K_Private_Instance Private_SetTabs(K_Private_Instance _Instance, K_Private_Tab[] _Tabs, Integer _SelectedTabId, Boolean _IsInit) {
	if (!_IsInit && _Instance.Tabs == _Tabs && _Instance.SelectedTabId == _SelectedTabId) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	Instance.Tabs = _Tabs;
	if (Instance.Tabs.count > C_TabsMax) {
		log("""WARNING [CMGame/Components/Tabs] Too many tabs ({{{Instance.Tabs.count}}} > {{{C_TabsMax}}}) in component `#{{{_Instance.Controls.Frame_TabsFlat.ControlId}}}`""");
	}

	for (TabKey => TabControl in Instance.Controls.Tabs) {
		declare Integer CMGame_ComponentsTabsFlat_TabId for TabControl.Quad_TabBackground = -1;
		if (Instance.Tabs.existskey(TabKey)) {
			CMGame_ComponentsTabsFlat_TabId = Instance.Tabs[TabKey].Id;
		} else {
			CMGame_ComponentsTabsFlat_TabId = -1;
		}
	}

	Instance = Private_SelectTabId(Instance, _SelectedTabId);
	MV::MarkForRedraw(Instance.Controls.Frame_TabsFlat, C_Private_RedrawTag_Tabs);

	return Instance;
}
K_Private_Instance Private_SetTabs(K_Private_Instance _Instance, Text[Integer] _Tabs, Integer _SelectedTabId, Boolean _IsInit) {
	declare K_Private_Tab[] Tabs;
	for (TabId => TabValue in _Tabs) {
		Tabs.add(K_Private_Tab { Id = TabId, Value = TabValue });
	}
	return Private_SetTabs(_Instance, Tabs, _SelectedTabId, _IsInit);
}
K_Private_Instance Private_SetTabs(K_Private_Instance _Instance, Text[Integer] _Tabs, Integer _SelectedTabId) {
	return Private_SetTabs(_Instance, _Tabs, _SelectedTabId, False);
}

/// Convert a navigation input type to a button icon bind
Text Private_NavigationInputTypeToButtonIconBind(Integer _NavigationInputType) {
	switch (_NavigationInputType) {
		case C_InputType_Up: return ButtonIcon::C_Bind_Up;
		case C_InputType_Right: return ButtonIcon::C_Bind_Right;
		case C_InputType_Left: return ButtonIcon::C_Bind_Left;
		case C_InputType_Down: return ButtonIcon::C_Bind_Down;
		case C_InputType_Select: return ButtonIcon::C_Bind_Select;
		case C_InputType_Cancel: return ButtonIcon::C_Bind_Cancel;
		case C_InputType_PageUp: return ButtonIcon::C_Bind_PageUp;
		case C_InputType_PageDown: return ButtonIcon::C_Bind_PageDown;
		case C_InputType_AppMenu: return ButtonIcon::C_Bind_AppMenu;
		case C_InputType_Action1: return ButtonIcon::C_Bind_Action1;
		case C_InputType_Action2: return ButtonIcon::C_Bind_Action2;
		case C_InputType_Action3: return ButtonIcon::C_Bind_Action3;
		case C_InputType_Action4: return ButtonIcon::C_Bind_Action4;
		case C_InputType_ScrollUp: return ButtonIcon::C_Bind_ScrollUp;
		case C_InputType_ScrollDown: return ButtonIcon::C_Bind_ScrollDown;
	}
	return "";
}

/// Redraw the component
Void Private_Hook_ComponentDraw(CMlScript _This, CMlFrame _Frame_TabsFlat, Text[] _Tags) {
	declare K_Private_Instance CMGame_ComponentsTabsFlat_Instance as Instance for _Frame_TabsFlat = C_Private_Instance_Null;

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Size, C_Private_RedrawTag_Tabs, C_Private_RedrawTag_Inputs])) {
		declare Real AllButtonsWidth = 0.;
		if (Instance.Properties.ButtonWidth > 0.) {
			if (Instance.Properties.InputPreviousTab != C_InputType_Null) AllButtonsWidth += Instance.Properties.ButtonWidth;
			if (Instance.Properties.InputNextTab != C_InputType_Null) AllButtonsWidth += Instance.Properties.ButtonWidth;
		}

		declare Real AllGapsWidth = ML::Max(0., (Instance.Tabs.count - 1) * Instance.Properties.TabGap);
		declare Real AllTabsWidth = 0.;
		declare Real[] TabsWidth;
		if (Instance.Properties.TabWidth == C_TabWidth_Same) {
			AllTabsWidth = ML::Max(0., Instance.Properties.Size.X - Instance.Properties.Padding.Left - Instance.Properties.Padding.Right - AllButtonsWidth - AllGapsWidth);
			declare Real TabWidth = 0.;
			if (Instance.Tabs.count > 0) TabWidth = AllTabsWidth / Instance.Tabs.count;
			for (I, 1, Instance.Tabs.count) {
				TabsWidth.add(TabWidth);
			}
		} else {
			for (Tab in Instance.Tabs) {
				declare Real Width = Instance.Controls.Label_TabWidth.ComputeWidth(Tab.Value, True) + (Instance.Properties.TabPadding * 2.);
				AllTabsWidth += Width;
				TabsWidth.add(Width);
			}
		}

		declare Vec2 Size = Instance.Properties.Size;
		if (Size.X == 0.) {
			Size.X = Instance.Properties.Padding.Left + AllButtonsWidth + AllTabsWidth + AllGapsWidth + Instance.Properties.Padding.Right;
		}

		Instance.DrawCache.Box = Box::New(Size, Instance.Properties.Margin, Instance.Properties.MarginRatio, Instance.Properties.Padding, Instance.Properties.PaddingRatio);

		Instance.Controls.Frame_Content.RelativePosition_V3 = Instance.DrawCache.Box.ContentArea.Pos;

		declare Real TabWidthRatio = 1.;
		declare Real ContentAreaSizeXWithoutButtons = ML::Max(0., Instance.DrawCache.Box.ContentArea.Size.X - AllButtonsWidth);
		if (
			ContentAreaSizeXWithoutButtons < AllTabsWidth + AllGapsWidth ||
			(Instance.Properties.TabWidth == C_TabWidth_TextFill && ContentAreaSizeXWithoutButtons > AllTabsWidth + AllGapsWidth)
		) {
			if (AllTabsWidth == 0.) {
				TabWidthRatio = 0.;
			} else {
				TabWidthRatio = ML::Max(0., ContentAreaSizeXWithoutButtons - AllGapsWidth) / AllTabsWidth;
			}
		}

		declare Real ShiftX = 0.;
		if (Instance.Properties.ButtonWidth > 0. && Instance.Properties.InputPreviousTab != C_InputType_Null) {
			ShiftX += Instance.Properties.ButtonWidth;
		}
		for (Key => ControlTab in Instance.Controls.Tabs) {
			if (TabsWidth.existskey(Key)) {
				declare Real TabWidth = TabsWidth[Key] * TabWidthRatio;
				declare Real LabelWidth = ML::Max(0., TabWidth - (Instance.Properties.TabPadding * 2.));
				ControlTab.Frame_Tab.RelativePosition_V3.X = ShiftX;
				ControlTab.Quad_TabBackground.Size = <TabWidth, Instance.DrawCache.Box.ContentArea.Size.Y>;
				if (LabelWidth > 0.) {
					ControlTab.Label_TabValue.Size = <LabelWidth, Instance.DrawCache.Box.ContentArea.Size.Y>;
					ControlTab.Label_TabValue.RelativePosition_V3 = <ControlTab.Quad_TabBackground.Size.X * 0.5, ControlTab.Quad_TabBackground.Size.Y * -0.5>;
					ControlTab.Label_TabValue.Visible = True;
				} else {
					ControlTab.Label_TabValue.Visible = False;
				}
				ShiftX += TabWidth + Instance.Properties.TabGap;
			}
		}

		if (Instance.Properties.ButtonWidth > 0.) {
			if (Instance.Properties.InputPreviousTab != C_InputType_Null) {
				declare Real Side = ML::Min(Instance.Properties.ButtonWidth, Instance.DrawCache.Box.ContentArea.Size.Y);
				ButtonIcon::SetSize(_This, Instance.Controls.ButtonIcon_PreviousTab, <Side, Side>);
				Instance.Controls.ButtonIcon_PreviousTab.RelativePosition_V3 = <Instance.Properties.ButtonWidth * 0.5, -Instance.DrawCache.Box.ContentArea.Size.Y * 0.5>;
			}
			if (Instance.Properties.InputNextTab != C_InputType_Null) {
				declare Real Side = ML::Min(Instance.Properties.ButtonWidth, Instance.DrawCache.Box.ContentArea.Size.Y);
				ButtonIcon::SetSize(_This, Instance.Controls.ButtonIcon_NextTab, <Side, Side>);
				Instance.Controls.ButtonIcon_NextTab.RelativePosition_V3 = <ShiftX - Instance.Properties.TabGap + (Instance.Properties.ButtonWidth * 0.5), -Instance.DrawCache.Box.ContentArea.Size.Y * 0.5>;
			}
		}
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Size, C_Private_RedrawTag_Align])) {
		MLHelpers::AlignContent(Instance.Controls.Frame_Align, Instance.DrawCache.Box.MarginArea.Size, Instance.Properties.HAlign, Instance.Properties.VAlign);
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_IsVisible])) {
		Instance.Controls.Frame_TabsFlat.Visible = Instance.Properties.IsVisible;
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Tabs, C_Private_RedrawTag_TabsFocus])) {
		for (TabKey => ControlTab in Instance.Controls.Tabs) {
			if (Instance.Tabs.existskey(TabKey)) {
				ControlTab.Label_TabValue.Value = Instance.Tabs[TabKey].Value;
				if (Instance.FocusedTabKeys.exists(TabKey)) {
					ControlTab.Label_TabValue.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Secondary);
					ControlTab.Quad_TabBackground.BgColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_BG_Secondary);
				} else if (TabKey == Instance.SelectedTabKey) {
					ControlTab.Label_TabValue.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_PrimaryEmphasis);
					ControlTab.Quad_TabBackground.BgColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_BG_PrimarySubtle);
				} else {
					ControlTab.Label_TabValue.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Primary);
					ControlTab.Quad_TabBackground.BgColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_BG_Primary);
				}
				ControlTab.Frame_Tab.Visible = True;
			} else {
				ControlTab.Frame_Tab.Visible = False;
			}
		}
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Inputs])) {
		if (Instance.Properties.ButtonWidth > 0) {
			if (Instance.Properties.InputPreviousTab == C_InputType_Null) {
				Instance.Controls.ButtonIcon_PreviousTab.Visible = False;
			} else {
				ButtonIcon::SetBind(_This, Instance.Controls.ButtonIcon_PreviousTab, Private_NavigationInputTypeToButtonIconBind(Instance.Properties.InputPreviousTab));
				Instance.Controls.ButtonIcon_PreviousTab.Visible = True;
			}
			if (Instance.Properties.InputNextTab == C_InputType_Null) {
				Instance.Controls.ButtonIcon_NextTab.Visible = False;
			} else {
				ButtonIcon::SetBind(_This, Instance.Controls.ButtonIcon_NextTab, Private_NavigationInputTypeToButtonIconBind(Instance.Properties.InputNextTab));
				Instance.Controls.ButtonIcon_NextTab.Visible = True;
			}
		} else {
			Instance.Controls.ButtonIcon_PreviousTab.Visible = False;
			Instance.Controls.ButtonIcon_NextTab.Visible = False;
		}
	}
}

	Instance.Properties.Margin = Box::C_Margin_Null;

/// Update the component
Void Private_Hook_ComponentAfterYield(CMlScript _This, CMlFrame _Frame_TabsFlat) {
	declare K_Private_Instance CMGame_ComponentsTabsFlat_Instance as Instance for _Frame_TabsFlat = C_Private_Instance_Null;

	if (
		Instance.Properties.IsNavigationEnabled &&
		(Instance.Properties.InputPreviousTab != C_InputType_Null || Instance.Properties.InputNextTab != C_InputType_Null)
	) {
		for (Input in NavigationInputs::GetInputList(_This)) {
			if (Input.Type == Instance.Properties.InputPreviousTab) {
				Instance = Private_SelectPreviousTab(Instance);
			} else if (Input.Type == Instance.Properties.InputNextTab) {
				Instance = Private_SelectNextTab(Instance);
			}
		}
	}
}
