/// Set the opacity of a control
Void Private_SetOpacity(CMlScript _This, CMlControl _Control, Real _Opacity, Boolean _IsAnimated) {
	_This.AnimMgr.Flush(_Control);
	if (_IsAnimated) {
		_This.AnimMgr.Add(_Control, "<anim opacity=\""^_Opacity^"\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	} else {
		switchtype (_Control) {
			case CMlQuad: (_Control as CMlQuad).Opacity = _Opacity;
			case CMlLabel: (_Control as CMlLabel).Opacity = _Opacity;
			default: assert(False, "Invalid control class");
		}
	}
}

/// Read the text of the button if the TTS feature is enabled
Void TTS_Focus(CMlFrame _Frame_Button, Boolean _IsFocused) {
	declare K_Button CMGame_Button2_Button for _Frame_Button = C_Button_Null;
	if (!CMGame_Button2_Button.IsButton || CMGame_Button2_Button.IsLocked) return;
	if (_IsFocused) CMGame_Button2_Button.Controls.Label_Value.TTS_Focus();
	else CMGame_Button2_Button.Controls.Label_Value.TTS_Unfocus();
}
Void TTS_Focus(CMlFrame _Frame_Button) {
	TTS_Focus(_Frame_Button, True);
}
Void TTS_Unfocus(CMlFrame _Frame_Button) {
	TTS_Focus(_Frame_Button, False);
}

/// Focus the button
Void Private_Focus(CMlScript _This, CMlFrame _Frame_Button, Boolean _IsAnimated) {
	declare K_State CMGame_Button2_State for _This.Page = C_State_Null;
	declare K_Button CMGame_Button2_Button for _Frame_Button = C_Button_Null;
	if (!CMGame_Button2_Button.IsButton || CMGame_Button2_Button.IsLocked) return;

	CMGame_Button2_Button.Controls.Quad_FocusIcon.Show();
	CMGame_Button2_Button.Controls.Quad_FocusBackground.Visible = ((CMGame_Button2_Button.Background.FocusColor != CMGame_Button2_State.DefaultColorFocus || CMGame_Button2_Button.Background.ImageFocus != "") && !CMGame_Button2_Button.IsBgEmpty);
	CMGame_Button2_Button.Controls.Quad_Image.Visible = (!CMGame_Button2_Button.Controls.Quad_FocusBackground.Visible || CMGame_Button2_Button.Background.ImageFocus != CMGame_Button2_Button.Background.Image) && !CMGame_Button2_Button.IsBgEmpty;
	CMGame_Button2_Button.Controls.Quad_Icon.Visible = !CMGame_Button2_Button.Controls.Quad_FocusIcon.Visible;

	if (CMGame_Button2_Button.IsBlink) {
		if (CMGame_Button2_Button.Controls.Quad_FocusBackground.Visible) {
			if (CMGame_Button2_State.QuadBlinking.existskey(_Frame_Button.Id)) {
				if (!CMGame_Button2_State.QuadBlinking[_Frame_Button.Id].exists(CMGame_Button2_Button.Controls.Quad_FocusBackground)) {
					CMGame_Button2_State.QuadBlinking[_Frame_Button.Id].add(CMGame_Button2_Button.Controls.Quad_FocusBackground);
				}
			} else {
				CMGame_Button2_State.QuadBlinking[_Frame_Button.Id] = [CMGame_Button2_Button.Controls.Quad_FocusBackground];
			}
		}
		if (CMGame_Button2_Button.Background.ContourImage != "") {
			if (CMGame_Button2_State.QuadBlinking.existskey(_Frame_Button.Id)) {
				if (!CMGame_Button2_State.QuadBlinking[_Frame_Button.Id].exists(CMGame_Button2_Button.Controls.Quad_Contour)) {
					CMGame_Button2_State.QuadBlinking[_Frame_Button.Id].add(CMGame_Button2_Button.Controls.Quad_Contour);
				}
			} else {
				CMGame_Button2_State.QuadBlinking[_Frame_Button.Id] = [CMGame_Button2_Button.Controls.Quad_Contour];
			}
		}
		CMGame_Button2_State.ResetBlink = True;
	} else {
		CMGame_Button2_State.QuadBlinking.removekey(_Frame_Button.Id);
	}

	Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Label_Value, 1., _IsAnimated);

	if (CMGame_Button2_Button.Controls.Quad_FocusBackground.Visible) {
		CMGame_Button2_Button.Controls.Label_Value.TextColor = Tools::GetVisibleColorOnBackground(CMGame_Button2_Button.Background.FocusColor, CMGame_Button2_Button.Label.FocusColor);
		if (CMGame_Button2_Button.Background.ImageFocusOpacity == -1.) {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusBackground, CMGame_Button2_Button.Background.Opacity, _IsAnimated);
			CMGame_Button2_Button.Controls.Quad_Image.Opacity = CMGame_Button2_Button.Background.Opacity;
		} else {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusBackground, CMGame_Button2_Button.Background.ImageFocusOpacity, _IsAnimated);
			CMGame_Button2_Button.Controls.Quad_Image.Opacity = CMGame_Button2_Button.Background.ImageFocusOpacity;
		}
	} else {
		CMGame_Button2_Button.Controls.Label_Value.TextColor = Tools::GetVisibleColorOnBackground(CMGame_Button2_Button.Background.Color, CMGame_Button2_Button.Label.FocusColor);
		if (CMGame_Button2_Button.Background.ImageFocusOpacity == -1.) {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Image, CMGame_Button2_Button.Background.Opacity, _IsAnimated);
		} else {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Image, CMGame_Button2_Button.Background.ImageFocusOpacity, _IsAnimated);
		}
	}

	if (CMGame_Button2_Button.Controls.Quad_FocusIcon.Visible) {
		if (CMGame_Button2_Button.Icon.ImageFocusOpacity == -1.) {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusIcon, CMGame_Button2_Button.Icon.Opacity, _IsAnimated);
			CMGame_Button2_Button.Controls.Quad_Icon.Opacity = CMGame_Button2_Button.Icon.Opacity;
		} else {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusIcon, CMGame_Button2_Button.Icon.ImageFocusOpacity, _IsAnimated);
			CMGame_Button2_Button.Controls.Quad_Icon.Opacity = CMGame_Button2_Button.Icon.ImageFocusOpacity;
		}
	} else if (CMGame_Button2_Button.Icon.ImageFocusOpacity != -1.) {
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Icon, CMGame_Button2_Button.Icon.ImageFocusOpacity, _IsAnimated);
	} else {
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Icon, CMGame_Button2_Button.Icon.Opacity, _IsAnimated);
	}
	if (CMGame_Button2_Button.Icon.ImageFocusOpacity == -1.) {
		if (_IsAnimated) {
			LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.Icon.Opacity, 250);
		} else {
			LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.Icon.Opacity, 0);
		}
	} else if (_IsAnimated) {
		LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.Icon.ImageFocusOpacity, 250);
	} else {
		LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.Icon.ImageFocusOpacity, 0);
	}

	if (CMGame_Button2_Button.Controls.Quad_Gradient.Visible) {
		if (CMGame_Button2_Button.Background.GradientOpacityFocus == -1.) {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Gradient, CMGame_Button2_Button.Background.Opacity, _IsAnimated);
		} else {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Gradient, CMGame_Button2_Button.Background.GradientOpacityFocus, _IsAnimated);
		}
	}

	if (CMGame_Button2_Button.Controls.Quad_Contour.Visible) {
		if (CMGame_Button2_Button.Background.ContourOpacityFocus == -1.) {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Contour, CMGame_Button2_Button.Background.Opacity, _IsAnimated);
		} else {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Contour, CMGame_Button2_Button.Background.ContourOpacityFocus, _IsAnimated);
		}
	}

	CMGame_Button2_Button.IsFocused = True;
}
Void Focus(CMlScript _This, CMlFrame _Frame_Button, Boolean _IsAnimated) {
	Private_Focus(_This, _Frame_Button, _IsAnimated);
	TTS_Focus(_Frame_Button);
}
Void Focus(CMlScript _This, CMlFrame _Frame_Button) {
	Focus(_This, _Frame_Button, True);
}

/// Unfocus the button
Void Private_Unfocus(CMlScript _This, CMlFrame _Frame_Button, Boolean _IsAnimated) {
	declare K_Button CMGame_Button2_Button for _Frame_Button = C_Button_Null;
	if (!CMGame_Button2_Button.IsButton || CMGame_Button2_Button.IsLocked) return;
	declare K_State CMGame_Button2_State for _This.Page = C_State_Null;

	CMGame_Button2_Button.Controls.Quad_Icon.Show();
	CMGame_Button2_Button.Controls.Quad_Image.Visible = !CMGame_Button2_Button.IsBgEmpty;
	CMGame_Button2_Button.Controls.Quad_FocusBackground.Hide();
	CMGame_Button2_Button.Controls.Quad_FocusIcon.Hide();
	CMGame_Button2_Button.Controls.Label_Value.TextColor = CMGame_Button2_Button.Label.Color;

	CMGame_Button2_State.QuadBlinking.removekey(_Frame_Button.Id);

	if (CMGame_Button2_Button.Icon.OpacityUnfocus != -1.) {
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Icon, CMGame_Button2_Button.Icon.OpacityUnfocus, _IsAnimated);
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusIcon, CMGame_Button2_Button.Icon.OpacityUnfocus, _IsAnimated);
		if (_IsAnimated) {
			LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.Icon.OpacityUnfocus, 250);
		} else {
			LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.Icon.OpacityUnfocus, 0);
		}
	} else if (CMGame_Button2_Button.OpacityUnfocus != -1.) {
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Icon, CMGame_Button2_Button.OpacityUnfocus, _IsAnimated);
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusIcon, CMGame_Button2_Button.OpacityUnfocus, _IsAnimated);
		if (_IsAnimated) {
			LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.OpacityUnfocus, 250);
		} else {
			LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.OpacityUnfocus, 0);
		}
	} else {
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Icon, CMGame_Button2_Button.Icon.Opacity, _IsAnimated);
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusIcon, CMGame_Button2_Button.Icon.Opacity, _IsAnimated);
		if (_IsAnimated) {
			LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.Icon.Opacity, 250);
		} else {
			LoadingSpinner::Fade(_This, CMGame_Button2_Button.Controls.Spinner_Loading, CMGame_Button2_Button.Icon.Opacity, 0);
		}
	}

	if (CMGame_Button2_Button.Background.OpacityUnfocus != -1.) {
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Image, CMGame_Button2_Button.Background.OpacityUnfocus, _IsAnimated);
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusBackground, CMGame_Button2_Button.Background.OpacityUnfocus, _IsAnimated);
	} else if (CMGame_Button2_Button.OpacityUnfocus != -1) {
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Image, CMGame_Button2_Button.OpacityUnfocus, _IsAnimated);
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusBackground, CMGame_Button2_Button.OpacityUnfocus, _IsAnimated);
	} else {
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Image, CMGame_Button2_Button.Background.Opacity, _IsAnimated);
		Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_FocusBackground, CMGame_Button2_Button.Background.Opacity, _IsAnimated);
	}

	if (CMGame_Button2_Button.Controls.Quad_Gradient.Visible) {
		if (CMGame_Button2_Button.Background.GradientOpacityUnfocus != -1.) {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Gradient, CMGame_Button2_Button.Background.GradientOpacityUnfocus, _IsAnimated);
		} else if (CMGame_Button2_Button.Background.OpacityUnfocus != -1.) {
			Private_SetOpacity(_This, CMGame_Button2_Button.Controls.Quad_Gradient, CMGame_Button2_Button.Background.OpacityUnfocus, _IsAnimated);
		} else {

/// Update the size of the button
Vec2 Private_ApplyProperties_Size(K_Button _Properties) {
	declare Vec2 SizeOfReference;
	if (_Properties.SizeHitbox == <-1., -1.>) {
		_Properties.Controls.Quad_Hitbox.Size = _Properties.Background.Size;
		SizeOfReference = _Properties.Background.Size;
	} else {
		_Properties.Controls.Quad_Hitbox.Size = _Properties.SizeHitbox;
		SizeOfReference = _Properties.SizeHitbox;
	}

	if (_Properties.TiltHitbox != 0.) {
		declare Real TiltRad = ML::DegToRad(_Properties.TiltHitbox);
		declare Real TiltWidthAtBase = SizeOfReference.Y * ML::Tan(TiltRad);
		declare Real CosTiltRad = ML::Cos(TiltRad);
		declare Real SinTiltRad = ML::Sin(TiltRad);
		_Properties.Controls.Quad_Hitbox.Size.X = (SizeOfReference.X - TiltWidthAtBase) * CosTiltRad;
		if (CosTiltRad == 0.) {
			_Properties.Controls.Quad_Hitbox.Size.Y = 0.;
		} else {
			_Properties.Controls.Quad_Hitbox.Size.Y = (SizeOfReference.Y / CosTiltRad) + ML::Abs((SizeOfReference.X - TiltWidthAtBase) * SinTiltRad);
		}
		if (SinTiltRad >= 0.) {
			_Properties.Controls.Quad_Hitbox.VerticalAlign = CMlControl::AlignVertical::Bottom;
			_Properties.Controls.Quad_Hitbox.RelativePosition_V3 = <0., -SizeOfReference.Y>;
		} else {
			_Properties.Controls.Quad_Hitbox.VerticalAlign = CMlControl::AlignVertical::Top;
			_Properties.Controls.Quad_Hitbox.RelativePosition_V3 = <0., 0.>;
		}
		_Properties.Controls.Quad_Hitbox.RelativeRotation = _Properties.TiltHitbox;
		_Properties.Controls.Frame_HitboxClip.Size = SizeOfReference;
	} else {
		_Properties.Controls.Quad_Hitbox.VerticalAlign = CMlControl::AlignVertical::Top;

		_Properties.Controls.Quad_FocusIcon.Size = <ML::Max(0., SizeOfReference.X - 5.), ML::Max(0., SizeOfReference.Y - 5.)>;
	} else {
		_Properties.Controls.Quad_Icon.Size = _Properties.Icon.Size;
		_Properties.Controls.Quad_FocusIcon.Size = _Properties.Icon.Size;
	}

	_Properties.Controls.Spinner_Loading.RelativeScale = _Properties.Loading.Scale;

	if (_Properties.Label.Size == <-1., -1.>) {
		_Properties.Controls.Label_Value.Size = <ML::Max(0., SizeOfReference.X - 5.), ML::Max(0., SizeOfReference.Y - 1.)>;
	} else {
		_Properties.Controls.Label_Value.Size = _Properties.Label.Size;
	}

	_Properties.Controls.Label_Value.RelativePosition_V3 = <SizeOfReference.X * _Properties.Label.PosX, SizeOfReference.Y * _Properties.Label.PosY>;
	_Properties.Controls.Quad_Icon.RelativePosition_V3 = <SizeOfReference.X * _Properties.Icon.PosX - _Properties.Controls.Quad_Icon.Size.X/2., SizeOfReference.Y * _Properties.Icon.PosY + _Properties.Controls.Quad_Icon.Size.Y/2.>;

	_Properties.Controls.Quad_Image.RelativePosition_V3 = <SizeOfReference.X * 0.5 - _Properties.Controls.Quad_Image.Size.X/2. + _Properties.Background.PosX, SizeOfReference.Y * -0.5 + _Properties.Controls.Quad_Image.Size.Y/2. + _Properties.Background.PosY>;

	Button.Loading.Scale = MV::GetProp_Real(_Frame_Button, C_Prop_LoadingScale, Button.Loading.Scale);
	if (MV::HasProp(_Frame_Button, C_Prop_LoadingPos)) {
		declare Vec2 Pos = MV::GetProp_Vec2(_Frame_Button, C_Prop_LoadingPos, <0., 0.>);

	}

/// Initialization hook

/// Redraw all buttons
Void RedrawButtons(CMlScript _This) {
	_This.Page.GetClassChildren(C_Class_FrameButtonContainer, _This.Page.MainFrame, True);
	foreach (Control in _This.Page.GetClassChildren_Result) {
		declare CMlFrame Frame_ButtonContainer = (Control as CMlFrame);
		Private_Hook_ComponentInit(_This, Frame_ButtonContainer.Parent);
	}
}

/// Yield hook
Void Private_Hook_PageAfterYield(CMlScript _This) {
	declare K_State CMGame_Button2_State for _This.Page = C_State_Null;
	if (CMGame_Button2_State.QuadBlinking.count > 0 && (Now >= CMGame_Button2_State.BlinkEndTime || CMGame_Button2_State.ResetBlink)) {
		foreach (Quads in CMGame_Button2_State.QuadBlinking) {
			foreach (Quad in Quads) {
				_This.AnimMgr.Flush(Quad);
				_This.AnimMgr.Add(Quad, "<anim opacity=\"0.3\" />", 2*C_AnimTime_Blink/5, CAnimManager::EAnimManagerEasing::QuadOut);
				_This.AnimMgr.AddChain(Quad, "<anim opacity=\"0.85\" />", 3*C_AnimTime_Blink/5, CAnimManager::EAnimManagerEasing::QuadOut);
			}
		}
		CMGame_Button2_State.BlinkEndTime = Now + C_AnimTime_Blink;
		CMGame_Button2_State.ResetBlink = False;
	}
	if (CMGame_Button2_State.DelayedStates.count > 0) {
		declare K_DelayedState[Ident] DelayedStates = CMGame_Button2_State.DelayedStates;
		foreach (ButtonId => DelayedState in DelayedStates) {
			if (DelayedState.ActivationTime > 0 && Now >= DelayedState.ActivationTime) {
				CMGame_Button2_State.DelayedStates.removekey(ButtonId);
				SetState(_This, DelayedState.Frame_Button, DelayedState.State);
			}
		}
	}

	// Force redraw component instances
	if (CMGame_Button2_State.RedrawRequested != RequestRedraw::GetComponentsRedrawCounter()) {
		CMGame_Button2_State.RedrawRequested = RequestRedraw::GetComponentsRedrawCounter();
		RedrawButtons(_This);
	}
}
