/**
 * Component: BannerAnimation@2
 */
#Const Version "2.0.1"
#Const ScriptName "Libs/Nadeo/CMGame/Components/BannerAnimation@2.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "MathLib" as ML
#Include "Libs/Nadeo/CMGame/Utils/Cast.Script.txt" as Cast
#Include "Libs/Nadeo/CMGame/Utils/ManiaView@3.Script.txt" as MV

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_Controls {
	CMlControl Control1;
	CMlControl Control2;
}

#Struct K_State {
	K_Controls Controls;
	Real InitialPosX;
	Integer AnimStartTime;
	Integer AnimDuration;
	Real ShiftX;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_ComponentName "CMGameBannerAnimation2"
#Const C_ComponentClass "cmgame-components-banner-animation-2"

#Const C_Class_Banner "class-banner-animation-2"

#Const C_AnimSpeedLow 4 // in pixel/sec
#Const C_AnimSpeedNormal 8 // in pixel/sec
#Const C_AnimSpeedHigh 12 // in pixel/sec

#Const C_State_Null K_State {
	Controls = K_Controls {},
	InitialPosX = 0.,
	AnimStartTime = -1,
	AnimDuration = 0,
	ShiftX = 0.
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Get the manialink of the frame instance component
 *
 *	@param	_ManiaLinkToInject				The manialink to animate (Careful do not add an id to your control, it will be duplicated in the component)
 *	@param	_Id												Id of the frame instance
 *	@param	_FrameSizeX								Size (in X) of the frame instance
 *	@param	_FrameSizeY								Size (in Y) of the frame instance
 *	@param	_FramePos						   		Pos (X and Y) of the frame instance
 *	@param	_VAlign										Vertical align of the frame instance
 *	@param	_HAlign										Horizontal align of the frame instance
 *
 *	@return														The manialink of the frame instance component
 */
Text InjectML(
	Text _ManiaLinkToInject,
	Text _Id,
	Vec2 _FrameSize,
	Vec2 _FramePos,
	CMlControl::AlignVertical _VAlign,
	CMlControl::AlignHorizontal _HAlign
) {
	declare Text ML =
	"""
	<frame id="frame-banner-{{{_Id}}}" size="{{{_FrameSize.X}}} {{{_FrameSize.Y}}}" pos="{{{_FramePos.X}}} {{{_FramePos.Y}}}" class="{{{C_Class_Banner}}}" halign="{{{Cast::ToText(_HAlign)}}}" valign="{{{Cast::ToText(_VAlign)}}}">
		{{{_ManiaLinkToInject}}}
		{{{_ManiaLinkToInject}}}
	</frame>
	""";
	return ML;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Private function to launch the animation
Void Private_LaunchAnim(CMlScript _This) {
	declare K_State CMGame_BannerAnimation2_State for _This.Page = C_State_Null;

	CMGame_BannerAnimation2_State.Controls.Control1.RelativePosition_V3.X = CMGame_BannerAnimation2_State.InitialPosX;
	CMGame_BannerAnimation2_State.Controls.Control2.RelativePosition_V3.X = CMGame_BannerAnimation2_State.InitialPosX + CMGame_BannerAnimation2_State.ShiftX;

	declare Real NewPosX = CMGame_BannerAnimation2_State.InitialPosX - CMGame_BannerAnimation2_State.ShiftX;
	_This.AnimMgr.Flush(CMGame_BannerAnimation2_State.Controls.Control1);
	_This.AnimMgr.Add(CMGame_BannerAnimation2_State.Controls.Control1, "<anim pos=\""^NewPosX^" "^CMGame_BannerAnimation2_State.Controls.Control1.RelativePosition_V3.Y^"\" />", CMGame_BannerAnimation2_State.AnimDuration, CAnimManager::EAnimManagerEasing::Linear);
	_This.AnimMgr.Flush(CMGame_BannerAnimation2_State.Controls.Control2);
	_This.AnimMgr.Add(CMGame_BannerAnimation2_State.Controls.Control2, "<anim pos=\""^CMGame_BannerAnimation2_State.InitialPosX^" "^CMGame_BannerAnimation2_State.Controls.Control2.RelativePosition_V3.Y^"\" />", CMGame_BannerAnimation2_State.AnimDuration, CAnimManager::EAnimManagerEasing::Linear);
	CMGame_BannerAnimation2_State.AnimStartTime = Now + CMGame_BannerAnimation2_State.AnimDuration;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Public function to set and launch the animation
Void SetAndLaunchAnimation(CMlScript _This, CMlControl _Control1, CMlControl _Control2, Real _InitialPosX, Real _ShiftX, Integer _Speed) {
	declare K_State CMGame_BannerAnimation2_State for _This.Page = C_State_Null;
	if (_Control1 != Null && _Control2 != Null) {
		CMGame_BannerAnimation2_State.Controls.Control1 = _Control1;
		CMGame_BannerAnimation2_State.Controls.Control2 = _Control2;
		CMGame_BannerAnimation2_State.InitialPosX = _InitialPosX;
		CMGame_BannerAnimation2_State.ShiftX = _ShiftX;
		CMGame_BannerAnimation2_State.AnimDuration = (ML::NearestInteger(CMGame_BannerAnimation2_State.ShiftX) / _Speed) * 1000;
		Private_LaunchAnim(_This);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Public function to stop the animation
Void StopAnimation(CMlScript _This) {
	declare K_State CMGame_BannerAnimation2_State for _This.Page = C_State_Null;
	CMGame_BannerAnimation2_State.AnimStartTime = -1;
	if (CMGame_BannerAnimation2_State.Controls.Control1 != Null) {
		_This.AnimMgr.Flush(CMGame_BannerAnimation2_State.Controls.Control1);
		CMGame_BannerAnimation2_State.Controls.Control1.RelativePosition_V3.X = CMGame_BannerAnimation2_State.InitialPosX;
	}
	if (CMGame_BannerAnimation2_State.Controls.Control2 != Null) {
		_This.AnimMgr.Flush(CMGame_BannerAnimation2_State.Controls.Control2);
		CMGame_BannerAnimation2_State.Controls.Control2.RelativePosition_V3.X = CMGame_BannerAnimation2_State.InitialPosX + CMGame_BannerAnimation2_State.ShiftX;
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialization hook
Void Private_Hook_PageMainInit(CMlScript _This) {
	declare K_State CMGame_BannerAnimation2_State for _This.Page = C_State_Null;
	CMGame_BannerAnimation2_State = C_State_Null;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Yield hook
Void Private_Hook_PageYieldAfter(CMlScript _This) {
	declare K_State CMGame_BannerAnimation2_State for _This.Page = C_State_Null;

	// Check if banner needs to be animated
	if (CMGame_BannerAnimation2_State.AnimStartTime > 0 && CMGame_BannerAnimation2_State.AnimStartTime <= Now) {
		Private_LaunchAnim(_This);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the component
MV::K_Component Component() {
	return MV::K_Component {
		Name = C_ComponentName,
		Path = ScriptName,
		Class = C_ComponentClass,
		Hooks = [
			MV::C_Hook_PageMainInit => """{{{C_ComponentName}}}::Private_Hook_PageMainInit""",
			MV::C_Hook_PageYieldAfter => """{{{C_ComponentName}}}::Private_Hook_PageYieldAfter"""
		]
	};
}
