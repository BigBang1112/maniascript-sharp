/**
 * Component: ClubName
 * =====================
 *
 * Confluence : https://confluence.ubisoft.com/display/Nadeo/ClubName
 *
 * Info
 * ----
 *
 * Display the name of a club with the verified icon
 *
 * Parameters
 * ----------
 *
 * - `data-size` (`Vec2`): size of the component bounding box
 * - `data-halign` (`Text`): horizontal alignment of the component. Can be `left`, `center` or `right`.
 * - `data-valign` (`Text`): vertical alignment of the component. Can be `top`, `center`, `center2` or `bottom`.
 * - `data-opacity` (`Real`): opacity of the component
 * - `data-textsize` (`Real`): textsize of the labels displaying the name and verified icon
 * - `data-textfont` (`Text`): font of the label displaying the name
 * - `data-textprefix` (`Text`): prefix of the label displaying the name
 * - `data-textcolor` (`Text`): color of the label displaying the name
 * - `data-textcompose` (`Text`): add a text to compose with the club name, works only with one compose for now ! Used to indicate track authors for example, e.g _("By %1") will display "By ClubName",
 */

}

/// Redraw the component (apply size, color...)
K_Result Private_Redraw(CMlScript _This, K_State _State, K_ClubName _ClubName) {
	declare K_Result Result = K_Result { State = _State, ClubName = _ClubName };
	declare Integer AnimDuration = C_Anim_Null;
	declare Boolean SizeUpdated = False;

	if (
		Result.ClubName.Dirty.exists(C_Property_Name) ||
		Result.ClubName.Dirty.exists(C_Property_Size) ||
		Result.ClubName.Dirty.exists(C_Property_TextSize) ||
		Result.ClubName.Dirty.exists(C_Property_TextFont) ||
		Result.ClubName.Dirty.exists(C_Property_TextPrefix) ||
		Result.ClubName.Dirty.exists(C_Property_TextCompose)
	) {
		SizeUpdated = True;

		// Check if there is something to display in the name part of the component
		declare Boolean HasNamePart = TL::Trim(Result.ClubName.ClubInfo.Name) != "" || TL::StripFormatting(TL::Trim(Result.ClubName.TextPrefix)) != "";

		// Display the verified icon
		Result.ClubName.Controls.Frame_VerifiedIcon.Visible = HasNamePart && _ClubName.ClubInfo.IsVerified;
		Result.ClubName.Controls.Quad_VerifiedIcon.Size = <Result.ClubName.Size.Y, Result.ClubName.Size.Y>;
		Result.ClubName.Controls.Quad_VerifiedIconShadow.Size = Result.ClubName.Controls.Quad_VerifiedIcon.Size;
		Result.ClubName.Controls.Quad_VerifiedIconShadow.RelativePosition_V3.Y = 0.5*Result.ClubName.Controls.Quad_VerifiedIcon.Size.Y;

		// Display the club name
		if (!HasNamePart) {
			Result.ClubName.Controls.Label_Name.Visible = False;
		} else {
			Result.ClubName.Controls.Label_Name.Visible = True;
			Result.ClubName.Controls.Label_Name.Value = Result.ClubName.ClubInfo.Name;
			Result.ClubName.Controls.Label_Name.TextSizeReal = Result.ClubName.TextSize;
			Result.ClubName.Controls.Label_Name.TextFont = Result.ClubName.TextFont;
			Result.ClubName.Controls.Label_Name.TextPrefix = Result.ClubName.TextPrefix;
			Result.ClubName.Controls.Label_Name.Size = Result.ClubName.Size;
			if (Result.ClubName.Controls.Frame_VerifiedIcon.Visible) {
				Result.ClubName.Controls.Label_Name.Size.X = ML::Max(0., Result.ClubName.Size.X - C_MarginBetweenNameAndLogo - Result.ClubName.Controls.Quad_VerifiedIcon.Size.X);
			}

			if (C_Debug_EnableLongName) {
				if (Result.ClubName.TextCompose == "") {
					Result.ClubName.Controls.Label_Name.Value = C_Debug_LongName_Name;
				} else {
					Result.ClubName.Controls.Label_Name.Value = TL::Compose(Result.ClubName.TextCompose, C_Debug_LongName_Name);
				}
			} else if (Result.ClubName.TextCompose != "") {
				Result.ClubName.Controls.Label_Name.Value = TL::Compose(Result.ClubName.TextCompose, Result.ClubName.ClubInfo.Name);
			} else {
				Result.ClubName.Controls.Label_Name.Value = Result.ClubName.ClubInfo.Name;
			}
			if (
				C_EnableRecommandedWidthWarning &&
				!Result.ClubName.RecommandedWidthWarningDisplayed &&
				Result.ClubName.Size.X < C_MinimumRecommandedWidth + C_MarginBetweenNameAndLogo + Result.ClubName.Controls.Quad_VerifiedIcon.Size.X
			) {
				Result.ClubName.RecommandedWidthWarningDisplayed = True;
				log("WARNING "^Result.ClubName.Controls.ClubName_Component.ControlId^Result.ClubName.Controls.ClubName_Component.Id^" is too narrow to display the longest club names. A width of at least "^C_MinimumRecommandedWidth + C_MarginBetweenNameAndLogo + Result.ClubName.Controls.Quad_VerifiedIcon.Size.X^" is recommanded (currently: "^Result.ClubName.Size.X^").");
			}

			Result.ClubName.Controls.Label_Name.RelativePosition_V3.Y = Result.ClubName.Size.Y * -0.5;

			declare Real StepSize = (Result.ClubName.TextSize - C_NameTextSizeMin) / Tools::C_FitLabelMaxSteps;
			Tools::FitLabelValue(Result.ClubName.Controls.Label_Name, Result.ClubName.TextSize, C_NameTextSizeMin, StepSize);
		}
		declare Boolean Translated = (Result.ClubName.TextCompose != ""); // label needs to be translated only if we have a text compose
		declare Real NameWidth = ML::Min(
			Result.ClubName.Controls.Label_Name.Size.X,
			Result.ClubName.Controls.Label_Name.ComputeWidth(Result.ClubName.Controls.Label_Name.Value, Translated)
		);

		// Align the verified icon at the end of the club name
		Result.ClubName.Controls.Frame_VerifiedIcon.RelativePosition_V3 = <NameWidth + C_MarginBetweenNameAndLogo, Result.ClubName.Size.Y * -0.5>;

		// Send a width update event
		if (Result.ClubName.Controls.Frame_VerifiedIcon.Visible) {
			Result.ClubName.Width = NameWidth + C_MarginBetweenNameAndLogo + Result.ClubName.Controls.Quad_VerifiedIcon.Size.X;
		} else {
			Result.ClubName.Width = NameWidth;
		}
		Result.State.DelayedPendingEvents.add(K_Event {
			Type = C_EventType_WidthUpdate,
			ClubName = Result.ClubName.Controls.ClubName_Component
		});
	}

	if (
		SizeUpdated ||
		Result.ClubName.Dirty.exists(C_Property_HAlign) ||
		Result.ClubName.Dirty.exists(C_Property_VAlign)
	) {
		Tools::AlignFrame(
			Result.ClubName.Controls.Frame_Align,
			<Result.ClubName.Width, Result.ClubName.Size.Y>,
			Result.ClubName.HAlign,
			Result.ClubName.VAlign
		);
	}

	if (
		Result.ClubName.Dirty.exists(C_Property_TextColor) ||
		Result.ClubName.Dirty.exists(C_Property_Name)
	) {
		Result.ClubName.Controls.Label_Name.TextColor = Result.ClubName.TextColor;
	}

	if (Result.ClubName.Dirty.exists(C_Property_Opacity)) {
		if (Result.ClubName.OpacityAnimDuration > 0) {
			AnimDuration = ML::Max(C_Anim_Instant, Result.ClubName.OpacityAnimDuration);
		} else {
			AnimDuration = C_Anim_Instant;
		}
	}

	if (AnimDuration == C_Anim_Instant) {
		_This.AnimMgr.Flush(Result.ClubName.Controls.Label_Name);
		Result.ClubName.Controls.Label_Name.Opacity = Result.ClubName.Opacity;
	} else if (AnimDuration > C_Anim_Instant) {
		_This.AnimMgr.Flush(Result.ClubName.Controls.Label_Name);
		_This.AnimMgr.Add(
			Result.ClubName.Controls.Label_Name,
			"""<a opacity="{{{Result.ClubName.Opacity}}}"/>""",
			AnimDuration,
			CAnimManager::EAnimManagerEasing::QuadOut
		);
	}

	Result.ClubName.Dirty = [];

	return Result;
}

/// Set the size of the component
K_ClubName Private_SetSize(K_ClubName _ClubName, Vec2 _Size) {
	declare Vec2 NewSize = <ML::Max(0., _Size.X), ML::Max(0., _Size.Y)>;
	if (NewSize == _ClubName.Size) return _ClubName;
	declare K_ClubName ClubName = _ClubName;
	ClubName.Size = NewSize;
	ClubName.RecommandedWidthWarningDisplayed = False;
	if (!ClubName.Dirty.exists(C_Property_Size)) {
		ClubName.Dirty.add(C_Property_Size);
	}
	return ClubName;
}

/// Initialize the component
Void Private_InitializeClubName(CMlScript _This, CMlFrame _Frame_ClubName) {
	assert(_Frame_ClubName.DataAttributeExists("size"), "The ClubName component '"^_Frame_ClubName.ControlId^_Frame_ClubName.Id^"' must have its 'data-size' attribute set");

	declare K_ClubName CMGame_ClubName_ClubName for _Frame_ClubName = C_ClubName_Null;
	CMGame_ClubName_ClubName = C_ClubName_Null;

	CMGame_ClubName_ClubName.Controls = K_Controls {
		ClubName_Component = _Frame_ClubName,
		Frame_Align = (_Frame_ClubName.GetFirstChild(C_Id_FrameAlign) as CMlFrame),
		Frame_VerifiedIcon = (_Frame_ClubName.GetFirstChild(C_Id_FrameVerifiedIcon) as CMlFrame),
		Label_Name = (_Frame_ClubName.GetFirstChild(C_Id_LabelName) as CMlLabel),
		Quad_VerifiedIcon = (_Frame_ClubName.GetFirstChild(C_Id_QuadVerifiedIcon) as CMlQuad),
		Quad_VerifiedIconShadow = (_Frame_ClubName.GetFirstChild(C_Id_QuadVerifiedIconShadow) as CMlQuad)
	};

	CMGame_ClubName_ClubName = Private_SetSize(CMGame_ClubName_ClubName, Tools::TextToVec2(_Frame_ClubName.DataAttributeGet("size")));
	if (_Frame_ClubName.DataAttributeExists("halign")) {
		CMGame_ClubName_ClubName = Private_SetHAlign(CMGame_ClubName_ClubName, Tools::TextToHAlign2(_Frame_ClubName.DataAttributeGet("halign")));
	}
	if (_Frame_ClubName.DataAttributeExists("valign")) {
		CMGame_ClubName_ClubName = Private_SetVAlign(CMGame_ClubName_ClubName, Tools::TextToVAlign2(_Frame_ClubName.DataAttributeGet("valign")));
	}
	if (_Frame_ClubName.DataAttributeExists("opacity")) {
		CMGame_ClubName_ClubName = Private_SetOpacity(CMGame_ClubName_ClubName, TL::ToReal(_Frame_ClubName.DataAttributeGet("opacity")), 0);
	}
	if (_Frame_ClubName.DataAttributeExists("textsize")) {
		CMGame_ClubName_ClubName = Private_SetTextSize(CMGame_ClubName_ClubName, TL::ToReal(_Frame_ClubName.DataAttributeGet("textsize")));
	}
	if (_Frame_ClubName.DataAttributeExists("textfont")) {
		CMGame_ClubName_ClubName = Private_SetTextFont(CMGame_ClubName_ClubName, _Frame_ClubName.DataAttributeGet("textfont"));
	}
	if (_Frame_ClubName.DataAttributeExists("textprefix")) {
		CMGame_ClubName_ClubName = Private_SetTextPrefix(CMGame_ClubName_ClubName, _Frame_ClubName.DataAttributeGet("textprefix"));
	}
	if (_Frame_ClubName.DataAttributeExists("textcolor")) {
		CMGame_ClubName_ClubName = Private_SetTextColor(CMGame_ClubName_ClubName, CL::HexToRgb(_Frame_ClubName.DataAttributeGet("textcolor")));
	}
	if (_Frame_ClubName.DataAttributeExists("textcompose")) {
		CMGame_ClubName_ClubName = Private_SetTextCompose(CMGame_ClubName_ClubName, _Frame_ClubName.DataAttributeGet("textcompose"));
	}

	CMGame_ClubName_ClubName.IsInitialized = True;

	declare K_State CMGame_ClubName_State for _This.Page = C_State_Null;
	declare K_Result Result = Private_Redraw(_This, CMGame_ClubName_State, CMGame_ClubName_ClubName);
	CMGame_ClubName_State = Result.State;
	CMGame_ClubName_ClubName = Result.ClubName;
}

/// Initialize the component script
Void Private_MainInit(CMlScript _This) {
	declare K_State CMGame_ClubName_State for _This.Page = C_State_Null;
	CMGame_ClubName_State = C_State_Null;
	_This.Page.GetClassChildren(C_Class_ClubName, _This.Page.MainFrame, True);
	foreach (Control in _This.Page.GetClassChildren_Result) {
		declare CMlFrame Frame_ClubName = (Control as CMlFrame).Parent;
		Private_InitializeClubName(_This, Frame_ClubName);
	}
}

/// Loop of the component
Void Private_MainLoop(CMlScript _This) {
	declare K_State CMGame_ClubName_State for _This.Page = C_State_Null;
	CMGame_ClubName_State.PendingEvents = CMGame_ClubName_State.DelayedPendingEvents;
	CMGame_ClubName_State.DelayedPendingEvents = [];
}
