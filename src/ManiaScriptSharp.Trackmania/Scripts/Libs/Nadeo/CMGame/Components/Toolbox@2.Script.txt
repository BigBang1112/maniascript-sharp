// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Enable or disable the navigation in the component
Void Private_UpdateNavigation(CMlScript _This, CMlFrame _Frame_NavigationContainer) {
	Navigation::EnableContainer(
		_This,
		_Frame_NavigationContainer,
		_Frame_NavigationContainer.Visible
	);
	switchtype (_This) {
		case CMlScriptIngame: {
			NavigationInputs::EnableNavigationInputs(_This, C_ComponentName, _Frame_NavigationContainer.Visible);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Open or close the component depending on its current visible state
Void Private_OpenClose(CMlScript _This) {
	declare K_State CMGame_Toolbox_State for _This.Page = C_State_Null;
	CMGame_Toolbox_State.Controls.Frame_NavigationContainer.Visible = !CMGame_Toolbox_State.Controls.Frame_NavigationContainer.Visible;
	Private_UpdateNavigation(_This, CMGame_Toolbox_State.Controls.Frame_NavigationContainer);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Handle navigtion focus
Void Private_Focus(CMlScript _This, CMlControl _From, CMlControl _To) {
	if (_From != Null) {
		Navigation::Unfocus(_This, _From);
		if (Button::IsButton(_From)) {
			Button::Unfocus(_This, _From as CMlFrame);
		} else if (Checkbox::IsCheckbox(_From)) {
			Checkbox::Unfocus(_This, _From as CMlFrame);
		} else if (TextField::IsTextField(_From)) {
			TextField::Unfocus(_This, _From as CMlFrame);
		}
	}

	if (_To != Null) {
		Navigation::Focus(_This, _To);
		if (Button::IsButton(_To)) {
			Button::Focus(_This, _To as CMlFrame);
		} else if (Checkbox::IsCheckbox(_To)) {
			Checkbox::Focus(_This, _To as CMlFrame);
		} else if (TextField::IsTextField(_To)) {
			TextField::Focus(_This, _To as CMlFrame);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Handle navigation selection
Void Private_Select(CMlScript _This, CMlControl _Control) {
	if (_Control == Null) return;
	declare K_State CMGame_Toolbox_State for _This.Page = C_State_Null;

	switch (_Control.ControlId) {
		case C_Id_ButtonClose: Private_OpenClose(_This);
		case C_Id_ButtonResetGuide: {
			TextField::SetTextValue(_This, CMGame_Toolbox_State.Controls.TextField_Guide, C_Path_Guide);
			TextField::SetTextValue(_This, CMGame_Toolbox_State.Controls.TextField_Opacity, ""^C_Opacity_Default);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Refresh all checkboxes and their corresponding tool
Void Private_RefreshTools(CMlScript _This) {
	if (_This.UserMgr == Null || _This.UserMgr.MainUser == Null) return;
	declare K_State CMGame_Toolbox_State for _This.Page = C_State_Null;
	declare persistent Boolean[Text] Persistent_Toolbox_ActivationStateList for _This.UserMgr.MainUser = [];
	declare persistent Real Persistent_Toolbox_GuideOpacity for _This.UserMgr.MainUser = C_Opacity_Default;
	declare persistent Text Persistent_Toolbox_GuideUrl for _This.UserMgr.MainUser = C_Path_Guide;

	TextField::SetTextValue(_This, CMGame_Toolbox_State.Controls.TextField_Guide, Persistent_Toolbox_GuideUrl);
	TextField::SetTextValue(_This, CMGame_Toolbox_State.Controls.TextField_Opacity, TL::FormatReal(Persistent_Toolbox_GuideOpacity, 2, False, False));
	foreach (ControlId => IsActivated in Persistent_Toolbox_ActivationStateList) {
		declare CMlFrame Checkbox = (CMGame_Toolbox_State.Controls.Frame_NavigationContainer.GetFirstChild(ControlId) as CMlFrame);
		if (Checkbox::IsCheckbox(Checkbox) && IsActivated != Checkbox::IsChecked(Checkbox)) {
			Checkbox::Check(_This, Checkbox, IsActivated);
		} else if (Checkbox == Null) {
			Persistent_Toolbox_ActivationStateList.removekey(ControlId);
		}
	}
}

	declare K_State CMGame_Toolbox_State for _This.Page = C_State_Null;
	Private_UpdateNavigation(_This, CMGame_Toolbox_State.Controls.Frame_NavigationContainer);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Loop of the component
Void Private_Hook_PageYieldAfter(CMlScript _This) {
	declare K_State CMGame_Toolbox_State for _This.Page = C_State_Null;

	// Manage navigation events
	for (Event in Navigation::GetPendingEvents(_This)) {
		switch (Event.Type) {
			case Navigation::C_EventType_Navigate: {
				if (Navigation::IsEventFromGroup(_This, Event, C_NavGroup_Default)) {
					if (Navigation::IsEventFromContainer(Event, CMGame_Toolbox_State.Controls.Frame_NavigationContainer)) {
						switch (Event.Input) {
							case Navigation::C_InputType_Select, Navigation::C_InputType_MouseClick: {
								Private_Select(_This, Event.To);
							}
							case Navigation::C_InputType_MouseOut, Navigation::C_InputType_MouseOver: {
								Private_Focus(_This, Event.From, Event.To);
							}
						}
					}
				}
			}
		}
	}

	// Refresh tools when switching context
	switchtype (_This as Context) {
		case CMlScriptIngame: {
			if (!CMGame_Toolbox_State.IsMainUserInitialized) {
				if (_This.UserMgr != Null && _This.UserMgr.MainUser != Null) {
					CMGame_Toolbox_State.IsMainUserInitialized = True;
					Private_RefreshTools(_This);
				}
			}
		}
		case CManiaAppTitleLayer: {
			if (CMGame_Toolbox_State.IsTitleControlReady != Context.TitleControl.IsReady) {
				CMGame_Toolbox_State.IsTitleControlReady = Context.TitleControl.IsReady;
				if (CMGame_Toolbox_State.IsTitleControlReady) {
					Private_RefreshTools(_This);
				}
			}
		}
	}

	// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
	// [MOUSE POS] Display the mouse position nearby the mouse cursor
	// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
	if (Checkbox::IsChecked(CMGame_Toolbox_State.Controls.Checkbox_MousePos)) {
		CMGame_Toolbox_State.Controls.Frame_MousePosDisplay.RelativePosition_V3 = <_This.MouseX, _This.MouseY> + <3., - 3.>;
		CMGame_Toolbox_State.Controls.Label_MousePos.SetText(""^TL::FormatReal(_This.MouseX, 2, False, False)^", "^TL::FormatReal(_This.MouseY, 2, False, False));
		CMGame_Toolbox_State.Controls.Frame_MouseRulerDisplay.Visible = _This.Input.MouseRightButton;
		if (_This.Input.MouseRightButton && CMGame_Toolbox_State.Pos_MouseRulerOrigin == <-1000., -1000.>) {
			CMGame_Toolbox_State.Pos_MouseRulerOrigin = <_This.MouseX, _This.MouseY>;
			CMGame_Toolbox_State.Controls.Quad_MouseRuler.RelativePosition_V3 = CMGame_Toolbox_State.Pos_MouseRulerOrigin;
		} else if (_This.Input.MouseRightButton) {
			CMGame_Toolbox_State.Controls.Quad_MouseRuler.Size.X = ML::Distance(<_This.MouseX, _This.MouseY>, CMGame_Toolbox_State.Pos_MouseRulerOrigin);
			CMGame_Toolbox_State.Controls.Quad_MouseRuler.RelativeRotation = -ML::RadToDeg(ML::OrientedAngle(<1., 0.>, <_This.MouseX, _This.MouseY> - <CMGame_Toolbox_State.Controls.Quad_MouseRuler.RelativePosition_V3.X, CMGame_Toolbox_State.Controls.Quad_MouseRuler.RelativePosition_V3.Y>));
			CMGame_Toolbox_State.Controls.Label_MouseRuler.SetText(""^TL::FormatReal(CMGame_Toolbox_State.Controls.Quad_MouseRuler.Size.X, 2, False, False));
			CMGame_Toolbox_State.Controls.Label_MouseRuler.Parent.RelativePosition_V3 = CMGame_Toolbox_State.Pos_MouseRulerOrigin + 0.5 * (<_This.MouseX, _This.MouseY> - <CMGame_Toolbox_State.Controls.Quad_MouseRuler.RelativePosition_V3.X, CMGame_Toolbox_State.Controls.Quad_MouseRuler.RelativePosition_V3.Y>);
			if (_This.Input.MouseRightButton && _This.Input.IsKeyPressed(54)) { //< ² (keycode 54)
				CMGame_Toolbox_State.Controls.Quad_MouseRuler.RelativeRotation = ML::NearestInteger(CMGame_Toolbox_State.Controls.Quad_MouseRuler.RelativeRotation / 10.) * 10.;
			}
		} else if (CMGame_Toolbox_State.Pos_MouseRulerOrigin != <-1000., -1000.>) {
			CMGame_Toolbox_State.Pos_MouseRulerOrigin = <-1000., -1000.>;
			CMGame_Toolbox_State.Controls.Quad_MouseRuler.Size.X = 0.;
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// An instance of the Checkbox component has been toggled
Void Private_OnCheckboxToggle(CMlScript _This, Checkbox::K_Event_Toggle _Event) {
	if (_Event.Component == Null) return;
	declare K_State CMGame_Toolbox_State for _This.Page = C_State_Null;

	switch (_Event.Component.ControlId) {
		case C_Id_CheckboxMousePos: CMGame_Toolbox_State.Controls.Frame_MousePosDisplay.Visible = _Event.IsChecked;
		case C_Id_CheckboxSafeArea: CMGame_Toolbox_State.Controls.Quad_SafeArea.Visible = _Event.IsChecked;
		case C_Id_CheckboxGuide: {
			declare Text ImageUrl = TextField::GetTextValue(CMGame_Toolbox_State.Controls.TextField_Guide);
			CMGame_Toolbox_State.Controls.Quad_Guide.Visible = _Event.IsChecked;
			if (_This.Http.IsValidUrl(ImageUrl)) CMGame_Toolbox_State.Controls.Quad_Guide.ChangeImageUrl(ImageUrl);
		}
	}

	// Save toolbox config
	if (_This.UserMgr != Null && _This.UserMgr.MainUser != Null) {
		declare persistent Boolean[Text] Persistent_Toolbox_ActivationStateList for _This.UserMgr.MainUser = [];
		Persistent_Toolbox_ActivationStateList[_Event.Component.ControlId] = _Event.IsChecked;
		_This.UserMgr.MainUser.PersistentSave();
	}
}

Void Private_OnTextFieldUpdate(CMlScript _This, TextField::K_Event_ValueUpdate _Event) {
	if (_Event.Component == Null) return;
	declare K_State CMGame_Toolbox_State for _This.Page = C_State_Null;

	switch (_Event.Component.ControlId) {
		case CMGame_Toolbox_State.Controls.TextField_Guide.ControlId: {
			if (_This.Http.IsValidUrl(_Event.Value) && _This.UserMgr != Null && _This.UserMgr.MainUser != Null) {
				declare persistent Text Persistent_Toolbox_GuideUrl for _This.UserMgr.MainUser = C_Path_Guide;
				CMGame_Toolbox_State.Controls.Quad_Guide.ChangeImageUrl(_Event.Value);
				Persistent_Toolbox_GuideUrl = _Event.Value;
				_This.UserMgr.MainUser.PersistentSave();
			}
		}
		case CMGame_Toolbox_State.Controls.TextField_Opacity.ControlId: {
			if (_This.UserMgr != Null && _This.UserMgr.MainUser != Null) {
				declare persistent Real Persistent_Toolbox_GuideOpacity for _This.UserMgr.MainUser = C_Opacity_Default;
				CMGame_Toolbox_State.Controls.Quad_Guide.Opacity = TL::ToReal(_Event.Value);
				Persistent_Toolbox_GuideOpacity = TL::ToReal(_Event.Value);
				_This.UserMgr.MainUser.PersistentSave();
			}
		}
	}
}
