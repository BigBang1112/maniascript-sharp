/**
 * Button@4 component
 * ==================
 *
 * Interactive element that the user can use to trigger an action
 *
 * Props
 * -----
 *
 * - `text-value` (`Text`): the text of the button
 * - `text-halign` (`Text`): horizontal alignment of the text displayed in the button
 *   Can be `left`, `center` or `right`
 * - `text-valign` (`Text`): vertical alignment of the text displayed in the button
 *   Can be `top`, `center`, `center2` or `bottom`
 * - `translate` (`Boolean`): enable or disable translation of the text
 * - `width-min` (`Real`): minimum width of the button
 *   Even if the button text is shorter, the button won't be narrower than this width.
 * - `width-max` (`Real`): maximum width of the button
 *   Even if the button text is longer, it will be reduced or cut so the button does not exceed this width.
 * - `width` (`Real`): forced width of the button
 *   This width will still be constrained by the minimum and maximum width.
 *   If the minimum width is 10, the maximum width 100, then if you set the width to 5 it will be clamped to 10.
 *   Same thing if you set the width to 200, it will be clamped to 100.
 * - `height` (`Real`): height of the button
 * - `margin` (`Margin`): margin around the component
 * - `margin-ratio` (`Margin`): margin for the component proportional to the component size
 * - `border` (`Border`): border around the component
 * - `border-ratio` (`Border`): border for the component proportional to the component size
 * - `padding` (`Padding`): padding inside the component
 * - `padding-ratio` (`Padding`): padding for the component proportional to the component size
 * - `halign` (`Text`): horizontal alignment of the component, can be `left`, `center` or `right`
 * - `valign` (`Text`): vertical alignment of the component, can be `top`, `center` or `bottom`
 * - `is-visible` (`Boolean`): the visibility of the component
 * - `text-default-style-list` (`Text`): space-separated list of styles to apply to the button text when it is in its default state
 * - `text-focus-style-list` (`Text`): space-separated list of styles to apply to the button text when it is in its focus state
 * - `background-default-style-list` (`Text`): space-separated list of styles to apply to the button background when it is in its default state
 * - `background-focus-style-list` (`Text`): space-separated list of styles to apply to the button background when it is in its focus state
 * - `foreground-default-style-list` (`Text`): space-separated list of styles to apply to the button foreground when it is in its default state
 * - `foreground-focus-style-list` (`Text`): space-separated list of styles to apply to the button foreground when it is in its focus state
 */

/// Convert an array of prop values to a properties structure
K_Private_Properties Private_ToProperties(Text[Text] _PropValueList) {
	declare K_Private_Properties Properties = C_Private_Properties_Default;

	Properties.Margin = Struct_Box::C_Margin_Null;
	Properties.MarginRatio = Struct_Box::C_Margin_Null;
	Properties.Border = Struct_Box::C_Border_Null;
	Properties.BorderRatio = Struct_Box::C_Border_Null;
	Properties.Padding = C_Private_Padding_Default;
	Properties.PaddingRatio = Struct_Box::C_Padding_Null;
	Properties.TextDefaultStyleList = [C_Private_Style_Label_Default];
	Properties.TextFocusStyleList = [C_Private_Style_Label_Focus];
	Properties.BackgroundDefaultStyleList = [C_Private_Style_Slice9_BackgroundDefault];
	Properties.BackgroundFocusStyleList = [C_Private_Style_Slice9_BackgroundFocus];
	Properties.ForegroundDefaultStyleList = [C_Private_Style_Slice9_ForegroundDefault];
	Properties.ForegroundFocusStyleList = [C_Private_Style_Slice9_ForegroundFocus];

	for (PropName => PropValue in _PropValueList) {
		switch (PropName) {
			case C_Prop_TextValue: Properties.TextValue = PropValue;
			case C_Prop_TextHAlign: Properties.TextHAlign = Cast::ToHAlign(PropValue);
			case C_Prop_TextVAlign: Properties.TextVAlign = Cast::ToVAlign(PropValue);
			case C_Prop_Translate: Properties.IsTranslated = Cast::ToBoolean(PropValue);
			case C_Prop_WidthMin: Properties.WidthMin = Cast::ToReal(PropValue);
			case C_Prop_WidthMax: Properties.WidthMax = Cast::ToReal(PropValue);
			case C_Prop_Width: Properties.Width = Cast::ToReal(PropValue);
			case C_Prop_Height: Properties.Height = Cast::ToReal(PropValue);
			case C_Prop_Margin: Properties.Margin = Struct_Box::ToMargin(PropValue);
			case C_Prop_MarginRatio: Properties.MarginRatio = Struct_Box::ToMargin(PropValue);
			case C_Prop_Border: Properties.Border = Struct_Box::ToBorder(PropValue);
			case C_Prop_BorderRatio: Properties.BorderRatio = Struct_Box::ToBorder(PropValue);
			case C_Prop_Padding: Properties.Padding = Struct_Box::ToPadding(PropValue);
			case C_Prop_PaddingRatio: Properties.PaddingRatio = Struct_Box::ToPadding(PropValue);
			case C_Prop_HAlign: Properties.HAlign = Cast::ToHAlign(PropValue);
			case C_Prop_VAlign: Properties.VAlign = Cast::ToVAlign(PropValue);
			case C_Prop_IsVisible: Properties.IsVisible = Cast::ToBoolean(PropValue);
			case C_Prop_TextDefaultStyleList: Properties.TextDefaultStyleList = TL::Split(C_Private_StyleListSeparator, PropValue);
			case C_Prop_TextFocusStyleList: Properties.TextFocusStyleList = TL::Split(C_Private_StyleListSeparator, PropValue);
			case C_Prop_BackgroundDefaultStyleList: Properties.BackgroundDefaultStyleList = TL::Split(C_Private_StyleListSeparator, PropValue);
			case C_Prop_BackgroundFocusStyleList: Properties.BackgroundFocusStyleList = TL::Split(C_Private_StyleListSeparator, PropValue);
			case C_Prop_ForegroundDefaultStyleList: Properties.ForegroundDefaultStyleList = TL::Split(C_Private_StyleListSeparator, PropValue);
			case C_Prop_ForegroundFocusStyleList: Properties.ForegroundFocusStyleList = TL::Split(C_Private_StyleListSeparator, PropValue);
		}
		if (!Properties.EnabledPropList.exists(PropName)) {
			Properties.EnabledPropList.add(PropName);
		}
	}

	return Properties;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the default properties
K_Private_Properties Private_GetDefaultProperties() {
	return Private_ToProperties([]);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Merge one style list into another
Text[] Private_MergeStyleList(Text[] _StyleListA, Text[] _StyleListB) {
	declare Text[] StyleList = _StyleListA;
	for (Style in _StyleListB) {
		StyleList.remove(Style); //< If the style is already in the list, remove it from the beginning of the list and add it at the end
		StyleList.add(Style);
	}
	return StyleList;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Merge one properties structure onto another
K_Private_Properties Private_MergeProperties(K_Private_Properties _PropertiesA, K_Private_Properties _PropertiesB) {
	declare K_Private_Properties Properties = _PropertiesA;

	for (PropName in _PropertiesB.EnabledPropList) {
		switch (PropName) {
			case C_Prop_TextValue: Properties.TextValue = _PropertiesB.TextValue;
			case C_Prop_TextHAlign: Properties.TextHAlign = _PropertiesB.TextHAlign;
			case C_Prop_TextVAlign: Properties.TextVAlign = _PropertiesB.TextVAlign;
			case C_Prop_Translate: Properties.IsTranslated = _PropertiesB.IsTranslated;
			case C_Prop_WidthMin: Properties.WidthMin = _PropertiesB.WidthMin;
			case C_Prop_WidthMax: Properties.WidthMax = _PropertiesB.WidthMax;
			case C_Prop_Width: Properties.Width = _PropertiesB.Width;
			case C_Prop_Height: Properties.Height = _PropertiesB.Height;
			case C_Prop_Margin: Properties.Margin = _PropertiesB.Margin;
			case C_Prop_MarginRatio: Properties.MarginRatio = _PropertiesB.MarginRatio;
			case C_Prop_Border: Properties.Border = _PropertiesB.Border;
			case C_Prop_BorderRatio: Properties.BorderRatio = _PropertiesB.BorderRatio;
			case C_Prop_Padding: Properties.Padding = _PropertiesB.Padding;
			case C_Prop_PaddingRatio: Properties.PaddingRatio = _PropertiesB.PaddingRatio;
			case C_Prop_HAlign: Properties.HAlign = _PropertiesB.HAlign;
			case C_Prop_VAlign: Properties.VAlign = _PropertiesB.VAlign;
			case C_Prop_IsVisible: Properties.IsVisible = _PropertiesB.IsVisible;
			case C_Prop_TextDefaultStyleList: Properties.TextDefaultStyleList = Private_MergeStyleList(
				Properties.TextDefaultStyleList,
				_PropertiesB.TextDefaultStyleList
			);
			case C_Prop_TextFocusStyleList: Properties.TextFocusStyleList = Private_MergeStyleList(
				Properties.TextFocusStyleList,
				_PropertiesB.TextFocusStyleList
			);
			case C_Prop_BackgroundDefaultStyleList: Properties.BackgroundDefaultStyleList = Private_MergeStyleList(
				Properties.BackgroundDefaultStyleList,
				_PropertiesB.BackgroundDefaultStyleList
			);
			case C_Prop_BackgroundFocusStyleList: Properties.BackgroundFocusStyleList = Private_MergeStyleList(
				Properties.BackgroundFocusStyleList,
				_PropertiesB.BackgroundFocusStyleList
			);
			case C_Prop_ForegroundDefaultStyleList: Properties.ForegroundDefaultStyleList = Private_MergeStyleList(
				Properties.ForegroundDefaultStyleList,
				_PropertiesB.ForegroundDefaultStyleList
			);
			case C_Prop_ForegroundFocusStyleList: Properties.ForegroundFocusStyleList = Private_MergeStyleList(
				Properties.ForegroundFocusStyleList,
				_PropertiesB.ForegroundFocusStyleList
			);
		}
		if (!Properties.EnabledPropList.exists(PropName)) {
			Properties.EnabledPropList.add(PropName);
		}
	}

	return Properties;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the instance is dirty
Boolean Private_IsDirty(K_Private_Instance _Instance) {
	return (_Instance.IsPropertiesDirty || _Instance.IsBoxDirty || _Instance.IsModifierListDirty);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the properties
K_Private_Instance Private_UpdateProperties(K_Private_Instance _Instance) {
	declare K_Private_Instance Instance = _Instance;

	Instance.IsPropertiesDirty = False;

	Instance.Properties = Private_GetDefaultProperties();
	Instance.Properties = Private_MergeProperties(Instance.Properties, Instance.StyleProperties);
	Instance.Properties = Private_MergeProperties(Instance.Properties, Instance.ControlProperties);

	return Instance;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the button modifiers
K_Private_Instance Private_UpdateModifierList(K_Private_Instance _Instance) {
	declare K_Private_Instance Instance = _Instance;

	Instance.IsModifierListDirty = False;

	if (_Instance.ModifierList.exists(C_Private_Modifier_Focus)) {
		Stylesheet::SetStyleList(_Instance.Controls.Label_TranslationEnabled, Instance.Properties.TextFocusStyleList);
		Stylesheet::SetStyleList(_Instance.Controls.Label_TranslationDisabled, Instance.Properties.TextFocusStyleList);
		Component_Slice9::SetStyleList(_Instance.Controls.Slice9_Background, Instance.Properties.BackgroundFocusStyleList);
		Component_Slice9::SetStyleList(_Instance.Controls.Slice9_Foreground, Instance.Properties.ForegroundFocusStyleList);
	} else {
		Stylesheet::SetStyleList(_Instance.Controls.Label_TranslationEnabled, Instance.Properties.TextDefaultStyleList);
		Stylesheet::SetStyleList(_Instance.Controls.Label_TranslationDisabled, Instance.Properties.TextDefaultStyleList);
		Component_Slice9::SetStyleList(_Instance.Controls.Slice9_Background, Instance.Properties.BackgroundDefaultStyleList);
		Component_Slice9::SetStyleList(_Instance.Controls.Slice9_Foreground, Instance.Properties.ForegroundDefaultStyleList);
	}

	return Instance;
}

/// Update the box
K_Private_Instance Private_UpdateBox(K_Private_Instance _Instance) {
	declare K_Private_Instance Instance = _Instance;

	Instance.IsBoxDirty = False;

	// Update text style and size before computing the width and height of the label
	Stylesheet::SetStyleList(Instance.Controls.Label_TmpTranslationEnabled, Stylesheet::GetStyleList(Instance.Controls.Label_TranslationEnabled));
	Stylesheet::SetStyleList(Instance.Controls.Label_TmpTranslationDisabled, Stylesheet::GetStyleList(Instance.Controls.Label_TranslationDisabled));
	Stylesheet::ApplyStyleList(Instance.Controls.Label_TmpTranslationEnabled);
	Stylesheet::ApplyStyleList(Instance.Controls.Label_TmpTranslationDisabled);
	Instance.Controls.Label_TmpTranslationEnabled.TextSizeReal = Stylesheet::GetTextSizeFitMax(Instance.Controls.Label_TmpTranslationEnabled);
	Instance.Controls.Label_TmpTranslationDisabled.TextSizeReal = Stylesheet::GetTextSizeFitMax(Instance.Controls.Label_TmpTranslationDisabled);
	Instance.Controls.Label_TmpTranslationEnabled.Visible = False;
	Instance.Controls.Label_TmpTranslationDisabled.Visible = False;

	declare Vec2 Size = <0., 0.>;

	if (Instance.Properties.Width >= 0.) {
		Size.X = Instance.Properties.Width;
	} else {
		declare Real LabelWidth = 0.;
		if (Instance.Properties.IsTranslated) {
			LabelWidth = MLHelpers::ComputeWidth(Instance.Controls.Label_TmpTranslationEnabled, Instance.Properties.TextValue);
		} else {
			LabelWidth = MLHelpers::ComputeWidth(Instance.Controls.Label_TmpTranslationDisabled, Instance.Properties.TextValue);
		}
		declare Real Ratio = 1. - (Instance.Properties.BorderRatio.Left + Instance.Properties.PaddingRatio.Left + Instance.Properties.PaddingRatio.Right + Instance.Properties.BorderRatio.Right);
		if (Ratio == 0.) {
			Size.X = 0.;
		} else {
			Size.X = (Instance.Properties.Border.Left + Instance.Properties.Padding.Left + LabelWidth + Instance.Properties.Padding.Right + Instance.Properties.Border.Right) / Ratio;
		}
	}

	if (Instance.Properties.WidthMax >= 0. && Size.X > Instance.Properties.WidthMax) {
		Size.X = Instance.Properties.WidthMax;
	}
	if (Instance.Properties.WidthMin >= 0. && Size.X < Instance.Properties.WidthMin) {
		Size.X = Instance.Properties.WidthMin;
	}

	if (Instance.Properties.Height >= 0.) {
		Size.Y = Instance.Properties.Height;
	} else {
		declare Real LabelHeight = 0.;
		if (Instance.Properties.IsTranslated) {
			LabelHeight = MLHelpers::ComputeHeight(Instance.Controls.Label_TmpTranslationEnabled, Instance.Properties.TextValue);
		} else {
			LabelHeight = MLHelpers::ComputeHeight(Instance.Controls.Label_TmpTranslationDisabled, Instance.Properties.TextValue);
		}
		declare Real Ratio = 1. - (Instance.Properties.BorderRatio.Top + Instance.Properties.PaddingRatio.Top + Instance.Properties.PaddingRatio.Bottom + Instance.Properties.BorderRatio.Bottom);
		if (Ratio == 0.) {
			Size.Y = 0.;
		} else {
			Size.Y = (Instance.Properties.Border.Top + Instance.Properties.Padding.Top + LabelHeight + Instance.Properties.Padding.Bottom + Instance.Properties.Border.Bottom) / Ratio;
		}
	}

	Instance.Box = Struct_Box::New(
		Size,
		Instance.Properties.Margin,
		Instance.Properties.MarginRatio,
		Instance.Properties.Border,
		Instance.Properties.BorderRatio,
		Instance.Properties.Padding,
		Instance.Properties.PaddingRatio
	);

	Component_Slice9::SetSize(Instance.Controls.Slice9_Background, Instance.Box.PaddingArea.Size);
	Component_Slice9::SetSize(Instance.Controls.Slice9_Foreground, Instance.Box.PaddingArea.Size);

	return Instance;
}

/// Update dirty values
K_Private_Instance Private_UpdateDirty(K_Private_Instance _Instance) {
	if (!Private_IsDirty(_Instance)) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	if (Instance.IsPropertiesDirty) {
		Instance = Private_UpdateProperties(Instance);
	}
	if (Instance.IsModifierListDirty) {
		Instance = Private_UpdateModifierList(Instance);
	}
	if (Instance.IsBoxDirty) {
		Instance = Private_UpdateBox(Instance);
	}

	return Instance;
}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Box, C_Private_RedrawTag_Align])) {
		MLHelpers::AlignContent(Instance.Controls.Frame_Align, Instance.Box.MarginArea.Size, Instance.Properties.HAlign, Instance.Properties.VAlign);
	}

		Instance.Controls.Frame_Button4.Visible = Instance.Properties.IsVisible;
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_TextValue])) {
		Instance.Controls.Label_TranslationEnabled.Value = Instance.Properties.TextValue;
		Instance.Controls.Label_TranslationDisabled.Value = Instance.Properties.TextValue;
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_TextValue, C_Private_RedrawTag_IsTranslated])) {
		Instance.Controls.Label_TranslationEnabled.Visible = Instance.Properties.IsTranslated && Instance.Properties.TextValue != "";

	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Box, C_Private_RedrawTag_TextValue, C_Private_RedrawTag_IsTranslated, C_Private_RedrawTag_ModifierList])) {

		}
		if (Instance.Controls.Label_TranslationDisabled.Visible) {
			Stylesheet::ApplyStyleList(Instance.Controls.Label_TranslationDisabled);
		}
	}
}
