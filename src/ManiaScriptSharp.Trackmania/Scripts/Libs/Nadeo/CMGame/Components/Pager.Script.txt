/**
 *	Pager component for ManiaView
 */
#Const Version		"1.0.0"
#Const ScriptName	"Libs/Nadeo/CMGame/Components/Pager.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Utils/ManiaView2.Script.txt" as MV

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constant
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Name "pager" //< Component name
#Const P "Pager_" //< Prefix use to differentiate functions/variables in the script

#Const C_ControlId_ButtonPageNext "Pager_button-page-next"
#Const C_ControlId_ButtonPagePrev "Pager_button-page-prev"
#Const C_ControlId_LabelPage "Pager_label-page"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Public
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Get the component
 *
 *	@return														The component id
 */
Text Component() {
	return MV::Component(
		C_Name,
		"""
<frame class="{{{P}}}container-pager">
	<quad pos="-10 -0.4" size="10 10" halign="right" valign="center" style="Icons64x64_1" substyle="ArrowDisabled" scriptevents="1" id="{{{C_ControlId_ButtonPagePrev}}}" />
	<label size="20 10" halign="center" valign="center" text="0/0" class="text-default" id="{{{C_ControlId_LabelPage}}}" />
	<quad pos="10 -0.4" size="10 10" valign="center" style="Icons64x64_1" substyle="ArrowDisabled" scriptevents="1" id="{{{C_ControlId_ButtonPageNext}}}" />
</frame>
		""",
		"""
#Const {{{P}}}C_EventType_ClickPageNext 0
#Const {{{P}}}C_EventType_ClickPagePrev 1

#Struct {{{P}}}K_Event {
	Integer Type;
	CMlFrame Pager;
}
		""",
		"""
declare {{{P}}}K_Event[] {{{P}}}PendingEvents;

Integer {{{P}}}Update(
	CMlFrame _Frame_Pager,
	Integer _ItemsPerPage,
	Integer _ItemsCount,
	Integer _NewPage
) {
	if (_ItemsPerPage <= 0) return 1;

	// Find page max
	declare PageMax = _ItemsCount / _ItemsPerPage;
	if (_ItemsCount % _ItemsPerPage != 0) PageMax += 1;
	declare CurrentPage = _NewPage;
	if (CurrentPage < 1) CurrentPage = 1;
	if (CurrentPage > PageMax) CurrentPage = PageMax;

	// Update pager UI
	if (_Frame_Pager != Null) {
		declare Label_Page <=> (_Frame_Pager.GetFirstChild("{{{C_ControlId_LabelPage}}}") as CMlLabel);
		declare Button_PagePrev <=> (_Frame_Pager.GetFirstChild("{{{C_ControlId_ButtonPagePrev}}}") as CMlQuad);
		declare Button_PageNext <=> (_Frame_Pager.GetFirstChild("{{{C_ControlId_ButtonPageNext}}}") as CMlQuad);
		Label_Page.Value = CurrentPage^"/"^PageMax;
		if (CurrentPage <= 1) {
			Button_PagePrev.Substyle = "ArrowDisabled";
		} else {
			Button_PagePrev.Substyle = "ArrowPrev";
		}
		if (CurrentPage >= PageMax) {
			Button_PageNext.Substyle = "ArrowDisabled";
		} else {
			Button_PageNext.Substyle = "ArrowNext";
		}
	}

	return CurrentPage;
}

***MainLoop***
***
{{{P}}}PendingEvents = [];

foreach (Event in PendingEvents) {
	if (Event.Type == CMlScriptEvent::Type::MouseClick) {
		if (Event.ControlId == "{{{C_ControlId_ButtonPageNext}}}") {
			{{{P}}}PendingEvents.add({{{P}}}K_Event {
				Type = {{{P}}}C_EventType_ClickPageNext,
				Pager = Event.Control.Parent.Parent
			});
		} else if (Event.ControlId == "{{{C_ControlId_ButtonPagePrev}}}") {
			{{{P}}}PendingEvents.add({{{P}}}K_Event {
				Type = {{{P}}}C_EventType_ClickPagePrev,
				Pager = Event.Control.Parent.Parent
			});
		}
	}
}
***
		""",
		[],
		[]
	);
}
