/**
 * Component: ProgressBar
 * ======================
 *
 * Parameters
 * ----------
 * - `data-size` (`Vec2`): size of the progress bar.
 * - `data-halign` (`Text`): horizontal alignment of the component.
 *      Can be `left`, `center` or `right`.
 * - `data-valign` (`Text`): vertical alignment of the component.
 *      Can be `top`, `center` or `bottom`.
 * - `data-levels-info` (`Text`): define leveling information.
 * - `data-levels-offset` (`Vec2`): add a two dimensionnal offset to the default position of the current and next levels frames.
 * - `data-levels-textcolor` (`Text`): text color of the current and next levels labels.
 * - `data-levels-textsize` (`Real`): text size of the current and next levels labels.
 * - `data-ratio-offset` (`Real`): add a two dimensionnal offset to the default position of the ratio label.
 * - `data-ratio-textcolor` (`Text`): text color of the ratio label.
 * - `data-ratio-textsize` (`Real`): text size of the ratio label.
 * - `data-styles` (`Text`): styles to apply to the component, new styles can be created with the `CreateStyle()` function.
 *      When several styles are applied to a component, the styles are merged together with the properties.
 *      of the latest style overriding the properties of the earlier ones.
 *      For example if "style-a" is `{ A = 1, B = 2, C = 3 }`, "style-b" is `{ A = 11 , C = 33}` and
 *      "style-c" is `{ A = 111 }` and we set the component style to `"style-a style-b style-c"`
 *      then the applied properties will be `{ A = 111, B = 2, C = 33 }`.
 */

/// Check if the given frame is an instance of the componnet
Void Private_AssertValidComponent(CMlFrame _Frame_ProgressBar) {
	assert(
		_Frame_ProgressBar != Null && _Frame_ProgressBar.Controls.count > 0 && _Frame_ProgressBar.Controls[0].HasClass(C_Class_ProgressBar),
		"The control '"^_Frame_ProgressBar.ControlId^_Frame_ProgressBar.Id^"' is not a valid ProgressBar component"
	);
}

/// Check if the given levels info are correct
Void Private_AssertValidLevelsInfo(K_Level[] _LevelsInfo) {
	foreach (Level => LevelInfo in _LevelsInfo) {
		if (Level < _LevelsInfo.count-1) {
			assert(
				LevelInfo.RequiredXP <= _LevelsInfo.get(Level+1, K_Level {}).RequiredXP,
				"The XP required to reach the level "^Level+2^" must be superior or equal to the XP required to reach the level "^Level+1^"."
			);
		}
	}
}

/// Enable or disable a property
Integer[] Private_EnableProperty(Integer[] _EnabledProperties, Integer _Property, Boolean _IsEnabled) {
	declare Integer[] EnabledProperties = _EnabledProperties;

	if (_IsEnabled) {
		if (!EnabledProperties.exists(_Property)) {
			EnabledProperties.add(_Property);
		}
	} else {
		EnabledProperties.remove(_Property);
	}

	return EnabledProperties;
}

/// Set and enable a property
K_Properties Private_SetProperty(K_Properties _TargetProperties, Integer _PropertyId, K_Properties _SourceProperties) {
	declare K_Properties Properties = _TargetProperties;

	Properties.Enabled = Private_EnableProperty(Properties.Enabled, _PropertyId, True);

	switch (_PropertyId) {
		case C_Property_Size: Properties.Size = _SourceProperties.Size;
		case C_Property_HAlign: Properties.HAlign = _SourceProperties.HAlign;
		case C_Property_VAlign: Properties.VAlign = _SourceProperties.VAlign;
		case C_Property_CurrentXP: Properties.CurrentXP = _SourceProperties.CurrentXP;
		case C_Property_LevelsInfo: Properties.LevelsInfo = _SourceProperties.LevelsInfo;
		case C_Property_LevelsOffset: Properties.LevelsOffset = _SourceProperties.LevelsOffset;
		case C_Property_LevelsTextColor: Properties.LevelsTextColor = _SourceProperties.LevelsTextColor;
		case C_Property_LevelsTextSize: Properties.LevelsTextSize = _SourceProperties.LevelsTextSize;
		case C_Property_RatioOffset: Properties.RatioOffset = _SourceProperties.RatioOffset;
		case C_Property_RatioTextColor: Properties.RatioTextColor = _SourceProperties.RatioTextColor;
		case C_Property_RatioTextSize: Properties.RatioTextSize = _SourceProperties.RatioTextSize;
	}

	return Properties;
}

/// Merge a base properties struct with other enabled properties
K_Properties Private_MergeProperties(K_Properties _Base, K_Properties[] _PropertiesToMerge) {
	declare K_Properties MergedProperties = _Base;

	foreach (Properties in _PropertiesToMerge) {
		foreach (PropertyId in Properties.Enabled) {
			MergedProperties = Private_SetProperty(MergedProperties, PropertyId, Properties);
		}
	}

	return MergedProperties;
}

/// Redraw the component (apply style, size, ...)
K_ProgressBar Private_RedrawProgressBar(CMlScript _This, K_ProgressBar _ProgressBar) {
	declare K_ProgressBar ProgressBar = _ProgressBar;
	declare K_State CMGame_ProgressBar_State for _This.Page = C_State_Null;

	// Merge properties
	declare K_Properties[] PropertiesToMerge = [CMGame_ProgressBar_State.DefaultProperties];
	foreach (Style in ProgressBar.Styles) {
		PropertiesToMerge.add(CMGame_ProgressBar_State.Styles.get(Style, K_Properties {}));
	}
	PropertiesToMerge.add(ProgressBar.CustomizedProperties);
	ProgressBar.MergedProperties = Private_MergeProperties(C_Properties_Default, PropertiesToMerge);

	// Update levels info
	declare Real LevelRatio = 1.;
	foreach (NextLevel => NextLevelInfo in ProgressBar.MergedProperties.LevelsInfo) {
		if (ProgressBar.MergedProperties.CurrentXP < NextLevelInfo.RequiredXP || NextLevel == ProgressBar.MergedProperties.LevelsInfo.count-1) {
			declare K_Level CurrentLevelInfo = ProgressBar.MergedProperties.LevelsInfo.get(Tools::Ternary(NextLevel > 0, NextLevel-1, ProgressBar.MergedProperties.LevelsInfo.count), K_Level {});
			ProgressBar.Controls.Label_CurrentLevel.SetText(""^NextLevel);
			ProgressBar.Controls.Label_NextLevel.SetText(""^NextLevel+1);
			ProgressBar.Controls.Label_CurrentLevel.Visible = CurrentLevelInfo.ImageUrl == "";
			ProgressBar.Controls.Label_NextLevel.Visible = NextLevelInfo.ImageUrl == "";
			ProgressBar.Controls.Quad_CurrentLevel.ChangeImageUrl(CurrentLevelInfo.ImageUrl);
			ProgressBar.Controls.Quad_NextLevel.ChangeImageUrl(NextLevelInfo.ImageUrl);
			ProgressBar.Controls.Label_RatioXP.SetText(TL::Compose("%1/%2", ""^ProgressBar.MergedProperties.CurrentXP, ""^NextLevelInfo.RequiredXP));
			if (NextLevelInfo.RequiredXP - CurrentLevelInfo.RequiredXP != 0) {
				LevelRatio = ML::Clamp(ML::ToReal(ProgressBar.MergedProperties.CurrentXP - CurrentLevelInfo.RequiredXP) / ML::ToReal(NextLevelInfo.RequiredXP - CurrentLevelInfo.RequiredXP), 0., 1.);
			}
			break;
		}
	}

	// Update positions
	ProgressBar.Controls.Frame_CurrentLevel.RelativePosition_V3 = <-0.5*ProgressBar.MergedProperties.Size.X - ProgressBar.MergedProperties.LevelsOffset.X, ProgressBar.MergedProperties.LevelsOffset.Y>;
	ProgressBar.Controls.Frame_NextLevel.RelativePosition_V3 = <0.5*ProgressBar.MergedProperties.Size.X + ProgressBar.MergedProperties.LevelsOffset.X, ProgressBar.MergedProperties.LevelsOffset.Y>;
	ProgressBar.Controls.Label_RatioXP.RelativePosition_V3 = ProgressBar.MergedProperties.RatioOffset;
	ProgressBar.Controls.Quad_ProgressBarXP.RelativePosition_V3 = <(LevelRatio-1)*ProgressBar.MergedProperties.Size.X, 0.>;

	// Update sizes
	ProgressBar.Controls.Label_RatioXP.Size = ProgressBar.MergedProperties.Size;
	ProgressBar.Controls.Frame_ProgressBarClip.Size = ProgressBar.MergedProperties.Size;
	ProgressBar.Controls.Quad_ProgressBarBackground.Size = ProgressBar.MergedProperties.Size;
	ProgressBar.Controls.Quad_ProgressBarXP.Size = ProgressBar.MergedProperties.Size;
	ProgressBar.Controls.Quad_NextLevel.Size = <ProgressBar.MergedProperties.Size.Y, ProgressBar.MergedProperties.Size.Y>;
	ProgressBar.Controls.Quad_CurrentLevel.Size = <ProgressBar.MergedProperties.Size.Y, ProgressBar.MergedProperties.Size.Y>;

	// Update other text properties
	ProgressBar.Controls.Label_CurrentLevel.TextColor = ProgressBar.MergedProperties.LevelsTextColor;
	ProgressBar.Controls.Label_CurrentLevel.TextSizeReal = ProgressBar.MergedProperties.LevelsTextSize;
	ProgressBar.Controls.Label_NextLevel.TextColor = ProgressBar.MergedProperties.LevelsTextColor;
	ProgressBar.Controls.Label_NextLevel.TextSizeReal = ProgressBar.MergedProperties.LevelsTextSize;
	ProgressBar.Controls.Label_RatioXP.TextColor = ProgressBar.MergedProperties.RatioTextColor;
	ProgressBar.Controls.Label_RatioXP.TextSizeReal = ProgressBar.MergedProperties.RatioTextSize;

	// Align the component
	Tools::Align(
		ProgressBar.Controls.Frame_Align,
		ProgressBar.MergedProperties.Size,
		ProgressBar.MergedProperties.HAlign,
		ProgressBar.MergedProperties.VAlign
	);

	return ProgressBar;
}

/// Customize a property
K_ProgressBar Private_SetCustomizedProperty(CMlScript _This, K_ProgressBar _ProgressBar, Integer _PropertyId, K_Properties _Properties, Boolean _IsRedrawRequired) {
	declare K_ProgressBar ProgressBar = _ProgressBar;
	ProgressBar.CustomizedProperties = Private_SetProperty(ProgressBar.CustomizedProperties, _PropertyId, _Properties);
	if (_IsRedrawRequired) ProgressBar = Private_RedrawProgressBar(_This, ProgressBar);
	return ProgressBar;
}

/// Reset a customized property
K_ProgressBar Private_ResetCustomizedProperty(CMlScript _This, K_ProgressBar _ProgressBar, Integer _PropertyId, Boolean _IsRedrawRequired) {
	declare K_ProgressBar ProgressBar = _ProgressBar;
	ProgressBar.CustomizedProperties.Enabled = Private_EnableProperty(ProgressBar.CustomizedProperties.Enabled, _PropertyId, False);
	if (_IsRedrawRequired) ProgressBar = Private_RedrawProgressBar(_This, ProgressBar);
	return ProgressBar;
}

/// Initialize the component
Void Private_InitializeProgressBar(CMlScript _This, CMlFrame _Frame_ProgressBar) {
	Private_AssertValidComponent(_Frame_ProgressBar);

	declare K_ProgressBar CMGame_ProgressBar for _Frame_ProgressBar = C_ProgressBar_Null;
	declare CMlFrame Frame_CurrentLevel = (_Frame_ProgressBar.GetFirstChild(C_Id_FrameCurrentLevel) as CMlFrame);
	declare CMlFrame Frame_NextLevel = (_Frame_ProgressBar.GetFirstChild(C_Id_FrameNextLevel) as CMlFrame);

	CMGame_ProgressBar = K_ProgressBar {
		Controls = K_Controls {
			Frame_Align = (_Frame_ProgressBar.GetFirstChild(C_Id_FrameAlign) as CMlFrame),
			Frame_CurrentLevel = Frame_CurrentLevel,
			Frame_NextLevel = Frame_NextLevel,
			Frame_ProgressBar = (_Frame_ProgressBar.GetFirstChild(C_Id_FrameProgressBar) as CMlFrame),
			Frame_ProgressBarClip = (_Frame_ProgressBar.GetFirstChild(C_Id_FrameProgressBarClip) as CMlFrame),
			Label_CurrentLevel = (Frame_CurrentLevel.GetFirstChild(C_Id_LabelLevel) as CMlLabel),
			Label_Level = (_Frame_ProgressBar.GetFirstChild(C_Id_LabelLevel) as CMlLabel),
			Label_NextLevel = (Frame_NextLevel.GetFirstChild(C_Id_LabelLevel) as CMlLabel),
			Label_RatioXP = (_Frame_ProgressBar.GetFirstChild(C_Id_LabelRatioXP) as CMlLabel),
			Quad_CurrentLevel = (Frame_CurrentLevel.GetFirstChild(C_Id_QuadLevel) as CMlQuad),
			Quad_Level = (_Frame_ProgressBar.GetFirstChild(C_Id_QuadLevel) as CMlQuad),
			Quad_NextLevel = (Frame_NextLevel.GetFirstChild(C_Id_QuadLevel) as CMlQuad),
			Quad_ProgressBarBackground = (_Frame_ProgressBar.GetFirstChild(C_Id_QuadProgressBarBackground) as CMlQuad),
			Quad_ProgressBarXP = (_Frame_ProgressBar.GetFirstChild(C_Id_QuadProgressBarXP) as CMlQuad)
		}
	};

	// Set custom properties from data attributes
	if (_Frame_ProgressBar.DataAttributeExists("size")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_Size, K_Properties { Size = Tools::TextToVec2(_Frame_ProgressBar.DataAttributeGet("size")) }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("halign")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_HAlign, K_Properties { HAlign = Tools::TextToHAlign2(_Frame_ProgressBar.DataAttributeGet("halign")) }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("valign")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_VAlign, K_Properties { VAlign = Tools::TextToVAlign2(_Frame_ProgressBar.DataAttributeGet("valign")) }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("current-xp")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_CurrentXP, K_Properties { CurrentXP = TL::ToInteger(_Frame_ProgressBar.DataAttributeGet("current-xp")) }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("levels-info")) {
		declare K_Level[] LevelsInfo;
		LevelsInfo.fromjson(_Frame_ProgressBar.DataAttributeGet("levels-info"));
		Private_AssertValidLevelsInfo(LevelsInfo);
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_LevelsInfo, K_Properties { LevelsInfo = LevelsInfo }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("levels-offset")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_LevelsOffset, K_Properties { LevelsOffset = Tools::TextToVec2(_Frame_ProgressBar.DataAttributeGet("levels-offset")) }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("levels-textcolor")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_LevelsTextColor, K_Properties { LevelsTextColor = CL::Hex6ToRgb(_Frame_ProgressBar.DataAttributeGet("levels-textcolor")) }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("levels-textsize")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_LevelsTextSize, K_Properties { LevelsTextSize = TL::ToReal(_Frame_ProgressBar.DataAttributeGet("levels-textsize")) }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("ratio-offset")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_RatioOffset, K_Properties { RatioOffset = Tools::TextToVec2(_Frame_ProgressBar.DataAttributeGet("ratio-offset")) }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("ratio-textcolor")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_RatioTextColor, K_Properties { RatioTextColor = CL::Hex6ToRgb(_Frame_ProgressBar.DataAttributeGet("ratio-textcolor")) }, False);
	}
	if (_Frame_ProgressBar.DataAttributeExists("ratio-textsize")) {
		CMGame_ProgressBar = Private_SetCustomizedProperty(_This, CMGame_ProgressBar, C_Property_RatioTextSize, K_Properties { RatioTextSize = TL::ToReal(_Frame_ProgressBar.DataAttributeGet("ratio-textsize")) }, False);
	}

	// Redraw the component
	CMGame_ProgressBar = Private_RedrawProgressBar(_This, CMGame_ProgressBar);
}

/// Initialize the component script
Void Private_MainInit(CMlScript _This) {
	_This.Page.GetClassChildren(C_Class_ProgressBar, _This.Page.MainFrame, True);
	foreach (Control in _This.Page.GetClassChildren_Result) {
		declare CMlFrame Frame_ProgressBar = (Control as CMlFrame).Parent;
		Private_InitializeProgressBar(_This, Frame_ProgressBar);
	}
}

/// Loop of the component
Void Private_MainLoop(CMlScript _This) {
	declare K_State CMGame_ProgressBar_State for _This.Page = C_State_Null;
	CMGame_ProgressBar_State.PendingEvents = [];
}

/// Check if the progress bar is visible or not
Boolean IsVisible(CMlFrame _Frame_ProgressBar) {
	Private_AssertValidComponent(_Frame_ProgressBar);
	declare K_ProgressBar CMGame_ProgressBar for _Frame_ProgressBar = C_ProgressBar_Null;
	return CMGame_ProgressBar.Controls.Frame_Align.Visible;
}

/// Show or hide the progress bar
Void SetVisibility(CMlScript _This, CMlFrame _Frame_ProgressBar, Boolean _IsVisible) {
	Private_AssertValidComponent(_Frame_ProgressBar);
	declare K_ProgressBar CMGame_ProgressBar for _Frame_ProgressBar = C_ProgressBar_Null;
	declare K_State CMGame_ProgressBar_State for _This.Page = C_State_Null;

	CMGame_ProgressBar.Controls.Frame_Align.Visible = _IsVisible;

	// Send visibility update event
	CMGame_ProgressBar_State.PendingEvents.add(K_Event {
		Type = C_EventType_VisibilityUpdate,
		IsVisible = CMGame_ProgressBar.Controls.Frame_Align.Visible
	});
}
Void Show(CMlScript _This, CMlFrame _Frame_ProgressBar) {
	SetVisibility(_This, _Frame_ProgressBar, True);
}
Void Hide(CMlScript _This, CMlFrame _Frame_ProgressBar) {
	SetVisibility(_This, _Frame_ProgressBar, False);
}

/// Set the default properties of all progress bars
Void SetDefaultProperties(CMlScript _This, K_Properties _DefaultProperties, Boolean _IsRedrawRequired) {
	declare K_State CMGame_ProgressBar_State for _This.Page = C_State_Null;
	CMGame_ProgressBar_State.DefaultProperties = _DefaultProperties;
	if (_IsRedrawRequired) {
		_This.Page.GetClassChildren(C_Class_ProgressBar, _This.Page.MainFrame, True);
		foreach (Control in _This.Page.GetClassChildren_Result) {
			declare CMlFrame Frame_ProgressBar = (Control as CMlFrame).Parent;
			declare K_ProgressBar CMGame_ProgressBar for Frame_ProgressBar = C_ProgressBar_Null;
			CMGame_ProgressBar = Private_RedrawProgressBar(_This, CMGame_ProgressBar);
		}
	}
}
Void SetDefaultProperties(CMlScript _This, K_Properties _DefaultProperties) {
	SetDefaultProperties(_This, _DefaultProperties, True);
}

	declare K_Properties[] PropertiesToMerge = [];
