/**
 * Component: TextField
 * ===========================
 *
 * The `TextField` component adds the ability to navigate to a `CMlEntry` or a `CMLTextedit` using the `Navigation` class.
 * With the `text-field-type` parameter, it is possible to initialize this component as a `CMlEntry` or a `CMlTextEdit`.
 *
 * Types of text-field
 * ----------
 * - C_TextFieldType_Entry         the type to define the component as a CMlEntry
 * - C_TextFieldType_TextEdit      the type to define the component as a CMlTextEdit
 *
 * Events
 * ----------
 * - C_Event_ValueUpdate           the event that is sent when the displayed text of the component is updated by the player
 *
 * Parameters
 * ----------
 * - `halign` (CMlControl::AlignHorizontal):	the horizontal alignment of the component.
 *      Manialink parameters : `left`, `center`, `right`
 * - `valign` (CMlControl::AlignVertical):	the vertical alignment of the comonent.
 *      Manialink parameters : `top`, `center`, `bottom`
 * - `size` (Vec2): the x and y size of the component
 * - `skew` (Real): the real skew to add a left and right margin depending on the background's skew
 * - `tooltip` (Text): the text to set the virtual keyboard tooltip
 * - `type` (Integer): the type of the text-field.
 *      Manialink parameters : `C_TextFieldType_Entry`, `C_TextFieldType_TextEdit`
 * - `button-styles` (Text): the styles to apply to the expendable button background
 * - `background-color-focused` (Vec3): the hexadecimal color of the background when the component is focused
 * - `background-color-unfocused` (Vec3): the hexadecimal color of the background when the component is not focused
 * - `background-empty-enabled` (Boolean): the boolean to set an empty background
 * - `background-opacity-focused` (Real): the real opacity of the background when the component is focused
 * - `background-opacity-unfocused` (Real): the real opacity of the background when the component is not focused
 * - `contour-color-edit` (Vec3): the hexadecimal color of the contour when the text is being edited
 * - `contour-color-focused` (Vec3): the hexadecimal color of the contour when the component is focused
 * - `contour-color-unfocused` (Vec3): the hexadecimal color of the contour when the component is not focused
 * - `contour-opacity-edit` (Real): the real opacity of the contour when the the text is being edited
 * - `contour-opacity-focused` (Real): the real opacity of the contour when the component is focused
 * - `contour-opacity-unfocused` (Real): the real opacity of the contour when the component is not focused
 * - `text-autonewline-enabled` (Boolean): the boolean to allow an automatic carriage return if the x-size of the component is too short (only for `C_TextFieldType_TextEdit` type)
 * - `text-color-focused` (Vec3): the hexadecimal color of the displayed text when the component is focused
 * - `text-color-unfocused` (Vec3): the hexadecimal color of the displayed text when the component is not focused
 * - `text-format` (Text): the formatting of the displayed text when the component is focused
 * - `text-maxlines` (Integer): the integer to set the maximum line of a text (only for `C_TextFieldType_TextEdit` type).
 * - `text-opacity-focused` (Real): the real opacity of the displayed text when the component is focused
 * - `text-opacity-unfocused` (Real): the real opacity of the displayed text when the component is not focused
 * - `text-placeholder` (Text): the text to display in the component when there is no text value. The opacity and the color of the placeholder cannot be changed.
 * - `text-prefix` (Text): the prefix of the text displayed in the text-field (the placeholder is not impacted, CMlTextEdit doesn't support text prefix)
 * - `text-size` (Real): the real size of the displayed text
 * - `text-value` (Text): the text value of the displayed text
 * - `text-valuetype` (CMlEntry::ValueType): the type of the displayed text.
 *      Manialink parameters : `Ml_Natural`, Ml_Integer, Ml_Real...
 */

///	The component is selected by the Navigation component
Void Private_Select(CMlScript _This, CMlFrame _Frame_TextField) {
	declare K_State CMGame_TextField2_State for _This.Page = C_State_Null;
	declare K_TextField CMGame_TextField2_TextField for _Frame_TextField = C_TextField_Null;

	switch (CMGame_TextField2_TextField.MergedProperties.Type) {
		case C_TextFieldType_Entry: {
			// Prepare the component for edition
			CMGame_TextField2_TextField.Controls.Entry_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorFocused;
			CMGame_TextField2_TextField.Controls.Entry_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityFocused;
			CMGame_TextField2_TextField.Controls.Entry_Label.Value = CMGame_TextField2_TextField.MergedProperties.TextValue;
			if (CMGame_TextField2_TextField.MergedProperties.TextValue == "") {
				CMGame_TextField2_TextField.Controls.Entry_Label.TextFormat = CMGame_TextField2_TextField.MergedProperties.TextFormatEntry;
			}

			// Start edition with a delay for the TextFormat to be correctly applied
			CMGame_TextField2_State.DelayedStartEdition = CMGame_TextField2_TextField.Controls.Entry_Label;
		}
		case C_TextFieldType_TextEdit: {
			// Prepare the component for edition
			CMGame_TextField2_TextField.Controls.TextEdit_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorFocused;
			CMGame_TextField2_TextField.Controls.TextEdit_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityFocused;
			if (CMGame_TextField2_TextField.MergedProperties.TextValue == "") {
				CMGame_TextField2_TextField.Controls.TextEdit_Label.TextFormat = CMGame_TextField2_TextField.MergedProperties.TextFormatTextEdit;
				CMGame_TextField2_TextField.Controls.TextEdit_Label.Value = "";
			}

			// Start edition with a delay for the TextFormat to be correctly applied
			CMGame_TextField2_State.DelayedStartEdition = CMGame_TextField2_TextField.Controls.TextEdit_Label;
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
///	The component is focused by the Navigation component
Void Private_Focus(CMlScript _This, CMlFrame _Frame_TextField, Boolean _IsForced, Boolean _IsAnimated) {
	declare K_TextField CMGame_TextField2_TextField for _Frame_TextField = C_TextField_Null;

	if (!CMGame_TextField2_TextField.IsFocused || _IsForced) {
		switch (CMGame_TextField2_TextField.MergedProperties.Type) {
			case C_TextFieldType_Entry: {
				_This.AnimMgr.Flush(CMGame_TextField2_TextField.Controls.Entry_Label);
				if (CMGame_TextField2_TextField.MergedProperties.TextValue == "") {
					CMGame_TextField2_TextField.Controls.Entry_Label.Opacity = C_Placeholder_Opacity;
					CMGame_TextField2_TextField.Controls.Entry_Label.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Secondary);
				} else if (_IsAnimated) {
					_This.AnimMgr.Add(CMGame_TextField2_TextField.Controls.Entry_Label, "<anim textcolor=\""^CL::RgbToHex6(CMGame_TextField2_TextField.MergedProperties.TextColorFocused)^"\" opacity=\""^CMGame_TextField2_TextField.MergedProperties.TextOpacityFocused^"\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
				} else {
					CMGame_TextField2_TextField.Controls.Entry_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorFocused;
					CMGame_TextField2_TextField.Controls.Entry_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityFocused;
				}
			}
			case C_TextFieldType_TextEdit: {
				_This.AnimMgr.Flush(CMGame_TextField2_TextField.Controls.TextEdit_Label);
				if (CMGame_TextField2_TextField.MergedProperties.TextValue == "") {
					CMGame_TextField2_TextField.Controls.TextEdit_Label.Opacity = C_Placeholder_Opacity;
					CMGame_TextField2_TextField.Controls.TextEdit_Label.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Secondary);
				} else { //< Animations are not working on CMlTextEdit
					CMGame_TextField2_TextField.Controls.TextEdit_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorFocused;
					CMGame_TextField2_TextField.Controls.TextEdit_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityFocused;
				}
			}
		}

		// Text to speech
		if (CMGame_TextField2_TextField.MergedProperties.TextValue != "") {
			ExpendableButton::SetTextTTSAlt(CMGame_TextField2_TextField.Controls.Button_Background, TL::Compose(C_TTS_TextDisplayed, CMGame_TextField2_TextField.MergedProperties.TextPlaceholder, CMGame_TextField2_TextField.MergedProperties.TextValue));
		} else if (CMGame_TextField2_TextField.MergedProperties.TextPlaceholder != "") {
			ExpendableButton::SetTextTTSAlt(CMGame_TextField2_TextField.Controls.Button_Background, TL::Compose(C_TTS_PlaceholderDisplayed, CMGame_TextField2_TextField.MergedProperties.TextPlaceholder));
		} else {
			ExpendableButton::SetTextTTSAlt(CMGame_TextField2_TextField.Controls.Button_Background, C_TTS_Empty);
		}

		// Update focus
		ExpendableButton::Focus(CMGame_TextField2_TextField.Controls.Button_Background, _IsAnimated);
		CMGame_TextField2_TextField.IsFocused = True;
	}
}

///	The component is unfocused by the Navigation component
Void Private_Unfocus(CMlScript _This, CMlFrame _Frame_TextField, Boolean _IsForced, Boolean _IsAnimated) {
	declare K_TextField CMGame_TextField2_TextField for _Frame_TextField = C_TextField_Null;

	if (CMGame_TextField2_TextField.IsFocused || _IsForced) {
		switch (CMGame_TextField2_TextField.MergedProperties.Type) {
			case C_TextFieldType_Entry: {
				_This.AnimMgr.Flush(CMGame_TextField2_TextField.Controls.Entry_Label);
				if (CMGame_TextField2_TextField.MergedProperties.TextValue == "") {
					CMGame_TextField2_TextField.Controls.Entry_Label.Opacity = C_Placeholder_Opacity;
					CMGame_TextField2_TextField.Controls.Entry_Label.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Secondary);
				} else if (_IsAnimated) {
					_This.AnimMgr.Add(CMGame_TextField2_TextField.Controls.Entry_Label, "<anim textcolor=\""^CL::RgbToHex6(CMGame_TextField2_TextField.MergedProperties.TextColorUnfocused)^"\" opacity=\""^CMGame_TextField2_TextField.MergedProperties.TextOpacityUnfocused^"\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
				} else {
					CMGame_TextField2_TextField.Controls.Entry_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorUnfocused;
					CMGame_TextField2_TextField.Controls.Entry_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityUnfocused;
				}
			}
			case C_TextFieldType_TextEdit: {
				_This.AnimMgr.Flush(CMGame_TextField2_TextField.Controls.TextEdit_Label);
				if (CMGame_TextField2_TextField.MergedProperties.TextValue == "") {
					CMGame_TextField2_TextField.Controls.TextEdit_Label.Opacity = C_Placeholder_Opacity;
					CMGame_TextField2_TextField.Controls.TextEdit_Label.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Secondary);
				} else { //< Animations are not working on CMlTextEdit
					CMGame_TextField2_TextField.Controls.TextEdit_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorUnfocused;
					CMGame_TextField2_TextField.Controls.TextEdit_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityUnfocused;
				}
			}
		}

		// Update focus
		ExpendableButton::Unfocus(CMGame_TextField2_TextField.Controls.Button_Background, _IsAnimated);
		CMGame_TextField2_TextField.IsFocused = False;
	}
}

/// Enable or disable a properties from the `Enabled` array
Integer[] Private_EnableProperty(Integer[] _EnabledProperties, Integer _Property, Boolean _Enabled) {
	declare Integer[] EnabledProperties = _EnabledProperties;

	if (_Enabled) {
		if (!EnabledProperties.exists(_Property)) {
			EnabledProperties.add(_Property);
		}
	} else {
		EnabledProperties.remove(_Property);
	}

	return EnabledProperties;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set and enable a property
K_Properties Private_SetProperty(K_Properties _TargetProperties, Integer _PropertyId, K_Properties _SourceProperties) {
	declare K_Properties Properties = _TargetProperties;

	Properties.Enabled = Private_EnableProperty(Properties.Enabled, _PropertyId, True);

	switch (_PropertyId) {
		case C_Property_HAlign: Properties.HAlign = _SourceProperties.HAlign;
		case C_Property_VAlign: Properties.VAlign = _SourceProperties.VAlign;
		case C_Property_Size: Properties.Size = _SourceProperties.Size;
		case C_Property_Skew: Properties.Skew = _SourceProperties.Skew;
		case C_Property_Tooltip: Properties.Tooltip = _SourceProperties.Tooltip;
		case C_Property_Type: Properties.Type = _SourceProperties.Type;
		case C_Property_ButtonStyles: Properties.ButtonStyles = _SourceProperties.ButtonStyles;
		case C_Property_BackgroundColorFocused: Properties.BackgroundColorFocused = _SourceProperties.BackgroundColorFocused;
		case C_Property_BackgroundColorUnfocused: Properties.BackgroundColorUnfocused = _SourceProperties.BackgroundColorUnfocused;
		case C_Property_BackgroundEmptyEnabled: Properties.BackgroundEmptyEnabled = _SourceProperties.BackgroundEmptyEnabled;
		case C_Property_BackgroundOpacityFocused: Properties.BackgroundOpacityFocused = _SourceProperties.BackgroundOpacityFocused;
		case C_Property_BackgroundOpacityUnfocused: Properties.BackgroundOpacityUnfocused = _SourceProperties.BackgroundOpacityUnfocused;
		case C_Property_ContourColorEdit: Properties.ContourColorEdit = _SourceProperties.ContourColorEdit;
		case C_Property_ContourColorFocused: Properties.ContourColorFocused = _SourceProperties.ContourColorFocused;
		case C_Property_ContourColorUnfocused: Properties.ContourColorUnfocused = _SourceProperties.ContourColorUnfocused;
		case C_Property_ContourOpacityEdit: Properties.ContourOpacityEdit = _SourceProperties.ContourOpacityEdit;
		case C_Property_ContourOpacityFocused: Properties.ContourOpacityFocused = _SourceProperties.ContourOpacityFocused;
		case C_Property_ContourOpacityUnfocused: Properties.ContourOpacityUnfocused = _SourceProperties.ContourOpacityUnfocused;
		case C_Property_TextAutoNewLineEnabled: Properties.TextAutoNewLineEnabled = _SourceProperties.TextAutoNewLineEnabled;
		case C_Property_TextColorFocused: Properties.TextColorFocused = _SourceProperties.TextColorFocused;
		case C_Property_TextColorUnfocused: Properties.TextColorUnfocused = _SourceProperties.TextColorUnfocused;
		case C_Property_TextFormat: {
			Properties.TextFormatEntry = _SourceProperties.TextFormatEntry;
			Properties.TextFormatTextEdit = _SourceProperties.TextFormatTextEdit;
		}
		case C_Property_TextMaxlines: Properties.TextMaxlines = _SourceProperties.TextMaxlines;
		case C_Property_TextOpacityFocused: Properties.TextOpacityFocused = _SourceProperties.TextOpacityFocused;
		case C_Property_TextOpacityUnfocused: Properties.TextOpacityUnfocused = _SourceProperties.TextOpacityUnfocused;
		case C_Property_TextPlaceholder: Properties.TextPlaceholder = _SourceProperties.TextPlaceholder;
		case C_Property_TextPrefix: Properties.TextPrefix = _SourceProperties.TextPrefix;
		case C_Property_TextSize: Properties.TextSize = _SourceProperties.TextSize;
		case C_Property_TextValue: Properties.TextValue = _SourceProperties.TextValue;
		case C_Property_TextValueType: Properties.TextValueType = _SourceProperties.TextValueType;
	}

	return Properties;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Merge a base properties with other enabled properties
K_Properties Private_MergeProperties(K_Properties _Base, K_Properties[] _PropertiesToMerge) {
	declare K_Properties MergedProperties = _Base;

	foreach (Properties in _PropertiesToMerge) {
		foreach (PropertyId in Properties.Enabled) {
			MergedProperties = Private_SetProperty(MergedProperties, PropertyId, Properties);
		}
	}

	return MergedProperties;
}

/// Redraw the component (apply style, width, ...)
Void Private_Redraw(CMlScript _This, CMlFrame _Frame_TextField) {
	declare K_State CMGame_TextField2_State for _This.Page = C_State_Null;
	declare K_TextField CMGame_TextField2_TextField for _Frame_TextField = C_TextField_Null;
	declare Text PreviousValue = CMGame_TextField2_TextField.MergedProperties.TextValue;

	CMGame_TextField2_TextField.MergedProperties = Private_MergeProperties(
		CMGame_TextField2_State.DefaultProperties,
		[CMGame_TextField2_TextField.CustomizedProperties]
	);

	// Send an event when the text value is updated
	if (PreviousValue != CMGame_TextField2_TextField.MergedProperties.TextValue) {
		declare K_Event_ValueUpdate[] CMGame_TextField_Events_ValueUpdate for _This.Page = [];
		CMGame_TextField_Events_ValueUpdate.add(K_Event_ValueUpdate {
			Component = _Frame_TextField,
			Value = CMGame_TextField2_TextField.MergedProperties.TextValue
		});
	}

	// Tooltip
	if (CMGame_TextField2_TextField.MergedProperties.Tooltip != "") {
		CMGame_TextField2_TextField.Controls.Entry_Label.ToolTip = CMGame_TextField2_TextField.MergedProperties.Tooltip;
		CMGame_TextField2_TextField.Controls.TextEdit_Label.ToolTip = CMGame_TextField2_TextField.MergedProperties.Tooltip;
	} else if (CMGame_TextField2_TextField.MergedProperties.TextPlaceholder != "") {
		CMGame_TextField2_TextField.Controls.Entry_Label.ToolTip = CMGame_TextField2_TextField.MergedProperties.TextPlaceholder;
		CMGame_TextField2_TextField.Controls.TextEdit_Label.ToolTip = CMGame_TextField2_TextField.MergedProperties.TextPlaceholder;
	} else {
		CMGame_TextField2_TextField.Controls.Entry_Label.ToolTip = "";
		CMGame_TextField2_TextField.Controls.TextEdit_Label.ToolTip = "";
	}

	// TextField label
	CMGame_TextField2_TextField.Controls.Entry_Label.Visible = CMGame_TextField2_TextField.MergedProperties.Type == C_TextFieldType_Entry;
	CMGame_TextField2_TextField.Controls.TextEdit_Label.Visible = CMGame_TextField2_TextField.MergedProperties.Type == C_TextFieldType_TextEdit;
	switch (CMGame_TextField2_TextField.MergedProperties.Type) {
		case C_TextFieldType_Entry: {
			CMGame_TextField2_TextField.Controls.Entry_Label.SetValueType(CMGame_TextField2_TextField.MergedProperties.TextValueType);
			/* AutoNewLine and MaxLine are deprecated on `CMlEntry`, they never had any effect.
			CMGame_TextField2_TextField.Controls.Entry_Label.AutoNewLine = CMGame_TextField2_TextField.MergedProperties.TextAutoNewLineEnabled;
			CMGame_TextField2_TextField.Controls.Entry_Label.MaxLine = CMGame_TextField2_TextField.MergedProperties.TextMaxlines;
			*/
			CMGame_TextField2_TextField.Controls.Entry_Label.Size = <CMGame_TextField2_TextField.MergedProperties.Size.X-4*MLHelpers::SkewX(CMGame_TextField2_TextField.MergedProperties.Size.Y, CMGame_TextField2_TextField.MergedProperties.Skew), CMGame_TextField2_TextField.MergedProperties.Size.Y - 2.>;
			CMGame_TextField2_TextField.Controls.Entry_Label.TextSizeReal = CMGame_TextField2_TextField.MergedProperties.TextSize;
			if (CMGame_TextField2_TextField.MergedProperties.TextValue == "") {
				CMGame_TextField2_TextField.Controls.Entry_Label.Opacity = C_Placeholder_Opacity;
				CMGame_TextField2_TextField.Controls.Entry_Label.TextFormat = CMlEntry::ETextFormat::Basic;
				CMGame_TextField2_TextField.Controls.Entry_Label.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Secondary);
				CMGame_TextField2_TextField.Controls.Entry_Label.Value = C_Prefix_Default^TL::GetTranslatedText(CMGame_TextField2_TextField.MergedProperties.TextPlaceholder);
			} else {
				CMGame_TextField2_TextField.Controls.Entry_Label.TextFormat = CMGame_TextField2_TextField.MergedProperties.TextFormatEntry;
				if (CMGame_TextField2_TextField.Controls.Entry_Label.TextFormat == CMlEntry::ETextFormat::Password || CMGame_TextField2_TextField.Controls.Entry_Label.TextFormat == CMlEntry::ETextFormat::Newpassword) {
					CMGame_TextField2_TextField.Controls.Entry_Label.Value = CMGame_TextField2_TextField.MergedProperties.TextValue;
				} else {
					CMGame_TextField2_TextField.Controls.Entry_Label.Value = CMGame_TextField2_TextField.MergedProperties.TextPrefix^CMGame_TextField2_TextField.MergedProperties.TextValue;
				}
				if (CMGame_TextField2_TextField.IsFocused) {
					CMGame_TextField2_TextField.Controls.Entry_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityFocused;
					CMGame_TextField2_TextField.Controls.Entry_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorFocused;
				} else {
					CMGame_TextField2_TextField.Controls.Entry_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityUnfocused;
					CMGame_TextField2_TextField.Controls.Entry_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorUnfocused;
				}
			}
		}
		case C_TextFieldType_TextEdit: {
			CMGame_TextField2_TextField.Controls.TextEdit_Label.AutoNewLine = CMGame_TextField2_TextField.MergedProperties.TextAutoNewLineEnabled;
			CMGame_TextField2_TextField.Controls.TextEdit_Label.MaxLine = CMGame_TextField2_TextField.MergedProperties.TextMaxlines;
			CMGame_TextField2_TextField.Controls.TextEdit_Label.Size = <CMGame_TextField2_TextField.MergedProperties.Size.X-4*MLHelpers::SkewX(CMGame_TextField2_TextField.MergedProperties.Size.Y, CMGame_TextField2_TextField.MergedProperties.Skew), CMGame_TextField2_TextField.MergedProperties.Size.Y - 2.>;
			CMGame_TextField2_TextField.Controls.TextEdit_Label.TextSizeReal = CMGame_TextField2_TextField.MergedProperties.TextSize;
			if (CMGame_TextField2_TextField.MergedProperties.TextValue == "") {
				CMGame_TextField2_TextField.Controls.TextEdit_Label.Opacity = C_Placeholder_Opacity;
				CMGame_TextField2_TextField.Controls.TextEdit_Label.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Secondary);
				CMGame_TextField2_TextField.Controls.TextEdit_Label.TextFormat = CMlTextEdit::EControlScriptEditorTextFormat::Basic;
				CMGame_TextField2_TextField.Controls.TextEdit_Label.Value = TL::GetTranslatedText(CMGame_TextField2_TextField.MergedProperties.TextPlaceholder);
			} else {
				CMGame_TextField2_TextField.Controls.TextEdit_Label.TextFormat = CMGame_TextField2_TextField.MergedProperties.TextFormatTextEdit;
				CMGame_TextField2_TextField.Controls.TextEdit_Label.Value = CMGame_TextField2_TextField.MergedProperties.TextValue;
				if (CMGame_TextField2_TextField.IsFocused) {
					CMGame_TextField2_TextField.Controls.TextEdit_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityFocused;
					CMGame_TextField2_TextField.Controls.TextEdit_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorFocused;
				} else {
					CMGame_TextField2_TextField.Controls.TextEdit_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityUnfocused;
					CMGame_TextField2_TextField.Controls.TextEdit_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorUnfocused;
				}
			}
		}
	}

	// TextField background
	CMGame_TextField2_TextField.Controls.Button_Background.Visible = !CMGame_TextField2_TextField.MergedProperties.BackgroundEmptyEnabled;
	if (CMGame_TextField2_TextField.Controls.Button_Background.Visible) {
		ExpendableButton::SetWidth(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.Size.X);
		ExpendableButton::SetHeight(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.Size.Y);
		if (CMGame_TextField2_TextField.MergedProperties.ButtonStyles.count > 0) {
			ExpendableButton::SetStyles(
				_This,
				CMGame_TextField2_TextField.Controls.Button_Background,
				CMGame_TextField2_TextField.MergedProperties.ButtonStyles
			);
		} else if (CMGame_TextField2_TextField.MergedProperties.Skew == C_Skew_Margin) { //< When the TextField is not skewed
			ExpendableButton::SetStyles(
				_This,
				CMGame_TextField2_TextField.Controls.Button_Background,
				[ExpendableButtonStyles::C_ButtonStyle_SquareStraight]
			);
		} else {
			ExpendableButton::SetStyles(
				_This,
				CMGame_TextField2_TextField.Controls.Button_Background,
				[ExpendableButtonStyles::C_ButtonStyle_SkewStraight]
			);
		}
		ExpendableButton::SetBgColorFocus(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.BackgroundColorFocused);
		ExpendableButton::SetBgColorDefault(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.BackgroundColorUnfocused);
		ExpendableButton::SetBgOpacityFocus(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.BackgroundOpacityFocused);
		ExpendableButton::SetBgOpacityDefault(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.BackgroundOpacityUnfocused);
		ExpendableButton::SetFgColorFocus(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.ContourColorFocused);
		ExpendableButton::SetFgColorDefault(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.ContourColorUnfocused);
		ExpendableButton::SetFgOpacityFocus(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.ContourOpacityFocused);
		ExpendableButton::SetFgOpacityDefault(CMGame_TextField2_TextField.Controls.Button_Background, CMGame_TextField2_TextField.MergedProperties.ContourOpacityUnfocused);
		if (CMGame_TextField2_TextField.IsFocused) {
			ExpendableButton::Focus(CMGame_TextField2_TextField.Controls.Button_Background, False);
		} else {
			ExpendableButton::Unfocus(CMGame_TextField2_TextField.Controls.Button_Background, False);
		}
	}

	// TextField hitbox
	// For an explanation of the calculation, see https://confluence.ubisoft.com/display/Nadeo/Hitboxes
	CMGame_TextField2_TextField.Controls.Frame_HitboxClip.Size = CMGame_TextField2_TextField.MergedProperties.Size;
	if (CMGame_TextField2_TextField.MergedProperties.Skew != C_Skew_Margin) {
		declare Real TextFieldVisibleWidth = CMGame_TextField2_TextField.MergedProperties.Size.X - MLHelpers::SkewX(CMGame_TextField2_TextField.MergedProperties.Size.Y, CMGame_TextField2_TextField.MergedProperties.Skew); //< Hypothenuse
		declare Real TextFieldSkewRad = ML::DegToRad(CMGame_TextField2_TextField.MergedProperties.Skew);
		CMGame_TextField2_TextField.Controls.Quad_NavigationZone.RelativeRotation = CMGame_TextField2_TextField.MergedProperties.Skew;
		CMGame_TextField2_TextField.Controls.Quad_NavigationZone.Size.X = TextFieldVisibleWidth * ML::Cos(TextFieldSkewRad);
		if (ML::Cos(TextFieldSkewRad) == 0.) {
			CMGame_TextField2_TextField.Controls.Quad_NavigationZone.Size.Y = 0.;
		} else {
			CMGame_TextField2_TextField.Controls.Quad_NavigationZone.Size.Y = (TextFieldVisibleWidth * ML::Sin(TextFieldSkewRad)) + (CMGame_TextField2_TextField.MergedProperties.Size.Y / ML::Cos(TextFieldSkewRad));
		}
	} else { //< When the TextField is not skewed
		CMGame_TextField2_TextField.Controls.Quad_NavigationZone.RelativeRotation = 0.;
		CMGame_TextField2_TextField.Controls.Quad_NavigationZone.Size = CMGame_TextField2_TextField.MergedProperties.Size;
		CMGame_TextField2_TextField.Controls.Frame_HitboxClip.Size = <0., 0.>; //< This default value disable the clipping
	}

	// Align the component
	Tools::Align(
		CMGame_TextField2_TextField.Controls.Frame_Align,
		CMGame_TextField2_TextField.MergedProperties.Size,
		CMGame_TextField2_TextField.MergedProperties.HAlign,
		CMGame_TextField2_TextField.MergedProperties.VAlign
	);

	// Update focus
	if (CMGame_TextField2_TextField.IsFocused) {
		Private_Focus(_This, _Frame_TextField, True, False);
	} else {
		Private_Unfocus(_This, _Frame_TextField, True, False);
	}
}

/// Update the component when edited
Void Private_Update(CMlScript _This) {
	declare K_State CMGame_TextField2_State for _This.Page = C_State_Null;
	declare CMlFrame Frame_TextField = CMGame_TextField2_State.FocusedTextField.Parent.Parent.Parent;
	declare K_TextField CMGame_TextField2_TextField for Frame_TextField = C_TextField_Null;
	declare Vec3 ContourColor = CMGame_TextField2_TextField.MergedProperties.ContourColorEdit;
	declare Real ContourOpacity = CMGame_TextField2_TextField.MergedProperties.ContourOpacityEdit;
	declare Boolean IsEventRequired;

	CMGame_TextField2_State.IsTextFieldFocused = CMGame_TextField2_State.FocusedTextField.IsFocused;
	if (!CMGame_TextField2_State.IsTextFieldFocused) {
		ContourColor = CMGame_TextField2_TextField.MergedProperties.ContourColorFocused;
		ContourOpacity = CMGame_TextField2_TextField.MergedProperties.BackgroundOpacityFocused;
		if (CMGame_TextField2_TextField.MergedProperties.ContourOpacityFocused != -1.) {
			ContourOpacity = CMGame_TextField2_TextField.MergedProperties.ContourOpacityFocused;
		}

		switchtype (CMGame_TextField2_State.FocusedTextField) {
			case CMlEntry: {
				declare CMlEntry Entry_Label = CMGame_TextField2_State.FocusedTextField as CMlEntry;
				IsEventRequired = CMGame_TextField2_TextField.MergedProperties.TextValue != Entry_Label.Value;
				CMGame_TextField2_TextField.MergedProperties.TextValue = Entry_Label.Value;
				if (Entry_Label.Value == "") {
					Entry_Label.Opacity = C_Placeholder_Opacity;
					Entry_Label.TextFormat = CMlEntry::ETextFormat::Basic;
					Entry_Label.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Secondary);
					Entry_Label.Value = C_Prefix_Default^TL::GetTranslatedText(CMGame_TextField2_TextField.MergedProperties.TextPlaceholder);
				} else {
					Entry_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityFocused;
					Entry_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorFocused;
					Entry_Label.TextFormat = CMGame_TextField2_TextField.MergedProperties.TextFormatEntry;
					if (Entry_Label.TextFormat != CMlEntry::ETextFormat::Password && Entry_Label.TextFormat != CMlEntry::ETextFormat::Newpassword) {
						Entry_Label.Value = CMGame_TextField2_TextField.MergedProperties.TextPrefix^Entry_Label.Value;
					}
				}
			}
			case CMlTextEdit: {
				declare CMlTextEdit TextEdit_Label = CMGame_TextField2_State.FocusedTextField as CMlTextEdit;
				IsEventRequired = CMGame_TextField2_TextField.MergedProperties.TextValue != TextEdit_Label.Value;
				CMGame_TextField2_TextField.MergedProperties.TextValue = TextEdit_Label.Value;
				if (TextEdit_Label.Value == "") {
					TextEdit_Label.Opacity = C_Placeholder_Opacity;
					TextEdit_Label.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_Secondary);
					TextEdit_Label.TextFormat = CMlTextEdit::EControlScriptEditorTextFormat::Basic;
					TextEdit_Label.Value = TL::GetTranslatedText(CMGame_TextField2_TextField.MergedProperties.TextPlaceholder);
				} else {
					TextEdit_Label.Opacity = CMGame_TextField2_TextField.MergedProperties.TextOpacityFocused;
					TextEdit_Label.TextColor = CMGame_TextField2_TextField.MergedProperties.TextColorFocused;
					TextEdit_Label.TextFormat = CMGame_TextField2_TextField.MergedProperties.TextFormatTextEdit;
				}
			}
		}

		// Send an event when the text value is updated
		if (IsEventRequired) {
			declare K_Event_ValueUpdate[] CMGame_TextField_Events_ValueUpdate for _This.Page = [];
			CMGame_TextField_Events_ValueUpdate.add(K_Event_ValueUpdate {
				Component = Frame_TextField,
				Value = CMGame_TextField2_TextField.MergedProperties.TextValue
			});
		}
	}

	ExpendableButton::SetFgColorFocus(CMGame_TextField2_TextField.Controls.Button_Background, ContourColor);
	ExpendableButton::SetFgOpacityFocus(CMGame_TextField2_TextField.Controls.Button_Background, ContourOpacity);
}

	}

	if (MV::HasProp(_Frame_TextField, C_Prop_BackgroundEmptyEnabled)) {

		Private_SetCustomizedProperty(_This, _Frame_TextField, C_Property_ContourColorEdit, K_Properties { ContourColorEdit = MV::GetProp_Color(_Frame_TextField, C_Prop_ContourColorEdit, CMGame_TextField2_State.DefaultProperties.ContourColorEdit) }, False);
	}
	if (MV::HasProp(_Frame_TextField, C_Prop_ContourColorFocused)) {
		Private_SetCustomizedProperty(_This, _Frame_TextField, C_Property_ContourColorFocused, K_Properties { ContourColorFocused = MV::GetProp_Color(_Frame_TextField, C_Prop_ContourColorFocused, CMGame_TextField2_State.DefaultProperties.ContourColorFocused) }, False);
	}
	if (MV::HasProp(_Frame_TextField, C_Prop_ContourColorUnfocused)) {
		Private_SetCustomizedProperty(_This, _Frame_TextField, C_Property_ContourColorUnfocused, K_Properties { ContourColorUnfocused = MV::GetProp_Color(_Frame_TextField, C_Prop_ContourColorUnfocused, CMGame_TextField2_State.DefaultProperties.ContourColorUnfocused) }, False);
	}
	if (MV::HasProp(_Frame_TextField, C_Prop_ContourOpacityEdit)) {
		Private_SetCustomizedProperty(_This, _Frame_TextField, C_Property_ContourOpacityEdit, K_Properties { ContourOpacityEdit = MV::GetProp_Real(_Frame_TextField, C_Prop_ContourOpacityEdit, CMGame_TextField2_State.DefaultProperties.ContourOpacityEdit) }, False);
	}
	if (MV::HasProp(_Frame_TextField, C_Prop_ContourOpacityFocused)) {
		Private_SetCustomizedProperty(_This, _Frame_TextField, C_Property_ContourOpacityFocused, K_Properties { ContourOpacityFocused = MV::GetProp_Real(_Frame_TextField, C_Prop_ContourOpacityFocused, CMGame_TextField2_State.DefaultProperties.ContourOpacityFocused) }, False);
	}
	if (MV::HasProp(_Frame_TextField, C_Prop_ContourOpacityUnfocused)) {
		Private_SetCustomizedProperty(_This, _Frame_TextField, C_Property_ContourOpacityUnfocused, K_Properties { ContourOpacityUnfocused = MV::GetProp_Real(_Frame_TextField, C_Prop_ContourOpacityUnfocused, CMGame_TextField2_State.DefaultProperties.ContourOpacityUnfocused) }, False);

/// Manialink page yield hook
Void Private_Hook_PageAfterYield(CMlScript _This) {
	declare K_State CMGame_TextField2_State for _This.Page = C_State_Null;

	// Manage navigation events
	if (!CMGame_TextField2_State.IsTextFieldFocused) {
		foreach (Event in Navigation::GetPendingEvents(_This)) {
			switch (Event.Input) {
				case Navigation::C_InputType_Select, Navigation::C_InputType_MouseClick: {
					declare K_TextField CMGame_TextField2_TextField for Event.To = C_TextField_Null;
					if (CMGame_TextField2_TextField.IsTextField && CMGame_TextField2_TextField.IsFocused) {
						Private_Select(_This, Event.To as CMlFrame);
					}
				}
			}
		}
	}

	// Manage delayed edition
	if (CMGame_TextField2_State.DelayedStartEdition != Null) {
		if (CMGame_TextField2_State.FocusedTextField != CMGame_TextField2_State.DelayedStartEdition) {
			CMGame_TextField2_State.FocusedTextField = CMGame_TextField2_State.DelayedStartEdition;
		} else {
			CMGame_TextField2_State.DelayedStartEdition = Null;
			switchtype (CMGame_TextField2_State.FocusedTextField as TypedControl) {
				case CMlEntry: TypedControl.StartEdition();
				case CMlTextEdit: TypedControl.StartEdition();
			}
		}
	}

	// Handle entry focused state (must be placed at the end of the main loop)
	if (
		CMGame_TextField2_State.FocusedTextField != Null &&
		CMGame_TextField2_State.IsTextFieldFocused != CMGame_TextField2_State.FocusedTextField.IsFocused
	) {
		Private_Update(_This);
	}
}
