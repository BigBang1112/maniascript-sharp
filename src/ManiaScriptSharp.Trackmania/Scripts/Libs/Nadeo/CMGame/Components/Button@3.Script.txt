/**
 * Button@3 component
 * ==================
 *
 * Interactive element that the user can use to trigger an action
 *
 * Props
 * -----
 *
 * - `width-min` (`Real`): minimum width of the button
 *   Even if the button text is shorter, the button won't be narrower than this width.
 * - `width-max` (`Real`): maximum width of the button
 *   Even if the button text is longer, it will be reduced or cut so the button does not exceed this width.
 * - `width` (`Real`): forced width of the button
 *   This width will still be constrained by the minimum and maximum width.
 *   If the minimum width is 10, the maximum width 100, then if you set the width to 5 it will be clamped to 10.
 *   Same thing if you set the width to 200, it will be clamped to 100.
 * - `height` (`Real`): height of the button
 * - `margin` (`Margin`): margin around the component
 * - `margin-ratio` (`Margin`): margin for the component proportional to the component size
 * - `padding` (`Padding`): padding inside the component
 * - `padding-ratio` (`Padding`): padding for the component proportional to the component size
 * - `halign` (`Text`): horizontal alignment of the component, can be `left`, `center` or `right`
 * - `valign` (`Text`): vertical alignment of the component, can be `top`, `center` or `bottom`
 * - `is-visible` (`Boolean`): the visibility of the component
 * - `text` (`Text`): the text of the button
 * - `text-size-fit` (`Vec3`): size of the button text processed by the `FitLabelValue()` function
 * - `text-halign` (`Text`): horizontal alignment of the text displayed in the button
 *   Can be `left`, `center` or `right`
 * - `text-valign` (`Text`): vertical alignment of the text displayed in the button
 *   Can be `top`, `center`, `center2` or `bottom`
 */

/// Add a modifier to the button

	if (_Instance.Modifiers.exists(_Modifier)) return _Instance;

	declare K_Private_Instance Instance = _Instance;
	Instance.Modifiers.add(_Modifier);
	MV::MarkForRedraw(Instance.Controls.Frame_Button3, C_Private_RedrawTag_Modifiers);
	return Instance;

/// Redraw the component
Void Private_Hook_ComponentDraw(CMlFrame _Frame_Button3, Text[] _Tags) {
	declare K_Private_Instance CMGame_ComponentsButton3_Instance as Instance for _Frame_Button3 = C_Private_Instance_Null;

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Modifiers])) {
		if (Instance.Modifiers.exists(C_Modifier_Focus)) {
			Private_CopyLabelProperties(Instance.Controls.Label_TextFocus, Instance.Controls.Label_Text);
		} else {
			Private_CopyLabelProperties(Instance.Controls.Label_TextDefault, Instance.Controls.Label_Text);
		}
	}

	// /!\ WARNING /!\
	// Some of the label properties (font, text size, prefix, ...) must be applied to the label before computing its width
	declare Boolean SizeUpdated = False;
	if (MV::HasRedrawTag(_Tags, [
		C_Private_RedrawTag_Init,
		C_Private_RedrawTag_Size,
		C_Private_RedrawTag_TextValue,
		C_Private_RedrawTag_TextSizeFit
	])) {
		SizeUpdated = True;

		Instance.Controls.Label_Text.Value = Instance.Properties.TextValue;
		Instance.Controls.Label_Text.TextSizeReal = Instance.Properties.TextSizeFit.X;

		declare Vec2 Size = <0., ML::Max(0., Instance.Properties.Height)>;
		if (Instance.Properties.Width >= 0.) {
			Size.X = Instance.Properties.Width;
		} else if (Instance.Properties.TextValue != "") {
			declare Real LabelWidth = Instance.Controls.Label_Text.ComputeWidth(Instance.Properties.TextValue);
			declare Real PaddingRatio = 1. - (Instance.Properties.PaddingRatio.Left + Instance.Properties.PaddingRatio.Right);
			if (PaddingRatio == 0.) {
				Size.X = Instance.Properties.Padding.Left + LabelWidth + Instance.Properties.Padding.Right;
			} else {
				Size.X = (Instance.Properties.Padding.Left + LabelWidth + Instance.Properties.Padding.Right) / PaddingRatio;
			}
		}

		if (Instance.Properties.WidthMax >= 0. && Size.X > Instance.Properties.WidthMax) {
			Size.X = Instance.Properties.WidthMax;
		}
		if (Instance.Properties.WidthMin >= 0. && Size.X < Instance.Properties.WidthMin) {
			Size.X = Instance.Properties.WidthMin;
		}

		Instance.DrawCache.Box = Box::New(Size, Instance.Properties.Margin, Instance.Properties.MarginRatio, Instance.Properties.Padding, Instance.Properties.PaddingRatio);

		Component_Slice9::SetSize(Instance.Controls.Slice9_BackgroundDefault, Instance.DrawCache.Box.PaddingArea.Size);
		Component_Slice9::SetSize(Instance.Controls.Slice9_BackgroundFocus, Instance.DrawCache.Box.PaddingArea.Size);
		Component_Slice9::SetSize(Instance.Controls.Slice9_ForegroundDefault, Instance.DrawCache.Box.PaddingArea.Size);
		Component_Slice9::SetSize(Instance.Controls.Slice9_ForegroundFocus, Instance.DrawCache.Box.PaddingArea.Size);
		Instance.Controls.Quad_Hitbox.Size = Instance.DrawCache.Box.PaddingArea.Size;

		Instance.Controls.Frame_Content.RelativePosition_V3 = Instance.DrawCache.Box.ContentArea.Pos;
		Instance.Controls.Frame_Content.Size = Instance.DrawCache.Box.ContentArea.Size;
		Instance.Controls.Label_Text.Size = Instance.DrawCache.Box.ContentArea.Size;
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Align]) || SizeUpdated) {
		MLHelpers::AlignContent(Instance.Controls.Frame_Align, Instance.DrawCache.Box.MarginArea.Size, Instance.Properties.HAlign, Instance.Properties.VAlign);
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_TextSizeFit]) || SizeUpdated) {
		MLHelpers::FitLabelValue(Instance.Controls.Label_Text, Instance.Properties.TextSizeFit);
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_TextAlign]) || SizeUpdated) {
		MLHelpers::AlignControl(Instance.Controls.Label_Text, Instance.DrawCache.Box.ContentArea.Size, Instance.Properties.TextHAlign, Instance.Properties.TextVAlign);
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_Modifiers]) || SizeUpdated) {
		if (Instance.Modifiers.exists(C_Modifier_Focus)) {
			Component_Slice9::CopyProperties(Instance.Controls.Slice9_BackgroundFocus, Instance.Controls.Slice9_Background);
			Component_Slice9::CopyProperties(Instance.Controls.Slice9_ForegroundFocus, Instance.Controls.Slice9_Foreground);
		} else {
			Component_Slice9::CopyProperties(Instance.Controls.Slice9_BackgroundDefault, Instance.Controls.Slice9_Background);
			Component_Slice9::CopyProperties(Instance.Controls.Slice9_ForegroundDefault, Instance.Controls.Slice9_Foreground);
		}
	}

	if (MV::HasRedrawTag(_Tags, [C_Private_RedrawTag_Init, C_Private_RedrawTag_IsVisible])) {
		Instance.Controls.Frame_Button3.Visible = Instance.Properties.IsVisible;
	}
}
