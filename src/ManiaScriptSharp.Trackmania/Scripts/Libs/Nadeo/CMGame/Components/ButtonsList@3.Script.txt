 * Component ButtonsList

 * Info
 * ----

 * Display a list of button icons with the action they perform

 * Props
 * -----
 *
 * - `halign` (`Text`): horizontal alignment of the button, can be `left`, `center` or `right`
 * - `valign` (`Text`): vertical alignment of the button, can be `top`, `center` or `bottom`
 * - `textsizefit` (`Vec3`): the font size of the label displaying the button action
 * - `textcolor` (`Hex6`): the color of the label displaying the button action
 * - `textfont` (`Text`): the font of the label displaying the button action
 * - `textprefix` (`Text`): the prefix of the label displaying the button action

/// Get the template of the component
Text Private_GetTemplate() {
	declare Text ML_Buttons = "";
	for (I, 0, C_ButtonsNb - 1) {
		ML_Buttons ^= """<frameinstance modelid="{{{C_ModelId_Button}}}" />""";
	}

	return
"""
<framemodel id="{{{C_ModelId_Button}}}">
	<CMGameButtonIcon3
		id="{{{C_Id_ButtonIconActionIcon}}}"
		$size="{{{C_ActionIconSize.X}}} {{{C_ActionIconSize.Y}}}"
		$halign="left"
	/>
	<label
		id="{{{C_Id_LabelActionName}}}"
		class="{{{Stylesheet::C_LabelStyle_NormalSemiBold}}}"
		size="{{{C_ActionNameSize.X}}} {{{C_ActionNameSize.Y}}}"
		halign="left"
		textemboss="1"
	/>
</framemodel>
<frame id="{{{C_Id_FrameAlign}}}">
	{{{ML_Buttons}}}
</frame>
""";
}

/// Redraw the component
Void Private_Hook_ComponentDraw(CMlScript _This, CMlFrame _Frame_ButtonsList, Text[] _Tags) {
	declare K_Instance CMGame_ComponentsButtonsList3_Instance as Instance for _Frame_ButtonsList = C_Instance_Null;

	if (MV::HasRedrawTag(_Tags, C_RedrawTag_Init, [C_RedrawTag_Buttons, C_RedrawTag_TextProperty])) {
		Instance.Size = <0., ML::Max(C_ActionIconSize.Y, C_ActionNameSize.Y)>;

		declare Text[] ActionBinds;
		declare Text[] ActionNames;
		for (LocalActionBind => LocalActionName in Instance.SortedButtons) {
			ActionBinds.add(LocalActionBind);
			ActionNames.add(LocalActionName);
		}

		for (Key => Button in Instance.Controls.Buttons) {
			declare Text LocalActionBind = ActionBinds.get(Key, "");
			declare Text LocalActionName = ActionNames.get(Key, "");
			if (LocalActionBind == "") {
				Button.ButtonIcon_ActionIcon.Visible = False;
			} else {
				ButtonIcon::SetBind(_This, Button.ButtonIcon_ActionIcon, LocalActionBind);
				Button.ButtonIcon_ActionIcon.Visible = True;
				Button.ButtonIcon_ActionIcon.RelativePosition_V3 = <Instance.Size.X, Instance.Size.Y * -0.5>;
				Instance.Size.X += C_ActionIconSize.X + C_ActionBindNameMargin;
			}
			if (LocalActionName == "") {
				Button.Label_ActionName.Visible = False;
			} else {
				Button.Label_ActionName.Value = LocalActionName;
				Button.Label_ActionName.Visible = True;
				Button.Label_ActionName.TextColor = Instance.TextColor;
				Button.Label_ActionName.TextFont = Instance.TextFont;
				Button.Label_ActionName.TextPrefix = Instance.TextPrefix;
				Button.Label_ActionName.RelativePosition_V3 = <Instance.Size.X, Instance.Size.Y * -0.5>;
				Tools::FitLabelValue(Button.Label_ActionName, Instance.TextSizeFit.X, Instance.TextSizeFit.Y, Instance.TextSizeFit.Z);
				Instance.Size.X += Tools::ComputeLabelValueWidth(Button.Label_ActionName, True) + C_ButtonMargin;
			}
		}
	}

	if (MV::HasRedrawTag(_Tags, C_RedrawTag_Init, [C_RedrawTag_Align, C_RedrawTag_Buttons, C_RedrawTag_TextProperty])) {
		Tools::AlignFrame(Instance.Controls.Frame_Align, Instance.Size, Instance.HAlign, Instance.VAlign);
	}
}

/// Initialize the component
Void Private_Hook_ComponentInit(CMlFrame _Frame_ButtonsList) {
	declare K_Instance CMGame_ComponentsButtonsList3_Instance as Instance for _Frame_ButtonsList = C_Instance_Null;

	Instance = C_Instance_Null;
	Instance.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_Text_White);
	Instance.TextFont = Stylesheet::GetFont(Stylesheet::C_Font_SemiBold);

	Instance.Controls = K_Controls {
		Frame_ButtonsList = _Frame_ButtonsList,
		Frame_Align = (_Frame_ButtonsList.GetFirstChild(C_Id_FrameAlign) as CMlFrame)
	};
	for (Control in Instance.Controls.Frame_Align.Controls) {
		Instance.Controls.Buttons.add(K_Control_Button {
			ButtonIcon_ActionIcon = ((Control as CMlFrame).GetFirstChild(C_Id_ButtonIconActionIcon) as CMlFrame),
			Label_ActionName = ((Control as CMlFrame).GetFirstChild(C_Id_LabelActionName) as CMlLabel)
		});
	}

	Instance.HAlign = MV::GetProp_HAlign(_Frame_ButtonsList, C_Prop_HAlign, Instance.HAlign);
	Instance.VAlign = MV::GetProp_VAlign(_Frame_ButtonsList, C_Prop_VAlign, Instance.VAlign);
	Instance.TextSizeFit = MV::GetProp_Vec3(_Frame_ButtonsList, C_Prop_TextSizeFit, Instance.TextSizeFit);
	Instance.TextColor = MV::GetProp_Color(_Frame_ButtonsList, C_Prop_TextColor, Instance.TextColor);
	Instance.TextFont = MV::GetProp_Text(_Frame_ButtonsList, C_Prop_TextFont, Instance.TextFont);
	Instance.TextPrefix = MV::GetProp_Text(_Frame_ButtonsList, C_Prop_TextPrefix, Instance.TextPrefix);

	MV::MarkForRedraw(_Frame_ButtonsList, [C_RedrawTag_Init]);
}

/// Sort the given buttons
Text[Text] Private_SortButtons(Text[Text] _Buttons) {
	declare Text[] ButtonOrder = [
		C_Bind_Select, C_Bind_A, C_Bind_Cross,
		C_Bind_Cancel, C_Bind_B, C_Bind_Circle,
		C_Bind_Action1, C_Bind_X, C_Bind_Square,
		C_Bind_Action2, C_Bind_Y, C_Bind_Triangle,
		C_Bind_PageUp, C_Bind_LB, C_Bind_L1,
		C_Bind_PageDown, C_Bind_RB, C_Bind_R1,
		C_Bind_Action3, C_Bind_LT, C_Bind_L2,
		C_Bind_Action4, C_Bind_RT, C_Bind_R2,
		C_Bind_View, C_Bind_Touchpad,
		C_Bind_AppMenu, C_Bind_Menu, C_Bind_Options,
		C_Bind_Left, C_Bind_Right, C_Bind_Up, C_Bind_Down,
		C_Bind_LeftStick, C_Bind_LeftStick_Left, C_Bind_LeftStick_Right, C_Bind_LeftStick_Up, C_Bind_LeftStick_Down,
		C_Bind_RightStick, C_Bind_RightStick_Left, C_Bind_RightStick_Right,
		C_Bind_ScrollUp, C_Bind_RightStick_Up,
		C_Bind_ScrollDown, C_Bind_RightStick_Down
	];

	declare Integer[Text] ButtonsToSort;
	for (ActionBind => ActionName in _Buttons) {
		if (ButtonOrder.exists(ActionBind)) {
			ButtonsToSort[ActionBind] = ButtonOrder.keyof(ActionBind);
		} else {
			ButtonsToSort[ActionBind] = ButtonOrder.count;
		}
	}
	ButtonsToSort = ButtonsToSort.sort();

	declare Text[Text] SortedButtons;
	for (ActionBind => Order in ButtonsToSort) {
		SortedButtons[ActionBind] = _Buttons[ActionBind];
	}

	return SortedButtons;
}

/// Set the buttons to display in the component
Void SetButtons(CMlFrame _Frame_ButtonsList, Text[Text] _Buttons, Boolean _SortButtons) {
	declare K_Instance CMGame_ComponentsButtonsList3_Instance for _Frame_ButtonsList = C_Instance_Null;
	if (
		CMGame_ComponentsButtonsList3_Instance.Buttons != _Buttons ||
		CMGame_ComponentsButtonsList3_Instance.SortButtons != _SortButtons
	) {
		CMGame_ComponentsButtonsList3_Instance.Buttons = _Buttons;
		CMGame_ComponentsButtonsList3_Instance.SortButtons = _SortButtons;

		if (CMGame_ComponentsButtonsList3_Instance.SortButtons) {
			CMGame_ComponentsButtonsList3_Instance.SortedButtons = Private_SortButtons(_Buttons);
		} else {
			CMGame_ComponentsButtonsList3_Instance.SortedButtons = _Buttons;
		}
		MV::MarkForRedraw(CMGame_ComponentsButtonsList3_Instance.Controls.Frame_ButtonsList, [C_RedrawTag_Buttons]);
	}
}
Void SetButtons(CMlFrame _Frame_ButtonsList, Text[Text] _Buttons) {
	SetButtons(_Frame_ButtonsList, _Buttons, True);
}

/// Get the size of the buttons list
Vec2 GetSize(CMlScript _This, CMlFrame _Frame_ButtonsList) {
	declare Text[] RedrawTags = MV::DoManualRedraw(_Frame_ButtonsList);
	if (RedrawTags.count > 0) {
		Private_Hook_ComponentDraw(_This, _Frame_ButtonsList, RedrawTags);
	}
	declare K_Instance CMGame_ComponentsButtonsList3_Instance for _Frame_ButtonsList = C_Instance_Null;
	return CMGame_ComponentsButtonsList3_Instance.Size;
}
