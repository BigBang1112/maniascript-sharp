/// Load an audio library configuration JSON
K_Private_State Private_LoadLibraryJson(K_Private_State _State, Text _AudioFilePath, Text _Json) {
	if (
		!TL::StartsWith("{", _Json) ||
		!TL::EndsWith("}", _Json) ||
		TL::Length(_Json) <= 2
	) {
		log("WARNING [Audio@2] The library json is not valid");
		return _State;
	}

	declare K_Private_State State = _State;

	declare K_Private_AudioItem[Text] NewLibrary;
	NewLibrary.fromjson(_Json);

	declare Text Separator;
	if (!TL::EndsWith("/", _AudioFilePath) && !TL::EndsWith("\\", _AudioFilePath)) {
		Separator = "/";
	}

	// Update library
	declare Text[] UsedAudioSourceFilePaths;
	for (NewSoundId => NewAudioItem in NewLibrary) {
		assert(
			NewAudioItem.Sounds.count > 0,
			"""[Audio@2] The `Sounds` array of `{{{NewSoundId}}}` in the json cannot be empty"""
		);

		declare K_Private_AudioItem AudioItem = NewAudioItem;
		for (SoundKey => Sound in AudioItem.Sounds) {
			declare Text FilePath = _AudioFilePath^Separator^Sound.FileName;
			AudioItem.Sounds[SoundKey].FileName = FilePath;
			if (!UsedAudioSourceFilePaths.exists(FilePath)) {
				UsedAudioSourceFilePaths.add(FilePath);
			}
		}
		AudioItem.LastPlayedSoundKey = C_Private_NoSoundPlayedYet;
		State.Library[NewSoundId] = AudioItem;
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Load an audio library configuration file
K_Private_State Private_LoadLibraryFile(
	K_Private_State _State,
	CManiaApp _Context,
	Text _ConfigFilePath,
	Text _AudioFilePath
) {

	if (_State.LibraryLoadingRequests.existskey(_ConfigFilePath)) return _State; //< We already are loading this file

	declare K_Private_State State = _State;
	State.LibraryLoadingRequests[_ConfigFilePath] = K_Private_LibraryLoadingRequest {
		ConfigFilePath = _ConfigFilePath,
		AudioFilePath = _AudioFilePath,
		Request = Http::CreateGet(_Context, _ConfigFilePath)
	};
	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the requests to load the library files
K_Private_State Private_UpdateLibraryLoadingRequests(K_Private_State _State, CManiaApp _Context) {
	declare K_Private_State State = _State;
	declare Text[] LoadingComplete;

	for (FilePath => LibraryLoadingRequest in State.LibraryLoadingRequests) {
		declare Http::K_Request UpdatedRequest = Http::Update(_Context, LibraryLoadingRequest.Request);
		if (!Http::IsProcessing(UpdatedRequest)) {
			if (Http::IsSuccess(UpdatedRequest)) {
				State = Private_LoadLibraryJson(
					State,
					LibraryLoadingRequest.AudioFilePath,
					Http::GetResult(UpdatedRequest)
				);
			}
			UpdatedRequest = Http::Destroy(_Context, UpdatedRequest);
			LoadingComplete.add(FilePath);
		}
		State.LibraryLoadingRequests[FilePath].Request = UpdatedRequest;
	}

	for (FilePath in LoadingComplete) {
		State.LibraryLoadingRequests.removekey(FilePath);
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get a new lifetime expiration date relative to now
Integer Private_GetNewLifetimeExpiration(CAudioSource _AudioSource) {
	declare Integer PlayLength = 0;
	if (_AudioSource != Null) {
		PlayLength = ML::NearestInteger(_AudioSource.PlayLength * 1000);
	}
	return TiL::GetMillisecondsSinceInit() + PlayLength + C_Private_ActiveAudioSourceLifetime;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add a new CAudioSources to the state
K_Private_State Private_AddAudioSource(
	K_Private_State _State,
	Text _SoundFileName,
	CAudioSource _AudioSource,
	Integer _PlaySoundTicketId
) {
	declare K_Private_State State = _State;

	if (!State.AudioSources.existskey(_SoundFileName)) {
		State.AudioSources[_SoundFileName] = [];
	}
	State.AudioSources[_SoundFileName].add(K_Private_ActiveAudioSource {
		AudioSourceId = _AudioSource.Id,
		LifetimeExpiration = Private_GetNewLifetimeExpiration(_AudioSource),
		PlaySoundTicketId = _PlaySoundTicketId
	});

	return State;
}

/// Create the CAudioSource for a list of sound from the library
K_Private_State Private_PreloadSounds(K_Private_State _State, CAudioManager _AudioMgr, Text[] _SoundIds) {
	declare K_Private_State State = _State;

	for (SoundId in _SoundIds) {
		declare K_Private_AudioItem AudioItem = State.Library.get(SoundId, C_Private_AudioItem_Null);
		for (Sound in AudioItem.Sounds) {
			if (State.AudioSources.get(Sound.FileName, []).count == 0) {
				declare CAudioSource AudioSource <=> _AudioMgr.CreateSound(
					Sound.FileName,
					C_Private_DefaultVolumedB,
					AudioItem.IsMusic,
					AudioItem.IsLooping,
					AudioItem.IsSpatialized
				);
				State = Private_AddAudioSource(State, Sound.FileName, AudioSource, C_Private_PlaySoundTicketId_Preload);
			}
		}
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Stop active AudioSource and delayed sounds
K_Private_State Private_StopSounds(K_Private_State _State, CAudioManager _AudioMgr, Integer[] _PlaySoundTicketIds) {
	declare K_Private_State State = _State;

	for (SoundFileName => ActiveAudioSources in State.AudioSources) {
		for (ActiveAudioSourceKey => ActiveAudioSource in ActiveAudioSources) {
			if (
				_PlaySoundTicketIds.exists(ActiveAudioSource.PlaySoundTicketId) &&
				_AudioMgr.Sounds.existskey(ActiveAudioSource.AudioSourceId)
			) {
				if (_AudioMgr.Sounds[ActiveAudioSource.AudioSourceId].IsPlaying) {
					_AudioMgr.Sounds[ActiveAudioSource.AudioSourceId].Stop();
					State.AudioSources[SoundFileName][ActiveAudioSourceKey].LifetimeExpiration = Private_GetNewLifetimeExpiration(Null);
					State.AudioSources[SoundFileName][ActiveAudioSourceKey].PlaySoundTicketId = C_Private_PlaySoundTicketId_Stop;
				}
			}
		}
	}

	declare Integer Key = 0;
	while (Key < State.DelayedSounds.count) {
		if (_PlaySoundTicketIds.exists(State.DelayedSounds[Key].PlaySoundTicketId)) {
			State.DelayedSounds.removekey(Key);
		} else {
			Key += 1;
		}
	}

	return State;
}
K_Private_State Private_StopSound(K_Private_State _State, CAudioManager _AudioMgr, Integer _PlaySoundTicketId) {
	return Private_StopSounds(_State, _AudioMgr, [_PlaySoundTicketId]);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Remove expired AudioSources
K_Private_ActiveAudioSource[][Text] Private_CleanUpExpiredAudioSources(
	K_Private_ActiveAudioSource[][Text] _AudioSources,
	CAudioManager _AudioMgr
) {
	declare K_Private_ActiveAudioSource[][Text] AudioSources;

	for (SoundFileName => ActiveAudioSources in _AudioSources) {
		for (ActiveAudioSource in ActiveAudioSources) {
			if (_AudioMgr.Sounds.existskey(ActiveAudioSource.AudioSourceId)) {
				if (
					ActiveAudioSource.LifetimeExpiration >= TiL::GetMillisecondsSinceInit() ||
					AudioSources.get(SoundFileName, []).count <= 0 //< Always keep at least one copy of a file AudioSource
				) {
					if (AudioSources.existskey(SoundFileName)) {
						AudioSources[SoundFileName].add(ActiveAudioSource);
					} else {
						AudioSources[SoundFileName] = [ActiveAudioSource];
					}
				} else {
					_AudioMgr.DestroySound(_AudioMgr.Sounds[ActiveAudioSource.AudioSourceId]);
				}
			}
		}
	}

	return AudioSources;
}

/// Get a value from a sound property array
Real Private_GetSoundProperty_Real(Real[] _PropertyValues) {
	if (_PropertyValues.count >= 3) {
		return _PropertyValues[ML::Rand(0, _PropertyValues.count - 1)];
	} else if (_PropertyValues.count == 2) {
		return ML::Rand(_PropertyValues[0], _PropertyValues[1]);
	} else if (_PropertyValues.count == 1) {
		return _PropertyValues[0];
	}
	return C_Private_DefaultSoundProperty_Real;
}
Integer Private_GetSoundProperty_Integer(Integer[] _PropertyValues) {
	if (_PropertyValues.count >= 3) {
		return _PropertyValues[ML::Rand(0, _PropertyValues.count - 1)];
	} else if (_PropertyValues.count == 2) {
		return ML::Rand(_PropertyValues[0], _PropertyValues[1]);
	} else if (_PropertyValues.count == 1) {
		return _PropertyValues[0];
	}
	return C_Private_DefaultSoundProperty_Integer;
}
Real Private_GetSoundProperty_Duration(Integer[] _PropertyValues) {
	if (_PropertyValues.count >= 3) {
		return _PropertyValues[ML::Rand(0, _PropertyValues.count - 1)] / 1000.;
	} else if (_PropertyValues.count == 2) {
		return ML::Rand(_PropertyValues[0], _PropertyValues[1]) / 1000.;
	} else if (_PropertyValues.count == 1) {
		return _PropertyValues[0] / 1000.;
	}
	return C_Private_DefaultSoundProperty_Duration;
}
Vec3 Private_GetSoundProperty_Vec3(Vec3[] _PropertyValues) {
	if (_PropertyValues.count >= 3) {
		return _PropertyValues[ML::Rand(0, _PropertyValues.count - 1)];
	} else if (_PropertyValues.count == 2) {
		return <
			ML::Rand(_PropertyValues[0].X, _PropertyValues[1].X),
			ML::Rand(_PropertyValues[0].Y, _PropertyValues[1].Y),
			ML::Rand(_PropertyValues[0].Z, _PropertyValues[1].Z)
		>;
	} else if (_PropertyValues.count == 1) {
		return _PropertyValues[0];
	}
	return C_Private_DefaultSoundProperty_Vec3;
}

/// Apply sound properties on an AudioSource
Void Private_ApplySoundProperties(CAudioSource _AudioSource, K_Private_Sound _SoundSource, K_SoundOptions _SoundOverride) {
	if (_SoundOverride.VolumedB.count > 0) {
		_AudioSource.VolumedB = Private_GetSoundProperty_Real(_SoundOverride.VolumedB);
	} else if (_SoundSource.VolumedB.count > 0) {
		_AudioSource.VolumedB = Private_GetSoundProperty_Real(_SoundSource.VolumedB);
	}
	if (_SoundOverride.Pitch.count > 0) {
		_AudioSource.Pitch = Private_GetSoundProperty_Real(_SoundOverride.Pitch);
	} else if (_SoundSource.Pitch.count > 0) {
		_AudioSource.Pitch = Private_GetSoundProperty_Real(_SoundSource.Pitch);
	}
	if (_SoundOverride.FadeDuration.count > 0) {
		_AudioSource.FadeDuration = Private_GetSoundProperty_Duration(_SoundOverride.FadeDuration);
	} else if (_SoundSource.FadeDuration.count > 0) {
		_AudioSource.FadeDuration = Private_GetSoundProperty_Duration(_SoundSource.FadeDuration);
	}
	if (_SoundOverride.RelativePosition.count > 0) {
		_AudioSource.RelativePosition = Private_GetSoundProperty_Vec3(_SoundOverride.RelativePosition);
	} else if (_SoundSource.RelativePosition.count > 0) {
		_AudioSource.RelativePosition = Private_GetSoundProperty_Vec3(_SoundSource.RelativePosition);
	}
	if (_SoundOverride.PanRadiusLfe.count > 0) {
		_AudioSource.PanRadiusLfe = Private_GetSoundProperty_Vec3(_SoundOverride.PanRadiusLfe);
	} else if (_SoundSource.PanRadiusLfe.count > 0) {
		_AudioSource.PanRadiusLfe = Private_GetSoundProperty_Vec3(_SoundSource.PanRadiusLfe);
	}
}

/// Play a sound from the library
K_Private_PlaySoundResult Private_StartSound(
	K_Private_State _State,
	CAudioManager _AudioMgr,
	Integer _Now,
	Text _SoundId,
	Integer _ForcePlaySoundTicketId,
	K_SoundOptions _SoundOptions
) {
	declare K_Private_PlaySoundResult Result = K_Private_PlaySoundResult {
		State = _State,
		PlaySoundTicketId = C_Private_PlaySoundTicketId_Error
	};

	// Skip silence sound
	if (_SoundId == C_SoundId_Silence) return Result;

	if (_SoundOptions.Delay.count > 0) {
		Result.State.PlaySoundTicketIdGenerator += 1;
		Result.PlaySoundTicketId = Result.State.PlaySoundTicketIdGenerator;

		declare K_SoundOptions SoundOptions = _SoundOptions;
		SoundOptions.Delay = [];
		Result.State.DelayedSounds.add(K_Private_DelayedSound {
			SoundId = _SoundId,
			SoundOptions = SoundOptions,
			PlaySoundTicketId = Result.PlaySoundTicketId,
			PlayTime = _Now + Private_GetSoundProperty_Integer(_SoundOptions.Delay)
		});
	} else {
		declare K_Private_AudioItem AudioItem = Result.State.Library.get(_SoundId, C_Private_AudioItem_Null);

		if (AudioItem.Sounds.count > 0) {
			if (_ForcePlaySoundTicketId == C_Private_UsePlaySoundTicketIdGenerator) {
				Result.State.PlaySoundTicketIdGenerator += 1;
				Result.PlaySoundTicketId = Result.State.PlaySoundTicketIdGenerator;
			} else {
				Result.PlaySoundTicketId = _ForcePlaySoundTicketId;
			}

			declare K_Private_Sound SoundToPlay;
			if (AudioItem.Sounds.count == 1) {
				SoundToPlay = AudioItem.Sounds[0];
			} else {
				declare Integer SoundKey = ML::Rand(0, AudioItem.Sounds.count - 1);
				if (SoundKey == AudioItem.LastPlayedSoundKey) {
					SoundKey = (SoundKey + 1) % AudioItem.Sounds.count;
				}
				SoundToPlay = AudioItem.Sounds[SoundKey];
				AudioItem.LastPlayedSoundKey = SoundKey;
				Result.State.Library[_SoundId] = AudioItem;
			}

			// Try to find an available AudioSource
			declare CAudioSource AvailableAudioSource;
			declare K_Private_ActiveAudioSource[] ActiveAudioSources = Result.State.AudioSources.get(SoundToPlay.FileName, []);
			for (ActiveAudioSourceKey => ActiveAudioSource in ActiveAudioSources) {
				if (
					_AudioMgr.Sounds.existskey(ActiveAudioSource.AudioSourceId) &&
					!_AudioMgr.Sounds[ActiveAudioSource.AudioSourceId].IsPlaying
				) {
					AvailableAudioSource <=> _AudioMgr.Sounds[ActiveAudioSource.AudioSourceId];
					Result.State.AudioSources[SoundToPlay.FileName][ActiveAudioSourceKey].LifetimeExpiration = Private_GetNewLifetimeExpiration(AvailableAudioSource);
					Result.State.AudioSources[SoundToPlay.FileName][ActiveAudioSourceKey].PlaySoundTicketId = Result.PlaySoundTicketId;
					break;
				}
			}

			// We could not find an available AudioSource, create a new one
			if (AvailableAudioSource == Null) {
				declare CAudioSource AudioSource <=> _AudioMgr.CreateSound(
					SoundToPlay.FileName,
					C_Private_DefaultVolumedB,
					AudioItem.IsMusic,
					AudioItem.IsLooping,
					AudioItem.IsSpatialized
				);
				if (AudioSource != Null) {
					AvailableAudioSource <=> AudioSource;
					Result.State = Private_AddAudioSource(Result.State, SoundToPlay.FileName, AvailableAudioSource, Result.PlaySoundTicketId);
				}
			}

			// Play the AudioSource
			if (AvailableAudioSource != Null) {
				Private_ApplySoundProperties(AvailableAudioSource, SoundToPlay, _SoundOptions);
				AvailableAudioSource.Play();
				// PlayCursor must set after playing the sound
				if (_SoundOptions.PlayCursor.count > 0) {
					AvailableAudioSource.PlayCursor = Private_GetSoundProperty_Duration(_SoundOptions.PlayCursor);
				} else if (SoundToPlay.PlayCursor.count > 0) {
					AvailableAudioSource.PlayCursor = Private_GetSoundProperty_Duration(SoundToPlay.PlayCursor);
				} else {
					AvailableAudioSource.PlayCursor = 0.;
				}
			}

			// Clean up the audio sources
			if (Result.State.AudioSourcesCleanUpTime <= TiL::GetMillisecondsSinceInit()) {
				Result.State.AudioSourcesCleanUpTime = TiL::GetMillisecondsSinceInit() + C_Private_ActiveAudioSourceLifetime;
				Result.State.AudioSources = Private_CleanUpExpiredAudioSources(Result.State.AudioSources, _AudioMgr);
			}
		}
	}

	return Result;
}

/// Play the delayed sounds
K_Private_State Private_PlayDelayedSounds(K_Private_State _State, CAudioManager _AudioMgr, Integer _Now) {
	declare K_Private_State State = _State;

	declare Integer Key = 0;
	while (Key < State.DelayedSounds.count) {
		if (State.DelayedSounds[Key].PlayTime <= _Now) {
			declare K_Private_PlaySoundResult Result = Private_StartSound(
				State,
				_AudioMgr,
				_Now,
				State.DelayedSounds[Key].SoundId,
				State.DelayedSounds[Key].PlaySoundTicketId,
				State.DelayedSounds[Key].SoundOptions
			);
			State = Result.State;
			State.DelayedSounds.removekey(Key);
		} else {
			Key += 1;
		}
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the sound balance
K_Private_State Private_SetSoundBalance(
	K_Private_State _State,
	CAudioManager _AudioMgr,
	K_SoundBalance _NewSoundBalance,
	Integer _TransitionStartTime,
	Integer _TransitionDuration
) {
	declare K_Private_State State = _State;

	if (_TransitionDuration > 0) {
		declare K_SoundBalance BaseSoundBalance = K_SoundBalance {
			Music = ML::Max(_AudioMgr.LimitMusicVolumedB, C_SoundBalanceVolumedBMin),
			SceneSound = ML::Max(_AudioMgr.LimitSceneSoundVolumedB, C_SoundBalanceVolumedBMin),
			UiSound = ML::Max(_AudioMgr.LimitUiSoundVolumedB, C_SoundBalanceVolumedBMin)
		};
		State.SoundBalanceTransition = K_Private_SoundBalanceTransition {
			StartTime = _TransitionStartTime,
			Duration = _TransitionDuration,
			BaseSoundBalance = BaseSoundBalance,
			DeltaSoundBalance = K_SoundBalance {
				Music = ML::Max(_NewSoundBalance.Music, C_SoundBalanceVolumedBMin) - BaseSoundBalance.Music,
				SceneSound = ML::Max(_NewSoundBalance.SceneSound, C_SoundBalanceVolumedBMin) - BaseSoundBalance.SceneSound,
				UiSound = ML::Max(_NewSoundBalance.UiSound, C_SoundBalanceVolumedBMin) - BaseSoundBalance.UiSound
			}
		};
	} else {
		State.SoundBalanceTransition = C_Private_SoundBalanceTransition_Null;
		_AudioMgr.LimitMusicVolumedB = ML::Max(_NewSoundBalance.Music, C_SoundBalanceVolumedBMin);
		_AudioMgr.LimitSceneSoundVolumedB = ML::Max(_NewSoundBalance.SceneSound, C_SoundBalanceVolumedBMin);
		_AudioMgr.LimitUiSoundVolumedB = ML::Max(_NewSoundBalance.UiSound, C_SoundBalanceVolumedBMin);
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the sound balance transition
K_Private_State Private_UpdateSoundBalanceTransition(
	K_Private_State _State,
	CAudioManager _AudioMgr,
	Integer _Now
) {
	declare K_Private_State State = _State;

	if (Now >= State.SoundBalanceTransition.StartTime) {
		if (_Now >= State.SoundBalanceTransition.StartTime + State.SoundBalanceTransition.Duration) {
			_AudioMgr.LimitMusicVolumedB = State.SoundBalanceTransition.BaseSoundBalance.Music + State.SoundBalanceTransition.DeltaSoundBalance.Music;
			_AudioMgr.LimitSceneSoundVolumedB = State.SoundBalanceTransition.BaseSoundBalance.SceneSound + State.SoundBalanceTransition.DeltaSoundBalance.SceneSound;
			_AudioMgr.LimitUiSoundVolumedB = State.SoundBalanceTransition.BaseSoundBalance.UiSound + State.SoundBalanceTransition.DeltaSoundBalance.UiSound;
			State.SoundBalanceTransition = C_Private_SoundBalanceTransition_Null;
		} else {
			_AudioMgr.LimitMusicVolumedB = AL::EaseLinear(
				_Now - State.SoundBalanceTransition.StartTime,
				State.SoundBalanceTransition.BaseSoundBalance.Music,
				State.SoundBalanceTransition.DeltaSoundBalance.Music,
				State.SoundBalanceTransition.Duration
			);
			_AudioMgr.LimitSceneSoundVolumedB = AL::EaseLinear(
				_Now - State.SoundBalanceTransition.StartTime,
				State.SoundBalanceTransition.BaseSoundBalance.SceneSound,
				State.SoundBalanceTransition.DeltaSoundBalance.SceneSound,
				State.SoundBalanceTransition.Duration
			);
			_AudioMgr.LimitUiSoundVolumedB = AL::EaseLinear(
				_Now - State.SoundBalanceTransition.StartTime,
				State.SoundBalanceTransition.BaseSoundBalance.UiSound,
				State.SoundBalanceTransition.DeltaSoundBalance.UiSound,
				State.SoundBalanceTransition.Duration
			);
		}
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize a new state
K_Private_State Private_InitializeState(K_Private_State _State, CManiaApp _Context, CAudioManager _AudioMgr) {
	// Cleanup data from a previous execution before initializing the state
	for (LibraryLoadingRequest in _State.LibraryLoadingRequests) {
		Http::Destroy(_Context, LibraryLoadingRequest.Request);
	}
	for (ActiveAudioSources in _State.AudioSources) {
		for (ActiveAudioSource in ActiveAudioSources) {
			if (_AudioMgr.Sounds.existskey(ActiveAudioSource.AudioSourceId)) {
				_AudioMgr.DestroySound(_AudioMgr.Sounds[ActiveAudioSource.AudioSourceId]);
			}
		}
	}

	return C_Private_State_Null;
}

/// Update the library
Void AfterYield(CManiaApp _Context) {
	declare K_Private_State CMGame_MenusAudio_State for _Context = C_Private_State_Null;

	if (CMGame_MenusAudio_State.LibraryLoadingRequests.count > 0) {
		CMGame_MenusAudio_State = Private_UpdateLibraryLoadingRequests(CMGame_MenusAudio_State, _Context);
	}

	if (CMGame_MenusAudio_State.DelayedSounds.count > 0) {
		CMGame_MenusAudio_State = Private_PlayDelayedSounds(CMGame_MenusAudio_State, _Context.Audio, Now);
	}

	if (CMGame_MenusAudio_State.SoundBalanceTransition.StartTime >= 0) {
		CMGame_MenusAudio_State = Private_UpdateSoundBalanceTransition(CMGame_MenusAudio_State, _Context.Audio, Now);
	}
}
