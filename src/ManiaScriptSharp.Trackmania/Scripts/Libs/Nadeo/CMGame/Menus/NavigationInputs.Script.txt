/// Convert an ::EMenuNavAction enum value to a C_InputType_XXX constant
Integer Private_ConvertNavActionToInputType(CManiaAppEvent::EMenuNavAction _NavAction) {
	switch (_NavAction) {
		case CManiaAppEvent::EMenuNavAction::Up: return C_InputType_Up;
		case CManiaAppEvent::EMenuNavAction::Right: return C_InputType_Right;
		case CManiaAppEvent::EMenuNavAction::Left: return C_InputType_Left;
		case CManiaAppEvent::EMenuNavAction::Down: return C_InputType_Down;
		case CManiaAppEvent::EMenuNavAction::Select: return C_InputType_Select;
		case CManiaAppEvent::EMenuNavAction::Cancel: return C_InputType_Cancel;
		case CManiaAppEvent::EMenuNavAction::PageUp: return C_InputType_PageUp;
		case CManiaAppEvent::EMenuNavAction::PageDown: return C_InputType_PageDown;
		case CManiaAppEvent::EMenuNavAction::AppMenu: return C_InputType_AppMenu;
		case CManiaAppEvent::EMenuNavAction::Action1: return C_InputType_Action1;
		case CManiaAppEvent::EMenuNavAction::Action2: return C_InputType_Action2;
		case CManiaAppEvent::EMenuNavAction::Action3: return C_InputType_Action3;
		case CManiaAppEvent::EMenuNavAction::Action4: return C_InputType_Action4;
		case CManiaAppEvent::EMenuNavAction::ScrollUp: return C_InputType_ScrollUp;
		case CManiaAppEvent::EMenuNavAction::ScrollDown: return C_InputType_ScrollDown;
	}
	return C_InputType_Null;
}

		case CMlScriptEvent::EMenuNavAction::Up: return C_InputType_Up;

		case CMlScriptEvent::EMenuNavAction::AppMenu: return C_InputType_AppMenu;

		case CMlScriptEvent::EMenuNavAction::Action2: return C_InputType_Action2;

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the nav action was generated by the mouse
Boolean Private_NavActionIsFromMouse(CManiaAppEvent::EMenuNavAction _NavAction) {
	return (
		_NavAction == CManiaAppEvent::EMenuNavAction::ScrollUp ||
		_NavAction == CManiaAppEvent::EMenuNavAction::ScrollDown
	);
}
Boolean Private_NavActionIsFromMouse(CMlScriptEvent::EMenuNavAction _NavAction) {
	return (
		_NavAction == CMlScriptEvent::EMenuNavAction::ScrollUp ||
		_NavAction == CMlScriptEvent::EMenuNavAction::ScrollDown
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the mouse right-click is used as a cancel key
Boolean Private_IsEnabledMouseRightClickForCancel(CMlScript _Context) {
	declare Integer CMGame_NavigationInputs_IsEnabledMouseRightClickForCancel as IsEnabled_MlScript for _Context = C_Private_Null;
	declare Integer IsEnabled_Local = IsEnabled_MlScript;
	if (IsEnabled_Local == C_Private_Null) {
		declare CManiaApp ContextParentApp <=> Context::GetParentApp(_Context);
		if (ContextParentApp != Null) {
			declare Integer CMGame_NavigationInputs_IsEnabledMouseRightClickForCancel as IsEnabled_ManiaApp for ContextParentApp = C_Private_Null;
			IsEnabled_Local = IsEnabled_ManiaApp;
		}
	}
	return (IsEnabled_Local == C_Private_True);
}

/// Get the inputs generated by a mania app
K_Input[] Private_GetManiaAppInputList(CManiaApp _Context, K_Input[] _InputList) {
	if (_Context == Null || !_Context.EnableMenuNavigationInputs) return _InputList;

	declare K_Input[] InputList = _InputList;

	declare CManiaAppEvent[] MenuNavigationEvents;
	switchtype (_Context as Context) {
		case CManiaAppTitle: {
			for (Event in Context.PendingEvents) {
				if (Event.Type == CManiaAppEvent::EType::MenuNavigation) {
					MenuNavigationEvents.add(Event);
				}
			}
		}
		case CManiaAppPlaygroundCommon: {
			for (Event in Context.PendingEvents) {
				if (Event.Type == CManiaAppEvent::EType::MenuNavigation) {
					MenuNavigationEvents.add(Event);
				}
			}
		}
		case CManiaplanetPlugin: {
			for (Event in Context.PendingEvents) {
				if (Event.Type == CManiaAppEvent::EType::MenuNavigation) {
					MenuNavigationEvents.add(Event);
				}
			}
		}
		case CAnyEditorPlugin: {
			for (Event in Context.PendingEvents) {
				if (Event.Type == CManiaAppEvent::EType::MenuNavigation) {
					MenuNavigationEvents.add(Event);
				}
			}
		}
		case CManiaAppBase: {
			for (Event in Context.PendingEvents) {
				if (Event.Type == CManiaAppEvent::EType::MenuNavigation) {
					MenuNavigationEvents.add(Event);
				}
			}
		}
		case CManiaAppBrowser: {
			for (Event in Context.PendingEvents) {
				if (Event.Type == CManiaAppEvent::EType::MenuNavigation) {
					MenuNavigationEvents.add(Event);
				}
			}
		}
	}

	for (Event in MenuNavigationEvents) {
		InputList.add(K_Input {
			Type = Private_ConvertNavActionToInputType(Event.MenuNavAction),
			IsFromMouse = Private_NavActionIsFromMouse(Event.MenuNavAction),
			IsActionAutoRepeat = Event.IsActionAutoRepeat,
			Control = Null
		});
	}

	return InputList;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the inputs generated by a manialink
K_Input[] Private_GetManialinkInputList(
	CMlScript _Context,
	K_Input[] _InputList,
	Boolean _ParentAppMenuNavigationEnabled
) {
	if (
		_Context == Null || (
			!_ParentAppMenuNavigationEnabled &&
			!_Context.EnableMenuNavigationInputs
		)
	) {
		return _InputList;
	}

	declare K_Input[] InputList = _InputList;

	for (Event in _Context.PendingEvents) {
		switch (Event.Type) {
			case CMlScriptEvent::Type::MenuNavigation: {
				InputList.add(K_Input {
					Type = Private_ConvertNavActionToInputType(Event.MenuNavAction),
					IsFromMouse = Private_NavActionIsFromMouse(Event.MenuNavAction),
					IsActionAutoRepeat = Event.IsActionAutoRepeat,
					Control = Null
				});
			}
			case CMlScriptEvent::Type::MouseClick: {
				// Create a local copy of the control instead of relying on the event reference
				// The reference is no longer valid anymore once the event is flushed (after a yield)
				// and would cause a runtime error if you tried to read it
				declare CMlControl Control = Event.Control;
				InputList.add(K_Input {
					Type = C_InputType_MouseClick,
					IsFromMouse = True,
					IsActionAutoRepeat = Event.IsActionAutoRepeat,
					Control = Control
				});
			}
			case CMlScriptEvent::Type::MouseRightClick: {
				if (!Private_IsEnabledMouseRightClickForCancel(_Context)) {
					declare CMlControl Control = Event.Control;
					InputList.add(K_Input {
						Type = C_InputType_MouseRightClick,
						IsFromMouse = True,
						IsActionAutoRepeat = Event.IsActionAutoRepeat,
						Control = Control
					});
				}
			}
			case CMlScriptEvent::Type::MouseOver: {
				declare CMlControl Control = Event.Control;
				InputList.add(K_Input {
					Type = C_InputType_MouseOver,
					IsFromMouse = True,
					IsActionAutoRepeat = Event.IsActionAutoRepeat,
					Control = Control
				});
			}
			case CMlScriptEvent::Type::MouseOut: {
				declare CMlControl Control = Event.Control;
				InputList.add(K_Input {
					Type = C_InputType_MouseOut,
					IsFromMouse = True,
					IsActionAutoRepeat = Event.IsActionAutoRepeat,
					Control = Control
				});
			}
		}
	}

	// The `CMlScriptEvent::Type::MouseRightClick` event can be captured by one specific page when the mouse cursor in on a control with `scriptevents="1"`
	// If we want to detect the mouse right-click in all pages to use it as a cancel key, we must check `CMlScript->MouseRightButton` instead
	// The other thing to take into account is that the `Private_GetManialinkInputList()` function can be called several times from different places during the same frame
	// It must return the `C_InputType_Cancel` for each call and not only the first one
	declare Boolean CMGame_NavigationInputs_PreviousMouseRightButton for _Context.Page = False;
	declare Integer CMGame_NavigationInputs_CanGenerateEventForNow for _Context.Page = -1;
	if (
		(
			(_Context.MouseRightButton && !CMGame_NavigationInputs_PreviousMouseRightButton) ||
			CMGame_NavigationInputs_CanGenerateEventForNow == Now
		) &&
		Private_IsEnabledMouseRightClickForCancel(_Context)
	) {
		CMGame_NavigationInputs_CanGenerateEventForNow = Now;
		InputList.add(K_Input {
			Type = C_InputType_Cancel,
			IsFromMouse = True,
			IsActionAutoRepeat = False,
			Control = Null
		});
	}
	CMGame_NavigationInputs_PreviousMouseRightButton = _Context.MouseRightButton;

	return InputList;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the inputs generated during the last frame
K_Input[] GetInputList(CManiaApp _Context) {
	return Private_GetManiaAppInputList(_Context, []);
}
K_Input[] GetInputList(CMlScript _Context) {
	declare K_Input[] InputList;
	declare CManiaApp ContextParentApp <=> Context::GetParentApp(_Context);
	declare Boolean ParentAppMenuNavigationEnabled = False;

	if (ContextParentApp != Null) {
		InputList = Private_GetManiaAppInputList(ContextParentApp, InputList);
		ParentAppMenuNavigationEnabled = ContextParentApp.EnableMenuNavigationInputs;
	}

	InputList = Private_GetManialinkInputList(_Context, InputList, ParentAppMenuNavigationEnabled);

	return InputList;
}
K_Input[] GetInputs(CManiaApp _Context) {
	return GetInputList(_Context);
}
K_Input[] GetInputs(CMlScript _Context) {
	return GetInputList(_Context);
}

/// Enable or disable the use of the right mouse click as a cancel key
Void EnableMouseRightClickForCancel(CManiaApp _Context, Boolean _Enabled) {
	declare Integer CMGame_NavigationInputs_IsEnabledMouseRightClickForCancel for _Context = C_Private_Null;
	if (_Enabled) {
		CMGame_NavigationInputs_IsEnabledMouseRightClickForCancel = C_Private_True;
	} else {
		CMGame_NavigationInputs_IsEnabledMouseRightClickForCancel = C_Private_False;
	}
}
Void EnableMouseRightClickForCancel(CMlScript _Context, Boolean _Enabled) {
	declare CManiaApp ContextParentApp <=> Context::GetParentApp(_Context);
	if (ContextParentApp != Null) EnableMouseRightClickForCancel(ContextParentApp, _Enabled);
	declare Integer CMGame_NavigationInputs_IsEnabledMouseRightClickForCancel for _Context = C_Private_Null;
	if (_Enabled) {
		CMGame_NavigationInputs_IsEnabledMouseRightClickForCancel = C_Private_True;
	} else {
		CMGame_NavigationInputs_IsEnabledMouseRightClickForCancel = C_Private_False;
	}
}

/// Enable or disable the menu navigation in the mania app
Boolean EnableNavigationInputs(CManiaApp _Context, Text _RequesterId, Boolean _Enabled) {
	if (_Context == Null) return False;

	declare Text[] CMGame_NavigationInputs_MenuNavigationRequesterIds for _Context = [];
	if (_Enabled) {
		if (!CMGame_NavigationInputs_MenuNavigationRequesterIds.exists(_RequesterId)) {
			CMGame_NavigationInputs_MenuNavigationRequesterIds.add(_RequesterId);
		}
	} else {
		CMGame_NavigationInputs_MenuNavigationRequesterIds.remove(_RequesterId);
	}

	// As long as there is at least one requester the menu navigation is enabled
	_Context.EnableMenuNavigationInputs = CMGame_NavigationInputs_MenuNavigationRequesterIds.count > 0;

	return _Context.EnableMenuNavigationInputs;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Enable or disable the menu navigation in the manialink
Boolean EnableNavigationInputs(CMlScript _Context, Text _RequesterId, Boolean _Enabled, Integer _Priority) {
	if (_Context == Null) return False;

	declare Integer[Text] CMGame_NavigationInputs_MenuNavigationRequesters for _Context.Page = [];
	if (_Enabled) {
		CMGame_NavigationInputs_MenuNavigationRequesters[_RequesterId] = _Priority;
	} else {
		CMGame_NavigationInputs_MenuNavigationRequesters.removekey(_RequesterId);
	}

	declare Integer HighestPriority = 0;
	for (RequesterPriority in CMGame_NavigationInputs_MenuNavigationRequesters) {
		if (RequesterPriority > HighestPriority) {
			HighestPriority = RequesterPriority;
		}
	}

	// As long as there is at least one requester, enable the menu navigation
	// The page with the highest priority number will generate the menu navigation events
	// (eg: 2 is a highest priority than 1)
	// If several pages share the highest priority, only one will generate events
	// and it will be the one that called `EnableMenuNavigation()` first.
	_Context.EnableMenuNavigation(
		CMGame_NavigationInputs_MenuNavigationRequesters.count > 0,
		False,
		False,
		Null,
		HighestPriority
	);

	return _Context.EnableMenuNavigationInputs;
}
Boolean EnableNavigationInputs(CMlScript _Context, Text _RequesterId, Boolean _Enabled) {
	return EnableNavigationInputs(_Context, _RequesterId, _Enabled, 0);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize the library in the mania app
Void Initialize(CManiaApp _Context) {
	if (_Context == Null) return;

	declare Text[] CMGame_NavigationInputs_MenuNavigationRequesterIds for _Context = [];
	CMGame_NavigationInputs_MenuNavigationRequesterIds = [];
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize the library in the manialink
Void Initialize(CMlScript _Context) {
	if (_Context == Null) return;

	declare Integer[Text] CMGame_NavigationInputs_MenuNavigationRequesters for _Context.Page = [];
	CMGame_NavigationInputs_MenuNavigationRequesters = [];
}
