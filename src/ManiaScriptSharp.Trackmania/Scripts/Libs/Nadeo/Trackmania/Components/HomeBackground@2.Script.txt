/// Delete the transition background video when we don't need it anymore
K_Private_Component Private_DeleteBackgroundVideoTransition(CMlScript _Context, K_Private_Component _Component) {
	if (
		_Component.BackgroundVideoTransitionState != C_Private_TransitionState_Complete ||
		_Component.BackgroundVideoTransitionId == NullId
	) {
		return _Component;
	}

	declare K_Private_Component Component = _Component;

	if (_Context.Video.Videos.existskey(Component.BackgroundVideoTransitionId)) {
		_Context.Video.DestroyVideo(_Context.Video.Videos[Component.BackgroundVideoTransitionId]);
	}
	Component.BackgroundVideoTransitionId = NullId;
	Component.Controls.Quad_BackgroundVideoTransition.Image = Null;

	return Component;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Play/Pause the background video
Void Private_SetBackgroundVideoIsPlaying(CMlScript _Context, K_Private_Component _Component, Boolean _IsPlaying) {
	if (_Component.BackgroundVideoActiveId != NullId && _Context.Video.Videos.existskey(_Component.BackgroundVideoActiveId)) {
		if (_IsPlaying && _Component.Controls.Quad_BackgroundVideoActive.Visible) {
			_Context.Video.Videos[_Component.BackgroundVideoActiveId].Play();
		} else {
			_Context.Video.Videos[_Component.BackgroundVideoActiveId].Pause();
		}
	}
	if (_Component.BackgroundVideoTransitionId != NullId && _Context.Video.Videos.existskey(_Component.BackgroundVideoTransitionId)) {
		if (_IsPlaying && _Component.Controls.Quad_BackgroundVideoTransition.Visible) {
			_Context.Video.Videos[_Component.BackgroundVideoTransitionId].Play();
		} else {
			_Context.Video.Videos[_Component.BackgroundVideoTransitionId].Pause();
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the camera scene is valid
Boolean Private_IsCameraSceneValid(CMlScript _Context, K_Private_Component _Component) {
	return (
		_Component.CameraScene.SceneId != NullId &&
		!MenuSceneManager::PageNeedToRecreateScene(_Context.Page, _Context.LocalUser)
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Select which video to use for the camera scene cube map
Void Private_UpdateCameraSceneCubeMap(CMlScript _Context, K_Private_Component _Component) {
	if (!Private_IsCameraSceneValid(_Context, _Component)) return;

	if (
		_Component.BackgroundVideoTransitionState == C_Private_TransitionState_Download &&
		_Component.BackgroundVideoTransitionId != NullId &&
		_Context.Video.Videos.existskey(_Component.BackgroundVideoTransitionId)
	) {
		_Context.MenuSceneMgr.CubeMapSetFromEquirect(
			_Component.CameraScene.SceneId,
			_Component.Controls.Quad_BackgroundVideoTransition
		);
	} else if (
		_Component.BackgroundVideoActiveId != NullId &&
		_Context.Video.Videos.existskey(_Component.BackgroundVideoActiveId)
	) {
		_Context.MenuSceneMgr.CubeMapSetFromEquirect(
			_Component.CameraScene.SceneId,
			_Component.Controls.Quad_BackgroundVideoActive
		);
	}
}

/// Set the state of background video transition
K_Private_Component Private_SetBackgroundVideoTransitionState(
	CMlScript _Context,
	K_Private_Component _Component,
	Integer _BackgroundVideoTransitionState,
	Boolean _IsInit
) {
	if (!_IsInit && _Component.BackgroundVideoTransitionState == _BackgroundVideoTransitionState) return _Component;

	declare K_Private_Component Component = _Component;

	Component.BackgroundVideoTransitionState = _BackgroundVideoTransitionState;

	_Context.AnimMgr.Flush(Component.Controls.Quad_BackgroundVideoActive);
	switch (Component.BackgroundVideoTransitionState) {
		case C_Private_TransitionState_Download: {
			Component.Controls.Quad_BackgroundVideoTransition.Visible = True;
			Component.Controls.Quad_BackgroundVideoActive.Visible = True;
			Component.Controls.Quad_BackgroundVideoActive.Opacity = 0.;
			Component.BackgroundVideoTransitionEndTime = -1;
		}
		case C_Private_TransitionState_Play: {
			Component.Controls.Quad_BackgroundVideoTransition.Visible = True;
			Component.Controls.Quad_BackgroundVideoActive.Visible = True;
			Component.Controls.Quad_BackgroundVideoActive.Opacity = 0.;
			_Context.AnimMgr.Add(Component.Controls.Quad_BackgroundVideoActive, """<anim opacity="1" />""", C_Private_BackgroundVideoTransitionDuration, CAnimManager::EAnimManagerEasing::Linear);
			Component.BackgroundVideoTransitionEndTime = Now + C_Private_BackgroundVideoTransitionDuration;
		}
		case C_Private_TransitionState_Complete: {
			Component.Controls.Quad_BackgroundVideoTransition.Visible = False;
			Component.Controls.Quad_BackgroundVideoActive.Visible = True;
			Component.Controls.Quad_BackgroundVideoActive.Opacity = 1.;
			Component.BackgroundVideoTransitionEndTime = -1;
		}
	}

	Component = Private_DeleteBackgroundVideoTransition(_Context, Component);
	Component = Private_UpdateCameraSceneCubeMapNextFrame(Component);
	Private_SetBackgroundVideoIsPlaying(_Context, Component, Component.IsEnabled);

	return Component;
}
K_Private_Component Private_SetBackgroundVideoTransitionState(CMlScript _Context, K_Private_Component _Component, Integer _BackgroundVideoState) {
	return Private_SetBackgroundVideoTransitionState(_Context, _Component, _BackgroundVideoState, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the state of background video transition
K_Private_Component Private_UpdateBackgroundVideoTransitionState(CMlScript _Context, K_Private_Component _Component) {
	if (_Component.BackgroundVideoTransitionState == C_Private_TransitionState_Complete) return _Component;

	declare K_Private_Component Component = _Component;

	switch (Component.BackgroundVideoTransitionState) {
		case C_Private_TransitionState_Download: {
			if (
				Component.BackgroundVideoActiveId == NullId ||
				!_Context.Video.Videos.existskey(Component.BackgroundVideoActiveId) ||
				!_Context.Video.Videos[Component.BackgroundVideoActiveId].DownloadInProgress
			) {
				Component = Private_SetBackgroundVideoTransitionState(_Context, Component, C_Private_TransitionState_Play);
			}
		}
		case C_Private_TransitionState_Play: {
			if (
				Component.BackgroundVideoTransitionEndTime < 0 ||
				Component.BackgroundVideoTransitionEndTime <= Now
			) {
				Component = Private_SetBackgroundVideoTransitionState(_Context, Component, C_Private_TransitionState_Complete);
			}
		}
	}

	return Component;
}

/// Set the video displayed in the background
K_Private_Component Private_SetBackgroundVideo(CMlScript _Context, K_Private_Component _Component, Text _BackgroundVideoActiveUrl, Boolean _IsInit) {
	if (!_IsInit && _Component.BackgroundVideoActiveUrl == _BackgroundVideoActiveUrl) return _Component;

	declare K_Private_Component Component = _Component;

	// Destroy previous transition video
	if (Component.BackgroundVideoTransitionId != NullId && _Context.Video.Videos.existskey(Component.BackgroundVideoTransitionId)) {
		_Context.Video.DestroyVideo(_Context.Video.Videos[Component.BackgroundVideoTransitionId]);
	}
	// Move active video to transition video
	Component.BackgroundVideoTransitionId = Component.BackgroundVideoActiveId;
	if (Component.BackgroundVideoTransitionId != NullId && _Context.Video.Videos.existskey(Component.BackgroundVideoTransitionId)) {
		Component.Controls.Quad_BackgroundVideoTransition.Image = _Context.Video.Videos[Component.BackgroundVideoTransitionId].Image;
		Component.Controls.Quad_BackgroundVideoTransition.Visible = True;
	} else {
		Component.Controls.Quad_BackgroundVideoTransition.Image = Null;
		Component.Controls.Quad_BackgroundVideoTransition.Visible = False;
	}

	// Create new active video
	Component.BackgroundVideoActiveUrl = _BackgroundVideoActiveUrl;
	declare CVideo ActiveVideo;
	if (Component.BackgroundVideoActiveUrl != "" && _Context.Http.IsValidUrl(Component.BackgroundVideoActiveUrl)) {
		ActiveVideo <=> _Context.Video.CreateVideo(Component.BackgroundVideoActiveUrl, True, True);
	}
	if (ActiveVideo == Null) {
		Component.BackgroundVideoActiveId = NullId;
		Component.Controls.Quad_BackgroundVideoActive.Image = Null;
	} else {
		Component.BackgroundVideoActiveId = ActiveVideo.Id;
		Component.Controls.Quad_BackgroundVideoActive.Image = ActiveVideo.Image;
	}

	declare Boolean Trackmania_ComponentsHomeBackground2_SkipBackgroundVideoTransition for _Context.System = False;
	if (_IsInit) {
		Component = Private_SetBackgroundVideoTransitionState(_Context, Component, C_Private_TransitionState_Complete);
	} else if (
		Component.BackgroundVideoActiveId != NullId &&
		_Context.Video.Videos.existskey(Component.BackgroundVideoActiveId) &&
		_Context.Video.Videos[Component.BackgroundVideoActiveId].DownloadInProgress
	) {
		Component = Private_SetBackgroundVideoTransitionState(_Context, Component, C_Private_TransitionState_Download);
	} else if (Trackmania_ComponentsHomeBackground2_SkipBackgroundVideoTransition) {
		Component = Private_SetBackgroundVideoTransitionState(_Context, Component, C_Private_TransitionState_Complete);
	} else {
		Component = Private_SetBackgroundVideoTransitionState(_Context, Component, C_Private_TransitionState_Play);
	}
	Trackmania_ComponentsHomeBackground2_SkipBackgroundVideoTransition = False;

	Component = Private_DeleteBackgroundVideoTransition(_Context, Component);
	Private_SetBackgroundVideoIsPlaying(_Context, Component, Component.IsEnabled);

	return Private_UpdateBackgroundVideoTransitionState(_Context, Component);
}
K_Private_Component Private_SetBackgroundVideo(CMlScript _Context, K_Private_Component _Component, Text _BackgroundVideoActiveUrl) {
	return Private_SetBackgroundVideo(_Context, _Component, _BackgroundVideoActiveUrl, False);
}

/// Set the state of foreground image transition
K_Private_Component Private_SetForegroundImageTransitionState(
	CMlScript _Context,
	K_Private_Component _Component,
	Integer _ForegroundImageTransitionState,
	Boolean _IsInit
) {
	if (!_IsInit && _Component.ForegroundImageTransitionState == _ForegroundImageTransitionState) return _Component;

	declare K_Private_Component Component = _Component;

	Component.ForegroundImageTransitionState = _ForegroundImageTransitionState;

	_Context.AnimMgr.Flush(Component.Controls.Quad_ForegroundImageActive);
	switch (Component.ForegroundImageTransitionState) {
		case C_Private_TransitionState_Download: {
			Component.Controls.Quad_ForegroundImageTransition.Visible = True;
			Component.Controls.Quad_ForegroundImageActive.Visible = True;
			Component.Controls.Quad_ForegroundImageActive.Opacity = 0.;
			Component.ForegroundImageTransitionEndTime = -1;
		}
		case C_Private_TransitionState_Play: {
			Component.Controls.Quad_ForegroundImageTransition.Visible = True;
			Component.Controls.Quad_ForegroundImageActive.Visible = True;
			Component.Controls.Quad_ForegroundImageActive.Opacity = 0.;
			_Context.AnimMgr.Add(Component.Controls.Quad_ForegroundImageActive, """<anim opacity="1" />""", C_Private_ForegroundImageTransitionDuration / 2, CAnimManager::EAnimManagerEasing::Linear);
			_Context.AnimMgr.Add(Component.Controls.Quad_ForegroundImageTransition, """<anim opacity="0" />""", Now + (C_Private_ForegroundImageTransitionDuration / 2), C_Private_ForegroundImageTransitionDuration / 2, CAnimManager::EAnimManagerEasing::Linear);
			Component.ForegroundImageTransitionEndTime = Now + C_Private_ForegroundImageTransitionDuration;
		}
		case C_Private_TransitionState_Complete: {
			Component.Controls.Quad_ForegroundImageTransition.Visible = False;
			Component.Controls.Quad_ForegroundImageActive.Visible = True;
			Component.Controls.Quad_ForegroundImageActive.Opacity = 1.;
			Component.ForegroundImageTransitionEndTime = -1;
		}
	}

	Component = Private_DeleteForegroundImageTransition(_Context, Component);

	return Component;
}
K_Private_Component Private_SetForegroundImageTransitionState(CMlScript _Context, K_Private_Component _Component, Integer _ForegroundImageState) {
	return Private_SetForegroundImageTransitionState(_Context, _Component, _ForegroundImageState, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the state of foreground image transition
K_Private_Component Private_UpdateForegroundImageTransitionState(CMlScript _Context, K_Private_Component _Component) {
	if (_Component.ForegroundImageTransitionState == C_Private_TransitionState_Complete) return _Component;

	declare K_Private_Component Component = _Component;

	switch (Component.ForegroundImageTransitionState) {
		case C_Private_TransitionState_Download: {
			if (!_Component.Controls.Quad_ForegroundImageActive.DownloadInProgress) {
				Component = Private_SetForegroundImageTransitionState(_Context, Component, C_Private_TransitionState_Play);
			}
		}
		case C_Private_TransitionState_Play: {
			if (
				Component.ForegroundImageTransitionEndTime < 0 ||
				Component.ForegroundImageTransitionEndTime <= Now
			) {
				Component = Private_SetForegroundImageTransitionState(_Context, Component, C_Private_TransitionState_Complete);
			}
		}
	}

	return Component;
}

/// Set the image displayed in the foreground
K_Private_Component Private_SetForegroundImage(CMlScript _Context, K_Private_Component _Component, Text _ForegroundImageActiveUrl, Boolean _IsInit) {
	if (!_IsInit && _Component.ForegroundImageActiveUrl == _ForegroundImageActiveUrl) return _Component;

	declare K_Private_Component Component = _Component;

	// Move active image to transition image
	Component.Controls.Quad_ForegroundImageTransition.ImageUrl = Component.Controls.Quad_ForegroundImageActive.ImageUrl;
	Component.Controls.Quad_ForegroundImageTransition.Visible = (Component.Controls.Quad_ForegroundImageTransition.ImageUrl != "");

	// Set new active image
	Component.ForegroundImageActiveUrl = _ForegroundImageActiveUrl;
	Component.Controls.Quad_ForegroundImageActive.ImageUrl = Component.ForegroundImageActiveUrl;

	declare Boolean Trackmania_ComponentsHomeBackground2_SkipForegroundImageTransition for _Context.System = False;
	if (_IsInit) {
		Component = Private_SetForegroundImageTransitionState(_Context, Component, C_Private_TransitionState_Complete);
	} else if (Component.Controls.Quad_ForegroundImageActive.DownloadInProgress) {
		Component = Private_SetForegroundImageTransitionState(_Context, Component, C_Private_TransitionState_Download);
	} else if (Trackmania_ComponentsHomeBackground2_SkipForegroundImageTransition) {
		Component = Private_SetForegroundImageTransitionState(_Context, Component, C_Private_TransitionState_Complete);
	} else {
		Component = Private_SetForegroundImageTransitionState(_Context, Component, C_Private_TransitionState_Play);
	}
	Trackmania_ComponentsHomeBackground2_SkipForegroundImageTransition = False;

	Component = Private_DeleteForegroundImageTransition(_Context, Component);

	return Private_UpdateForegroundImageTransitionState(_Context, Component);
}
K_Private_Component Private_SetForegroundImage(CMlScript _Context, K_Private_Component _Component, Text _ForegroundImageActiveUrl) {

/// Select if the default or special car is displayed
K_Private_Component Private_SetIsSpecialCarDisplayed(CMlScript _Context, K_Private_Component _Component, Boolean _IsSpecialCarDisplayed) {
	if (!Private_IsCameraSceneValid(_Context, _Component)) return _Component;

	declare K_Private_Component Component = _Component;

	Component.CameraScene.IsSpecialCarDisplayed = _IsSpecialCarDisplayed;

	if (Component.CameraScene.IsSpecialCarDisplayed && Component.CameraScene.SpecialCarItemId != NullId) {
		Component.CameraScene.CurrentCarItemId = Component.CameraScene.SpecialCarItemId;
		switch (Component.CameraScene.SpecialCarModelName) {
			case C_Private_RallyCarModelName: Component.CarPosition = C_Private_CarPosition_RallyCar;
			case C_Private_SnowCarModelName: Component.CarPosition = C_Private_CarPosition_SnowCar;
			case C_Private_DesertCarModelName: Component.CarPosition = C_Private_CarPosition_DesertCar;
			default: Component.CarPosition = C_Private_CarPosition_SportCar;
		}
	} else {
		Component.CameraScene.CurrentCarItemId = Component.CameraScene.DefaultCarItemId;
		Component.CarPosition = C_Private_CarPosition_SportCar;
	}

	if (Component.CameraScene.DefaultCarItemId != NullId) {
		_Context.MenuSceneMgr.ItemSetVisible(
			Component.CameraScene.SceneId,
			Component.CameraScene.DefaultCarItemId,
			Component.CameraScene.DefaultCarItemId == Component.CameraScene.CurrentCarItemId
		);
	}
	if (Component.CameraScene.SpecialCarItemId != NullId) {
		_Context.MenuSceneMgr.ItemSetVisible(
			Component.CameraScene.SceneId,
			Component.CameraScene.SpecialCarItemId,
			Component.CameraScene.SpecialCarItemId == Component.CameraScene.CurrentCarItemId
		);
	}

	return Component;
}

/// Set the car steer direction from the player's inputs
K_Private_Component Private_SetCarSteerDirection(K_Private_Component _Component, CMlScriptEvent::EMenuNavAction _MenuNavAction) {
	if (
		(_Component.CarSteerDirection == C_Private_CarSteerDirection_Left && _MenuNavAction == CMlScriptEvent::EMenuNavAction::Left) ||
		(_Component.CarSteerDirection == C_Private_CarSteerDirection_Right && _MenuNavAction == CMlScriptEvent::EMenuNavAction::Right)
	) {
		return _Component;
	}

	declare K_Private_Component Component = _Component;

	switch (Component.CarSteerDirection) {
		case C_Private_CarSteerDirection_Left: {
			if (_MenuNavAction == CMlScriptEvent::EMenuNavAction::Right) {
				Component = Private_SetCarSteerDirection(Component, C_Private_CarSteerDirection_Neutral);
			}
		}
		case C_Private_CarSteerDirection_Neutral: {
			if (_MenuNavAction == CMlScriptEvent::EMenuNavAction::Right) {
				Component = Private_SetCarSteerDirection(Component, C_Private_CarSteerDirection_Right);
			} else if (_MenuNavAction == CMlScriptEvent::EMenuNavAction::Left) {
				Component = Private_SetCarSteerDirection(Component, C_Private_CarSteerDirection_Left);
			}
		}
		case C_Private_CarSteerDirection_Right: {
			if (_MenuNavAction == CMlScriptEvent::EMenuNavAction::Left) {
				Component = Private_SetCarSteerDirection(Component, C_Private_CarSteerDirection_Neutral);
			}
		}
	}

	return Component;
}

/// Update the car steer animation
K_Private_Component Private_UpdateCarSteerAnim(K_Private_Component _Component) {
	if (_Component.CarSteerAnim.StartTime < 0) return _Component;

	declare K_Private_Component Component = _Component;

	if (Component.CarSteerAnim.EndTime <= Now) {
		Component.CarSteerAnim.StartTime = -1;
		Component.CarSteerAnim.CurrentValue = Component.CarSteerAnim.TargetValue;
		Component.IsWaitingCarStateUpdate = True;
	} else {
		Component.CarSteerAnim.CurrentValue = AL::EaseInOutQuad(
			Now - Component.CarSteerAnim.StartTime,
			Component.CarSteerAnim.StartValue,
			Component.CarSteerAnim.DeltaValue,
			Component.CarSteerAnim.Duration
		);
		Component.IsWaitingCarStateUpdate = True;
	}

	return Component;
}

/// Set the car boost
K_Private_Component Private_SetCarBoost(K_Private_Component _Component, Integer _CarBoost, Boolean _IsInit) {
	if (!_IsInit && _Component.CarBoost == _CarBoost) return _Component;

	declare K_Private_Component Component = _Component;

	Component.CarBoost = _CarBoost;
	if (Component.CarBoost == C_Private_CarBoost_TransformUp) {
		Component.CarBoostTransformCooldownEndTime = Now + C_Private_CarBoostTransformCooldownDuration;
	}
	Component.CarBoostHoverCooldownEndTime = Now + C_Private_CarBoostHoverCooldownDuration;

	switch (Component.CarBoost) {
		case C_Private_CarBoost_HoverUp: {
			Component.CarBoostAnim = Private_CreateAnim(
				Component.CarBoostAnim.CurrentValue,
				C_Private_CarBoostHoverPosY,
				C_Private_CarBoostHoverAnimDuration
			);
		}
		case C_Private_CarBoost_HoverDown: {
			Component.CarBoostAnim = Private_CreateAnim(
				Component.CarBoostAnim.CurrentValue,
				C_Private_CarBoostOffPosY,
				C_Private_CarBoostHoverAnimDuration
			);
		}
		case C_Private_CarBoost_HoverLoop: {
			Component.CarBoostAnim = Private_CreateAnim(
				Component.CarBoostAnim.CurrentValue,
				C_Private_CarBoostHoverPosY,
				0
			);
		}
		case C_Private_CarBoost_TransformUp: {
			Component.CarBoostAnim = Private_CreateAnim(
				Component.CarBoostAnim.CurrentValue,
				C_Private_CarBoostTransformPosY,
				C_Private_CarBoostTransformAnimDuration
			);
		}
		case C_Private_CarBoost_TransformDown: {
			Component.CarBoostAnim = Private_CreateAnim(
				Component.CarBoostAnim.CurrentValue,
				C_Private_CarBoostOffPosY,
				C_Private_CarBoostTransformAnimDuration
			);
		}
	}

	return Component;
}
K_Private_Component Private_SetCarBoost(K_Private_Component _Component, Integer _CarBoost) {
	return Private_SetCarBoost(_Component, _CarBoost, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the car boost from the player's inputs
K_Private_Component Private_SetCarBoost(K_Private_Component _Component, CMlScriptEvent::EMenuNavAction _MenuNavAction) {
	if (_Component.CarBoostHoverCooldownEndTime > Now ||
		_Component.CarBoost == C_Private_CarBoost_TransformUp ||
		_Component.CarBoost == C_Private_CarBoost_TransformDown ||
		_Component.CarBoost == C_Private_CarBoost_HoverUp ||
		_Component.CarBoost == C_Private_CarBoost_HoverDown ||
		(_Component.CarBoost == C_Private_CarBoost_Off && _MenuNavAction == CMlScriptEvent::EMenuNavAction::Down)
	) {
		return _Component;
	}

	declare K_Private_Component Component = _Component;

	switch (Component.CarBoost) {
		case C_Private_CarBoost_Off: {
			if (_MenuNavAction == CMlScriptEvent::EMenuNavAction::Up) {
				if (
					C_Private_CarBoostTransformProbability > 0. &&
					Component.CarBoostTransformCooldownEndTime <= Now &&
					ML::Rand(0., 100.) <= C_Private_CarBoostTransformProbability
				) {
					Component = Private_SetCarBoost(Component, C_Private_CarBoost_TransformUp);
				} else {
					Component = Private_SetCarBoost(Component, C_Private_CarBoost_HoverUp);
				}
			}
		}
		case C_Private_CarBoost_HoverLoop: {
			if (_MenuNavAction == CMlScriptEvent::EMenuNavAction::Down) {
				Component = Private_SetCarBoost(Component, C_Private_CarBoost_HoverDown);
			} else if (
				_MenuNavAction == CMlScriptEvent::EMenuNavAction::Up &&
				C_Private_CarBoostTransformProbability > 0. &&
				Component.CarBoostTransformCooldownEndTime <= Now &&
				ML::Rand(0., 100.) <= C_Private_CarBoostTransformProbability
			) {
				Component = Private_SetCarBoost(Component, C_Private_CarBoost_TransformUp);
			}
		}
	}

	return Component;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the car boost animation
K_Private_Component Private_UpdateCarBoostAnim(CMlScript _Context, K_Private_Component _Component) {
	if (_Component.CarBoostAnim.StartTime < 0) return _Component;

	declare K_Private_Component Component = _Component;

	switch (Component.CarBoost) {
		case C_Private_CarBoost_HoverUp, C_Private_CarBoost_TransformUp: {
			if (Component.CarBoostAnim.EndTime <= Now) {
				Component.CarBoostAnim.StartTime = -1;
				Component.CarBoostAnim.CurrentValue = Component.CarBoostAnim.TargetValue;
				Component.IsWaitingCarStateUpdate = True;
				if (Component.CarBoost == C_Private_CarBoost_HoverUp) {
					Component = Private_SetCarBoost(Component, C_Private_CarBoost_HoverLoop);
				} else {
					Component = Private_SetIsSpecialCarDisplayed(_Context, Component, !Component.CameraScene.IsSpecialCarDisplayed);
					Component = Private_SetCarBoost(Component, C_Private_CarBoost_TransformDown);
				}
			} else {
				Component.CarBoostAnim.CurrentValue = AL::EaseInOutQuad(
					Now - Component.CarBoostAnim.StartTime,
					Component.CarBoostAnim.StartValue,
					Component.CarBoostAnim.DeltaValue,
					Component.CarBoostAnim.Duration
				);
				Component.IsWaitingCarStateUpdate = True;
			}
		}
		case C_Private_CarBoost_HoverDown: {
			if (Component.CarBoostAnim.EndTime <= Now) {
				Component.CarBoostAnim.StartTime = -1;
				Component.CarBoostAnim.CurrentValue = Component.CarBoostAnim.TargetValue;
				Component.IsWaitingCarStateUpdate = True;
				Component = Private_SetCarBoost(Component, C_Private_CarBoost_Off);
			} else {
				Component.CarBoostAnim.CurrentValue = AL::EaseOutBounce(
					Now - Component.CarBoostAnim.StartTime,
					Component.CarBoostAnim.StartValue,
					Component.CarBoostAnim.DeltaValue,
					Component.CarBoostAnim.Duration
				);
				Component.IsWaitingCarStateUpdate = True;
			}
		}
		case C_Private_CarBoost_HoverLoop: {
			Component.CarBoostAnim.CurrentValue = C_Private_CarBoostHoverPosY + (C_Private_CarBoostHoverAnimAmplitude * (ML::Sin((Now - Component.CarBoostAnim.StartTime) * C_Private_CarBoostHoverAnimFrequency)));
			Component.IsWaitingCarStateUpdate = True;
		}
		case C_Private_CarBoost_TransformDown: {
			if (Component.CarBoostAnim.EndTime <= Now) {
				Component.CarBoostAnim.StartTime = -1;
				Component.CarBoostAnim.CurrentValue = Component.CarBoostAnim.TargetValue;
				Component.IsWaitingCarStateUpdate = True;
				Component = Private_SetCarBoost(Component, C_Private_CarBoost_Off);
			} else {
				Component.CarBoostAnim.CurrentValue = AL::EaseOutQuart(
					Now - Component.CarBoostAnim.StartTime,
					Component.CarBoostAnim.StartValue,
					Component.CarBoostAnim.DeltaValue,
					Component.CarBoostAnim.Duration
				);
				Component.IsWaitingCarStateUpdate = True;
			}
		}
	}

	return Component;
}

/// Play an emote animation on the pilot
K_Private_Component Private_PlayPilotEmote(CMlScript _Context, K_Private_Component _Component) {
	if (
		!Private_IsCameraSceneValid(_Context, _Component) ||
		_Component.CameraScene.PilotItemId == NullId ||
		_Component.PilotEmoteCooldownEndTime > Now
	) {
		return _Component;
	}

	declare K_Private_Component Component = _Component;
	Component.PilotEmoteCooldownEndTime = Now + C_Private_PilotEmoteCooldownDuration;
	_Context.MenuSceneMgr.ItemTriggerPlayerGesture(Component.CameraScene.SceneId, Component.CameraScene.PilotItemId);
	return Component;
}

	Integer _MenuMediaUpdate,
	CampaignStore::K_MediaActive _BackgroundVideo,
	CampaignStore::K_MediaActive _ForegroundImage,

) {

		Component = Private_SetForegroundImage(_Context, Component, _ForegroundImage.Url);

/// `MainInit` section of the component
Void Private_MainInit(CMlScript _Context) {
	declare CMlFrame[] Trackmania_ComponentsHomeBackground2_HomeBackgroundList for _Context.Page = [];
	Trackmania_ComponentsHomeBackground2_HomeBackgroundList = [];
	_Context.Page.GetClassChildren(C_Private_Class_ComponentContainer, _Context.Page.MainFrame, True);
	foreach (Control in _Context.Page.GetClassChildren_Result) {
		declare CMlFrame Frame_ComponentContainer = (Control as CMlFrame);
		Trackmania_ComponentsHomeBackground2_HomeBackgroundList.add(Frame_ComponentContainer.Parent);
		Private_InitComponentInstance(_Context, Frame_ComponentContainer.Parent);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// `MainLoop` section of the component
Void Private_MainLoop(CMlScript _Context) {
	if (_Context.PageIsVisible) {
		declare CMlFrame[] Trackmania_ComponentsHomeBackground2_HomeBackgroundList for _Context.Page = [];
		for (Frame_HomeBackground in Trackmania_ComponentsHomeBackground2_HomeBackgroundList) {
			declare K_Private_Component Trackmania_ComponentsHomeBackground2_Component as Component for Frame_HomeBackground = C_Private_Component_Null;

			if (
				Component.CameraScene.SceneId != NullId &&
				MenuSceneManager::PageNeedToRecreateScene(_Context.Page, _Context.LocalUser)
			) {
				declare Integer Trackmania_ComponentsHomeBackground2_CameraSceneReloadCounter for _Context.System = 0;
				Component = Private_ReloadCameraScene(_Context, Component, Trackmania_ComponentsHomeBackground2_CameraSceneReloadCounter);
			}

			if (Component.UpdateCameraSceneCubeMapTime != -1 && Component.UpdateCameraSceneCubeMapTime < Now) {
				Component.UpdateCameraSceneCubeMapTime = -1;
				Private_UpdateCameraSceneCubeMap(_Context, Component);
			}

			if (Component.IsEnabled) {
				declare Integer Trackmania_ComponentsHomeBackground2_CameraSceneReloadCounter for _Context.System = 0;

				if (Component.MenuMediaUpdate != CampaignStore::GetMenuMediaUpdate()) {
					Component = Private_SetMediaActive(_Context, Component, CampaignStore::GetMenuMediaUpdate(), CampaignStore::GetMediaActive(CampaignStore::C_CampaignMediaType_HomeBackground), CampaignStore::GetMediaActive(CampaignStore::C_CampaignMediaType_HomeForeground));
				}
				if (Component.CameraScene.ReloadCounter != Trackmania_ComponentsHomeBackground2_CameraSceneReloadCounter) {
					Component = Private_ReloadCameraScene(_Context, Component, Trackmania_ComponentsHomeBackground2_CameraSceneReloadCounter);
				}

				if (Component.IsInputEnabled) {
					foreach (Event in ShareMenuNavigation::GetPendingEvents(_Context)) {
						if (
							Event.MenuNavAction == CMlScriptEvent::EMenuNavAction::Left ||
							Event.MenuNavAction == CMlScriptEvent::EMenuNavAction::Right
						) {
							Component = Private_SetCarSteerDirection(Component, Event.MenuNavAction);
						} else if (
							Event.MenuNavAction == CMlScriptEvent::EMenuNavAction::Up ||
							Event.MenuNavAction == CMlScriptEvent::EMenuNavAction::Down
						) {
							Component = Private_PlayPilotEmote(_Context, Component);
							Component = Private_SetCarBoost(Component, Event.MenuNavAction);
						}
					}
				}

				Component = Private_UpdateCarSteerAnim(Component);
				Component = Private_UpdateCarBoostAnim(_Context, Component);
				Component = Private_UpdateCarState(_Context, Component);
				Component = Private_UpdateBackgroundVideoTransitionState(_Context, Component);
				Component = Private_UpdateForegroundImageTransitionState(_Context, Component);
			}
		}
	}
}
