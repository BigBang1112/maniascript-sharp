// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the images displayed in the loading screen
Void Private_UpdateLoadingScreenDisplay(
	K_Private_ControlLoadingScreen _ControlLoadingScreen,
	LoadingScreenStore::K_LoadingScreenImage _LoadingScreenImage,
	Text _DefaultMessage
) {
	_ControlLoadingScreen.Quad_Media.ImageUrl = _LoadingScreenImage.MediaUrl;
	_ControlLoadingScreen.Quad_Media.Visible = (_ControlLoadingScreen.Quad_Media.ImageUrl != "");

	if (_LoadingScreenImage.Message == "") {
		_ControlLoadingScreen.Label_Message.Value = _DefaultMessage;
	} else {
		_ControlLoadingScreen.Label_Message.Value = _LoadingScreenImage.Message;
	}
	if (_ControlLoadingScreen.Label_Message.Value != "") {
		if (_ControlLoadingScreen.IsFullscreen) {
			MLHelpers::FitLabelValue(_ControlLoadingScreen.Label_Message, C_Private_TextSize_MessageFullscreen);
		} else {
			MLHelpers::FitLabelValue(_ControlLoadingScreen.Label_Message, C_Private_TextSize_MessageSmallscreen);
		}
	}
	_ControlLoadingScreen.Label_Message.Visible = (_ControlLoadingScreen.Label_Message.Value != "");
	_ControlLoadingScreen.Quad_MessageBackground.Visible = _ControlLoadingScreen.Label_Message.Visible;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the loading screen to display
K_Private_Component Private_SetLoadingScreen(K_Private_Component _Component, Integer _TitleMenuLoadingScreenUpdate, LoadingScreenStore::K_LoadingScreen _LoadingScreen) {
	declare K_Private_Component Component = _Component;

	Component.TitleMenuLoadingScreenUpdate = _TitleMenuLoadingScreenUpdate;

	if (_LoadingScreen.ImageList.count >= 4) {
		for (ControlKey => ControlLoadingScreen in Component.Controls.FourScreens.LoadingScreenList) {
			Private_UpdateLoadingScreenDisplay(
				ControlLoadingScreen,
				_LoadingScreen.ImageList.get(ControlKey, LoadingScreenStore::C_LoadingScreenImage_Null),
				"" //< Do not display the default message in the four screens layout
			);
		}

		Component.Controls.FourScreens.Frame_FourScreens.Visible = True;
		Component.Controls.Fullscreen.Frame_LoadingScreen.Visible = False;
	} else {
		Private_UpdateLoadingScreenDisplay(
			Component.Controls.Fullscreen,
			_LoadingScreen.ImageList.get(0, LoadingScreenStore::C_LoadingScreenImage_Null),
			_LoadingScreen.DefaultMessage
		);

		Component.Controls.FourScreens.Frame_FourScreens.Visible = False;
		Component.Controls.Fullscreen.Frame_LoadingScreen.Visible = True;
	}

	return Component;
}

/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize the component instance
Void Private_InitComponentInstance(CMlScript _Context, CMlFrame _Frame_LoadingScreen) {
	declare K_Private_Component Trackmania_ComponentsLoadingScreen_Component as Component for _Frame_LoadingScreen = C_Private_Component_Null;
	Component = C_Private_Component_Null;

	Component.Controls.Frame_LoadingScreen <=> _Frame_LoadingScreen;
	Component.Controls.Fullscreen.IsFullscreen = True;
	Component.Controls.Fullscreen.Frame_LoadingScreen <=> (Component.Controls.Frame_LoadingScreen.GetFirstChild(C_Private_Frame_Fullscreen) as CMlFrame);
	Component.Controls.Label_TOTD <=> (Component.Controls.Frame_LoadingScreen.GetFirstChild(C_Private_Label_TOTD) as CMlLabel);
	Component.Controls.Fullscreen.Quad_Media <=> (Component.Controls.Frame_LoadingScreen.GetFirstChild(C_Private_Quad_Media) as CMlQuad);
	Component.Controls.Fullscreen.Quad_MessageBackground <=> (Component.Controls.Frame_LoadingScreen.GetFirstChild(C_Private_Quad_MessageBackground) as CMlQuad);
	Component.Controls.Fullscreen.Label_Message <=> (Component.Controls.Frame_LoadingScreen.GetFirstChild(C_Private_Label_Message) as CMlLabel);
	Component.Controls.FourScreens.Frame_FourScreens <=> (Component.Controls.Frame_LoadingScreen.GetFirstChild(C_Private_Frame_FourScreens) as CMlFrame);
	declare CMlFrame Frame_SmallScreenList <=> (Component.Controls.FourScreens.Frame_FourScreens.GetFirstChild(C_Private_Frame_SmallscreenList) as CMlFrame);
	for (Control in Frame_SmallScreenList.Controls) {
		declare CMlFrame Frame_LoadingScreen <=> (Control as CMlFrame);
		Component.Controls.FourScreens.LoadingScreenList.add(K_Private_ControlLoadingScreen {
			IsFullscreen = False,
			Frame_LoadingScreen = Frame_LoadingScreen,
			Quad_Media = (Frame_LoadingScreen.GetFirstChild(C_Private_Quad_Media) as CMlQuad),
			Quad_MessageBackground = (Frame_LoadingScreen.GetFirstChild(C_Private_Quad_MessageBackground) as CMlQuad),
			Label_Message = (Frame_LoadingScreen.GetFirstChild(C_Private_Label_Message) as CMlLabel)
		});
	}

	MLHelpers::FitLabelValue(Component.Controls.Label_TOTD, C_Private_TextSize_TOTD);

	Component = Private_SetLoadingScreen(
		Component,
		LoadingScreenStore::GetCurrentTitleMenuLoadingScreenUpdate(This),
		LoadingScreenStore::GetCurrentTitleMenuLoadingScreen(This)
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// `MainInit` section of the component
Void Private_MainInit(CMlScript _Context) {
	declare CMlFrame[] Trackmania_ComponentsLoadingScreen_LoadingScreenList for _Context.Page = [];
	Trackmania_ComponentsLoadingScreen_LoadingScreenList = [];
	_Context.Page.GetClassChildren(C_Private_Class_ComponentContainer, _Context.Page.MainFrame, True);
	foreach (Control in _Context.Page.GetClassChildren_Result) {
		declare CMlFrame Frame_ComponentContainer = (Control as CMlFrame);
		Trackmania_ComponentsLoadingScreen_LoadingScreenList.add(Frame_ComponentContainer.Parent);
		Private_InitComponentInstance(_Context, Frame_ComponentContainer.Parent);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// `MainLoop` section of the component
Void Private_MainLoop(CMlScript _Context) {
	if (_Context.PageIsVisible) {
		declare CMlFrame[] Trackmania_ComponentsLoadingScreen_LoadingScreenList for _Context.Page = [];
		for (Frame_LoadingScreen in Trackmania_ComponentsLoadingScreen_LoadingScreenList) {
			if (Frame_LoadingScreen.Visible) {
				declare K_Private_Component Trackmania_ComponentsLoadingScreen_Component as Component for Frame_LoadingScreen = C_Private_Component_Null;
				if (Component.TitleMenuLoadingScreenUpdate != LoadingScreenStore::GetCurrentTitleMenuLoadingScreenUpdate(This)) {
					Component = Private_SetLoadingScreen(
						Component,
						LoadingScreenStore::GetCurrentTitleMenuLoadingScreenUpdate(This),
						LoadingScreenStore::GetCurrentTitleMenuLoadingScreen(This)
					);
				}
			}
		}
	}
}
