/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the component instance
K_Private_Component Private_ComponentUpdatePrestigeProgression(K_Private_Component _Component, K_Private_GlobalState _GlobalState) {
	declare K_Private_Component Component = _Component;

	for (PrestigeBlockKey => PrestigeBlock in Component.Controls.PrestigeBlockList) {
		declare K_PrestigeProgression PrestigeProgression = _GlobalState.PrestigeProgressionList.get(PrestigeBlockKey + 1, C_PrestigeProgression_Null);

		if (PrestigeProgression.Type == C_PrestigeType_Null) {
			PrestigeBlock.Frame_PrestigeBlock.Visible = False;
		} else {
			declare Text PrestigeTypeName = "";
			switch (PrestigeProgression.Type) {
				case C_PrestigeType_Bronze: {
					PrestigeBlock.Quad_PrestigeIcon.ImageUrl = MenuConst::C_ImageUrl_Medal_Bronze;
					PrestigeTypeName = C_Private_PrestigeTypeNameList.get(CMode::EMedal::Bronze, "");
				}
				case C_PrestigeType_Silver: {
					PrestigeBlock.Quad_PrestigeIcon.ImageUrl = MenuConst::C_ImageUrl_Medal_Silver;
					PrestigeTypeName = C_Private_PrestigeTypeNameList.get(CMode::EMedal::Silver, "");
				}
				case C_PrestigeType_Gold: {
					PrestigeBlock.Quad_PrestigeIcon.ImageUrl = MenuConst::C_ImageUrl_Medal_Gold;
					PrestigeTypeName = C_Private_PrestigeTypeNameList.get(CMode::EMedal::Gold, "");
				}
				case C_PrestigeType_Author: {
					PrestigeBlock.Quad_PrestigeIcon.ImageUrl = MenuConst::C_ImageUrl_Medal_Nadeo;
					PrestigeTypeName = C_Private_PrestigeTypeNameList.get(CMode::EMedal::Author, "");
				}
				default: {
					PrestigeBlock.Quad_PrestigeIcon.ImageUrl = "";
				}
			}

			declare Real SizeX_PrestigeStatSide = MLHelpers::SkewX(C_Private_Size_PrestigeStatList.Y, Stylesheet::C_Trackmania_SkewAngle);
			declare Real SizeX_PrestigeStat = 0.;
			if (PrestigeProgression.StatNext > 0) {
				SizeX_PrestigeStat = ML::Max(0., C_Private_Size_PrestigeStatList.X - SizeX_PrestigeStatSide - (C_Private_GapX_PrestigeStat * (PrestigeProgression.StatNext - 1))) / PrestigeProgression.StatNext;
			}
			for (PrestigeStatKey => PrestigeStat in PrestigeBlock.PrestigeStatList) {
				if (PrestigeStatKey + 1 <= PrestigeProgression.StatNext) {
					PrestigeStat.Frame_PrestigeStat.RelativePosition_V3.X = (SizeX_PrestigeStat + C_Private_GapX_PrestigeStat) * PrestigeStatKey;
					PrestigeStat.Quad_Left.Size = <SizeX_PrestigeStatSide, C_Private_Size_PrestigeStatList.Y>;
					PrestigeStat.Quad_Right.Size = <SizeX_PrestigeStatSide, C_Private_Size_PrestigeStatList.Y>;
					PrestigeStat.Quad_Center.Size = <ML::Max(0., SizeX_PrestigeStat - SizeX_PrestigeStatSide), C_Private_Size_PrestigeStatList.Y>;
					PrestigeStat.Quad_Center.RelativePosition_V3.X = PrestigeStat.Quad_Left.Size.X;
					PrestigeStat.Quad_Right.RelativePosition_V3.X = PrestigeStat.Quad_Center.RelativePosition_V3.X + PrestigeStat.Quad_Center.Size.X;
					PrestigeStat.Frame_PrestigeStat.Visible = True;

					if (PrestigeStatKey + 1 <= PrestigeProgression.StatCurrent) {
						PrestigeStat.Quad_Left.Colorize = Stylesheet::GetColorRGB(Stylesheet::C_Color_AccentOne);
						PrestigeStat.Quad_Center.BgColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_AccentOne);
						PrestigeStat.Quad_Right.Colorize = Stylesheet::GetColorRGB(Stylesheet::C_Color_AccentOne);
					} else {
						PrestigeStat.Quad_Left.Colorize = Stylesheet::GetColorRGB(Stylesheet::C_Color_AccentThree);
						PrestigeStat.Quad_Center.BgColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_AccentThree);
						PrestigeStat.Quad_Right.Colorize = Stylesheet::GetColorRGB(Stylesheet::C_Color_AccentThree);
					}
				} else {
					PrestigeStat.Frame_PrestigeStat.Visible = False;
				}
			}

			if (C_Private_EnableMedalCounter && PrestigeProgression.IsComplete) {
				PrestigeBlock.Label_PrestigeTotal.Value = TL::Compose(C_Private_Text_NbMedals, ""^PrestigeProgression.StatTotal);
				MLHelpers::FitLabelValue(PrestigeBlock.Label_PrestigeTotal, C_Private_TextSize_PrestigeTotal);
				PrestigeBlock.Label_PrestigeTotal.Visible = True;
			} else {
				PrestigeBlock.Label_PrestigeTotal.Visible = False;
			}

			if (PrestigeProgression.Level > 0) {
				for (PrestigeLevelKey => Quad_PrestigeLevel in PrestigeBlock.PrestigeLevelList) {
					if (PrestigeLevelKey + 1 <= PrestigeProgression.Level) {
						Quad_PrestigeLevel.ImageUrl = C_Private_ImageUrl_PrestigeLevelSeasonOn;
					} else {
						Quad_PrestigeLevel.ImageUrl = C_Private_ImageUrl_PrestigeLevelSeasonOff;
					}
				}
				PrestigeBlock.Frame_PrestigeLevelList.Visible = True;
				PrestigeBlock.Label_PrestigeUnlock.Visible = False;
			} else if (PrestigeTypeName != "") {
				PrestigeBlock.Label_PrestigeUnlock.Value = TL::Compose(C_Private_Text_PrestigeUnlock, PrestigeTypeName);
				MLHelpers::FitLabelValue(PrestigeBlock.Label_PrestigeUnlock, C_Private_TextSize_PrestigeUnlock);
				PrestigeBlock.Label_PrestigeUnlock.Visible = True;
				PrestigeBlock.Frame_PrestigeLevelList.Visible = False;
			} else {
				PrestigeBlock.Label_PrestigeUnlock.Visible = False;
				PrestigeBlock.Frame_PrestigeLevelList.Visible = False;
			}

			PrestigeBlock.Quad_PrestigeIcon.Visible = (PrestigeBlock.Quad_PrestigeIcon.ImageUrl != "");
			PrestigeBlock.Frame_PrestigeBlock.Visible = True;
		}
	}

	Component.PrestigeProgressionUpdate = _GlobalState.PrestigeProgressionUpdate;

	return Component;
}

/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the current season
Text Private_GetSeason() {
	switch (Tools::GetTimestampSeason(TiL::GetCurrent())) {
		case Tools::C_Season_Winter: return "Winter";
		case Tools::C_Season_Spring: return "Spring";
		case Tools::C_Season_Summer: return "Summer";
		case Tools::C_Season_Fall: return "Fall";
	}
	return "Winter";
}

/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Start a task to get the seasonal prestige list
K_Private_GlobalState Private_GetPrestigeList(CMlScript _Context, K_Private_GlobalState _GlobalState, Boolean _ForceUpdate) {
	if (
		!_ForceUpdate && (
			(!C_Private_Debug_CanAlwaysGetPrestige && !_GlobalState.CanGetPrestigeList) ||
			Task::IsInitialized(_GlobalState.Task_GetPrestigeList)
		)
	) {
		return _GlobalState;
	}

	declare K_Private_GlobalState GlobalState = _GlobalState;

	declare Integer Year = Tools::GetYear();
	declare Text Season = Private_GetSeason();
	if (C_Private_Debug_ForcePrestige) {
		Year = C_Private_Debug_PrestigeYear;
		Season = C_Private_Debug_PrestigeSeason;
	}
	GlobalState.Task_GetPrestigeList = Task::DestroyAndCreate(
		GlobalState.Task_GetPrestigeList,
		_Context.UserMgr,
		_Context.UserMgr.Prestige_GetPrestigeList(_Context.UserMgr.MainUser.Id, CUserV2Manager::EPrestigeMode::Season, Year, Season)
	);

	return GlobalState;
}
K_Private_GlobalState Private_GetPrestigeList(CMlScript _Context, K_Private_GlobalState _GlobalState) {
	return Private_GetPrestigeList(_Context, _GlobalState, False);
}

/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the prestige progression with the seasonal list of prestige
K_Private_GlobalState Private_UpdatePrestigeProgressionList(K_Private_GlobalState _GlobalState, CTaskResult_PrestigeList _TaskResult) {
	declare K_Private_GlobalState GlobalState = _GlobalState;

	for (Prestige in _TaskResult.PrestigeList) {
		if (Prestige.CategoryLevel >= C_PrestigeTypeRange.X && Prestige.CategoryLevel <= C_PrestigeTypeRange.Y) {
			// Update existing progression when possible
			declare K_PrestigeProgression PrestigeProgression = GlobalState.PrestigeProgressionList.get(Prestige.CategoryLevel, C_PrestigeProgression_Null);
			PrestigeProgression.Type = Prestige.CategoryLevel;
			PrestigeProgression.Level = Prestige.PrestigeLevel;
			PrestigeProgression.IsComplete = (Prestige.StatCurrentValue > Prestige.StatValueForNextLevel);
			PrestigeProgression.StatTotal = Prestige.StatCurrentValue;
			if (C_Private_EnableMedalCounter && PrestigeProgression.IsComplete) {
				PrestigeProgression.StatNext = 1;
				PrestigeProgression.StatCurrent = 1;
			} else {
				PrestigeProgression.StatNext = ML::Max(1, Prestige.StatValueForNextLevel - Prestige.StatValueForCurrentLevel);
				PrestigeProgression.StatCurrent = ML::Clamp(Prestige.StatCurrentValue - Prestige.StatValueForCurrentLevel, 0, PrestigeProgression.StatNext);
			}
			PrestigeProgression.SkinName = Prestige.RewardDisplayName;
			PrestigeProgression.SkinUrl = Prestige.RewardFileUrl;
			PrestigeProgression.SkinOptions = Prestige.SkinOptions;
			GlobalState.PrestigeProgressionList[Prestige.CategoryLevel] = PrestigeProgression;
		}
	}

	GlobalState.PrestigeProgressionUpdate += 1;

	return GlobalState;
}

/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize the component instance
K_Private_Component Private_ComponentInit(CMlFrame _Frame_PrestigeProgression) {
	declare K_Private_Component Component = C_Private_Component_Null;

	Component.Controls.Frame_PrestigeProgression = _Frame_PrestigeProgression;
	declare CMlFrame Frame_PrestigeBlockList <=> (Component.Controls.Frame_PrestigeProgression.GetFirstChild(C_Private_Id_Frame_PrestigeBlockList) as CMlFrame);
	for (Control in Frame_PrestigeBlockList.Controls) {
		declare CMlFrame Frame_PrestigeBlock <=> (Control as CMlFrame);
		declare K_Private_ControlPrestigeBlock ControlPrestigeBlock = K_Private_ControlPrestigeBlock {
			Frame_PrestigeBlock = Frame_PrestigeBlock,
			Quad_PrestigeIcon = (Frame_PrestigeBlock.GetFirstChild(C_Private_Id_Quad_PrestigeIcon) as CMlQuad),
			Label_PrestigeTotal = (Frame_PrestigeBlock.GetFirstChild(C_Private_Id_Label_PrestigeTotal) as CMlLabel),
			Label_PrestigeUnlock = (Frame_PrestigeBlock.GetFirstChild(C_Private_Id_Label_PrestigeUnlock) as CMlLabel)
		};
		declare CMlFrame Frame_PrestigeStatList <=> (Frame_PrestigeBlock.GetFirstChild(C_Private_Id_Frame_PrestigeStatList) as CMlFrame);
		for (Control in Frame_PrestigeStatList.Controls) {
			declare CMlFrame Frame_PrestigeStat <=> (Control as CMlFrame);
			ControlPrestigeBlock.PrestigeStatList.add(K_Private_ControlPrestigeStat {
				Frame_PrestigeStat = Frame_PrestigeStat,
				Quad_Left = (Frame_PrestigeStat.GetFirstChild(C_Private_Id_Quad_PrestigeStatLeft) as CMlQuad),
				Quad_Center = (Frame_PrestigeStat.GetFirstChild(C_Private_Id_Quad_PrestigeStatCenter) as CMlQuad),
				Quad_Right = (Frame_PrestigeStat.GetFirstChild(C_Private_Id_Quad_PrestigeStatRight) as CMlQuad)
			});
		}
		ControlPrestigeBlock.Frame_PrestigeLevelList <=> (Frame_PrestigeBlock.GetFirstChild(C_Private_Id_Frame_PrestigeLevelList) as CMlFrame);
		for (Control in ControlPrestigeBlock.Frame_PrestigeLevelList.Controls) {
			ControlPrestigeBlock.PrestigeLevelList.add(Control as CMlQuad);
		}
		Component.Controls.PrestigeBlockList.add(ControlPrestigeBlock);
	}
	Component.Controls.Quad_navZone <=> (Component.Controls.Frame_PrestigeProgression.GetFirstChild(C_NavZone) as CMlQuad);

	return Component;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// `MainInit` section of the component
Void Private_MainInit(CMlScript _Context) {
	declare K_Private_GlobalState Trackmania_ComponentsPrestigeProgression_GlobalState as GlobalState for _Context.System = C_Private_GlobalState_Null;
	for (PrestigeType, C_PrestigeTypeRange.X, C_PrestigeTypeRange.Y) {
		if (!GlobalState.PrestigeProgressionList.existskey(PrestigeType)) {
			declare K_PrestigeProgression PrestigeProgressionOffline = C_PrestigeProgression_Null;
			PrestigeProgressionOffline.Type = PrestigeType;
			PrestigeProgressionOffline.StatNext = 1;
			GlobalState.PrestigeProgressionList[PrestigeType] = PrestigeProgressionOffline;
		}
	}

	declare CMlFrame[] Trackmania_ComponentsPrestigeProgression_ComponentControlList for _Context.Page = [];
	Trackmania_ComponentsPrestigeProgression_ComponentControlList = [];
	_Context.Page.GetClassChildren(C_Private_Class_ComponentContainer, _Context.Page.MainFrame, True);
	foreach (Control in _Context.Page.GetClassChildren_Result) {
		declare CMlFrame Frame_ComponentContainer = (Control as CMlFrame);
		Trackmania_ComponentsPrestigeProgression_ComponentControlList.add(Frame_ComponentContainer.Parent);

		declare K_Private_Component Trackmania_ComponentsPrestigeProgression_Component as Component for Frame_ComponentContainer.Parent = C_Private_Component_Null;
		Component = Private_ComponentInit(Frame_ComponentContainer.Parent);
		Component = Private_ComponentUpdatePrestigeProgression(Component, GlobalState);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// `MainLoop` section of the component
Void Private_MainLoop(CMlScript _Context) {
	declare K_Private_GlobalState Trackmania_ComponentsPrestigeProgression_GlobalState as GlobalState for _Context.System = C_Private_GlobalState_Null;

	if (_Context.LocalUser != Null) {
		declare Boolean Trackmania_ComponentsPrestigeProgression_IsPrestigeProgressionDirty for _Context.LocalUser = False;
		if (Trackmania_ComponentsPrestigeProgression_IsPrestigeProgressionDirty) {
			Trackmania_ComponentsPrestigeProgression_IsPrestigeProgressionDirty = False;
			GlobalState = Private_GetPrestigeList(_Context, GlobalState, True);
		}
	}

	if (Task::IsInitialized(GlobalState.Task_GetPrestigeList)) {
		GlobalState.Task_GetPrestigeList = Task::Update(GlobalState.Task_GetPrestigeList);
		if (!Task::IsRunning(GlobalState.Task_GetPrestigeList)) {
			declare CTaskResult_PrestigeList TaskResult <=> Task::GetSourceTask_PrestigeList(GlobalState.Task_GetPrestigeList);
			if (Task::IsSuccess(GlobalState.Task_GetPrestigeList) && TaskResult != Null) {
				GlobalState.CanGetPrestigeList = False;
				GlobalState = Private_UpdatePrestigeProgressionList(GlobalState, TaskResult);
			}
			GlobalState.Task_GetPrestigeList = Task::Destroy(GlobalState.Task_GetPrestigeList);
		}
	}

	declare CMlFrame[] Trackmania_ComponentsPrestigeProgression_ComponentControlList for _Context.Page = [];
	for (ComponentControl in Trackmania_ComponentsPrestigeProgression_ComponentControlList) {
		declare K_Private_Component Trackmania_ComponentsPrestigeProgression_Component as Component for ComponentControl = C_Private_Component_Null;
		if (Component.PrestigeProgressionUpdate != GlobalState.PrestigeProgressionUpdate) {
			Component = Private_ComponentUpdatePrestigeProgression(Component, GlobalState);
		}
	}
}
