/** Add a button widget
 *	The button can have one of three behaviour
 *	- C_ButtonBehaviour_SendEvent: the button will send a {{{P}}}C_EventType_ButtonEvent event containing the _Event parameter
 *	- C_ButtonBehaviour_Search: the button will send a {{{P}}}C_EventType_Search event with all search widgets values
 *	- C_ButtonBehaviour_Reset: the button will reset all search widgets to their default value and send a {{{P}}}C_EventType_Search event
 *	- C_ButtonBehaviour_Close: the button will send a {{{P}}}C_EventType_Close event
 */
Trackmania_Search_K_Widgets AddWidgetButton(Trackmania_Search_K_Widgets _Widgets, Integer _WidgetId, Integer _Behaviour, Text _Event, Text _Value) {
	declare Trackmania_Search_K_Widgets Widgets = _Widgets;
	assert(!Widgets.All.existskey(_WidgetId), "[Search] Widget with id '"^_WidgetId^"' already in the list.");

	Widgets.Buttons[_WidgetId] = Trackmania_Search_K_Widget_Button {
		Value = _Value,
		Behaviour = _Behaviour,
		Event = _Event
	};

	Widgets.All[_WidgetId] = Trackmania_Search_K_Widget {
		Id = _WidgetId,
		Type = C_Widget_Button,
		Label = ""
	};

	return Widgets;
}
Trackmania_Search_K_Widgets AddWidgetButton(Trackmania_Search_K_Widgets _Widgets, Integer _WidgetId, Integer _Behaviour, Text _Value) {
	return AddWidgetButton(_Widgets, _WidgetId, _Behaviour, "", _Value);
}
