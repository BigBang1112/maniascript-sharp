/// Update the car in the scene
K_Private_Component Private_UpdateCameraCar(CMlScript _Context, K_Private_Component _Component) {
	if (_Component.SceneId == NullId) return _Component;

	declare K_Private_Component Component = _Component;

	if (_Context.LocalUser != Null) {
		Component.UserColor = _Context.LocalUser.Color;
	}

	if (Component.CarItemId != NullId) {
		_Context.MenuSceneMgr.ItemDestroy(Component.SceneId, Component.CarItemId);
		Component.CarItemId = NullId;
	}

	Component.CarItemId = _Context.MenuSceneMgr.ItemCreate(Component.SceneId, C_Private_CarModelName, Component.SkinName, Component.SkinUrl, Component.SkinOptions);

	if (Component.CarItemId != NullId) {
		_Context.MenuSceneMgr.ItemSetLocation(Component.SceneId, Component.CarItemId, C_Private_CarPosition, C_Private_CarRotation, C_Private_CarIsTurntable);
		_Context.MenuSceneMgr.ItemSetVehicleState(Component.SceneId, Component.CarItemId, C_Private_CarSteer,	C_Private_CarBrake, C_Private_CarFrontLight, C_Private_CarTurboLevel, C_Private_CarBoostLevel, C_Private_CarBurnoutSmoke);
		if (_Context.LocalUser != Null) {
			_Context.MenuSceneMgr.ItemSetPlayerState(Component.SceneId, Component.CarItemId, _Context.LocalUser);
		}
	}


	return Component;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Create the scene
K_Private_Component Private_CreateCameraScene(CMlScript _Context, K_Private_Component _Component) {
	if (_Component.SceneId != NullId || !_Component.UseSkinViewer) return _Component;

	declare K_Private_Component Component = _Component;

	Component.SceneId = _Context.MenuSceneMgr.SceneCreate(C_Private_SceneLayout);
	Component.Controls.Camera_SkinViewer.Visible = (Component.SceneId != NullId);
	if (_Context.LocalUser != Null) {
		MenuSceneManager::PageSynchronizeScene(_Context.Page, _Context.LocalUser);
	}

	if (Component.SceneId != NullId) {
		_Context.MenuSceneMgr.CameraSetLocation(Component.SceneId, C_Private_CameraPosition, C_Private_CameraRotation, C_Private_CameraFOV);
	}

	return Component;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Destroy the scene
K_Private_Component Private_DestroyCameraScene(CMlScript _Context, K_Private_Component _Component) {
	if (_Component.SceneId == NullId) return _Component;

	declare K_Private_Component Component = _Component;

	if (Component.CarItemId != NullId) {
		_Context.MenuSceneMgr.ItemDestroy(Component.SceneId, Component.CarItemId);
		Component.CarItemId = NullId;
	}
	_Context.MenuSceneMgr.SceneDestroy(Component.SceneId);
	Component.SceneId = NullId;
	Component.Controls.Camera_SkinViewer.Visible = False;

	return Component;
}

/// Get the name of the prestige season
Text Private_GetPrestigeSeasonName(Text _PrestigeSeason) {
	switch (_PrestigeSeason) {
		case C_Private_PrestigeSeason_Spring: return C_Private_Text_SeasonSpring;
		case C_Private_PrestigeSeason_Summer: return C_Private_Text_SeasonSummer;
		case C_Private_PrestigeSeason_Fall: return C_Private_Text_SeasonFall;
		case C_Private_PrestigeSeason_Winter: return C_Private_Text_SeasonWinter;
	}

/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //

	switch (_PrestigeType) {
		case C_PrestigeType_Bronze: return C_Private_PrestigeTypeNameList.get(CMode::EMedal::Bronze, "");

/// Update the prestige progression display
K_Private_Component Private_ComponentUpdatePrestigeProgression(
	CMlScript _Context,
	K_Private_Component _Component,
	K_Private_GlobalState _GlobalState
) {
	declare K_Private_Component Component = _Component;

	Component.PrestigeProgressionUpdate = _GlobalState.PrestigeProgressionUpdate;

	// Find the lowest progressing prestige type (bronze < silver < gold < author)
	declare K_PrestigeProgression LastUnlockedPrestigeProgression;
	declare K_PrestigeProgression NextUnlockedPrestigeProgression;
	for (PrestigeProgression in _GlobalState.PrestigeProgressionList) {
		if (
			PrestigeProgression.LevelCurrent >= 0 &&
			PrestigeProgression.LevelCurrent < PrestigeProgression.LevelMax && (
				LastUnlockedPrestigeProgression.Type == C_PrestigeType_Null ||
				PrestigeProgression.Type < LastUnlockedPrestigeProgression.Type
			)
		) {
			LastUnlockedPrestigeProgression = PrestigeProgression;
		}
	}

	// If all prestiges are complete, find the highest completed one
	if (LastUnlockedPrestigeProgression.Type == C_PrestigeType_Null) {
		for (PrestigeProgression in _GlobalState.PrestigeProgressionList) {
			if (
				PrestigeProgression.LevelCurrent >= PrestigeProgression.LevelMax && (
					LastUnlockedPrestigeProgression.Type == C_PrestigeType_Null ||
					PrestigeProgression.Type > LastUnlockedPrestigeProgression.Type
				)
			) {
				LastUnlockedPrestigeProgression = PrestigeProgression;
			}
		}
	}
	// If the found prestige is locked at level 0, select the previous completed prestige
	else if (LastUnlockedPrestigeProgression.LevelCurrent == 0 && LastUnlockedPrestigeProgression.Type > C_PrestigeType_Bronze) {
		NextUnlockedPrestigeProgression = LastUnlockedPrestigeProgression;
		LastUnlockedPrestigeProgression = K_PrestigeProgression {};
		for (PrestigeProgression in _GlobalState.PrestigeProgressionList) {
			if (
				PrestigeProgression.Type < NextUnlockedPrestigeProgression.Type &&
				PrestigeProgression.LevelCurrent >= PrestigeProgression.LevelMax && (
					LastUnlockedPrestigeProgression.Type == C_PrestigeType_Null ||
					PrestigeProgression.Type > LastUnlockedPrestigeProgression.Type
				)
			) {
				LastUnlockedPrestigeProgression = PrestigeProgression;
			}
		}
	}

	if (NextUnlockedPrestigeProgression.Type == C_PrestigeType_Null) {
		NextUnlockedPrestigeProgression = LastUnlockedPrestigeProgression;
	}

	Component.SkinName = LastUnlockedPrestigeProgression.SkinName;
	Component.SkinUrl = LastUnlockedPrestigeProgression.SkinUrl;
	Component.SkinOptions = LastUnlockedPrestigeProgression.SkinOptions;

	if (LastUnlockedPrestigeProgression.Type == C_PrestigeType_Null) {
		Component.Controls.Quad_Medal.Visible = False;
		Component.Controls.Label_PrestigeSeasonLevel.Visible = False;
		Component.Controls.Frame_PrestigeTypeLevel.Visible = False;
		Component.Controls.Label_PrestigeNextLevel.Visible = False;
		Component.Controls.Quad_NavZone.Visible = False;
	} else {
		Component.Controls.Label_PrestigeSeasonLevel.Value = TL::Compose(
			C_Private_Text_PrestigeLevel,
			Private_GetPrestigeSeasonName(_GlobalState.PrestigeSeason),
			""^_GlobalState.PrestigeYear,
			Private_GetPrestigeTypeName(LastUnlockedPrestigeProgression.Type),
			""^LastUnlockedPrestigeProgression.LevelCurrent
		);
		MLHelpers::FitLabelValue(Component.Controls.Label_PrestigeSeasonLevel, C_Private_PrestigeSeasonLevelTextSize);


		if (NextUnlockedPrestigeProgression.Next > 1) {
			switch (NextUnlockedPrestigeProgression.Type) {
				case C_PrestigeType_Bronze: {
					Component.Controls.Label_PrestigeNextLevel.Value = TL::Compose(C_Private_Text_NextLevelBronzeSeveral, ""^NextUnlockedPrestigeProgression.Next);
				}
				case C_PrestigeType_Silver: {
					Component.Controls.Label_PrestigeNextLevel.Value = TL::Compose(C_Private_Text_NextLevelSilverSeveral, ""^NextUnlockedPrestigeProgression.Next);
				}
				case C_PrestigeType_Gold: {
					Component.Controls.Label_PrestigeNextLevel.Value = TL::Compose(C_Private_Text_NextLevelGoldSeveral, ""^NextUnlockedPrestigeProgression.Next);
				}
				case C_PrestigeType_Author: {
					Component.Controls.Label_PrestigeNextLevel.Value = TL::Compose(C_Private_Text_NextLevelAuthorSeveral, ""^NextUnlockedPrestigeProgression.Next);
				}
				default: {
					Component.Controls.Label_PrestigeNextLevel.Value = "";
				}
			}
		} else if (NextUnlockedPrestigeProgression.Next == 1) {
			switch (NextUnlockedPrestigeProgression.Type) {
				case C_PrestigeType_Bronze: {
					Component.Controls.Label_PrestigeNextLevel.Value = C_Private_Text_NextLevelBronzeOne;
				}
				case C_PrestigeType_Silver: {
					Component.Controls.Label_PrestigeNextLevel.Value = C_Private_Text_NextLevelSilverOne;
				}
				case C_PrestigeType_Gold: {
					Component.Controls.Label_PrestigeNextLevel.Value = C_Private_Text_NextLevelGoldOne;
				}
				case C_PrestigeType_Author: {
					Component.Controls.Label_PrestigeNextLevel.Value = C_Private_Text_NextLevelAuthorOne;
				}
				default: {
					Component.Controls.Label_PrestigeNextLevel.Value = "";
				}
			}
		} else {
			Component.Controls.Label_PrestigeNextLevel.Value = "";
		}
		MLHelpers::FitLabelValue(Component.Controls.Label_PrestigeNextLevel, C_Private_PrestigeNextLevelTextSize);

		// Get the maximum number of level controls to display
		declare Integer LevelControlListCount = 0;
		for (PrestigeProgression in _GlobalState.PrestigeProgressionList) {
			if (LevelControlListCount < PrestigeProgression.LevelMax) {
				LevelControlListCount = PrestigeProgression.LevelMax;
			}
		}
		LevelControlListCount = ML::Clamp(LevelControlListCount, 0, Component.Controls.QuadList_PrestigeTypeLevel.count);

		// Create the level controls layout
		declare Layout::K_Element[] ElementList = [
			Layout::K_Element {
				Id = "RootContainer",
				Direction = Layout::C_Direction_LeftToRight,
				Width = Layout::NewSizingFixed(Component.SizePrestigeLevelList.X),
				Height = Layout::NewSizingFixed(Component.SizePrestigeLevelList.Y),
				ChildGap = C_Private_ContainerChildGap
			}
		];
		for (I, 1, LevelControlListCount) {
			ElementList.add(Layout::K_Element {
				Id = Layout::NewId("PrestigeTypeLevel", I),
				ParentId = "RootContainer",
				Width = Layout::NewSizingGrow(),
				Height = Layout::NewSizingGrow()
			});
		}
		declare Layout::K_Layout ComponentLayout = Layout::NewLayout(Component.SizePrestigeLevelList, "RootContainer", ElementList);

		// Update the level controls properties
		declare Integer Level = 1;
		for (Quad_PrestigeTypeLevel in Component.Controls.QuadList_PrestigeTypeLevel) {
			if (Level < 1 || Level > LastUnlockedPrestigeProgression.LevelMax) {
				Quad_PrestigeTypeLevel.ImageUrl = "";
			} else {
				declare Integer PositionIndex = Level + (LevelControlListCount - ML::Min(LastUnlockedPrestigeProgression.LevelMax, Component.Controls.QuadList_PrestigeTypeLevel.count));
				Quad_PrestigeTypeLevel.Size = Layout::GetSize(ComponentLayout, Layout::NewId("PrestigeTypeLevel", PositionIndex));
				Quad_PrestigeTypeLevel.RelativePosition_V3 = Layout::GetPosition(ComponentLayout, Layout::NewId("PrestigeTypeLevel", PositionIndex)) + <Quad_PrestigeTypeLevel.Size.X, Quad_PrestigeTypeLevel.Size.Y * -0.5>;

				if (Level <= LastUnlockedPrestigeProgression.LevelCurrent) {
					Quad_PrestigeTypeLevel.ImageUrl = C_Private_ImageUrl_PrestigeLevelSeasonOn;
				} else {
					Quad_PrestigeTypeLevel.ImageUrl = C_Private_ImageUrl_PrestigeLevelSeasonOff;
				}
			}
			Quad_PrestigeTypeLevel.Visible = (Quad_PrestigeTypeLevel.ImageUrl != "");
			Level += 1;
		}

		if (Component.UseSkinViewer) {
			Component.Controls.Quad_Medal.ImageUrl = "";
		} else {
			Component.Controls.Quad_Medal.ImageUrl = Private_GetPrestigeTypeMedalImageUrl(LastUnlockedPrestigeProgression.Type);
		}

		Component.Controls.Quad_Medal.Visible = (Component.Controls.Quad_Medal.ImageUrl != "");
		Component.Controls.Label_PrestigeSeasonLevel.Visible = (Component.Controls.Label_PrestigeSeasonLevel.Value != "");
		Component.Controls.Frame_PrestigeTypeLevel.Visible = True;
		Component.Controls.Label_PrestigeNextLevel.Visible = (Component.Controls.Label_PrestigeNextLevel.Value != "");
		Component.Controls.Quad_NavZone.Visible = True;
	}

	Component = Private_UpdateCameraCar(_Context, Component);

	return Component;
}

/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the current season
Text Private_GetSeason() {
	switch (Tools::GetTimestampSeason(TiL::GetCurrent())) {
		case Tools::C_Season_Winter: return C_Private_PrestigeSeason_Winter;
		case Tools::C_Season_Spring: return C_Private_PrestigeSeason_Spring;
		case Tools::C_Season_Summer: return C_Private_PrestigeSeason_Summer;
		case Tools::C_Season_Fall: return C_Private_PrestigeSeason_Fall;
	}
	return "Winter";
}

/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Start a task to get the seasonal prestige list
K_Private_GlobalState Private_GetPrestigeList(CMlScript _Context, K_Private_GlobalState _GlobalState, Boolean _ForceUpdate) {
	declare Integer LiveQuarterlyCampaignId = CampaignStore::GetLiveQuarterlyCampaignId();
	if (
		!_ForceUpdate && (
			_GlobalState.LiveQuarterlyCampaignId == LiveQuarterlyCampaignId ||
			LiveQuarterlyCampaignId == CampaignStruct::C_Campaign_NullId
		) && (
			(!C_Private_Debug_CanAlwaysGetPrestige && !_GlobalState.CanGetPrestigeList) ||
			Task::IsInitialized(_GlobalState.Task_GetPrestigeList)
		)
	) {
		return _GlobalState;
	}

	declare K_Private_GlobalState GlobalState = _GlobalState;

	declare Integer Year = Tools::GetYear();
	declare Text Season = Private_GetSeason();
	if (C_Private_Debug_ForcePrestige) {
		Year = C_Private_Debug_PrestigeYear;
		Season = C_Private_Debug_PrestigeSeason;
	} else if (LiveQuarterlyCampaignId != CampaignStruct::C_Campaign_NullId) {
		declare CampaignStruct::LibCampaignStruct_K_CampaignName CampaignName = CampaignStruct::ParseCampaignName(CampaignStore::GetLiveQuarterlyCampaign().Name);
		if (CampaignName.IsValid) {
			Year = Cast::ToInteger(CampaignName.Year);
			switch (TL::ToLowerCase(CampaignName.Season)) {
				case "winter": Season = C_Private_PrestigeSeason_Winter;
				case "spring": Season = C_Private_PrestigeSeason_Spring;
				case "summer": Season = C_Private_PrestigeSeason_Summer;
				case "fall": Season = C_Private_PrestigeSeason_Fall;
			}
		}
	}
	GlobalState.LiveQuarterlyCampaignId = LiveQuarterlyCampaignId;
	GlobalState.PrestigeYear = Year;
	GlobalState.PrestigeSeason = Season;
	GlobalState.Task_GetPrestigeList = Task::DestroyAndCreate(
		GlobalState.Task_GetPrestigeList,
		_Context.UserMgr,
		_Context.UserMgr.Prestige_GetPrestigeList(_Context.UserMgr.MainUser.Id, CUserV2Manager::EPrestigeMode::Season, Year, Season)
	);

	return GlobalState;
}
K_Private_GlobalState Private_GetPrestigeList(CMlScript _Context, K_Private_GlobalState _GlobalState) {
	return Private_GetPrestigeList(_Context, _GlobalState, False);
}

/// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the prestige progression with the seasonal list of prestige
K_Private_GlobalState Private_UpdatePrestigeProgressionList(K_Private_GlobalState _GlobalState, CTaskResult_PrestigeList _TaskResult) {
	declare K_Private_GlobalState GlobalState = _GlobalState;

	for (Prestige in _TaskResult.PrestigeList) {
		if (Prestige.CategoryLevel >= C_PrestigeTypeRange.X && Prestige.CategoryLevel <= C_PrestigeTypeRange.Y) {
			// Update existing progression when possible
			declare K_PrestigeProgression PrestigeProgression = GlobalState.PrestigeProgressionList.get(Prestige.CategoryLevel, K_PrestigeProgression {});
			PrestigeProgression.Type = Prestige.CategoryLevel;
			PrestigeProgression.LevelCurrent = Prestige.PrestigeLevel;
			PrestigeProgression.LevelMax = Prestige.PrestigeLevelMax;
			PrestigeProgression.Next = ML::Max(0, Prestige.StatValueForNextLevel - Prestige.StatCurrentValue);
			PrestigeProgression.SkinName = Prestige.RewardDisplayName;
			PrestigeProgression.SkinUrl = Prestige.RewardFileUrl;
			PrestigeProgression.SkinOptions = Prestige.SkinOptions;
			GlobalState.PrestigeProgressionList[PrestigeProgression.Type] = PrestigeProgression;
		}
	}

	GlobalState.PrestigeProgressionUpdate += 1;

	return GlobalState;
}

/** Set the visibility of the component
 * This will destroy or create the 3d scene
 */
K_Private_Component Private_SetIsComponentVisible(
	CMlScript _Context,
	K_Private_Component _Component,
	Boolean _IsVisible,
	Boolean _IsInit
) {
	if (!_IsInit && _Component.IsVisible == _IsVisible) return _Component;

	declare K_Private_Component Component = _Component;

	Component.IsVisible = _IsVisible;
	Component.Controls.Frame_PrestigeProgression.Visible = Component.IsVisible;

	if (_IsVisible && Component.UseSkinViewer) {
		Component = Private_CreateCameraScene(_Context, Component);
		Component = Private_UpdateCameraCar(_Context, Component);
	} else {
		Component = Private_DestroyCameraScene(_Context, Component);
	}

	return Component;
}
K_Private_Component Private_SetIsComponentVisible(CMlScript _Context, K_Private_Component _Component, Boolean _IsVisible) {
	return Private_SetIsComponentVisible(_Context, _Component, _IsVisible, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// `MainInit` section of the component
Void Private_MainInit(CMlScript _Context) {
	declare K_Private_GlobalState Trackmania_ComponentsPrestigeProgression2_GlobalState as GlobalState for _Context.System = K_Private_GlobalState {};
	for (PrestigeType, C_PrestigeTypeRange.X, C_PrestigeTypeRange.Y) {
		if (!GlobalState.PrestigeProgressionList.existskey(PrestigeType)) {
			declare K_PrestigeProgression PrestigeProgressionOffline = K_PrestigeProgression {};
			PrestigeProgressionOffline.Type = PrestigeType;
			GlobalState.PrestigeProgressionList[PrestigeType] = PrestigeProgressionOffline;
		}
	}

	declare CMlFrame[] Trackmania_ComponentsPrestigeProgression2_ComponentControlList for _Context.Page = [];
	Trackmania_ComponentsPrestigeProgression2_ComponentControlList = [];
	_Context.Page.GetClassChildren(C_Private_Class_ComponentContainer, _Context.Page.MainFrame, True);
	foreach (Control in _Context.Page.GetClassChildren_Result) {
		declare CMlFrame Frame_ComponentContainer = (Control as CMlFrame);
		Trackmania_ComponentsPrestigeProgression2_ComponentControlList.add(Frame_ComponentContainer.Parent);

		declare K_Private_Component Trackmania_ComponentsPrestigeProgression2_Component as Component for Frame_ComponentContainer.Parent = K_Private_Component {};
		Component = Private_ComponentInit(_Context, Frame_ComponentContainer.Parent);
		Component = Private_ComponentUpdatePrestigeProgression(_Context, Component, GlobalState);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// `MainLoop` section of the component
Void Private_MainLoop(CMlScript _Context) {
	declare K_Private_GlobalState Trackmania_ComponentsPrestigeProgression2_GlobalState as GlobalState for _Context.System = K_Private_GlobalState {};

	if (_Context.LocalUser != Null) {
		declare Boolean Trackmania_ComponentsPrestigeProgression2_IsPrestigeProgressionDirty for _Context.LocalUser = False;
		if (Trackmania_ComponentsPrestigeProgression2_IsPrestigeProgressionDirty) {
			Trackmania_ComponentsPrestigeProgression2_IsPrestigeProgressionDirty = False;
			GlobalState = Private_GetPrestigeList(_Context, GlobalState, True);
		}
	}

	if (Task::IsInitialized(GlobalState.Task_GetPrestigeList)) {
		GlobalState.Task_GetPrestigeList = Task::Update(GlobalState.Task_GetPrestigeList);
		if (!Task::IsRunning(GlobalState.Task_GetPrestigeList)) {
			declare CTaskResult_PrestigeList TaskResult <=> Task::GetSourceTask_PrestigeList(GlobalState.Task_GetPrestigeList);
			if (Task::IsSuccess(GlobalState.Task_GetPrestigeList) && TaskResult != Null) {
				GlobalState.CanGetPrestigeList = False;
				GlobalState = Private_UpdatePrestigeProgressionList(GlobalState, TaskResult);
			}
			GlobalState.Task_GetPrestigeList = Task::Destroy(GlobalState.Task_GetPrestigeList);
		}
	}

	declare CMlFrame[] Trackmania_ComponentsPrestigeProgression2_ComponentControlList for _Context.Page = [];
	for (ComponentControl in Trackmania_ComponentsPrestigeProgression2_ComponentControlList) {
		declare K_Private_Component Trackmania_ComponentsPrestigeProgression2_Component as Component for ComponentControl = K_Private_Component {};
		if (Component.PrestigeProgressionUpdate != GlobalState.PrestigeProgressionUpdate) {
			Component = Private_ComponentUpdatePrestigeProgression(_Context, Component, GlobalState);
		}

		if (Component.IsVisible) {
			if (_Context.LocalUser != Null) {
				if (Component.SceneId != NullId && MenuSceneManager::PageNeedToRecreateScene(_Context.Page, _Context.LocalUser)) {
					Component = Private_DestroyCameraScene(_Context, Component);
					Component = Private_CreateCameraScene(_Context, Component);
				}
				if (Component.SceneId != NullId && Component.UserColor != _Context.LocalUser.Color) {
					Component = Private_UpdateCameraCar(_Context, Component);
				}
			}
		}
	}
}
