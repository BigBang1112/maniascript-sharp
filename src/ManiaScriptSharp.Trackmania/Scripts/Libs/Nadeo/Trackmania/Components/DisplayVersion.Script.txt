	return MV::Component(
		C_Name,

<label class="{{{P}}}label-version" size="72 6" valign="center2" opacity="0.9" textsize="1" textfont="{{{Stylesheet::GetFont(Stylesheet::C_Font_Weight_ExtraBold)}}}" textprefix="$t$i" focusareacolor1="00000000" focusareacolor2="00000000" scriptevents="1" />
		""",
		"""
#Include "TextLib" as {{{P}}}TL
#Include "Libs/Nadeo/CMGame/Utils/BuildInfo@2.Script.txt" as {{{P}}}BuildInfo
#Include "Libs/Nadeo/CMGame/Utils/Config.Script.txt" as {{{P}}}Config
		""",
		"""
/* The `_ExeVersion` argument is kept for backward compatibility
 * The exe version is now provided directly in the _BuildInfo argument
 */
Void {{{P}}}UpdateVersion(CMlLabel[] _Labels_Version, Text _BuildInfo, Text _ExeVersion, Text _ConfigName) {
	declare Text Version = _BuildInfo;
	if (_ConfigName != "" && _ConfigName != "production") {
		if (Version != "") Version ^= "-";
		Version ^= _ConfigName;
	}
	foreach (Label_Version in _Labels_Version) {
		Label_Version.Value = Version;
	}
}
Void {{{P}}}UpdateVersion(CMlLabel[] _Labels_Version, Text _BuildInfo, Text _ConfigName) {
	{{{P}}}UpdateVersion(_Labels_Version, _BuildInfo, "", _ConfigName);
}

Text {{{P}}}GetBuildInfo() {
	return {{{P}}}BuildInfo::FormatBuildInfo_Short({{{P}}}BuildInfo::GetBuildInfo(System));
}
// Backward compatibility
Text {{{P}}}GetTitleVersion() {
	return {{{P}}}GetBuildInfo();
}

Text {{{P}}}GetClipboard() {
	return {{{P}}}BuildInfo::FormatBuildInfo_Clipboard({{{P}}}BuildInfo::GetBuildInfo(System));
}

***MainInit***
***
declare CMlLabel[] {{{P}}}Labels_Version;
declare Integer {{{P}}}BuildInfoDirtyCounter;
declare Integer {{{P}}}ConfigDirtyCounter;
***

***MainStart***
***
Page.GetClassChildren("{{{P}}}label-version", Page.MainFrame, True);
foreach (Control in Page.GetClassChildren_Result) {
	declare CMlLabel Label_Version = (Control as CMlLabel);
	{{{P}}}Labels_Version.add(Label_Version);
}

{{{P}}}BuildInfoDirtyCounter = {{{P}}}BuildInfo::GetDirtyCounter(System);
{{{P}}}ConfigDirtyCounter = {{{P}}}Config::GetDirtyCounter();

{{{P}}}UpdateVersion({{{P}}}Labels_Version, {{{P}}}GetBuildInfo(), {{{P}}}Config::Get().ConfigName);
***

***MainLoop***
***
if (PageIsVisible) {
	if (
		{{{P}}}BuildInfoDirtyCounter != {{{P}}}BuildInfo::GetDirtyCounter(System) ||
		{{{P}}}ConfigDirtyCounter != {{{P}}}Config::GetDirtyCounter()
	) {
		{{{P}}}BuildInfoDirtyCounter = {{{P}}}BuildInfo::GetDirtyCounter(System);
		{{{P}}}ConfigDirtyCounter = {{{P}}}Config::GetDirtyCounter();
		{{{P}}}UpdateVersion({{{P}}}Labels_Version, {{{P}}}GetBuildInfo(), {{{P}}}Config::Get().ConfigName);
	}

	foreach (Event in PendingEvents) {
		if (Event.Type == CMlScriptEvent::Type::MouseClick && Event.Control != Null && Event.Control.HasClass("{{{P}}}label-version")) {
			System.ClipboardSet({{{P}}}GetClipboard());
		}
	}
}
***
		""",
		[],
		[]
	);
}
