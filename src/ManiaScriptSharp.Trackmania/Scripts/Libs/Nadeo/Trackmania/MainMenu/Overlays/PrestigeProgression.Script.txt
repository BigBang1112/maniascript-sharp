// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Name "Overlay_PrestigeProgression"
#Const P "OverlayPrestigeProgression_" //< Prefix used to differentiate functions/variables in the script
#Const C_NavGroup_Mouse "navgroup-prestige-progression-mouse"

#Const C_Private_Id_PrestigeProgression_Main "trackmania-main-menu-overlays-prestige-progression-id-prestige-progression-main"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the page manialink
Text GetManialink() {
	return MV::Create(
C_Name, 3,
"""
<frameinstance
	modelid="{{{PrestigeProgression::C_Name}}}"
	id="{{{C_Private_Id_PrestigeProgression_Main}}}"
	class="{{{Navigation::C_Class}}}"
	pos="145 45"
	z-index="{{{MainMenuConst::C_ZIndex_PrestigeProgression}}}"
	data-halign="right"
	data-valign="bottom"
	data-use-skin-viewer="true"
	data-nav-inputs="select"
	data-nav-targets="_"
	data-nav-group="{{{C_NavGroup_Mouse}}}"
	data-nav-zone="{{{PrestigeProgression::C_NavZone}}}"
/>
""",
"""
""",
"""
Boolean {{{P}}}UpdatePageIsVisible(CMlFrame _PrestigeProgression_Main, Boolean _PageIsVisible) {
	{{{PrestigeProgression::P}}}SetIsComponentVisible(This, _PrestigeProgression_Main, _PageIsVisible);
	return _PageIsVisible;
}

Void {{{P}}}UpdatePrestigeProgressionFocus(CMlControl _PrestigeProgression_Main) {
	AnimMgr.Flush(_PrestigeProgression_Main);
	if ({{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroup_Mouse}}}") == _PrestigeProgression_Main) {
		AnimMgr.Add(_PrestigeProgression_Main, "<anim scale=\"1.01\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	} else {
		AnimMgr.Add(_PrestigeProgression_Main, "<anim scale=\"1\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	}
}

Void Focus(CMlControl _From, CMlControl _To) {
	if ({{{Transition::P}}}IsInTransition(This)) return;

	if (_From != Null) {
		{{{Navigation::P}}}Unfocus(This, _From);

		if (_From.ControlId == "{{{C_Private_Id_PrestigeProgression_Main}}}") {
			{{{P}}}UpdatePrestigeProgressionFocus(_From);
		}
	}

	if (_To != Null) {
		{{{Navigation::P}}}Focus(This, _To);

		if (_To.ControlId == "{{{C_Private_Id_PrestigeProgression_Main}}}") {
			{{{P}}}UpdatePrestigeProgressionFocus(_To);
		}
	}
}

Void Select(CMlControl _Control) {
	if (_Control == Null || {{{Transition::P}}}IsInTransition(This)) return;

	if (_Control.ControlId == "{{{C_Private_Id_PrestigeProgression_Main}}}") {
		if ({{{Router::P}}}GetCurrentRoutePath(This) != "{{{RoutePaths::C_Path_CreatePrestigeRecap}}}") {
			{{{Router::P}}}SetParentPath(This, "{{{RoutePaths::C_Path_CreatePrestigeRecap}}}", {{{Router::P}}}GetCurrentRoutePath(This));
		}
		{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_CreatePrestigeRecap}}}");

		{{{Navigation::P}}}Unfocus(This, _Control);
		{{{P}}}UpdatePrestigeProgressionFocus(_Control);
	}
}

***MainInit***
***
declare CMlFrame PrestigeProgression_Main;
declare Boolean LocalPageIsVisible;
***

***MainStart***
***
PrestigeProgression_Main <=> (Page.GetFirstChild("{{{C_Private_Id_PrestigeProgression_Main}}}") as CMlFrame);
LocalPageIsVisible = {{{P}}}UpdatePageIsVisible(PrestigeProgression_Main, PageIsVisible);
***

***MainLoop***
***
if (LocalPageIsVisible != PageIsVisible) {
	LocalPageIsVisible = {{{P}}}UpdatePageIsVisible(PrestigeProgression_Main, PageIsVisible);
}

if ({{{Router::P}}}PageIsRunning(This)) {
	foreach (Event in {{{Navigation::P}}}GetPendingEvents(This)) {
		if (
			Event.Type == {{{Navigation::P}}}C_EventType_NavigateMouse &&
			{{{Navigation::P}}}IsEventFromGroup(This, Event, "{{{C_NavGroup_Mouse}}}")
		) {
			switch (Event.Mouse) {
				case CMlScriptEvent::Type::MouseClick: {
					Select(Event.To);
				}
				case CMlScriptEvent::Type::MouseOver, CMlScriptEvent::Type::MouseOut: {
					Focus(Event.From, Event.To);
				}
			}
		}
	}
}
***
""",
[
	Router::ComponentV2(),
	PrestigeProgression::ComponentV2(),
	Transition::ComponentV2(),
	Navigation::ComponentV2()
],
[]
	);
}
