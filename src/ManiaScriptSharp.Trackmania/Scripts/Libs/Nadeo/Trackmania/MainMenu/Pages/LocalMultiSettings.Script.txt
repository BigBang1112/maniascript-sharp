	if (_From != Null) {
		{{{Navigation::P}}}Unfocus(This, _From);
		if ({{{Button::P}}}IsButton(_From)) {
			{{{Button::P}}}Unfocus(This, (_From as CMlFrame));
		} else if ({{{ExpendableButton::P}}}IsButton(_From)) {
			{{{ExpendableButton::P}}}Unfocus(This, (_From as CMlFrame));
		}  else if ({{{Scroll::P}}}IsScroll(_From)) {
			{{{Scroll::P}}}Unfocus(This, (_From as CMlFrame));
		} else if (_From.HasClass("{{{C_Class_Setting}}}")) {
			FocusSetting(_State, _Controls, _From as CMlFrame, False, _IsMouse);
			UpdateControllerInfo(_Controls, _State.IsMovingTracks);
		}
	}

	if (_To != Null) {
		{{{Navigation::P}}}Focus(This, _To);
		if ({{{Button::P}}}IsButton(_To)) {
			{{{Button::P}}}Focus(This, (_To as CMlFrame));
		} else if ({{{ExpendableButton::P}}}IsButton(_To)) {
			{{{ExpendableButton::P}}}Focus(This, (_To as CMlFrame));
		} else if ({{{Scroll::P}}}IsScroll(_To)) {
			{{{Scroll::P}}}Focus(This, (_To as CMlFrame));
		} else if (_To.HasClass("{{{C_Class_Setting}}}")) {
			FocusSetting(_State, _Controls, _To as CMlFrame, True, _IsMouse);
			UpdateControllerInfo(_Controls, _State.IsMovingTracks);
		}
	}
}

}

// Update the given setting
K_State UpdateSetting(K_State _State, K_Controls _Controls, Integer _SettingId, Integer _ShiftNb) {
	declare K_State State = _State;
	declare CMlFrame Frame_Setting = (_Controls.Frame_Settings.GetFirstChild("frame-setting-"^_SettingId) as CMlFrame);

	if (_Controls.Frame_Settings.Visible) {
		declare CMlLabel Label_Value = (Frame_Setting.GetFirstChild("label-setting-value") as CMlLabel);

		switch (_SettingId) {
			case C_Setting_NbPlayers: {
				declare Integer NbIndex = ML::Max(0, State.PlayersNbPossibilities.keyof(State.NbPlayers));
				if (_ShiftNb == C_SettingShift_Next) {
					NbIndex += 1;
				} else if (_ShiftNb == C_SettingShift_Previous) {
					NbIndex -= 1;
				}
				if (NbIndex < 0) {
					State.NbPlayers = State.PlayersNbPossibilities[State.PlayersNbPossibilities.count - 1];
				} else {
					State.NbPlayers = State.PlayersNbPossibilities[NbIndex % State.PlayersNbPossibilities.count];
				}
				Label_Value.SetText(TL::ToText(State.NbPlayers));
			}
			case C_Setting_GameMode: {
				if (State.ModeNames.count > 1) {
					declare Integer ModeIndex = State.ModeIndex + _ShiftNb;
					if (ModeIndex >= State.ModeNames.count) ModeIndex = 0;
					else if (ModeIndex < 0) ModeIndex = State.ModeNames.count-1;

					State.ModeIndex = ModeIndex;
					State.ModeScriptName = C_ModeNames_ScriptNames[State.ModeNames[State.ModeIndex]];
					State.ScriptSettings = {{{ScriptSettingsEditor::P}}}C_DefaultScriptSettings[State.ModeIndex];
					Label_Value.SetText(State.ModeNames[State.ModeIndex]);
					CheckTrackTypes(State, _Controls);
					UpdateInfoParagraph(_Controls, C_Setting_Names.get(_SettingId, ""), GetDescription(State, _SettingId), True);
				}
			}
			case C_Setting_TrackPack: {
				if (_ShiftNb == C_SettingShift_Next) {
					State.CurrentTrackPack = (State.CurrentTrackPack + 1) % State.TrackPacks.count;
				} else if (_ShiftNb == C_SettingShift_Previous) {
					State.CurrentTrackPack -= 1;
					if (State.CurrentTrackPack < 0) State.CurrentTrackPack = State.TrackPacks.count - 1;
				}
				Label_Value.SetText(State.TrackPacks[State.CurrentTrackPack]);
				UpdateScrollElementsTotal(State, _Controls, False);
				if (State.LocalMultiType == {{{Const::C_LocalMultiType_SplitScreen}}}) {
					CheckTrackTypes(State, _Controls);
				} else {
					UpdateErrorMessage("", _Controls);
				}
			}
		}

		Private_TTSContextUpdate(Frame_Setting, True);
	}

	/* Lock the setting if hidden */
	{{{Navigation::P}}}Lock(Frame_Setting, !_Controls.Frame_Settings.Visible);

	return State;
}

Boolean IsCurrentTrackPackLiveCampaign(K_State _State) {
	return _State.CurrentTrackPack >= C_TrackPack_LiveCampaignWhite && _State.CurrentTrackPack <= C_TrackPack_LiveCampaignBlack;
}

/// Check permission for selected tracks
Boolean HasPermissionToPlayTracks(K_State _State) {
	switch (_State.CurrentTrackPack) {
		case C_TrackPack_LiveWeeklyShortsCampaign: {
			if (!{{{UserStore::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayCurrentWeeklyShortsCampaign)) {
				ClubEdition::GoToPage(This, {{{Router::P}}}GetCurrentRoutePath(This), RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_PlayCurrentWeeklyShortsCampaign));
				return False;
			}
		}
		case C_TrackPack_LiveCampaignWhite, C_TrackPack_LiveCampaignGreen, C_TrackPack_LiveCampaignBlue, C_TrackPack_LiveCampaignRed, C_TrackPack_LiveCampaignBlack: {
			if (!UserStore::HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayEntireOfficialQuarterlyCampaign)) {
				foreach (Map in _State.MapLists[_State.CurrentTrackPack]) {
					if (RBAC::TrackRequiresEntireCampaignAccess(CampaignStore::GetLiveQuarterlyCampaign(), Map.Uid)) {
						ClubEdition::GoToPage(This, {{{Router::P}}}GetCurrentRoutePath(This), RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_PlayEntireOfficialQuarterlyCampaign));
						return False;
					}
				}
			}
		}
		case C_TrackPack_LiveMonthlyCampaign: {
			if (!{{{UserStore::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayCurrentOfficialMonthlyCampaign)) {
				ClubEdition::GoToPage(This, {{{Router::P}}}GetCurrentRoutePath(This), RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_PlayCurrentOfficialMonthlyCampaign));
				return False;
			}
		}
	}

	return True;
}

Void Play(K_State _State, K_Controls _Controls) {
	if (!HasPermissionToPlayTracks(_State)) return;

	declare Text[] MapUrls;
	foreach (Map in _State.MapLists[_State.CurrentTrackPack]) {
		if (Map.LocalFileName != "") {
			MapUrls.add(Map.LocalFileName);
		} else if (Map.DownloadUrl != "") {
			MapUrls.add(Map.DownloadUrl);
		}
	}
	if (MapUrls.count > 0 && _State.NbPlayers > 0) {
		switch (_State.LocalMultiType) {
			case {{{Const::C_LocalMultiType_HotSeat}}}: {
				LoadingScreenStore::UpdateTitleMenuLoadingScreen_Random(This, "{{{Text_Title_HotSeat}}}");
				LoadingScreenStore::StartTrackingLoadingScreen(This, LoadingScreenStore::GetCurrentTitleMenuLoadingScreen(This));
				declare Settings = "<script_settings><setting name=\"S_PlayersNb\" value=\"" ^ _State.NbPlayers ^ "\" type=\"integer\"/> </script_settings>";
				TitleControl::PlayMapList(TitleControl, MapUrls, "{{{Const::C_ModeScript_HotSeat}}}", Settings);
				UpdateControllerInfo(_Controls, _State.IsMovingTracks);
			}
			case {{{Const::C_LocalMultiType_SplitScreen}}}: {
				if (TrackTypeCorrespondingToGameMode(_State)) {
					if (_State.ModeScriptName != "") {
						declare CTitleControl::ESplitScreenLayout SplitScreenLayout for System;
						if (_State.NbPlayers == 2) SplitScreenLayout = CTitleControl::ESplitScreenLayout::Vertical;
						else if (_State.NbPlayers == 4)	SplitScreenLayout = CTitleControl::ESplitScreenLayout::Four;
						LoadingScreenStore::UpdateTitleMenuLoadingScreen_Random(This, "{{{Text_Title_SplitScreen}}}");
						LoadingScreenStore::StartTrackingLoadingScreen(This, LoadingScreenStore::GetCurrentTitleMenuLoadingScreen(This));
						TitleControl::PlaySplitScreen(TitleControl, SplitScreenLayout, MapUrls, _State.ModeScriptName, _State.ScriptSettingsXML);
						UpdateControllerInfo(_Controls, _State.IsMovingTracks);
					}
				} else {
					UpdateErrorMessage("{{{Text_MapType}}}", _Controls);
				}
			}
		}
	}	else {
		UpdateErrorMessage(_("You must select at least one track to play"), _Controls);
	}
}

K_State Select(K_State _State, CMlControl _Control, K_Controls _Controls) {
	if (_Control == Null || {{{Transition::P}}}IsInTransition(This)) return _State;

	declare K_State State = _State;

	switch (_Control.ControlId) {
		case "button-back": {
			State = Cancel(State, _Controls);
		}
		case "button-script-settings": {
			if (State.IsMovingTracks) State = SetIsMovingTracks(State, _Controls, False, "");
			ShowScriptSettingsEditor(State, _Controls);
		}
		case "button-play": {
			if (State.IsMovingTracks) State = SetIsMovingTracks(State, _Controls, False, "");
			Play(State, _Controls);
		}
	}
	return State;
}

K_State UpdateAvailableTrackPacks(K_State _State, K_Controls _Controls) {
	declare K_State State = _State;

	State.TrackPacks = [];
	//L16N [LocalMultiSettings] Selector of a custom track pack in the local multi modes
	State.TrackPacks[C_TrackPack_Custom] = _("|TrackPack|Custom");

	if (State.LiveWeeklyShortsCampaign.IsLoaded) {
		//L16N [LocalMultiSettings] Title of the track pack displaying the current weekly shorts campaign. "Weekly Shorts" is considered to be a proper name, so do NOT translate it if you do not need to
		State.TrackPacks[C_TrackPack_LiveWeeklyShortsCampaign] = TL::Compose(_("Weekly Shorts - Week %1"), ""^State.LiveWeeklyShortsCampaign.Week);
	} else if (State.CurrentTrackPack == C_TrackPack_LiveWeeklyShortsCampaign) {
		// Reset track pack
		State = ForceCurrentTrackPack(State, _Controls, C_TrackPack_Custom);
	}

	if (State.LiveCampaign.IsLoaded) {
		declare {{{CampaignStruct::P}}}K_CampaignName CampaignName = {{{CampaignStruct::P}}}ParseCampaignName(State.LiveCampaign.Name);
		declare Text SeasonName = CampaignName.Season;
		if (CampaignName.IsValid) {
			SeasonName = C_Season_Names[CampaignName.Season];
		}
		for (TrackPackIndex, C_TrackPack_LiveCampaignWhite, C_TrackPack_LiveCampaignBlack) {
			declare Text Color = CL::RgbToHex3(C_DifficultyColors.get(TrackPackIndex-2, C_DifficultyColors[Difficulty::C_Difficulty_White]));
			if (UserMgr != Null && UserMgr.MainUser != Null && UserMgr.MainUser.Config.Interface_ColorblindMode) Color = "FFF";

			State.TrackPacks[TrackPackIndex] = TL::Compose(
				//L16N [LocalMultiSettings] Selector of a track pack in the local multi modes, this is a campaign track pack, %1 will be replaced by the season of the campaign, %2 by the year of the campaign, %3 by the color code of the difficulty (do not translate this), and %4 by the difficulty, e.g. "Winter 2024 $<$FFFwhite$>", which will be displayed as "Winter 2024 white"
				_("|TrackPack|%1 %2 $<$%3%4$>"),
				SeasonName,
				CampaignName.Year,
				Color,
				C_DifficultyNames.get(TrackPackIndex-2, "")
			);
		}
	} else if (IsCurrentTrackPackLiveCampaign(State)) {
		// Reset track pack
		State = ForceCurrentTrackPack(State, _Controls, C_TrackPack_Custom);
	}

	if (State.LiveMonthlyCampaign.IsLoaded) {
		State.TrackPacks[C_TrackPack_LiveMonthlyCampaign] = TL::Compose(C_Month_Names[State.LiveMonthlyCampaign.Month-1], ""^State.LiveMonthlyCampaign.Year);
	} else if (State.CurrentTrackPack == C_TrackPack_LiveMonthlyCampaign) {
		// Reset track pack
		State = ForceCurrentTrackPack(State, _Controls, C_TrackPack_Custom);
	}

	_Controls.Button_PreviousTrackPack.Visible = State.TrackPacks.count > 1;
	_Controls.Button_NextTrackPack.Visible = State.TrackPacks.count > 1;

	return State;
}

// Update available mode names to display in the game mode setting
K_State UpdateAvailableModeNames(K_State _State) {
	declare K_State State = _State;
	declare Text[Integer] ModeNames = [];

	if (State.LocalMultiType == {{{Const::C_LocalMultiType_HotSeat}}}) {
		ModeNames = C_HotSeatModeName;
	} else if (State.LocalMultiType == {{{Const::C_LocalMultiType_SplitScreen}}}) {
		ModeNames = C_ModeNames;
		//@HACK remove this when KO mode fixed in splitscreen (see /trackmania-next/tmnext/-/issues/1081 and /trackmania-next/tmnext/-/issues/5219)
		ModeNames.removekey({{{Const::C_ModeIndex_Knockout}}});
		ModeNames.removekey({{{Const::C_ModeIndex_Teams}}});
		ModeNames.removekey({{{Const::C_ModeIndex_Stunt}}});
	}

	State.ModeNames = [];
	State.ModeNamesIndexes = [];
	foreach (ModeIndex => ModeName in ModeNames) {
		State.ModeNames.add(ModeName);
		State.ModeNamesIndexes.add(ModeIndex);
	}

	return State;
}

K_State InitLocalMultiType(K_State _State, K_Controls _Controls, CMlFrame _Frame_Global) {
	declare K_State State = _State;

	State = UpdateAvailableModeNames(State);
	State = UpdateAvailableTrackPacks(State, _Controls);
	declare Text[Integer] TabNames = [];
	switch (State.LocalMultiType) {
		case {{{Const::C_LocalMultiType_HotSeat}}}: {
			State.PlayersNbPossibilities = {{{HotSeatPlayersNb}}};
			TabNames = C_TabNames_HotSeat;
			_Controls.Frame_ScriptSettings.Hide();
			_Controls.Button_PreviousGameMode.Hide();
			_Controls.Button_NextGameMode.Hide();
			State.ModeIndex = 0;
		}
		case {{{Const::C_LocalMultiType_SplitScreen}}}: {
			TabNames = C_TabNames_SplitScreen;
			_Controls.Frame_ScriptSettings.Show();
			_Controls.Button_PreviousGameMode.Show();
			_Controls.Button_NextGameMode.Show();
			State.ModeIndex = {{{Const::C_ModeIndex_TimeAttack}}};
			State.ModeScriptName = C_ModeNames_ScriptNames[State.ModeNames[State.ModeIndex]];
			State.ScriptSettings = {{{ScriptSettingsEditor::P}}}C_DefaultScriptSettings[State.ModeIndex];
			State.ScriptSettingsXML = "<script_settings><setting name=\"S_IsSplitScreen\" value=\"True\" type=\"boolean\"/> </script_settings>";
			State.PlayersNbPossibilities = {{{SplitScreenPlayersNb}}};
		}
	}
	_Controls.Label_SettingGameMode.SetText(State.ModeNames[State.ModeIndex]);
	{{{Navigation::P}}}Lock(_Controls.Button_PreviousGameMode, !_Controls.Button_PreviousGameMode.Visible);
	{{{Navigation::P}}}Lock(_Controls.Button_NextGameMode, !_Controls.Button_NextGameMode.Visible);
	{{{Navigation::P}}}Lock(_Controls.Button_ScriptSettings, !_Controls.Frame_ScriptSettings.Visible);
	{{{NavigationTabs::P}}}SetTabs(This, _Controls.Frame_Tabs, TabNames, False);
	{{{NavigationTabs::P}}}RedrawNavigationTabs(This, _Controls.Frame_Tabs, False);
	_Controls.Label_MovingTracks.RelativePosition_V3.X = _Controls.Frame_Tabs.RelativePosition_V3.X + {{{NavigationTabs::P}}}GetWidth(_Controls.Frame_Tabs) + 3.;
	State.NbPlayers = State.PlayersNbPossibilities[0];
	_Controls.Label_SettingPlayersNb.Value = TL::ToText(State.NbPlayers);
	_Controls.Label_SettingTrackPack.Value = State.TrackPacks[State.CurrentTrackPack];

	State.MapLists[C_TrackPack_Custom] = [];

	return State;
}

						declare Integer TrackKey for Frame_Element;
						TrackKey = C_Track_NullKey;
						HideTrack(Frame_Element);

					Event.Scroll.Visible = Event.ElementsVisible.count > 0;

				case {{{Scroll::P}}}C_EventType_Select: {

								State.MapLists[State.CurrentTrackPack][State.SelectedTrackKey] = State.MapLists[State.CurrentTrackPack][TmpKey];
								State.MapLists[State.CurrentTrackPack][TmpKey] = TmpTrack;
								State = SetIsMovingTracks(State, Controls, False, "");
							}
						} else { // Open track options pop up
							if (State.CurrentTrackPack != C_TrackPack_Custom && State.MapLists[State.CurrentTrackPack].count == 1) {
								//L16N [LocalMultiSettings] Error message when the user is trying to select the last track of the pack

foreach (Event in PendingEvents) {
	if (CampaignStore::IsCampaignStoreEvent(Event.CustomEventType)) {
		switch (Event.Type) {
			case CMlScriptEvent::Type::PluginCustomEvent: {
				switch (Event.CustomEventType) {
					case CampaignStore::C_Event_CampaignLoadComplete, CampaignStore::C_Event_CampaignUpdateFailed: {
						if (
							Event.CustomEventData.count > 0 &&
							(
								State.IsWaitingForLiveCampaignLoad ||
								State.IsWaitingForLiveWeeklyShortsCampaignLoad
							)
						) {
							switch (Event.CustomEventData[0]) {
								case TL::ToText(State.LiveCampaign.Id): {
									if (!State.IsWaitingForLiveMonthlyCampaignLoad && !State.IsWaitingForLiveWeeklyShortsCampaignLoad) DisplayLoadingSpinner(State, Controls, False);
									State.LiveCampaign = CampaignStore::GetCampaignFromId(State.LiveCampaign.Id);
									State = UpdateMapListsFromLiveCampaign(State, Controls);
									State.IsWaitingForLiveCampaignLoad = False;
								}
								case TL::ToText(State.LiveWeeklyShortsCampaign.Id): {
									if (!State.IsWaitingForLiveMonthlyCampaignLoad && !State.IsWaitingForLiveCampaignLoad) DisplayLoadingSpinner(State, Controls, False);
									State.LiveWeeklyShortsCampaign = CampaignStore::GetCampaignFromId(State.LiveWeeklyShortsCampaign.Id);
									State = UpdateMapListsFromLiveWeeklyShortsCampaign(State, Controls);
									State.IsWaitingForLiveWeeklyShortsCampaignLoad = False;
								}
							}
							if (Event.CustomEventType == CampaignStore::C_Event_CampaignUpdateFailed && Event.CustomEventData.count > 1) {
								UpdateErrorMessage(Event.CustomEventData[1], Controls);
							}
						}
					}
					case CampaignStore::C_Event_LiveWeeklyShortsCampaignUpdated: {
						if (State.IsWaitingForLiveWeeklyShortsCampaignLoad) {
							declare Status = CampaignStore::GetLiveWeeklyShortsCampaignStatus();
							if (
								Status == CampaignStore::C_LiveCampaignStatus_NoCampaigns ||
								Status == CampaignStore::C_LiveCampaignStatus_Error
							) {
								// No weekly campaign to preload or error while preloading, stop here.
								if (!State.IsWaitingForLiveMonthlyCampaignLoad && !State.IsWaitingForLiveCampaignLoad) DisplayLoadingSpinner(State, Controls, False);
								State.LiveWeeklyShortsCampaign = CampaignStore::GetLiveWeeklyShortsCampaign();
								State = UpdateMapListsFromLiveWeeklyShortsCampaign(State, Controls);
								UpdateErrorMessage(CampaignStore::GetLiveWeeklyShortsCampaignError(), Controls);
								State.IsWaitingForLiveWeeklyShortsCampaignLoad = False;
							} else if (Status == CampaignStore::C_LiveCampaignStatus_Loaded) {
								State.LiveWeeklyShortsCampaign = CampaignStore::GetCampaignFromId(CampaignStore::GetLiveWeeklyShortsCampaign().Id);
								if (!State.LiveWeeklyShortsCampaign.IsLoaded && State.LiveWeeklyShortsCampaign.Id != {{{CampaignStruct::P}}}C_Campaign_NullId) {
									CampaignStore::Action_LoadCampaign(This, State.LiveWeeklyShortsCampaign.Id);
								} else {
									//L16N [LocalMultiSettings] Error while loading the official campaign
									UpdateErrorMessage(_("Error while retrieving the campaign playlists."), Controls);
									State = UpdateMapListsFromLiveWeeklyShortsCampaign(State, Controls);
									if (!State.IsWaitingForLiveMonthlyCampaignLoad && !State.IsWaitingForLiveCampaignLoad) DisplayLoadingSpinner(State, Controls, False);
									State.IsWaitingForLiveWeeklyShortsCampaignLoad = False;
								}
							}
						}
					}
					case CampaignStore::C_Event_LiveQuarterlyCampaignUpdated: {
						if (State.IsWaitingForLiveCampaignLoad) {
							declare Status = CampaignStore::GetLiveQuarterlyCampaignStatus();
							if (
								Status == CampaignStore::C_LiveCampaignStatus_NoCampaigns ||
								Status == CampaignStore::C_LiveCampaignStatus_Error
							) {
								// No quarterly campaign to preload or error while preloading, stop here.
								if (!State.IsWaitingForLiveMonthlyCampaignLoad && !State.IsWaitingForLiveWeeklyShortsCampaignLoad) DisplayLoadingSpinner(State, Controls, False);
								State.LiveCampaign = CampaignStore::GetLiveQuarterlyCampaign();
								State = UpdateMapListsFromLiveCampaign(State, Controls);
								UpdateErrorMessage(CampaignStore::GetLiveQuarterlyCampaignError(), Controls);
								State.IsWaitingForLiveCampaignLoad = False;
							} else if (Status == CampaignStore::C_LiveCampaignStatus_Loaded) {
								State.LiveCampaign = CampaignStore::GetCampaignFromId(CampaignStore::GetLiveQuarterlyCampaign().Id);
								if (!State.LiveCampaign.IsLoaded && State.LiveCampaign.Id != {{{CampaignStruct::P}}}C_Campaign_NullId) {
									CampaignStore::Action_LoadCampaign(This, State.LiveCampaign.Id);
								} else {
									//L16N [LocalMultiSettings] Error while loading the official campaign
									UpdateErrorMessage(_("Error while retrieving the campaign playlists."), Controls);
									State = UpdateMapListsFromLiveCampaign(State, Controls);
									if (!State.IsWaitingForLiveMonthlyCampaignLoad && !State.IsWaitingForLiveWeeklyShortsCampaignLoad) DisplayLoadingSpinner(State, Controls, False);
									State.IsWaitingForLiveCampaignLoad = False;
								}
							}
						}
					}
					case CampaignStore::C_Event_LiveMonthlyCampaignUpdated: {
						if (State.IsWaitingForLiveMonthlyCampaignLoad) {
							declare Status = CampaignStore::GetLiveMonthlyCampaignStatus();
							if (
								Status == CampaignStore::C_LiveCampaignStatus_NoCampaigns ||
								Status == CampaignStore::C_LiveCampaignStatus_Error
							) {
								if (!State.IsWaitingForLiveCampaignLoad && !State.IsWaitingForLiveWeeklyShortsCampaignLoad) DisplayLoadingSpinner(State, Controls, False);
								State.LiveMonthlyCampaign = CampaignStore::GetLiveMonthlyCampaign();
								State = UpdateMapListsFromMonthlyCampaign(State, Controls);
								UpdateErrorMessage(CampaignStore::GetLiveMonthlyCampaignError(), Controls);
								State.IsWaitingForLiveMonthlyCampaignLoad = False;
							} else {
								// Other values could be `C_LiveCampaignStatus_Loading` and we should continue to wait until the end of the loading.
								// Or `C_LiveCampaignStatus_Loaded` and then we must wait for the `C_Event_MonthlyCampaignLoadComplete` event.
								State.LiveMonthlyCampaign = CampaignStore::GetLiveMonthlyCampaign();
							}
						}
					}
					case CampaignStore::C_Event_MonthlyCampaignLoadComplete: {
						if (
							State.IsWaitingForLiveMonthlyCampaignLoad &&
							Event.CustomEventData.count > 0 &&
							Event.CustomEventData[0] == ""^State.LiveMonthlyCampaign.Id
						) {
							if (!State.IsWaitingForLiveCampaignLoad && !State.IsWaitingForLiveWeeklyShortsCampaignLoad) DisplayLoadingSpinner(State, Controls, False);
							State.LiveMonthlyCampaign = CampaignStore::GetMonthlyCampaignFromId(State.LiveMonthlyCampaign.Id);
							State = UpdateMapListsFromMonthlyCampaign(State, Controls);
							State.IsWaitingForLiveMonthlyCampaignLoad = False;
						}
					}
					case CampaignStore::C_Event_CampaignListUpdateError: {
						if (
							State.IsWaitingForLiveMonthlyCampaignLoad &&
							Event.CustomEventData.count > 1 &&
							TL::ToInteger(Event.CustomEventData[0]) == {{{CampaignStruct::P}}}C_CampaignType_Monthly
						) {
							UpdateErrorMessage(Event.CustomEventData[1], Controls);
							if (!State.IsWaitingForLiveCampaignLoad && !State.IsWaitingForLiveWeeklyShortsCampaignLoad) DisplayLoadingSpinner(State, Controls, False);
							State = UpdateMapListsFromMonthlyCampaign(State, Controls);
							State.IsWaitingForLiveMonthlyCampaignLoad = False;
						}
					}
				}
			}
		}
	}
