/// Create the camera scene
K_Private_State Private_CreateCameraScene(CMlScript _Context, K_Private_State _State) {
	if (_State.SceneId != NullId) return _State;

	declare K_Private_State State = _State;

	State.SceneId = _Context.MenuSceneMgr.SceneCreate(C_Private_SceneLayout);
	State.Camera_Scene.Visible = (State.SceneId != NullId);
	State.PrestigeProgressionUpdate = PrestigeProgression::GetPrestigeProgressionUpdate(_Context);
	MenuSceneManager::PageSynchronizeScene(_Context.Page, _Context.LocalUser);

	if (_Context.LocalUser != Null) {
		State.UserColor = _Context.LocalUser.Color;
	}

	if (State.SceneId != NullId) {
		_Context.MenuSceneMgr.PlaneReflectEnable(State.SceneId, C_Private_PlaneReflectOpacity, Null, Null, Null, Null, C_Private_PlaneReflectAngle);
		_Context.MenuSceneMgr.CameraSetLocation(State.SceneId, C_Private_CameraPosition, C_Private_CameraRotation, C_Private_CameraFOV);

		declare PrestigeProgression::K_PrestigeProgression[Integer] PrestigeProgressionList = PrestigeProgression::GetPrestigeProgressionList(_Context);

		for (PrestigeType, C_Private_PrestigeTypeRange.X, C_Private_PrestigeTypeRange.Y) {
			if (!State.CarItemIdList.existskey(PrestigeType) || State.CarItemIdList[PrestigeType] == NullId) {
				declare PrestigeProgression::K_PrestigeProgression PrestigeProgression = PrestigeProgressionList.get(PrestigeType, PrestigeProgression::K_PrestigeProgression {});
				declare Ident CarItemId = _Context.MenuSceneMgr.ItemCreate(State.SceneId, C_Private_CarModelName, PrestigeProgression.SkinName, PrestigeProgression.SkinUrl, PrestigeProgression.SkinOptions);
				if (CarItemId != NullId) {
					State.CarItemIdList[PrestigeType] = CarItemId;
					_Context.MenuSceneMgr.ItemSetLocation(State.SceneId, CarItemId, C_Private_CarPositionList.get(PrestigeType, <0., 0., 0.>), C_Private_CarRotationList.get(PrestigeType, 0.), C_Private_CarIsTurntable);
					_Context.MenuSceneMgr.ItemSetVehicleState(State.SceneId, CarItemId, C_Private_CarSteerList.get(PrestigeType, 0.),	C_Private_CarBrake, C_Private_CarFrontLight, C_Private_CarTurboLevel, C_Private_CarBoostLevel, C_Private_CarBurnoutSmoke);
					_Context.MenuSceneMgr.ItemSetPlayerState(State.SceneId, CarItemId, _Context.LocalUser);
				}
			}
		}
	}

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Destroy the camera scene
K_Private_State Private_DestroyCameraScene(CMlScript _Context, K_Private_State _State) {
	if (_State.SceneId == NullId) return _State;

	declare K_Private_State State = _State;

	for (CarItemId in State.CarItemIdList) {
		if (CarItemId != NullId) {
			_Context.MenuSceneMgr.ItemDestroy(State.SceneId, CarItemId);
		}
	}
	State.CarItemIdList = [];

	_Context.MenuSceneMgr.SceneDestroy(State.SceneId);
	State.SceneId = NullId;

	State.Camera_Scene.Visible = False;

	return State;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Focus an element of the page
Void Private_Focus(CMlScript _Context, CMlControl _From, CMlControl _To) {
	if (Transition::IsInTransition(_Context)) return;

	if (_From != Null) {
		Navigation::Unfocus(_Context, _From);

		if (ExpendableButtonBase::IsButton(_From)) {
			ExpendableButtonBase::Unfocus(_Context, (_From as CMlFrame));
		}
	}

	if (_To != Null) {
		Navigation::Focus(_Context, _To);

		if (ExpendableButtonBase::IsButton(_To)) {
			ExpendableButtonBase::Focus(_Context, (_To as CMlFrame));
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Select an element of the page
Void Private_Select(CMlScript _Context, CMlControl _Control) {
	if (_Control == Null || Transition::IsInTransition(_Context)) return;

	switch (_Control.ControlId) {
		case C_Private_Id_Button_Garage: {
			if (Router::GetCurrentRoutePath(_Context) != RoutePaths::C_Path_Garage) {
				Router::SetParentPath(_Context, RoutePaths::C_Path_Garage, Router::GetCurrentRoutePath(_Context));
			}
			Router::Push(_Context, RoutePaths::C_Path_Garage);
		}
		case C_Private_Id_Button_Back: {
			Router::PushParent(_Context);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// The page main start section
Void Private_MainStart(CMlScript _Context) {
	declare K_Private_State Trackmania_MainMenuPagesPrestigeRecap_State as State for _Context.Page = C_Private_State_Null;
	State = C_Private_State_Null;

	State.Camera_Scene <=> (_Context.Page.GetFirstChild(C_Private_Id_Camera_Scene) as CMlCamera);
	State.PrestigeProgression_Recap <=> (_Context.Page.GetFirstChild(C_Private_Id_PrestigeProgression_Recap) as CMlFrame);

	MLHelpers::FitLabelValue((_Context.Page.GetFirstChild(C_Private_Id_Label_Explaination) as CMlLabel), C_Private_TextSize_Explaination);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// The page main loop section
Void Private_MainLoop(CMlScript _Context) {
	if (Router::PageIsRunning(_Context, RoutePaths::C_Path_CreatePrestigeRecap)) {
		declare K_Private_State Trackmania_MainMenuPagesPrestigeRecap_State as State for _Context.Page = C_Private_State_Null;

		foreach (Event in Router::GetPendingEvents(_Context)) {
			switch (Event.Type) {
				case Router::C_Event_EnteringRoute: {
					Private_Focus(_Context, Navigation::GetFocusedControl(_Context, C_Private_NavGroup_Main), _Context.Page.GetFirstChild(C_Private_Id_Button_Garage));
					ControllerInfo::SetButtons(["A" => _("Select"), "B" => _("Back")]);
					State = Private_CreateCameraScene(_Context, State);
					PrestigeProgression::SetIsComponentVisible(_Context, State.PrestigeProgression_Recap, True);
				}
			}
		}

		foreach (Event in Navigation::GetPendingEvents(_Context)) {
			switch (Event.Type) {
				case Navigation::C_EventType_NavigateInput: {
					if (Navigation::IsEventFromGroup(_Context, Event, C_Private_NavGroup_Main)) {
						if (Event.Input == CMlScriptEvent::EMenuNavAction::Select) {
							Private_Select(_Context, Event.To);
						} else {
							Private_Focus(_Context, Event.From, Event.To);
						}
					} else if (Navigation::IsEventFromGroup(_Context, Event, C_Private_NavGroup_Global)) {
						if (Event.Input == CMlScriptEvent::EMenuNavAction::Cancel) {
							Private_Select(_Context, _Context.Page.GetFirstChild(C_Private_Id_Button_Back));
						}
					}
				}
				case Navigation::C_EventType_NavigateMouse: {
					if (Navigation::IsEventFromGroup(_Context, Event, C_Private_NavGroup_Main)) {
						if (Event.Mouse == CMlScriptEvent::Type::MouseClick) {
							Private_Select(_Context, Event.To);
						} else if (Event.Mouse == CMlScriptEvent::Type::MouseOver) {
							Private_Focus(_Context, Event.From, Event.To);
						}
					}
				}
			}
		}

		if (
			(State.SceneId != NullId && MenuSceneManager::PageNeedToRecreateScene(_Context.Page, _Context.LocalUser)) ||
			(State.SceneId != NullId && _Context.LocalUser != Null && State.UserColor != _Context.LocalUser.Color) ||
			State.PrestigeProgressionUpdate != PrestigeProgression::GetPrestigeProgressionUpdate(_Context)
		) {
			State = Private_DestroyCameraScene(_Context, State);
			State = Private_CreateCameraScene(_Context, State);
		}
	} else {
		foreach (Event in Router::GetPendingEvents(_Context)) {
			switch (Event.Type) {
				case Router::C_Event_LeavingRoute: {
					declare K_Private_State Trackmania_MainMenuPagesPrestigeRecap_State as State for _Context.Page = C_Private_State_Null;
					State = Private_DestroyCameraScene(_Context, State);
					PrestigeProgression::SetIsComponentVisible(_Context, State.PrestigeProgression_Recap, False);
				}
			}
		}
	}
}
