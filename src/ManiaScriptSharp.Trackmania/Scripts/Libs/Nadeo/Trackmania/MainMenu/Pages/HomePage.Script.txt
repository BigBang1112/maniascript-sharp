Void SetErrorMessage(CMlLabel _Label, Text _Message) {
	_Label.Value = _Message;
	AnimMgr.Flush(_Label);
	AnimMgr.Add(_Label, "<a opacity=\"1.\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	AnimMgr.Add(_Label, "<a opacity=\"0.\" />", Now+3000, 500, CAnimManager::EAnimManagerEasing::QuadOut);
}

Boolean CanGoBackToTitleScreen() {
	return (
		Platform::IsFromFirstParty(System) &&
		Platform::GetPlatform(System) != CSystemPlatform::ESystemPlatform::Luna
	);
}

Void LockUnlockNavigation(K_State _State) {
	declare Boolean IsOpenedPopUp;

	foreach (PopUpMultiChoice in _State.Controls.PopUpMultiChoiceList) {
		if ({{{PopUpMultiChoice::P}}}IsDisplayed(PopUpMultiChoice)) {
			{{{PopUpMultiChoice::P}}}LockPopUp(This, PopUpMultiChoice, Splashscreen::IsVisible(), Splashscreen::IsVisible());
			IsOpenedPopUp = True;
		}
	}

	{{{Navigation::P}}}EnableGroup(This, "{{{C_NavGroupName}}}", !Splashscreen::IsVisible() && !IsOpenedPopUp);
	{{{Router::P}}}LockOverlay(This, "{{{RoutePaths::C_Path_OverlayName_ClubWidget}}}", Splashscreen::IsVisible() || IsOpenedPopUp);
	{{{Router::P}}}LockOverlay(This, "{{{RoutePaths::C_Path_OverlayName_ProfileWidget}}}", Splashscreen::IsVisible() || IsOpenedPopUp);
}

Void UpdateControllerInfo(K_State _State) {
	if (Splashscreen::IsVisible() || _State.Controls.PopUpMultiChoice_CrossPlay.Visible) {
		ControllerInfo::SetButtons(["A" => _("Select")]);
	} else if (_State.Controls.PopUpMultiChoice_QuitGame.Visible) {
		ControllerInfo::SetButtons(["A" => _("Select"), "B" => _("Back")]);
	} else {
		declare Text Echap = _("Quit");
		if (CanGoBackToTitleScreen()) Echap = _("Back");
		ControllerInfo::SetButtons(["A" => _("Select"), "{{{Const::C_NavInput_ProfileWidget}}}" => "{{{Const::C_Text_ProfileWidget}}}", "B" => Echap]);
	}
}

Void Focus(CMlControl _From, CMlControl _To) {
	if ({{{Transition::P}}}IsInTransition(This)) return;

	if (_From != Null) {
		{{{Navigation::P}}}Unfocus(This, _From);
		if ({{{Button::P}}}IsButton(_From)) {
			{{{Button::P}}}Unfocus(This, (_From as CMlFrame));
		} else if ({{{ExpendableButton::P}}}IsButton(_From)) {
			{{{ExpendableButton::P}}}Unfocus(This, (_From as CMlFrame));
		} else if ({{{Checkbox::P}}}IsCheckbox(_From)) {
			{{{Checkbox::P}}}Unfocus(This, _From as CMlFrame);
		} else if (_From.HasClass("news-pager")) {
			AnimMgr.Flush(_From);

K_State UpdateNewsDisplay(K_State _State, Integer _Shift, Boolean _ManualSwitch, Boolean _Anim) {
	declare K_State State = _State;

	State.NewsChangeTime = Now + C_NewsDisplayDuration;

	if (State.News.count <= 0) {
		State.CurrentNewsIndex = 0;
	} else {
		State.CurrentNewsIndex += _Shift;
		if (State.CurrentNewsIndex < 0) State.CurrentNewsIndex = State.News.count - 1;
		else if (State.CurrentNewsIndex > State.News.count - 1) State.CurrentNewsIndex = 0;
	}
	State.Controls.Frame_News_Tabs.Visible = State.News.count > 1;

	if (State.News.existskey(State.CurrentNewsIndex)) {
		declare {{{NewsStruct::P}}}K_News News = State.News[State.CurrentNewsIndex];

		if (News.Id == C_FakeNews_GlobalRanking_Id) {
			declare LeaderboardAPI::K_ResponseFromGetPlayerTrophyRanking UserRanking = {{{UserStore::P}}}GetUserTrophyRankings();
			declare {{{CampaignStruct::P}}}K_Ranking TopRank = CampaignStore::GetTopRank(UserRanking.Ranking);
			if (UserRanking.Ranking.count > 0) {
				State.Controls.Label_Top.Value = "{{{Text_Top}}}";
				State.Controls.Label_TopNumber.Value = Tools::FormatTop(TopRank.Position, False);
				State.News[State.CurrentNewsIndex].Title = C_Zone_Translations.get(TopRank.ZoneName, TL::GetTranslatedText(TopRank.ZoneName));
			} else {
				State.Controls.Label_Top.Value = "";
				//L16N [HomePage] The player did not received a ranking yet
				State.Controls.Label_TopNumber.Value = _("Not ranked");
				State.News[State.CurrentNewsIndex].Title = "";
			}
			Tools::FitLabelValue(State.Controls.Label_TopNumber, 18., 8., 2.);
		} else if (News.Source == {{{NewsStruct::P}}}C_Source_Ubisoft) {
			Tracking::SendNewsImpression(
				ParentApp.TrackingMgr,
				System,
				News.Placement,
				News.Id,
				News.DisplayTime
			);
		}

		if (News.DisplayTime > 0) {
			State.NewsChangeTime = Now + (News.DisplayTime * 1000);
		}

		State.Controls.Quad_Tmp_News_Img.ChangeImageUrl({{{Button::P}}}GetImage(State.Controls.Button_News));
		declare Text MediaUrl = News.MediaUrl;
		if (MediaUrl != "") {
			{{{Button::P}}}SetImage(This, State.Controls.Button_News, MediaUrl);
			{{{Button::P}}}SetImageFocus(This, State.Controls.Button_News, MediaUrl);
		} else {
			{{{Button::P}}}SetImage(This, State.Controls.Button_News, "{{{Const::C_ImageUrl_Home_News_BG}}}");
			{{{Button::P}}}SetImageFocus(This, State.Controls.Button_News, "{{{Const::C_ImageUrl_Home_News_BG}}}");
		}
		// Hide the shadow gradient for global ranking and show it for classic news
		if (News.Id == C_FakeNews_GlobalRanking_Id) {
			{{{Button::P}}}SetGradientOpacityFocus(This, State.Controls.Button_News, 0.);
			{{{Button::P}}}SetGradientOpacityUnfocus(This, State.Controls.Button_News, 0.);
		} else {
			{{{Button::P}}}SetGradientOpacityFocus(This, State.Controls.Button_News, {{{C_NewsGradientOpacityFocus}}});
			{{{Button::P}}}SetGradientOpacityUnfocus(This, State.Controls.Button_News, {{{C_NewsGradientOpacityUnfocus}}});
		}

		// Start fade animation
		if (_Anim) {
			State.Controls.Quad_Tmp_News_Img.Opacity = 1.;
			AnimMgr.Flush(State.Controls.Label_News);
			AnimMgr.Flush(State.Controls.Quad_Tmp_News_Img);
			AnimMgr.Flush(State.Controls.Quad_News_Button_On);
			declare Integer FadeInDuration = Tools::Ternary(_ManualSwitch, C_ManualFadeInDuration, C_AutoFadeInDuration);
			declare Integer FadeOutDuration = Tools::Ternary(_ManualSwitch, C_ManualFadeOutDuration, C_AutoFadeOutDuration);
			declare Integer FadeGap = Tools::Ternary(_ManualSwitch, C_ManualFadeGap, C_AutoFadeGap);
			State.IsManualNewsSwitch = _ManualSwitch;
			AnimMgr.Add(State.Controls.Label_News, "<anim opacity=\"0.\" />", FadeInDuration, CAnimManager::EAnimManagerEasing::Linear);
			AnimMgr.Add(State.Controls.Quad_Tmp_News_Img, "<anim opacity=\"0.\" />", FadeInDuration + FadeGap + FadeOutDuration, CAnimManager::EAnimManagerEasing::Linear);
			AnimMgr.Add(State.Controls.Quad_News_Button_On, "<anim opacity=\"0.\" />", FadeInDuration, CAnimManager::EAnimManagerEasing::Linear);
			// Hide global ranking labels for classic news
			if (News.Id != C_FakeNews_GlobalRanking_Id && State.Controls.Label_GlobalRanking.Opacity > 0.) {
				AnimMgr.Flush(State.Controls.Label_GlobalRanking);
				AnimMgr.Flush(State.Controls.Label_Top);
				AnimMgr.Flush(State.Controls.Label_TopNumber);
				AnimMgr.Add(State.Controls.Label_GlobalRanking, "<anim opacity=\"0.\" />", FadeInDuration, CAnimManager::EAnimManagerEasing::Linear);
				AnimMgr.Add(State.Controls.Label_Top, "<anim opacity=\"0.\" />", FadeInDuration, CAnimManager::EAnimManagerEasing::Linear);
				AnimMgr.Add(State.Controls.Label_TopNumber, "<anim opacity=\"0.\" />", FadeInDuration, CAnimManager::EAnimManagerEasing::Linear);
			}
			State.NewsAnimEndTime = Now + FadeInDuration + FadeGap;
		} else { // Only set news data
			if (State.Controls.Frame_NewsTabsButtonsOff.Controls.existskey(State.CurrentNewsIndex)) {
				State.Controls.Quad_News_Button_On.RelativePosition_V3.X = State.Controls.Frame_NewsTabsButtonsOff.Controls[State.CurrentNewsIndex].RelativePosition_V3.X;
			} else {
				State.Controls.Quad_News_Button_On.RelativePosition_V3.X = 0.;
			}
			State.Controls.Label_News.Value = State.News[State.CurrentNewsIndex].Title;
			{{{Button::P}}}SetTextToSpeech(This, State.Controls.Button_News, State.Controls.Label_News.Value);
			Tools::FitLabelValue(State.Controls.Label_News, 2., 0.75, 0.25);
		}
	} else {
		State.Controls.Label_News.Value = "";
		{{{Button::P}}}SetImage(This, State.Controls.Button_News, "{{{Const::C_ImageUrl_Home_News_BG}}}");
		{{{Button::P}}}SetImageFocus(This, State.Controls.Button_News, "{{{Const::C_ImageUrl_Home_News_BG}}}");
		{{{Button::P}}}SetTextToSpeech(This, State.Controls.Button_News, State.Controls.Label_News.Value);
		{{{Button::P}}}SetGradientOpacityFocus(This, State.Controls.Button_News, {{{C_NewsGradientOpacityFocus}}});
		{{{Button::P}}}SetGradientOpacityUnfocus(This, State.Controls.Button_News, {{{C_NewsGradientOpacityUnfocus}}});
	}

	return State;
}

Void AlignNewsTabs(K_State _State) {
	declare Real SizeX_AllButtons = 0.;
	if (_State.News.count > 0) {
		SizeX_AllButtons = (_State.News.count * {{{Size_NewsTabsButton.X}}}) + (_State.News.count - 1 * {{{SizeX_NewsTabsButtonGap}}});
	}
	_State.Controls.Quad_NewsTabsBg.Size.X = SizeX_AllButtons + {{{2 * Size_NewsTabsButton.X}}};
	foreach (Key => Control in _State.Controls.Frame_NewsTabsButtonsOff.Controls) {
		if (Key < _State.News.count) {
			Control.RelativePosition_V3.X = -SizeX_AllButtons - {{{Size_NewsTabsButton.X}}} + (Key * ({{{Size_NewsTabsButton.X}}} + {{{SizeX_NewsTabsButtonGap}}}));
			Control.Visible = True;
		} else {
			Control.Visible = False;
		}
	}
	foreach (Key => Control in _State.Controls.Frame_NewsTabsButtonsFocus.Controls) {
		if (Key < _State.News.count) {
			Control.RelativePosition_V3.X = -SizeX_AllButtons - {{{Size_NewsTabsButton.X}}} + (Key * ({{{Size_NewsTabsButton.X}}} + {{{SizeX_NewsTabsButtonGap}}}));
			Control.Visible = True;
		} else {
			Control.Visible = False;
		}
	}
}

K_State UpdateNews(K_State _State, Boolean _Anim) {
	declare K_State State = _State;

	State.News.clear();
	State.News.add(GetGlobalRankingFakeNews());
	foreach (New in {{{NewsStore::P}}}GetNews({{{NewsStruct::P}}}C_Placement_Homepage, C_MaxNewsDisplayed-1)) {

K_State Select(CMlControl _Control, K_State _State) {
	if (_Control == Null || {{{Transition::P}}}IsInTransition(This)) return _State;
	declare K_State State = _State;

	switch (_Control.ControlId) {
		case "button-play": {
			switch ({{{UserStore::P}}}GetCurrentPlayTabIndex()) {
				case {{{Const::C_TabIndex_Solo}}}: {{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_Solo}}}");
				case {{{Const::C_TabIndex_Live}}}: {{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_Live}}}");
				case {{{Const::C_TabIndex_Local}}}: {{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_Local}}}");
				default: {{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_Solo}}}");
			}
		}
		case "button-clubs": {
			if ({{{UserStore::P}}}IsOfflineMode()) {
				{{{PopUpMultiChoice::P}}}Open(This, State.Controls.PopUpMultiChoice_Offline);
			} else {
				{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_Clubs}}}");
			}
		}
		case "button-create": {{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_Create}}}");
		case "button-ubi-connect": State.Task_ShowUbisoftConnect = Task::DestroyAndCreate(State.Task_ShowUbisoftConnect, UserMgr, UserMgr.UbisoftConnect_Show(MainUser::GetMainUserId()));
		case "button-news": {
			if (State.News.existskey(State.CurrentNewsIndex)) {
				declare {{{NewsStruct::P}}}K_News News = State.News[State.CurrentNewsIndex];
				// Open the profile page if the players click on their global ranking
				if (News.Id == C_FakeNews_GlobalRanking_Id) {
					{{{Router::P}}}SetParentPath(This, "{{{RoutePaths::C_Path_Profile}}}", {{{Router::P}}}GetCurrentRoutePath(This));
					{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_Profile}}}");
				} else if (News.Source == {{{NewsStruct::P}}}C_Source_Nadeo) {
					SplashscreenStore::Action_CreateImpression(This, News.Id, 0, True);
					if (!Redirect::IsRunning(State.NewsRedirect)) {
						State.NewsRedirect = Redirect::Destroy(This, State.NewsRedirect);
						State.NewsRedirect = Redirect::Create(This, News.ActionType, News.ActionIdentifier);
						if (!Redirect::IsRunning(State.NewsRedirect) && !State.NewsRedirect.IsSuccess) {
							SetErrorMessage(State.Controls.Label_Error, Redirect::C_Text_RedirectionFailed);
						}
					}
				} else if (!Task::IsInitialized(State.Task_OpenNewsLink)) {
					// We put every links in a list and will try to open them all until one work
					State.PendingLinksList = News.Links;
					State.NewsInteraction = K_NewsInteraction {
						NewsId = News.Id,
						Placement = News.Placement
					};
					State.Task_OpenNewsLink = Task::Destroy(State.Task_OpenNewsLink);
				}
			}
		}
		case "button-settings": {
			{{{Router::P}}}SetParentPath(This, "{{{RoutePaths::C_Path_Settings}}}", "{{{RoutePaths::C_Path_Home}}}");
			{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_Settings}}}");
		}
	}

	if (_Control.HasClass("news-pager") && _Control.DataAttributeExists("key")) {
		declare Integer Shift = TL::ToInteger(_Control.DataAttributeGet("key")) - State.CurrentNewsIndex;
		if (Shift != 0) {
			State = UpdateNewsDisplay(State, Shift, True);
		}
	}

	return State;
}
