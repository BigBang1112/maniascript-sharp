Void Focus(K_State _State, CMlControl _From, CMlControl _To) {
	if ({{{Transition::P}}}IsInTransition(This)) return;

	if (_From != Null) {
		{{{Navigation::P}}}Unfocus(This, _From);
		if ({{{ExpendableButton::P}}}IsButton(_From)) {
			{{{ExpendableButton::P}}}Unfocus(This, (_From as CMlFrame));
		} else if ({{{Button::P}}}IsButton(_From)) {
			{{{Button::P}}}Unfocus(This, (_From as CMlFrame));
			if (_From.Id == _State.Controls.Button_TOTD.Id) SetTOTDDayTextColor(_State.Controls.Label_DayTOTD, False);
			if (_From.DataAttributeExists("secondarylabel")) {
				SetLabelOpacity(_From.Parent.GetFirstChild(_From.DataAttributeGet("secondarylabel")) as CMlLabel, False);
			}
		}
	}

	if (_To != Null) {
		{{{Navigation::P}}}Focus(This, _To);
		if ({{{ExpendableButton::P}}}IsButton(_To)) {
			{{{ExpendableButton::P}}}Focus(This, (_To as CMlFrame));
		} else if ({{{Button::P}}}IsButton(_To)) {
			{{{Button::P}}}Focus(This, (_To as CMlFrame));
			if (_To.Id == _State.Controls.Button_TOTD.Id) SetTOTDDayTextColor(_State.Controls.Label_DayTOTD, True);
			if (_To.DataAttributeExists("secondarylabel")) {
				SetLabelOpacity(_To.Parent.GetFirstChild(_To.DataAttributeGet("secondarylabel")) as CMlLabel, True);
			}
		}
	}
}

K_State Select(K_State _State, CMlControl _Control) {
	if (_Control == Null || {{{Transition::P}}}IsInTransition(This)) return _State;

	if ({{{UserStore::P}}}IsOfflineMode() && _Control.Id != _State.Controls.Button_Back.Id) {
		{{{PopUpMultiChoice::P}}}Open(This, _State.Controls.PopUpMultiChoice_Offline);
		return _State;
	}

	declare K_State State = _State;

	switch (_Control.ControlId) {
		case "button-cup-of-the-week": {
			{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_COTWChannelDisplay}}}", True);
		}
		case "button-ranked": {
			if (!{{{UserStore::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayMatchmaking)) {
				ClubEdition::GoToPage(This, {{{Router::P}}}GetCurrentRoutePath(This), RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_PlayMatchmaking));
			} else {
				if (State.CouldPlayRanked) {
					{{{Router::P}}}SetParentPath(This, "{{{RoutePaths::C_Path_MatchmakingMainPage}}}", "{{{RoutePaths::C_Path_Live}}}");
					{{{Router::P}}}Push(
						This,
						"{{{RoutePaths::C_Path_MatchmakingMainPage}}}",
						[
							"{{{RoutePaths::C_Query_ForceMode}}}" => {{{MatchmakingStruct::P}}}GetActiveRankedMode(),
							"{{{Router::C_Query_DisablePrivilege}}}" => Privileges::PrivilegesToQuery([CPrivilegeMgr::EPrivilege::PlayMultiplayerModeWithUGC]),
							"{{{Router::C_Query_EnablePrivilege}}}" => Privileges::PrivilegesToQuery([CPrivilegeMgr::EPrivilege::PlayMultiplayerMode])
						]
					);
				} else {
					{{{PopUpMultiChoice::P}}}Open(This, _State.Controls.PopUpMultiChoice_Ranked);
				}
			}
		}
		case "button-totd": {
			{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_TOTDChannelDisplay}}}", True);
		}
		case "button-arcade": {
			// Check that the user has the privilege to view UGC
			State.UGCPrivilegeCheck = Privileges::Check_DestroyAndCreate(
				State.UGCPrivilegeCheck,
				PrivilegeMgr,
				MainUser::GetMainUserId(),
				CPrivilegeMgr::EPrivilege::UseUserCreatedContent
			);
		}
		case "button-events": {
			{{{Router::P}}}SetParentPath(This, "{{{RoutePaths::C_Path_Events}}}", "{{{RoutePaths::C_Path_Live}}}");
			{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_Events}}}");
		}
		case "button-back": {{{Router::P}}}PushParent(This);
	}

	return State;
}

// Check the campaign medals to display a warning for players that don't have enough Bronze medals and want to play Ranked
K_State CheckLiveQuarterlyCampaignMedals(K_State _State, {{{CampaignStruct::P}}}K_Campaign _Campaign) {
	declare K_State State = _State;

	// Count the bronze medals in all the campaign
	declare Integer BronzeMedalsNb = 0;
	foreach (MapData in _Campaign.Playlist) {
		declare Medal = ScoreMgr.Map_GetMedal(
			MainUser::GetMainUserId(),
			MapData.Uid,
			"{{{Const::C_ScopeType_PersonalBest}}}",
			"",
			"{{{Const::C_GameMode_TimeAttack}}}",
			""
		);
		if (Medal >= Medals::C_ScoreMgrMedal_Bronze) {
			BronzeMedalsNb += 1;
		}
	}

	// Determine if player should play Ranked mode with medals numbers
	State.CouldPlayRanked = BronzeMedalsNb >= Config::Get().MinBronzeMedalNbToPlayRanked;

	return State;
}

/// Update the Cup of the Week status
K_State Private_SetStatus(K_State _State) {
	declare K_State State = _State;

	declare Integer CountdownDelta = TiL::GetDelta(""^State.COTW.CountdownTimestamp, TiL::GetCurrent());
	declare Integer QualificationStartDelta = TiL::GetDelta(""^State.COTW.QualificationStartTimestamp, TiL::GetCurrent());
	declare Integer QualificationEndDelta = TiL::GetDelta(""^State.COTW.QualificationEndTimestamp, TiL::GetCurrent());
	declare Integer CampaignEndDelta = TiL::GetDelta(""^State.COTW.CampaignEndTimestamp, TiL::GetCurrent());

	if (State.COTW.CampaignId == {{{CampaignStruct::P}}}C_Campaign_NullId) {
		State.COTW.Status = C_Private_Status_NoCup;
		State.COTW.StatusUpdateTime = -1;
		State.Controls.Label_FooterCOTW.Value = "";
		State.Controls.Quad_FooterCOTW.Visible = False;
	} else if (CountdownDelta > 0) {
		State.COTW.Status = C_Private_Status_WaitingCup;
		State.COTW.StatusUpdateTime = Now + (CountdownDelta * 1000);
		State.Controls.Label_FooterCOTW.Value = "";
		State.Controls.Quad_FooterCOTW.Visible = False;
	} else if (QualificationStartDelta > 0) {
		State.COTW.Status = C_Private_Status_QualificationCountdown;
		State.COTW.StatusUpdateTime = Now + (QualificationStartDelta * 1000);
		// Update `Label_FooterCOTW` in the main loop
		State.Controls.Quad_FooterCOTW.Visible = True;
	} else if (QualificationEndDelta > 0) {
		State.COTW.Status = C_Private_Status_QualificationOngoing;
		State.COTW.StatusUpdateTime = Now + (QualificationEndDelta * 1000);
		// Update `Label_FooterCOTW` in the main loop
		State.Controls.Quad_FooterCOTW.Visible = True;
	} else if (CampaignEndDelta > 0) {
		State.COTW.Status = C_Private_Status_QualificationEnded;
		State.COTW.StatusUpdateTime = Now + (CampaignEndDelta * 1000);
		State.Controls.Label_FooterCOTW.Value = C_Private_Text_QualificationEnded;
		State.Controls.Quad_FooterCOTW.Visible = True;
	} else {
		State.COTW.Status = C_Private_Status_NextCup;
		State.COTW.StatusUpdateTime = -1;
		State.Controls.Label_FooterCOTW.Value = C_Private_Text_CupEnded;
		State.Controls.Quad_FooterCOTW.Visible = True;
	}

	// If there is a cup but it has not started yet, enable loading of the next cup
	// /!\ Be careful when editing this, we purposefully set the value to `True` but we do not set it to `False`
	if (
		State.COTW.Status != C_Private_Status_NoCup &&
		State.COTW.Status != C_Private_Status_NextCup
	) {
		State.COTW.CanLoadNextCupOfTheWeekCampaign = True;
	}

	if (State.Controls.Label_FooterCOTW.Value != "") {
		MLHelpers::FitLabelValue(State.Controls.Label_FooterCOTW, C_Private_TextSize_COTWFooter);
	}

	return State;
}

/// Set the current Cup of the Week campaign
K_State Private_SetCupOfTheWeekCampaign(
	K_State _State,
	{{{CampaignStruct::P}}}K_Campaign _CupOfTheWeekCampaign,
	Integer _CupOfTheWeekCampaignUpdateTime
) {
	declare K_State State = _State;

	State.COTW.CampaignId = _CupOfTheWeekCampaign.Id;

	State.COTW.CampaignEndTimestamp = _CupOfTheWeekCampaign.EndTimestamp;
	State.COTW.QualificationStartTimestamp = State.COTW.CampaignEndTimestamp + CampaignStore::C_CupOfTheWeekStartTimeOffset;
	State.COTW.CountdownTimestamp = State.COTW.QualificationStartTimestamp - C_Private_CountdownDuration;
	State.COTW.QualificationEndTimestamp = State.COTW.QualificationStartTimestamp + C_Private_QualificationDuration;
	State.COTW.NextCupOfTheWeekCampaignUpdateTime = _CupOfTheWeekCampaignUpdateTime;

	if (State.COTW.QualificationStartTimestamp > 0) {
		State.Controls.Label_DateCOTW.Value = TiL::FormatDate(""^State.COTW.QualificationStartTimestamp, TiL::EDateFormats::Date);
	} else {
		State.Controls.Label_DateCOTW.Value = "";
	}

	MLHelpers::FitLabelValue(State.Controls.Label_DateCOTW, C_Private_TextSize_COTWDate);

	State = Private_SetStatus(State);

	return State;
}
