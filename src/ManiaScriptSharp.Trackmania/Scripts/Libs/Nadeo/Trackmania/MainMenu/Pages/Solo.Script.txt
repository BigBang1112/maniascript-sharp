Text GetQuarterlyCampaignName({{{CampaignStruct::P}}}K_CampaignName _CampaignName) {
	if (_CampaignName.IsValid) {
		return _CampaignName.TranslatedName;
	}
	return _CampaignName.Name;
}

Text GetQuarterlyCampaignTTS({{{CampaignStruct::P}}}K_CampaignName _CampaignName, Text _CampaignText) {
	if (_CampaignName.IsValid) {
		return TL::Compose("%1 - %2 %3", _CampaignText, C_Season_Names.get(_CampaignName.Season, ""), _CampaignName.Year);
	}
	return TL::Compose("%1 - %2", _CampaignText, _CampaignName.Name);
}

Void SetLabelOpacity(CMlLabel _Label, Boolean _IsFocused) {
	AnimMgr.Flush(_Label);
	if (_IsFocused) {

	}

			AnimMgr.Add(Control, "<anim opacity=\""^Tools::Ternary(_IsFocused, 1., 0.8)^"\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		}
	}
}

K_State Focus(K_State _State, CMlControl _From, CMlControl _To) {
	if ({{{Transition::P}}}IsInTransition(This)) return _State;
	declare K_State State = _State;

	if (_From != Null) {
		{{{Navigation::P}}}Unfocus(This, _From);
		if ({{{ExpendableButton::P}}}IsButton(_From)) {
			{{{ExpendableButton::P}}}Unfocus(This, (_From as CMlFrame));
		} else if ({{{Button::P}}}IsButton(_From)) {
			{{{Button::P}}}Unfocus(This, (_From as CMlFrame));
			switch (_From.Id) {
				case State.Controls.Button_TOTD.Id: {
					SetTOTDDayTextColor(State.Controls.Label_DayTOTD, False);
					FocusButtonFrames(State.Controls.Frame_Rank_TOTD, False);
				}
				case State.Controls.Button_Campaign.Id: {
					if (State.ShowQuarterlyCampaignTimer) {
						declare Integer SeasonInfoAppearanceTimestamp = State.SwitchSeasonInfoTime - C_DelayBetweenSeasonInfoSwitch + C_Duration_SeasonInfoHiddenAnim;
						foreach (Frame in [State.Controls.Frame_Timer_Quarterly, State.Controls.Frame_Rank_Quarterly]) {
							if (!Frame.Visible) { //< Replace the opacity animation of the frame that will soon be displayed
								foreach (Control in Frame.Controls) {
									AnimMgr.Flush(Control);
									AnimMgr.Add(Control, "<anim opacity=\"0.8\" />", SeasonInfoAppearanceTimestamp, C_Duration_SeasonInfoShownedAnim, CAnimManager::EAnimManagerEasing::QuadOut);
								}
							} else if (SeasonInfoAppearanceTimestamp < Now) { //< Update opacity only if visible frame is not disappearing
								FocusButtonFrames(Frame, False);
							}
						}
					} else {
						FocusButtonFrames(State.Controls.Frame_Rank_Quarterly, False);
						FocusButtonFrames(State.Controls.Frame_Timer_Quarterly, False);
					}
				}
			}
			if (_From.DataAttributeExists("secondarylabel")) {
				SetLabelOpacity(_From.Parent.GetFirstChild(_From.DataAttributeGet("secondarylabel")) as CMlLabel, False);
			}
		}
	}

	if (_To != Null) {
		{{{Navigation::P}}}Focus(This, _To);
		if ({{{ExpendableButton::P}}}IsButton(_To)) {
			{{{ExpendableButton::P}}}Focus(This, (_To as CMlFrame));
		} else if ({{{Button::P}}}IsButton(_To)) {
			{{{Button::P}}}Focus(This, (_To as CMlFrame));
			switch (_To.Id) {
				case State.Controls.Button_TOTD.Id: {
					SetTOTDDayTextColor(State.Controls.Label_DayTOTD, True);
					FocusButtonFrames(State.Controls.Frame_Rank_TOTD, True);
				}
				case State.Controls.Button_Campaign.Id: {
					if (State.ShowQuarterlyCampaignTimer) State.SwitchSeasonInfoTime = Now;
					FocusButtonFrames(State.Controls.Frame_Rank_Quarterly, True);
					FocusButtonFrames(State.Controls.Frame_Timer_Quarterly, True);
				}
			}
			if (_To.DataAttributeExists("secondarylabel")) {
				SetLabelOpacity(_To.Parent.GetFirstChild(_To.DataAttributeGet("secondarylabel")) as CMlLabel, True);
			}
		}
	}

	return State;
}

Void UpdateTimer(K_State _State, CMlFrame _Frame_Timer, Integer _CampaignEndTimestamp) {
	declare CMlLabel Label_Timer <=> _Frame_Timer.GetFirstChild("label-timer") as CMlLabel;
	declare Integer TimerLeft = TiL::GetDelta(""^_CampaignEndTimestamp, TiL::GetCurrent());
	declare Integer Days = TimerLeft/(3600*24);

	if (Days >= 1) { // More than a day
		if (Days > 1) {
			//L16N Message displayed to indicate when the new season starts. %1 will be replaced by a number, e.g "23 days"
			Label_Timer.Value = TL::Compose(_("%1 days"), ""^Days);
		} else {
			declare Integer Hours = TimerLeft/3600 - Days*24 + 1; // +1 to avoid "and 0 hours"
			if (Hours > 1) {
				//L16N Message displayed to indicate when the new season starts. %1 and %2 will be replaced by a number, e.g "1 day and 12 hours"
				Label_Timer.Value = TL::Compose(_("%1 day and %2 hours"), ""^Days, ""^Hours);
			} else {
				//L16N Message displayed to indicate when the new season starts. %1 will be replaced by a number, e.g "1 day and 1 hour"
				Label_Timer.Value = TL::Compose(_("%1 day and %2 hour"), ""^Days, ""^Hours);
			}
		}
	} else {
		//L16N Message displayed to indicate when the new season starts. %1 will be replaced by a timer, e.g "12:10:53"
		Label_Timer.Value = TL::Compose(_("%1"), TL::TimeToText(TimerLeft*1000 + 1, False)); // Add 1 second on the timer to avoid "-0:00" on screen
	}
}

Void UpdatePlayerLiveCampaignRank(K_State _State, Integer _CampaignId, CMlFrame _Frame_Rank, CMlLabel _Label_RankZone, CMlLabel _Label_RankPosition, Boolean _IsMapRankings) {
	if (_Frame_Rank == Null || _Label_RankZone == Null || _Label_RankPosition == Null) return;
	declare {{{CampaignStruct::P}}}K_Campaign Campaign = CampaignStore::GetCampaignFromId(_CampaignId);
	_Frame_Rank.Visible = Campaign.Id != {{{CampaignStruct::P}}}C_Campaign_NullId;

	if (_Frame_Rank.Visible) {
		declare {{{CampaignStruct::P}}}K_Ranking TopRank = CampaignStore::GetTopRank(Campaign.CampaignRanking.Rankings);
		if (_IsMapRankings) { // Need to retrieve the map for the TOTD rankings
			declare {{{CampaignStruct::P}}}K_Map MapData;
			foreach (Map in Campaign.Playlist) {
				if (Map.Uid == _State.CurrentTOTDMapUid) {
					MapData = Map;
					break;
				}
			}
			TopRank = CampaignStore::GetTopRank(MapData.Rankings);
		}

		if (TopRank.ZoneName != "" && TopRank.Position > 0) {
			_Label_RankZone.Value = C_Zone_Translations.get(TopRank.ZoneName, TL::GetTranslatedText(TopRank.ZoneName));
			// Display a rank suffix only if player in Top 10 (after that the player rank will be displayed like "1K")
			if (TopRank.Position < C_SoloRankFormatLimit) {
				_Label_RankPosition.Value = TL::FormatRank(TopRank.Position, False);
			} else if (TopRank.Position < C_RankFormatLimit) {
				_Label_RankPosition.Value = TL::ToText(TopRank.Position);
			} else {
				_Label_RankPosition.Value = Tools::FormatTop(TopRank.Position, False, C_RankFormatLimit);
			}
		} else {
			_Label_RankPosition.Value = "";
			_Label_RankZone.Value = "{{{Text_NotRanked}}}";
		}
		Tools::FitLabelValue(_Label_RankZone, 2.5, 1., 0.25);
		_Label_RankZone.RelativePosition_V3.X = _Label_RankPosition.RelativePosition_V3.X + ML::Min(_Label_RankPosition.Size.X, _Label_RankPosition.ComputeWidth(_Label_RankPosition.Value));
	}
}

Void UpdatePlayerLiveCampaignRank(K_State _State, Integer _CampaignId, CMlFrame _Frame_Rank, CMlLabel _Label_RankZone, CMlLabel _Label_RankPosition) {
	UpdatePlayerLiveCampaignRank(_State, _CampaignId, _Frame_Rank, _Label_RankZone, _Label_RankPosition, False);
}

Void AnimFrames(CMlFrame _FrameToShow, CMlFrame _FrameToHide) {
	declare Vec2 Pos_HiddenStart = <{{{Pos_FrameSeasonInfoVisible.X}}} - 5., 0.>;
	declare CMlControl FocusedControl = {{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroup_Main}}}");
	declare Real FocusedOpacity = Tools::Ternary(FocusedControl != Null && FocusedControl.HasClass("{{{C_Class_CampaignButton}}}"), 1., 0.8);

	// Flush animations
	AnimMgr.Flush(_FrameToShow);
	AnimMgr.Flush(_FrameToHide);

	// Hide previous information frame directly
	AnimMgr.Add(_FrameToHide, "<anim hidden=\"1\"/>", C_Duration_SeasonInfoHiddenAnim, CAnimManager::EAnimManagerEasing::QuadOut);
	foreach (Control in _FrameToHide.Controls) {
		AnimMgr.Add(Control, "<anim opacity=\"0.\" />", C_Duration_SeasonInfoHiddenAnim, CAnimManager::EAnimManagerEasing::QuadOut);
	}

	// Show new information frame when previous animation is over
	_FrameToShow.RelativePosition_V3 = Pos_HiddenStart;
	AnimMgr.Add(_FrameToShow, "<anim pos=\""^{{{Pos_FrameSeasonInfoVisible.X}}}^" "^{{{Pos_FrameSeasonInfoVisible.Y}}}^"\" hidden=\"0\"/>", Now + C_Duration_SeasonInfoHiddenAnim, C_Duration_SeasonInfoShownedAnim, CAnimManager::EAnimManagerEasing::QuadOut);
	foreach (Control in _FrameToShow.Controls) {
		if ((Control is CMlQuad) || (Control is CMlLabel)) {
			AnimMgr.Flush(Control);
			if (Control is CMlQuad) (Control as CMlQuad).Opacity = 0.;
			else (Control as CMlLabel).Opacity = 0.;
			AnimMgr.Add(Control, "<anim opacity=\""^FocusedOpacity^"\" />", Now + C_Duration_SeasonInfoHiddenAnim, C_Duration_SeasonInfoShownedAnim, CAnimManager::EAnimManagerEasing::QuadOut);
		}
	}
}

Void ResetAnimFrames(CMlFrame _Frame) {
	declare CMlControl FocusedControl = {{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroup_Main}}}");
	declare Real FocusedOpacity = Tools::Ternary(FocusedControl != Null && FocusedControl.HasClass("{{{C_Class_CampaignButton}}}"), 1., 0.8);

	AnimMgr.Flush(_Frame);
	_Frame.RelativePosition_V3 = {{{Pos_FrameSeasonInfoVisible}}};
	foreach (Control in _Frame.Controls) {
		AnimMgr.Flush(Control);
		if (Control is CMlQuad) {
			(Control as CMlQuad).Opacity = FocusedOpacity;
		} else if (Control is CMlLabel) {
			(Control as CMlLabel).Opacity = FocusedOpacity;
		}
	}
}

K_State UpdateQuarterlyCampaign(K_State _State) {
	declare K_State State = _State;
	declare {{{CampaignStruct::P}}}K_Campaign LiveQuarterlyCampaign = CampaignStore::GetLiveQuarterlyCampaign();

	if (LiveQuarterlyCampaign.Id != {{{CampaignStruct::P}}}C_Campaign_NullId) {
		declare {{{CampaignStruct::P}}}K_CampaignName CampaignName = {{{CampaignStruct::P}}}ParseCampaignName(LiveQuarterlyCampaign.Name);

		// Update current campaign relative end
		if (LiveQuarterlyCampaign.IsSeasonLive) {
			State.CurrentQuarterlyCampaignEndTimestamp = LiveQuarterlyCampaign.EndTimestamp;
		}

		// Update current campaign button
		{{{LoadingSpinner::P}}}StopAnimation(This, State.Controls.Frame_LoadingSpinner);
		{{{Button::P}}}SetValue(This, State.Controls.Button_Campaign, GetQuarterlyCampaignName(CampaignName));
		{{{Button::P}}}SetTextToSpeech(This, State.Controls.Button_Campaign, GetQuarterlyCampaignTTS(CampaignName, "{{{Text_Campaign}}}"));
		{{{Button::P}}}SetVisibilityAndNavigation(State.Controls.Button_Campaign, True);
		State.Controls.Quad_CampaignForeground.Show();
		State.Controls.Label_Campaign.Show();
		UpdateQuarterlyButtonMedia(State, LiveQuarterlyCampaign, CampaignName);
	} else {
		State.CurrentQuarterlyCampaignEndTimestamp = 0;
		State.Controls.Label_Campaign.Hide();
		State.Controls.Frame_Rank_Quarterly.Hide();
		State.Controls.Frame_Timer_Quarterly.Hide();
		State.Controls.Quad_CampaignForeground.Hide();
		{{{Button::P}}}SetVisibilityAndNavigation(State.Controls.Button_Campaign, False);
		{{{LoadingSpinner::P}}}StartAnimation(This, State.Controls.Frame_LoadingSpinner, "{{{Text_Retrieving}}}");
	}

	// Update campaign rank visibility
	ResetAnimFrames(_State.Controls.Frame_Rank_Quarterly);
	UpdatePlayerLiveCampaignRank(
		State,
		LiveQuarterlyCampaign.Id,
		State.Controls.Frame_Rank_Quarterly,
		State.Controls.Label_RankZone_Quarterly,
		State.Controls.Label_RankPosition_Quarterly
	);

	// Update campaign timer visibility
	ResetAnimFrames(_State.Controls.Frame_Timer_Quarterly);
	State.ShowQuarterlyCampaignTimer = TiL::GetDelta(""^State.CurrentQuarterlyCampaignEndTimestamp, TiL::GetCurrent())/(3600*24) <= C_MaxDaysLeftToShowQuarterlyCampaignTimer;
	State.Controls.Frame_Timer_Quarterly.Visible = State.ShowQuarterlyCampaignTimer && !State.Controls.Frame_Rank_Quarterly.Visible && State.CurrentQuarterlyCampaignEndTimestamp > 0;
	if (State.ShowQuarterlyCampaignTimer) State.SwitchSeasonInfoTime = Now + C_DelayBetweenSeasonInfoSwitch;

	return State;
}

Integer GetMedalFromMapUidList(Text[] _MapUidList) {
	// Count the medals in the map list
	declare Integer BronzeMedalsNb = 0;
	declare Integer SilverMedalsNb = 0;
	declare Integer GoldMedalsNb = 0;
	declare Integer AuthorMedalsNb = 0;
	foreach (MapUid in _MapUidList) {
		declare Integer Medal = ScoreMgr.Map_GetMedal(
			MainUser::GetMainUserId(),
			MapUid,
			"{{{Const::C_ScopeType_PersonalBest}}}",
			"",
			"{{{Const::C_GameMode_TimeAttack}}}",
			""
		);
		if (Medal >= Medals::C_ScoreMgrMedal_Bronze) BronzeMedalsNb += 1;
		if (Medal >= Medals::C_ScoreMgrMedal_Silver) SilverMedalsNb += 1;
		if (Medal >= Medals::C_ScoreMgrMedal_Gold) GoldMedalsNb += 1;
		if (Medal >= Medals::C_ScoreMgrMedal_Author) AuthorMedalsNb += 1;
	}

	// To return the best medal record set on all maps
	declare Integer AllCampaignMedal = Medals::C_ScoreMgrMedal_None;
	if (AuthorMedalsNb == _MapUidList.count) AllCampaignMedal = Medals::C_ScoreMgrMedal_Author;
	else if (GoldMedalsNb == _MapUidList.count) AllCampaignMedal = Medals::C_ScoreMgrMedal_Gold;
	else if (SilverMedalsNb == _MapUidList.count) AllCampaignMedal = Medals::C_ScoreMgrMedal_Silver;
	else if (BronzeMedalsNb == _MapUidList.count) AllCampaignMedal = Medals::C_ScoreMgrMedal_Bronze;
	return AllCampaignMedal;
}
