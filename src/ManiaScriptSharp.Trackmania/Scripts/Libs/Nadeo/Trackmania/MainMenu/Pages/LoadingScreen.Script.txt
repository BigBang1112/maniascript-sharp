#Const Version "1.5.0"
#Const ScriptName "Libs/Nadeo/Trackmania/MainMenu/Pages/LoadingScreen.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Includes
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Utils/Layers.Script.txt" as Layers
#Include "Libs/Nadeo/CMGame/Utils/ManiaView2.Script.txt" as MV
#Include "Libs/Nadeo/Trackmania/Components/LoadingScreen.Script.txt" as LoadingScreen

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Name "Page_LoadingScreen"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the manialink page
Text GetManialink() {
	return MV::Create(
C_Name, 3,
"""
<frame id="frame-global" z-index="-1">
	<frameinstance modelid="{{{LoadingScreen::C_Name}}}" />
</frame>
""",
"""
""",
"""
""",
[
	LoadingScreen::ComponentV2()
],
[]
	);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Create and display layer
Void Load() {
	Layers::Create(C_Name);
	declare CUILayer LoadingScreenLayer <=> Layers::Get(C_Name);
	if (LoadingScreenLayer != Null) {
		LoadingScreenLayer.Type = CUILayer::EUILayerType::LoadingScreen;
		LoadingScreenLayer.IsVisible = True;
		LoadingScreenLayer.ManialinkPage = GetManialink();
	}
}
