Boolean Private_IsParentOf(Integer[Integer] _FolderReverseTree, Integer _ParentFolderId, Integer _ChildFolderId) {
	if (_ParentFolderId == _ChildFolderId) return True;

	declare Integer ParentFolderId = _FolderReverseTree.get(_ChildFolderId, ClubStruct::C_FolderNullId);
	declare Integer[] TraversedFolderIdList = [_ChildFolderId];
	while (ParentFolderId != ClubStruct::C_FolderNullId) {
		if (ParentFolderId == _ParentFolderId) {
			return True;
		}
		// There is a loop in the parent hierarchy, escape it
		if (TraversedFolderIdList.exists(ParentFolderId)) {
			return False;
		}

		TraversedFolderIdList.add(ParentFolderId);
		ParentFolderId = _FolderReverseTree.get(ParentFolderId, ClubStruct::C_FolderNullId);
	}

	return ParentFolderId == _ParentFolderId;
}

ClubStruct::LibStructuresClub_K_Folder[Integer] Private_FilterFolderList(
	ClubStruct::LibStructuresClub_K_Folder[] _FolderList,
	ClubStruct::LibStructuresClub_K_Activity _ActivityToMove
) {
	declare ClubStruct::LibStructuresClub_K_Folder[Integer] FilteredFolderList;

	if (_ActivityToMove.ParentFolderId != ClubStruct::C_FolderNullId) {
		FilteredFolderList = [
			0 => ClubStruct::LibStructuresClub_K_Folder {
				Id = ClubStruct::C_FolderNullId,
				ParentFolderId = ClubStruct::C_FolderNullId,
				Name = C_Private_Text_FolderHomePage,
				MediaUrl = "",
				CreatorAccountId = "",
				LatestEditorAccountId = ""
			}
		];
	}

	/// Filter the folder list to prevent the creation of parent-child loop. A includes B includes C includes A for example.
	if (_ActivityToMove.Type == ClubStruct::C_ClubActivity_Folder) {
		// Link child to parent
		declare Integer[Integer] FolderReverseTree;
		for (Folder in _FolderList) {
			FolderReverseTree[Folder.Id] = Folder.ParentFolderId;
		}

		for (Folder in _FolderList) {
			if (
				_ActivityToMove.ParentFolderId != Folder.Id &&
				!Private_IsParentOf(FolderReverseTree, _ActivityToMove.Id, Folder.Id)
			) {
				FilteredFolderList[FilteredFolderList.count] = Folder;
			}
		}
	} else {
		for (Folder in _FolderList) {
			FilteredFolderList[FilteredFolderList.count] = Folder;
		}
	}

	return FilteredFolderList;
}
