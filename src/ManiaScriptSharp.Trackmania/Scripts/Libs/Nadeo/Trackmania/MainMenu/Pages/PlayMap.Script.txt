/// Get the page manialink
Text GetManialink() {
	//L16N [PlayMap] Title of a page where the player can select a track and play on it
	declare Text Text_PlayMap = _("Play a track");

	return MV::Create(
C_Name, 3,
"""
<frameinstance modelid="{{{TrackBrowser::C_Name}}}"
	id="frame-track-browser"
	z-index="{{{Const::C_ZIndex_BrowserAPI}}}"
	data-show-only-playable="1"
/>
""",
"""
#Include "Libs/Nadeo/CMGame/Utils/EnvironmentInfo.Script.txt" as EnvironmentInfo
#Include "Libs/Nadeo/CMGame/Utils/Tools.Script.txt" as Tools
#Include "Libs/Nadeo/Trackmania/MainMenu/TitleControl.Script.txt" as TitleControl
#Include "Libs/Nadeo/Trackmania/Modes/Stunt/Constants.Script.txt" as StuntConst
#Include "Libs/Nadeo/Trackmania/Stores/LoadingScreen.Script.txt" as LoadingScreenStore
#Include "TextLib" as TL

#Const C_RoyalDecalUrl "file://Media/Manialinks/Nadeo/Trackmania/Modes/Royal/Decal_Royal.dds"
""",
"""
***MainLoop***
***
if ({{{Router::P}}}PageIsRunning(This, "{{{RoutePaths::C_Path_PlayMap}}}")) {
	foreach (Event in {{{Router::P}}}GetPendingEvents(This)) {
		switch (Event.Type) {
			case {{{Router::P}}}C_Event_EnteringRoute: {
				declare Text[Text] Query = {{{Router::P}}}GetCurrentRouteQuery(This);
				if (!Query.existskey("{{{RoutePaths::C_Query_LeaveClubEditioninfo}}}")) {
					{{{TrackBrowser::P}}}Display(This, True);
					{{{TrackBrowser::P}}}SetMapTypeFilters(This, [
						"{{{Const::C_MapType_Race}}}",
						"{{{Const::C_MapType_Royal}}}",
						"{{{Const::C_MapType_Stunt}}}",
						"{{{Const::C_MapType_Platform}}}"
					]);
				}
			}
		}
	}

	foreach (Event in {{{TrackBrowser::P}}}GetPendingEvents(This)) {
		switch (Event.Type) {
			case {{{TrackBrowser::P}}}C_Event_Exit: {{{Router::P}}}PushParent(This);
			case {{{TrackBrowser::P}}}C_Event_Select: {
				if (Event.Track.IsPlayable && EnvironmentInfo::IsPlayable(Event.Track.CollectionName)) {
					declare Text MapPath = Tools::Ternary(Event.Track.DownloadUrl != "", Event.Track.DownloadUrl, Event.Track.LocalFileName);

					switch (Event.Track.Type) {
						case "{{{Const::C_MapType_Race}}}": {
							LoadingScreenStore::UpdateTitleMenuLoadingScreen_Random(This, Event.Track.Name);
							LoadingScreenStore::StartTrackingLoadingScreen(This, LoadingScreenStore::GetCurrentTitleMenuLoadingScreen(This));
							TitleControl::PlayMap(TitleControl, MapPath, "{{{Const::C_ModeScript_PlayMap}}}", "");
						}
						case "{{{Const::C_MapType_Royal}}}": {
							LoadingScreenStore::UpdateTitleMenuLoadingScreen_Random(This, Event.Track.Name);
							LoadingScreenStore::StartTrackingLoadingScreen(This, LoadingScreenStore::GetCurrentTitleMenuLoadingScreen(This));
							TitleControl::PlayMap(TitleControl, MapPath, "{{{Const::C_ModeScript_RoyalSolo}}}", TL::Join("", [
								"<root>",
									"<setting name=\"S_DecoImageUrl_Checkpoint\" value=\""^TL::MLEncode(C_RoyalDecalUrl)^"\" type=\"text\"/>",
								"</root>"
							]));
						}
						case "{{{Const::C_MapType_Stunt}}}": {
							LoadingScreenStore::UpdateTitleMenuLoadingScreen_Random(This, Event.Track.Name);
							LoadingScreenStore::StartTrackingLoadingScreen(This, LoadingScreenStore::GetCurrentTitleMenuLoadingScreen(This));
							TitleControl::PlayMap(TitleControl, MapPath, "{{{Const::C_ModeScript_StuntSolo}}}", TL::Join("", [
								"<root>",
									"<setting name=\"S_DecoImageUrl_Checkpoint\" value=\""^TL::MLEncode(StuntConst::C_ImageUrl_StuntLogo)^"\" type=\"text\"/>",
								"</root>"
							]));
						}
						case "{{{Const::C_MapType_Platform}}}": {
							LoadingScreenStore::UpdateTitleMenuLoadingScreen_Random(This, Event.Track.Name);
							LoadingScreenStore::StartTrackingLoadingScreen(This, LoadingScreenStore::GetCurrentTitleMenuLoadingScreen(This));
							TitleControl::PlayMap(TitleControl, MapPath, "{{{Const::C_ModeScript_PlatformSolo}}}", TL::Join("", [
								"<root>",
									"<setting name=\"S_DecoImageUrl_Checkpoint\" value=\""^TL::MLEncode("{{{Const::C_ImageUrl_PlatformLogo}}}")^"\" type=\"text\"/>",
								"</root>"
							]));
						}
					}

					{{{TrackBrowser::P}}}ReFocus(This);
				}
			}
		}
	}
}
***
""",
[
	Transition::ComponentV2(),
	Navigation::ComponentV2(),
	Router::ComponentV2(),
	TrackBrowser::ComponentV2(),
	MenuSounds::ComponentV2()
],
[]
	);
}
