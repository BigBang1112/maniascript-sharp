Void SetLabelOpacity(CMlLabel _Label, Boolean _IsFocused) {
	AnimMgr.Flush(_Label);
	if (_IsFocused) {
		AnimMgr.Add(_Label, "<anim opacity=\"1.\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	} else {
		AnimMgr.Add(_Label, "<anim opacity=\"0.8\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	}
}

Void Focus(CMlControl _From, CMlControl _To) {
	if ({{{Transition::P}}}IsInTransition(This)) return;

	if (_From != Null) {
		{{{Navigation::P}}}Unfocus(This, _From);
		if ({{{ExpendableButton::P}}}IsButton(_From)) {
			{{{ExpendableButton::P}}}Unfocus(This, (_From as CMlFrame));
		} else if ({{{Button::P}}}IsButton(_From)) {
			{{{Button::P}}}Unfocus(This, (_From as CMlFrame));
			if (_From.DataAttributeExists("secondarylabel")) {
				SetLabelOpacity(_From.Parent.GetFirstChild(_From.DataAttributeGet("secondarylabel")) as CMlLabel, False);
			}
		}
	}

	if (_To != Null) {
		{{{Navigation::P}}}Focus(This, _To);
		if ({{{ExpendableButton::P}}}IsButton(_To)) {
			{{{ExpendableButton::P}}}Focus(This, (_To as CMlFrame));
		} else if ({{{Button::P}}}IsButton(_To)) {
			{{{Button::P}}}Focus(This, (_To as CMlFrame));
			if (_To.DataAttributeExists("secondarylabel")) {
				SetLabelOpacity(_To.Parent.GetFirstChild(_To.DataAttributeGet("secondarylabel")) as CMlLabel, True);
			}
		}
	}
}

K_State UpdateStep(K_State _State, Integer _Step) {
	declare K_State State = _State;
	declare CMlControl[CMlScriptEvent::EMenuNavAction] NavigationMemory for State.Controls.Button_Back;
	declare CMlFrame Frame_CreateType = (State.Controls.Frame_Global.GetFirstChild("frame-create-type") as CMlFrame);
	declare CMlFrame Frame_Controller = (State.Controls.Frame_Global.GetFirstChild("frame-controller") as CMlFrame);
	declare CMlFrame Frame_DifficultyMouse = (State.Controls.Frame_Global.GetFirstChild("frame-difficulty-mouse") as CMlFrame);
	declare CMlFrame Frame_DifficultyController = (State.Controls.Frame_Global.GetFirstChild("frame-difficulty-gamepad") as CMlFrame);
	declare CMlFrame Frame_Enviro = (State.Controls.Frame_Global.GetFirstChild("frame-enviro") as CMlFrame);
	declare CMlFrame Frame_Mood = (State.Controls.Frame_Global.GetFirstChild("frame-mood") as CMlFrame);
	declare Boolean BackButtonFocused = (
		{{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroup_Page}}}") != Null &&
		{{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroup_Page}}}").ControlId == State.Controls.Button_Back.ControlId
	);

	// Reset navigation
	NavigationMemory = [];
	foreach (Step => NavGroup in C_SettingsStep_NavGroups) {
		{{{Navigation::P}}}EnableGroup(This, NavGroup, Step == _Step);
	}

	// Frames visibility
	State.Controls.Frame_MainMenu.Visible = _Step == C_SettingsStep_MainMenu;
	Frame_CreateType.Visible = _Step == C_SettingsStep_CreateType;
	Frame_Controller.Visible = _Step == C_SettingsStep_Controller;
	Frame_DifficultyMouse.Visible = _Step == C_SettingsStep_DifficultyMouse;
	Frame_DifficultyController.Visible = _Step == C_SettingsStep_DifficultyGamepad;
	Frame_Enviro.Visible = _Step == C_SettingsStep_Enviro;
	Frame_Mood.Visible = _Step == C_SettingsStep_Mood;
	if (_Step != C_SettingsStep_BrowseMaps) {
		DisplayTrackBrowser(State, False, False);
	}

	// Update step
	State.CurrentStep = _Step;
	switch (State.CurrentStep) {
		case C_SettingsStep_MainMenu: {
			Focus({{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroup_Page}}}"), State.Controls.Button_Create);
			{{{Navigation::P}}}ChangeControlTarget(This, State.Controls.Button_Back, "up", State.Controls.Button_Create.ControlId);
			{{{Navigation::P}}}ChangeControlTarget(This, State.Controls.Button_Back, "right", State.Controls.Button_Create.ControlId);

			{{{Navigation::P}}}ChangeControlTarget(This, State.Controls.Button_Back, "right", State.Controls.Button_CreateStunt.ControlId);

			} else {
				declare CMlControl Button_Mouse = Frame_Controller.GetFirstChild("button-mouse");
				Focus({{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroup_Page}}}"), Button_Mouse);
				{{{Navigation::P}}}ChangeControlTarget(This, State.Controls.Button_Back, "up", Button_Mouse.ControlId);
				{{{Navigation::P}}}ChangeControlTarget(This, State.Controls.Button_Back, "right", Button_Mouse.ControlId);
			}
		}
		case C_SettingsStep_DifficultyMouse: {
			declare CMlControl Button_MouseSimple = Frame_DifficultyMouse.GetFirstChild("button-difficulty-mouse-simple");
			Focus({{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroup_Page}}}"), Button_MouseSimple);
			{{{Navigation::P}}}ChangeControlTarget(This, State.Controls.Button_Back, "up", Button_MouseSimple.ControlId);
			{{{Navigation::P}}}ChangeControlTarget(This, State.Controls.Button_Back, "right", Button_MouseSimple.ControlId);
		}
		case C_SettingsStep_DifficultyGamepad: {
			Focus({{{Navigation::P}}}GetFocusedControl(This, "{{{C_NavGroup_Page}}}"), Frame_DifficultyController.GetFirstChild("button-difficulty-gamepad-simple"));

K_State Cancel(K_State _State) {
	declare K_State State = _State;

	switch (State.CurrentStep) {
		case C_SettingsStep_MainMenu: {{{Router::P}}}PushParent(This);
		case C_SettingsStep_BrowseMaps: State = UpdateStep(State, C_SettingsStep_MainMenu);
		case C_SettingsStep_CreateType: State = UpdateStep(State, C_SettingsStep_MainMenu);
		case C_SettingsStep_Controller: {
			if (
				{{{Router::P}}}GetCurrentParentPath(This) == "{{{RoutePaths::C_Path_ServerReview}}}" ||
				{{{Router::P}}}GetCurrentParentPath(This) == "{{{RoutePaths::C_Path_LocalMultiSettings}}}"
			) {
				{{{Router::P}}}PushParent(This);
			} else if ({{{Router::P}}}GetCurrentParentPath(This) == "{{{RoutePaths::C_Path_CreateCampaign}}}") {
				declare CampaignStruct::LibCampaignStruct_K_Map Map;
				declare {{{Router::P}}}K_URL_Config Config = {{{Router::P}}}K_URL_Config {
					SaveHistory = True,
					ResetPreviousPagesDisplayed = True,
					KeepPreviousPagesDisplayed = False,
					HidePreviousPage = True,
					ShowParentPage = True
				};
				{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_CreateCampaign}}}", ["{{{RoutePaths::C_Query_Map}}}" => Map.tojson()], Config);
			} else if (Platform::IsFromFirstParty(System)) {
				State = UpdateStep(State, C_SettingsStep_MainMenu);
			} else {
				State = UpdateStep(State, C_SettingsStep_CreateType);
			}
		}
		case C_SettingsStep_DifficultyMouse: {
			if (TitleControl::IsStuntMapType(State.LaunchParams.MapType) || TitleControl::IsPlatformMapType(State.LaunchParams.MapType)) {
				State = UpdateStep(State, C_SettingsStep_CreateType);
			} else {
				State = UpdateStep(State, C_SettingsStep_Controller);
			}
		}
		case C_SettingsStep_DifficultyGamepad: {
			if (Platform::IsFromFirstParty(System)) {
				State.CurrentStep = C_SettingsStep_Controller;
				State = Cancel(State);
			} else {
				State = UpdateStep(State, C_SettingsStep_Controller);
			}
		}
		case C_SettingsStep_Enviro: {
			if (TitleControl::IsRoyalMapType(State.LaunchParams.MapType)) {
				State = UpdateStep(State, C_SettingsStep_CreateType);
			} else if (State.LaunchParams.Controller == C_Controller_Mouse) {
				State = UpdateStep(State, C_SettingsStep_DifficultyMouse);
			} else {
				State = UpdateStep(State, C_SettingsStep_DifficultyGamepad);
			}
		}
		case C_SettingsStep_Mood: {
			State = UpdateStep(State, C_SettingsStep_Enviro);
		}
	}

	return State;
}

K_State LaunchEditor(K_State _State) {
	declare K_State State = _State;
	declare Text[] EditorPluginScripts;
	declare Text[] EditorPluginSettings;

	// When opening the gamepad editor, we don't want to allow plugins that were saved in the profile
	// opened in previous Map Editor session to be opened alongside GamepadTrackEditor.Script.txt
	declare Boolean	OnlyUseForcedPlugins = TitleControl::IsRaceMapType(State.LaunchParams.MapType) && State.LaunchParams.Controller == C_Controller_Gamepad;

	// Add the 'PlayMapAtSave' plugin
	EditorPluginScripts.add("TrackMania/Nadeo/Trackmania/PlayMapAtSave.Script.txt");
	if (
		{{{Router::P}}}GetCurrentParentPath(This) == "{{{RoutePaths::C_Path_CreateCampaign}}}" ||
		TitleControl::IsStuntMapType(State.LaunchParams.MapType) ||
		TitleControl::IsPlatformMapType(State.LaunchParams.MapType) ||
		TitleControl::IsRoyalMapType(State.LaunchParams.MapType)
	) {
		EditorPluginSettings.add("<root><setting name=\"S_ShowMenuAfterSave\" value=\""^False^"\" type=\"boolean\"/></root>");
	} else {
		EditorPluginSettings.add("");
	}

	// Add the 'GamepadTrackEditor' plugin (only available for race maps)
	if (State.LaunchParams.Controller == C_Controller_Gamepad && TitleControl::IsRaceMapType(State.LaunchParams.MapType)) {
		EditorPluginScripts.add("TrackMania/Nadeo/TMEditor/GamepadTrackEditor.Script.txt");
		EditorPluginSettings.add("<root><setting name=\"S_EditorLevel\" value=\""^State.LaunchParams.Difficulty^"\" type=\"integer\"/></root>");
	}

	// Add the 'ColorAndAnimation' plugin (except in the Gamepad Editor)
	if (State.LaunchParams.Controller != C_Controller_Gamepad) {
		EditorPluginScripts.add("TrackMania/Nadeo/TMGame/ColorAndAnimation.Script.txt");
		EditorPluginSettings.add("");
	}

	// Add the 'TrackSettings' plugin only for race map with PC Editor
	if (
		State.LaunchParams.Controller != C_Controller_Gamepad &&
		TitleControl::IsRaceMapType(State.LaunchParams.MapType)
	) {
		EditorPluginScripts.add("TrackMania/Nadeo/Trackmania/TrackSettings.Script.txt");
		EditorPluginSettings.add("");
	}

	// Update the loading screen
	declare Text LoadingScreenTitle = Tools::Ternary(State.LaunchParams.MapToEdit.Name != "", State.LaunchParams.MapToEdit.Name, "{{{C_Text_TrackEditor}}}");
	LoadingScreenStore::UpdateTitleMenuLoadingScreen_Random(This, LoadingScreenTitle);

	// Enforce a valid enviro
	if (State.LaunchParams.Enviro == C_Enviro_Ask) {
		State.LaunchParams.Enviro = C_Enviro_Stadium;
	}

	// Launch the editor
	Share::ResetLastMapCreated();
	if (State.LaunchParams.MapToEdit.Uid != "") {
		declare Text FileName = State.LaunchParams.MapToEdit.LocalFileName;
		if (FileName == "") FileName = State.LaunchParams.MapToEdit.DownloadUrl;
		TitleControl::EditMap(
			TitleControl,
			FileName,
			"",
			"",
			EnvironmentInfo::C_PlayerModel_CarSport,
			EditorPluginScripts,
			EditorPluginSettings,
			State.LaunchParams.Difficulty != C_Difficulty_Simple,
			OnlyUseForcedPlugins
		);
	} else if (TitleControl::IsRoyalMapType(State.LaunchParams.MapType)) {
		declare Text BaseMap = "RoyalBaseMap.Map.Gbx";
		declare Text Mood = EnvironmentInfo::GetCollectionMood(State.LaunchParams.Enviro, State.LaunchParams.Mood);
		switch (State.LaunchParams.Enviro) {
			case EnvironmentInfo::C_Collection_Stadium: {
				BaseMap = "RoyalBaseMap.Map.Gbx";
			}
			case EnvironmentInfo::C_Collection_RedIsland: {
				BaseMap = "RoyalBaseMapRedIsland.Map.Gbx";
			}
			case EnvironmentInfo::C_Collection_GreenCoast: {
				BaseMap = "RoyalBaseMapGreenCoast.Map.Gbx";
			}
			case EnvironmentInfo::C_Collection_BlueBay: {
				BaseMap = "RoyalBaseMapBlueBay.Map.Gbx";
			}
			case EnvironmentInfo::C_Collection_WhiteShore: {
				BaseMap = "RoyalBaseMapWhiteShore.Map.Gbx";
			}
			default: {
				BaseMap = "RoyalBaseMap.Map.Gbx";
				Mood = EnvironmentInfo::GetCollectionMood(EnvironmentInfo::C_Collection_Stadium, State.LaunchParams.Mood);
			}
		}
		TitleControl::EditNewMapFromBaseMap(
			TitleControl,
			BaseMap,
			Mood,
			"",
			EnvironmentInfo::C_PlayerModel_CarSport,
			State.LaunchParams.MapType,
			EditorPluginScripts,
			EditorPluginSettings,
			OnlyUseForcedPlugins
		);
	} else {
		// Force stadium environment with mouse simple editor
		declare Boolean IsMouseSimpleEditor = (State.LaunchParams.Controller == C_Controller_Mouse && State.LaunchParams.Difficulty == C_Difficulty_Simple);
		if (IsMouseSimpleEditor) {
			State.LaunchParams.Enviro = C_Enviro_Stadium;
		}
		TitleControl::EditNewMap(
			TitleControl,
			State.LaunchParams.Enviro,
			EnvironmentInfo::GetCollectionMood(State.LaunchParams.Enviro, State.LaunchParams.Mood),
			"",
			EnvironmentInfo::C_PlayerModel_CarSport,
			State.LaunchParams.MapType,
			IsMouseSimpleEditor,
			EditorPluginScripts,
			EditorPluginSettings,
			OnlyUseForcedPlugins
		);
	}

	// Coming back from the editor
	declare Share::K_MapInfo MapInfo = Share::GetLastMapCreated();
	DisplayTrackBrowser(State, False, False);
	if (Share::GetGoToServerReview()) {
		if ({{{UserStore::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_AccessServerReview)) {
			declare CampaignStruct::LibCampaignStruct_K_Map Map = Share::ConvertToMap(MapInfo);
			{{{Router::P}}}SetParentPath(This, "{{{RoutePaths::C_Path_ServerReview}}}", {{{Router::P}}}GetCurrentRoutePath(This));
			{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_ServerReview}}}", ["{{{RoutePaths::C_Query_Map}}}" => Map.tojson(), "{{{RoutePaths::C_Query_IsWeeklyTracksReview}}}" => ""^Share::GetGoToWeeklyTracksReview()]);
		} else {
			ClubEdition::GoToPage(This, "{{{RoutePaths::C_Path_Create}}}", RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_AccessServerReview));
		}
	} else if (Share::GetGoToHotSeat()) {
		if ({{{UserStore::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_PlayHotSeat)) {
			declare CampaignStruct::LibCampaignStruct_K_Map Map = Share::ConvertToMap(MapInfo);
			{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_LocalMultiSettings}}}", ["{{{RoutePaths::C_Query_Map}}}" => Map.tojson(), "{{{RoutePaths::C_Query_LocalMultiType}}}" => "{{{Const::C_LocalMultiType_HotSeat}}}"]);
		} else {
			ClubEdition::GoToPage(This, "{{{RoutePaths::C_Path_Local}}}", RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_PlayHotSeat));
		}
	} else if ({{{Router::P}}}GetCurrentParentPath(This) == "{{{RoutePaths::C_Path_CreateCampaign}}}") {
		declare CampaignStruct::LibCampaignStruct_K_Map Map = Share::ConvertToMap(MapInfo);
		declare {{{Router::P}}}K_URL_Config Config = {{{Router::P}}}K_URL_Config {
			SaveHistory = True,
			ResetPreviousPagesDisplayed = True,
			KeepPreviousPagesDisplayed = False,
			HidePreviousPage = True,
			ShowParentPage = True
		};
		{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_CreateCampaign}}}", ["{{{RoutePaths::C_Query_Map}}}" => Map.tojson()], Config);
	} else {
		if (MapInfo.Uid != "") {
			State = UpdateStep(State, C_SettingsStep_BrowseMaps);
		} else {
			State = UpdateStep(State, C_SettingsStep_MainMenu);
		}
	}
	Share::ResetLastMapCreated();

	return State;
}

K_State Select(K_State _State, CMlControl _SelectedControl) {
	if (_SelectedControl == Null || {{{Transition::P}}}IsInTransition(This)) return _State;
	declare K_State State = _State;

	switch (_SelectedControl.ControlId) {
		case "button-back": State = Cancel(State);
		case "button-edit": State = UpdateStep(State, C_SettingsStep_BrowseMaps);
		case "button-club-items": {
			{{{Router::P}}}SetParentPath(This, "{{{RoutePaths::C_Path_LibraryClubItems}}}", {{{Router::P}}}GetCurrentRoutePath(This));
			{{{Router::P}}}Push(This, "{{{RoutePaths::C_Path_LibraryClubItems}}}");
		}
		case "button-create": {
			if (DataFileMgr.UserSave_AvailableSpace < C_MinSpaceNeededForTrackCreationInkB) {
				OpenDiskSpacePopUp(_State.Controls.Frame_MultiChoice);
			} else {
				State.LaunchParams = K_LaunchParams {
					Controller = GetQuickStartEditorGamepad(),
					Difficulty = GetQuickStartEditorDifficulty(),
					MapType = GetQuickStartEditorMapType(),
					Enviro = GetQuickStartEditorEnviro(),
					Mood = GetQuickStartEditorMood(),
					MapToEdit = CampaignStruct::LibCampaignStruct_K_Map {}
				};
				if (IsQuickStartEditorEnabled() && State.LaunchParams.MapType != "") {
					if (State.LaunchParams.Enviro != C_Enviro_Ask) {
						State = LaunchEditor(State);
					} else {
						State = UpdateStep(State, C_SettingsStep_Enviro);
					}
				} else if (Platform::IsFromFirstParty(System)) {
					State = Select(State, State.Controls.Button_CreateRace);
				} else {
					State = UpdateStep(State, C_SettingsStep_CreateType);
				}
			}
		}
		case "button-create-race": {
			State.LaunchParams.MapType = "{{{Const::C_MapType_Race}}}";
			if (IsQuickStartEditorEnabled()) {
				if (State.LaunchParams.Enviro != C_Enviro_Ask) {
					State = LaunchEditor(State);
				} else {
					State = UpdateStep(State, C_SettingsStep_Enviro);
				}
			} else {
				State = UpdateStep(State, C_SettingsStep_Controller);
			}
		}
		case "button-create-royal": {
			State.LaunchParams.MapType = "{{{Const::C_MapType_Royal}}}";
			if (IsQuickStartEditorEnabled()) {
				if (State.LaunchParams.Enviro != C_Enviro_Ask) {
					State = LaunchEditor(State);
				} else {
					State = UpdateStep(State, C_SettingsStep_Enviro);
				}
			} else if ({{{UserStore::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_OpenAdvancedMapEditor)) {
				State = UpdateStep(State, C_SettingsStep_Enviro); //< Simple editor not available when creating from base map
			} else {
				ClubEdition::GoToPage(This, {{{Router::P}}}GetCurrentRoutePath(This), RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_OpenAdvancedMapEditor));
			}
		}
		case "button-create-stunt": {
			State.LaunchParams.MapType = "{{{Const::C_MapType_Stunt}}}";
			State.LaunchParams.Controller = C_Controller_Mouse;
			if (IsQuickStartEditorEnabled()) {
				if (State.LaunchParams.Enviro != C_Enviro_Ask) {
					State = LaunchEditor(State);
				} else {
					State = UpdateStep(State, C_SettingsStep_Enviro);
				}
			} else {
				State = UpdateStep(State, C_SettingsStep_DifficultyMouse);
			}
		}
		case "button-create-platform": {
			State.LaunchParams.MapType = "{{{Const::C_MapType_Platform}}}";
			State.LaunchParams.Controller = C_Controller_Mouse;
			if (IsQuickStartEditorEnabled()) {
				if (State.LaunchParams.Enviro != C_Enviro_Ask) {
					State = LaunchEditor(State);
				} else {
					State = UpdateStep(State, C_SettingsStep_Enviro);
				}
			} else {
				State = UpdateStep(State, C_SettingsStep_DifficultyMouse);
			}
		}
		case "button-mouse": {
			State.LaunchParams.Controller = C_Controller_Mouse;
			// Editing a map done in advanced editor
			if (State.LaunchParams.MapToEdit.Uid != "" && !State.LaunchParams.MapToEdit.CreatedWithSimpleEditor) {
				if (!{{{UserStore::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_OpenAdvancedMapEditor)) {
					ClubEdition::GoToPage(This, {{{Router::P}}}GetCurrentRoutePath(This), RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_OpenAdvancedMapEditor));
				} else {
					State = UpdateStep(State, C_SettingsStep_Mood);
				}
			} else {
				State = UpdateStep(State, C_SettingsStep_DifficultyMouse);
			}
		}
		case "button-gamepad": {
			State.LaunchParams.Controller = C_Controller_Gamepad;
			State = UpdateStep(State, C_SettingsStep_DifficultyGamepad);
		}
		default: {
			if (_SelectedControl.DataAttributeExists("difficulty")) {
				State.LaunchParams.Difficulty = TL::ToInteger(_SelectedControl.DataAttributeGet("difficulty"));
				if (State.LaunchParams.Difficulty == C_Difficulty_Simple && {{{UserStore::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_OpenSimpleMapEditor)) {
					if (State.LaunchParams.Controller == C_Controller_Mouse) {
						State = LaunchEditor(State);
					} else {
						State = UpdateStep(State, C_SettingsStep_Enviro);
					}
				} else if (State.LaunchParams.Difficulty != C_Difficulty_Simple && {{{UserStore::P}}}HasGamePermission(PrivilegeMgr, RBAC::C_GamePermission_OpenAdvancedMapEditor)) {
					State = UpdateStep(State, C_SettingsStep_Enviro);
				} else {
					if (State.LaunchParams.Difficulty == C_Difficulty_Simple) {
						ClubEdition::GoToPage(This, {{{Router::P}}}GetCurrentRoutePath(This), RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_OpenSimpleMapEditor));
					} else {
						ClubEdition::GoToPage(This, {{{Router::P}}}GetCurrentRoutePath(This), RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_OpenAdvancedMapEditor));
					}
				}
			} else if (_SelectedControl.DataAttributeExists("enviro")) {
				State.LaunchParams.Enviro = _SelectedControl.DataAttributeGet("enviro");
				if (IsQuickStartEditorEnabled()) {
					State = LaunchEditor(State);
				} else {
					State = UpdateStep(State, C_SettingsStep_Mood);
				}
			} else if (_SelectedControl.DataAttributeExists("mood")) {
				State.LaunchParams.Mood = _SelectedControl.DataAttributeGet("mood");
				State = LaunchEditor(State);
			}
		}
	}

	return State;
}
