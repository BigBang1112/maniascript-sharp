/// Update the campaign button images
Void Private_UpdateCampaignMedia(CMlScript _Context, K_Private_Instance _Instance) {
	declare Text ShortsMediaUrl = CampaignStore::GetMediaActive(CampaignStore::C_CampaignMediaType_SoloWeeklyShorts).Url;
	if (ShortsMediaUrl != "") {
		Button::SetImage(_Context, _Instance.Controls.Button_WeeklyShorts, ShortsMediaUrl);
	} else {
		Button::SetImage(_Context, _Instance.Controls.Button_WeeklyShorts, C_Private_ImageUrl_ButtonWeeklyShortsImage);
	}

	declare Text ButtonImageUrl = C_Private_ImageUrl_ButtonWeeklyGrandsImage;
	declare Text GrandsMediaActiveUrl = CampaignStore::GetMediaActive(CampaignStore::C_CampaignMediaType_SoloWeeklyGrands).Url;
	if (GrandsMediaActiveUrl != "") {
		ButtonImageUrl = GrandsMediaActiveUrl;
	} else {
		declare Text LiveWeeklyGrandsCampaignMediaUrl = CampaignStore::GetLiveWeeklyGrandsCampaign().MediaUrl;
		if (LiveWeeklyGrandsCampaignMediaUrl != "") {
			ButtonImageUrl = LiveWeeklyGrandsCampaignMediaUrl;
		}
	}
	Button::SetImage(_Context, _Instance.Controls.Button_WeeklyGrands, ButtonImageUrl);
}

/// Get the worst medal of the campaign
Integer Private_GetCampaignMedal(CampaignStruct::LibCampaignStruct_K_Campaign _Campaign) {
	declare Integer WorstMedal = Medals::C_ScoreMgrMedal_Author;
	declare Text GameMode = MainMenuConst::C_GameMode_TimeAttack;
	if (_Campaign.Type == CampaignStruct::C_CampaignType_WeeklyGrands) {
		GameMode = MainMenuConst::C_GameMode_CampaignClone;
	}
	foreach (Map in _Campaign.Playlist) {
		declare Integer Medal = ScoreMgr.Map_GetMedal(
			MainUser::GetMainUserId(),
			Map.Uid,
			MainMenuConst::C_ScopeType_PersonalBest,
			"",
			GameMode,
			""
		);
		if (WorstMedal > Medal) {
			WorstMedal = Medal;
			if (WorstMedal <= Medals::C_ScoreMgrMedal_None) {
				break;
			}
		}
	}
	return WorstMedal;
}

/// Set the campaign medal stack
Void Private_SetCampaignMedalStack(Integer _CampaignId, CMlFrame _MedalStack_Campaign) {
	if (_CampaignId == CampaignStruct::C_Campaign_NullId) {
		MedalStack::SetMedalsNumber(_MedalStack_Campaign, 0);
	} else {
		MedalStack::SetMedalsNumber(_MedalStack_Campaign, Private_GetCampaignMedal(CampaignStore::GetCampaignFromId(_CampaignId)));
	}
}

/// Set the season info rank
Void Private_SetSeasonInfoRank(K_Private_SeasonInfo _SeasonInfo, CampaignStruct::LibCampaignStruct_K_Campaign _Campaign) {
	declare CampaignStruct::LibCampaignStruct_K_Ranking TopRank = CampaignStore::GetTopRank(_Campaign.CampaignRanking.Rankings);
	if (TopRank.ZoneName != "" && TopRank.Position > 0) {
		_SeasonInfo.Label_RankZone.Value = C_Private_ZoneTranslationList.get(TopRank.ZoneName, TL::GetTranslatedText(TopRank.ZoneName));
		// Display a rank suffix only if player in Top 10 (after that the player rank will be displayed like "1K")
		if (TopRank.Position < C_Private_SuffixFormatLimit) {
			_SeasonInfo.Label_RankPosition.Value = TL::FormatRank(TopRank.Position, False);
		} else if (TopRank.Position < C_Private_FullFormatLimit) {
			_SeasonInfo.Label_RankPosition.Value = TL::ToText(TopRank.Position);
		} else {
			_SeasonInfo.Label_RankPosition.Value = Tools::FormatTop(TopRank.Position, False, C_Private_FullFormatLimit);
		}
	} else {
		_SeasonInfo.Label_RankPosition.Value = "";
		_SeasonInfo.Label_RankZone.Value = C_Private_Text_NotRanked;
	}

	MLHelpers::FitLabelValue(_SeasonInfo.Label_RankZone, C_Private_TextSize_RankZone);
	_SeasonInfo.Label_RankZone.RelativePosition_V3.X = _SeasonInfo.Label_RankPosition.RelativePosition_V3.X + MLHelpers::ComputeConstrainedLabelValueWidth(_SeasonInfo.Label_RankPosition);
}

/// Set the season info timer
Void Private_SetSeasonInfoTimer(K_Private_SeasonInfo _SeasonInfo) {
	if (_SeasonInfo.EndTime >= 0) {
		declare Integer TimeLeft = _SeasonInfo.EndTime - Now;
		declare Integer DayDuration = 24 * 60 * 60 * 1000;
		declare Integer Days = TimeLeft / DayDuration;
		if (Days >= 1) { // More than a day
			if (Days > 1) {
				//L16N Message displayed to indicate when the new season starts. %1 will be replaced by a number, e.g "23 days"
				_SeasonInfo.Label_Timer.Value = TL::Compose(_("%1 days"), ""^Days);
			} else {
				declare Integer HourDuration = 60 * 60 * 1000;
				declare Integer Hours = ((TimeLeft % DayDuration) / HourDuration) + 1; // +1 to avoid "and 0 hours"
				if (Hours > 1) {
					//L16N Message displayed to indicate when the new season starts. %1 and %2 will be replaced by a number, e.g "1 day and 12 hours"
					_SeasonInfo.Label_Timer.Value = TL::Compose(_("%1 day and %2 hours"), ""^Days, ""^Hours);
				} else {
					//L16N Message displayed to indicate when the new season starts. %1 will be replaced by a number, e.g "1 day and 1 hour"
					_SeasonInfo.Label_Timer.Value = TL::Compose(_("%1 day and %2 hour"), ""^Days, ""^Hours);
				}
			}
		} else if (TimeLeft > 0) {
			//L16N Message displayed to indicate when the new season starts. %1 will be replaced by a timer, e.g "12:10:53"
			_SeasonInfo.Label_Timer.Value = TL::Compose(_("%1"), TL::TimeToText(TimeLeft + 1000, False)); // Add 1 second on the timer to avoid "-0:00" on screen
		} else {
			_SeasonInfo.Label_Timer.Value = C_Private_Text_Now;
		}
	} else {
		_SeasonInfo.Label_Timer.Value = "";
	}
}

		switchtype (Control as TypedControl) {
			case CMlLabel: {
				_Context.AnimMgr.Flush(TypedControl);
				_Context.AnimMgr.Add(TypedControl, """<anim opacity="{{{_Opacity}}}" />""", _StartTime, _Duration, CAnimManager::EAnimManagerEasing::Linear);
			}
			case CMlQuad: {
				_Context.AnimMgr.Flush(TypedControl);
				_Context.AnimMgr.Add(TypedControl, """<anim opacity="{{{_Opacity}}}" />""", _StartTime, _Duration, CAnimManager::EAnimManagerEasing::Linear);
			}
		}
	}
}

/// Set the visibility of the season info frame
Void Private_SetIsSeasonInfoVisible(CMlScript _Context, CMlFrame _Frame_SeasonInfo, Boolean _IsVisible, Boolean _IsFocused) {
	_Context.AnimMgr.Flush(_Frame_SeasonInfo);
	if (_IsVisible) {
		_Context.AnimMgr.Add(_Frame_SeasonInfo, """<anim pos="{{{C_Private_Pos_FrameSeasonInfoEnd.X}}} {{{C_Private_Pos_FrameSeasonInfoEnd.Y}}}" hidden="0" />""", Now + C_Private_SeasonInfoHideAnimDuration, C_Private_SeasonInfoShowAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
		declare Real Opacity = C_Private_Opacity_Unfocus;
		if (_IsFocused) Opacity = C_Private_Opacity_Focus;
		Private_SetSeasonInfoOpacity(_Context, _Frame_SeasonInfo, Opacity, Now + C_Private_SeasonInfoHideAnimDuration, C_Private_SeasonInfoShowAnimDuration);
	} else {
		_Frame_SeasonInfo.RelativePosition_V3 = C_Private_Pos_FrameSeasonInfoEnd;
		_Context.AnimMgr.Add(_Frame_SeasonInfo, """<anim hidden="1" />""", Now, C_Private_SeasonInfoHideAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
		_Context.AnimMgr.Add(_Frame_SeasonInfo, """<anim pos="{{{C_Private_Pos_FrameSeasonInfoStart.X}}} {{{C_Private_Pos_FrameSeasonInfoStart.Y}}}" />""", Now + C_Private_SeasonInfoHideAnimDuration, 0, CAnimManager::EAnimManagerEasing::QuadOut);
		Private_SetSeasonInfoOpacity(_Context, _Frame_SeasonInfo, 0., Now, C_Private_SeasonInfoHideAnimDuration);
	}
}

/// Shift displayed season info
K_Private_SeasonInfo Private_ShiftSeasonInfo(CMlScript _Context, K_Private_SeasonInfo _SeasonInfo, Integer _StartTime) {
	declare K_Private_SeasonInfo SeasonInfo = _SeasonInfo;

	if (SeasonInfo.List.count <= 0) {
		SeasonInfo.Current = C_Private_SeasonInfo_Null;
	} else if (SeasonInfo.List.count == 1) {
		SeasonInfo.Current = SeasonInfo.List[0];
	} else if (SeasonInfo.List.count > 1) {
		if (SeasonInfo.List.exists(SeasonInfo.Current)) {
			SeasonInfo.Current = SeasonInfo.List[((Now - _StartTime) / C_Private_SeasonInfoDisplayDuration) % SeasonInfo.List.count];
		} else {
			SeasonInfo.Current = SeasonInfo.List[0];
		}
	}

	Private_SetIsSeasonInfoVisible(_Context, SeasonInfo.Frame_Rank, SeasonInfo.Current == C_Private_SeasonInfo_Rank, SeasonInfo.IsFocused);
	Private_SetIsSeasonInfoVisible(_Context, SeasonInfo.Frame_Timer, SeasonInfo.Current == C_Private_SeasonInfo_Timer, SeasonInfo.IsFocused);

	return SeasonInfo;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Enable or disable a season info
K_Private_SeasonInfo Private_SetIsSeasonInfoEnabled(K_Private_SeasonInfo _SeasonInfo, Integer _SeasonInfoId, Boolean _IsEnabled) {
	if (
		(_IsEnabled && _SeasonInfo.List.exists(_SeasonInfoId)) ||
		(!_IsEnabled && !_SeasonInfo.List.exists(_SeasonInfoId))
	) {
		return _SeasonInfo;
	}

	declare K_Private_SeasonInfo SeasonInfo = _SeasonInfo;

	if (_IsEnabled) {
		SeasonInfo.List.add(_SeasonInfoId);
		SeasonInfo.List = SeasonInfo.List.sort();
	} else {
		SeasonInfo.List.remove(_SeasonInfoId);
	}

	return SeasonInfo;
}

/// Update the visibility of the season info (timer and rank)
K_Private_Instance Private_UpdateSeasonInfoVisibility(CMlScript _Context, K_Private_Instance _Instance) {
	declare K_Private_Instance Instance = _Instance;

	// Weekly Shorts
	declare CampaignStruct::LibCampaignStruct_K_Campaign LiveWeeklyShortsCampaign = CampaignStore::GetLiveWeeklyShortsCampaign();
	declare CampaignStruct::LibCampaignStruct_K_Campaign LoadedWeeklyShortsCampaign = CampaignStore::GetCampaignFromId(LiveWeeklyShortsCampaign.Id);

	Instance.SeasonInfo_WeeklyShorts = Private_SetIsSeasonInfoEnabled(
		Instance.SeasonInfo_WeeklyShorts,
		C_Private_SeasonInfo_Rank,
		(
			LoadedWeeklyShortsCampaign.Id != CampaignStruct::C_Campaign_NullId &&
			LoadedWeeklyShortsCampaign.IsLoaded
		)
	);
	Private_SetSeasonInfoRank(Instance.SeasonInfo_WeeklyShorts, LoadedWeeklyShortsCampaign);

	if (
		LiveWeeklyShortsCampaign.Id != CampaignStruct::C_Campaign_NullId &&
		LiveWeeklyShortsCampaign.EndTimestamp > 0 &&
		LiveWeeklyShortsCampaign.IsSeasonLive
	) {
		declare Integer TimeLeft = TiL::GetDelta(""^LiveWeeklyShortsCampaign.EndTimestamp, TiL::GetCurrent());
		if (TimeLeft > 0 && TimeLeft <= C_Private_SeasonTimerMaxDuration) {
			Instance.SeasonInfo_WeeklyShorts.EndTime = Now + (TimeLeft * 1000) + 5000; //< Add a 5 seconds margin
		} else {
			Instance.SeasonInfo_WeeklyShorts.EndTime = -1;
		}
	} else {
		Instance.SeasonInfo_WeeklyShorts.EndTime = -1;
	}
	Instance.SeasonInfo_WeeklyShorts = Private_SetIsSeasonInfoEnabled(
		Instance.SeasonInfo_WeeklyShorts,
		C_Private_SeasonInfo_Timer,
		Instance.SeasonInfo_WeeklyShorts.EndTime >= 0
	);

	Instance.SeasonInfo_WeeklyShorts = Private_ShiftSeasonInfo(_Context, Instance.SeasonInfo_WeeklyShorts, Instance.SeasonInfoStartTime);

	// Weekly Grands
	declare CampaignStruct::LibCampaignStruct_K_Campaign LiveWeeklyGrandsCampaign = CampaignStore::GetLiveWeeklyGrandsCampaign();
	declare CampaignStruct::LibCampaignStruct_K_Campaign LoadedWeeklyGrandsCampaign = CampaignStore::GetCampaignFromId(LiveWeeklyGrandsCampaign.Id);

	Instance.SeasonInfo_WeeklyGrands = Private_SetIsSeasonInfoEnabled(
		Instance.SeasonInfo_WeeklyGrands,
		C_Private_SeasonInfo_Rank,
		(
			LoadedWeeklyGrandsCampaign.Id != CampaignStruct::C_Campaign_NullId &&
			LoadedWeeklyGrandsCampaign.IsLoaded
		)
	);
	Private_SetSeasonInfoRank(Instance.SeasonInfo_WeeklyGrands, LoadedWeeklyGrandsCampaign);

	if (
		LiveWeeklyGrandsCampaign.Id != CampaignStruct::C_Campaign_NullId &&
		LiveWeeklyGrandsCampaign.EndTimestamp > 0 &&
		LiveWeeklyGrandsCampaign.IsSeasonLive
	) {
		declare Integer TimeLeft = TiL::GetDelta(""^LiveWeeklyGrandsCampaign.EndTimestamp, TiL::GetCurrent());
		if (TimeLeft > 0 && TimeLeft <= C_Private_SeasonTimerMaxDuration) {
			Instance.SeasonInfo_WeeklyGrands.EndTime = Now + (TimeLeft * 1000) + 5000; //< Add a 5 seconds margin
		} else {
			Instance.SeasonInfo_WeeklyGrands.EndTime = -1;
		}
	} else {
		Instance.SeasonInfo_WeeklyGrands.EndTime = -1;
	}
	Instance.SeasonInfo_WeeklyGrands = Private_SetIsSeasonInfoEnabled(
		Instance.SeasonInfo_WeeklyGrands,
		C_Private_SeasonInfo_Timer,
		Instance.SeasonInfo_WeeklyGrands.EndTime >= 0
	);

	Instance.SeasonInfo_WeeklyGrands = Private_ShiftSeasonInfo(_Context, Instance.SeasonInfo_WeeklyGrands, Instance.SeasonInfoStartTime);

	// Set season update timer
	if (
		Instance.SeasonTimerUpdateTime < 0 && (
			Instance.SeasonInfo_WeeklyShorts.EndTime >= 0 ||
			Instance.SeasonInfo_WeeklyGrands.EndTime >= 0
		)
	) {
		Instance.SeasonTimerUpdateTime = Now;
	} else if (
		Instance.SeasonTimerUpdateTime >= 0 &&
		Instance.SeasonInfo_WeeklyShorts.EndTime < 0 &&
		Instance.SeasonInfo_WeeklyGrands.EndTime < 0
	) {
		Instance.SeasonTimerUpdateTime = -1;
	}

	return Instance;
}

/// Set the focus state of the season info frame
K_Private_SeasonInfo Private_SetIsSeasonInfoFocused(CMlScript _Context, K_Private_SeasonInfo _SeasonInfo, Boolean _IsFocused) {
	declare K_Private_SeasonInfo SeasonInfo = _SeasonInfo;

	SeasonInfo.IsFocused = _IsFocused;

	// Only visible season info need to be updated
	if (SeasonInfo.Current == C_Private_SeasonInfo_Rank) {
		Private_SetIsSeasonInfoVisible(_Context, SeasonInfo.Frame_Rank, True, SeasonInfo.IsFocused);
	} else if (SeasonInfo.Current == C_Private_SeasonInfo_Timer) {
		Private_SetIsSeasonInfoVisible(_Context, SeasonInfo.Frame_Timer, True, SeasonInfo.IsFocused);
	}

	return SeasonInfo;
}

/// Update the order of the buttons
K_Private_Instance Private_UpdateButtonOrder(CMlScript _Context, K_Private_Instance _Instance) {
	declare K_Private_Instance Instance = _Instance;

	declare Boolean DisplayWeeklyGrandsFirst = False;
	declare Boolean IsConfigValid = _Context.UserMgr.Config_GetBoolValue("DisplayWeeklyGrandsFirst", DisplayWeeklyGrandsFirst);

	declare Vec2 Pos_ButtonTop = MLHelpers::SkewX(C_Private_Pos_ButtonTop, 10.);
	declare Vec2 Pos_ButtonBottom = MLHelpers::SkewX(C_Private_Pos_ButtonBottom, 10.);

	if (IsConfigValid && DisplayWeeklyGrandsFirst) {
		Instance.DefaultFocus = Instance.Controls.Button_WeeklyGrands;
		Instance.Controls.Frame_WeeklyGrands.RelativePosition_V3 = Pos_ButtonTop;
		Instance.Controls.Frame_WeeklyShorts.RelativePosition_V3 = Pos_ButtonBottom;
		Navigation::ChangeControlTargets(
			_Context,
			Instance.Controls.Button_WeeklyGrands,
			["up", "down"],
			["_", C_Private_Id_Button_WeeklyShorts]
		);
		Navigation::ChangeControlTargets(
			_Context,
			Instance.Controls.Button_WeeklyShorts,
			["up", "down"],
			[C_Private_Id_Button_WeeklyGrands, C_Private_Id_Button_Back]
		);
		Navigation::ChangeControlTarget(
			_Context,
			Instance.Controls.Button_Back,
			"up",
			C_Private_Id_Button_WeeklyShorts
		);
	} else {
		Instance.DefaultFocus = Instance.Controls.Button_WeeklyShorts;
		Instance.Controls.Frame_WeeklyShorts.RelativePosition_V3 = Pos_ButtonTop;
		Instance.Controls.Frame_WeeklyGrands.RelativePosition_V3 = Pos_ButtonBottom;
		Navigation::ChangeControlTargets(
			_Context,
			Instance.Controls.Button_WeeklyShorts,
			["up", "down"],
			["_", C_Private_Id_Button_WeeklyGrands]
		);
		Navigation::ChangeControlTargets(
			_Context,
			Instance.Controls.Button_WeeklyGrands,
			["up", "down"],
			[C_Private_Id_Button_WeeklyShorts, C_Private_Id_Button_Back]
		);
		Navigation::ChangeControlTarget(
			_Context,
			Instance.Controls.Button_Back,
			"up",
			C_Private_Id_Button_WeeklyGrands
		);
	}

	return Instance;
}

/// Focus an element of the page
K_Private_Instance Private_Focus(CMlScript _Context, K_Private_Instance _Instance, CMlControl _From, CMlControl _To) {
	if (Transition::IsInTransition(_Context)) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	if (_From != Null) {
		Navigation::Unfocus(_Context, _From);

		if (ExpendableButtonBase::IsButton(_From)) {
			ExpendableButtonBase::Unfocus(_Context, (_From as CMlFrame));
		} else if (Button::IsButton(_From)) {
			Button::Unfocus(_Context, (_From as CMlFrame));
		}

		if (_From.HasClass(C_Private_Class_HasSeasonInfo)) {
			if (_From.ControlId == C_Private_Id_Button_WeeklyShorts) {
				Instance.SeasonInfo_WeeklyShorts = Private_SetIsSeasonInfoFocused(_Context, Instance.SeasonInfo_WeeklyShorts, False);
			} else if (_From.ControlId == C_Private_Id_Button_WeeklyGrands) {
				Instance.SeasonInfo_WeeklyGrands = Private_SetIsSeasonInfoFocused(_Context, Instance.SeasonInfo_WeeklyGrands, False);
			}
		}
	}

	if (_To != Null) {
		Navigation::Focus(_Context, _To);

		if (ExpendableButtonBase::IsButton(_To)) {
			ExpendableButtonBase::Focus(_Context, (_To as CMlFrame));
		} else if (Button::IsButton(_To)) {
			Button::Focus(_Context, (_To as CMlFrame));
		}

		if (_To.HasClass(C_Private_Class_HasSeasonInfo)) {
			if (_To.ControlId == C_Private_Id_Button_WeeklyShorts) {
				Instance.SeasonInfo_WeeklyShorts = Private_SetIsSeasonInfoFocused(_Context, Instance.SeasonInfo_WeeklyShorts, True);
			} else if (_To.ControlId == C_Private_Id_Button_WeeklyGrands) {
				Instance.SeasonInfo_WeeklyGrands = Private_SetIsSeasonInfoFocused(_Context, Instance.SeasonInfo_WeeklyGrands, True);
			}
		}
	}

	return Instance;
}

/// Select an element of the page
Void Private_Select(CMlScript _Context, CMlControl _Control) {
	if (_Control == Null || Transition::IsInTransition(_Context)) return;

	switch (_Control.ControlId) {
		case C_Private_Id_Button_WeeklyShorts: {
			declare CampaignStruct::LibCampaignStruct_K_Campaign LiveWeeklyShortsCampaign = CampaignStore::GetLiveWeeklyShortsCampaign();
			if (LiveWeeklyShortsCampaign.Id != CampaignStruct::C_Campaign_NullId) {
				Router::SetParentPath(_Context, RoutePaths::C_Path_CampaignDisplay, RoutePaths::C_Path_SoloWeeklyTracks);
				Router::Push(_Context,
					RoutePaths::C_Path_CampaignDisplay,	[
						RoutePaths::C_Query_Campaign => ""^LiveWeeklyShortsCampaign.tojson(),
						Router::C_Query_DisablePrivilege => Privileges::PrivilegesToQuery([CPrivilegeMgr::EPrivilege::UseUserCreatedContentRequired])
					],
					True
				);
			}
		}
		case C_Private_Id_Button_WeeklyGrands: {
			declare CampaignStruct::LibCampaignStruct_K_Campaign LiveWeeklyGrandsCampaign = CampaignStore::GetLiveWeeklyGrandsCampaign();
			if (LiveWeeklyGrandsCampaign.Id != CampaignStruct::C_Campaign_NullId) {
				Router::SetParentPath(_Context, RoutePaths::C_Path_CampaignDisplay, RoutePaths::C_Path_SoloWeeklyTracks);
				Router::Push(_Context,
					RoutePaths::C_Path_CampaignDisplay,	[
						RoutePaths::C_Query_Campaign => ""^LiveWeeklyGrandsCampaign.tojson(),
						Router::C_Query_DisablePrivilege => Privileges::PrivilegesToQuery([CPrivilegeMgr::EPrivilege::UseUserCreatedContentRequired])
					],
					True
				);
			}
		}
		case C_Private_Id_Button_Back: {
			Router::PushParent(_Context);
		}
	}
}

/// The page main loop section
Void Private_MainLoop(CMlScript _Context) {
	if (Router::PageIsRunning(_Context, RoutePaths::C_Path_SoloWeeklyTracks)) {
		declare K_Private_Instance Trackmania_MainMenuPagesWeeklyTracks_Instance as Instance for _Context.Page = K_Private_Instance {};

		foreach (Event in Router::GetPendingEvents(_Context)) {
			switch (Event.Type) {
				case Router::C_Event_EnteringRoute: {
					ControllerInfo::SetButtons(["A" => _("Select"), "B" => _("Back")]);

					Instance = Private_UpdateButtonOrder(_Context, Instance);
					declare CMlControl FocusedControl = Navigation::GetFocusedControl(_Context, C_Private_NavGroup_Main);
					if (FocusedControl == Instance.Controls.Button_Back) {
						Instance = Private_Focus(_Context, Instance, FocusedControl, Instance.DefaultFocus);
					}

					CampaignStore::Action_LoadLiveWeeklyShortsCampaign(_Context, CampaignAPI::C_CachePreload_CampaignWeeklyShorts);
					declare CampaignStruct::LibCampaignStruct_K_Campaign LiveWeeklyShortsCampaign = CampaignStore::GetLiveWeeklyShortsCampaign();
					if (LiveWeeklyShortsCampaign.Id != CampaignStruct::C_Campaign_NullId) {
						CampaignStore::Action_LoadCampaignPlayerGlobalRankings(_Context, LiveWeeklyShortsCampaign.Id, LiveWeeklyShortsCampaign.SeasonUid);
					}
					CampaignStore::Action_LoadLiveWeeklyGrandsCampaign(_Context, CampaignAPI::C_CachePreload_CampaignWeeklyGrands);
					declare CampaignStruct::LibCampaignStruct_K_Campaign LiveWeeklyGrandsCampaign = CampaignStore::GetLiveWeeklyGrandsCampaign();
					if (LiveWeeklyGrandsCampaign.Id != CampaignStruct::C_Campaign_NullId) {
						CampaignStore::Action_LoadCampaignPlayerGlobalRankings(_Context, LiveWeeklyGrandsCampaign.Id, LiveWeeklyGrandsCampaign.SeasonUid);
					}

					CampaignStore::Action_LoadMediaActive(_Context, C_Private_MediaUpdateCooldown);
					Private_UpdateCampaignMedia(_Context, Instance);

					Private_SetCampaignMedalStack(LiveWeeklyShortsCampaign.Id, Instance.Controls.MedalStack_WeeklyShorts);
					Private_SetCampaignMedalStack(LiveWeeklyGrandsCampaign.Id, Instance.Controls.MedalStack_WeeklyGrands);

					Instance.SeasonInfoSwitchTime = Now + C_Private_SeasonInfoDisplayDuration;
					Instance.SeasonInfoStartTime = Now;

					// Set `CanRefresh` to `True` only once per page display
					// We need to avoid spamming the API to refresh the campaign
					Instance.SeasonInfo_WeeklyShorts.CanRefresh = True;
					Instance.SeasonInfo_WeeklyGrands.CanRefresh = True;

					Instance = Private_UpdateSeasonInfoVisibility(_Context, Instance);
				}
			}
		}

		foreach (Event in Navigation::GetPendingEvents(_Context)) {
			switch (Event.Type) {
				case Navigation::C_EventType_NavigateInput: {
					if (Navigation::IsEventFromGroup(_Context, Event, C_Private_NavGroup_Main)) {
						if (Event.Input == CMlScriptEvent::EMenuNavAction::Select) {
							Private_Select(_Context, Event.To);
						} else {
							Instance = Private_Focus(_Context, Instance, Event.From, Event.To);
						}
					} else if (Navigation::IsEventFromGroup(_Context, Event, C_Private_NavGroup_Global)) {
						if (Event.Input == CMlScriptEvent::EMenuNavAction::Cancel) {
							Private_Select(_Context, Instance.Controls.Button_Back);
						}
					}
				}
				case Navigation::C_EventType_NavigateMouse: {
					if (Navigation::IsEventFromGroup(_Context, Event, C_Private_NavGroup_Main)) {
						if (Event.Mouse == CMlScriptEvent::Type::MouseClick) {
							Private_Select(_Context, Event.To);
						} else if (Event.Mouse == CMlScriptEvent::Type::MouseOver) {
							Instance = Private_Focus(_Context, Instance, Event.From, Event.To);
						}
					}
				}
			}
		}

		// Store events
		foreach (Event in _Context.PendingEvents) {
			if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
				switch (Event.CustomEventType) {
					case CampaignStore::C_Event_LiveWeeklyShortsCampaignUpdated: {
						if (CampaignStore::GetLiveWeeklyShortsCampaignStatus() == CampaignStore::C_LiveCampaignStatus_Loaded) {
							declare CampaignStruct::LibCampaignStruct_K_Campaign LiveWeeklyShortsCampaign = CampaignStore::GetLiveWeeklyShortsCampaign();
							if (LiveWeeklyShortsCampaign.Id != CampaignStruct::C_Campaign_NullId) {
								if (!LiveWeeklyShortsCampaign.IsLoaded || LiveWeeklyShortsCampaign.FailedToGetMapUrl) {
									CampaignStore::Action_LoadCampaign(_Context, LiveWeeklyShortsCampaign.Id);
								}
								Instance = Private_UpdateSeasonInfoVisibility(_Context, Instance);
							}
						}
					}
					case CampaignStore::C_Event_LiveWeeklyGrandsCampaignUpdated: {
						if (CampaignStore::GetLiveWeeklyGrandsCampaignStatus() == CampaignStore::C_LiveCampaignStatus_Loaded) {
							declare CampaignStruct::LibCampaignStruct_K_Campaign LiveWeeklyGrandsCampaign = CampaignStore::GetLiveWeeklyGrandsCampaign();
							if (LiveWeeklyGrandsCampaign.Id != CampaignStruct::C_Campaign_NullId) {
								if (!LiveWeeklyGrandsCampaign.IsLoaded || LiveWeeklyGrandsCampaign.FailedToGetMapUrl) {
									CampaignStore::Action_LoadCampaign(_Context, LiveWeeklyGrandsCampaign.Id);
								}
								Instance = Private_UpdateSeasonInfoVisibility(_Context, Instance);
								Private_UpdateCampaignMedia(_Context, Instance);
							}
						}
					}
					case CampaignStore::C_Event_CampaignLoadComplete, CampaignStore::C_Event_CampaignUpdateFailed: {
						switch (Cast::ToInteger(Event.CustomEventData.get(0, ""))) {
							case CampaignStore::GetLiveWeeklyShortsCampaignId(): {
								Private_SetCampaignMedalStack(CampaignStore::GetLiveWeeklyShortsCampaignId(), Instance.Controls.MedalStack_WeeklyShorts);
							}
							case CampaignStore::GetLiveWeeklyGrandsCampaignId(): {
								Private_SetCampaignMedalStack(CampaignStore::GetLiveWeeklyGrandsCampaignId(), Instance.Controls.MedalStack_WeeklyGrands);
							}
						}
					}
					case CampaignStore::C_Event_CampaignPlayerGlobalRankingsUpdated: {
						if (Event.CustomEventData.count > 0) {
							switch (Cast::ToInteger(Event.CustomEventData[0])) {
								case CampaignStore::GetLiveWeeklyShortsCampaignId(): {
									Instance = Private_UpdateSeasonInfoVisibility(_Context, Instance);
								}
								case CampaignStore::GetLiveWeeklyGrandsCampaignId(): {
									Instance = Private_UpdateSeasonInfoVisibility(_Context, Instance);
								}
							}
						}
					}
					case CampaignStore::C_Event_MediaActiveUpdated: {
						Private_UpdateCampaignMedia(_Context, Instance);
					}
				}
			}
		}

		if (Instance.SeasonInfoSwitchTime <= Now) {
			Instance.SeasonInfoSwitchTime = Now + C_Private_SeasonInfoDisplayDuration;
			Instance.SeasonInfo_WeeklyShorts = Private_ShiftSeasonInfo(_Context, Instance.SeasonInfo_WeeklyShorts, Instance.SeasonInfoStartTime);
			Instance.SeasonInfo_WeeklyGrands = Private_ShiftSeasonInfo(_Context, Instance.SeasonInfo_WeeklyGrands, Instance.SeasonInfoStartTime);
		}

		if (
			Instance.SeasonTimerUpdateTime >= 0 &&
			Instance.SeasonTimerUpdateTime <= Now
		) {
			Instance.SeasonTimerUpdateTime = Now + C_Private_SeasonTimerUpdateInterval;
			if (Instance.SeasonInfo_WeeklyShorts.Current == C_Private_SeasonInfo_Timer) {
				Private_SetSeasonInfoTimer(Instance.SeasonInfo_WeeklyShorts);
			}
			if (Instance.SeasonInfo_WeeklyGrands.Current == C_Private_SeasonInfo_Timer) {
				Private_SetSeasonInfoTimer(Instance.SeasonInfo_WeeklyGrands);
			}
		}

		if (
			Instance.SeasonInfo_WeeklyShorts.CanRefresh &&
			Instance.SeasonInfo_WeeklyShorts.EndTime >= 0 &&
			Instance.SeasonInfo_WeeklyShorts.EndTime <= Now
		) {
			Instance.SeasonInfo_WeeklyShorts.CanRefresh = False;
			CampaignStore::Action_LoadLiveWeeklyShortsCampaign(_Context, CampaignAPI::C_CachePreload_CampaignWeeklyShorts);
		}

		if (
			Instance.SeasonInfo_WeeklyShorts.CanRefresh &&
			Instance.SeasonInfo_WeeklyShorts.EndTime >= 0 &&
			Instance.SeasonInfo_WeeklyShorts.EndTime <= Now
		) {
			Instance.SeasonInfo_WeeklyShorts.CanRefresh = False;
			CampaignStore::Action_LoadLiveWeeklyGrandsCampaign(_Context, CampaignAPI::C_CachePreload_CampaignWeeklyGrands);
		}
	}
}
