/// Update the info display
Void Private_UpdateInfoDisplay(CManiaAppTitleLayer _Context, K_Private_Instance _Instance) {
	if (_Instance.IsJoiningServer) {
		LoadingSpinner::StartAnimWithFade(_Context, _Instance.Controls.LoadingSpinner_Info, C_Private_Text_Joining);
		ExpendableButtonBase::RemoveStyle(_Context, _Instance.Controls.Button_Join, ExpendableButton::Style_PageNextAnimated(), False);
		ExpendableButtonBase::AddStyle(_Context, _Instance.Controls.Button_Join, ExpendableButton::Style_ButtonLoading());
	} else {
		LoadingSpinner::StopAnimWithFade(_Context, _Instance.Controls.LoadingSpinner_Info);
		ExpendableButtonBase::RemoveStyle(_Context, _Instance.Controls.Button_Join, ExpendableButton::Style_ButtonLoading(), False);
		ExpendableButtonBase::AddStyle(_Context, _Instance.Controls.Button_Join, ExpendableButton::Style_PageNextAnimated());
	}

	declare Vec2 Pos_Hidden = MLHelpers::SkewX(<-126., 20.>, 10.);
	declare Vec2 Pos_Visible = MLHelpers::SkewX(<-125., 20.>, 10.);
	_Context.AnimMgr.Flush(_Instance.Controls.Label_Info);
	if (!_Instance.IsJoiningServer && _Instance.InfoMessage != "") {
		_Instance.Controls.Label_Info.Opacity = 0.;
		_Instance.Controls.Label_Info.RelativePosition_V3 = Pos_Hidden;
		_Instance.Controls.Label_Info.Value = _Instance.InfoMessage;
		if (_Instance.InfoType == MainMenuConst::C_InfoType_Error) {
			_Instance.Controls.Label_Info.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_TextError);
		} else {
			_Instance.Controls.Label_Info.TextColor = Stylesheet::GetColorRGB(Stylesheet::C_Color_TextLight);
		}
		_Context.AnimMgr.Add(_Instance.Controls.Label_Info, """<anim pos="{{{Pos_Visible.X}}} {{{Pos_Visible.Y}}}" opacity="1" hidden="0" />""", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	} else {
		_Context.AnimMgr.Add(_Instance.Controls.Label_Info, """<anim pos="{{{Pos_Hidden.X}}} {{{Pos_Hidden.Y}}}" opacity="0" hidden="1" />""", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	}
}

/// Set the joining state
K_Private_Instance Private_SetIsJoiningServer(CManiaAppTitleLayer _Context, K_Private_Instance _Instance, Boolean _IsInit) {
	declare Boolean IsJoiningServer = (
		!Privileges::Check_IsComplete(_Instance.PlayPrivilegeCheck) ||
		Http::IsRunning(_Instance.JoinLinkRequest)
	);

	if (!_IsInit && _Instance.IsJoiningServer == IsJoiningServer) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	Instance.IsJoiningServer = IsJoiningServer;
	Private_UpdateInfoDisplay(_Context, Instance);

	return Instance;
}
K_Private_Instance Private_SetIsJoiningServer(CManiaAppTitleLayer _Context, K_Private_Instance _Instance) {
	return Private_SetIsJoiningServer(_Context, _Instance, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the info message
K_Private_Instance Private_SetInfoMessage(CManiaAppTitleLayer _Context, K_Private_Instance _Instance, Text _InfoMessage, Integer _InfoType, Boolean _IsInit) {
	if (!_IsInit && _Instance.InfoMessage == _InfoMessage && _Instance.InfoType == _InfoType) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	Instance.InfoMessage = _InfoMessage;

K_Private_Instance Private_SetInfoMessage(CManiaAppTitleLayer _Context, K_Private_Instance _Instance, Text _InfoMessage, Integer _InfoType) {
	return Private_SetInfoMessage(_Context, _Instance, _InfoMessage, _InfoType, False);

/// Update the Cup of the Week status
K_Private_Instance Private_SetStatus(CManiaAppTitleLayer _Context, K_Private_Instance _Instance) {
	declare K_Private_Instance Instance = _Instance;

	declare Integer CountdownDelta = TiL::GetDelta(""^Instance.CountdownTimestamp, TiL::GetCurrent());
	declare Integer QualificationStartDelta = TiL::GetDelta(""^Instance.QualificationStartTimestamp, TiL::GetCurrent());
	declare Integer QualificationEndDelta = TiL::GetDelta(""^Instance.QualificationEndTimestamp, TiL::GetCurrent());
	declare Integer CampaignEndDelta = TiL::GetDelta(""^Instance.CampaignEndTimestamp, TiL::GetCurrent());

	if (Instance.CampaignId == CampaignStruct::C_Campaign_NullId) {
		Instance.Status = C_Private_Status_NoCup;
		Instance.StatusUpdateTime = -1;
		Instance.Controls.Label_COTWStatus.Value = "";
	} else if (CountdownDelta > 0) {
		Instance.Status = C_Private_Status_WaitingCup;
		Instance.StatusUpdateTime = Now + (CountdownDelta * 1000);
		Instance.Controls.Label_COTWStatus.Value = "";
	} else if (QualificationStartDelta > 0) {
		Instance.Status = C_Private_Status_QualificationCountdown;
		Instance.StatusUpdateTime = Now + (QualificationStartDelta * 1000);
		// Update `Label_COTWStatus` in the main loop
	} else if (QualificationEndDelta > 0) {
		Instance.Status = C_Private_Status_QualificationOngoing;
		Instance.StatusUpdateTime = Now + (QualificationEndDelta * 1000);
		// Update `Label_COTWStatus` in the main loop
	} else if (CampaignEndDelta > 0) {
		Instance.Status = C_Private_Status_QualificationEnded;
		Instance.StatusUpdateTime = Now + (CampaignEndDelta * 1000);
		Instance.Controls.Label_COTWStatus.Value = C_Private_Text_QualificationEnded;
	} else {
		Instance.Status = C_Private_Status_NextCup;
		Instance.StatusUpdateTime = -1;
		Instance.Controls.Label_COTWStatus.Value = C_Private_Text_CupEnded;
	}

	// Update popup header image
	if (
		Instance.Status == C_Private_Status_QualificationCountdown ||
		Instance.Status == C_Private_Status_QualificationOngoing
	) {
		PopUpBackground::SetHeaderImageUrl(_Context, Instance.Controls.PopUpBackground_Main, MainMenuConst::C_ImageUrl_PopUp_TOTD_Header_COTD);
	} else {
		PopUpBackground::SetHeaderImageUrl(_Context, Instance.Controls.PopUpBackground_Main, MainMenuConst::C_ImageUrl_PopUp_TOTD_Header);
	}

	// If there is a cup but it has not started yet, enable loading of the next cup
	// /!\ Be careful when editing this, we purposefully set the value to `True` but we do not set it to `False`
	if (
		Instance.Status != C_Private_Status_NoCup &&
		Instance.Status != C_Private_Status_NextCup
	) {
		Instance.CanLoadNextCupOfTheWeekCampaign = True;
	}

	return Instance;
}

/// Set the current Cup of the Week campaign
K_Private_Instance Private_SetCupOfTheWeekCampaign(
	CManiaAppTitleLayer _Context,
	K_Private_Instance _Instance,
	CampaignStruct::LibCampaignStruct_K_Campaign _CupOfTheWeekCampaign,
	Integer _CupOfTheWeekCampaignUpdateTime
) {
	declare K_Private_Instance Instance = _Instance;

	Instance.CampaignId = _CupOfTheWeekCampaign.Id;
	if (_CupOfTheWeekCampaign.Playlist.count > 0) {
		Instance.Map = CampaignStore::GetMap(_CupOfTheWeekCampaign.Id, _CupOfTheWeekCampaign.Playlist[0].Uid);
	} else {
		Instance.Map = CampaignStruct::LibCampaignStruct_K_Map {};
	}

	Instance.CampaignEndTimestamp = _CupOfTheWeekCampaign.EndTimestamp;
	Instance.QualificationStartTimestamp = Instance.CampaignEndTimestamp + CampaignStore::C_CupOfTheWeekStartTimeOffset;
	Instance.CountdownTimestamp = Instance.QualificationStartTimestamp - C_Private_CountdownDuration;
	Instance.QualificationEndTimestamp = Instance.QualificationStartTimestamp + C_Private_QualificationDuration;
	Instance.NextCupOfTheWeekCampaignUpdateTime = _CupOfTheWeekCampaignUpdateTime;

	Instance.Controls.Label_COTWDate.Value = TL::Compose(C_Private_Text_COTWDate, TiL::FormatDate(""^Instance.QualificationStartTimestamp, TiL::EDateFormats::Date), TiL::FormatDate(""^Instance.QualificationStartTimestamp, TiL::EDateFormats::TimeShort));
	Instance.Controls.Label_COTWDay.Value = ""^CampaignStore::GetDayCOTW();

	MLHelpers::FitLabelValue(Instance.Controls.Label_COTWTakePlace, C_Private_TextSize_COTWTakePlace);
	MLHelpers::FitLabelValue(Instance.Controls.Label_COTWDate, C_Private_TextSize_COTWDate);
	MLHelpers::FitLabelValue(Instance.Controls.Label_COTWExplanation, C_Private_TextSize_COTWExplanation);

	Instance = Private_SetStatus(_Context, Instance);

	Private_UpdateMapRankings(_Context, Instance);

	return Instance;
}

	if (!_IsInit && _Instance.IsPageActive == _IsPageActive) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	Instance.IsPageActive = _IsPageActive;
	if (Instance.IsPageActive) {
		ControllerInfo::SetButtons(["A" => Translations::C_Text_ActionSelect, "B" => Translations::C_Text_ActionBack]);
	}
	Private_UpdateNavigation(_Context, Instance);

	return Instance;
}
K_Private_Instance Private_SetIsPageActive(CManiaAppTitleLayer _Context, K_Private_Instance _Instance, Boolean _IsActive) {
	return Private_SetIsPageActive(_Context, _Instance, _IsActive, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Focus an element of the page
Void Private_Focus(CManiaAppTitleLayer _Context, CMlControl _From, CMlControl _To) {
	if (Transition::IsInTransition(_Context)) return;

	if (_From != Null) {
		Navigation::Unfocus(_Context, _From);

		if (ExpendableButtonBase::IsButton(_From)) {
			ExpendableButtonBase::Unfocus(_Context, (_From as CMlFrame));
		}
	}

	if (_To != Null) {
		Navigation::Focus(_Context, _To);

		if (ExpendableButtonBase::IsButton(_To)) {
			ExpendableButtonBase::Focus(_Context, (_To as CMlFrame));
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Select an element of the page
K_Private_Instance Private_Select(CManiaAppTitleLayer _Context, K_Private_Instance _Instance, CMlControl _Control) {
	if (_Control == Null || Transition::IsInTransition(_Context)) return _Instance;

	declare K_Private_Instance Instance = _Instance;

	switch (_Control.ControlId) {
		case C_Private_Id_Button_Join: {
			if (!UserStore::HasGamePermission(_Context.PrivilegeMgr, RBAC::C_GamePermission_PlayCOTWChannel)) {
				ClubEdition::GoToPage(_Context, Router::GetCurrentRoutePath(_Context), RBAC::GetGamePermissionMessage(RBAC::C_GamePermissionMessage_PlayCOTWChannel));
			} else if (
				_Context.TitleControl.IsReady &&
				Privileges::Check_IsComplete(Instance.PlayPrivilegeCheck) &&
				!Http::IsInitialized(Instance.JoinLinkRequest)
			) {
				Instance.PlayPrivilegeCheck = Privileges::Check_DestroyAndCreate(
					Instance.PlayPrivilegeCheck,
					_Context.PrivilegeMgr,
					MainUser::GetMainUserId(),
					CPrivilegeMgr::EPrivilege::PlayMultiplayerSession
				);
				Instance = Private_SetInfoMessage(_Context, Instance, "", MainMenuConst::C_InfoType_None);
				Instance = Private_SetIsJoiningServer(_Context, Instance);
			}
		}
	}

	return Instance;
}

/// The page main loop section
Void Private_MainLoop(CManiaAppTitleLayer _Context) {
	declare K_Private_Instance Trackmania_MainMenuPagesCOTWChannelDisplay_Instance as Instance for _Context.Page = K_Private_Instance {};

	if (Router::PageIsRunning(_Context, RoutePaths::C_Path_COTWChannelDisplay)) {
		foreach (Event in Router::GetPendingEvents(_Context)) {
			switch (Event.Type) {
				case Router::C_Event_EnteringRoute: {
					Instance = Private_SetIsPageActive(_Context, Instance, True);
					Instance = Private_SetCupOfTheWeekCampaign(_Context, Instance, CampaignStore::GetLiveCupOfTheWeekCampaign(), CampaignStore::GetLiveCupOfTheWeekCampaignsUpdateTime());

					CampaignStore::Action_LoadLiveCupOfTheWeekCampaign(_Context, CampaignAPI::C_CachePreload_CampaignCupOfTheWeek);
				}
			}
		}

		if (Instance.StatusUpdateTime > 0 && Instance.StatusUpdateTime <= Now) {
			Instance = Private_SetStatus(_Context, Instance);
		}
		if (
			Instance.Status == C_Private_Status_NextCup &&
			Instance.CanLoadNextCupOfTheWeekCampaign &&
			Instance.NextCupOfTheWeekCampaignUpdateTime >= 0 &&
			Instance.NextCupOfTheWeekCampaignUpdateTime <= Now
		) {
			CampaignStore::Action_LoadLiveCupOfTheWeekCampaign(_Context, CampaignAPI::C_CachePreload_CampaignCupOfTheWeek);
			Instance.CanLoadNextCupOfTheWeekCampaign = False;
		}
		if (Instance.Status == C_Private_Status_QualificationCountdown) {
			Instance.Controls.Label_COTWStatus.Value = TL::Compose(C_Private_Text_QualificationStartsIn, TL::TimeToText(Instance.StatusUpdateTime - Now, False, False));
		} else if (Instance.Status == C_Private_Status_QualificationOngoing) {
			Instance.Controls.Label_COTWStatus.Value = TL::Compose(C_Private_Text_QualificationEndsIn, TL::TimeToText(Instance.StatusUpdateTime - Now, False, False));
		}

		// Navigation events
		foreach (Event in Navigation::GetPendingEvents(_Context)) {
			switch (Event.Type) {
				case Navigation::C_EventType_NavigateInput: {
					if (Navigation::IsEventFromGroup(_Context, Event, C_Private_NavGroup_Global)) {
						switch (Event.Input) {
							case CMlScriptEvent::EMenuNavAction::Cancel: Router::PushParent(_Context);
						}
					} else if (Navigation::IsEventFromGroup(_Context, Event, C_Private_NavGroup_Main)) {
						switch (Event.Input) {
							case CMlScriptEvent::EMenuNavAction::Select: Instance = Private_Select(_Context, Instance, Event.To);
							default: Private_Focus(_Context, Event.From, Event.To);
						}
					}
				}
				case Navigation::C_EventType_NavigateMouse: {
					if (Navigation::IsEventFromGroup(_Context, Event, C_Private_NavGroup_Main)) {
						switch (Event.Mouse) {
							case CMlScriptEvent::Type::MouseClick: Instance = Private_Select(_Context, Instance, Event.To);
							case CMlScriptEvent::Type::MouseOver: Private_Focus(_Context, Event.From, Event.To);
						}
					}
				}
			}
		}

		// Campaign store events
		foreach (Event in _Context.PendingEvents) {
			switch (Event.Type) {
				case CMlScriptEvent::Type::PluginCustomEvent: {
					switch (Event.CustomEventType) {
						case CampaignStore::C_Event_LiveCupOfTheWeekCampaignUpdated: {
							if (CampaignStore::GetLiveCupOfTheWeekCampaignStatus() == CampaignStore::C_LiveCampaignStatus_Loaded) {
								declare CampaignStruct::LibCampaignStruct_K_Campaign LiveCupOfTheWeekCampaign = CampaignStore::GetLiveCupOfTheWeekCampaign();

								Instance = Private_SetCupOfTheWeekCampaign(_Context, Instance, LiveCupOfTheWeekCampaign, CampaignStore::GetLiveCupOfTheWeekCampaignsUpdateTime());

								if (LiveCupOfTheWeekCampaign.Id != CampaignStruct::C_Campaign_NullId) {
									declare CampaignStruct::LibCampaignStruct_K_Campaign LoadedCupOfTheWeekCampaign = CampaignStore::GetCampaignFromId(LiveCupOfTheWeekCampaign.Id);
									if (!LoadedCupOfTheWeekCampaign.IsLoaded || LoadedCupOfTheWeekCampaign.FailedToGetMapUrl) {
										CampaignStore::Action_LoadCampaign(_Context, LiveCupOfTheWeekCampaign.Id);
									}
								}
							}
						}
						case CampaignStore::C_Event_CampaignLoadComplete, CampaignStore::C_Event_CampaignUpdateFailed: {
							switch (Cast::ToInteger(Event.CustomEventData.get(0, ""))) {
								case CampaignStore::GetLiveCupOfTheWeekCampaignId(): {
									Instance = Private_SetCupOfTheWeekCampaign(_Context, Instance, CampaignStore::GetLiveCupOfTheWeekCampaign(), CampaignStore::GetLiveCupOfTheWeekCampaignsUpdateTime());
								}
							}
						}
					}
				}
			}
		}

		// Popup background events
		foreach (Event in PopUpBackground::GetPendingEvents(_Context)) {
			if (Event.PopUpBackground != Null && Event.PopUpBackground.Id == Instance.Controls.PopUpBackground_Main.Id) {
				switch (Event.Type) {
					case PopUpBackground::C_EventType_Close: Router::PushParent(_Context);
				}
			}
		}

		// Check multiplayer privilege
		if (!Privileges::Check_IsComplete(Instance.PlayPrivilegeCheck)) {
			Instance.PlayPrivilegeCheck = Privileges::Check_Update(Instance.PlayPrivilegeCheck);
			if (Privileges::Check_IsComplete(Instance.PlayPrivilegeCheck)) {
				if (Privileges::Check_HasPrivilege(Instance.PlayPrivilegeCheck)) {
					Instance.JoinLinkWaitingTime = 0;
					Instance.JoinLinkRequest = Http::DestroyAndCreate(
						Instance.JoinLinkRequest,
						ChannelAPI::GetJoinLink(MainMenuConst::C_ChannelUid_CupOfTheWeek)
					);
				}
				Instance.PlayPrivilegeCheck = Privileges::Check_Destroy(Instance.PlayPrivilegeCheck);
				Instance = Private_SetIsJoiningServer(_Context, Instance);
			}
		}

		// JoinLink request
		if (Http::IsInitialized(Instance.JoinLinkRequest)) {
			Instance.JoinLinkRequest = Http::Update(Instance.JoinLinkRequest);
			if (!Http::IsRunning(Instance.JoinLinkRequest)) {
				if (Http::IsSuccess(Instance.JoinLinkRequest)) {
					declare Text JoinLink = ChannelAPI::GetResponseFromGetJoinLink(Instance.JoinLinkRequest);
					if (JoinLink != "") {
						LoadingScreenStore::UpdateTitleMenuLoadingScreen_Random(_Context, C_Private_Text_COTW);
						LoadingScreenStore::StartTrackingLoadingScreen(_Context, LoadingScreenStore::GetCurrentTitleMenuLoadingScreen(_Context));
						TitleControl::OpenLinkAndWait(_Context, _Context.TitleControl, JoinLink, CMlScript::LinkType::ManialinkBrowser);
						CampaignStore::Action_LoadLiveCupOfTheWeekCampaign(_Context, CampaignAPI::C_CachePreload_CampaignCupOfTheWeek);
					} else if (Instance.JoinLinkWaitingTime < C_Private_JoinlinkWaitingTimeMax) {
						Instance.JoinLinkRequest = Http::RetryWithDelay(Instance.JoinLinkRequest, Now + C_Private_JoinlinkRetryInterval);
						Instance.JoinLinkWaitingTime += C_Private_JoinlinkRetryInterval;
					} else {
						Instance = Private_SetInfoMessage(_Context, Instance, C_Private_Text_ServersOffline, MainMenuConst::C_InfoType_Error);
					}
				} else {
					Instance = Private_SetInfoMessage(_Context, Instance, TL::Compose(C_Private_Text_JoinError, ""^Http::GetStatusCode(Instance.JoinLinkRequest)), MainMenuConst::C_InfoType_Error);
				}
				if (!Http::IsWaitingRetry(Instance.JoinLinkRequest)) {
					Instance.JoinLinkRequest = Http::Destroy(Instance.JoinLinkRequest);
				}
				Instance = Private_SetIsJoiningServer(_Context, Instance);
			}
		}
	} else {
		foreach (Event in Router::GetPendingEvents(_Context)) {
			switch (Event.Type) {
				case Router::C_Event_LeavingRoute: {
					Instance.PlayPrivilegeCheck = Privileges::Check_Destroy(Instance.PlayPrivilegeCheck);
					Instance.JoinLinkRequest = Http::Destroy(Instance.JoinLinkRequest);
					Instance = Private_SetInfoMessage(_Context, Instance, "", MainMenuConst::C_InfoType_None);
					Instance = Private_SetIsJoiningServer(_Context, Instance);
					Instance = Private_SetIsPageActive(_Context, Instance, False);
				}
			}
		}
	}
}
