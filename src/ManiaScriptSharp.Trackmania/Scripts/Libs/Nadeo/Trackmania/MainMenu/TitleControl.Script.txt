/// Check that the given script is for the 'TM_Royal' map type
Boolean IsRoyalScript(Text _Script) {
	return (
		TL::Find(Const::C_ScriptModeName_RoyalOnline, _Script, True, True) ||
		TL::Find(Const::C_ScriptModeName_RoyalStarsOnline, _Script, True, True) ||
		TL::Find(Const::C_ScriptModeName_RoyalTimeAttack, _Script, True, True) ||
		TL::Find(Const::C_ScriptModeName_RoyalValidation, _Script, True, True)
	);
}

/// Check that the given script is for the 'TM_Stunt' map type
Boolean IsStuntScript(Text _Script) {
	return (
		TL::Find(Const::C_ScriptModeName_StuntSolo, _Script, True, True) ||
		TL::Find(Const::C_ScriptModeName_StuntMulti, _Script, True, True) ||
		TL::Find(Const::C_ScriptModeName_StuntValidation, _Script, True, True)
	);
}

/// Check that the given script is for the 'TM_Platform' map type
Boolean IsPlatformScript(Text _Script) {
	return (
		TL::Find(Const::C_ScriptModeName_PlatformSolo, _Script, True, True) ||
		TL::Find(Const::C_ScriptModeName_PlatformMulti, _Script, True, True) ||
		TL::Find(Const::C_ScriptModeName_PlatformValidation, _Script, True, True)
	);
}

/// Check that the given script is for the 'TM_Race' map type
Boolean IsRaceScript(Text _Script) {
	return C_ModeNames_ScriptNames.exists(_Script) && !IsRoyalScript(_Script) && !IsStuntScript(_Script) && !IsPlatformScript(_Script);
}

/// Get the map type corresponding to the given script
Text GetMapTypeForScript(Text _Script) {
	if (IsRoyalScript(_Script)) return Const::C_MapType_Royal_NoConcat;
	if (IsStuntScript(_Script)) return Const::C_MapType_Stunt_NoConcat;
	if (IsPlatformScript(_Script)) return Const::C_MapType_Platform_NoConcat;
	return Const::C_MapType_Race_NoConcat;
}

/// Check that the given map type is a 'TM_Royal' map type
Boolean IsRoyalMapType(Text _MapType) {
	return TL::Find(Const::C_ScriptModeName_Royal, _MapType, True, True);
}

/// Check that the given map type is a 'TM_Stunt' map type
Boolean IsStuntMapType(Text _MapType) {
	return TL::Find(Const::C_ScriptModeName_Stunt, _MapType, True, True);
}

/// Check that the given map type is a 'TM_Platform' map type
Boolean IsPlatformMapType(Text _MapType) {
	return TL::Find(Const::C_ScriptModeName_Platform, _MapType, True, True);
}

/// Check that the given map type is a 'TM_Race' map type
Boolean IsRaceMapType(Text _MapType) {
	return TL::Find(Const::C_ScriptModeName_Race, _MapType, True, True);
}

/// Get the default mode corresponding to the given map type
Text GetGameModeFromMapType(Text _MapType, Boolean _IsCloneEnabled) {
	if (IsRoyalMapType(_MapType)) return Const::C_GameMode_RoyalTimeAttack;
	if (IsStuntMapType(_MapType)) return Const::C_GameMode_Stunt;
	if (IsPlatformMapType(_MapType)) return Const::C_GameMode_Platform;
	if (_IsCloneEnabled) return Const::C_GameMode_CampaignClone;
	return Const::C_GameMode_TimeAttack;
}
Text GetGameModeFromMapType(Text _MapType) {
	return GetGameModeFromMapType(_MapType, False);
}

Void WaitIsReady(CTitleControl _TitleControl) {
	while (!_TitleControl.IsReady) yield;
}

Void PlayAgainstReplay(CTitleControl _TitleControl, Text _ReplayPath, Text _ModePath, Text _SettingsXml) {

	WaitIsReady(_TitleControl);
}
