Void Action_LoadFavoriteMapList(
	CMlScript _This,
	Text[] _MapTypeList,
	Boolean _SortByNameElseByDate,
	Boolean _SortOrderAsc,
	Boolean _OnlyPlayable,
	Boolean _OnlyMine
) {
	declare Text SortByNameElseByDate = "0";
	if (_SortByNameElseByDate) SortByNameElseByDate = "1";
	declare Text SortOrderAsc = "0";
	if (_SortOrderAsc) SortOrderAsc = "1";
	declare Text OnlyPlayable = "0";
	if (_OnlyPlayable) OnlyPlayable = "1";
	declare Text OnlyMine = "0";
	if (_OnlyMine) OnlyMine = "1";

	_This.SendCustomEvent(
		C_Action_LoadFavoriteMaps,
		[
			_MapTypeList.tojson(),
			SortByNameElseByDate,
			SortOrderAsc,
			OnlyPlayable,
			OnlyMine
		]
	);
}

Void Action_GetMap(CMlScript _This, Text _MapUid) {
	_This.SendCustomEvent(C_Action_GetMap, [_MapUid]);
}

Void Action_RemoveMapFromFavorites(CMlScript _This, Text _MapUid) {
	_This.SendCustomEvent(C_Action_RemoveMapFromFavorites, [_MapUid]);
}

Void Action_AddMapToFavorites(CMlScript _This, Text _MapUid) {
	_This.SendCustomEvent(C_Action_AddMapToFavorites, [_MapUid]);
}

Void Action_ToggleCurrentMapFromFavorites(CMlScript _This) {
	_This.SendCustomEvent(C_Action_ToggleCurrentMapFromFavorites, []);
}

CampaignStruct::LibCampaignStruct_K_Map[Integer] GetFavoriteMaps() {
	declare K_State Trackmania_MapStore_State for System = C_State_Null;
	return Trackmania_MapStore_State.FavoriteMapList;
}

Integer GetFavoriteMapsTotal() {
	declare K_State Trackmania_MapStore_State for System = C_State_Null;
	return Trackmania_MapStore_State.FavoriteMapList.count;
}

// Reset the current map status
Void ResetCurrentMap() {
	declare K_State Trackmania_MapStore_State for System = C_State_Null;
	Trackmania_MapStore_State.CurrentMap = K_CurrentMap {
		Status = C_CurrentMapStatus_Null,
		ErrorCode = C_ErrorCode_Null,
		ErrorCodeTask = C_ErrorCodeTask_Null,
		IsMapUploadedToNS = False,
		IsFavorite = False,
		Uid = ""
	};
}

// Get information about the current map status
K_CurrentMap GetCurrentMap() {
	declare K_State Trackmania_MapStore_State for System = C_State_Null;
	return Trackmania_MapStore_State.CurrentMap;
}

Void MainInit(CMlScript _This) {
	Store::Subscribe(_This, C_StoreId);
	ResetCurrentMap();
}
