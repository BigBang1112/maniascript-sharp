/**
 * Share player specific data between the server game mode and the clients UI
 */
#Const Version "1.0.0"
#Const ScriptName "Libs/Nadeo/Trackmania/Modes/RoyalTimeAttack/PlayerData.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structure
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_Times {
	Integer[] SegmentRaceTimes;
	Integer Total;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Default_PlayerTimesEventUpdate 0
#Const C_Default_PlayerTimes K_Times {
	SegmentRaceTimes = [],
	Total = 0
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Send a player times event to the clients (server side only)
Void SendPlayerTimesEvent(CSmPlayer _Player, K_Times _Times) {
	if (_Player == Null) return;
	declare netwrite Integer Net_RoyalTimeAttack_PlayerData_TimesEventUpdate for _Player = C_Default_PlayerTimesEventUpdate;
	declare netwrite K_Times Net_RoyalTimeAttack_PlayerData_Times for _Player = C_Default_PlayerTimes;
	Net_RoyalTimeAttack_PlayerData_Times = _Times;
	Net_RoyalTimeAttack_PlayerData_TimesEventUpdate += 1;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the player times event counter value (client side only)
Integer GetPlayerTimesEventUpdate(CSmPlayer _Player) {
	if (_Player == Null) return C_Default_PlayerTimesEventUpdate;
	declare netread Integer Net_RoyalTimeAttack_PlayerData_TimesEventUpdate for _Player = C_Default_PlayerTimesEventUpdate;
	return Net_RoyalTimeAttack_PlayerData_TimesEventUpdate;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the player times (client side only)
K_Times GetPlayerTimes(CSmPlayer _Player) {
	if (_Player == Null) return C_Default_PlayerTimes;
	declare netread K_Times Net_RoyalTimeAttack_PlayerData_Times for _Player = C_Default_PlayerTimes;
	return Net_RoyalTimeAttack_PlayerData_Times;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize player data (server side only)
Void InitializePlayer(CSmPlayer _Player) {
	if (_Player == Null) return;
	SendPlayerTimesEvent(_Player, C_Default_PlayerTimes);
}
