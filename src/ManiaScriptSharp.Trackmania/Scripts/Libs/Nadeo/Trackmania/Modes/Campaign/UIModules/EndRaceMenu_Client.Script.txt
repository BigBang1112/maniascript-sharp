K_State AddSound(K_State _State, Text _SoundName, Integer _PlayTime) {
	if (_PlayTime < Now) return _State;
	declare K_State State = _State;
	declare K_SoundToPlay[] NewWaitList;
	declare NewSoundInserted = False;
	declare NewSound = K_SoundToPlay {
		Name = _SoundName,
		PlayTime = _PlayTime
	};
	for (Index, 0, State.SoundsToPlay.count-1) {
		if (!NewSoundInserted && _PlayTime < State.SoundsToPlay[Index].PlayTime) {
			NewWaitList.add(NewSound);
			NewSoundInserted = True;
		}
		NewWaitList.add(State.SoundsToPlay[Index]);
	}
	if (!NewSoundInserted) NewWaitList.add(NewSound);
	State.SoundsToPlay = NewWaitList;

	return State;
}

K_State RemoveSounds(K_State _State, Text[] _SoundNames) {
	declare K_State State = _State;
	declare K_SoundToPlay[] NewWaitList;
	foreach (Sound in State.SoundsToPlay) {
		if (!_SoundNames.exists(Sound.Name)) {
			NewWaitList.add(Sound);
		}
	}
	foreach (SoundName in _SoundNames) {
		AudioManager::StopSound(This, SoundName);
	}
	State.SoundsToPlay = NewWaitList;

	return State;
}

K_State SetNewMedal(K_State _State, Integer _NewMedal) {
	declare K_State State = _State;
	declare Quad_Medal <=> (State.Controls.Frame_NewMedal.GetFirstChild("quad-medal") as CMlQuad);
	declare Quad_Medal_Anim <=> (State.Controls.Frame_NewMedal.GetFirstChild("quad-medal-anim") as CMlQuad);
	declare Quad_Banner <=> (State.Controls.Frame_NewMedal.GetFirstChild("quad-medal-banner") as CMlQuad);
	declare Quad_BannerGlow <=> (State.Controls.Frame_NewMedal.GetFirstChild("quad-medal-banner-glow") as CMlQuad);
	declare Integer TransitionDuration = C_Animation_MedalGain_TransitionDuration; //< 250 if no current medal
	if (State.CurrentMedal > 0) TransitionDuration = C_Animation_MedalGain_TransitionDuration * 2; //< 500 otherwise
	State.BannerSlideAnimation = K_Animation { IsAnimationFinished = True };
	State.MedalsAnimation = K_Animation {
		Images = [],
		GainSounds = [],
		StartTime = Now + C_Animation_MedalGain_Delay + C_Animation_MedalGain_BackgroundIntroDuration + TransitionDuration,
		IsAnimationFinished = False,
		LightOnTime = -1
	};

	// Prepare banners and medals
	if (!State.CanChallengeOpponents) {
		Quad_Banner.ChangeImageUrl("{{{Images::C_ImageUrl_ModeMenu_Banner_MedalsShort}}}");
		Quad_Banner.Size = {{{Size_NewMedalBannerSmall}}};
		Quad_BannerGlow.ChangeImageUrl("{{{Images::C_ImageUrl_ModeMenu_Banner_MedalsShortGlow}}}");
		Quad_BannerGlow.Size = {{{Size_NewMedalBannerSmall}}};
	} else {
		Quad_Banner.ChangeImageUrl("{{{Images::C_ImageUrl_ModeMenu_Banner_Medals}}}");
		Quad_Banner.Size = {{{Size_NewMedalBanner}}};
		Quad_BannerGlow.ChangeImageUrl("{{{Images::C_ImageUrl_ModeMenu_Banner_MedalsGlow}}}");
		Quad_BannerGlow.Size = {{{Size_NewMedalBanner}}};
	}
	Quad_Medal_Anim.ChangeImageUrl("");
	Quad_Medal.ChangeImageUrl("");
	Quad_Medal.Visible = State.CurrentMedal > 0;
	Quad_BannerGlow.Visible = State.CurrentMedal > 0;
	if (Quad_BannerGlow.Visible) Quad_BannerGlow.Opacity = 1.;
	else Quad_BannerGlow.Opacity = 0.;

	// Initialize medal list
	for (Medal, State.CurrentMedal, _NewMedal) {
		if (Medal == 0) continue;
		if (State.CurrentMedal > 0 && Medal == State.CurrentMedal) {
			Quad_Medal.ChangeImageUrl(GetMedalImage(Medal));
			Quad_Medal_Anim.ChangeImageUrl(GetMedalImage(Medal));
		} else {
			State.MedalsAnimation.Images[Medal] = GetMedalImage(Medal);
			State.MedalsAnimation.GainSounds[Medal] = GetMedalGainSound(Medal);
		}
	}

	// Display the medal frame for the upcoming medal animations
	State.Controls.Frame_NewMedal.Hide();
	State.Controls.Frame_NewMedal.RelativePosition_V3 = MLHelpers::SkewX(<0., {{{PosY_Banners}}}-200.>, 10.);
	AnimMgr.Flush(State.Controls.Frame_NewMedal);
	AnimMgr.Add(State.Controls.Frame_NewMedal, "<anim hidden=\"0\" pos=\"0. "^{{{PosY_Banners}}}^"\"/>", Now + C_Animation_MedalGain_Delay, C_Animation_MedalGain_BackgroundIntroDuration, CAnimManager::EAnimManagerEasing::Linear);
	State = AddSound(State, "{{{C_Sound_MedalIntro}}}", Now + C_Animation_MedalGain_Delay);

	// Update current medal
	State.CurrentMedal = _NewMedal;

	return State;
}

Void Focus(K_State _State, CMlControl _From, CMlControl _To) {
	if ({{{Transition::P}}}IsInTransition(This)) return;

	if (_From != Null) {
		{{{Navigation::P}}}Unfocus(This, _From);
		if ({{{Button::P}}}IsButton(_From)) {
			{{{Button::P}}}Unfocus(This, (_From as CMlFrame));
		} else if ({{{ExpendableButton::P}}}IsButton(_From)) {
			{{{ExpendableButton::P}}}Unfocus(This, (_From as CMlFrame));
		}
	}

	if (_To != Null) {
		{{{Navigation::P}}}Focus(This, _To);
		if ({{{Button::P}}}IsButton(_To)) {
			{{{Button::P}}}Focus(This, (_To as CMlFrame));
		} else if ({{{ExpendableButton::P}}}IsButton(_To)) {
			{{{ExpendableButton::P}}}Focus(This, (_To as CMlFrame));
			if (_To.Parent != Null && _To.Parent.DataAttributeExists("grid-row") && _To.Parent.DataAttributeGet("grid-row") == "{{{C_Row_Footer}}}") {
				{{{Navigation::P}}}ChangeControlTarget(This, _State.Controls.Button_Replay, "down", _To.ControlId);
				{{{Navigation::P}}}ChangeControlTarget(This, _State.Controls.Button_Improve, "up", _To.ControlId);
			}
		}
	}
}

Void Select(K_State _State, CMlControl _Control, {{{CampaignStruct::P}}}K_PlayerCloseToMedal PlayerCloseToMedal) {
	if (_Control == Null) return;

	switch (_Control.ControlId) {
		case "button-improve": SendCustomEvent("{{{Consts::C_EndRaceMenuEvent_Improve}}}", []);
		case "button-challenge": SendCustomEvent("{{{Consts::C_EndRaceMenuEvent_Challenge}}}", [PlayerCloseToMedal.AccountId, PlayerCloseToMedal.Medal]);
		case "button-opponents": SendCustomEvent("{{{Consts::C_RaceMenuEvent_ChangeOpponents}}}", []);
		case "button-next-map": SendCustomEvent("{{{Consts::C_RaceMenuEvent_NextMap}}}", []);
		case "button-exit": SendCustomEvent("{{{Consts::C_RaceMenuEvent_Exit}}}", []);
		case "button-vote-like": TrackVote::LikeTrack(This, Map, _State.Controls.Button_VoteLike, _State.Controls.Button_VoteDislike);
		case "button-vote-dislike": TrackVote::DislikeTrack(This, Map, _State.Controls.Button_VoteLike, _State.Controls.Button_VoteDislike);
		case "button-replay": {
			if ({{{Button::P}}}GetValue(_State.Controls.Button_Replay) == "{{{Text_SaveReplay}}}") {
				SendCustomEvent("{{{Consts::C_EndRaceMenuEvent_SaveReplay}}}", []);
			}
		}
	}
}

/// Update the library
Void Yield() {
	foreach (Event in PendingEvents) {
		switch (Event.PlaygroundType) {
			case CManiaAppPlaygroundEvent::Type::LayerCustomEvent: {
				declare Text[] EventData = Event.CustomEventData;
				switch (Event.CustomEventType) { //< Filters and passes UI events to the mode
					case Consts::C_EndRaceMenuEvent_Improve: SendCustomEvent(Event.CustomEventType, EventData);
					case Consts::C_EndRaceMenuEvent_Challenge: SendCustomEvent(Event.CustomEventType, EventData);
					case Consts::C_RaceMenuEvent_ChangeOpponents: SendCustomEvent(Event.CustomEventType, EventData);
					case Consts::C_RaceMenuEvent_NextMap: SendCustomEvent(Event.CustomEventType, EventData);
					case Consts::C_RaceMenuEvent_Exit: SendCustomEvent(Event.CustomEventType, EventData);
					case Consts::C_EndRaceMenuEvent_SaveReplay: SendCustomEvent(Event.CustomEventType, EventData);
					case Consts::C_EndRaceMenuEvent_SkipOutro: SendCustomEvent(Event.CustomEventType, EventData);
					case Consts::C_RaceMenuEvent_MapRecordsUpdated: SendCustomEvent(Event.CustomEventType, EventData);
				}
			}
		}
	}
