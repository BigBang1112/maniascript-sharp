Vec2 GetPosOfEventFromKey(Integer _Key) {
	if (_Key == 0) return <0., 0.>;
	return <-C_Size_Event.X, (C_Size_Event.Y + 1.) * (_Key - 1.) * -1.>;
}

Void UpdateEventLineContent(CMlFrame _Frame, Structs::K_Event _Event) {
	declare Integer TMWC2023_EventSerial for _Frame = -1;
	TMWC2023_EventSerial = _Event.Serial;

	declare Teams::K_Team[Integer] TeamsInfo = Teams::GetTeamsInfo(This);

	declare CMlQuad Quad_TeamLogo <=> (_Frame.GetFirstChild(C_Id_QuadTeamLogo) as CMlQuad);
	declare Text TeamLogo = Teams::GetClanLogoUrl(This, TeamsInfo, _Event.ClanId);
	if (TeamLogo == "") {
		Quad_TeamLogo.Colorize = <1., 1., 1.>;
		if (_Event.ClanId == 1) Quad_TeamLogo.ImageUrl = C_DefaultClan1Logo;
		else Quad_TeamLogo.ImageUrl = C_DefaultClan2Logo;
	} else {
		Quad_TeamLogo.Colorize = MLHelpers::C_DefaultColorize;
		Quad_TeamLogo.ImageUrl = TeamLogo;
	}

	declare CMlLabel Label <=> (_Frame.GetFirstChild(C_Id_LabelEvent) as CMlLabel);
	declare CMlQuad Quad_TeamLogoBg <=> (_Frame.GetFirstChild(C_Id_QuadTeamLogoBg) as CMlQuad);
	if (_Event.ClanId == 1) {
		Quad_TeamLogoBg.BgColor = Stylesheet::C_Color_Vec3_TMWC2023_BlueTeam_Primary;
		Label.TextColor = Stylesheet::C_Color_Vec3_TMWC2023_BlueTeam_Primary;
	} else if (_Event.ClanId == 2) {
		Quad_TeamLogoBg.BgColor = Stylesheet::C_Color_Vec3_TMWC2023_RedTeam_Primary;
		Label.TextColor = Stylesheet::C_Color_Vec3_TMWC2023_RedTeam_Primary;
	} else {
		Quad_TeamLogoBg.BgColor = Stylesheet::C_Color_Vec3_TMWC2023_Default;
		Label.TextColor = Stylesheet::C_Color_Vec3_TMWC2023_Default;
		Quad_TeamLogo.ImageUrl = "";
	}

	declare Text PlayerName;
	if (_Event.WebServicesUserId != "") {
		// In addition to its return value, `UserMgr.FindDisplayName()` will also update the value of the `IsFirstPartyDisplayName` variable.
		declare Boolean IsFirstPartyDisplayName;
		PlayerName = Teams::GetPlayerName(Teams::GetPlayersInfo(This), _Event.WebServicesUserId,  UserMgr.FindDisplayName(_Event.WebServicesUserId, IsFirstPartyDisplayName), IsFirstPartyDisplayName);
		if (PlayerName == "") { // Should never happen except for fake player. Also the FirstParty logo will not be displayed for them
			foreach (Score in Scores) {
				if (Score.User.WebServicesUserId != _Event.WebServicesUserId) continue;
				PlayerName = Score.User.Name;
				break;
			}
		}
	}

	switch (_Event.Type) {
		case Structs::C_EventType_Respawn: Label.Value = TL::Compose("%1 respawns", PlayerName);
		case Structs::C_EventType_WorldRecord: Label.Value = TL::Compose("%1 sets a new WR (%2)", PlayerName, _Event.Value);
		case Structs::C_EventType_PersonalBest: Label.Value = TL::Compose("%1 sets a new PB (%2)", PlayerName, _Event.Value);
		case Structs::C_EventType_Points: {
			if (_Event.Value == "1") Label.Value = TL::Compose("+1 point for %1", Teams::GetClanName(This, TeamsInfo, _Event.ClanId));
			else Label.Value = TL::Compose("+%1 points for %2", _Event.Value, Teams::GetClanName(This, TeamsInfo, _Event.ClanId));
		}
		default: Label.Value = _Event.Serial ^ " - " ^ _Event.Value;
	}
	Tools::FirstPartyLogoMatchLuminance(Label);
	Tools::FitLabelValue(Label, C_TextSize_LabelEvent, 1.5, 0.1);
}

Void CopyEventLineContent(CMlFrame _FrameSrc, CMlFrame _FrameDst) {
	declare Integer TMWC2023_EventSerial as TMWC2023_EventSerial_Src for _FrameSrc = -1;
	declare Integer TMWC2023_EventSerial as TMWC2023_EventSerial_Dst for _FrameDst = -1;
	TMWC2023_EventSerial_Dst = TMWC2023_EventSerial_Src;

	declare CMlQuad Quad_TeamLogoBg_Src <=> (_FrameSrc.GetFirstChild(C_Id_QuadTeamLogoBg) as CMlQuad);
	declare CMlQuad Quad_TeamLogoBg_Dst <=> (_FrameDst.GetFirstChild(C_Id_QuadTeamLogoBg) as CMlQuad);
	Quad_TeamLogoBg_Dst.BgColor = Quad_TeamLogoBg_Src.BgColor;

	declare CMlQuad Quad_TeamLogo_Src <=> (_FrameSrc.GetFirstChild(C_Id_QuadTeamLogo) as CMlQuad);
	declare CMlQuad Quad_TeamLogo_Dst <=> (_FrameDst.GetFirstChild(C_Id_QuadTeamLogo) as CMlQuad);
	Quad_TeamLogo_Dst.ImageUrl = Quad_TeamLogo_Src.ImageUrl;

	declare CMlLabel Label_Src <=> (_FrameSrc.GetFirstChild(C_Id_LabelEvent) as CMlLabel);
	declare CMlLabel Label_Dst <=> (_FrameDst.GetFirstChild(C_Id_LabelEvent) as CMlLabel);
	Label_Dst.Value = Label_Src.Value;
	Label_Dst.TextColor = Label_Src.TextColor;
}

Void UpdateEventLinePosition(CMlFrame _Frame, Integer _Key) {
	AnimMgr.Flush(_Frame);
	_Frame.Visible = True;
	_Frame.RelativePosition_V3 = GetPosOfEventFromKey(_Key);
	AnimMgr.Add(_Frame, "<a " ^ MLHelpers::Pos(GetPosOfEventFromKey(_Key + 1)) ^" />", C_EventAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
}

K_State UpdateEvents(K_State _State, Integer _QueueUpdate, Structs::K_Event[Integer] _Queue) {
	declare K_State State = _State;
	State.QueueUpdate = _QueueUpdate;
	declare Boolean NewAnimation;

	declare Integer NextEventSerialToDisplay = -1;
	if (_Queue.count > 0) {
		foreach (Serial => Event in _Queue) {
			if (!State.DisplayedEvents.existskey(Serial)) {
				NextEventSerialToDisplay = Serial;
				break;
			}
		}
	}

	foreach (Key => Frame in State.Controls.Frames_Events reverse) {
		// Previous in the order the frames, but not in the order of the loop
		declare CMlFrame PreviousFrame = State.Controls.Frames_Events[Key - 1];
		declare Integer TMWC2023_EventSerial as TMWC2023_EventSerial_PreviousFrame for PreviousFrame = -1;

		if (NextEventSerialToDisplay > -1 && TMWC2023_EventSerial_PreviousFrame >= 0 && Key > 0) {
			NewAnimation = True;
			CopyEventLineContent(PreviousFrame, Frame);
			UpdateEventLinePosition(Frame, Key);
		}

		declare Integer TMWC2023_EventSerial for Frame = -1;

		if (TMWC2023_EventSerial >= 0 && (!_Queue.existskey(TMWC2023_EventSerial) || Key == C_MaxEventsDisplayed)) {
			TMWC2023_EventSerial = -1;
			NewAnimation = True;

			AnimMgr.Flush(Frame);
			AnimMgr.Add(Frame, "<a pos=\"0 "^ Frame.RelativePosition_V3.Y ^"\" hidden=\"1\"/>", C_EventAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
		}
	}

	if (NextEventSerialToDisplay > -1) {
		State.DisplayedEvents[NextEventSerialToDisplay] = _Queue[NextEventSerialToDisplay];
		State.LastEventSerial = NextEventSerialToDisplay;

		NewAnimation = True;

		UpdateEventLineContent(State.Controls.Frames_Events[0], _Queue[NextEventSerialToDisplay]);
		UpdateEventLinePosition(State.Controls.Frames_Events[0], 0);
	}

	if (NewAnimation) {
		State.AnimationCooldown = Now + C_EventAnimDuration;
	} else {
		State.AnimationCooldown = -1;
	}

	return State;
}


K_State SetVisibility(K_State _State, Boolean _DisplayModule, Boolean _IsSpectating) {
	declare K_State State = _State;

	State.DisplayModule = _DisplayModule;
	State.IsSpectating = _IsSpectating;
	State.IsVisible = (
		State.DisplayModule &&
		State.IsSpectating
	);

	State.Controls.Frame_Global.Visible = State.IsVisible;
	return State;
}
