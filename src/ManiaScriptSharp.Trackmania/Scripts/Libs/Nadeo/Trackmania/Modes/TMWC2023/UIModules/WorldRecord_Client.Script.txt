K_State SetVisibility(K_State _State, Boolean _DisplayModule, Boolean _IsSpectating, Structs::K_Record _WorldRecord) {
	declare K_State State = _State;

	State.DisplayModule = _DisplayModule;
	State.IsSpectating = _IsSpectating;
	State.IsVisible = (
		State.DisplayModule &&
		State.IsSpectating &&
		_WorldRecord.Time > 0 &&
		State.PlayerName != ""
	);

	AnimMgr.Flush(State.Controls.Frame_AnimVisibility);
	if (State.IsVisible) {
		AnimMgr.Add(State.Controls.Frame_AnimVisibility, "<a pos=\"0 0\" hidden=\"0\" />", C_VisibilityAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
	} else {
		AnimMgr.Add(State.Controls.Frame_AnimVisibility, "<a pos=\""^ - C_Size_Background.X - C_Size_Title.X ^" 0\" hidden=\"1\" />", C_VisibilityAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
	}
	return State;
}

K_State SetRecord(K_State _State, Integer _RecordUpdate, Structs::K_Record _Record, Boolean _RetrieveDisplayName) {
	declare K_State State = _State;

	State.RecordUpdate = _RecordUpdate;

	if (_Record.WebServicesUserId == "") {
		State.Controls.Label_Record.Value = "";
		State.PlayerName = "";
	} else {
		declare Text PlayerName;
		declare Text LogoUrl;

		foreach (Team in Teams::GetAllTeamsInfo(This)) {
			foreach (Player in Team.Players) {
				if (Player.AccountId == _Record.WebServicesUserId) {
					PlayerName = Player.Name;
					LogoUrl = Team.LogoUrl;
					break;
				}
			}
			if (PlayerName != "") break;
		}

		// not a TMGL player, trying to get their Ubisoft name
		if (PlayerName == "") {
			PlayerName = UserStore::GetUserMgrPlayerName(_Record.WebServicesUserId);
		}

		State.PlayerName = PlayerName;

		if (State.PlayerName != "") {
			declare Text RecordTime = TL::TimeToText(_Record.Time, True, True);
			State.Controls.Label_Record.Value = RecordTime;
			State.Controls.Quad_PlayerIcon.ImageUrl = LogoUrl;

			State.Controls.Label_Player.Value = State.PlayerName;
			Tools::FirstPartyLogoMatchLuminance(State.Controls.Label_Player);
			if (LogoUrl == "") {
				State.Controls.Label_Player.Size.X = C_Size_LabelPlayerName.X;
			} else {
				State.Controls.Label_Player.Size.X = C_Size_LabelPlayerName.X - C_Size_Icons.X - C_Size_MarginIcon;
			}
			Tools::FitLabelValue(State.Controls.Label_Player, C_TextSize_Labels, 1., 0.25);

			// Adjust center
			declare Real CumulatedSize = C_Size_Icons.X + C_Size_MarginIcon;

			CumulatedSize += ML::Min(State.Controls.Label_Record.ComputeWidth(RecordTime), State.Controls.Label_Record.Size.X);

			State.Controls.Label_Separator.RelativePosition_V3.X = CumulatedSize;

			CumulatedSize += C_Size_LabelSeparator.X;

			State.Controls.Quad_PlayerIcon.RelativePosition_V3.X = CumulatedSize;

			if (LogoUrl != "") {
				CumulatedSize += C_Size_Icons.X + C_Size_MarginIcon;
			}
			State.Controls.Label_Player.RelativePosition_V3.X = CumulatedSize;

			CumulatedSize += ML::Min(State.Controls.Label_Player.ComputeWidth(State.PlayerName), State.Controls.Label_Player.Size.X);

			State.Controls.Frame_Record.RelativePosition_V3.X = C_Size_Background.X / 2. - CumulatedSize / 2.;
		} else if (_RetrieveDisplayName) {
			State.Task_RetrieveDisplayName = Task::DestroyAndCreate(State.Task_RetrieveDisplayName, UserMgr, UserMgr.RetrieveDisplayName(MainUser::GetMainUserId(), [_Record.WebServicesUserId]));
		}
	}

	return SetVisibility(State, _State.DisplayModule, _State.IsSpectating, _Record);
}
