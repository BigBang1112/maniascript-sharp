// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_Record {
	Text WebServicesUserId;
	Integer Time;
}

#Struct K_Event {
	Integer Serial; // unique value
	Integer Time;
	Integer Type;
	Integer ClanId;
	Text WebServicesUserId;
	Text Value;
}

#Struct K_PlayerStats {
	Integer Top1;
	Integer Crashes;
	Integer NumberOfRuns;
}

#Struct K_HttpPlayerStats {
	Integer top1;
	Integer crashes;
	Integer numberOfRuns;
}
#Struct K_HttpGetPostPlayerStats {
	Text competitionUid;
	K_HttpPlayerStats[Text] stats;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Cast `K_HttpPlayerStats` into `K_PlayerStats`
K_PlayerStats ToPlayerStats(K_HttpPlayerStats _HttpPlayerStats) {
	return K_PlayerStats {
		Top1 = _HttpPlayerStats.top1,
		Crashes = _HttpPlayerStats.crashes,
		NumberOfRuns = _HttpPlayerStats.numberOfRuns
	};
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Cast `K_PlayerStats` into `K_HttpPlayerStats`
K_HttpPlayerStats ToHttpPlayerStats(K_PlayerStats _PlayerStats) {
	return K_HttpPlayerStats {
		top1 = _PlayerStats.Top1,
		crashes = _PlayerStats.Crashes,
		numberOfRuns = _PlayerStats.NumberOfRuns
	};
}
