/**
 * Read clone metadata in the track file
 */
#Const Version "1.0.0"
#Const ScriptName "Libs/Nadeo/Trackmania/Modes/Clone/MetadataMode.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/Trackmania/Modes/Clone/Clone.Script.txt" as Clone

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Private_ClonesMaxNb 128

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the clone feature is enabled
Boolean IsCloneEnabled(CMap _Map) {
	declare metadata Boolean Nadeo_IsCloneEnabled for _Map = Clone::C_IsCloneEnableDefault;
	return Nadeo_IsCloneEnabled;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the list of clone configs
Int2[] GetCloneConfigList(CMap _Map) {
	// /!\ DO NOT CHANGE THIS VARIABLE NAME /!\
	// The C++ reads the `Race_AuthorClones` variable to update the map header
	declare metadata Int2[] Race_AuthorClones for _Map = [];
	return Race_AuthorClones.slice(0, C_Private_ClonesMaxNb);
}
