/// Get the trophy image from its level
Text Private_GetTrophyImage(Integer _TrophyLevel) {
	switch (_TrophyLevel) {
		case 1: return Const::C_ImageUrl_Trophy_512_Bronze1;
		case 2: return Const::C_ImageUrl_Trophy_512_Bronze2;
		case 3: return Const::C_ImageUrl_Trophy_512_Bronze3;
		case 4: return Const::C_ImageUrl_Trophy_512_Silver1;
		case 5: return Const::C_ImageUrl_Trophy_512_Silver2;
		case 6: return Const::C_ImageUrl_Trophy_512_Silver3;
		case 7: return Const::C_ImageUrl_Trophy_512_Gold1;
		case 8: return Const::C_ImageUrl_Trophy_512_Gold2;
		case 9: return Const::C_ImageUrl_Trophy_512_Gold3;
	}
	return Const::C_ImageUrl_Trophy_512_None;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the trophy reward for the competition match
K_TrophyReward Private_GetTrophyReward(Integer[Integer][Integer] _RankingTrophyList, Integer _Rank) {
	if (_Rank <= 0) {
		return K_TrophyReward {
			ImageUrl = Private_GetTrophyImage(0),
			Level = 0,
			Count = 0
		};
	}

	for (Rank => TrophyList in _RankingTrophyList.sortkey()) {
		if (_Rank <= Rank) {
			for (Level => Count in TrophyList.sortkeyreverse()) {
				return K_TrophyReward {
					ImageUrl = Private_GetTrophyImage(Level),
					Level = Level,
					Count = Count
				};
			}
		}
	}

	return K_TrophyReward {
		ImageUrl = Private_GetTrophyImage(0),
		Level = 0,
		Count = 0
	};
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the trophy reward for the competition match
K_TrophyReward GetQualifiersTrophyReward(Text _CompetitionType, Integer _Rank) {
	return Private_GetTrophyReward(C_QualifiersTrophyRewardList.get(_CompetitionType, []), _Rank);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the trophy reward for the competition match
K_TrophyReward GetMatchTrophyReward(Text _CompetitionType, Text _Division, Integer _Rank) {
	return Private_GetTrophyReward(C_MatchTrophyRewardList.get(_CompetitionType, []).get(_Division, []), _Rank);
}
