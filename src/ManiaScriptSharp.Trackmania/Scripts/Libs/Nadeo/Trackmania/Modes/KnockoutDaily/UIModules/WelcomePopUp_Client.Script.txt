Void SetTrophyImage(CMlQuad _Quad_Trophy, CMlLabel _Label_Reward, Text _Division, Text _CompetitionType) {
	if (_Quad_Trophy == Null) return;

	declare Competition::K_TrophyReward TrophyReward = Competition::GetMatchTrophyReward(_CompetitionType, _Division, 1);
	if (TrophyReward.Level > 0 && TrophyReward.Count > 0) {
		_Quad_Trophy.ImageUrl = TrophyReward.ImageUrl;
		_Quad_Trophy.Visible = True;
		_Label_Reward.Value = TL::Compose(C_Text_Reward, ""^TrophyReward.Count, ""^TrophyReward.Level);
		_Label_Reward.Visible = True;
	} else {
		_Quad_Trophy.Visible = False;
		_Label_Reward.Visible = False;
	}
}

Text SetCompetitionName(CMlLabel _Label_Title, CMlLabel _Label_Header, CMlLabel _Label_Info, Text _CompetitionName) {
	if (_CompetitionName == "") {
		_Label_Title.Value = C_Text_TitleCOTD;
		_Label_Header.Value = C_Text_HeaderCOTD;
		_Label_Info.Value = C_Text_InfoCOTD;
	} else {
		declare Text TranslatedCompetitionName = TL::GetTranslatedText(_CompetitionName);
		_Label_Title.Value = TranslatedCompetitionName;
		_Label_Header.Value = TL::Compose(C_Text_HeaderGeneric, TranslatedCompetitionName);
		_Label_Info.Value = C_Text_InfoGeneric;
	}
	return _CompetitionName;
}

Void Focus(CMlControl _From, CMlControl _To) {
	if (_From != Null) {
		{{{Navigation::P}}}Unfocus(This, _From);
		if ({{{ExpendableButton::P}}}IsButton(_From)) {
			{{{ExpendableButton::P}}}Unfocus(This, (_From as CMlFrame));
		}

	if (_To != Null) {
		{{{Navigation::P}}}Focus(This, _To);
		if ({{{ExpendableButton::P}}}IsButton(_To)) {
			{{{ExpendableButton::P}}}Focus(This, (_To as CMlFrame));
		}
	}
}

Void Toggle(CMlFrame _Frame_PopUp, Boolean _Enabled) {
	_Frame_PopUp.Visible = _Enabled;
}

Void Select(CMlFrame _Frame_PopUp, CMlControl _Control) {
	if (_Control == Null) return;

	switch (_Control.ControlId) {
		case "button-ok": Toggle(_Frame_PopUp, False);
	}
