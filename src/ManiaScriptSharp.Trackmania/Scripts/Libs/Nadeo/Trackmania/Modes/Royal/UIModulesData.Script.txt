/**
 *	Share data between client UI modules
 */
#Const Version		"2.1.0"
#Const ScriptName	"Libs/Nadeo/Trackmania/Modes/Royal/UIModulesData.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Default_Size <0., 0.>
#Const C_Default_Position <0., 0.>
#Const C_Default_Timestamp 0

#Const C_SpecModeActivationDelay 1000

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the size of the given UI module
Void SetSize(CMlScript _Context, Text _UIModuleId, Vec2 _Size) {
	if (_Context == Null) return;
	declare Vec2[Text] Royal_UIModulesData_Sizes for _Context = [];
	Royal_UIModulesData_Sizes[_UIModuleId] = _Size;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the size of the given UI module
Vec2 GetSize(CMlScript _Context, Text _UIModuleId) {
	if (_Context == Null) return C_Default_Size;
	declare Vec2[Text] Royal_UIModulesData_Sizes for _Context = [];
	return Royal_UIModulesData_Sizes.get(_UIModuleId, C_Default_Size);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the position of the given UI module
Void SetPosition(CMlScript _Context, Text _UIModuleId, Vec2 _Position) {
	if (_Context == Null) return;
	declare Vec2[Text] Royal_UIModulesData_Positions for _Context = [];
	Royal_UIModulesData_Positions[_UIModuleId] = _Position;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the position of the given UI module
Vec2 GetPosition(CMlScript _Context, Text _UIModuleId) {
	if (_Context == Null) return C_Default_Position;
	declare Vec2[Text] Royal_UIModulesData_Positions for _Context = [];
	return Royal_UIModulesData_Positions.get(_UIModuleId, C_Default_Position);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the ending timestamp for the completion of a SegmentComplete UI animation
Void SetSegmentCompleteEndAnimTimestamp(CMlScript _Context, Integer _EndAnimTimestamp) {
	if (_Context == Null) return;
	declare Integer Royal_UIModulesData_SegmentComplete_EndAnimTimestamp for _Context = C_Default_Timestamp;
	Royal_UIModulesData_SegmentComplete_EndAnimTimestamp = _EndAnimTimestamp;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the ending timestamp for the completion of a SegmentComplete UI animation
Integer GetSegmentCompleteEndAnimTimestamp(CMlScript _Context) {
	if (_Context == Null) return C_Default_Timestamp;
	declare Integer Royal_UIModulesData_SegmentComplete_EndAnimTimestamp for _Context = C_Default_Timestamp;
	return Royal_UIModulesData_SegmentComplete_EndAnimTimestamp;
}
