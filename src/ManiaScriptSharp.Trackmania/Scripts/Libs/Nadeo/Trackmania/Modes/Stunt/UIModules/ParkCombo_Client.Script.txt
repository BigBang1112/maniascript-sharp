/// Get the remaining time in the given combo
Integer GetRemainingTime(PlayerData::K_ParkComboEvent _ParkComboEvent) {
	return _ParkComboEvent.EndTime - ML::Max(0, _ParkComboEvent.StartTime - GameTime) - GameTime;
}

/// Update the combo timer
K_State UpdateComboTimer(K_State _State, PlayerData::K_ParkComboEvent _ParkComboEvent, Boolean _IsFasterScoreAnim) {
	declare K_State State = _State;
	declare Integer RemainingTime = GetRemainingTime(_ParkComboEvent);

	// Trigger score animation
	State.ScoreAnim = UIModulesData::K_ScoreAnim {
		CurrentScore = Tools::Ternary(State.ScoreAnim.CurrentScore <= _ParkComboEvent.Score, State.ScoreAnim.CurrentScore, 0),
		OldScore = Tools::Ternary(State.ScoreAnim.CurrentScore <= _ParkComboEvent.Score, State.ScoreAnim.CurrentScore, 0),
		NewScore = _ParkComboEvent.Score,
		Duration = Tools::Ternary(_IsFasterScoreAnim, Const::C_AnimDuration_FasterScoreIncrease, Const::C_AnimDuration_ScoreIncrease),
		StartTime = Now
	};

	// Prepare controls for the upcoming animations
	if (RemainingTime > 0 && _ParkComboEvent.Duration > 0) {
		State.Controls.Quad_ComboScoreBg.Visible = _ParkComboEvent.Score > 0;
		State.Controls.Quad_ComboTimer.BgColor = CL::Hex6ToRgb(C_ColorBg_Parks.get(_ParkComboEvent.ParkId, "FFFFFF"));
		State.Controls.Quad_ComboTimer.Size.X = ML::Max(0., {{{Size_ComboTimer.X}}} * (ML::ToReal(RemainingTime) / ML::ToReal(_ParkComboEvent.Duration)));
		State.Controls.Label_ComboScore.TextColor = CL::Hex6ToRgb(C_ColorBg_Parks.get(_ParkComboEvent.ParkId, "FFFFFF"));
		if (!State.Controls.Quad_ScreenGradient.Visible) State.Controls.Quad_ScreenGradient.Colorize = CL::Hex6ToRgb(C_ColorBg_Parks.get(_ParkComboEvent.ParkId, "FFFFFF"));
		if (_ParkComboEvent.Score <= 0) State.Controls.Label_ComboScore.SetText("");
	}

	// Update the frame visibility
	AnimMgr.Flush(_State.Controls.Frame_Combo);
	if (RemainingTime > 0) {
		AnimMgr.Add(_State.Controls.Frame_Combo, "<anim size=\""^{{{Size_FrameCombo.X}}}^" "^{{{Size_FrameCombo.Y}}}^"\" hidden=\"0\"/>", 500, CAnimManager::EAnimManagerEasing::QuadIn);
	} else {
		AnimMgr.Add(_State.Controls.Frame_Combo, "<anim size=\"0 "^{{{Size_FrameCombo.Y}}}^"\" hidden=\"1\"/>", 500, CAnimManager::EAnimManagerEasing::QuadOut);
	}

	// Gradient animations
	if (RemainingTime > 0) {
		AnimMgr.Add(State.Controls.Quad_ScreenGradient, "<anim opacity=\"1.\" colorize=\""^C_ColorBg_Parks.get(_ParkComboEvent.ParkId, "FFFFFF")^"\" hidden=\"0\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	} else {
		AnimMgr.Add(State.Controls.Quad_ScreenGradient, "<anim opacity=\"0.\" hidden=\"1\"/>", 500, CAnimManager::EAnimManagerEasing::QuadOut);
	}

	// Timer opacity animation
	if (RemainingTime > 0 && !_ParkComboEvent.IsPaused) {
		AnimMgr.Add(State.Controls.Quad_ComboTimer, "<anim opacity=\"1.\" />", 200, CAnimManager::EAnimManagerEasing::Linear);
	}	else {
		AnimMgr.Add(State.Controls.Quad_ComboTimer, "<anim opacity=\"0.6\" />", 200, CAnimManager::EAnimManagerEasing::Linear);
	}

	return State;
}

/// Reset the module
K_State ResetModule(K_State _State) {
	declare K_State State = _State;

	State.ScoreAnim = UIModulesData::K_ScoreAnim {};
	State.LastParkComboEvent = PlayerData::C_Default_PlayerParkComboEvent;
	State = UpdateComboTimer(State, PlayerData::C_Default_PlayerParkComboEvent, False);

	return State;
}

/// Trigger a combo event
K_State TriggerParkComboEvent(K_State _State, CSmPlayer _Player) {
	if (_Player == Null) return _State;
	declare K_State State = _State;
	declare Boolean IsSpectatingAnotherTarget = IsSpectatorClient && State.ParkComboEventUpdate < 0; //< Faster animations when switching target as a spectator

	State.ParkComboEventUpdate = PlayerData::GetPlayerParkComboEventUpdate(_Player);
	State.LastParkComboEvent = PlayerData::GetPlayerParkComboEvent(_Player);
	if (State.LastParkComboEvent.IsEndingActiveCombo || !(State.LastParkComboEvent.EndTime < 0 && State.ScoreAnim.CurrentScore < State.ScoreAnim.NewScore)) {
		State = UpdateComboTimer(State, State.LastParkComboEvent, IsSpectatingAnotherTarget);
	}

	return State;
}

/// Update the visibility of the whole manialink UI
K_State UpdateVisibility(K_State _State, Boolean _IsPageVisible, Boolean _IsModuleDisplayed) {
	declare K_State State = _State;

	State.IsPageVisible = _IsPageVisible;
	State.IsModuleDisplayed = _IsModuleDisplayed;
	State.IsVisible = State.IsPageVisible && State.IsModuleDisplayed && State.TargetId != NullId;
	State.Controls.Frame_Global.Visible = State.IsVisible;

	// Reset module if hidden
	if (!State.Controls.Frame_Global.Visible) {
		State = ResetModule(State);
	}

	return State;
}

/// Check if the current target has changed
Boolean IsTargetUpdateRequired(K_State _State, CSmPlayer _Player) {
	return (
		(_Player != Null && _State.TargetId != _Player.Id) ||
		(_Player == Null && _State.TargetId != NullId)
	);
}

/// Set the player currently followed by the camera
K_State SetTarget(K_State _State, CSmPlayer _Player) {
	declare K_State State = _State;

	// Update target data
	if (_Player == Null) {
		State.TargetId = NullId;
	} else {
		State.ParkComboEventUpdate = -1;
		State.TargetId = _Player.Id;
	}

	// Refresh UI
	State = UpdateVisibility(State, State.IsPageVisible, State.IsModuleDisplayed);
	State = ResetModule(State);

	return State;
}
