/// Display the new figure made by the player
Void DisplayFigure(K_State _State, PlayerData::K_FigureEvent _FigureEvent) {
	if (GameTime > _FigureEvent.Timestamp + 0.8*Const::C_AnimDuration_NewFigure) return;
	declare CMlFrame Frame_Align = (_State.Controls.Frame_Figure.GetFirstChild("frame-align") as CMlFrame);
	declare CMlQuad Quad_Background = (_State.Controls.Frame_Figure.GetFirstChild("quad-background") as CMlQuad);
	declare CMlLabel Label_FigureName = (_State.Controls.Frame_Figure.GetFirstChild("label-figure-name") as CMlLabel);
	declare CMlLabel Label_FigureAngle = (_State.Controls.Frame_Figure.GetFirstChild("label-figure-angle") as CMlLabel);
	declare CMlLabel Label_FigurePoints = (_State.Controls.Frame_Figure.GetFirstChild("label-figure-points") as CMlLabel);
	declare CMlLabel Label_FigureMultiplier = (_State.Controls.Frame_Figure.GetFirstChild("label-figure-multiplier") as CMlLabel);
	declare Integer DisplayDuration = ML::Max(0, _FigureEvent.Timestamp + Const::C_AnimDuration_NewFigure - GameTime);
	declare CVideo LandingVideo = _State.LandingVideos.get(_FigureEvent.LandingType, Null);
	declare Text LandingSound = C_SoundNames_FiguresLanding.get(_FigureEvent.LandingType, "");

	// Update figure info
	Label_FigureName.SetText(_FigureEvent.Name);
	Label_FigureAngle.SetText(""^_FigureEvent.Angle^"° |");
	Label_FigurePoints.SetText(TL::Compose(MainMenuConst::C_Text_Pts, TL::FormatInteger(_FigureEvent.Points, 1, True)));
	Label_FigureMultiplier.SetText(Tools::Ternary(_FigureEvent.Multiplier > Const::C_DefaultLandingMultiplier, "x"^TL::FormatReal(_FigureEvent.Multiplier, 1, False, False), ""));
	Label_FigureMultiplier.TextColor = CL::Hex6ToRgb(C_Colors_FiguresLanding.get(_FigureEvent.LandingType, "FFFFFF"));
	Label_FigureAngle.RelativePosition_V3.X = Label_FigureName.RelativePosition_V3.X + Label_FigureName.ComputeWidth(Label_FigureName.Value) + 1.;
	Label_FigurePoints.RelativePosition_V3.X = Label_FigureAngle.RelativePosition_V3.X + Label_FigureAngle.ComputeWidth(Label_FigureAngle.Value) + 1.;
	Label_FigureMultiplier.RelativePosition_V3.X = Label_FigurePoints.RelativePosition_V3.X + Label_FigurePoints.ComputeWidth(Label_FigurePoints.Value) + 1.;
	Quad_Background.Size.X = ML::Abs(Label_FigureMultiplier.RelativePosition_V3.X + Label_FigureMultiplier.ComputeWidth(Label_FigureMultiplier.Value) + 1.);
	Tools::AlignFrame(Frame_Align, Quad_Background.Size, CMlControl::AlignHorizontal::HCenter, CMlControl::AlignVertical::VCenter);

	// Update and play landing video
	if (LandingVideo != Null) {
		_State.Controls.Quad_FigureLanding.Size = 0.05 * C_VideoResolutions_FiguresLanding.get(_FigureEvent.LandingType, <0., 0.>);
		_State.Controls.Quad_FigureLanding.Image = LandingVideo.Image;
		_State.Controls.Quad_FigureLanding.Show();
		LandingVideo.Stop();
		LandingVideo.Play();
	}

	// Play sound landing
	if (LandingSound != "") {
		AudioManager::PlaySound(This, LandingSound);
	}

	// Reset for the upcoming animations
	Label_FigureName.Opacity = 0.;
	Label_FigureAngle.Opacity = 0.;
	Label_FigurePoints.Opacity = 0.;
	Label_FigureMultiplier.Opacity = 0.;
	Quad_Background.Opacity = 0.;
	_State.Controls.Frame_Figure.Scale = 0.9;

	// Animations
	AnimMgr.Flush(_State.Controls.Frame_Figure);
	AnimMgr.Add(_State.Controls.Frame_Figure, "<anim scale=\"1.\" hidden=\"0\" />", 500, CAnimManager::EAnimManagerEasing::BackOut);
	AnimMgr.Add(_State.Controls.Frame_Figure, "<anim scale=\"0.9\" hidden=\"1\" />", Now + DisplayDuration - 250, 500, CAnimManager::EAnimManagerEasing::BackIn);

	AnimMgr.Flush(Quad_Background);
	AnimMgr.Add(Quad_Background, "<anim opacity=\"0.6\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	AnimMgr.Add(Quad_Background, "<anim opacity=\"0.\" />", Now + DisplayDuration, 250, CAnimManager::EAnimManagerEasing::QuadOut);

	foreach (Label in [Label_FigureName, Label_FigureAngle, Label_FigurePoints, Label_FigureMultiplier]) {
		AnimMgr.Flush(Label);
		AnimMgr.Add(Label, "<anim opacity=\"1.\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		AnimMgr.Add(Label, "<anim opacity=\"0.\" />", Now + DisplayDuration, 250, CAnimManager::EAnimManagerEasing::QuadOut);
	}
}

/// Reset the module
K_State ResetModule(K_State _State) {
	declare K_State State = _State;

	State.Controls.Frame_Figure.Hide();
	State.Controls.Quad_FigureLanding.Hide();
	foreach (LandingVideo in State.LandingVideos) {
		LandingVideo.Stop();
	}

	return State;
}

/// Trigger a figure event
K_State TriggerFigureEvent(K_State _State, CSmPlayer _Player) {
	if (_Player == Null) return _State;
	declare K_State State = _State;

	State.FigureEventUpdate = PlayerData::GetPlayerFigureEventUpdate(_Player);
	if (PlayerData::GetPlayerFigureEvent(_Player).Name != "") {
		DisplayFigure(State, PlayerData::GetPlayerFigureEvent(_Player));
	} else {
		State = ResetModule(State);
	}

	return State;
}

/// Update the visibility of the whole manialink UI
K_State UpdateVisibility(K_State _State, Boolean _IsPageVisible, Boolean _IsModuleDisplayed) {
	declare K_State State = _State;

	State.IsPageVisible = _IsPageVisible;
	State.IsModuleDisplayed = _IsModuleDisplayed;
	State.IsVisible = State.IsPageVisible && State.IsModuleDisplayed && State.TargetId != NullId;
	State.Controls.Frame_Global.Visible = State.IsVisible;

	// Reset module if hidden
	if (!State.Controls.Frame_Global.Visible) {
		State = ResetModule(State);
	}

	// Create or destroy landing videos
	if (State.Controls.Frame_Global.Visible) {
		foreach (LandingType => LandingVideo in C_VideoUrls_FiguresLanding) {
			State.LandingVideos[LandingType] = Video.CreateVideo(LandingVideo, False, True, True);
		}
	} else {
		foreach (LandingType => LandingVideo in State.LandingVideos) {
			if (LandingVideo != Null && LandingVideo.Id != NullId && Video.Videos.existskey(LandingVideo.Id)) {
				State.LandingVideos.removekey(LandingType);
				Video.DestroyVideo(LandingVideo);
			}
		}
	}

	return State;
}

/// Check if the current target has changed
Boolean IsTargetUpdateRequired(K_State _State, CSmPlayer _Player) {
	return (
		(_Player != Null && _State.TargetId != _Player.Id) ||
		(_Player == Null && _State.TargetId != NullId)
	);
}

/// Set the player currently followed by the camera
K_State SetTarget(K_State _State, CSmPlayer _Player) {
	declare K_State State = _State;

	// Update target data
	if (_Player == Null) {
		State.TargetId = NullId;
	} else {
		State.FigureEventUpdate = -1;
		State.TargetId = _Player.Id;
	}

	// Refresh UI
	State = UpdateVisibility(State, State.IsPageVisible, State.IsModuleDisplayed);
	State = ResetModule(State);

	return State;
}
