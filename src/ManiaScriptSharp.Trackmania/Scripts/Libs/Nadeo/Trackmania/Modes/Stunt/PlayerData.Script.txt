/**
 * Share player specific data between the server game mode and the clients UI
 */
#Const Version "1.1.0"
#Const ScriptName "Libs/Nadeo/Trackmania/Modes/Stunt/PlayerData.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structure
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_FigureEvent {
	Integer Angle;
	Integer ParkId;
	Integer Points;
	Integer Timestamp;
	Real Multiplier;
	Text LandingType;
	Text Name;
}

#Struct K_ParkComboEvent {
	Boolean IsEndingActiveCombo;
	Boolean IsPaused;
	Integer Duration;
	Integer EndTime;
	Integer ParkId;
	Integer Score;
	Integer StartTime;
}

#Struct K_ParkData {
	Integer Id;
	Integer Score;
}
#Struct K_ParkEvent {
	K_ParkData[] Parks;
	Boolean IsParksInitialisation;
}

#Struct K_Stats {
	Integer FiguresCount;
	Integer[Text] LandingsCount;
	Text BestFigure;
}

#Struct K_FinishEvent {
	Integer TotalScore;
	K_ParkData[] Parks;
	K_Stats Stats;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Default_PlayerFigureEventUpdate 0
#Const C_Default_PlayerFigureEvent K_FigureEvent {
	Angle = 0,
	ParkId = -1, //< C_ParkId_Null
	Points = 0,
	Timestamp = 0,
	Multiplier = 1.0,
	LandingType = "",
	Name = ""
}
#Const C_Default_PlayerTotalScoreEventUpdate 0
#Const C_Default_PlayerTotalScore 0
#Const C_Default_PlayerParkComboEventUpdate 0
#Const C_Default_PlayerParkComboEvent K_ParkComboEvent {
	IsEndingActiveCombo = True,
	IsPaused = False,
	Duration = -1,
	EndTime = -1,
	ParkId = -1, //< C_ParkId_Null
	Score = 0,
	StartTime = -1
}
#Const C_Default_PlayerParkEventUpdate 0
#Const C_Default_PlayerParkEvent K_ParkEvent {
	Parks = [],
	IsParksInitialisation = False
}
#Const C_Default_PlayerFinishEventUpdate 0
#Const C_Default_PlayerFinishEvent K_FinishEvent {
	TotalScore = 0,
	Parks = [],
	Stats = K_Stats {}
}

// MARK: Functions

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Send a player total score event to the clients (server side only)
Void SendPlayerTotalScoreEvent(CSmPlayer _Player, Integer _TotalScore) {
	if (_Player == Null) return;
	declare netwrite Integer Net_Stunt_PlayerData_TotalScoreEventUpdate for _Player = C_Default_PlayerTotalScoreEventUpdate;
	declare netwrite Integer Net_Stunt_PlayerData_TotalScore for _Player = C_Default_PlayerTotalScore;
	Net_Stunt_PlayerData_TotalScore = _TotalScore;
	Net_Stunt_PlayerData_TotalScoreEventUpdate += 1;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the player total score event counter value (client side only)
Integer GetPlayerTotalScoreEventUpdate(CSmPlayer _Player) {
	if (_Player == Null) return C_Default_PlayerTotalScoreEventUpdate;
	declare netread Integer Net_Stunt_PlayerData_TotalScoreEventUpdate for _Player = C_Default_PlayerTotalScoreEventUpdate;
	return Net_Stunt_PlayerData_TotalScoreEventUpdate;
}

/// Initialize player data (server side only)
Void InitializePlayer(CSmPlayer _Player) {
	if (_Player == Null) return;
	SendPlayerTotalScoreEvent(_Player, C_Default_PlayerTotalScore);
	SendPlayerFigureEvent(_Player, C_Default_PlayerFigureEvent);
	SendPlayerParkComboEvent(_Player, C_Default_PlayerParkComboEvent);
	SendPlayerParkEvent(_Player, C_Default_PlayerParkEvent);
	SendPlayerFinishEvent(_Player, C_Default_PlayerFinishEvent);
}
