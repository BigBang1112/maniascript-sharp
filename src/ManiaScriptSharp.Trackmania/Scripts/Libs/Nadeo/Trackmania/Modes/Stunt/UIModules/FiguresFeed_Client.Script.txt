/// Display the new figure made by the player

/// Update the figure display
Void UpdateFigureDisplay(K_State _State, PlayerData::K_FigureEvent _FigureEvent) {
	if (GameTime > _FigureEvent.Timestamp + Const::C_AnimDuration_NewFigure + Const::C_AnimDuration_FigureInFeed) return;
	declare Boolean IsEventDisplayed;

	foreach (Index => Control in _State.Controls.Frame_FiguresFeed.Controls) {
		declare CMlFrame Frame_Figure = Control as CMlFrame;

		AnimMgr.Flush(Frame_Figure);
		if (!IsEventDisplayed && !Frame_Figure.Visible) {
			AnimMgr.Add(Frame_Figure, "<anim pos=\"0 0\" />", Now + 500, 500, CAnimManager::EAnimManagerEasing::QuadOut);
			foreach (Control in Frame_Figure.Controls) {
				AnimMgr.Flush(Control);
				AnimMgr.Add(Control, "<anim opacity=\""^Tools::Ternary(Control.ControlId == "quad-background", 0.6, 1.)^"\" />", Now + 500, 500, CAnimManager::EAnimManagerEasing::QuadOut);
				AnimMgr.Add(Control, "<anim opacity=\"0.\" />", Now + Const::C_AnimDuration_FigureInFeed, 2000, CAnimManager::EAnimManagerEasing::QuadOut);
			}
			DisplayFigure(Frame_Figure, _FigureEvent);
			IsEventDisplayed = True;
		} else if (Frame_Figure.Visible) {
			declare Real NewPosY = Frame_Figure.RelativePosition_V3.Y - {{{SizeY_Figure}}} - {{{Spacing}}};
			AnimMgr.Add(Frame_Figure, "<anim pos=\"0 "^NewPosY^"\" />", 500, CAnimManager::EAnimManagerEasing::QuadOut);
			if (NewPosY < {{{-SizeY_Figures}}}) {
				AnimMgr.AddChain(Frame_Figure, "<anim pos=\"2 0\" hidden=\"1\" />", 0, CAnimManager::EAnimManagerEasing::QuadOut);
			}
		}
	}
}

/// Reset the module
K_State ResetModule(K_State _State) {
	declare K_State State = _State;

	State.DelayedFigureEvent = PlayerData::C_Default_PlayerFigureEvent;
	foreach (Control in _State.Controls.Frame_FiguresFeed.Controls) {
		Control.RelativePosition_V3 = <2., 0.>;
		AnimMgr.Flush(Control);
		Control.Hide();
	}

	return State;
}

/// Trigger a figure event
K_State TriggerFigureEvent(K_State _State, CSmPlayer _Player) {
	if (_Player == Null) return _State;
	declare K_State State = _State;

	State.FigureEventUpdate = PlayerData::GetPlayerFigureEventUpdate(_Player);
	if (PlayerData::GetPlayerFigureEvent(_Player).Name != "") {
		UpdateFigureDisplay(State, State.DelayedFigureEvent);
		State.DelayedFigureEvent = PlayerData::GetPlayerFigureEvent(_Player);
	} else {
		State = ResetModule(State);
	}

	return State;
}

/// Update the visibility of the whole manialink UI
K_State UpdateVisibility(K_State _State, Boolean _IsPageVisible, Boolean _IsModuleDisplayed) {
	declare K_State State = _State;

	State.IsPageVisible = _IsPageVisible;
	State.IsModuleDisplayed = _IsModuleDisplayed;
	State.IsVisible = State.IsPageVisible && State.IsModuleDisplayed && State.TargetId != NullId;
	State.Controls.Frame_Global.Visible = State.IsVisible;

	// Reset module if hidden
	if (!State.Controls.Frame_Global.Visible) {
		State = ResetModule(State);
	}

	return State;
}

/// Check if the current target has changed
Boolean IsTargetUpdateRequired(K_State _State, CSmPlayer _Player) {
	return (
		(_Player != Null && _State.TargetId != _Player.Id) ||
		(_Player == Null && _State.TargetId != NullId)
	);
}

/// Set the player currently followed by the camera
K_State SetTarget(K_State _State, CSmPlayer _Player) {
	declare K_State State = _State;

	// Update target data
	if (_Player == Null) {
		State.TargetId = NullId;
	} else {
		State.FigureEventUpdate = -1;
		State.TargetId = _Player.Id;
	}

	// Refresh UI
	State = UpdateVisibility(State, State.IsPageVisible, State.IsModuleDisplayed);
	State = ResetModule(State);

	return State;
}
