/// Update the medals displayed
Void DisplayMedal(K_State _State, Integer _MedalNumber) {
	if (_MedalNumber == Medals::C_ScoreMgrMedal_None) return;
	switch (_MedalNumber) {
		case Medals::C_ScoreMgrMedal_Bronze: {
			AnimMgr.Flush(_State.Controls.Quad_MedalBronze);
			AnimMgr.Add(_State.Controls.Quad_MedalBronze, "<anim pos=\"-5 0\" hidden=\"0\" />", 500, CAnimManager::EAnimManagerEasing::BackOut);
		}
		case Medals::C_ScoreMgrMedal_Silver: {
			AnimMgr.Flush(_State.Controls.Quad_MedalSilver);
			AnimMgr.Add(_State.Controls.Quad_MedalSilver, "<anim pos=\"-9 0\" hidden=\"0\" />", 500, CAnimManager::EAnimManagerEasing::BackOut);
		}
		case Medals::C_ScoreMgrMedal_Gold: {
			AnimMgr.Flush(_State.Controls.Quad_MedalGold);
			AnimMgr.Add(_State.Controls.Quad_MedalGold, "<anim pos=\"-13 0\" hidden=\"0\" />", 500, CAnimManager::EAnimManagerEasing::BackOut);
		}
		case Medals::C_ScoreMgrMedal_Author: {
			AnimMgr.Flush(_State.Controls.Quad_MedalAuthor);
			AnimMgr.Add(_State.Controls.Quad_MedalAuthor, "<anim pos=\"-17 0\" hidden=\"0\" />", 500, CAnimManager::EAnimManagerEasing::BackOut);
		}
	}

	if (!IsSpectatorClient) {
		AudioManager::StopSound(This, "RankUp");
		AudioManager::PlaySound(This, "RankUp");
	}
}
Void HideMedals(K_State _State) {
	foreach (Quad_Medal in _State.Controls.Frame_MedalsImages.Controls) {
		AnimMgr.Flush(Quad_Medal);
		Quad_Medal.RelativePosition_V3.X = 0.;
		Quad_Medal.Hide();
	}
}

/// Update the score if the players beat their PB
K_State DisplayPBMessage(K_State _State) {
	declare K_State State = _State;

	State.Controls.Label_TotalScore.Value = C_Text_NewRecord;
	Tools::FitLabelValue(State.Controls.Label_TotalScore, 2.5, 0.5, 0.25);
	AnimMgr.Flush(State.Controls.Frame_TotalScore);
	AnimMgr.Add(State.Controls.Frame_TotalScore, "<anim scale=\"1.1\" />", 500, CAnimManager::EAnimManagerEasing::QuadOut);
	AnimMgr.Add(State.Controls.Frame_TotalScore, "<anim scale=\"1.\" />", Now + 500, 100, CAnimManager::EAnimManagerEasing::QuadOut);

	if (!IsSpectatorClient) {
		AudioManager::StopSound(This, "SquadInvitation");
		AudioManager::PlaySound(This, "SquadInvitation");
	}

	return State;
}

/// Update the data of a given park
Void UpdateParkPoints(K_State _State, CMlFrame _Frame_Park, PlayerData::K_ParkData _ParkData, Boolean _IsParksInitialisation, Boolean _NewCurrentPark) {
	declare CMlFrame Frame_PointsBg <=> (_Frame_Park.GetFirstChild("frame-points-bg") as CMlFrame);
	declare CMlLabel Label_Points <=> (_Frame_Park.GetFirstChild("label-points") as CMlLabel);

	// Update colors and positions
	if (_IsParksInitialisation) {
		if (_ParkData.Id == Const::C_ParkId_BestStunt) {
			_Frame_Park.RelativePosition_V3.Y = -_State.TotalParkNb * {{{Size_Park.Y}}} - 1.;
			{{{ExpendableAsset::P}}}SetColor(This, Frame_PointsBg, CL::Hex6ToRgb(C_ColorBg_BestStunt), False);
			{{{ExpendableAsset::P}}}RedrawAsset(This, Frame_PointsBg, False);
			Label_Points.TextColor = <1., 1., 1.>;
		} else {
			_Frame_Park.RelativePosition_V3.Y = -(_State.Controls.Frame_Parks.Controls.keyof(_Frame_Park) - 1) * {{{Size_Park.Y}}};
			Label_Points.RelativePosition_V3.X = {{{-Size_Park.X * 0.9}}};
			Label_Points.TextColor = C_ColorText_Parks.get(_ParkData.Id, <1., 1., 1.>);
			{{{ExpendableAsset::P}}}SetColor(This, Frame_PointsBg, CL::Hex6ToRgb(C_ColorBg_Parks.get(_ParkData.Id, "FFFFFF")), False);
			{{{ExpendableAsset::P}}}SetStyles(This, Frame_PointsBg, ["{{{C_Style_ColorizeLeft}}}"], False);
			{{{ExpendableAsset::P}}}SetWidth(This, Frame_PointsBg, {{{Size_Park.X}}}, False);
			{{{ExpendableAsset::P}}}SetHeight(This, Frame_PointsBg, {{{Size_Park.Y}}}, False);
			{{{ExpendableAsset::P}}}RedrawAsset(This, Frame_PointsBg, False);
		}
	}

	// Update score
	if (_ParkData.Id == Const::C_ParkId_BestStunt && _ParkData.Score <= 0) {
		Label_Points.SetText("{{{Const::C_Text_BestStunt}}}");
		Label_Points.Opacity = 1.;
	} else {
		declare Text NewScore = Tools::Ternary(_ParkData.Id == Const::C_ParkId_BestStunt, "+ "^TL::FormatInteger(_ParkData.Score, 1, True), TL::FormatInteger(_ParkData.Score, 1, True));
		if (Label_Points.Value != NewScore && _ParkData.Score > 0) {
			AnimMgr.Add(Label_Points, "<anim scale=\"1.1\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			AnimMgr.Add(Label_Points, "<anim scale=\"1.\" />", Now + 250, 100, CAnimManager::EAnimManagerEasing::QuadOut);
		}
		Label_Points.SetText(NewScore);
		Label_Points.Opacity = Tools::Ternary(_ParkData.Score > 0, 1., 0.4);
	}

	// Current park animation
	if (_NewCurrentPark) {
		if (_ParkData.Id == _State.CurrentParkId) {
			_Frame_Park.ZIndex = 2.;
			if (Label_Points.Scale > 1.) {
				Label_Points.Scale = 1.;
				AnimMgr.Flush(Label_Points);
			}
			AnimMgr.Add(Label_Points, "<anim pos=\""^{{{-Size_Park.X * 1.1}}}^" "^Label_Points.RelativePosition_V3.Y^"\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			{{{ExpendableAsset::P}}}AddStyle(This, Frame_PointsBg, "{{{ExpendableAsset::C_Style_LeftEdgeSoloRounded}}}", False);
			{{{ExpendableAsset::P}}}SetWidth(This, Frame_PointsBg, {{{Size_Park.X * 1.2}}}, False);
			{{{ExpendableAsset::P}}}SetHeight(This, Frame_PointsBg, {{{Size_Park.Y * 1.2}}}, False);
			{{{ExpendableAsset::P}}}RedrawAsset(This, Frame_PointsBg, True);
		} else {
			_Frame_Park.ZIndex = 1.;
			if (Label_Points.Scale > 1.) {
				Label_Points.Scale = 1.;
				AnimMgr.Flush(Label_Points);
			}
			AnimMgr.Add(Label_Points, "<anim pos=\""^{{{-Size_Park.X * 0.9}}}^" "^Label_Points.RelativePosition_V3.Y^"\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			{{{ExpendableAsset::P}}}RemoveStyle(This, Frame_PointsBg, "{{{ExpendableAsset::C_Style_LeftEdgeSoloRounded}}}", False);
			{{{ExpendableAsset::P}}}SetWidth(This, Frame_PointsBg, {{{Size_Park.X}}}, False);
			{{{ExpendableAsset::P}}}SetHeight(This, Frame_PointsBg, {{{Size_Park.Y}}}, False);
			{{{ExpendableAsset::P}}}RedrawAsset(This, Frame_PointsBg, True);
		}
	}

	// Fit label
	if (_ParkData.Id == _State.CurrentParkId) {
		Tools::FitLabelValue(Label_Points, 3., 1., 0.25);
	} else {
		Tools::FitLabelValue(Label_Points, 2., 1., 0.25);
	}
}

/// Update all parks data
K_State UpdateAllParksPoints(K_State _State, PlayerData::K_ParkEvent _ParkEvent, Boolean _NewCurrentPark) {
	declare K_State State = _State;

	// Update the park numbers for this map
	if (State.TotalParkNb != _ParkEvent.Parks.count-1) {
		State.TotalParkNb = _ParkEvent.Parks.count-1;
		State.Controls.Frame_ClipParks.RelativePosition_V3.Y = (State.TotalParkNb + 1) * {{{Size_Park.Y}}} + 8.;
	}

	// Update all parks display
	foreach (Key => Control in State.Controls.Frame_Parks.Controls) {
		Control.Visible = _ParkEvent.Parks.existskey(Key);
		if (_ParkEvent.Parks.existskey(Key)) {
			UpdateParkPoints(State, Control as CMlFrame, _ParkEvent.Parks[Key], _ParkEvent.IsParksInitialisation, _NewCurrentPark);
		}
	}

	// Update helpers text
	if (_ParkEvent.IsParksInitialisation) {
		foreach (Key => Control in State.Controls.Frame_ParksHelpers.Controls) {
			Control.Visible = _ParkEvent.Parks.existskey(Key);
			if (Control.Visible) {
				if (_ParkEvent.Parks[Key].Id == Const::C_ParkId_BestStunt) {
					(Control as CMlLabel).SetText("{{{Const::C_Text_BestStunt}}}");
				} else {
					(Control as CMlLabel).SetText(C_ColorName_Parks.get(_ParkEvent.Parks[Key].Id, ""));
				}
			}
		}
	}

	return State;
}

/// Trigger a park event
K_State TriggerParkEvent(K_State _State, CSmPlayer _Player) {
	if (_Player == Null) return _State;
	declare K_State State = _State;

	State.ParkEventUpdate = PlayerData::GetPlayerParkEventUpdate(_Player);
	State = UpdateAllParksPoints(State, PlayerData::GetPlayerParkEvent(_Player), False);

	return State;
}

/// Trigger a combo event
K_State TriggerParkComboEvent(K_State _State, CSmPlayer _Player) {
	if (_Player == Null) return _State;
	declare K_State State = _State;

	State.ParkComboEventUpdate = PlayerData::GetPlayerParkComboEventUpdate(_Player);
	State.CurrentParkId = PlayerData::GetPlayerParkComboEvent(_Player).ParkId;
	declare Boolean NewCurrentPark = False;
	if (State.PreviousParkId != State.CurrentParkId) {
		NewCurrentPark = True;
		State.PreviousParkId = State.CurrentParkId;
	}
	declare PlayerData::K_ParkEvent PlayerParkEvent = PlayerData::GetPlayerParkEvent(_Player);
	PlayerParkEvent.IsParksInitialisation = False;
	State = UpdateAllParksPoints(State, PlayerParkEvent, NewCurrentPark);

	return State;
}

/// Trigger a total score event
K_State TriggerTotalScoreEvent(K_State _State, CSmPlayer _Player) {
	if (_Player == Null) return _State;
	declare K_State State = _State;
	declare Boolean IsSpectatingAnotherTarget = IsSpectatorClient && State.TotalScoreEventUpdate < 0; //< Faster animations when switching target as a spectator

	// Trigger score animation
	State.ScoreAnim = UIModulesData::K_ScoreAnim {
		CurrentScore = Tools::Ternary(State.ScoreAnim.CurrentScore <= PlayerData::GetPlayerTotalScore(_Player), State.ScoreAnim.CurrentScore, 0),
		OldScore = Tools::Ternary(State.ScoreAnim.CurrentScore <= PlayerData::GetPlayerTotalScore(_Player), State.ScoreAnim.CurrentScore, 0),
		NewScore = PlayerData::GetPlayerTotalScore(_Player),
		Duration = Tools::Ternary(IsSpectatingAnotherTarget, Const::C_AnimDuration_FasterScoreIncrease, Const::C_AnimDuration_ScoreIncrease),
		StartTime = Now
	};

	// Update total score event update value
	State.TotalScoreEventUpdate = PlayerData::GetPlayerTotalScoreEventUpdate(_Player);

	// Reset total score
	if (State.ScoreAnim.NewScore <= 0 || IsSpectatingAnotherTarget) {
		State.Controls.Label_TotalScore.SetText(C_Text_TotalScore);
		Tools::FitLabelValue(State.Controls.Label_TotalScore, 3., 1., 0.25);
		HideMedals(State);
	} else if (State.Controls.Label_TotalScore.TextSizeReal < 3.) {
		State.Controls.Label_TotalScore.TextSizeReal = 3.;
	}

	return State;
}

	// Parent frame
	AnimMgr.Flush(State.Controls.Frame_Helpers);
	AnimMgr.Add(State.Controls.Frame_Helpers, "<anim hidden=\""^Tools::Ternary(_IsVisible, 0, 1)^"\" />", Tools::Ternary(_IsVisible, 0, 500), CAnimManager::EAnimManagerEasing::QuadOut);

	return State;
}

/// Update the visibility of the whole manialink UI
K_State UpdateVisibility(K_State _State, Boolean _PageIsVisible, Boolean _DisplayModule) {
	declare K_State State = _State;

	State.PageIsVisible = _PageIsVisible;
	State.DisplayModule = _DisplayModule;
	State.IsVisible = State.PageIsVisible && State.DisplayModule && State.TargetId != NullId;
	State.Controls.Frame_Global.Visible = State.IsVisible;

	// Reset module if hidden
	if (!State.Controls.Frame_Global.Visible) {
		State.CurrentParkId = Const::C_ParkId_Null;
		State.ParkEventUpdate = -1;
		State.ParkComboEventUpdate = -1;
		State.TotalScoreEventUpdate = -1;
		State.ScoreAnim = UIModulesData::K_ScoreAnim {};
		State.Controls.Label_TotalScore.SetText(C_Text_TotalScore);
		Tools::FitLabelValue(State.Controls.Label_TotalScore, 3., 1., 0.25);
		State = UpdateAllParksPoints(State, PlayerData::C_Default_PlayerParkEvent, True);
		HideMedals(State);
	}

	// Clip parks animation
	AnimMgr.Flush(State.Controls.Frame_ClipParks);
	if (State.Controls.Frame_Global.Visible) {
		AnimMgr.Add(State.Controls.Frame_ClipParks, "<anim size=\""^{{{Size_ClipParks.X}}}^" "^{{{Size_ClipParks.Y}}}^"\" hidden=\"0\" />", Now + 250, 500, CAnimManager::EAnimManagerEasing::QuadOut);
	} else {
		State.Controls.Frame_ClipParks.Size.X = 0.;
		State.Controls.Frame_ClipParks.Hide();
	}

	return State;
}

/// Check if the current target has changed
Boolean IsTargetUpdateRequired(K_State _State, CSmPlayer _Player) {
	return (
		(_Player != Null && _State.TargetId != _Player.Id) ||
		(_Player == Null && _State.TargetId != NullId)
	);
}

/// Set the player currently followed by the camera
K_State SetTarget(K_State _State, CSmPlayer _Player) {
	declare K_State State = _State;

	// Update target data
	if (_Player == Null) {
		State.TargetId = NullId;
	} else {
		State.ParkEventUpdate = -1;
		State.ParkComboEventUpdate = -1;
		State.TotalScoreEventUpdate = -1;
		State.ScoreAnim = UIModulesData::K_ScoreAnim {};
		State.TargetId = _Player.Id;
	}

	// Refresh UI
	State = UpdateVisibility(State, State.PageIsVisible, State.DisplayModule);
	State = UpdateAllParksPoints(State, PlayerData::K_ParkEvent {
		Parks = PlayerData::GetPlayerParkEvent(_Player).Parks,
		IsParksInitialisation = True
	}, False);

	return State;
}
