/// Update the visibility of the whole manialink UI
K_State UpdateVisibility(K_State _State, Boolean _IsPageVisible, Boolean _IsModuleDisplayed) {
	declare K_State State = _State;
	declare persistent Boolean Persistent_Stunt_HasHiddenWelcomeStuntScreen for UserMgr.MainUser;

	State.IsPageVisible = _IsPageVisible;
	State.IsModuleDisplayed = _IsModuleDisplayed;
	State.IsVisible = State.IsPageVisible && State.IsModuleDisplayed && !Persistent_Stunt_HasHiddenWelcomeStuntScreen;
	State.TriggerEnableInputsDelay = Tools::Ternary(State.IsVisible, Now + {{{Const::C_AnimDuration_UnskippableWelcomeStunt}}}, 0);

	// Inform the mode of the player choice
	if (InputPlayer != Null && InputPlayer.User != Null) {
		if (Persistent_Stunt_HasHiddenWelcomeStuntScreen) {
			SendCustomEvent("{{{UIModules_WelcomeStunt_Common::C_Event_PlayStunt}}}", [InputPlayer.User.WebServicesUserId]);
		}
	}

	// Update navigation
	State.IsMouse = !Tools::IsUsingMouse(Input);
	EnableMenuNavigation(State.IsVisible, False, False, Null, 0);
	{{{Navigation::P}}}UseLocalEvents(This, State.IsVisible);

	// Update landing multipliers
	foreach (LandindType => LandingColor in Const::C_Colors_FiguresLanding) {
		declare CMlLabel Label_NextStunt = State.Controls.Frame_Global.GetFirstChild("label-next-stunt-"^LandindType) as CMlLabel;
		Label_NextStunt.SetText(TL::Compose("{{{C_Text_ScoreMultiplier}}}", "$"^CL::RgbToHex3(CL::Hex6ToRgb(LandingColor))^"x"^TL::FormatReal(RaceData::GetLandingMultipliers(Teams[0]).get(LandindType, 1.), 1, False, False)));
	}

	// Update controls visibility
	AnimMgr.Flush(State.Controls.Frame_Global);
	AnimMgr.Add(State.Controls.Frame_Global, "<anim hidden=\""^Tools::Ternary(State.IsVisible, 0, 1)^"\" />", Tools::Ternary(State.IsVisible, 0, 250), CAnimManager::EAnimManagerEasing::QuadOut);
	foreach (Control in [State.Controls.Button_PlayStunt, State.Controls.Checkbox_NeverShowAgain, State.Controls.MenuButtonList_WelcomeStunt]) {
		Control.Hide();
		AnimMgr.Flush(Control);
		if (State.IsVisible) {
			AnimMgr.Add(Control, "<anim hidden=\"0\" />", State.TriggerEnableInputsDelay, 0, CAnimManager::EAnimManagerEasing::QuadOut);
		}
	}
	foreach (Class in ["{{{C_Class_QuadCentered}}}", "{{{C_Class_TextExtraBold}}}", "{{{C_Class_TextSemiBold}}}"]) {
		Page.GetClassChildren(Class, State.Controls.Frame_Global, True);
		foreach (Control in Page.GetClassChildren_Result) {
			AnimMgr.Flush(Control);
			if (Control.ControlId == "quad-dimmer") {
				AnimMgr.Add(Control, "<anim opacity=\""^Tools::Ternary(State.IsVisible, 0.7, 0.)^"\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			} else {
				AnimMgr.Add(Control, "<anim opacity=\""^Tools::Ternary(State.IsVisible, 1., 0.)^"\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			}
		}
	}

	return State;
}

/// Handle navigation focus
Void Focus(CMlControl _From, CMlControl _To) {
	if ({{{Transition::P}}}IsInTransition(This)) return;

	if (_From != Null) {
		{{{Navigation::P}}}Unfocus(This, _From);
		if ({{{ExpendableButton::P}}}IsButton(_From)) {
			{{{ExpendableButton::P}}}Unfocus(This, _From as CMlFrame);
		} else if ({{{Checkbox::P}}}IsCheckbox(_From)) {
			{{{Checkbox::P}}}Unfocus(This, _From as CMlFrame);
		}
	}

	if (_To != Null) {
		{{{Navigation::P}}}Focus(This, _To);
		if ({{{ExpendableButton::P}}}IsButton(_To)) {
			{{{ExpendableButton::P}}}Focus(This, _To as CMlFrame);
		} else if ({{{Checkbox::P}}}IsCheckbox(_To)) {
			{{{Checkbox::P}}}Focus(This, _To as CMlFrame);
		}
	}
}

/// Handle navigation selection
Void Select(CMlControl _Control) {
	if (_Control == Null) return;

	switch (_Control.ControlId) {
		case "button-play-stunt": {
			EnableMenuNavigation(False, False, False, Null, 0);
			SendCustomEvent("{{{UIModules_WelcomeStunt_Common::C_Event_PlayStunt}}}", [InputPlayer.User.WebServicesUserId]);
		}
		case "checkbox-never-show-again": {
			declare persistent Boolean Persistent_Stunt_HasHiddenWelcomeStuntScreen for UserMgr.MainUser;
			Persistent_Stunt_HasHiddenWelcomeStuntScreen = {{{Checkbox::P}}}Toggle(This, _Control as CMlFrame);
			UserMgr.MainUser.PersistentSave();
		}
	}
}

/// Update the library
Void Yield() {
	foreach (Event in PendingEvents) {
		switch (Event.PlaygroundType) {
			case CManiaAppPlaygroundEvent::Type::LayerCustomEvent: {
				switch (Event.CustomEventType) {
					case UIModules_WelcomeStunt_Common::C_Event_PlayStunt: {
						assert(Event.CustomEventData.count == 1, "'C_Event_PlayStunt' needs 1 parameter: 'WebServicesUserId'.");
						SendCustomEvent(UIModules_WelcomeStunt_Common::C_Event_PlayStunt, [Event.CustomEventData[0]]);
					}
				}
			}
		}
	}
