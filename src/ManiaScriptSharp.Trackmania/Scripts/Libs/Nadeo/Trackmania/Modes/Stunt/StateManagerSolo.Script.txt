// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/**
 * Enabling this setting will have a big impact on the script performance on crowded servers.
 * The library will check every player on the server on every frame.
 * If the script uses 'MB_Yield()' instead of 'yield', it shouldn't have to enforce the state at every frame.
 * Each new player will be initialized when they join the server.
 */
#Const C_ForceStatesEveryFrame False

#Const C_State_Waiting 0
#Const C_State_WelcomeStunt 1
#Const C_State_StartRaceMenu 2
#Const C_State_Playing 3
#Const C_State_StuntsResult 4
#Const C_State_EndRaceMenu 5

#Const C_DefaultStates []

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Globales
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
declare Integer[] G_ForcedStates;

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Apply the given states to the given player
Void Private_Apply(CPlayer _Player, Integer[] _States) {
	if (_Player == Null) return;

	UIModules_EndRaceMenu::SetVisibility(_Player, False);
	UIModules_FiguresFeed::SetVisibility(_Player, False);
	UIModules_NewFigure::SetVisibility(_Player, False);
	UIModules_ParkCombo::SetVisibility(_Player, False);
	UIModules_ParksPoints::SetVisibility(_Player, False);
	UIModules_PauseMenu::SetVisibility(_Player, False);
	UIModules_ScorestableHelper::SetVisibility(_Player, False);
	UIModules_StartRaceMenu::SetVisibility(_Player, False);
	UIModules_StuntsResult::SetVisibility(_Player, False);
	UIModules_WelcomeStunt::SetVisibility(_Player, False);

	foreach (State in _States) {
		switch (State) {
			case C_State_WelcomeStunt: {
				UIModules_WelcomeStunt::SetVisibility(_Player, True);
			}
			case C_State_StartRaceMenu: {
				UIModules_StartRaceMenu::SetVisibility(_Player, True);
			}
			case C_State_Playing: {
				UIModules_FiguresFeed::SetVisibility(_Player, True);
				UIModules_NewFigure::SetVisibility(_Player, True);
				UIModules_ParkCombo::SetVisibility(_Player, True);
				UIModules_ParksPoints::SetVisibility(_Player, True);
				UIModules_PauseMenu::SetVisibility(_Player, True);
				UIModules_ScorestableHelper::SetVisibility(_Player, True);
			}
			case C_State_StuntsResult: {
				UIModules_StuntsResult::SetVisibility(_Player, True);
				UIModules_PauseMenu::SetVisibility(_Player, True);
			}
			case C_State_EndRaceMenu: {
				UIModules_EndRaceMenu::SetVisibility(_Player, True);
			}
			default: {
				UIModules_PauseMenu::SetVisibility(_Player, True);
			}
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Add a state to a player
Void Add(CPlayer _Player, Integer _State) {
	if (_Player == Null) return;
	declare Integer[] LibStateMgr_States for _Player = C_DefaultStates;
	if (!LibStateMgr_States.exists(_State)) {
		LibStateMgr_States.add(_State);
		if (G_ForcedStates.count <= 0) {
			Private_Apply(_Player, LibStateMgr_States);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Remove a state from a player
Void Remove(CPlayer _Player, Integer _State) {
	if (_Player == Null) return;
	declare Integer[] LibStateMgr_States for _Player = C_DefaultStates;
	if (LibStateMgr_States.exists(_State)) {
		LibStateMgr_States.remove(_State);
		if (G_ForcedStates.count <= 0) {
			Private_Apply(_Player, LibStateMgr_States);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the states of a player
Integer[] Get(CPlayer _Player) {
	if (_Player == Null) return [];
	declare Integer[] LibStateMgr_States for _Player = C_DefaultStates;
	return LibStateMgr_States;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the given state to a player
Void Set(CPlayer _Player, Integer _State) {
	if (_Player == Null) return;
	declare Integer[] LibStateMgr_States for _Player = C_DefaultStates;
	LibStateMgr_States = [_State];
	if (G_ForcedStates.count <= 0) {
		Private_Apply(_Player, LibStateMgr_States);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Remove all state from a player
Void Reset(CPlayer _Player) {
	if (_Player == Null) return;
	declare Integer[] LibStateMgr_States for _Player = C_DefaultStates;
	LibStateMgr_States = C_DefaultStates;
	if (G_ForcedStates.count <= 0) {
		Private_Apply(_Player, LibStateMgr_States);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the player has the given state
Boolean HasState(CPlayer _Player, Integer _State) {
	if (_Player == Null) return False;
	declare Integer[] LibStateMgr_States for _Player = C_DefaultStates;
	return LibStateMgr_States.exists(_State) || G_ForcedStates.exists(_State);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the players have the given state as forced
Boolean HasForcedState(Integer _State) {
	return G_ForcedStates.exists(_State);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Force the states applied to all players
/// To disable the forced states use an empty array
Void ForcePlayersStates(Integer[] _States) {
	G_ForcedStates = _States;

	foreach (Player in AllPlayers) {
		declare Boolean LibStateMgr_IsForcedStates for Player = False;
		// Enable
		if (G_ForcedStates.count > 0) {
			LibStateMgr_IsForcedStates = True;
			Private_Apply(Player, G_ForcedStates);
		}
		// Disable
		else {
			LibStateMgr_IsForcedStates = False;
			declare Integer[] LibStateMgr_States for Player = C_DefaultStates;
			Private_Apply(Player, LibStateMgr_States);
		}
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the library
Void Yield() {
	if (C_ForceStatesEveryFrame && G_ForcedStates.count > 0) {
		foreach (Player in AllPlayers) {
			declare Boolean LibStateMgr_IsForcedStates for Player = False;
			if (!LibStateMgr_IsForcedStates) {
				LibStateMgr_IsForcedStates = True;
				Private_Apply(Player, G_ForcedStates);
			}
		}
	}

	UIModules_EndRaceMenu::Yield();
	UIModules_PauseMenu::Yield();
	UIModules_StartRaceMenu::Yield();
	UIModules_StuntsResult::Yield();
	UIModules_WelcomeStunt::Yield();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize a player
Void InitializePlayer(CPlayer _Player) {
	if (_Player == Null) return;

	UIModules_EndRaceMenu::InitializePlayer(_Player);
	UIModules_FiguresFeed::InitializePlayer(_Player);
	UIModules_NewFigure::InitializePlayer(_Player);
	UIModules_ParkCombo::InitializePlayer(_Player);
	UIModules_ParksPoints::InitializePlayer(_Player);
	UIModules_PauseMenu::InitializePlayer(_Player);
	UIModules_ScorestableHelper::InitializePlayer(_Player);
	UIModules_StartRaceMenu::InitializePlayer(_Player);
	UIModules_StuntsResult::InitializePlayer(_Player);
	UIModules_WelcomeStunt::InitializePlayer(_Player);

	Reset(_Player);

	if (G_ForcedStates.count > 0) {
		declare Boolean LibStateMgr_IsForcedStates for _Player = False;
		LibStateMgr_IsForcedStates = True;
		Private_Apply(_Player, G_ForcedStates);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Unload the library
Void Unload() {
	ForcePlayersStates([]);

	foreach (Player in AllPlayers) {
		InitializePlayer(Player);
	}

	UIModules_EndRaceMenu::Unload();
	UIModules_FiguresFeed::Unload();
	UIModules_NewFigure::Unload();
	UIModules_ParkCombo::Unload();
	UIModules_ParksPoints::Unload();
	UIModules_PauseMenu::Unload();
	UIModules_ScorestableHelper::Unload();
	UIModules_StartRaceMenu::Unload();
	UIModules_StuntsResult::Unload();
	UIModules_WelcomeStunt::Unload();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Load the library
Void Load() {
	Unload();

	UIModules_EndRaceMenu::Load();
	UIModules_FiguresFeed::Load();
	UIModules_NewFigure::Load();
	UIModules_ParkCombo::Load();
	UIModules_ParksPoints::Load();
	UIModules_PauseMenu::Load();
	UIModules_ScorestableHelper::Load();
	UIModules_StartRaceMenu::Load();
	UIModules_StuntsResult::Load();
	UIModules_WelcomeStunt::Load();

	UIModules::InitializeCustomizableModule(UIModules_FiguresFeed::GetConfig());
	UIModules::InitializeCustomizableModule(UIModules_NewFigure::GetConfig());
	UIModules::InitializeCustomizableModule(UIModules_ParkCombo::GetConfig());
	UIModules::InitializeCustomizableModule(UIModules_ParksPoints::GetConfig());
	UIModules::InitializeCustomizableModule(UIModules_StuntsResult::GetConfig());
	UIModules::InitializeCustomizableModule(UIModules_WelcomeStunt::GetConfig());

	UIModules::LoadModules([
		UIModules_EndRaceMenu::C_Id,
		UIModules_FiguresFeed::GetId(),
		UIModules_NewFigure::GetId(),
		UIModules_ParkCombo::GetId(),
		UIModules_ParksPoints::GetId(),
		UIModules_PauseMenu::C_Id,
		UIModules_ScorestableHelper::C_Id,
		UIModules_StartRaceMenu::C_Id,
		UIModules_StuntsResult::GetId(),
		UIModules_WelcomeStunt::GetId()
	]);

	// Unload the spectator module to avoid a conflict with the replay in the start and end race menu
	// Otherwise the camera won't follow the ghost and will switch to camera type `Follow` and target `AllPlayers`
	UIModules::UnloadModules([
		UIModules_Chrono_Default::GetId(),
		UIModules_LapsCounter_Default::GetId(),
		UIModules_PauseMenu_Default::C_Id,
		UIModules_SpectatorBase_Default::GetId()
	]);
}
