/**
 * UI module: WelcomeStunt server side
 */
#Const Version "1.0.0"
#Const ScriptName "Libs/Nadeo/Trackmania/Modes/Stunt/UIModules/WelcomeStunt_Server.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Includes
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Modes/UIModules_Common.Script.txt" as UIModules_Common
#Include "Libs/Nadeo/Trackmania/Modes/Stunt/UIModules/WelcomeStunt_Common.Script.txt" as UIModules_WelcomeStunt_Common

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_Event {
	Text Type;
	Text AccountId;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const UIModules_WelcomeStunt_Common::C_Event_PlayStunt as C_Event_PlayStunt

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Globales
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
declare K_Event[] G_PendingEvents;

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the id of the UI module
Text GetId() {
	return UIModules_WelcomeStunt_Common::C_Id;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the configuration of the UI module
UIModules_Common::K_ModuleConfig GetConfig() {
	return UIModules_WelcomeStunt_Common::C_UIModuleConfig;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get last pending events received by the server
K_Event[] GetPendingEvents() {
	return G_PendingEvents;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/**
 * Show or hide the UI
 *
 * @param _Player       The player to update
 * @param _IsVisible    True to show, False otherwise
 */
Void SetVisibility(CPlayer _Player, Boolean _IsVisible) {
	if (_Player == Null) return;
	declare CUIConfig UI <=> UIManager.GetUI(_Player);
	if (UI == Null) return;

	declare netwrite Boolean Net_Stunt_WelcomeStunt_IsVisible for UI;
	Net_Stunt_WelcomeStunt_IsVisible = _IsVisible;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize a player
Void InitializePlayer(CPlayer _Player) {
	if (_Player == Null) return;
	SetVisibility(_Player, False);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Unload the library
Void Unload() {
	foreach (Player in AllPlayers) {
		InitializePlayer(Player);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Load the library
Void Load() {
	Unload();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the library
Void Yield() {
	G_PendingEvents = [];
	foreach (Event in UIManager.PendingEvents) {
		switch (Event.Type) {
			case CUIConfigEvent::EType::OnLayerCustomEvent: {
				switch (Event.CustomEventType) {
					case C_Event_PlayStunt: {
						G_PendingEvents.add(K_Event {
							Type = Event.CustomEventType,
							AccountId = Event.CustomEventData[0]
						});
					}
				}
			}
		}
	}
}
