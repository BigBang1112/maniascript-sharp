	State.IsPageVisible = _IsPageVisible;
	State.IsModuleDisplayed = _IsModuleDisplayed;

/// Update the scores and stats of the player
Void UpdateScoresAndStats(K_State _State, PlayerData::K_FinishEvent _FinishEvent) {
	if (!_State.IsVisible && _State.Controls.Frame_Global.Visible) return;

	_State.Controls.Frame_StuntResults.Visible = _FinishEvent.Parks.count > 0;
	_State.Controls.Label_NoResults.Visible = !_State.Controls.Frame_StuntResults.Visible;
	if (_State.Controls.Frame_StuntResults.Visible) {
		// Update parks and best stunt score
		_State.Controls.Label_BestFigureScore.SetText("+"^TL::FormatInteger(_FinishEvent.Parks.get(Const::C_ParkId_BestStunt, PlayerData::K_ParkData {}).Score, 1, True));
		_State.Controls.Label_BestFigure.SetText(_FinishEvent.Stats.BestFigure);
		Tools::FitLabelValue(_State.Controls.Label_BestFigure, 3., 2., 0.1);
		foreach (Index => Control in _State.Controls.Frame_ParksScores.Controls) {
			declare CMlFrame Frame_ParkScore = Control as CMlFrame;
			Frame_ParkScore.Visible = _FinishEvent.Parks.existskey(Index+1);
			if (Frame_ParkScore.Visible) {
				declare CMlFrame Asset_Background = Frame_ParkScore.GetFirstChild("asset-background") as CMlFrame;
				{{{ExpendableAsset::P}}}SetColor(This, Asset_Background, CL::Hex6ToRgb(C_ColorBg_Parks.get(_FinishEvent.Parks[Index+1].Id, "FFFFFF")), False);
				{{{ExpendableAsset::P}}}RedrawAsset(This, Asset_Background, False);
				(Frame_ParkScore.GetFirstChild("label-park-score") as CMlLabel).SetText(TL::FormatInteger(_FinishEvent.Parks[Index+1].Score, 1, True));
				Frame_ParkScore.RelativePosition_V3.X = (Index+1) * {{{Size_ParkPoints.X}}};
			}
		}

		// Update total score
		_State.Controls.Label_TotalScore.SetText(TL::FormatInteger(_FinishEvent.TotalScore, 1, True));
		_State.Controls.Label_Points.RelativePosition_V3.X = 0.5 * ML::Min(_State.Controls.Label_TotalScore.ComputeWidth(_State.Controls.Label_TotalScore.Value, True), {{{SizeX_TotalScore}}}) + 5.;

		// Update landings
		_State.Controls.Label_StuntsCount.SetText(""^_FinishEvent.Stats.FiguresCount);
		_State.Controls.Label_Stunts.RelativePosition_V3.X = 0.5 * ML::Min(_State.Controls.Label_StuntsCount.ComputeWidth(_State.Controls.Label_StuntsCount.Value, True), {{{SizeX_TotalStunts}}}) + 5.;
		foreach (LandingType => Count in _FinishEvent.Stats.LandingsCount) {
			(_State.Controls.Frame_Global.GetFirstChild("label-"^LandingType^"-landing-count") as CMlLabel).SetText(""^Count);
		}
	}
}

/// Update the library
Void Yield() {
	foreach (Event in PendingEvents) {
		switch (Event.PlaygroundType) {
			case CManiaAppPlaygroundEvent::Type::LayerCustomEvent: {
				switch (Event.CustomEventType) {
					case UIModules_StuntsResult_Common::C_Event_SkipStuntsResult: {
						assert(Event.CustomEventData.count == 1, "'C_Event_SkipStuntsResult' needs 1 parameter: 'WebServicesUserId'.");
						SendCustomEvent(UIModules_StuntsResult_Common::C_Event_SkipStuntsResult, [Event.CustomEventData[0]]);
					}
				}
			}
		}
	}
