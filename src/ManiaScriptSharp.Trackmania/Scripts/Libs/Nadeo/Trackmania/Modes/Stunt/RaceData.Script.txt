/**
 * Share race data between the server game mode and the clients UI
 */
#Const Version "1.0.0"
#Const ScriptName "Libs/Nadeo/Trackmania/Modes/Stunt/RaceData.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if it's stunt validation mode or not (client side only)
Boolean IsMapValidation(CTeam _Team) {
	declare netread Boolean Net_Stunt_RaceData_IsMapValidation for _Team = False;
	return Net_Stunt_RaceData_IsMapValidation;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Update the boolean that indicates whether it's stunt validation mode or not (server side only)
Void SetIsMapValidation(CTeam _Team, Boolean _IsMapValidation) {
	declare netwrite Boolean Net_Stunt_RaceData_IsMapValidation for _Team = False;
	Net_Stunt_RaceData_IsMapValidation = _IsMapValidation;
}
