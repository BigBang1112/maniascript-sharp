/// Get the vertical position of the frame according to the given rank
Real GetFrameVerticalPosFromRank(Integer _Rank) {
	return -{{{C_PlayerFrameHeight + C_Margin}}} * _Rank;
}

/// Get the ID of the best player
Integer GetBestPlayerId(Struct::K_HotSeatPlayer[Integer] _HotSeatPlayers) {
	declare Integer BestPlayerID = Const::C_PlayerID_Null;
	foreach (Id => HotSeatPlayer in _HotSeatPlayers) {
		if (HotSeatPlayer.BestTime > 0 && (BestPlayerID < 0 || HotSeatPlayer.Rank < _HotSeatPlayers[BestPlayerID].Rank)) {
			BestPlayerID = Id;
		}
	}
	return BestPlayerID;
}

/// Update the best time for the given player
Void SetPlayerBestTime(CMlFrame _Frame_HotSeatPlayer, Integer _PlayerID, Integer _PlayerBestTime, Integer _BestPlayerID, Integer _BestTime) {
	declare Label_BestTime <=> (_Frame_HotSeatPlayer.GetFirstChild("label-best-time") as CMlLabel);

	if (_PlayerBestTime == Const::C_BestTime_None) Label_BestTime.SetText("--:--.---");
	//L16N acronym of the term Did Not Finish (https://en.wikipedia.org/wiki/Did_Not_Finish)
	else if (_PlayerBestTime == Const::C_BestTime_RaceNotFinished) Label_BestTime.SetText(_("|DidNotFinish|DNF"));
	else if (_PlayerID == _BestPlayerID) Label_BestTime.SetText(TL::TimeToText(_PlayerBestTime, True, True));
	else Label_BestTime.SetText("+"^TL::TimeToText(_PlayerBestTime-_BestTime, True, True));
}

/// Update the energy for the given player
Void SetPlayerEnergy(CMlFrame _Frame_HotSeatPlayer, Integer _EnergyAmount, Integer _MaxEnergyAmount) {
	if (_Frame_HotSeatPlayer == Null) return;
	declare CMlFrame Frame_EnergyGaugeClip <=> (_Frame_HotSeatPlayer.GetFirstChild("frame-energy-gauge-clip") as CMlFrame);

	Frame_EnergyGaugeClip.Visible = _MaxEnergyAmount > 0 && _EnergyAmount > 0;
	if (Frame_EnergyGaugeClip.Visible) {
		Frame_EnergyGaugeClip.Size.X = {{{Size_EnergyGauge.X}}} * (_EnergyAmount / ML::ToReal(_MaxEnergyAmount));
	}
}

/// Init players' displays
Void InitHotSeatPlayers(K_State _State, Struct::K_HotSeatPlayer[Integer] _HotSeatPlayers, Integer _MaxEnergyAmount) {
	declare Integer BestPlayerID = GetBestPlayerId(_HotSeatPlayers);

	// Hide all frames
	foreach (Frame_HotSeatPlayer in _State.Controls.Frame_HotSeatPlayers.Controls) {
		Frame_HotSeatPlayer.Hide();
	}

	// Assign a frame to each each player and init it
	foreach (PlayerID => HotSeatPlayer in _HotSeatPlayers) {
		if (_State.Controls.Frame_HotSeatPlayers.Controls.existskey(HotSeatPlayer.Rank)) {
			declare CMlFrame Frame_HotSeatPlayer = _State.Controls.Frame_HotSeatPlayers.Controls[HotSeatPlayer.Rank] as CMlFrame;
			declare CMlLabel Label_Name <=> (Frame_HotSeatPlayer.GetFirstChild("label-name") as CMlLabel);
			declare CMlLabel Label_NameClipped <=> (Frame_HotSeatPlayer.GetFirstChild("label-name-clipped") as CMlLabel);
			declare Integer HotSeat_PlayerID for Frame_HotSeatPlayer;

			// Assign PlayerID to frame
			HotSeat_PlayerID = PlayerID;

			// Update controls
			(Frame_HotSeatPlayer.GetFirstChild("quad-logo") as CMlQuad).ImageUrl = HotSeatPlayer.PlayerLogo;
			(Frame_HotSeatPlayer.GetFirstChild("quad-logo-shadow") as CMlQuad).ImageUrl = HotSeatPlayer.PlayerLogo;
			(Frame_HotSeatPlayer.GetFirstChild("quad-logo-bg") as CMlQuad).BgColor = Clans::GetClanColor(HotSeatPlayer.Clan);
			(Frame_HotSeatPlayer.GetFirstChild("quad-energy-gauge") as CMlQuad).BgColor = Clans::GetClanColor(HotSeatPlayer.Clan);
			Label_Name.SetText(HotSeatPlayer.PlayerName);
			Label_NameClipped.SetText(HotSeatPlayer.PlayerName);
			Label_NameClipped.TextColor = Clans::GetClanForegroundColor(HotSeatPlayer.Clan);
			Tools::FitLabelValue(Label_NameClipped, 3., 2., 0.25);
			Tools::FitLabelValue(Label_Name, 3., 2., 0.25);

			// Set best time and energy display
			SetPlayerBestTime(Frame_HotSeatPlayer, PlayerID, HotSeatPlayer.BestTime, BestPlayerID, _HotSeatPlayers.get(BestPlayerID, Struct::K_HotSeatPlayer {}).BestTime);
			SetPlayerEnergy(Frame_HotSeatPlayer, HotSeatPlayer.EnergyAmount, _MaxEnergyAmount);

			// Set the frame correct position and display
			Frame_HotSeatPlayer.RelativePosition_V3.Y = GetFrameVerticalPosFromRank(HotSeatPlayer.Rank);
			Frame_HotSeatPlayer.Show();
		}
	}
}

/// Update the screen bars with the color of the current player
Void UpdateScreenBars(K_State _State, Boolean _IsPlaying) {
	foreach (Control in _State.Controls.Frame_ScreenBars.Controls) {
		AnimMgr.Flush(Control);
		if (_IsPlaying) {
			AnimMgr.Add(Control, "<anim opacity=\"0.9\" colorize=\""^CL::RgbToHex6(Clans::GetClanColor({{{NetShare::P}}}GetHotSeatPlayers().get(_State.CurrentPlayerID, Struct::K_HotSeatPlayer {}).Clan))^"\" hidden=\"0\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		} else {
			AnimMgr.Add(Control, "<anim opacity=\"0.\" hidden=\"1\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		}
	}
}

/// Refresh players' displays with rank change animations
Void RefreshHotSeatPlayers(K_State _State, Struct::K_HotSeatPlayer[Integer] _HotSeatPlayers, Integer _MaxEnergyAmount) {
	declare Integer BestPlayerID = GetBestPlayerId(_HotSeatPlayers);

	foreach (Control_HotSeatPlayer in _State.Controls.Frame_HotSeatPlayers.Controls) {
		declare CMlFrame Frame_HotSeatPlayer = Control_HotSeatPlayer as CMlFrame;
		declare Integer HotSeat_PlayerID for Frame_HotSeatPlayer = Const::C_PlayerID_Null;

		if (_HotSeatPlayers.existskey(HotSeat_PlayerID)) {
			declare Struct::K_HotSeatPlayer HotSeatPlayer = _HotSeatPlayers[HotSeat_PlayerID];

			// Set best time and energy display
			SetPlayerBestTime(Frame_HotSeatPlayer, HotSeat_PlayerID, HotSeatPlayer.BestTime, BestPlayerID, _HotSeatPlayers.get(BestPlayerID, Struct::K_HotSeatPlayer {}).BestTime);
			SetPlayerEnergy(Frame_HotSeatPlayer, HotSeatPlayer.EnergyAmount, _MaxEnergyAmount);

			// Move frame at the correct position and display
			AnimMgr.Flush(Frame_HotSeatPlayer);
			AnimMgr.Add(Frame_HotSeatPlayer, "<anim pos=\"0 "^GetFrameVerticalPosFromRank(HotSeatPlayer.Rank)^"\" />", Now, 250, CAnimManager::EAnimManagerEasing::QuadOut);
			Frame_HotSeatPlayer.Show();
		} else {
			Frame_HotSeatPlayer.Hide();
		}
	}
}

///	Set current player on display
K_State SetCurrentPlayer(K_State _State, Struct::K_HotSeatPlayer[Integer] _HotSeatPlayers, Integer _CurrentPlayerID) {
	declare K_State State = _State;

	foreach (Control_HotSeatPlayer in State.Controls.Frame_HotSeatPlayers.Controls) {
		declare CMlFrame Frame_HotSeatPlayer = Control_HotSeatPlayer as CMlFrame;
		declare CMlLabel Label_BestTime <=> (Frame_HotSeatPlayer.GetFirstChild("label-best-time") as CMlLabel);
		declare CMlQuad Quad_LogoBg <=> (Frame_HotSeatPlayer.GetFirstChild("quad-logo-bg") as CMlQuad);
		declare CMlQuad Quad_TimeBg <=> (Frame_HotSeatPlayer.GetFirstChild("quad-time-bg") as CMlQuad);
		declare Integer HotSeat_PlayerID for Frame_HotSeatPlayer = Const::C_PlayerID_Null;
		declare Struct::K_HotSeatPlayer HotSeatPlayer = _HotSeatPlayers.get(HotSeat_PlayerID, Struct::K_HotSeatPlayer {});

		AnimMgr.Flush(Label_BestTime);
		AnimMgr.Flush(Quad_LogoBg);
		AnimMgr.Flush(Quad_TimeBg);

		if (HotSeat_PlayerID == _CurrentPlayerID) {
			State.Controls.Frame_CurrentHotSeatPlayer = Frame_HotSeatPlayer;
			AnimMgr.Add(Label_BestTime, "<anim textcolor=\""^CL::RgbToHex6(Clans::GetClanForegroundColor(HotSeatPlayer.Clan))^"\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			AnimMgr.Add(Quad_LogoBg, "<anim opacity=\"0.8\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			AnimMgr.Add(Quad_TimeBg, "<anim opacity=\"0.8\" bgcolor=\""^CL::RgbToHex6(Clans::GetClanColor(HotSeatPlayer.Clan))^"\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		} else {
			AnimMgr.Add(Label_BestTime, "<anim textcolor=\"{{{Stylesheet::GetColorHex6(Stylesheet::C_Color_TextLight)}}}\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			AnimMgr.Add(Quad_LogoBg, "<anim opacity=\"0.4\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			AnimMgr.Add(Quad_TimeBg, "<anim opacity=\"0.4\" bgcolor=\"{{{Stylesheet::GetColorHex6(Stylesheet::C_Color_BGDark)}}}\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
			AnimMgr.Add(Frame_HotSeatPlayer.GetFirstChild("quad-energy-gauge"), "<anim opacity=\"0.95\" />", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		}
	}

	return State;
}

/// Play an animation on the current player anergy gauge
Void PlayEnergyGaugeAnim(CMlFrame _Frame_HotSeatPlayer, Boolean _Play) {
	if (_Frame_HotSeatPlayer == Null) return;
	declare Quad_EnergyGauge <=> (_Frame_HotSeatPlayer.GetFirstChild("quad-energy-gauge") as CMlQuad);
	if (_Play) Quad_EnergyGauge.Opacity = 0.75 + (ML::Cos(Now % ML::NearestInteger(ML::Tau * 200) / 200.) * 0.25);
	else Quad_EnergyGauge.Opacity = 1.;
}
