// Update the record text and image opacity
Void SetRecordOpacity(CMlLabel _LabelStarReward, CMlLabel _Label_StarTime, CMlQuad _QuadStar, Boolean _IsRecordPossible) {
	declare Real Opacity = 0.6;
	if (_IsRecordPossible) {
		Opacity = 1.;
	}

	_LabelStarReward.Opacity = Opacity;
	_Label_StarTime.Opacity = Opacity;
	_QuadStar.Opacity = Opacity;
}

// Show only one digit for the minutes if the time is less than 10mins (example : 2:45.922 instead of 02:45.922)
Text CustomTimeToText(Integer _Time) {
	declare Text Value = TL::TimeToText(_Time, True, True);
	if (_Time < C_TenMinutes && TL::StartsWith("0", Value)) {
		return TL::SubString(Value, 1, TL::Length(Value));
	}
	return Value;
}

// Update the segment record display
K_State UpdateSegmentRecordDisplay(K_State _State, Integer _CurrentPlayerRaceTime) {
	declare K_State State = _State;
	State.PossibleRecords.clear();

	// The keys in the State.SegmentStarTimes array are the segment index
	// 1 is white, 2 is green, ...
	if (State.SegmentStarTimes.existskey(State.TargetSegment.Current)) {
		State.Controls.Frame_Records.Visible = True;
		State.Controls.Label_NoRecord.Visible = False;

		// Update stars rewards number
		State.Controls.Label_StarReward_Author.Value = TL::ToText(State.StarRewards.Record);
		State.Controls.Label_StarReward_Gold.Value = TL::ToText(State.StarRewards.Gold);
		State.Controls.Label_StarReward_Silver.Value = TL::ToText(State.StarRewards.Silver);
		State.Controls.Label_StarReward_Bronze.Value = TL::ToText(State.StarRewards.Bronze);

		// Update record times
		State.Controls.Label_StarTime_Author.Value = CustomTimeToText(State.SegmentStarTimes[State.TargetSegment.Current].Record);
		State.Controls.Label_StarTime_Gold.Value = CustomTimeToText(State.SegmentStarTimes[State.TargetSegment.Current].Gold);
		State.Controls.Label_StarTime_Silver.Value = CustomTimeToText(State.SegmentStarTimes[State.TargetSegment.Current].Silver);

		// Update records opacity
		SetRecordOpacity(State.Controls.Label_StarReward_Author, State.Controls.Label_StarTime_Author, State.Controls.Quad_Star_Author, _CurrentPlayerRaceTime <= State.SegmentStarTimes[State.TargetSegment.Current].Record);
		SetRecordOpacity(State.Controls.Label_StarReward_Gold, State.Controls.Label_StarTime_Gold, State.Controls.Quad_Star_Gold, _CurrentPlayerRaceTime <= State.SegmentStarTimes[State.TargetSegment.Current].Gold);
		SetRecordOpacity(State.Controls.Label_StarReward_Silver, State.Controls.Label_StarTime_Silver, State.Controls.Quad_Star_Silver, _CurrentPlayerRaceTime <= State.SegmentStarTimes[State.TargetSegment.Current].Silver);

		// Store possible records
		State.PossibleRecords.add(State.SegmentStarTimes[State.TargetSegment.Current].Record);
		State.PossibleRecords.add(State.SegmentStarTimes[State.TargetSegment.Current].Gold);
		State.PossibleRecords.add(State.SegmentStarTimes[State.TargetSegment.Current].Silver);
	} else {
		// No record yet
		State.Controls.Frame_Records.Visible = False;
		State.Controls.Label_NoRecord.Visible = True;
	}

	return State;
}

// Update the new record display
Void UpdateNewRecordDisplay(K_State _State, Integer _Time, Text _AccountId, Integer _TeamId) {
	// It is not possible to get the name of a bot from its account id
	// But we can get it from the user instead
	declare CUser RecordUser;
	for (Score in Scores) {
		if (Score.User != Null && Score.User.WebServicesUserId == _AccountId) {
			RecordUser <=> Score.User;
			break;
		}
	}

	// Update text infos
	if (RecordUser != Null) {
		{{{PlayerName::P}}}SetUser(This, _State.Controls.PlayerName_NewRecord, RecordUser, True, False);
	} else {
		{{{PlayerName::P}}}SetAccountId(This, _State.Controls.PlayerName_NewRecord, _AccountId, True, False);
	}
	_State.Controls.Label_NewRecordTime.Value = CustomTimeToText(_Time);

	// Update text and background colors corresponding to the team
	_State.Controls.Quad_BgNewRecord.BgColor = Clans::GetClanColor(_TeamId);
	declare Vec3 ForegroundColor = Clans::GetClanForegroundColor(_TeamId);
	_State.Controls.Label_NewRecord.TextColor = ForegroundColor;
	{{{PlayerName::P}}}SetTextColor(This, _State.Controls.PlayerName_NewRecord, ForegroundColor, True);
	_State.Controls.Label_NewRecordTime.TextColor = ForegroundColor;

	// Display animation
	_State.Controls.Frame_AnimNewRecord.RelativePosition_V3.X = {{{C_Size_DisplayClip.X}}};
	AnimMgr.Flush(_State.Controls.Frame_AnimNewRecord);
	AnimMgr.Add(_State.Controls.Frame_AnimNewRecord, "<a pos=\"0 0\" hidden=\"0\"/>", C_AnimDuration_NewRecord_FadeIn, CAnimManager::EAnimManagerEasing::QuadIn);
	AnimMgr.Add(_State.Controls.Frame_AnimNewRecord, "<a pos=\""^C_Size_DisplayClip.X^" 0\" hidden=\"1\"/>", Now + C_AnimDuration_NewRecord_DisplayTime, C_AnimDuration_NewRecord_FadeIn, CAnimManager::EAnimManagerEasing::QuadIn);
}

// Update the visibility of the whole manialink UI
K_State SetVisibility(K_State _State, Boolean _PageIsVisible, Boolean _DisplayModule, Real _TeamScoreHeight) {
	declare K_State State = _State;

	State.PageIsVisible = _PageIsVisible;
	State.DisplayModule = _DisplayModule;
	State.IsVisible = State.PageIsVisible && State.DisplayModule && State.TargetId != NullId;

	AnimMgr.Flush(State.Controls.Frame_DisplayAnimation);
	if (State.IsVisible) {
		AnimMgr.Add(State.Controls.Frame_DisplayAnimation, "<a pos=\"0 0\" hidden=\"0\" />", C_AnimDuration_ShowHideModule, CAnimManager::EAnimManagerEasing::QuadIn);
	} else {
		AnimMgr.Add(State.Controls.Frame_DisplayAnimation, "<a pos=\""^C_Size_DisplayClip.X^" 0\" hidden=\"1\" />", C_AnimDuration_ShowHideModule, CAnimManager::EAnimManagerEasing::QuadIn);
	}

	if (State.TeamScoreHeight != _TeamScoreHeight) {
		State.TeamScoreHeight = _TeamScoreHeight;
		State.Controls.Frame_TeamScoreHeight.RelativePosition_V3.Y = -State.TeamScoreHeight - C_MarginWithTeamScore;
	}

	if (State.IsVisible) {
		UIModulesData::SetPosition(This, {{{dump(GetId())}}}, State.Controls.Frame_TeamScoreHeight.AbsolutePosition_V3);
	} else {
		UIModulesData::SetPosition(This, {{{dump(GetId())}}}, State.Controls.Frame_Global.GetFirstChild("{{{UIModules::GetCustomizableUIModuleFrameId(UIModules_SegmentRecord_Common::C_UIModuleConfig)}}}").RelativePosition_V3);
	}

	return State;
}

// Update the star times data
K_State SetSegmentStarTimes(
	K_State _State,
	Integer _SegmentStarTimesUpdate,
	RaceData::K_StarTimes[Integer] _SegmentStarTimes
) {
	declare K_State State = _State;

	// Detect new record on the target current segment
	if (
		State.TargetId != NullId && (
			(
				!State.SegmentStarTimes.existskey(State.TargetSegment.Current) &&
				_SegmentStarTimes.existskey(State.TargetSegment.Current)
			) || (
				_SegmentStarTimes.existskey(State.TargetSegment.Current) &&
				State.SegmentStarTimes[State.TargetSegment.Current].Record != _SegmentStarTimes[State.TargetSegment.Current].Record
			)
		)
	) {
		UpdateNewRecordDisplay(
			_State,
			_SegmentStarTimes[State.TargetSegment.Current].Record,
			_SegmentStarTimes[State.TargetSegment.Current].RecordAccountId,
			_SegmentStarTimes[State.TargetSegment.Current].RecordTeamId
		);
	}

	State.SegmentStarTimesUpdate = _SegmentStarTimesUpdate;
	State.SegmentStarTimes = _SegmentStarTimes;

	State.UpdateSegmentRecordTime = Now + C_AnimDuration_NewRecord_DisplayTime;

	return State;
}

// Update the star rewards data
K_State SetStarRewards(
	K_State _State,
	Integer _StarRewardsUpdate,
	RaceData::K_StarRewards _StarRewards
) {
	declare K_State State = _State;

	State.StarRewardsUpdate = _StarRewardsUpdate;
	State.StarRewards = _StarRewards;

	State.UpdateSegmentRecordTime = Now;

	return State;
}

// Update the number of points earned for holding a segment record
K_State SetSegmentRecordPoints(K_State _State, Integer _SegmentRecordPoints) {
	declare K_State State = _State;

	State.SegmentRecordPoints = _SegmentRecordPoints;
	State.Controls.Label_NewRecord.SetText(TL::Compose("%1 %2", "{{{Text_NewRecord}}}", "+"^State.SegmentRecordPoints));

	return State;
}

// Update the targeted player
K_State SetTarget(K_State _State, CSmPlayer _Target) {
	declare K_State State = _State;
	declare Boolean NewTarget = False;

	if (_Target == Null) {
		State.TargetId = NullId;
		State.TargetSegment = PlayerData::K_Segment {};
	} else {
		if (State.TargetId != _Target.Id) NewTarget = True;
		State.TargetId = _Target.Id;
		State.TargetSegment = PlayerData::GetPlayerSegment(_Target);
	}

	State = SetVisibility(State, State.PageIsVisible, State.DisplayModule, UIModulesData::GetSize(This, UIModules_TeamScore_Common::C_Id).Y);

	if (NewTarget) {
		State.UpdateSegmentRecordTime = Now;
	} else { // same target but new track segment
		State.UpdateSegmentRecordTime = Now + C_DelayBetweenSegments;
	}

	return State;
}

// Get the index of the possible record corresponding to the current race time
Integer GetPossibleRecordIndex(K_State _State, Integer _CurrentRaceTime) {
	for (Index => Time in _State.PossibleRecords) {
		if (_CurrentRaceTime <= Time) return Index;
	}
	return _State.PossibleRecords.count;
}
