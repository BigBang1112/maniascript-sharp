/// Update the visibility of the whole manialink UI
K_State SetVisibility(K_State _State, Boolean _PageIsVisible, Boolean _DisplayModule) {
	declare K_State State = _State;

	State.PageIsVisible = _PageIsVisible;
	State.DisplayModule = _DisplayModule;
	State.IsVisible = State.PageIsVisible && State.DisplayModule;
	State.Controls.Frame_Global.Visible = State.IsVisible;

	return State;
}

/// Update the position of the whole manialink UI depending on the given UI module
Vec2 SetPosition(K_State _State, Text _UIModuleId) {
	declare UIModules_Common::K_ModuleConfig UIModuleConfig = UIModules_SegmentsLeaders_Common::C_UIModuleConfig;
	declare Vec2 Pos_LinkedUIModule = UIModulesData::GetPosition(This, _UIModuleId);
	declare Vec2 Size_LinkedUIModule = UIModulesData::GetSize(This, _UIModuleId);
	declare Real UIModuleMargin = 2.;

	// On Sony consoles every player names must be in the safe area
	// see: /trackmania-next/tmnext/-/issues/4660
	_State.Controls.Frame_UIModule.RelativePosition_V3 = <
		Tools::Ternary(Platform::IsSony(System), ML::Clamp(UIModuleConfig.Properties.Position.X, -144., 144.), UIModuleConfig.Properties.Position.X),
		UIModuleConfig.Properties.Position.Y - ML::Abs(UIModuleConfig.Properties.Position.Y - Pos_LinkedUIModule.Y) - Size_LinkedUIModule.Y - UIModuleMargin
	>;

	return Pos_LinkedUIModule;
}

// Update the display of the segments leaders
Void UpdateSegmentLeaderInfo(K_State _State, CMlFrame _Frame_SegmentLeader, RaceData::K_StarTimes _SegmentStarTime) {
	declare CMlFrame PlayerName_SegmentLeader <=> (_Frame_SegmentLeader.GetFirstChild("playername-segment-leader") as CMlFrame);
	declare CMlQuad Quad_TeamLogo <=> (_Frame_SegmentLeader.GetFirstChild("quad-team-logo") as CMlQuad);
	declare CUser RecordUser;
	for (Score in Scores) {
		if (Score.User != Null && Score.User.WebServicesUserId == _SegmentStarTime.RecordAccountId) {
			RecordUser <=> Score.User;
			break;
		}
	}

	// Update clan logo and player name
	Quad_TeamLogo.ImageUrl = Clans::GetClanLogo(_SegmentStarTime.RecordTeamId);
	if (RecordUser != Null) {
		{{{PlayerName::P}}}SetUser(This, PlayerName_SegmentLeader, RecordUser, True, True);
	} else {
		{{{PlayerName::P}}}SetAccountId(This, PlayerName_SegmentLeader, _SegmentStarTime.RecordAccountId, True, True);
	}
}

/// Flush all animations of a segment leader frame
Void FlushAnimations(CMlFrame _Frame_SegmentLeader) {
	AnimMgr.Flush(_Frame_SegmentLeader);
	foreach (Control in _Frame_SegmentLeader.Controls) {
		AnimMgr.Flush(Control);
	}
}

/// Play an animation that extends the size of the cliping frame
Void PlayAnimation_ClipExtension(K_State _State, RaceData::K_StarTimes[Integer] _NewSegmentStarTimes) {
	// Flush animations
	AnimMgr.Flush(_State.Controls.Frame_AnimationClip);

	// Show header if there is at least one record
	if (_NewSegmentStarTimes.count <= 0) {
		_State.Controls.Frame_AnimationClip.Size.Y = 0.;
		_State.Controls.Frame_AnimationClip.Hide();
	} else if (!_State.Controls.Frame_AnimationClip.Visible) {
		AnimMgr.Add(_State.Controls.Frame_AnimationClip, "<anim size=\""^{{{C_Size_DisplayClip.X}}}^" "^{{{C_Size_DisplayClip.Y}}}^"\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
		_State.Controls.Frame_AnimationClip.Show();
	}
}

/// Play an animation when the first record is set on a segment
Void PlayAnimation_FirstSegmentRecord(K_State _State, CMlFrame _Frame_SegmentLeader) {
	declare CMlFrame PlayerName_SegmentLeader <=> (_Frame_SegmentLeader.GetFirstChild("playername-segment-leader") as CMlFrame);
	declare CMlQuad Quad_Background <=> (_Frame_SegmentLeader.GetFirstChild("quad-background") as CMlQuad);
	declare CMlQuad Quad_Flag <=> (_Frame_SegmentLeader.GetFirstChild("quad-segment-flag") as CMlQuad);
	declare CMlQuad Quad_TeamLogo <=> (_Frame_SegmentLeader.GetFirstChild("quad-team-logo") as CMlQuad);

	// Initialize
	_Frame_SegmentLeader.Hide();
	_Frame_SegmentLeader.RelativePosition_V3.X = 2.;
	foreach (Control in _Frame_SegmentLeader.Controls) {
		switchtype (Control as TypedControl) {
			case CMlQuad: TypedControl.Opacity = 0.;
			case CMlLabel: TypedControl.Opacity = 0.;
			default: {
				if ({{{PlayerName::P}}}IsPlayerName(Control)) {
					{{{PlayerName::P}}}SetOpacity(This, Control as CMlFrame, 0.);
				}
			}
		}
	}

	// Flush animations
	FlushAnimations(_Frame_SegmentLeader);

	// Play animations
	AnimMgr.Add(_Frame_SegmentLeader, "<anim pos=\"0 "^_Frame_SegmentLeader.RelativePosition_V3.Y^"\" hidden=\"0\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	AnimMgr.Add(Quad_Background, "<anim opacity=\"0.6\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	AnimMgr.Add(Quad_Flag, "<anim opacity=\"1.\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	AnimMgr.Add(Quad_TeamLogo, "<anim opacity=\"1.\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	{{{PlayerName::P}}}SetOpacity(This, PlayerName_SegmentLeader, 1., 250);
}

/// Play an animation when a new record is replacing a segment record
K_State PlayAnimation_NewSegmentRecordIn(K_State _State, CMlFrame _Frame_SegmentLeader) {
	declare K_State State = _State;
	declare CMlFrame PlayerName_SegmentLeader <=> (_Frame_SegmentLeader.GetFirstChild("playername-segment-leader") as CMlFrame);
	declare CMlLabel Label_NewRecord <=> (_Frame_SegmentLeader.GetFirstChild("label-new-record") as CMlLabel);
	declare CMlQuad Quad_TeamLogo <=> (_Frame_SegmentLeader.GetFirstChild("quad-team-logo") as CMlQuad);

	// Initialize
	Quad_TeamLogo.Opacity = 1.;
	Quad_TeamLogo.RelativePosition_V3 = {{{Pos_TeamLogo}}};
	{{{PlayerName::P}}}SetOpacity(This, PlayerName_SegmentLeader, 1.);
	PlayerName_SegmentLeader.RelativePosition_V3 = {{{Pos_SegmentLeaderName}}};
	Label_NewRecord.Opacity = 0.;
	Label_NewRecord.RelativePosition_V3 = {{{Pos_NewRecord}}} - <2., 0.>;

	// Flush animations
	AnimMgr.Flush(PlayerName_SegmentLeader);
	AnimMgr.Flush(Label_NewRecord);
	AnimMgr.Flush(Quad_TeamLogo);

	// Play animations
	{{{PlayerName::P}}}SetOpacity(This, PlayerName_SegmentLeader, 0., 250);
	AnimMgr.Add(PlayerName_SegmentLeader, "<anim pos=\""^{{{Pos_SegmentLeaderName.X+2.}}}^" "^{{{Pos_SegmentLeaderName.Y}}}^"\" opacity=\"0.\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	AnimMgr.Add(Label_NewRecord, "<anim pos=\""^{{{Pos_NewRecord.X}}}^" "^{{{Pos_NewRecord.Y}}}^"\" opacity=\"1.\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	AnimMgr.Add(Quad_TeamLogo, "<anim pos=\""^{{{Pos_TeamLogo.X+2.}}}^" "^{{{Pos_TeamLogo.Y}}}^"\" opacity=\"0.\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);

	// Delay the out animation
	if (State.DelayedAnimations.existskey(Now + 2500)) {
		if (!State.DelayedAnimations[Now + 2500].exists(_Frame_SegmentLeader)) {
			State.DelayedAnimations[Now + 2500].add(_Frame_SegmentLeader);
		}
	} else {
		State.DelayedAnimations[Now + 2500] = [_Frame_SegmentLeader];
	}

	return State;
}

/// Play an animation when a new record is replacing a segment record

	AnimMgr.AddChain(Label_NewRecord, "<anim pos=\""^{{{Pos_NewRecord.X-2.}}}^" "^{{{Pos_NewRecord.Y}}}^"\" opacity=\"0.\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
	AnimMgr.AddChain(Quad_TeamLogo, "<anim pos=\""^{{{Pos_TeamLogo.X}}}^" "^{{{Pos_TeamLogo.Y}}}^"\" opacity=\"1.\"/>", 250, CAnimManager::EAnimManagerEasing::QuadOut);
}

// Update the star times data
K_State SetSegmentStarTimes(K_State _State, Integer _SegmentStarTimesUpdate, RaceData::K_StarTimes[Integer] _SegmentStarTimes) {
	declare K_State State = _State;

	// Update the clip size
	if (State.SegmentStarTimes.count <= 0 || _SegmentStarTimes.count != State.SegmentStarTimes.count) {
		PlayAnimation_ClipExtension(State, _SegmentStarTimes);
	}

	// Update segment leaders
	foreach (SegmentIndex => Control in State.Controls.Frame_SegmentsLeaders.Controls) {
		declare RaceData::K_StarTimes OldSegmentStarsTime = State.SegmentStarTimes.get(SegmentIndex+1, RaceData::K_StarTimes {});
		declare RaceData::K_StarTimes NewSegmentStarsTime = _SegmentStarTimes.get(SegmentIndex+1, RaceData::K_StarTimes {});
		if (NewSegmentStarsTime.Record > 0) {
			if (OldSegmentStarsTime.Record > 0 && NewSegmentStarsTime.Record < OldSegmentStarsTime.Record) {
				State = PlayAnimation_NewSegmentRecordIn(State, Control as CMlFrame);
			} else if (OldSegmentStarsTime.Record <= 0) {
				UpdateSegmentLeaderInfo(State, Control as CMlFrame, NewSegmentStarsTime);
				PlayAnimation_FirstSegmentRecord(State, Control as CMlFrame);
			}
		} else {
			FlushAnimations(Control as CMlFrame);
			Control.Hide();
		}
	}

	// Update with new data
	State.SegmentStarTimesUpdate = _SegmentStarTimesUpdate;
	State.SegmentStarTimes = _SegmentStarTimes;

	return State;
}

// Update the number of points earned for holding a segment record
K_State SetSegmentRecordPoints(K_State _State, Integer _SegmentRecordPoints) {
	declare K_State State = _State;

	State.SegmentRecordPoints = _SegmentRecordPoints;
	State.Controls.Label_Points.SetText("+"^State.SegmentRecordPoints);

	return State;
}
