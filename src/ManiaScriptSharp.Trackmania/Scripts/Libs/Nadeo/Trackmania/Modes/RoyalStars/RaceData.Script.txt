/**
 *	Share race data between the server game mode and the clients UI
 */
#Const Version		"3.1.0"
#Const ScriptName	"Libs/Nadeo/Trackmania/Modes/RoyalStars/RaceData.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_StarTimes {
	Integer Record;
	Integer Gold;
	Integer Silver;
	Text RecordAccountId;
	Integer RecordTeamId;
}

#Struct K_StarRewards {
	Integer First;
	Integer Record;
	Integer Gold;
	Integer Silver;
	Integer Bronze;
}

#Struct K_TeamProgress {
	Integer Segment;
	Integer StarsNb;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Default_UpdateCounter 0
#Const C_Default_SegmentStarTimes []
#Const C_Default_StarRewards []
#Const C_Default_TeamsProgress []
#Const C_Default_SegmentRecordPoints 0

#Const C_StarRewards_Null K_StarRewards {
	First = 0,
	Record = 0,
	Gold = 0,
	Silver = 0,
	Bronze = 0
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the times the players must beat on each segment to get stars (server side only)
Void SetSegmentStarTimes(CTeam _Team, K_StarTimes[Integer] _SegmentStarTimes) {
	if (_Team == Null) return;
	declare netwrite Integer Net_RoyalStars_RaceData_SegmentStarTimesUpdate for _Team = C_Default_UpdateCounter;
	declare netwrite K_StarTimes[Integer] Net_RoyalStars_RaceData_SegmentStarTimes for _Team = C_Default_SegmentStarTimes;
	Net_RoyalStars_RaceData_SegmentStarTimes = _SegmentStarTimes;
	Net_RoyalStars_RaceData_SegmentStarTimesUpdate += 1;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the segment star times update counter (client side only)
Integer GetSegmentStarTimesUpdate(CTeam _Team) {
	if (_Team == Null) return C_Default_UpdateCounter;
	declare netread Integer Net_RoyalStars_RaceData_SegmentStarTimesUpdate for _Team = C_Default_UpdateCounter;
	return Net_RoyalStars_RaceData_SegmentStarTimesUpdate;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the segment star times (client side only)
K_StarTimes[Integer] GetSegmentStarTimes(CTeam _Team) {
	if (_Team == Null) return C_Default_SegmentStarTimes;
	declare netread K_StarTimes[Integer] Net_RoyalStars_RaceData_SegmentStarTimes for _Team = C_Default_SegmentStarTimes;
	return Net_RoyalStars_RaceData_SegmentStarTimes;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the number of stars earned when finishing a race (server side only)
Void SetStarRewards(
	CTeam _Team,
	Integer _First,
	Integer _Record,
	Integer _Gold,
	Integer _Silver,
	Integer _Bronze
) {
	if (_Team == Null) return;
	declare netwrite Integer Net_RoyalStars_RaceData_StarRewardsUpdate for _Team = C_Default_UpdateCounter;
	declare netwrite Integer[] Net_RoyalStars_RaceData_StarRewards for _Team = C_Default_StarRewards;
	Net_RoyalStars_RaceData_StarRewards = [_First, _Record, _Gold, _Silver, _Bronze];
	Net_RoyalStars_RaceData_StarRewardsUpdate += 1;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the number of stars earned when finishing a race update counter (client side only)
Integer GetStarRewardsUpdate(CTeam _Team) {
	if (_Team == Null) return C_Default_UpdateCounter;
	declare netread Integer Net_RoyalStars_RaceData_StarRewardsUpdate for _Team = C_Default_UpdateCounter;
	return Net_RoyalStars_RaceData_StarRewardsUpdate;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the number of stars earned when finishing a race (client side only)
K_StarRewards GetStarRewards(CTeam _Team) {
	if (_Team == Null) return C_StarRewards_Null;
	declare netread Integer[] Net_RoyalStars_RaceData_StarRewards for _Team = C_Default_StarRewards;
	declare K_StarRewards StarRewards = C_StarRewards_Null;
	if (Net_RoyalStars_RaceData_StarRewards.count >= 5) {
		StarRewards.First = Net_RoyalStars_RaceData_StarRewards[0];
		StarRewards.Record = Net_RoyalStars_RaceData_StarRewards[1];
		StarRewards.Gold = Net_RoyalStars_RaceData_StarRewards[2];
		StarRewards.Silver = Net_RoyalStars_RaceData_StarRewards[3];
		StarRewards.Bronze = Net_RoyalStars_RaceData_StarRewards[4];
	}
	return StarRewards;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the team progress in the race (server side only)
Void SetTeamProgress(CTeam _Team, Integer _TeamId, Integer _Segment, Integer _StarsNb) {
	if (_Team == Null) return;
	declare netwrite Integer Net_RoyalStars_RaceData_TeamsProgressUpdate for _Team = C_Default_UpdateCounter;
	declare netwrite K_TeamProgress[Integer] Net_RoyalStars_RaceData_TeamsProgress for _Team = C_Default_TeamsProgress;
	if (
		!Net_RoyalStars_RaceData_TeamsProgress.existskey(_TeamId) ||
		Net_RoyalStars_RaceData_TeamsProgress[_TeamId].Segment != _Segment ||
		Net_RoyalStars_RaceData_TeamsProgress[_TeamId].StarsNb != _StarsNb
	) {
		Net_RoyalStars_RaceData_TeamsProgress[_TeamId] = K_TeamProgress {
			Segment = _Segment,
			StarsNb = _StarsNb
		};
		Net_RoyalStars_RaceData_TeamsProgressUpdate += 1;
	}
}
Void ResetTeamsProgress(CTeam _Team) {
	if (_Team == Null) return;
	declare netwrite Integer Net_RoyalStars_RaceData_TeamsProgressUpdate for _Team = C_Default_UpdateCounter;
	declare netwrite K_TeamProgress[Integer] Net_RoyalStars_RaceData_TeamsProgress for _Team = C_Default_TeamsProgress;
	Net_RoyalStars_RaceData_TeamsProgress = C_Default_TeamsProgress;
	Net_RoyalStars_RaceData_TeamsProgressUpdate += 1;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the teams progress update counter (client side only)
Integer GetTeamsProgressUpdate(CTeam _Team) {
	if (_Team == Null) return C_Default_UpdateCounter;
	declare netread Integer Net_RoyalStars_RaceData_TeamsProgressUpdate for _Team = C_Default_UpdateCounter;
	return Net_RoyalStars_RaceData_TeamsProgressUpdate;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //

K_TeamProgress[Integer] GetTeamsProgress(CTeam _Team) {

}
