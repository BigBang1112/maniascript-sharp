/**
 *	Share player specific data between the server game mode and the clients UI
 */
#Const Version		"3.0.2"
#Const ScriptName	"Libs/Nadeo/Trackmania/Modes/RoyalStars/PlayerData.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structure
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_Segment {
	Integer Current;
	Integer Next;
	Integer Loop;
}

#Struct K_RaceFinishEvent {
	K_Segment OldSegment;
	K_Segment NewSegment;
	Integer[] OldStars;
	Integer[] NewStars;
	Integer[] EarnedStars;
	Integer Time;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_Default_PlayerSegment K_Segment {
	Current = 1,
	Next = 2,
	Loop = 0
}
#Const C_Default_PlayerStars []
#Const C_Default_PlayerNeedSynchronization True
#Const C_Default_PlayerSynchronizationPoint 0
#Const C_Default_PlayerRaceFinishEventUpdate 0
#Const C_Default_PlayerRaceFinishEvent K_RaceFinishEvent {
	OldSegment = K_Segment {
		Current = 1,
		Next = 2,
		Loop = 0
	},
	NewSegment = K_Segment {
		Current = 1,
		Next = 2,
		Loop = 0
	},
	OldStars = [],
	NewStars = [],
	EarnedStars = [],
	Time = 0
}

#Const C_AnimDuration_ShowStars 1500
#Const C_AnimDuration_AddStars 1000
#Const C_AnimDuration_UpdateSegment 1000

#Const C_StarLevel_First 0
#Const C_StarLevel_Record 1
#Const C_StarLevel_Gold 2
#Const C_StarLevel_Silver 3
#Const C_StarLevel_Bronze 4

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the segment progression of a player (server side only)
Void SetPlayerSegment(CPlayer _Player, K_Segment _PlayerSegment) {
	if (_Player == Null) return;
	declare netwrite K_Segment Net_RoyalStars_PlayerData_PlayerSegment for _Player = C_Default_PlayerSegment;
	Net_RoyalStars_PlayerData_PlayerSegment = _PlayerSegment;
}
Void ResetPlayerSegment(CPlayer _Player) {
	SetPlayerSegment(_Player, C_Default_PlayerSegment);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the segment progression of a player (client side only)
K_Segment GetPlayerSegment(CPlayer _Player) {
	if (_Player == Null) return C_Default_PlayerSegment;
	declare netread K_Segment Net_RoyalStars_PlayerData_PlayerSegment for _Player = C_Default_PlayerSegment;
	return Net_RoyalStars_PlayerData_PlayerSegment;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the stars owned by player (server side only)
Void SetPlayerStars(CPlayer _Player, Integer[] _Stars) {
	if (_Player == Null) return;
	declare netwrite Integer[] Net_RoyalStars_PlayerData_PlayerStars for _Player = C_Default_PlayerStars;
	Net_RoyalStars_PlayerData_PlayerStars = _Stars;
}
Void ResetPlayerStars(CPlayer _Player) {
	SetPlayerStars(_Player, C_Default_PlayerStars);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the stars owned by player (client side only)
Integer[] GetPlayerStars(CPlayer _Player) {
	if (_Player == Null) return C_Default_PlayerStars;
	declare netread Integer[] Net_RoyalStars_PlayerData_PlayerStars for _Player = C_Default_PlayerStars;
	return Net_RoyalStars_PlayerData_PlayerStars;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Send a player segment progression event to the clients (server side only)
Void SendPlayerRaceFinishEvent(
	CSmPlayer _Player,
	K_Segment _OldSegment,
	K_Segment _NewSegment,
	Integer[] _OldStars,
	Integer[] _NewStars,
	Integer[] _EarnedStars,
	Integer _Time
) {
	if (_Player == Null) return;
	declare netwrite Integer Net_RoyalStars_PlayerData_RaceFinishEventUpdate for _Player = C_Default_PlayerRaceFinishEventUpdate;
	declare netwrite K_RaceFinishEvent Net_RoyalStars_PlayerData_RaceFinishEvent for _Player = C_Default_PlayerRaceFinishEvent;
	Net_RoyalStars_PlayerData_RaceFinishEvent = K_RaceFinishEvent {
		OldSegment = _OldSegment,
		NewSegment = _NewSegment,
		OldStars = _OldStars,
		NewStars = _NewStars,
		EarnedStars = _EarnedStars,
		Time = _Time
	};
	Net_RoyalStars_PlayerData_RaceFinishEventUpdate += 1;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the player race finish event counter value (client side only)
Integer GetPlayerRaceFinishEventUpdate(CSmPlayer _Player) {
	if (_Player == Null) return C_Default_PlayerRaceFinishEventUpdate;
	declare netread Integer Net_RoyalStars_PlayerData_RaceFinishEventUpdate for _Player = C_Default_PlayerRaceFinishEventUpdate;
	return Net_RoyalStars_PlayerData_RaceFinishEventUpdate;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the player race finish event (client side only)
K_RaceFinishEvent GetPlayerRaceFinishEvent(CSmPlayer _Player) {
	if (_Player == Null) return C_Default_PlayerRaceFinishEvent;
	declare netread K_RaceFinishEvent Net_RoyalStars_PlayerData_RaceFinishEvent for _Player = C_Default_PlayerRaceFinishEvent;
	return Net_RoyalStars_PlayerData_RaceFinishEvent;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set if the player data need to be synchronized on the client (server side only)
Void SetPlayerNeedSynchronization(CPlayer _Player, Boolean _PlayerNeedSynchronization) {
	if (_Player == Null) return;
	declare Boolean RoyalStars_PlayerData_PlayerNeedSynchronization for _Player = C_Default_PlayerNeedSynchronization;
	RoyalStars_PlayerData_PlayerNeedSynchronization = _PlayerNeedSynchronization;
}
Void ResetPlayerNeedSynchronization(CPlayer _Player) {
	SetPlayerNeedSynchronization(_Player, C_Default_PlayerNeedSynchronization);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Check if the player data need to be synchronized on the client (server side only)
Boolean PlayerNeedSynchronization(CPlayer _Player) {
	if (_Player == Null) return False;
	declare Boolean RoyalStars_PlayerData_PlayerNeedSynchronization for _Player = C_Default_PlayerNeedSynchronization;
	return RoyalStars_PlayerData_PlayerNeedSynchronization;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Request the synchronization of the player data on the client (server side only)
Void SynchronizePlayer(CSmPlayer _Player) {
	if (_Player == Null || _Player.Score == Null) return;
	declare Boolean RoyalStars_PlayerData_PlayerNeedSynchronization for _Player = C_Default_PlayerNeedSynchronization;
	declare netwrite Integer Net_RoyalStars_PlayerData_PlayerSynchronizationPoint for _Player.Score = C_Default_PlayerSynchronizationPoint;
	RoyalStars_PlayerData_PlayerNeedSynchronization = False;
	Net_RoyalStars_PlayerData_PlayerSynchronizationPoint += 1;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the player synchronization point (client side only)
Integer GetPlayerSynchronizationPoint(CSmPlayer _Player) {
	if (_Player == Null || _Player.Score == Null) return C_Default_PlayerSynchronizationPoint;
	declare netread Integer Net_RoyalStars_PlayerData_PlayerSynchronizationPoint for _Player.Score = C_Default_PlayerSynchronizationPoint;
	return Net_RoyalStars_PlayerData_PlayerSynchronizationPoint;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Initialize player data (server side only)
Void InitializePlayer(CPlayer _Player) {
	if (_Player == Null) return;
	ResetPlayerSegment(_Player);
	ResetPlayerStars(_Player);
	ResetPlayerNeedSynchronization(_Player);
}
