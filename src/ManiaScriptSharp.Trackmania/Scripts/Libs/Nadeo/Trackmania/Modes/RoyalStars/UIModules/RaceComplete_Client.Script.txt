// Update the race finish display
Void UpdateRaceFinishDisplay(K_State _State, PlayerData::K_RaceFinishEvent _RaceFinishEvent) {
	if (_RaceFinishEvent.EarnedStars.count <= 0) return;

	declare Integer NbStarsTopRow = 0;
	declare Integer NbStarsBottomRow = _RaceFinishEvent.EarnedStars.count;
	if (_RaceFinishEvent.EarnedStars.count > 2) {
		NbStarsTopRow = _RaceFinishEvent.EarnedStars.count / 2;
		NbStarsBottomRow = _RaceFinishEvent.EarnedStars.count - NbStarsTopRow;
	}

	// Update stars layout
	foreach (Index => Control in _State.Controls.Frame_Stars.Controls) {
		declare CMlQuad Quad_Star = Control as CMlQuad;

		AnimMgr.Flush(Quad_Star);
		Quad_Star.Visible = Index < _RaceFinishEvent.EarnedStars.count;
		if (Quad_Star.Visible) {
			declare Integer CurrentRowIndex = Tools::Ternary(Index < NbStarsTopRow, Index, Index - NbStarsTopRow);
			declare Integer NbStarsCurrentRow = Tools::Ternary(Index < NbStarsTopRow, NbStarsTopRow, NbStarsBottomRow);

			Quad_Star.Opacity = 0.;
			Quad_Star.RelativeScale = 1.;
			Quad_Star.RelativeRotation = 0.;
			Quad_Star.RelativePosition_V3.X = {{{Size_Star.X+4}}}*(-0.5*NbStarsCurrentRow + CurrentRowIndex + 0.5);
			Quad_Star.RelativePosition_V3.Y = 0.85*{{{Size_Star.Y}}}*Tools::Ternary(Index < NbStarsTopRow, 1, 0);
			switch (_RaceFinishEvent.EarnedStars[Index]) {
				case PlayerData::C_StarLevel_Gold: Quad_Star.Colorize = {{{Stylesheet::GetColorRGB(Stylesheet::C_Color_MedalGold)}}};
				case PlayerData::C_StarLevel_Silver: Quad_Star.Colorize = {{{Stylesheet::GetColorRGB(Stylesheet::C_Color_MedalSilver)}}};
				case PlayerData::C_StarLevel_Bronze: Quad_Star.Colorize = {{{Stylesheet::GetColorRGB(Stylesheet::C_Color_MedalBronze)}}};
				default: Quad_Star.Colorize = {{{Stylesheet::GetColorRGB(Stylesheet::C_Color_MedalAuthorLight)}}};
			}
		}

	}

	// Update labels
	_State.Controls.Label_RaceTime.SetText(TL::TimeToText(_RaceFinishEvent.Time, True, True));
	switch (_RaceFinishEvent.EarnedStars[0]) {

	}

	// Reset controls for the upcoming animations
	_State.Controls.Frame_Result.RelativePosition_V3 = <0., -15.>;
	_State.Controls.Quad_OverlayBg.Opacity = 0.;
	_State.Controls.Label_RaceTime.Opacity = 0.;
	_State.Controls.Label_RaceInfo.Opacity = 0.;
	_State.Controls.Frame_Overlay.Show();

	// Flush animations
	AnimMgr.Flush(_State.Controls.Frame_Overlay);
	AnimMgr.Flush(_State.Controls.Frame_Result);
	AnimMgr.Flush(_State.Controls.Label_RaceInfo);
	AnimMgr.Flush(_State.Controls.Label_RaceTime);
	AnimMgr.Flush(_State.Controls.Quad_OverlayBg);

			AnimMgr.Add(Quad_Star, "<anim opacity=\"0.\"/>", AnimOutStartTime, C_AnimDuration_StarOut, CAnimManager::EAnimManagerEasing::QuadOut);
		}
	}
}

// Update the visibility of the whole manialink UI
K_State SetVisibility(K_State _State, Boolean _PageIsVisible, Boolean _DisplayModule) {
	declare K_State State = _State;
	State.PageIsVisible = _PageIsVisible;
	State.DisplayModule = _DisplayModule;
	State.IsVisible = State.PageIsVisible && State.DisplayModule;
	State.Controls.Frame_Global.Visible = State.IsVisible;
	return State;
}

// Synchronize player data with the server
// Prevent triggering an old race finish event when initializing the UI
K_State SynchronizePlayer(K_State _State, CSmPlayer _Target) {
	if (_Target == Null) return _State;

	declare K_State State = _State;

	State.SynchronizationPoint = PlayerData::GetPlayerSynchronizationPoint(_Target);
	State.RaceFinishEventUpdate = PlayerData::GetPlayerRaceFinishEventUpdate(_Target);

	return State;
}

// Trigger a race finish event
K_State TriggerRaceFinishEvent(K_State _State, CSmPlayer _Target) {
	if (_Target == Null) return _State;

	declare K_State State = _State;

	State.RaceFinishEventUpdate = PlayerData::GetPlayerRaceFinishEventUpdate(_Target);
	UpdateRaceFinishDisplay(State, PlayerData::GetPlayerRaceFinishEvent(_Target));

	return State;
}

// Update the targeted player
K_State SetTarget(K_State _State, CSmPlayer _Target) {
	declare K_State State = _State;

	if (_Target == Null) {
		State.TargetId = NullId;
	} else {
		State.TargetId = _Target.Id;
		State = SynchronizePlayer(State, _Target);
	}

	return State;
}
