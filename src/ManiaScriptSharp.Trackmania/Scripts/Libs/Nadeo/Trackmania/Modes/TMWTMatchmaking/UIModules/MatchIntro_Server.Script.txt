/**
 * UI module: MatchIntro server side
 */
#Const Version "1.1.0"
#Const ScriptName "Libs/Nadeo/Trackmania/Modes/TMWTMatchmaking/UIModules/MatchIntro_Server.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Includes
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Modes/UIModules_Common.Script.txt" as UIModules_Common
#Include "Libs/Nadeo/Trackmania/Modes/TMWTMatchmaking/UIModules/MatchIntro_Common.Script.txt" as UIModules_MatchIntro_Common

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_DefaultBackgroundUrl ""
#Const C_DefaultLogoUrl ""
#Const C_DefaultDisplayEndTime -1
#Const C_DefaultVisibility False

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the id of the UI module
Text GetId() {
	return UIModules_MatchIntro_Common::C_Id;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the configuration of the UI module
UIModules_Common::K_ModuleConfig GetConfig() {
	return UIModules_MatchIntro_Common::C_UIModuleConfig;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the url of the background displayed in the intro sequence
Void SetBackgroundUrl(Text _BackgroundUrl) {
	declare netwrite Text Net_TMWTMatchmaking_MatchIntro_BackgroundUrl for Teams[0] = C_DefaultBackgroundUrl;
	Net_TMWTMatchmaking_MatchIntro_BackgroundUrl = _BackgroundUrl;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the url of the logo displayed in the intro sequence
Void SetLogoUrl(Text _LogoUrl) {
	declare netwrite Text Net_TMWTMatchmaking_MatchIntro_LogoUrl for Teams[0] = C_DefaultLogoUrl;
	Net_TMWTMatchmaking_MatchIntro_LogoUrl = _LogoUrl;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the time at which the sequence will stop
Void SetDisplayEndTime(Integer _DisplayEndTime) {
	declare netwrite Integer Net_TMWTMatchmaking_MatchIntro_DisplayEndTime for Teams[0] = C_DefaultDisplayEndTime;
	Net_TMWTMatchmaking_MatchIntro_DisplayEndTime = _DisplayEndTime;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Show or hide the UI
 *
 *	@param	_Player										The player to update
 *	@param	_Display									True to show
 *																		False to hide
 */
Void SetVisibility(CPlayer _Player, Boolean _Display) {
	if (_Player == Null) return;
	declare CUIConfig UI <=> UIManager.GetUI(_Player);
	if (UI == Null) return;

	declare netwrite Boolean Net_TMWTMatchmaking_MatchIntro_IsVisible for UI = C_DefaultVisibility;
	Net_TMWTMatchmaking_MatchIntro_IsVisible = _Display;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Initialize a player
 *
 *	@param	_Player										The player to initialize
 */
Void InitializePlayer(CPlayer _Player) {
	if (_Player == Null) return;

	SetVisibility(_Player, C_DefaultVisibility);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Unload the library
Void Unload() {
	SetBackgroundUrl(C_DefaultBackgroundUrl);
	SetLogoUrl(C_DefaultBackgroundUrl);
	SetDisplayEndTime(C_DefaultDisplayEndTime);

	foreach (Player in AllPlayers) {
		InitializePlayer(Player);
	}
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Load the library
Void Load() {
	Unload();
}
