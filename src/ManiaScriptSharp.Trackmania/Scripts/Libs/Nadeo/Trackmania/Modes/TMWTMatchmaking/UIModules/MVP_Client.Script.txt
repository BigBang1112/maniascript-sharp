K_State SetVisibility(K_State _State, Boolean _DisplayModule) {
	declare K_State State = _State;

	State.DisplayModule = _DisplayModule;
	State.IsVisible = (
		State.DisplayModule &&
		State.MVPAccountId != ""
	);

	AnimMgr.Flush(State.Controls.Frame_AnimVisibility);
	if (State.IsVisible) {
		AnimMgr.Add(State.Controls.Frame_AnimVisibility, "<a pos=\"0 0\" hidden=\"0\" />", C_VisibilityAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
	} else {
		AnimMgr.Add(State.Controls.Frame_AnimVisibility, "<a pos=\""^ - C_Size_Background.X - C_Size_Title.X ^" 0\" hidden=\"1\" />", C_VisibilityAnimDuration, CAnimManager::EAnimManagerEasing::QuadOut);
	}
	return State;
}

K_State SetMVP(K_State _State, Integer _MVPUpdate, Text _MVPAccountId, Int2 _MVPScore) {
	declare K_State State = _State;

	State.MVPUpdate = _MVPUpdate;
	State.MVPAccountId = _MVPAccountId;
	State.MVPScore = _MVPScore;

	// Try to find the user to display the name of the bots correctly
	declare CUser MVPUser;
	for (Score in Scores) {
		if (Score.User != Null && Score.User.WebServicesUserId == State.MVPAccountId) {
			MVPUser <=> Score.User;
			break;
		}
	}
	if (MVPUser == Null) {
		{{{PlayerName::P}}}SetAccountId(This, State.Controls.PlayerName_MVP, State.MVPAccountId, True);
	} else {
		{{{PlayerName::P}}}SetUser(This, State.Controls.PlayerName_MVP, MVPUser, True);
	}

	if (State.MVPScore.X >= 0) {
		State.Controls.Label_Points.Value = ""^State.MVPScore.X;
	} else {
		State.Controls.Label_Points.Value = "0";
	}
	if (State.MVPScore.Y >= 0) {
		State.Controls.Label_Time.Value = TL::TimeToText(State.MVPScore.Y, True, True);
	} else {
		State.Controls.Label_Time.Value = "-";
	}
	MLHelpers::FitLabelValue(State.Controls.Label_Points, C_TextSize_Score);
	MLHelpers::FitLabelValue(State.Controls.Label_Time, C_TextSize_Score);

	return SetVisibility(State, _State.DisplayModule);
}
