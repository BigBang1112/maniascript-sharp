K_State SetVisibility(K_State _State, Boolean _DisplayModule) {
	declare K_State State = _State;

	State.DisplayModule = _DisplayModule;

	State.Controls.Frame_Global.Visible = State.DisplayModule;

	return State;
}

Void UpdateClansColors(K_State _State) {
	foreach (Clan => ControlsClan in _State.Controls.Clans) {
		foreach (Player in ControlsClan.Players) {
			if (Clan == 2) {
				Player.Quad_PlayerBg.Colorize = TMWTStylesheet::GetVec3Color(TMWTStylesheet::C_Class_Gradient_RedTeam, TMWTStylesheet::C_Theme_TMGL);
			} else {
				Player.Quad_PlayerBg.Colorize = TMWTStylesheet::GetVec3Color(TMWTStylesheet::C_Class_Gradient_BlueTeam, TMWTStylesheet::C_Theme_TMGL);
			}
		}
	}
}

Void UpdateTeamsInfo(K_State _State) {
	foreach (Clan => ControlsClan in _State.Controls.Clans) {
		ControlsClan.Quad_ClanLogo.ImageUrl = C_DefaultClanLogoList.get(Clan, C_DefaultClanLogoList[1]);
		if (Teams.existskey(Clan - 1)) {
			ControlsClan.Label_ClanName.Value = Teams[Clan - 1].Name;
			MLHelpers::FitLabelValue(ControlsClan.Label_ClanName, C_TextSize_LabelClanName);
		} else {
			ControlsClan.Label_ClanName.Value = "";
		}
	}
}

K_State SetBackgroundUrl(K_State _State, Text _BackgroundUrl) {
	declare K_State State = _State;

	State.BackgroundUrl = _BackgroundUrl;
	State.Controls.Quad_Background.ImageUrl = State.BackgroundUrl;
	if (State.Controls.Quad_Background.ImageUrl == "") {
		State.Controls.Quad_Background.Opacity = 0.7;
	} else {
		State.Controls.Quad_Background.Opacity = 1.;
	}

	return State;
}

K_State SetLogoUrl(K_State _State, Text _LogoUrl) {
	declare K_State State = _State;

	State.LogoUrl = _LogoUrl;
	State.Controls.Quad_Logo.ImageUrl = State.LogoUrl;
	State.Controls.Quad_Logo.Visible = (State.Controls.Quad_Logo.ImageUrl != "");

	return State;
}

Void UpdateLabelMatchInfo(K_State _State) {
	declare Text NewValue = C_Text_WaitingForPlayers;

	if (_State.DisplayEndTime > 0) {
		if (_State.DisplayEndTime > GameTime) {
			NewValue = TL::Compose(C_Text_Countdown, ""^(((_State.DisplayEndTime - GameTime) / 1000) + 1));
		} else {
			NewValue = C_Text_MatchStarting;
		}
	}

	if (_State.Controls.Label_MatchInfo.Value != NewValue) {
		_State.Controls.Label_MatchInfo.Value = NewValue;
		MLHelpers::FitLabelValue(_State.Controls.Label_MatchInfo, C_TextSize_LabelMatchInfo);
	}
}

K_State SetDisplayEndTime(K_State _State, Integer _DisplayEndTime) {
	declare K_State State = _State;

	State.DisplayEndTime = _DisplayEndTime;
	UpdateLabelMatchInfo(State);

	return State;
}

Void UpdatePlayers(K_State _State, CSmPlayer[][Integer] _ClansPlayers) {
	foreach (Clan => ControlsClan in _State.Controls.Clans) {
		foreach (Key => ControlsPlayer in ControlsClan.Players) {
			if (_ClansPlayers[Clan].existskey(Key)) {
				declare CSmPlayer Player <=> _ClansPlayers[Clan][Key];

					} else {
						ControlsPlayer.Quad_PlayerFlag.ImageUrl = Player.User.CountryFlagUrl;
					}
					{{{PlayerName::P}}}SetUser(This, ControlsPlayer.PlayerName_Main, Player.User, True);
					ControlsPlayer.Quad_PlayerFlag.Visible = (ControlsPlayer.Quad_PlayerFlag.ImageUrl != "");
					ControlsPlayer.PlayerName_Main.Visible = True;
				} else {
					ControlsPlayer.Quad_PlayerFlag.Visible = False;
					ControlsPlayer.PlayerName_Main.Visible = False;
				}
				ControlsPlayer.Frame_Player.Visible = True;
			} else {
				ControlsPlayer.Frame_Player.Visible = False;
			}
		}
	}
