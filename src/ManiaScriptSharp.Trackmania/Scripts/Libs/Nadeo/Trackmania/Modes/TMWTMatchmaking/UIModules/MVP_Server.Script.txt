// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants

#Const C_DefaultVisibility False

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the id of the UI module
Text GetId() {
	return UIModules_MVP_Common::C_Id;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Get the configuration of the UI module
UIModules_Common::K_ModuleConfig GetConfig() {
	return UIModules_MVP_Common::C_UIModuleConfig;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Set the match MVP
Void SetMVP(CSmScore _Score) {
	declare netwrite Text Net_TMWTMatchmaking_MVP_AccountId for Teams[0] = "";
	declare netwrite Int2 Net_TMWTMatchmaking_MVP_Score for Teams[0] = <-1, -1>;
	declare netwrite Integer Net_TMWTMatchmaking_MVP_Update for Teams[0] = 0;
	if (_Score != Null && _Score.User != Null) {
		Net_TMWTMatchmaking_MVP_AccountId = _Score.User.WebServicesUserId;
		if (_Score.BestRaceTimes.count > 0) {
			Net_TMWTMatchmaking_MVP_Score = <Scores::GetPlayerMatchPoints(_Score), _Score.BestRaceTimes[-1]>;
		} else {
			Net_TMWTMatchmaking_MVP_Score = <Scores::GetPlayerMatchPoints(_Score), -1>;
		}
	} else {
		Net_TMWTMatchmaking_MVP_AccountId = "";
		Net_TMWTMatchmaking_MVP_Score = <-1, -1>;
	}
	Net_TMWTMatchmaking_MVP_Update = Now;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Show or hide the UI
 *
 *	@param	_Player										The player to update
 *	@param	_Display									True to show
 *																		False to hide
 */
Void SetVisibility(CPlayer _Player, Boolean _Display) {
	if (_Player == Null) return;
	declare CUIConfig UI <=> UIManager.GetUI(_Player);
	if (UI == Null) return;

	declare netwrite Boolean Net_TMWTMatchmaking_MVP_IsVisible for UI = C_DefaultVisibility;
	Net_TMWTMatchmaking_MVP_IsVisible = _Display;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Initialize a player
 *
 *	@param	_Player										The player to initialize
 */
Void InitializePlayer(CPlayer _Player) {
	if (_Player == Null) return;

	SetVisibility(_Player, C_DefaultVisibility);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Unload the library
Void Unload() {
	foreach (Player in AllPlayers) {
		InitializePlayer(Player);
	}
	SetMVP(Null);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Load the library
Void Load() {
	Unload();
}
