// Create a K_TeamMember structure from a score
LibStructuresMatchmaking_K_TeamMember New_TeamMember(CSmScore _Score) {
	if (_Score == Null || _Score.User == Null) return C_TeamMember_Null;

	declare Integer BestTime = -1;
	if (_Score.BestRaceTimes.count > 0) BestTime = _Score.BestRaceTimes[_Score.BestRaceTimes.count - 1];

	return LibStructuresMatchmaking_K_TeamMember {
		AccountId = _Score.User.WebServicesUserId,
		Name = _Score.User.Name,
		ClubTag = _Score.User.ClubTag,
		FlagUrl = _Score.User.CountryFlagUrl,
		Score = _Score.Points,
		BestTime = BestTime,
		IsFakeUser = _Score.User.IsFakeUser
	};
}

/// Get the active ranked mode
Text GetActiveRankedMode() {
	switch (Config::Get().RankedMode) {
		case C_Mode_Teams3v3: return C_Mode_Teams3v3;
		case C_Mode_Teams2v2: return C_Mode_Teams2v2;
	}
	return C_Mode_Teams2v2;
}
