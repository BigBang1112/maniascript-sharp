/**
 * Component: JoinLinkAPI, API for join links
 */
#Const Version		"1.0.0"
#Const ScriptName "Libs/Nadeo/Trackmania/API/JoinLinkAPI.Script.txt"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Libraries
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Include "Libs/Nadeo/CMGame/Utils/Config.Script.txt" as Config
#Include "Libs/Nadeo/CMGame/Utils/Http.Script.txt" as Http

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Constants
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Const C_API_JoinLink "/api/joinlink"
#Const C_Headers [
	"Accept" => "application/json",
	"Content-Type" => "application/json"
]

#Const C_Route_GetPlayerJoinLinks "/me"

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Structures
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
#Struct K_PlayerJoinLink {
	Text Name;
	Text JoinLink;
	Boolean IsSpectator;
	Boolean IsCompleted;
	Text MediaUrl;
}
#Struct K_HttpPlayerJoinLink {
	Text name;
	Text joinLink;
	Boolean isSpectator;
	Boolean isCompleted;
	Text mediaUrl;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// MARK: Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Start a request to get the player joinlinks
Http::K_Request GetPlayerJoinLinks() {
	return Http::CreateGet(Config::Get().APIMeetUrl^C_API_JoinLink^C_Route_GetPlayerJoinLinks, C_Headers);
}

/// Parse the response to the GetPlayerJoinLinks request
K_PlayerJoinLink[] GetResponseFromGetPlayerJoinLinks(Http::K_Request _Request) {
	declare K_HttpPlayerJoinLink[] HttpResponse;
	HttpResponse.fromjson(Http::GetResult(_Request));

	declare K_PlayerJoinLink[] PlayerJoinLinks;
	foreach (HttpPlayerJoinLink in HttpResponse) {
		PlayerJoinLinks.add(K_PlayerJoinLink {
			Name = HttpPlayerJoinLink.name,
			JoinLink = HttpPlayerJoinLink.joinLink,
			IsSpectator = HttpPlayerJoinLink.isSpectator,
			IsCompleted = HttpPlayerJoinLink.isCompleted,
			MediaUrl = HttpPlayerJoinLink.mediaUrl
		});
	}

	return PlayerJoinLinks;
}
